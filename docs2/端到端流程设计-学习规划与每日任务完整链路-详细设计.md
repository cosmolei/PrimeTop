# 端到端流程设计 — 学习规划与每日任务完整链路

## 1. 文档概述

### 1.1 目的

本文档描述 PrimeTop **学习规划与每日任务**功能的端到端完整链路，覆盖从学生设定学习目标到最终学习数据沉淀与计划自动调整的全过程，包括客户端交互、服务端处理、跨模块协作、异常降级等环节，供开发人员直接编码实现。

### 1.2 范围

覆盖以下学习规划场景：
- 目标设定与计划自动生成（学期/考试冲刺/自定义）
- 每日任务智能生成（凌晨定时 + 实时调整）
- 任务执行与状态追踪（开始/暂停/完成/跳过）
- 打卡与学习习惯养成
- 计划复盘与动态调整
- 提醒通知与家长同步

### 3.2 目标创建请求定义

```typescript
// 客户端 → 服务端：创建学习目标
interface CreateGoalRequest {
  /** 目标标题 */
  title: string;                           // 如 "期末考试数学90分以上"

  /** 目标类型 */
  goalType: 'daily'                        // 每日小目标
           | 'weekly'                      // 周目标
           | 'monthly'                     // 月目标
           | 'exam_preparation'            // 考试备考
           | 'semester';                   // 学期目标

  /** 关联学科ID (null=全科) */
  subjectId?: string;

  /** 目标考试 (exam_preparation必填) */
  targetExam?: string;                     // 如 "期中考试" / "中考" / "高考"

  /** 截止日期 */
  targetDate: string;                      // ISO 8601 date

  /** 当前水平（自评 or 系统填充） */
  currentLevel?: string;                   // 如 "中等" / "80分左右"

  /** 期望水平 */
  targetLevel?: string;                    // 如 "优秀" / "90分以上"

  /** 每日可用学习时长（分钟） */
  dailyAvailableMinutes?: number;          // 默认 30, 范围 10-180

  /** 学习偏好 */
  preference?: {
    /** 偏重方向 */
    focusAreas?: ('weakness_recovery'      // 薄弱点补救
                  | 'new_knowledge'        // 新知识学习
                  | 'exam_drill'           // 考试刷题
                  | 'review')[];           // 复习巩固
    /** 难度偏好 */
    difficulty?: 'easy' | 'moderate' | 'intensive';
  };

  /** 来源标记 */
  source?: 'manual'                        // 手动创建
          | 'report_suggestion'            // 学情报告建议
          | 'exam_result'                  // 考试结果触发
          | 'onboarding';                  // 新手引导
}
```

### 3.3 服务端目标创建流程

```python
# service/plan/goal_service.py

class GoalService:
    async def create_goal(
        self,
        user_id: str,
        req: CreateGoalRequest,
        ctx: RequestContext,
    ) -> GoalVO:
        """
        创建学习目标主流程
        """
        # 1. 参数校验
        self._validate_goal(req)

        # 2. 获取用户画像上下文（用于系统填充当前水平）
        profile = await self.profile_repo.get_learning_profile(user_id)

        # 3. 自动填充 currentLevel（若用户未填）
        if not req.current_level:
            req.current_level = self._infer_current_level(profile, req.subject_id)

        # 4. 校验活跃目标数量上限（同时最多 3 个活跃目标）
        active_count = await self.goal_repo.count_active_goals(user_id)
        if active_count >= 3:
            raise BizError(
                ErrorCode.GOAL_LIMIT_EXCEEDED,
                "当前已有3个进行中的学习目标，请先完成或放弃后再创建新目标"
            )

        # 5. 校验截止日期合理性
        min_days = self._get_min_days(req.goal_type)
        days_until = (parse_date(req.target_date) - date.today()).days
        if days_until < min_days:
            raise BizError(
                ErrorCode.GOAL_DATE_TOO_SHORT,
                f"{req.goal_type}类型目标至少需要{min_days}天"
            )

        # 6. 持久化目标
        goal = Goal(
            goal_id=uuid4(),
            user_id=user_id,
            title=req.title,
            goal_type=req.goal_type,
            subject_id=req.subject_id,
            target_exam=req.target_exam,
            target_date=req.target_date,
            current_level=req.current_level,
            target_level=req.target_level,
            status=GoalStatus.ACTIVE,
            metadata={
                'daily_available_minutes': req.daily_available_minutes or 30,
                'preference': req.preference,
                'source': req.source or 'manual',
            },
        )
        await self.goal_repo.save(goal)

        # 7. 发布目标创建事件（触发计划生成）
        await self.event_bus.publish(GoalCreatedEvent(
            goal_id=goal.goal_id,
            user_id=user_id,
            goal_type=req.goal_type,
        ))

        return GoalVO.from_entity(goal)

    def _get_min_days(self, goal_type: str) -> int:
        return {
            'daily': 1,
            'weekly': 7,
            'monthly': 14,
            'exam_preparation': 7,
            'semester': 30,
        }.get(goal_type, 7)
```

### 4.2 计划生成编排流程

```python
# service/plan/plan_generation_orchestrator.py

class PlanGenerationOrchestrator:
    """
    计划生成编排器 — 协调多服务完成计划生成
    """

    async def generate_plan(
        self,
        goal_id: str,
        user_id: str,
        trigger: str = 'goal_created',
    ) -> StudyPlan:
        """
        计划生成主流程
        """
        # 1. 加载目标
        goal = await self.goal_repo.get_by_id(goal_id)
        if not goal or goal.user_id != user_id:
            raise BizError(ErrorCode.GOAL_NOT_FOUND)

        # 2. 并行获取上下文数据
        profile, mastery_map, mistake_summary, chapters = await asyncio.gather(
            self.profile_repo.get_learning_profile(user_id),
            self.mastery_repo.get_subject_mastery(user_id, goal.subject_id),
            self.mistake_repo.get_weak_point_summary(
                user_id,
                subject_id=goal.subject_id,
                limit=10,
            ),
            self.chapter_repo.get_chapters_by_subject(
                goal.subject_id,
                user_id=user_id,
            ),
        )

        # 3. 构建计划生成 Prompt
        prompt = self.prompt_builder.build_plan_prompt(
            goal=goal,
            profile=profile,
            mastery_map=mastery_map,
            weak_points=mistake_summary.weak_points,
            chapters=chapters,
            daily_minutes=goal.metadata.get('daily_available_minutes', 30),
        )

        # 4. 调用 AI 生成计划
        ai_result = await self.ai_service.generate_structured(
            prompt=prompt,
            response_schema=PlanSchema,
            model_hint='reasoning',      # 使用推理增强模型
            temperature=0.3,             # 低温度，确保结构化输出稳定
            max_tokens=4096,
        )

        # 5. 解析与校验 AI 输出
        plan_data = PlanSchema.parse(ai_result.content)
        self._validate_plan_data(plan_data, goal)

        # 6. 废弃旧计划（如有）
        if trigger != 'goal_created':
            await self.plan_repo.deactivate_user_plans(
                user_id,
                goal_id=goal_id,
                reason=f'plan_regenerate:{trigger}',
            )

        # 7. 持久化计划与里程碑
        plan = await self._save_plan_with_milestones(
            user_id=user_id,
            goal_id=goal_id,
            plan_data=plan_data,
        )

        # 8. 立即生成今日任务（如果当天还没有）
        today_batch = await self.task_repo.get_today_batch(user_id)
        if not today_batch:
            await self.task_generator.generate_daily_tasks(user_id, plan.plan_id)

        # 9. 发布计划生成事件
        await self.event_bus.publish(PlanCreatedEvent(
            plan_id=plan.plan_id,
            user_id=user_id,
            goal_id=goal_id,
            milestone_count=len(plan_data.milestones),
        ))

        return plan
```

## 5. P3 每日任务生成

### 5.1 触发方式

| 触发源 | 触发时机 | 说明 |
|--------|---------|------|
| 定时任务 | 每日 05:00 CST | 为活跃用户批量生成 |
| 用户首次打开APP | 检测今日无任务时实时生成 | 补偿定时任务遗漏 |
| 计划调整后 | 计划变更事件触发 | 重新生成当日剩余任务 |
| 任务完成率 < 30% | 中午 12:00 检测 | 简化下午任务 |

### 5.2 任务类型体系

```typescript
/** 任务类型枚举 */
type TaskType =
  | 'chapter_study'          // 章节学习（同步课堂）
  | 'knowledge_explain'      // AI知识点讲解
  | 'practice_chapter'       // 章节练习
  | 'practice_topic'         // 知识点专项练习
  | 'practice_adaptive'      // 自适应练习
  | 'mistake_retry'          // 错题重练
  | 'mistake_review'         // 错题复习（遗忘曲线调度）
  | '背诵检测'               // 文科背诵检测
  | 'essay_writing'          // 作文练习
  | 'exam_simulation'        // 模拟测试
```

### 5.3 任务生成核心算法

```python
# service/task/daily_task_generator.py

class DailyTaskGenerator:
    """
    每日任务生成器 — 综合多个维度智能组装每日任务清单
    """

    # 默认任务时间分配（分钟）
    DEFAULT_TIME_MAP = {
        'chapter_study': 15,
        'practice_chapter': 10,
        'practice_topic': 10,
        'practice_adaptive': 12,
        'mistake_retry': 8,
        'mistake_review': 8,
        'knowledge_explain': 10,
        'exam_simulation': 30,
        '背诵检测': 8,
        'essay_writing': 20,
    }

    async def generate_daily_tasks(
        self,
        user_id: str,
        plan_id: str,
        target_date: date = None,
    ) -> DailyTaskBatch:
        target_date = target_date or date.today()

        # 1. 幂等校验：同用户同日期同计划不重复生成
        existing = await self.task_repo.get_batch(user_id, plan_id, target_date)
        if existing and existing.status != BatchStatus.CANCELLED:
            return existing

        # 2. 并行获取上下文
        plan, profile, active_milestone, mastery_map, due_reviews, weak_points = (
            await asyncio.gather(
                self.plan_repo.get_by_id(plan_id),
                self.profile_repo.get_learning_profile(user_id),
                self.milestone_repo.get_active_milestone(user_id, plan_id),
                self.mastery_repo.get_subject_mastery(user_id, plan.subject_id),
                self.review_scheduler.get_due_reviews(
                    user_id,
                    target_date=target_date,
                    limit=10,
                ),
                self.weakness_repo.get_top_weak_points(
                    user_id,
                    subject_id=plan.subject_id,
                    limit=5,
                ),
            )
        )

        # 3. 计算可用总时长
        daily_minutes = plan.metadata.get('daily_available_minutes', 30)

        # 4. 获取当前里程碑的任务模板比例
        template = active_milestone.daily_task_template if active_milestone else {
            'new_knowledge_ratio': 0.3,
            'weakness_review_ratio': 0.3,
            'mistake_retry_ratio': 0.2,
            'comprehensive_ratio': 0.2,
        }

        # 5. 计算各分类可用时间
        time_budget = {
            'new_knowledge': int(daily_minutes * template['new_knowledge_ratio']),
            'weakness_review': int(daily_minutes * template['weakness_review_ratio']),
            'mistake_retry': int(daily_minutes * template['mistake_retry_ratio']),
            'comprehensive': int(daily_minutes * template['comprehensive_ratio']),
        }

        # 6. 按分类生成任务
        tasks: list[DailyTaskItem] = []

        # 6a. 新知识学习任务（来自当前里程碑目标章节）
        tasks.extend(await self._generate_new_knowledge_tasks(
            user_id, active_milestone, time_budget['new_knowledge'], mastery_map,
        ))

        # 6b. 薄弱点复习任务
        tasks.extend(await self._generate_weakness_tasks(
            user_id, weak_points, time_budget['weakness_review'], plan.subject_id,
        ))

        # 6c. 错题重练任务
        tasks.extend(await self._generate_mistake_tasks(
            user_id, due_reviews, time_budget['mistake_retry'],
        ))

        # 6d. 综合练习任务
        tasks.extend(await self._generate_comprehensive_tasks(
            user_id, plan.subject_id, mastery_map, time_budget['comprehensive'],
        ))

        # 7. 排序与时间微调
        tasks = self._optimize_ordering(tasks)
        total_estimated = sum(t.estimated_minutes for t in tasks)

        # 超时裁剪：去掉优先级最低的任务直到总时长 ≤ daily_minutes
        if total_estimated > daily_minutes:
            tasks = self._trim_tasks(tasks, daily_minutes)

        # 8. 持久化任务批次
        batch = DailyTaskBatch(
            batch_id=uuid4(),
            user_id=user_id,
            plan_id=plan_id,
            batch_date=target_date,
            status=BatchStatus.PENDING,
            total_count=len(tasks),
            estimated_min=sum(t.estimated_minutes for t in tasks),
            items=tasks,
        )
        await self.task_repo.save_batch(batch)

        return batch

    async def _generate_new_knowledge_tasks(
        self,
        user_id: str,
        milestone: Milestone,
        budget_minutes: int,
        mastery_map: MasteryMap,
    ) -> list[DailyTaskItem]:
        """
        基于当前里程碑目标章节生成学习任务
        """
        tasks = []
        remaining = budget_minutes

        for chapter_id in (milestone.target_chapters or []):
            if remaining <= 0:
                break

            mastery = mastery_map.get_chapter_mastery(chapter_id)
            if mastery and mastery.overall > 0.85:
                # 已掌握的章节跳过
                continue

            # 章节学习任务
            tasks.append(DailyTaskItem(
                task_id=uuid4(),
                task_type='chapter_study',
                title=f'学习章节：{chapter_id}',
                source='plan_milestone',
                priority=Priority.HIGH,
                estimated_minutes=min(15, remaining),
                ref_id=chapter_id,
                ref_type='chapter',
                params={'chapter_id': chapter_id},
            ))
            remaining -= 15

            # 配套练习（如果时间允许）
            if remaining >= 10:
                tasks.append(DailyTaskItem(
                    task_id=uuid4(),
                    task_type='practice_chapter',
                    title=f'章节练习：{chapter_id}',
                    source='plan_milestone',
                    priority=Priority.MEDIUM,
                    estimated_minutes=10,
                    ref_id=chapter_id,
                    ref_type='chapter',
                    params={
                        'chapter_id': chapter_id,
                        'difficulty': 'adaptive',
                        'count': 5,
                    },
                ))
                remaining -= 10

        return tasks

    async def _generate_mistake_tasks(
        self,
        user_id: str,
        due_reviews: list[DueReview],
        budget_minutes: int,
    ) -> list[DailyTaskItem]:
        """
        基于遗忘曲线到期复习的错题生成重练任务
        """
        if not due_reviews:
            return []

        # 按紧急度排序（超期天数越大越紧急）
        due_reviews.sort(key=lambda r: r.overdue_days, reverse=True)

        tasks = []
        remaining = budget_minutes

        # 取最紧急的错题组
        review_batch = due_reviews[:min(5, len(due_reviews))]
        task_minutes = min(8, remaining)

        if task_minutes > 0:
            tasks.append(DailyTaskItem(
                task_id=uuid4(),
                task_type='mistake_review',
                title=f'错题复习（{len(review_batch)}道到期）',
                source='mistake_review',
                priority=Priority.HIGH,
                estimated_minutes=task_minutes,
                params={
                    'review_ids': [r.review_id for r in review_batch],
                    'knowledge_point_ids': list(set(
                        r.knowledge_point_id for r in review_batch
                    )),
                },
            ))
            remaining -= task_minutes

        return tasks

    def _optimize_ordering(self, tasks: list[DailyTaskItem]) -> list[DailyTaskItem]:
        """
        任务排序策略：
        1. 高优先级优先
        2. 同优先级内，错题复习 > 薄弱点 > 新知识 > 综合练习
        3. 交替不同类型任务，避免连续同类型疲劳
        """
        type_order = {
            'mistake_review': 0, 'mistake_retry': 1,
            'practice_topic': 2,
            'chapter_study': 3, 'knowledge_explain': 4,
            'practice_chapter': 5, 'practice_adaptive': 6,
            'comprehensive': 7,
        }
        # 按优先级降序 + 类型序排列
        tasks.sort(key=lambda t: (-t.priority.value, type_order.get(t.task_type, 99)))

        # 交错排列：避免连续两个同类型
        result = []
        while tasks:
            # 找第一个与末尾不同类型的
            for i, t in enumerate(tasks):
                if not result or t.task_type != result[-1].task_type:
                    result.append(tasks.pop(i))
                    break
            else:
                result.append(tasks.pop(0))
        return result

    def _trim_tasks(self, tasks: list[DailyTaskItem], budget: int) -> list[DailyTaskItem]:
        """裁剪超时任务：从最低优先级开始移除"""
        total = sum(t.estimated_minutes for t in tasks)
        # 按优先级升序排列副本（低优先级在前）
        sorted_by_priority = sorted(enumerate(tasks), key=lambda x: x[1].priority.value)
        remove_indices = set()
        for idx, task in sorted_by_priority:
            if total <= budget:
                break
            total -= task.estimated_minutes
            remove_indices.add(idx)
        return [t for i, t in enumerate(tasks) if i not in remove_indices]
```

```python
# service/task/task_completion_handler.py

class TaskCompletionHandler:

    async def handle_completion(
        self,
        user_id: str,
        task_id: str,
        req: TaskCompleteRequest,
    ) -> TaskCompletionResult:
        """
        任务完成后的一揽子处理
        """
        # 1. 加载任务并校验状态
        task = await self.task_repo.get_by_id(task_id)
        if task.user_id != user_id:
            raise BizError(ErrorCode.FORBIDDEN)
        if task.status != TaskStatus.IN_PROGRESS:
            raise BizError(ErrorCode.TASK_NOT_IN_PROGRESS)

        # 2. 更新任务状态
        task.status = TaskStatus.COMPLETED
        task.actual_minutes = req.actual_minutes
        task.result = req.result
        task.feedback = req.feedback
        task.completed_at = datetime.utcnow()
        await self.task_repo.update(task)

        # 3. 更新批次统计
        batch = await self.task_repo.get_batch_by_id(task.batch_id)
        batch.completed_count += 1
        batch.actual_minutes = (batch.actual_minutes or 0) + req.actual_minutes

        # 4. 判断批次是否全部完成
        all_done = batch.completed_count + batch.skipped_count >= batch.total_count
        if all_done:
            batch.status = BatchStatus.COMPLETED
        else:
            batch.status = BatchStatus.IN_PROGRESS
        await self.task_repo.update_batch(batch)

        # 5. 发布任务完成事件
        await self.event_bus.publish(TaskCompletedEvent(
            task_id=task_id,
            user_id=user_id,
            batch_id=task.batch_id,
            plan_id=task.plan_id,
            task_type=task.task_type,
            actual_minutes=req.actual_minutes,
            result=req.result,
            feedback=req.feedback,
        ))

        # 6. 尝试打卡
        checkin_result = await self.checkin_service.try_check_in(user_id)

        # 7. 计算奖励
        rewards = await self._calc_rewards(task, batch, checkin_result)

        return TaskCompletionResult(
            task_id=task_id,
            status=TaskStatus.COMPLETED,
            checkin=checkin_result,
            rewards=rewards,
            batch_progress={
                'completed': batch.completed_count,
                'total': batch.total_count,
                'rate': batch.completed_count / batch.total_count,
            },
        )

    async def _calc_rewards(
        self,
        task: DailyTaskItem,
        batch: DailyTaskBatch,
        checkin: CheckInResult,
    ) -> RewardsVO:
        rewards = RewardsVO(items=[])

        # 基础积分：每个任务 +5 分
        rewards.items.append(RewardItem(
            type='points',
            amount=5,
            reason='task_complete',
        ))

        # 全勤奖励：当日任务全部完成 +20 分
        if batch.completed_count == batch.total_count:
            rewards.items.append(RewardItem(
                type='points',
                amount=20,
                reason='daily_all_complete',
            ))

        # 连续打卡奖励
        if checkin.streak_updated and checkin.current_streak in (3, 7, 14, 30, 60, 100):
            rewards.items.append(RewardItem(
                type='badge',
                badge_id=f'streak_{checkin.current_streak}',
                reason='streak_milestone',
            ))

        return rewards
```

### 8.2 打卡判定逻辑

打卡服务在接收到 `TaskCompletedEvent` 后自动触发判定：

```python
# service/checkin/checkin_service.py

class CheckInService:

    async def try_check_in(self, user_id: str) -> CheckInResult:
        """
        判定今日是否达成打卡条件并执行打卡
        """
        today = date.today()

        # 幂等检查：今日已打过卡
        existing = await self.checkin_repo.get_record(user_id, today)
        if existing:
            return CheckInResult(already_checked_in=True, current_streak=existing.streak)

        # 查询今日任务批次
        batch = await self.task_repo.get_today_batch(user_id)
        if not batch:
            return CheckInResult(checked_in=False, reason='no_tasks')

        # 加载打卡规则
        rules = await self.checkin_repo.get_active_rules()

        # 按优先级匹配规则
        for rule in rules:
            if self._evaluate_rule(rule, batch):
                # 满足条件，执行打卡
                streak = await self._update_streak(user_id, today)
                record = CheckInRecord(
                    record_id=uuid4(),
                    user_id=user_id,
                    checkin_date=today,
                    rule_id=rule.id,
                    matched_condition=rule.condition_expr,
                    streak=streak,
                    tasks_completed=batch.completed_count,
                    study_minutes=batch.actual_minutes or 0,
                )
                await self.checkin_repo.save(record)

                # 发布打卡事件
                await self.event_bus.publish(CheckInEvent(
                    user_id=user_id,
                    streak=streak,
                    tasks_completed=batch.completed_count,
                    study_minutes=batch.actual_minutes or 0,
                ))

                return CheckInResult(
                    checked_in=True,
                    current_streak=streak,
                    streak_updated=True,
                )

        return CheckInResult(checked_in=False, reason='conditions_not_met')

    def _evaluate_rule(self, rule: CheckInRule, batch: DailyTaskBatch) -> bool:
        """
        评估打卡规则条件
        """
        expr = rule.condition_expr
        conditions = expr.get('conditions', [])
        logic = expr.get('type', 'AND')

        results = []
        for cond in conditions:
            metric = cond['metric']
            operator = cond['operator']
            value = cond['value']

            if metric == 'study_duration_sec':
                actual = (batch.actual_minutes or 0) * 60
            elif metric == 'questions_completed':
                actual = batch.completed_count
            elif metric == 'completion_rate':
                actual = batch.completed_count / max(batch.total_count, 1)
            else:
                actual = 0

            if operator == '>=':
                results.append(actual >= value)
            elif operator == '>':
                results.append(actual > value)
            elif operator == '==':
                results.append(actual == value)

        if logic == 'AND':
            return all(results)
        elif logic == 'OR':
            return any(results)
        return False

    async def _update_streak(self, user_id: str, today: date) -> int:
        """
        更新连续打卡天数
        """
        last_record = await self.checkin_repo.get_last_record(user_id)
        if last_record:
            days_diff = (today - last_record.checkin_date).days
            if days_diff == 1:
                # 连续：累加
                return last_record.streak + 1
            elif days_diff == 0:
                # 同一天（不应走到这里，幂等已检查）
                return last_record.streak
            else:
                # 断了：重新从1开始
                return 1
        else:
            return 1
```

```python
        return

            plan_id=event.plan_id,
            progress=progress,
```

```python
# service/plan/plan_review_service.py

class PlanReviewService:

    async def weekly_review(self, user_id: str, plan_id: str) -> WeeklyReviewVO:
        """
        每周复盘：分析执行情况，生成调整建议
        """
        # 1. 统计本周任务数据
        week_start = get_week_start()    # 本周一
        week_end = get_week_end()        # 本周日

        batches = await self.task_repo.get_batches_in_range(
            user_id, plan_id, week_start, week_end,
        )

        stats = self._calc_weekly_stats(batches)

        # 2. 对比计划预期
        plan = await self.plan_repo.get_by_id(plan_id)
        milestone = await self.milestone_repo.get_active_milestone(user_id, plan_id)

        progress_gap = stats.completion_rate - (milestone.expected_daily_rate * 7)

        # 3. AI 生成复盘报告
        review_prompt = self.prompt_builder.build_review_prompt(
            weekly_stats=stats,
            plan=plan,
            milestone=milestone,
            progress_gap=progress_gap,
        )

        review_content = await self.ai_service.generate(
            prompt=review_prompt,
            model_hint='default',
            temperature=0.5,
        )

        # 4. 生成调整建议
        adjustments = self._generate_adjustments(stats, milestone, progress_gap)

        # 5. 持久化复盘记录
        review = PlanReview(
            review_id=uuid4(),
            user_id=user_id,
            plan_id=plan_id,
            review_type='weekly',
            period_start=week_start,
            period_end=week_end,
            stats=stats,
            ai_summary=review_content,
            adjustments=adjustments,
        )
        await self.review_repo.save(review)

        # 6. 如果需要调整，触发下周任务模板更新
        if adjustments.get('template_update'):
            await self._apply_template_adjustments(milestone, adjustments['template_update'])

        return WeeklyReviewVO.from_entity(review)

    def _generate_adjustments(
        self,
        stats: WeeklyStats,
        milestone: Milestone,
        progress_gap: float,
    ) -> dict:
        """
        基于执行情况生成调整策略
        """
        adjustments = {}

        # 完成率过低：减少任务量，增加复习比例
        if stats.completion_rate < 0.5:
            adjustments['template_update'] = {
                'daily_task_count': max(milestone.daily_task_template.get('daily_task_count', 5) - 2, 3),
                'weakness_review_ratio': 0.4,     # 增加薄弱点复习
                'new_knowledge_ratio': 0.2,       # 减少新知识
            }
            adjustments['daily_minutes'] = max(
                milestone.daily_task_template.get('daily_minutes', 30) - 10, 15
            )

        # 完成率很高：适当增加挑战
        elif stats.completion_rate > 0.9 and stats.avg_accuracy > 0.85:
            adjustments['template_update'] = {
                'new_knowledge_ratio': 0.4,       # 加快进度
                'weakness_review_ratio': 0.2,
                'comprehensive_ratio': 0.3,       # 增加综合练习
            }

        # 错题重练正确率低：增加错题复习
        if stats.mistake_retry_accuracy < 0.6:
            adjustments.setdefault('template_update', {})
            adjustments['template_update']['mistake_retry_ratio'] = 0.3

        return adjustments
```

## 14. 数据一致性保障

### 14.1 幂等设计

| 操作 | 幂等键 | 说明 |
|------|--------|------|
| 目标创建 | `user_id + title + goal_type` | 防止重复提交 |
| 每日任务生成 | `user_id + plan_id + batch_date` | 同一天不重复生成 |
| 任务完成 | `task_id` | 同一任务只能完成一次 |
| 打卡 | `user_id + checkin_date` | 同一天只打一次卡 |
| 复盘生成 | `user_id + plan_id + review_type + period_start` | 同一周期不重复生成 |

### 14.2 并发控制

- **任务完成**：乐观锁（`version` 字段），冲突时重试
- **打卡判定**：分布式锁（`checkin:{user_id}:{date}`），防止并发重复打卡
- **计划状态变更**：CAS 操作，状态机前置校验
- **每日任务生成**：幂等键 + 唯一索引，防止定时任务重复触发
