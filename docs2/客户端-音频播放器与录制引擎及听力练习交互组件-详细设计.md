# 客户端-音频播放器与录制引擎及听力练习交互组件-详细设计

## 1. 概述

### 1.1 文档目的

本文档详细设计 PrimeTop 客户端统一的音频播放与录制引擎，以及在此之上构建的学科听力练习交互组件。该模块是多个核心业务场景的底层能力：

- **英语听力练习**：考试听力模拟、课文跟读、口语发音评估
- **语文背诵检测**：课文朗读、古诗文背诵语音检测
- **AI 语音对话**：TTS 流式播放、ASR 语音输入
- **发音评测**：单词/句子发音打分、音素级纠正建议
- **音频类学习资源**：教育内容音频版、微课旁白播放

### 1.2 设计目标

| 目标 | 指标 |
|------|------|
| 播放器首帧延迟 | ≤ 500ms（本地文件）/ ≤ 800ms（流式） |
| 录制音质 | 16kHz / 16-bit PCM，满足 ASR 要求 |
| 内存占用（音频缓冲） | ≤ 15MB |
| 并发播放轨道 | 支持 1 路主播放 + 1 路背景音 |
| 弱网容错 | 支持断点续播、自动降级码率 |
| 电量优化 | 后台暂停录音、屏幕熄屏策略 |

### 1.3 技术选型

| 层次 | 技术方案 | 说明 |
|------|----------|------|
| 跨平台框架 | Flutter | 项目客户端主框架 |
| 音频播放 | `just_audio` + `audio_session` | 支持本地/网络/流式播放 |
| 音频录制 | `record` / 平台原生 Channel | 支持高质量 PCM/AAC 录制 |
| 波形可视化 | CustomPainter + 实时 PCM 采样 | 轻量级波形渲染 |
| 文本同步高亮 | LRC 解析器 + RichText 控制器 | 歌词式同步高亮 |
| 原生能力桥接 | Flutter Platform Channel | iOS AVAudioEngine / Android AudioRecord |
| 音频编解码 | 平台原生 AAC/Opus | 传输压缩，本地 PCM 处理 |

#### 4.1.3 流式 TTS 播放策略

AI 语音对话场景下，TTS 音频通过 SSE/WebSocket 分片到达，需要边收边播：

```dart
/// 流式 TTS 播放器
class StreamTtsPlayer implements PlayerEngine {
  final Queue<List<int>> _chunkQueue = Queue();  // PCM 分片队列
  final StreamController<List<int>> _audioController = StreamController.broadcast();
  bool _isFlushing = false;
  
  /// 接收 TTS 音频分片
  void feedChunk(Uint8List pcmChunk) {
    _chunkQueue.addLast(pcmChunk);
    
    // 首片到达立即开始播放（降低首帧延迟）
    if (_currentState == PlaybackState.loading && _chunkQueue.length >= 2) {
      _startPlayback();
    }
  }
  
  /// 标记 TTS 流结束
  void feedEnd() {
    _isFlushing = true;
    _checkComplete();
  }
  
  void _startPlayback() {
    _currentState = PlaybackState.playing;
    // 将分片队列连接到音频输出
    _pumpChunks();
  }
  
  void _pumpChunks() {
    while (_chunkQueue.isNotEmpty) {
      final chunk = _chunkQueue.removeFirst();
      _audioController.add(chunk);
    }
    
    if (_isFlushing && _chunkQueue.isEmpty) {
      _currentState = PlaybackState.completed;
      _statusController.add(_buildStatus());
    }
  }
}
```

#### 4.2.2 录音质量保障策略

```dart
/// 录音质量监控器
class RecordingQualityMonitor {
  static const double _minAcceptableVolume = -35.0; // dBFS，低于此值提示声音太小
  static const double _maxClippingVolume = -3.0;    // dBFS，高于此值提示爆音
  static const double _minSnr = 15.0;               // 信噪比下限
  
  /// 实时分析 PCM 数据块
  QualityAnalysis analyze(Uint8List pcmChunk) {
    final samples = _bytesToSamples(pcmChunk);
    
    return QualityAnalysis(
      volume: _calculateRMS(samples),
      peakVolume: _calculatePeak(samples),
      isClipping: _detectClipping(samples),
      snr: _estimateSNR(samples),
      suggestion: _generateSuggestion(samples),
    );
  }
  
  String? _generateSuggestion(List<int> samples) {
    final rms = _calculateRMS(samples);
    
    if (rms < _minAcceptableVolume) {
      return '声音太小，请靠近麦克风或提高音量';
    }
    if (_detectClipping(samples)) {
      return '声音过大，可能导致爆音，请稍微远离麦克风';
    }
    return null;
  }
  
  double _calculateRMS(List<int> samples) {
    if (samples.isEmpty) return 0;
    double sumSquares = 0;
    for (final s in samples) {
      sumSquares += (s * s) / 32768.0;  // 16-bit normalize
    }
    final rms = sqrt(sumSquares / samples.length);
    return 20 * log(rms.abs() + 1e-10) / ln10;  // 转 dBFS
  }
}
```

#### 4.2.3 VAD 语音活动检测

```dart
/// 基于能量 + 过零率的 VAD 检测器
class VadDetector {
  final VadConfig config;
  final RingBuffer<double> _energyHistory = RingBuffer(30);  // 30 帧历史
  final RingBuffer<double> _zcrHistory = RingBuffer(30);     // 过零率历史
  
  bool _isSpeaking = false;
  DateTime? _speechStartTime;
  DateTime? _lastSpeechTime;
  
  /// 处理一帧 PCM 数据，返回 VAD 决策
  VadDecision process(Uint8List pcmChunk, int sampleRate) {
    final samples = _bytesToSamples(pcmChunk);
    final energy = _calculateEnergy(samples);
    final zcr = _calculateZeroCrossingRate(samples);
    
    _energyHistory.add(energy);
    _zcrHistory.add(zcr);
    
    final dynamicThreshold = _calculateDynamicThreshold();
    final isVoiceActive = energy > dynamicThreshold && zcr < _zcrUpperBound();
    
    return _updateState(isVoiceActive);
  }
  
  VadDecision _updateState(bool isVoiceActive) {
    final now = DateTime.now();
    
    if (isVoiceActive) {
      if (!_isSpeaking) {
        _isSpeaking = true;
        _speechStartTime = now;
        return VadDecision.speechStart;
      }
      _lastSpeechTime = now;
      return VadDecision.speaking;
    }
    
    if (_isSpeaking && _lastSpeechTime != null) {
      final silenceDuration = now.difference(_lastSpeechTime!);
      if (silenceDuration >= config.silenceTimeout) {
        _isSpeaking = false;
        return VadDecision.speechEnd;
      }
    }
    
    return VadDecision.silence;
  }
}

enum VadDecision {
  silence,        // 静默
  speechStart,    // 语音开始
  speaking,       // 语音持续
  speechEnd,      // 语音结束
}
```

### 4.3 音频会话管理 (AudioSessionManager)

#### 4.3.1 设计说明

音频会话管理器负责协调多个音频功能的优先级和系统级音频焦点，避免以下冲突场景：

- AI 语音对话时突然收到听力练习推送音
- 播放听力音频时用户切换到后台播放音乐
- 录音过程中收到电话中断
- 多个页面同时尝试播放音频

```dart
/// 音频会话管理器
class AudioSessionManager {
  final List<AudioSession> _activeSessions = [];
  AudioSession? _currentForeground;
  
  /// 请求音频会话
  Future<AudioSessionPermission> requestSession(AudioSession session) async {
    // 检查优先级冲突
    final conflict = _findConflict(session);
    
    if (conflict != null) {
      switch (_resolveStrategy(session, conflict)) {
        case ConflictResolution.deny:
          return AudioSessionPermission.denied(conflict);
        case ConflictResolution.duckOthers:
          await _duck(conflict, reduceBy: 0.3);
          break;
        case ConflictResolution.pauseOthers:
          await _pause(conflict);
          break;
        case ConflictResolution.interrupt:
          await _interrupt(conflict);
          break;
      }
    }
    
    _activeSessions.add(session);
    _configureSystemAudioSession(session);
    
    return AudioSessionPermission.granted();
  }
  
  /// 释放音频会话
  Future<void> releaseSession(AudioSession session) async {
    _activeSessions.remove(session);
    
    // 恢复被降低/暂停的会话
    final resumable = _activeSessions.where((s) => s.wasInterrupted);
    for (final s in resumable) {
      await _resume(s);
    }
  }
  
  ConflictResolution _resolveStrategy(AudioSession newSession, AudioSession existing) {
    // 优先级矩阵
    if (newSession.category == AudioCategory.recording &&
        existing.category == AudioCategory.playback) {
      return ConflictResolution.duckOthers;  // 录音时降低播放音量
    }
    if (newSession.category == AudioCategory.alarm &&
        existing.category != AudioCategory.alarm) {
      return ConflictResolution.interrupt;    // 闹钟中断一切
    }
    if (newSession.priority > existing.priority) {
      return ConflictResolution.pauseOthers;
    }
    return ConflictResolution.deny;
  }
}

/// 音频会话
class AudioSession {
  final String id;
  final AudioCategory category;
  final int priority;               // 优先级 1-10
  final bool allowsDucking;         // 是否允许其他应用降低其音量
  final bool mixWithOthers;         // 是否允许与其他音频混音
  bool wasInterrupted = false;      // 是否被中断
```

```dart
/// 音频缓存管理器
class AudioCacheManager {
  static const _maxCacheSize = 200 * 1024 * 1024; // 200MB
  final String _cacheDir;
  final Map<String, CachedAudio> _index = {};
  
  /// 获取或下载音频文件
  Future<String> getOrDownload(AudioAsset asset) async {
    // 1. 检查持久缓存
    final cached = _index[asset.id];
    if (cached != null && await File(cached.path).exists()) {
      cached.lastAccess = DateTime.now();
      return cached.path;
    }
    
    // 2. 检查预加载缓存
    final preloaded = _preloadCache[asset.id];
    if (preloaded != null) {
      return preloaded.path;
    }
    
    // 3. 下载并缓存
    return await _downloadAndCache(asset);
  }
  
  /// 清理过期缓存
  Future<void> cleanup() async {
    final now = DateTime.now();
    final expired = _index.entries
        .where((e) => now.difference(e.value.lastAccess).inDays > 7)
        .toList();
    
    for (final entry in expired) {
      await File(entry.value.path).delete();
      _index.remove(entry.key);
    }
    
    // 如果仍超容量，按 LRU 清理
    final totalSize = _index.values.fold<int>(0, (sum, a) => sum + a.size);
    if (totalSize > _maxCacheSize) {
      final sorted = _index.entries.toList()
        ..sort((a, b) => a.value.lastAccess.compareTo(b.value.lastAccess));
      
      var remaining = totalSize;
      for (final entry in sorted) {
        if (remaining <= _maxCacheSize * 0.8) break;
        await File(entry.value.path).delete();
        remaining -= entry.value.size;
        _index.remove(entry.key);
      }
    }
  }
}
```

```dart
///
/// 用于录音中显示实时波形，帮助用户了解录音状态。
/// 也用于发音评测场景，对比标准发音和用户发音波形。
class WaveformWidget extends StatefulWidget {
  final Stream<Uint8List>? pcmStream;     // 实时 PCM 数据流
  final List<double>? staticWaveform;     // 静态波形数据（回放模式）
  final WaveformStyle style;
  final bool showCenterLine;
  final double height;
  
  const WaveformWidget({
    super.key,
    this.pcmStream,
    this.staticWaveform,
    this.style = const WaveformStyle(),
    this.showCenterLine = false,
    this.height = 60,
  });
  
  @override
  State<WaveformWidget> createState() => _WaveformWidgetState();
}

class _WaveformWidgetState extends State<WaveformWidget>
    with SingleTickerProviderStateMixin {
  late CustomPainter _painter;
  final List<double> _samples = [];
  static const _maxSamples = 100;
  
  @override
  void initState() {
    super.initState();
    
    if (widget.pcmStream != null) {
      widget.pcmStream!.listen(_onPcmData);
    }
    
    _painter = WaveformPainter(
      samples: _samples,
      style: widget.style,
    );
  }
  
  void _onPcmData(Uint8List pcm) {
    // 降采样：将 PCM 块压缩为单个幅度值
    final amplitude = _calculateAmplitude(pcm);
    
    setState(() {
      _samples.add(amplitude);
      if (_samples.length > _maxSamples) {
        _samples.removeAt(0);
      }
    });
  }
  
  double _calculateAmplitude(Uint8List pcm) {
    // 16-bit PCM samples
    double max = 0;
    for (int i = 0; i < pcm.length - 1; i += 2) {
      final sample = (pcm[i + 1] << 8) | pcm[i];
      final signed = sample.toSigned(16);
      final normalized = signed.abs() / 32768.0;
      if (normalized > max) max = normalized;
    }
    return max;
  }
  
  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      painter: WaveformPainter(
        samples: List.from(_samples),
        style: widget.style,
      ),
      size: Size(double.infinity, widget.height),
    );
  }
}

/// 波形样式
class WaveformStyle {
  final Color activeColor;        // 当前波形颜色
  final Color inactiveColor;      // 历史波形颜色
  final Color centerLineColor;    // 中心线颜色
  final double barWidth;          // 每条波形柱宽度
  final double barSpacing;        // 波形柱间距
  final bool mirrored;            // 是否上下镜像（对称显示）
  
  const WaveformStyle({
    this.activeColor = const Color(0xFF4F46E5),
    this.inactiveColor = const Color(0xFFE0E0E0),
    this.centerLineColor = const Color(0xFFBDBDBD),
    this.barWidth = 3.0,
    this.barSpacing = 2.0,
    this.mirrored = true,
  });
}

/// 波形渲染器
class WaveformPainter extends CustomPainter {
  final List<double> samples;
  final WaveformStyle style;
  
  WaveformPainter({required this.samples, required this.style});
  
  @override
  void paint(Canvas canvas, Size size) {
    final centerY = size.height / 2;
    final totalBarWidth = style.barWidth + style.barSpacing;
    final maxBars = (size.width / totalBarWidth).floor();
    
    // 只绘制最新的 N 条
    final start = samples.length > maxBars ? samples.length - maxBars : 0;
    
    for (int i = start; i < samples.length; i++) {
      final x = (i - start) * totalBarWidth;
      final amplitude = samples[i] * (size.height / 2 * 0.9);
      
      final paint = Paint()
        ..color = (i == samples.length - 1)
            ? style.activeColor
            : style.inactiveColor.withOpacity(0.3 + 0.7 * (i - start) / maxBars)
        ..strokeWidth = style.barWidth
        ..strokeCap = StrokeCap.round;
      
      if (style.mirrored) {
        // 上下镜像对称
        canvas.drawLine(
          Offset(x, centerY - amplitude),
          Offset(x, centerY + amplitude),
          paint,
        );
      } else {
        // 仅向上
        canvas.drawLine(
          Offset(x, size.height),
          Offset(x, size.height - amplitude * 2),
          paint,
        );
      }
    }
  }
  
  @override
  bool shouldRepaint(covariant WaveformPainter old) => true;
}
```

```dart
/// AI 语音对话音频管理器
class VoiceChatAudioManager {
  final PlayerEngine _player;
  final RecorderEngine _recorder;
  final AudioSessionManager _sessionManager;
  
  /// AI 语音对话播放流程
  ///
  /// 1. 用户停止说话 → VAD 检测 speechEnd
  /// 2. 停止录音，上传音频获取 ASR 结果
  /// 3. ASR 文本发送给 AI 获取回答
  /// 4. AI 回答触发 TTS 流式合成

  /// 6. 播放完成 → 恢复录音监听
  
  Future<void> startVoiceChatLoop() async {
    while (_isActive) {
      // 阶段 1: 监听用户说话
      final recording = await _listenForUserSpeech();
      
      // 阶段 2: 请求 AI 回答
      _state = VoiceChatState.thinking;
      final ttsStream = await _aiService.requestVoiceAnswer(recording);
      
      // 阶段 3: 播放 TTS 流
      _state = VoiceChatState.speaking;
      await _playTtsStream(ttsStream);
      
      // 阶段 4: 恢复监听
      _state = VoiceChatState.listening;
    }
  }
  
  Future<Uint8List> _listenForUserSpeech() async {
    final completer = Completer<Uint8List>();
    final chunks = <Uint8List>[];
    
    await _recorder.initialize(RecordingConfig(
      sampleRate: 16000,
      channels: 1,
      bitDepth: 16,
      format: RecordingFormat.pcm,
      echoCancellation: true,  // 回声消除，防止 TTS 残音
      vadConfig: VadConfig(
        enabled: true,
        silenceTimeout: const Duration(seconds: 2),
        autoStopOnSilence: true,
      ),
    ));
    
    _recorder.pcmStream.listen((chunk) {
      chunks.add(chunk);
    });
    
    _recorder.statusStream.listen((status) {
      if (status.state == RecorderState.completed) {
        final allBytes = Uint8List.fromList(chunks.expand((c) => c).toList());
        completer.complete(allBytes);
      }
    });
    
    await _recorder.start();
    return completer.future;
  }
}
```

```dart
enum DuplexMode {
  halfDuplex,    // 半双工：AI 说话时静音麦克风
  fullDuplex,    // 全双工：AI 说话时允许打断

/// 回声消除控制器
class EchoCancellationController {
  DuplexMode mode = DuplexMode.halfDuplex;
  
  /// 半双工策略：AI 播放完毕后才开始监听
  /// 优点：简单可靠，无回声干扰
  /// 缺点：无法打断 AI，体验不够自然
  void halfDuplexStrategy() {
    // 1. 播放 TTS 时完全禁用录音
    // 2. TTS 播放结束后延迟 300ms 启动录音（避免尾音）
    // 3. 录音期间禁止任何播放
  }
  
  /// 全双工策略：AI 播放时也监听麦克风，检测到用户说话则打断
  /// 优点：自然对话体验，支持打断
  /// 缺点：需要回声消除算法支持
  void fullDuplexStrategy() {
    // 1. 播放 TTS 时同时开启麦克风
    // 2. 使用 AEC (Acoustic Echo Cancellation) 消除回声
    // 3. VAD 检测到用户语音时触发打断
    // 4. 打断后立即停止 TTS 播放
  }
  
  /// 检测打断条件
  bool shouldInterrupt(Uint8List micInput) {
    // 在 TTS 播放期间，检测麦克风输入是否为真实用户语音
    // 而非 TTS 的回声
    final energy = _calculateEnergy(micInput);
    
    // 能量阈值 + 持续时间判断
    return energy > _interruptThreshold && 
           _sustainedDuration > const Duration(milliseconds: 500);
  }
}
```

```dart
/// 录音文件上传器（支持断点续传）
class RecordingUploader {
  final int _chunkSize = 256 * 1024;  // 256KB 分片
  
  /// 上传录音文件
  Future<String> upload(
    String filePath, {
    String? sessionId,
    void Function(double progress)? onProgress,
  }) async {
    final file = File(filePath);
    final fileSize = await file.length();
    final chunks = (fileSize / _chunkSize).ceil();
    
    // 检查是否有未完成的续传会话
    final resumeOffset = sessionId != null
        ? await _getResumeOffset(sessionId)
        : 0;
    
    final raf = await file.open();
    await raf.setPosition(resumeOffset);
    
    for (int i = (resumeOffset ~/ _chunkSize); i < chunks; i++) {
      final chunkData = await raf.read(_chunkSize);
      
      try {
        await _api.uploadChunk(
          sessionId: sessionId!,
          chunkIndex: i,
          totalChunks: chunks,
          data: chunkData,
        );
      } catch (e) {
        // 网络失败，保存进度，等待重试
        await _saveResumeState(sessionId, i * _chunkSize);
        rethrow;
      }
      
      onProgress?.call((i + 1) / chunks);
    }
    
    await raf.close();
    
    // 通知服务端合并完成
    final audioUrl = await _api.completeUpload(sessionId!);
    await _clearResumeState(sessionId);
    
    return audioUrl;
  }
}
```

| AUDIO_005 | SESSION_CONFLICT | 音频会话冲突 | 暂停低优先级会话 |
| REC_001 | MIC_PERMISSION_DENIED | 麦克风权限被拒 | 引导用户去设置开启 |
| REC_002 | MIC_BUSY | 麦克风被占用 | 提示关闭其他录音应用 |
| REC_003 | STORAGE_FULL | 存储空间不足 | 提示清理空间 |
| REC_004 | RECORD_TOO_SHORT | 录音过短 (<0.5s) | 提示录音时长不足 |
| REC_005 | RECORD_TOO_LONG | 超出最大时长 | 自动截断并提交 |
| REC_006 | NO_SPEECH_DETECTED | VAD 未检测到语音 | 提示重新录音 |
| REC_007 | LOW_QUALITY | 录音质量过低 | 提示环境噪声或麦克风问题 |
| TTS_001 | TTS_SERVICE_ERROR | TTS 服务异常 | 降级为纯文本显示 |
| TTS_002 | TTS_RATE_LIMIT | TTS 并发限制 | 排队等待或降级 |
| ASR_001 | ASR_SERVICE_ERROR | ASR 服务异常 | 提示重试 |
| ASR_002 | ASR_LOW_CONFIDENCE | ASR 识别置信度低 | 提供文本输入替代 |

### 9.2 统一错误处理

```dart
/// 音频模块统一错误处理
class AudioErrorHandler {
  /// 处理播放错误
  static AudioErrorAction handlePlayerError(AudioError error) {
    switch (error.code) {
      case 'AUDIO_001': // 加载失败
        return AudioErrorAction.retry(
          maxRetries: 3,
          backoff: const [Duration(seconds: 1), Duration(seconds: 3), Duration(seconds: 5)],
          fallback: AudioErrorAction.showAlert(
            title: '音频加载失败',
            message: '请检查网络后重试',
            actions: [AlertAction.retry, AlertAction.cancel],
          ),
        );
        
      case 'AUDIO_003': // 下载超时
        return AudioErrorAction.retryWithFallback(
          maxRetries: 2,
          fallback: AudioErrorAction.useCache(),
        );
        
      case 'AUDIO_005': // 会话冲突
        return AudioErrorAction.defer(
          delay: const Duration(milliseconds: 500),
        );
        
      default:
        return AudioErrorAction.showAlert(
          title: '播放异常',
          message: error.message,
          actions: [AlertAction.retry, AlertAction.cancel],
        );
    }
  }
  
  /// 处理录音错误
  static AudioErrorAction handleRecorderError(AudioError error) {
    switch (error.code) {
      case 'REC_001': // 麦克风权限
        return AudioErrorAction.showPermissionGuide(
          permission: 'microphone',
          guide: '请在设置中允许 PrimeTop 访问麦克风',
        );
        
      case 'REC_002': // 麦克风被占用
        return AudioErrorAction.showAlert(
          title: '麦克风被占用',
          message: '请关闭其他正在使用麦克风的应用后重试',
          actions: [AlertAction.retry, AlertAction.cancel],
        );
        
      case 'REC_006': // 无语音
        return AudioErrorAction.showAlert(
          title: '未检测到语音',
          message: '请靠近麦克风，用正常音量说话',
          actions: [AlertAction.retry],
          autoDismissAfter: const Duration(seconds: 3),
        );
        
      default:
        return AudioErrorAction.showAlert(
          title: '录音异常',
          message: error.message,
          actions: [AlertAction.retry, AlertAction.cancel],
        );
    }
  }
}
```

## 10. 性能优化

### 10.1 内存管理

```dart
/// 音频内存管理器
class AudioMemoryManager {
  static const _maxBufferBytes = 10 * 1024 * 1024;  // 10MB 播放缓冲
  static const _maxRecordingBufferBytes = 5 * 1024 * 1024;  // 5MB 录音缓冲
  
  /// 监听内存压力，主动释放资源
  void onMemoryPressure() {
    // 1. 释放音频预加载缓存（保留当前播放的）
    _preloadCache.evictAll(keepKeys: {_currentPlayingId});
    
    // 2. 清理波形采样历史
    _waveformBuffer.clear();
    
    // 3. 释放未使用的 PlayerEngine 实例
    _playerPool.releaseIdle();
    
    // 4. 触发 GC
    SystemChannels.system.invokeMethod('SystemSound.play');
  }
}
```

### 10.2 电量优化策略

| 策略 | 触发条件 | 措施 |
|------|----------|------|
