# 学科与科目基础数据管理详细设计

## 1. 概述

### 1.1 设计目标

学科与科目基础数据管理系统是 PrimeTop 全学段 AI 辅助学习软件的核心基础数据服务，负责：
- 维护学科基础信息、分类体系、年级适配范围
- 提供学科相关配置（图标、颜色、排序等）
- 支持多维度学科查询与筛选
- 为其他业务模块提供学科维度数据支持
- 支持学科数据的动态配置与灰度发布

### 1.2 适用场景

- 学生设置年级时自动展示可选学科
- 拍题答疑时识别题目所属学科
- 错题本按学科分类整理
- 学情分析按学科生成能力画像
- 学习计划按学科分配时间
- 首页展示学科学习进度卡片
- 教材版本按学科分类管理

### 1.3 模块边界

**包含：**
- 学科基础信息管理
- 学科分类体系
- 学科与年级映射关系
- 学科 UI 配置

**不包含：**
- 学科内容（章节、知识点、题目）- 由其他模块管理
- 学科能力评估 - 由学情分析模块负责
- 学科工具（计算器、画板等）- 由学科工具模块负责

```sql
CREATE TABLE subjects (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '学科ID',

    icon_url VARCHAR(255) COMMENT '学科图标URL',
    color_hex VARCHAR(7) COMMENT '学科主题色（十六进制）',
    description TEXT COMMENT '学科描述',
    sort_order INT NOT NULL DEFAULT 0 COMMENT '排序权重（越小越靠前）',
    min_grade TINYINT COMMENT '适用最低年级（1-12，含幼儿园0）',
    max_grade TINYINT COMMENT '适用最高年级（1-12，含幼儿园0）',
    min_stage VARCHAR(10) COMMENT '适用最低学段：kindergarten(幼儿园),primary(小学),junior(初中),senior(高中)',
    max_stage VARCHAR(10) COMMENT '适用最高学段：kindergarten(幼儿园),primary(小学),junior(初中),senior(高中)',
    is_active BOOLEAN NOT NULL DEFAULT TRUE COMMENT '是否启用',
    is_required BOOLEAN NOT NULL DEFAULT TRUE COMMENT '是否必修（默认所有学生都需要学）',
    allow_optional BOOLEAN DEFAULT FALSE COMMENT '是否允许学生自主选择添加',
    parent_id BIGINT COMMENT '父学科ID（用于学科细分，如：初中物理、高中物理）',
    extra_config JSON COMMENT '扩展配置（学科特定配置）',

    -- 时间字段
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    created_by BIGINT COMMENT '创建人ID',
    updated_by BIGINT COMMENT '更新人ID',

    -- 索引
    INDEX idx_code (code),
    INDEX idx_category (category),
    INDEX idx_active_sort (is_active, sort_order),
    INDEX idx_parent (parent_id),
    INDEX idx_grade_range (min_grade, max_grade),
    INDEX idx_stage_range (min_stage, max_stage),
    FOREIGN KEY (parent_id) REFERENCES subjects(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='学科基础信息表';
```

## 4. 核心实现代码示例

### 4.1 学科服务核心逻辑

```typescript
// subject.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Subject, SubjectGrade, StudentSubject } from '../entities';
import { CacheService } from '../cache/cache.service';
import { SubjectCategory, Stage } from '../enums';

@Injectable()
export class SubjectService {
  constructor(
    @InjectRepository(Subject)
    private readonly subjectRepo: Repository<Subject>,
    @InjectRepository(SubjectGrade)
    private readonly subjectGradeRepo: Repository<SubjectGrade>,
    @InjectRepository(StudentSubject)
    private readonly studentSubjectRepo: Repository<StudentSubject>,
    private readonly cacheService: CacheService,
  ) {}

  /**
   * 获取所有学科（带缓存）
   */
  async getAllSubjects(options: {
    category?: SubjectCategory;
    gradeLevel?: number;
    stage?: Stage;
    onlyActive?: boolean;
    withChildren?: boolean;
  }): Promise<Subject[]> {
    const cacheKey = this.buildCacheKey(options);
    const cached = await this.cacheService.get(cacheKey);

    if (cached) {
      return JSON.parse(cached);
    }

    const queryBuilder = this.subjectRepo.createQueryBuilder('subject');

    // 筛选条件
    if (options.category) {
      queryBuilder.andWhere('subject.category = :category', { category: options.category });
    }
    if (options.onlyActive !== false) {
      queryBuilder.andWhere('subject.isActive = true');
    }

    // 年级学段过滤（联合查询 subject_grades 表）
    if (options.gradeLevel || options.stage) {
      queryBuilder.innerJoin(
        'subject.grades',
        'grade',
        'grade.isAvailable = true'
      );

      if (options.gradeLevel) {
        queryBuilder.andWhere('grade.gradeLevel = :gradeLevel', { gradeLevel: options.gradeLevel });
      }
      if (options.stage) {
        queryBuilder.andWhere('grade.stage = :stage', { stage: options.stage });
      }
    }

    queryBuilder.orderBy('subject.sortOrder', 'ASC');

    const subjects = await queryBuilder.getMany();

    // 加载子学科
    if (options.withChildren) {
      await Promise.all(
        subjects.map(async (subject) => {
          subject.children = await this.getChildren(subject.id);
        })
      );
    }

    // 缓存 10 分钟
    await this.cacheService.set(cacheKey, JSON.stringify(subjects), 600);

    return subjects;
  }

  /**
   * 获取学科详情
   */
  async getSubjectById(subjectId: number): Promise<Subject | null> {
    // 先查缓存
    const cacheKey = `subject:info:${subjectId}`;
    const cached = await this.cacheService.get(cacheKey);

    if (cached) {
      return JSON.parse(cached);
    }

    const subject = await this.subjectRepo.findOne({
      where: { id: subjectId },
      relations: ['parent', 'grades'],
    });

    if (subject) {
      // 缓存 30 分钟
      await this.cacheService.set(cacheKey, JSON.stringify(subject), 1800);
    }

    return subject;
  }

  /**
   * 获取学生当前学习的学科列表
   */
  async getStudentSubjects(
    userId: number,
    options: { onlyActive?: boolean } = {}
  ): Promise<StudentSubject[]> {
    const cacheKey = options.onlyActive !== false
      ? `student:${userId}:subjects:active`
      : `student:${userId}:subjects`;

    const cached = await this.cacheService.get(cacheKey);

    if (cached) {
      return JSON.parse(cached);
    }

    const queryBuilder = this.studentSubjectRepo
      .createQueryBuilder('ss')
      .leftJoinAndSelect('ss.subject', 'subject')
      .where('ss.userId = :userId', { userId });

    if (options.onlyActive !== false) {
      queryBuilder.andWhere('ss.isActive = true');
    }

    const studentSubjects = await queryBuilder.getMany();

    // 缓存 5 分钟
    await this.cacheService.set(cacheKey, JSON.stringify(studentSubjects), 300);

    return studentSubjects;
  }

  /**
   * 学生添加学科（选修）
   */
  async addSubjectForStudent(
    userId: number,
    subjectId: number,
    gradeLevel: number,
    stage: Stage,
    semester: number
  ): Promise<StudentSubject> {
    // 1. 检查学科是否存在且允许选修
    const subject = await this.getSubjectById(subjectId);
    if (!subject || !subject.isActive) {
      throw new BusinessException('学科不存在或已下架');
    }

    if (!subject.allowOptional) {
      throw new BusinessException('该学科不允许自主添加');
    }

    // 2. 检查该年级学段是否有该学科
    const subjectGrade = await this.subjectGradeRepo.findOne({
      where: {
        subjectId,
        gradeLevel,
        stage,
        semester,
        isAvailable: true,
      },
    });

    if (!subjectGrade) {
      throw new BusinessException('该年级学段暂未开设此学科');
    }

    // 3. 检查是否已经添加
    const existing = await this.studentSubjectRepo.findOne({
      where: {
        userId,
        subjectId,
        gradeLevel,
        stage,
        semester,
        isActive: true,
      },
    });

    if (existing) {
      throw new BusinessException('您已添加该学科');
    }

    // 4. 创建选科记录
    const studentSubject = this.studentSubjectRepo.create({
      userId,
      subjectId,
      gradeLevel,
      stage,
      semester,
      isActive: true,
      isOptional: true,
      startAt: new Date(),
    });

    await this.studentSubjectRepo.save(studentSubject);

    // 5. 清除缓存
    await this.cacheService.del(`student:${userId}:subjects:active`);

    return studentSubject;
  }

  /**
   * 学生移除学科
   */
  async removeSubjectForStudent(userId: number, subjectId: number): Promise<void> {
    const studentSubject = await this.studentSubjectRepo.findOne({
      where: {
        userId,
        subjectId,
        isActive: true,
      },
    });

    if (!studentSubject) {
      throw new BusinessException('未找到该学科');
    }

    if (!studentSubject.isOptional) {
      throw new BusinessException('必修学科不能移除');
    }

    // 软删除（设置 isActive = false）
    studentSubject.isActive = false;
    studentSubject.endAt = new Date();

    await this.studentSubjectRepo.save(studentSubject);

    // 清除缓存
    await this.cacheService.del(`student:${userId}:subjects:active`);
  }

  /**
   * 批量为学生初始化必修学科
   */
  async initializeRequiredSubjects(
    userId: number,
    gradeLevel: number,
    stage: Stage,
    semester: number
  ): Promise<void> {
    // 1. 获取该年级学段的必修学科
    const requiredSubjectGrades = await this.subjectGradeRepo
      .createQueryBuilder('sg')
      .leftJoinAndSelect('sg.subject', 'subject')
      .where('sg.gradeLevel = :gradeLevel', { gradeLevel })
      .andWhere('sg.stage = :stage', { stage })
      .andWhere('sg.semester = :semester', { semester })
      .andWhere('sg.isAvailable = true')
      .andWhere('sg.isRequired = true')
      .andWhere('subject.isActive = true')
      .andWhere('subject.isRequired = true')
      .getMany();

    // 2. 批量创建选科记录
    const studentSubjects = requiredSubjectGrades.map((sg) => ({
      userId,
      subjectId: sg.subjectId,
      gradeLevel,
      stage,
      semester,
      isActive: true,
      isOptional: false,
      startAt: new Date(),
    }));

    await this.studentSubjectRepo.save(studentSubjects);

    // 3. 清除缓存
    await this.cacheService.del(`student:${userId}:subjects:active`);
  }

  /**
   * 构建缓存 Key
   */
  private buildCacheKey(options: {
    category?: SubjectCategory;
    gradeLevel?: number;
    stage?: Stage;
    onlyActive?: boolean;
  }): string {
    const parts = ['subject'];

    if (options.category) parts.push(`cat:${options.category}`);
    if (options.gradeLevel) parts.push(`grade:${options.gradeLevel}`);
    if (options.stage) parts.push(`stage:${options.stage}`);
    if (options.onlyActive !== false) parts.push('active');

    return parts.join(':');
  }

  /**
   * 获取子学科
   */
  private async getChildren(parentId: number): Promise<Subject[]> {
    return this.subjectRepo.find({
      where: { parentId, isActive: true },
      order: { sortOrder: 'ASC' },
    });
  }
}
```

### 4.3 缓存管理

```typescript
// cache.service.ts
import { Injectable } from '@nestjs/common';
import { Redis } from 'ioredis';

@Injectable()
export class CacheService {
  private readonly redis: Redis;

  constructor() {
    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: parseInt(process.env.REDIS_PORT || '6379'),
      password: process.env.REDIS_PASSWORD,
      db: parseInt(process.env.REDIS_DB || '0'),
    });
  }

  /**
   * 获取缓存
   */
  async get(key: string): Promise<string | null> {
    return await this.redis.get(key);
  }

  /**
   * 设置缓存
   */
  async set(key: string, value: string, ttl?: number): Promise<void> {
    if (ttl) {
      await this.redis.setex(key, ttl, value);
    } else {
      await this.redis.set(key, value);
    }
  }

  /**
   * 删除缓存
   */
  async del(key: string): Promise<void> {
    await this.redis.del(key);
  }

  /**
   * 批量删除缓存（支持通配符）
   */
  async delPattern(pattern: string): Promise<void> {
    const keys = await this.redis.keys(pattern);
    if (keys.length > 0) {
      await this.redis.del(...keys);
    }
  }

  /**
   * 清除学科相关缓存
   */
  async clearSubjectCache(subjectId?: number): Promise<void> {
    if (subjectId) {
      // 清除特定学科缓存
      await this.del(`subject:info:${subjectId}`);
    } else {
      // 清除所有学科缓存
      await this.delPattern('subject:*');
    }
  }

  /**
   * 清除学生选科缓存
   */
  async clearStudentSubjectCache(userId: number): Promise<void> {
    await this.del(`student:${userId}:subjects`);
    await this.del(`student:${userId}:subjects:active`);
  }
}
```

```typescript
// exception.filter.ts
import { ExceptionFilter, Catch, ArgumentsHost, HttpException } from '@nestjs/common';
import { Response } from 'express';

@Catch()
export class AllExceptionsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();

    let code = 500;
    let message = '服务器内部错误';

    if (exception instanceof HttpException) {
      code = exception.getStatus();
      message = exception.message;
    } else if (exception instanceof BusinessException) {
      code = exception.code;
      message = exception.message;
    }

    response.status(code).json({
      code,
      message,
      data: null,
      timestamp: new Date().toISOString(),
    });
  }
}

// 业务异常类
export class BusinessException extends Error {
  constructor(
    public readonly code: number,
    message: string
  ) {
    super(message);
  }
}
```
