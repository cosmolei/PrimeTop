## 1. 概述

### 1.1 功能定位

本组件为 PrimeTop 客户端提供**数学函数图形绘制**与**交互式几何画板**两大核心能力，供学生在理科学习、解题过程、知识探索等场景中使用。组件以 Flutter CustomPainter + Canvas 为底层渲染引擎，支持触控交互、动态参数调节、多函数叠加显示。

### 1.2 适用场景

| 场景 | 用途 | 触发方式 |
|------|------|----------|
| AI 辅导讲解 | AI 生成函数图像辅助讲解概念 | AI 回答卡片内嵌图形 |
| 拍题答疑解析 | 解析中附带函数/几何示意图 | 解析卡片内嵌图形 |
| 同步课堂学习 | 章节内容中的图形化知识展示 | 教材内容卡片内嵌 |
| 自由探索工具 | 学生自主输入函数、绘制几何图形 | 学科工具箱独立入口 |
| 练习答题 | 几何作答题、函数选择题的图形交互 | 答题组件内嵌 |

### 1.3 设计原则

1. **轻量内嵌**：作为 Widget 可嵌入任意页面，不依赖独立页面。
2. **触控友好**：所有操作支持手势交互（拖拽、缩放、点击）。
3. **适龄适配**：低年级简化操作入口，高年级开放完整功能。
4. **离线可用**：核心绘制引擎纯客户端计算，不依赖网络。
5. **性能优先**：60fps 渲染，函数曲线采样自适应屏幕分辨率。

```dart
/// 函数表达式类型
enum FunctionType {
  linear,        // 一次函数: y = kx + b
  quadratic,     // 二次函数: y = ax² + bx + c
  polynomial,    // 多项式: y = aₙxⁿ + ... + a₁x + a₀
  trigonometric, // 三角函数: y = A·sin(ωx + φ) + k 等
  exponential,   // 指数函数: y = a·bˣ + k
  logarithmic,   // 对数函数: y = a·ln(bx) + k
  rational,      // 分式函数: y = P(x) / Q(x)
  parametric,    // 参数方程: x=f(t), y=g(t)
  piecewise,     // 分段函数
  implicit,      // 隐函数: F(x,y) = 0（如圆锥曲线）
}

/// 函数曲线定义
class FunctionCurve {
  final String id;                  // 唯一标识
  final FunctionType type;
  final String expression;          // 原始表达式字符串，如 "2x² - 3x + 1"
  final String displayName;         // 显示名，如 "y = 2x² - 3x + 1"
  final Map<String, double> params; // 参数值 {a: 2, b: -3, c: 1}
  final List<ParamConstraint> paramConstraints; // 参数约束（可调范围）

  /// 样式
  final CurveStyle style;
  final bool visible;
  final bool animated;              // 是否启用动画（如描点动画）

  /// 动态参数绑定（与滑块关联）
  final Map<String, String>? paramBindings; // { "a": "slider_1" }

}
```

### 5.1 函数曲线采样与绘制

```dart
/// 曲线渲染器
class CurveRenderer {
  final Canvas canvas;
  final Viewport viewport;
  final Size canvasSize;

  /// 渲染一条函数曲线
  void renderCurve(FunctionCurve curve, GraphTheme theme) {
    final points = _sampleCurve(curve);
    final screenPoints = points.map(
      (p) => viewport.mathToScreen(p, canvasSize)
    ).toList();

    final paint = Paint()
      ..color = curve.style.color
      ..strokeWidth = curve.style.strokeWidth
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    if (curve.style.lineStyle == LineStyle.dashed) {
      paint.path = _buildDashedPath(screenPoints);
    }

    // 绘制曲线路径
    final path = Path();
    if (screenPoints.isNotEmpty) {
      path.moveTo(screenPoints.first.dx, screenPoints.first.dy);
      for (int i = 1; i < screenPoints.length; i++) {
        path.lineTo(screenPoints[i].dx, screenPoints[i].dy);
      }
    }
    canvas.drawPath(path, paint);

    // 绘制关键点
    if (curve.style.showPoints) {
      _renderKeyPoints(curve, theme);
    }
  }

  /// 自适应采样：在曲率大的区域增加采样密度
  List<Point2D> _sampleCurve(FunctionCurve curve) {
    final rect = viewport.visibleMathRect;
    final pixelWidth = canvasSize.width;
    final targetSamplesPerPixel = 2.0; // 每像素2个采样点
    final baseStep = rect.width / (pixelWidth * targetSamplesPerPixel);

    final points = <Point2D>[];
    double x = rect.left;
    while (x <= rect.right) {
      final y = _evaluateExpr(curve, x);
      if (y.isFinite) {
        points.add(Point2D(x, y));
      }
      // 根据曲率自适应步进
      final step = _adaptiveStep(curve, x, baseStep);
      x += step;
    }
    return points;
  }

  /// 自适应步进：曲率大时减小步进
  double _adaptiveStep(FunctionCurve curve, double x, double baseStep) {
    final y0 = _evaluateExpr(curve, x - baseStep);
    final y1 = _evaluateExpr(curve, x);
    final y2 = _evaluateExpr(curve, x + baseStep);

    if (!y0.isFinite || !y1.isFinite || !y2.isFinite) {
      return baseStep;
    }

    // 二阶差分近似曲率
    final curvature = ((y2 - y1) - (y1 - y0)).abs();
    if (curvature > 10) return baseStep / 4;
    if (curvature > 1) return baseStep / 2;
    return baseStep;
  }

  double _evaluateExpr(FunctionCurve curve, double x) {
    return MathExpressionEvaluator.evaluate(
      curve.expression, curve.params, x,
    );
  }
}
```

```dart
    final gridPaint = Paint()
      ..color = theme.gridColor.withOpacity(0.3)
      ..strokeWidth = 0.5;

    final rect = viewport.visibleMathRect;
    final spacing = _autoGridSpacing(rect.width);

    // 竖线
    double x = (rect.left / spacing).ceil() * spacing;
    while (x <= rect.right) {
      final sx = viewport.mathToScreen(Point2D(x, 0), size).dx;
      canvas.drawLine(Offset(sx, 0), Offset(sx, size.height), gridPaint);
      x += spacing;
    }

    // 横线
    double y = (rect.top / spacing).ceil() * spacing;
    while (y <= rect.bottom) {
      final sy = viewport.mathToScreen(Point2D(0, y), size).dy;
      canvas.drawLine(Offset(0, sy), Offset(size.width, sy), gridPaint);
      y += spacing;
    }
  }

  void _drawAxes() {
    final axisPaint = Paint()
      ..color = theme.axisColor
      ..strokeWidth = 1.5;

    final origin = viewport.mathToScreen(Point2D(0, 0), size);

    // X 轴
    if (origin.dy >= 0 && origin.dy <= size.height) {
      canvas.drawLine(
        Offset(0, origin.dy), Offset(size.width, origin.dy), axisPaint,
      );
      // 箭头
      _drawArrow(canvas, Offset(size.width - 8, origin.dy - 4),
                  Offset(size.width, origin.dy),
                  Offset(size.width - 8, origin.dy + 4), axisPaint);
    }

    // Y 轴
    if (origin.dx >= 0 && origin.dx <= size.width) {
      canvas.drawLine(
        Offset(origin.dx, 0), Offset(origin.dx, size.height), axisPaint,
      );
      _drawArrow(canvas, Offset(origin.dx - 4, 8),
                  Offset(origin.dx, 0),
                  Offset(origin.dx + 4, 8), axisPaint);
    }
  }

  /// 自动计算网格间距（1-2-5 序列）
  double _autoGridSpacing(double viewWidth) {
    final targetLines = 8; // 目标网格线数
    final rawSpacing = viewWidth / targetLines;
    final magnitude = pow(10, (log(rawSpacing) / ln10).floor()).toDouble();
    final normalized = rawSpacing / magnitude;
    if (normalized < 1.5) return magnitude;
    if (normalized < 3.5) return 2 * magnitude;
    if (normalized < 7.5) return 5 * magnitude;
    return 10 * magnitude;
  }
}
```

```dart
class HitTestEngine {
  static const double defaultTolerance = 8.0; // 像素

  /// 检测屏幕坐标命中了哪个图形
  static HitResult? hitTest({
    required Point2D screenPoint,
    required List<GeoShape> shapes,
    required Viewport viewport,
    required Size canvasSize,
    double tolerance = defaultTolerance,
  }) {
    final mathPoint = viewport.screenToMath(screenPoint, canvasSize);
    final mathTolerance = tolerance * viewport.scaleX;

    // 反向遍历（后绘制的在上层）
    for (int i = shapes.length - 1; i >= 0; i--) {
      final shape = shapes[i];
      if (shape.hitTest(mathPoint, mathTolerance)) {
        // 检查是否命中控制点
        for (int j = 0; j < shape.controlPoints.length; j++) {
          if ((shape.controlPoints[j] - mathPoint).distance < mathTolerance) {
            return HitResult(
              shapeId: shape.id,
              controlPointIndex: j,
              type: HitType.controlPoint,
            );
          }
        }
        return HitResult(
          shapeId: shape.id,
          controlPointIndex: null,
          type: HitType.body,
        );
      }
    }
    return null;
  }
}

class HitResult {
  final String shapeId;
  final int? controlPointIndex;
  final HitType type;
}
```

```dart
    if (_pendingPoints.length >= _requiredPointsForTool(_currentTool)) {

    _pendingPoints.clear();

          return GeoLineSegment(
            start: _pendingPoints[0],
            end: currentMouse,
          );
        }
      case InteractionMode.drawCircle:
        if (_pendingPoints.length == 1) {
          return GeoCircle(
            center: _pendingPoints[0],
            radius: (_pendingPoints[0] - currentMouse).distance,
          );
        }
      // ... 其他工具
      default:
        return null;
    }
  }
```

```dart
class MathExpressionEvaluator {
  /// 评估函数表达式在 x 处的值
  static double evaluate(
    String expression,
    Map<String, double> params,
    double x,
  ) {
    // 1. 预处理：将表达式标准化
    final normalized = _normalize(expression);

    // 2. 替换参数和变量
    final substituted = _substitute(normalized, params, x);

    // 3. 构建 AST
    final ast = _parse(substituted);

    // 4. 递归求值
    return _evaluateAST(ast);
  }

  static String _normalize(String expr) {
    return expr
        .replaceAll('×', '*')
        .replaceAll('÷', '/')
        .replaceAll('π', '${pi}')
        .replaceAll('²', '^2')
        .replaceAll('³', '^3')
        .replaceAllMapped(
          RegExp(r'(\d)([a-zA-Z(])'), // 隐式乘法: 2x → 2*x
          (m) => '${m[1]}*${m[2]}',
        )
        .replaceAllMapped(
          RegExp(r'\)([\da-zA-Z(])'), // )( → )*(  , )2 → )*2
          (m) => ')*${m[1]}',
        );
  }

  /// 使用递归下降解析器构建 AST
  static ASTNode _parse(String expr) {
    final parser = ExprParser(TokenLexer(expr));
    return parser.parseExpression();
  }

  static double _evaluateAST(ASTNode node) {
    return node.accept(_EvaluatorVisitor());
  }
}
```

```dart
  final CurveAnimationType type;

  /// 根据动画进度返回可见的采样点子集
  List<Point2D> getVisiblePoints(
    List<Point2D> allPoints,
    double progress, // 0.0 ~ 1.0
  ) {
    switch (type) {
      case CurveAnimationType.drawFromLeft:
        final count = (allPoints.length * progress).ceil();
        return allPoints.sublist(0, count);

      case CurveAnimationType.morphFromLine:

        return allPoints.map((p) {
          final lineY = p.x * (allPoints.last.y / allPoints.last.x);
          final targetY = p.y;
          final y = lineY + (targetY - lineY) * progress;
          return Point2D(p.x, y);
        }).toList();

      default:
        return allPoints;
    }
  }
```

/// AI 回答中的图形指令解析
class GraphInstructionParser {
  /// 从 AI 回答文本中提取图形指令
  ///
  /// AI 回答格式约定：
  /// ```graph
  /// type: function
  /// curves:
  ///   - y = 2x^2 - 3x + 1
  ///   - y = x - 1
  /// viewport: [-5, 5, -10, 10]
  /// highlight: intersection
  /// ```
  static GraphInstruction? parse(String aiResponse) {
    final graphBlock = RegExp(r'```graph\n([\s\S]*?)```')
        .firstMatch(aiResponse)?.group(1);
    if (graphBlock == null) return null;

    final lines = graphBlock.split('\n').where((l) => l.trim().isNotEmpty);

    GraphType? type;
    final curves = <String>[];
    Viewport? viewport;
    String? highlight;

    for (final line in lines) {
      if (line.startsWith('type:')) {
        type = line.substring(5).trim() == 'function'
            ? GraphType.function : GraphType.geometry;
      } else if (line.startsWith('  - ')) {
        curves.add(line.substring(4).trim());
      } else if (line.startsWith('viewport:')) {
        final parts = line.substring(9).trim()
            .split(',').map(double.parse).toList();
        viewport = Viewport.fromBounds(parts[0], parts[1], parts[2], parts[3]);
      } else if (line.startsWith('highlight:')) {
        highlight = line.substring(10).trim();
      }
    }

    return GraphInstruction(
      type: type ?? GraphType.function,
      curveExpressions: curves,
      viewport: viewport,
      highlight: highlight,
    );
  }
}

class GraphInstruction {
  final GraphType type;
  final List<String> curveExpressions;
  final Viewport? viewport;
  final String? highlight; // intersection, extremum, tangent, etc.

```dart
        // 绘制错误提示占位
        _renderErrorPlaceholder(canvas, curve, size);
      }
    }
  }

  void _renderErrorPlaceholder(Canvas canvas, FunctionCurve curve, Size size) {
    final paint = Paint()..color = Colors.red.withOpacity(0.3);
    canvas.drawRect(
      Rect.fromLTWH(size.width - 120, 10, 110, 30),
      paint,
    );
    // 绘制 "渲染失败" 提示
  }
}
```

### 14.1 表达式解析测试

| 用例 | 输入 | 预期 |
|------|------|------|
| 隐式乘法 | `2x` | `2*x` → 正确求值 |

| 用例 | 操作 | 预期 |
|------|------|------|
| 缩放 | 双指放大2x | 网格间距减半，曲线平滑 |

### 15.2 集成示例（AI 辅导页）

```dart
/// AI 回答消息气泡中的图形渲染
class AiMessageBubble extends StatelessWidget {
  final AiMessage message;

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        // 文本内容
        MarkdownBody(data: message.textBeforeGraph),

        // 自动检测并渲染图形
        if (message.hasGraphInstruction)
          _buildEmbeddedGraph(message.graphInstruction!),

        // 继续的文本内容
        if (message.textAfterGraph != null)
          MarkdownBody(data: message.textAfterGraph!),
      ],
    );
  }

  Widget _buildEmbeddedGraph(GraphInstruction instruction) {
    return Container(
      height: 240,
      margin: const EdgeInsets.symmetric(vertical: 8),
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: Colors.grey.shade300),
      ),
      child: ClipRRect(
        borderRadius: BorderRadius.circular(12),
        child: MathGraphWidget(
          config: GraphConfig.embedded(
            viewport: instruction.viewport,
          ),
          initialCurves: instruction.toFunctionCurves(),
          interactive: true,  // 可平移/缩放
          showToolbar: false, // 嵌入模式不显示工具栏
          onCurveTapped: (curve) {
            // 点击曲线可以展开到大图查看
            Navigator.of(context).push(
              MaterialPageRoute(
                builder: (_) => FullScreenGraphPage(curves: [curve]),
              ),
            );
          },
        ),
      ),
    );
  }
}
```
