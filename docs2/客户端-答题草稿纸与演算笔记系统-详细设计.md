# 客户端 - 答题草稿纸与演算笔记系统 详细设计

> **模块定位**：为学生提供虚拟草稿纸，在答题、解题、考试等学习场景中支持手写演算、绘制草图、快捷笔记，演算过程可回溯、可关联题目、可导出。
> **关联模块**：练习与测评系统、考试模拟与真题练习、AI智能辅导、错题整理、富文本与学科内容渲染引擎
> **优先级**：P1（V1.0 阶段增强体验）

---

## 1. 功能概述

### 1.1 核心诉求

学生在纸面做题时有草稿纸辅助演算，但在 APP 内答题时缺少类似体验。本系统提供虚拟草稿纸，让学生能：

1. **自由书写/绘制**：用手写笔或手指在 Canvas 上自由演算
2. **快捷输入**：通过符号键盘快速插入数学符号、公式片段
3. **关联题目**：草稿自动绑定到当前答题/解题场景
4. **演算回溯**：复习时可回看当时思路，辅助错因分析
5. **智能辅助**：识别手写公式并提供计算验证

### 1.2 使用场景

| 场景 | 触发方式 | 草稿生命周期 |
|------|---------|-------------|
| 练习答题 | 答题页面右上角"草稿纸"按钮 | 跟随练习会话，练习结束后归档 |
| 考试模拟 | 答题页面固定入口 | 跟随考试会话，交卷后归档 |
| AI解题 | AI对话页面"草稿"快捷按钮 | 跟随对话消息，可持久保存 |
| 错题订正 | 错题详情页"我的演算" | 持久保存，覆盖更新 |
| 自由演算 | 首页/工具箱入口 | 用户手动管理 |

## 4. Canvas 渲染引擎

### 4.1 渲染策略

```
┌─────────────────────────────────────────┐
│              CanvasWidget               │
│  ┌─────────────────────────────────┐    │
│  │  Background Layer (网格/底图)    │    │
│  ├─────────────────────────────────┤    │
│  │  Strokes Layer (已确认笔画)      │    │
│  ├─────────────────────────────────┤    │
│  │  Active Stroke (当前正在绘制)    │    │
│  ├─────────────────────────────────┤    │
│  │  Objects Layer (文本/公式块)     │    │
│  ├─────────────────────────────────┤    │
│  │  Selection Layer (选中框/手柄)   │    │
│  └─────────────────────────────────┘    │
│                                         │
│  渲染策略:                               │
│  - Background: 仅重绘配置变更时          │
│  - Strokes: 增量绘制（新笔画append）     │
│  - Active: 每帧重绘当前笔画             │
│  - Objects: 仅重绘变更对象              │
└─────────────────────────────────────────┘
```

/// 分层画布绘制器
class ScratchpadPainter extends CustomPainter {
  final ScratchpadDoc doc;
  final Stroke? activeStroke;
  final CanvasTransform transform;
  final String? selectedId;

  ScratchpadPainter({
    required this.doc,
    this.activeStroke,
    required this.transform,
    this.selectedId,
  });

  @override
  void paint(Canvas canvas, Size size) {
    // 应用画布变换
    canvas.save();
    canvas.translate(transform.offsetX, transform.offsetY);
    canvas.scale(transform.scale);

    // Layer 1: 背景网格
    _drawBackground(canvas, size);

    // Layer 2: 已确认笔画
    for (final stroke in doc.strokes) {
      if (!stroke.erased) {
        _drawStroke(canvas, stroke);
      }
    }

    // Layer 3: 活跃笔画
    if (activeStroke != null) {
      _drawStroke(canvas, activeStroke!);
    }

    // Layer 4: 文本/公式块
    for (final tb in doc.textBlocks) {
      _drawTextBlock(canvas, tb);
    }
    for (final fb in doc.formulas) {
      _drawFormulaBlock(canvas, fb);
    }

    // Layer 5: 选中态
    if (selectedId != null) {
      _drawSelection(canvas, selectedId!);
    }

    canvas.restore();
  }

  void _drawStroke(Canvas canvas, Stroke stroke) {
    final paint = Paint()
      ..color = _hexToColor(stroke.pen.color, stroke.pen.opacity)
      ..strokeWidth = stroke.pen.size
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round
      ..style = PaintingStyle.stroke;

    if (stroke.pen.isHighlighter) {
      paint.blendMode = BlendMode.multiply;
    }

    final path = Path();
    if (stroke.points.isEmpty) return;

    path.moveTo(stroke.points.first.x, stroke.points.first.y);

    // 使用贝塞尔曲线平滑
    for (int i = 1; i < stroke.points.length; i++) {
      final prev = stroke.points[i - 1];
      final curr = stroke.points[i];
      final midX = (prev.x + curr.x) / 2;
      final midY = (prev.y + curr.y) / 2;
      path.quadraticBezierTo(prev.x, prev.y, midX, midY);
    }

    // 末端
    final last = stroke.points.last;
    path.lineTo(last.x, last.y);

    canvas.drawPath(path, paint);
  }

  void _drawBackground(Canvas canvas, Size size) {
    final config = doc.canvasConfig;
    if (!config.showGrid) {
      canvas.drawRect(
        Rect.fromLTWH(0, 0, config.width, config.height),
        Paint()..color = _hexToColor(config.bgColor),
      );
      return;
    }

    // 根据网格类型绘制
    final gridPaint = Paint()
      ..color = const Color(0xFFE0E0E0)
      ..strokeWidth = 0.5;

    switch (config.gridPattern) {
      case GridPattern.square:
        for (double x = 0; x <= config.width; x += config.gridSize) {
          canvas.drawLine(Offset(x, 0), Offset(x, config.height), gridPaint);
        }
        for (double y = 0; y <= config.height; y += config.gridSize) {
          canvas.drawLine(Offset(0, y), Offset(config.width, y), gridPaint);
        }
        break;
      case GridPattern.lined:
        for (double y = config.gridSize; y <= config.height; y += config.gridSize) {
          canvas.drawLine(Offset(0, y), Offset(config.width, y), gridPaint);
        }
        break;
      case GridPattern.dot:
        for (double x = config.gridSize; x <= config.width; x += config.gridSize) {
          for (double y = config.gridSize; y <= config.height; y += config.gridSize) {
            canvas.drawCircle(Offset(x, y), 1.0, gridPaint);
          }
        }
        break;
      case GridPattern.coordinate:
        // 坐标纸：细格+粗格
        final thinPaint = Paint()..color = const Color(0xFFE8E8E8)..strokeWidth = 0.3;
        final thickPaint = Paint()..color = const Color(0xFFCCCCCC)..strokeWidth = 0.8;
        for (double x = 0; x <= config.width; x += config.gridSize / 5) {
          canvas.drawLine(
            Offset(x, 0), Offset(x, config.height),
            x % config.gridSize == 0 ? thickPaint : thinPaint,
          );
        }
        for (double y = 0; y <= config.height; y += config.gridSize / 5) {
          canvas.drawLine(
            Offset(0, y), Offset(config.width, y),
            y % config.gridSize == 0 ? thickPaint : thinPaint,
          );
        }
        break;
      case GridPattern.blank:
      case GridPattern.music:
        break;
    }
  }

  @override
  bool shouldRepaint(ScratchpadPainter oldDelegate) {
    return oldDelegate.doc != doc ||
        oldDelegate.activeStroke != activeStroke ||
        oldDelegate.transform != transform ||
        oldDelegate.selectedId != selectedId;
  }
}

  final ScratchpadRepository _repository;
  final ScratchpadDoc _doc;

  // 撤销/重做栈
  final _undoStack = <ScratchpadAction>[];
  final _redoStack = <ScratchpadAction>[];
  static const _maxHistorySize = 50;

  // 选中态
  String? selectedElementId;

  // 自动保存
  Timer? _autoSaveTimer;

  ScratchpadController({
    required ScratchpadRepository repository,
    required ScratchpadDoc doc,
  })  : _repository = repository,
        _doc = doc {
    _startAutoSave();
  }

  // ===== 笔画操作 =====

  /// 添加笔画（绘制完成时调用）
  void addStroke(Stroke stroke) {
    _executeAction(AddStrokeAction(stroke));
  }

  /// 擦除笔画（橡皮擦命中时）
  void eraseStroke(String strokeId) {
    final idx = _doc.strokes.indexWhere((s) => s.id == strokeId);
    if (idx >= 0) {
      _executeAction(EraseStrokeAction(strokeId, idx));
    }
  }

  /// 撤销选中笔画
  void eraseSelected() {
    if (selectedElementId != null) {
      eraseStroke(selectedElementId!);
      selectedElementId = null;
      notifyListeners();
    }
  }

  // ===== 文本/公式块操作 =====

  void insertTextBlock(TextBlock block) {
    _executeAction(AddTextBlockAction(block));
  }

  void insertFormulaBlock(FormulaBlock block) {
    _executeAction(AddFormulaBlockAction(block));
  }

  void moveElement(String id, Offset newPosition) {
    _executeAction(MoveElementAction(id, newPosition));
  }

  // ===== 撤销/重做 =====

  bool get canUndo => _undoStack.isNotEmpty;
  bool get canRedo => _redoStack.isNotEmpty;

  void undo() {
    if (_undoStack.isEmpty) return;
    final action = _undoStack.removeLast();
    action.undo(_doc);
    _redoStack.add(action);
    notifyListeners();
  }

  void redo() {
    if (_redoStack.isEmpty) return;
    final action = _redoStack.removeLast();
    action.execute(_doc);
    _undoStack.add(action);
    notifyListeners();
  }

  void _executeAction(ScratchpadAction action) {
    action.execute(_doc);
    _undoStack.add(action);
    _redoStack.clear();
    if (_undoStack.length > _maxHistorySize) {
      _undoStack.removeAt(0);
    }
    notifyListeners();
  }

  // ===== 笔迹识别（可选端侧AI） =====

  Future<void> recognizeSelected() async {
    if (selectedElementId == null) return;

    final stroke = _doc.strokes.firstWhere(
      (s) => s.id == selectedElementId,
    );

    // 调用端侧手写识别
    final result = await EdgeAIManager.instance.recognizeHandwriting(
      strokes: [stroke],
    );

    if (result.confidence > 0.7) {
      // 弹出识别结果，用户确认后替换为文本
      // 实际替换由UI层触发 confirmReplaceWithText()
    }
  }

  // ===== 快照 =====

  /// 创建手动快照
  Future<void> createSnapshot(String label) async {
    final snapshot = ScratchpadSnapshot(
      id: const Uuid().v4(),
      docId: _doc.id,
      label: label,
      strokes: List.from(_doc.strokes),
      createdAt: DateTime.now(),
    );
    await _repository.saveSnapshot(snapshot);
  }

  // ===== 持久化 =====

  void _startAutoSave() {
    _autoSaveTimer = Timer.periodic(
      const Duration(seconds: 5),
      (_) => _save(),
    );
  }

  Future<void> _save() async {
    await _repository.saveDoc(_doc);
  }

  @override
  void dispose() {
    _autoSaveTimer?.cancel();
    _save(); // 退出时保存
    super.dispose();
  }
}

/// 操作抽象（Command 模式）
abstract class ScratchpadAction {
  void execute(ScratchpadDoc doc);
  void undo(ScratchpadDoc doc);
}

class AddStrokeAction implements ScratchpadAction {
  final Stroke stroke;
  AddStrokeAction(this.stroke);

  @override
  void execute(ScratchpadDoc doc) {
    doc.strokes.add(stroke);
  }

  @override
  void undo(ScratchpadDoc doc) {
    doc.strokes.removeWhere((s) => s.id == stroke.id);
  }
}

class EraseStrokeAction implements ScratchpadAction {
  final String strokeId;
  final int index;
  EraseStrokeAction(this.strokeId, this.index);

  @override
  void execute(ScratchpadDoc doc) {
    final stroke = doc.strokes[index];
    // 标记为已擦除，而非删除（保留重做能力）
    doc.strokes[index] = stroke.copyWith(erased: true);
  }

  @override
  void undo(ScratchpadDoc doc) {
    final stroke = doc.strokes[index];
    doc.strokes[index] = stroke.copyWith(erased: false);
  }
}

class AddTextBlockAction implements ScratchpadAction {
  final TextBlock block;
  AddTextBlockAction(this.block);

  @override
  void execute(ScratchpadDoc doc) {
    doc.textBlocks.add(block);
  }

  @override
  void undo(ScratchpadDoc doc) {
    doc.textBlocks.removeWhere((b) => b.id == block.id);
  }
}

class AddFormulaBlockAction implements ScratchpadAction {
  final FormulaBlock block;
  AddFormulaBlockAction(this.block);

  @override
  void execute(ScratchpadDoc doc) {
    doc.formulas.add(block);
  }

  @override
  void undo(ScratchpadDoc doc) {
    doc.formulas.removeWhere((b) => b.id == block.id);
  }
}

class MoveElementAction implements ScratchpadAction {
  final String elementId;
  final Offset newPosition;
  Offset? _oldPosition;

  MoveElementAction(this.elementId, this.newPosition);

  @override
  void execute(ScratchpadDoc doc) {
    final tbIdx = doc.textBlocks.indexWhere((b) => b.id == elementId);
    if (tbIdx >= 0) {
      _oldPosition = doc.textBlocks[tbIdx].position;
      doc.textBlocks[tbIdx] = doc.textBlocks[tbIdx].copyWith(position: newPosition);
      return;
    }
    final fbIdx = doc.formulas.indexWhere((b) => b.id == elementId);
    if (fbIdx >= 0) {
      _oldPosition = doc.formulas[fbIdx].position;
      doc.formulas[fbIdx] = doc.formulas[fbIdx].copyWith(position: newPosition);
    }
  }

  @override
  void undo(ScratchpadDoc doc) {
    if (_oldPosition == null) return;
    final tbIdx = doc.textBlocks.indexWhere((b) => b.id == elementId);
    if (tbIdx >= 0) {
      doc.textBlocks[tbIdx] = doc.textBlocks[tbIdx].copyWith(position: _oldPosition!);
      return;
    }
    final fbIdx = doc.formulas.indexWhere((b) => b.id == elementId);
    if (fbIdx >= 0) {
      doc.formulas[fbIdx] = doc.formulas[fbIdx].copyWith(position: _oldPosition!);
    }
  }
}

### 8.2 Hive 本地实现

```dart
class ScratchpadRepositoryHive implements ScratchpadRepository {
  static const _boxName = 'scratchpad_docs';
  static const _snapshotBoxName = 'scratchpad_snapshots';

  Future<Box<ScratchpadDoc>> get _box => Hive.openBox<ScratchpadDoc>(_boxName);
  Future<Box<ScratchpadSnapshot>> get _snapshotBox =>
      Hive.openBox<ScratchpadSnapshot>(_snapshotBoxName);

  @override
  Future<ScratchpadDoc?> findByContext({
    required ScratchpadScene scene,
    String? questionId,
    String? sessionId,
  }) async {
    final box = await _box;
    final results = box.values.where((doc) {
      if (doc.scene != scene) return false;
      if (questionId != null && doc.questionId != questionId) return false;
      if (sessionId != null && doc.sessionId != sessionId) return false;
      return true;
    });
    // 返回最近更新的
    return results.isEmpty
        ? null
        : results.reduce((a, b) =>
            a.updatedAt.isAfter(b.updatedAt) ? a : b);
  }

  @override
  Future<void> saveDoc(ScratchpadDoc doc) async {
    final box = await _box;
    await box.put(doc.id, doc);
  }

  @override
  Future<List<ScratchpadDoc>> listDocs({
    required String userId,
    ScratchpadScene? scene,
    int limit = 20,
    String? cursor,
  }) async {
    final box = await _box;
    var results = box.values.where((d) => d.userId == userId).toList();
    if (scene != null) {
      results = results.where((d) => d.scene == scene).toList();
    }
    results.sort((a, b) => b.updatedAt.compareTo(a.updatedAt));
    // cursor分页
    if (cursor != null) {
      final idx = results.indexWhere((d) => d.id == cursor);
      if (idx >= 0) results = results.sublist(idx + 1);
    }
    return results.take(limit).toList();
  }

  @override
  Future<void> deleteDoc(String docId) async {
    final box = await _box;
    await box.delete(docId);
  }

  @override
  Future<void> saveSnapshot(ScratchpadSnapshot snapshot) async {
    final box = await _snapshotBox;
    await box.put(snapshot.id, snapshot);
  }

  @override
  Future<List<ScratchpadSnapshot>> getSnapshots(String docId) async {
    final box = await _snapshotBox;
    return box.values.where((s) => s.docId == docId).toList()
      ..sort((a, b) => b.createdAt.compareTo(a.createdAt));
  }

  @override
  Future<void> deleteSnapshot(String snapshotId) async {
    final box = await _snapshotBox;
    await box.delete(snapshotId);
  }
}

class ScratchpadExporter {
  /// 导出为 PNG 图片
  Future<Uint8List> exportToPng(ScratchpadDoc doc, {\n    double scale = 2.0,  // 高清倍率
    bool includeGrid = false,
  }) async {
    final recorder = ui.PictureRecorder();
    final canvas = Canvas(recorder);
    final size = Size(doc.canvasConfig.width, doc.canvasConfig.height);

    // 使用 ScratchpadPainter 渲染到离屏 Canvas
    final painter = ScratchpadPainter(
      doc: includeGrid ? doc : doc.copyWith(
        canvasConfig: doc.canvasConfig.copyWith(showGrid: false),
      ),
      transform: CanvasTransform.identity(),
    );
    painter.paint(canvas, size);

    final picture = recorder.endRecording();
    final image = await picture.toImage(
      (size.width * scale).toInt(),
      (size.height * scale).toInt(),
    );
    final byteData = await image.toByteData(format: ui.ImageByteFormat.png);
    return byteData!.buffer.asUint8List();
  }

  /// 导出为 PDF（含题目标题）
  Future<String> exportToPdf(ScratchpadDoc doc, {String? title}) async {
    final pngBytes = await exportToPng(doc);
    final pdf = pw.Document();
    pdf.addPage(
      pw.Page(
        pageFormat: PdfPageFormat.a4,
        build: (context) => pw.Column(
          crossAxisAlignment: pw.CrossAxisAlignment.start,
          children: [
            if (title != null)
              pw.Text(title, style: pw.TextStyle(fontSize: 14)),
            pw.SizedBox(height: 8),
            pw.Image(pw.MemoryImage(pngBytes)),
          ],
        ),
      ),
    );
    final filePath = await _savePdf(pdf, doc.id);
    return filePath;
  }
}

/// 笔画数据采样压缩（存储时）
Stroke compressStroke(Stroke stroke) {
  if (stroke.points.length < 10) return stroke;
  final compressed = <StrokePoint>[];
  final step = (stroke.points.length / 200).ceil();
  for (int i = 0; i < stroke.points.length; i += step) {
    compressed.add(stroke.points[i]);
  }
  // 确保包含最后一个点
  if (compressed.last != stroke.points.last) {
    compressed.add(stroke.points.last);
  }
  return stroke.copyWith(points: compressed);
}

    if (doc.archivedAt == null &&
        doc.scene != ScratchpadScene.freeform &&
        DateTime.now().difference(doc.updatedAt).inHours > 2) {
      // 超过2小时未更新的会话草稿自动归档
      await repo.saveDoc(doc.copyWith(archivedAt: DateTime.now()));
    }
  }
}
```
