# 汉字笔画顺序检测与评估系统 - 详细设计

## 1. 模块概述

### 1.1 功能定位

汉字笔画顺序检测与评估系统是 PrimeTop 拼音识字启蒙模块的核心子功能，负责对学生（幼儿及小学低年级）的手写汉字笔画顺序进行实时检测、评估和反馈。该系统通过捕捉用户在触摸屏上的书写轨迹，与标准笔画顺序进行智能比对，提供：
- 笔画顺序正确性判断
- 笔画质量评估（起笔、行笔、收笔）
- 实时纠错指导
- 笔画动画演示回放
- 书写进度统计与能力追踪

### 1.2 核心价值

| 价值维度 | 说明 |
|---------|------|
| 教育价值 | 帮助学生养成正确的笔画书写习惯，为后续语文学习打好基础 |
| 交互价值 | 提供即时反馈，增强学习趣味性和成就感 |
| 技术价值 | 展示手写识别、路径比对算法等AI能力在教育场景的应用 |
| 产品价值 | 区别于传统识字工具，提供更完整的汉字学习闭环 |

### 1.3 与拼音识字启蒙模块的关系

```
拼音识字启蒙模块
    ├── 拼音学习子模块
    ├── 汉字识记子模块
    │   ├── 汉字卡片展示
    │   ├── 笔画动画演示
    │   └── 笔画顺序检测（本文档）
    ├── 练习评估子模块
    └── 学习调度子模块
```

**边界界定：**
- 本模块负责：笔画轨迹捕捉、顺序比对、实时反馈
- 不属于本模块：汉字内容元数据（由拼音识字启蒙模块提供）、语音识别、拼音学习

### 4.1 轨迹数据格式

```typescript
/** 单个笔画数据点 */
interface StrokePoint {
  x: number;        // X坐标（相对于画布左上角）
  y: number;        // Y坐标
  t: number;        // 时间戳（毫秒）
  p?: number;       // 压力值（0-1，部分设备支持）
}

/** 单个笔画 */
interface Stroke {
  index: number;    // 笔画序号（从1开始）
  points: StrokePoint[];  // 轨迹点数组
  startTime: number;      // 起笔时间
  endTime: number;        // 收笔时间
  duration: number;       // 笔画时长（毫秒）
}

/** 完整书写数据 */
interface HandwritingData {
  userId: number;
  hanziId: string;
  hanziChar: string;
  sessionId: string;
  canvasSize: { width: number; height: number };
  strokes: Stroke[];
  totalDuration: number;
}
```

### 4.2 Flutter 轨迹采集实现

```dart
/// 书写画布组件
class StrokeCanvasWidget extends StatefulWidget {
  final String hanziChar;
  final Function(Stroke) onStrokeDetected;
  final Function(HandwritingData) onWritingComplete;
  
  const StrokeCanvasWidget({
    Key? key,
    required this.hanziChar,
    required this.onStrokeDetected,
    required this.onWritingComplete,
  }) : super(key: key);

  @override
  _StrokeCanvasWidgetState createState() => _StrokeCanvasWidgetState();
}

class _StrokeCanvasWidgetState extends State<StrokeCanvasWidget> {
  final List<Offset> _currentPoints = [];
  final List<Stroke> _strokes = [];
  final GlobalKey _canvasKey = GlobalKey();
  
  int _currentStrokeIndex = 0;
  DateTime? _strokeStartTime;
  bool _isDrawing = false;

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onPanStart: (details) {
        setState(() {
          _isDrawing = true;
          _currentPoints.clear();
          _strokeStartTime = DateTime.now();
          _currentPoints.add(details.localPosition);
        });
      },
      onPanUpdate: (details) {
        if (_isDrawing) {
          setState(() {
            _currentPoints.add(details.localPosition);
          });
        }
      },
      onPanEnd: (details) async {
        if (_isDrawing && _currentPoints.isNotEmpty) {
          _isDrawing = false;
          
          final stroke = _createStrokeFromPoints();
          _strokes.add(stroke);
          _currentStrokeIndex++;
          
          // 触发笔画检测
          widget.onStrokeDetected(stroke);
          
          // 检查是否完成整个汉字
          final standardStrokeCount = await _getStandardStrokeCount();
          if (_strokes.length >= standardStrokeCount) {
            final handwritingData = _createHandwritingData();
            widget.onWritingComplete(handwritingData);
          }
        }
      },
      child: CustomPaint(
        painter: StrokePainter(_strokes, _currentPoints),
        size: Size.infinite,
      ),
    );
  }

  Stroke _createStrokeFromPoints() {
    final points = _currentPoints.map((p) => StrokePoint(
      x: p.dx,
      y: p.dy,
      t: DateTime.now().millisecondsSinceEpoch,
    )).toList();
    
    final startTime = _strokeStartTime?.millisecondsSinceEpoch ?? DateTime.now().millisecondsSinceEpoch;
    final endTime = DateTime.now().millisecondsSinceEpoch;
    
    return Stroke(
      index: _currentStrokeIndex + 1,
      points: points,
      startTime: startTime,
      endTime: endTime,
      duration: endTime - startTime,
    );
  }

  HandwritingData _createHandwritingData() {
    return HandwritingData(
      userId: CurrentUser.id,
      hanziId: widget.hanziChar, // 实际应从内容服务获取hanziId
      hanziChar: widget.hanziChar,
      sessionId: Uuid().v4(),
      canvasSize: {
        'width': _canvasKey.currentContext?.size?.width ?? 0,
        'height': _canvasKey.currentContext?.size?.height ?? 0,
      },
      strokes: _strokes,
      totalDuration: _strokes.fold(0, (sum, stroke) => sum + stroke.duration),
    );
  }

  Future<int> _getStandardStrokeCount() async {
    // 从内容服务获取标准笔画数
    final hanziInfo = await ContentApi.getHanziInfo(widget.hanziChar);
    return hanziInfo.strokes;
  }
}

/// 笔画绘制器
class StrokePainter extends CustomPainter {
  final List<Stroke> strokes;
  final List<Offset> currentPoints;

  StrokePainter(this.strokes, this.currentPoints);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.black
      ..strokeWidth = 4.0
      ..strokeCap = StrokeCap.round
      ..style = PaintingStyle.stroke;

    // 绘制已完成笔画
    for (final stroke in strokes) {
      final path = Path();
      if (stroke.points.isNotEmpty) {
        path.moveTo(stroke.points.first.x, stroke.points.first.y);
        for (final point in stroke.points.skip(1)) {
          path.lineTo(point.x, point.y);
        }
      }
      canvas.drawPath(path, paint);
    }

    // 绘制当前笔画
    if (currentPoints.isNotEmpty) {
      final currentPath = Path();
      currentPath.moveTo(currentPoints.first.dx, currentPoints.first.dy);
      for (final point in currentPoints.skip(1)) {
        currentPath.lineTo(point.dx, point.dy);
      }
      canvas.drawPath(currentPath, paint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}
```

## 5. 服务端检测算法

### 5.1 笔画顺序检测算法

```java
/**
 * 笔画顺序检测器
 * 通过比对用户书写顺序与标准笔画顺序，判断顺序正确性
 */
@Component
public class StrokeOrderDetector {
    
    /**
     * 检测笔画顺序是否正确
     */
    public OrderDetectionResult detectOrder(
            HandwritingData handwriting, 
            List<StandardStroke> standardStrokes) {
        
        List<Stroke> userStrokes = handwriting.getStrokes();
        int totalStrokes = standardStrokes.size();
        
        if (userStrokes.size() != totalStrokes) {
            return OrderDetectionResult.builder()
                .isCorrect(false)
                .errorType(userStrokes.size() < totalStrokes ? 
                    StrokeErrorType.MISSING_STROKE : StrokeErrorType.EXTRA_STROKE)
                .strokeIndex(userStrokes.size() + 1)
                .confidence(1.0)
                .build();
        }
        
        // 逐笔比对顺序
        int correctCount = 0;
        List<StrokeError> errors = new ArrayList<>();
        
        for (int i = 0; i < userStrokes.size(); i++) {
            Stroke userStroke = userStrokes.get(i);
            StandardStroke standardStroke = standardStrokes.get(i);
            
            StrokeMatchResult match = matchStroke(userStroke, standardStroke);
            
            if (match.isMatch()) {
                correctCount++;
            } else {
                errors.add(StrokeError.builder()
                    .strokeIndex(i + 1)
                    .errorType(match.getErrorType())
                    .confidence(match.getConfidence())
                    .userStroke(userStroke)
                    .correctStroke(standardStroke)
                    .build());
            }
        }
        
        double orderScore = (double) correctCount / totalStrokes * 100;
        
        return OrderDetectionResult.builder()
            .isCorrect(correctCount == totalStrokes)
            .correctCount(correctCount)
            .totalCount(totalStrokes)
            .score(orderScore)
            .errors(errors)
            .confidence(orderScore / 100)
            .build();
    }
    
    /**
     * 匹配单个笔画
     */
    private StrokeMatchResult matchStroke(Stroke userStroke, StandardStroke standardStroke) {
        // 1. 方向匹配
        DirectionMatch directionMatch = matchDirection(userStroke, standardStroke);
        
        // 2. 轨迹形状匹配
        ShapeMatch shapeMatch = matchShape(userStroke, standardStroke);
        
        // 3. 位置匹配
        PositionMatch positionMatch = matchPosition(userStroke, standardStroke);
        
        // 综合判断
        double directionWeight = 0.4;
        double shapeWeight = 0.4;
        double positionWeight = 0.2;
        
        double overallScore = directionMatch.getScore() * directionWeight +
                             shapeMatch.getScore() * shapeWeight +
                             positionMatch.getScore() * positionWeight;
        
        if (overallScore >= 0.7) {
            return StrokeMatchResult.match(overallScore);
        }
        
        // 判断错误类型
        StrokeErrorType errorType;
        if (directionMatch.getScore() < 0.3) {
            errorType = StrokeErrorType.WRONG_DIRECTION;
        } else if (shapeMatch.getScore() < 0.3) {
            errorType = StrokeErrorType.DEVIATED_PATH;
        } else {
            errorType = StrokeErrorType.WRONG_ORDER;
        }
        
        return StrokeMatchResult.error(errorType, overallScore);
    }
    
    /**
     * 匹配笔画方向
     */
    private DirectionMatch matchDirection(Stroke userStroke, StandardStroke standardStroke) {
        Offset userStart = userStroke.getPoints().get(0);
        Offset userEnd = userStroke.getPoints().get(userStroke.getPoints().size() - 1);
        
        Offset standardStart = standardStroke.getStartPoint();
        Offset standardEnd = standardStroke.getEndPoint();
        
        // 计算方向向量
        Offset userVector = Offset(
            userEnd.getX() - userStart.getX(),
            userEnd.getY() - userStart.getY()
        );
        Offset standardVector = Offset(
            standardEnd.getX() - standardStart.getX(),
            standardEnd.getY() - standardStart.getY()
        );
        
        // 归一化
        double userLength = Math.sqrt(userVector.getX() * userVector.getX() + 
                                       userVector.getY() * userVector.getY());
        double standardLength = Math.sqrt(standardVector.getX() * standardVector.getX() + 
                                          standardVector.getY() * standardVector.getY());
        
        if (userLength < 1 || standardLength < 1) {
            return DirectionMatch.builder().score(0.5).build();
        }
        
        double cosAngle = (userVector.getX() * standardVector.getX() + 
                          userVector.getY() * standardVector.getY()) / 
                         (userLength * standardLength);
        
        // 方向得分：cos值范围 -1 到 1，映射到 0 到 1
        double directionScore = (cosAngle + 1) / 2;
        
        return DirectionMatch.builder()
            .score(directionScore)
            .angle(Math.acos(cosAngle) * 180 / Math.PI)
            .build();
    }
    
    /**
     * 匹配笔画形状（使用动态时间规整 DTW 算法）
     */
    private ShapeMatch matchShape(Stroke userStroke, StandardStroke standardStroke) {
        List<Offset> userPoints = userStroke.getPoints();
        List<Offset> standardPoints = standardStroke.getPoints();
        
        // 重采样到相同数量的点
        List<Offset> resampledUser = resample(userPoints, 50);
        List<Offset> resampledStandard = resample(standardPoints, 50);
        
        // 计算 DTW 距离
        double dtwDistance = calculateDTW(resampledUser, resampledStandard);
        
        // 归一化距离，转换为相似度
        double maxLength = Math.max(userPoints.size(), standardPoints.size());
        double normalizedDistance = dtwDistance / maxLength;
        double similarityScore = Math.max(0, 1 - normalizedDistance / 100);
        
        return ShapeMatch.builder()
            .score(similarityScore)
            .distance(dtwDistance)
            .build();
    }
    
    /**
     * 动态时间规整算法
     */
    private double calculateDTW(List<Offset> series1, List<Offset> series2) {
        int n = series1.size();
        int m = series2.size();
        
        double[][] dtw = new double[n + 1][m + 1];
        
        // 初始化
        for (int i = 0; i <= n; i++) {
            dtw[i][0] = Double.POSITIVE_INFINITY;
        }
        for (int j = 0; j <= m; j++) {
            dtw[0][j] = Double.POSITIVE_INFINITY;
        }
        dtw[0][0] = 0;
        
        // 计算 DTW 矩阵
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                double cost = euclideanDistance(series1.get(i - 1), series2.get(j - 1));
                dtw[i][j] = cost + Math.min(
                    Math.min(dtw[i - 1][j], dtw[i][j - 1]),
                    dtw[i - 1][j - 1]
                );
            }
        }
        
        return dtw[n][m];
    }
    
    private double euclideanDistance(Offset p1, Offset p2) {
        return Math.sqrt(
            Math.pow(p1.getX() - p2.getX(), 2) + 
            Math.pow(p1.getY() - p2.getY(), 2)
        );
    }
    
    /**
     * 重采样轨迹点
     */
    private List<Offset> resample(List<Offset> points, int targetSize) {
        if (points.size() <= targetSize) {
            return new ArrayList<>(points);
        }
        
        List<Offset> resampled = new ArrayList<>();
        double step = (double) (points.size() - 1) / (targetSize - 1);
        
        for (int i = 0; i < targetSize; i++) {
            int index = (int) Math.round(i * step);
            resampled.add(points.get(index));
        }
        
        return resampled;
    }
    
    /**
     * 匹配笔画位置
     */
    private PositionMatch matchPosition(Stroke userStroke, StandardStroke standardStroke) {
        Offset userCenter = calculateCenter(userStroke.getPoints());
        Offset standardCenter = calculateCenter(standardStroke.getPoints());
        
        double distance = euclideanDistance(userCenter, standardCenter);
        
        // 假设画布大小为 300x300，归一化距离
        double maxDistance = Math.sqrt(300 * 300 + 300 * 300);
        double normalizedDistance = distance / maxDistance;
        double positionScore = Math.max(0, 1 - normalizedDistance * 2);
        
        return PositionMatch.builder()
            .score(positionScore)
            .distance(distance)
            .build();
    }
    
    private Offset calculateCenter(List<Offset> points) {
        double sumX = points.stream().mapToDouble(Offset::getX).sum();
        double sumY = points.stream().mapToDouble(Offset::getY).sum();
        return Offset.of(sumX / points.size(), sumY / points.size());
    }
}
```

### 5.2 笔画质量评估算法

```java
/**
 * 笔画质量评估器
 * 评估笔画的书写的平滑度、笔锋、粗细等质量指标
 */
@Component
public class StrokeQualityEvaluator {
    
    /**
     * 评估单个笔画的质量
     */
    public QualityAssessment evaluateQuality(Stroke stroke) {
        List<Offset> points = stroke.getPoints();
        
        if (points.size() < 2) {
            return QualityAssessment.builder()
                .overallScore(0)
                .smoothnessScore(0)
                .stabilityScore(0)
                .level(QualityLevel.NEED_IMPROVEMENT)
                .build();
        }
        
        // 1. 平滑度评估
        double smoothness = evaluateSmoothness(points);
        
        // 2. 稳定性评估（抖动程度）
        double stability = evaluateStability(points);
        
        // 3. 笔锋评估
        double strokeTip = evaluateStrokeTip(points);
        
        // 综合评分
        double overallScore = smoothness * 0.4 + stability * 0.4 + strokeTip * 0.2;
        
        QualityLevel level = determineQualityLevel(overallScore);
        
        return QualityAssessment.builder()
            .overallScore((int) (overallScore * 100))
            .smoothnessScore((int) (smoothness * 100))
            .stabilityScore((int) (stability * 100))
            .strokeTipScore((int) (strokeTip * 100))
            .level(level)
            .build();
    }
    
    /**
     * 评估平滑度（基于曲率变化）
     */
    private double evaluateSmoothness(List<Offset> points) {
        if (points.size() < 3) return 1.0;
        
        double totalCurvature = 0;
        int count = 0;
        
        for (int i = 1; i < points.size() - 1; i++) {
            double angle1 = calculateAngle(points.get(i - 1), points.get(i));
            double angle2 = calculateAngle(points.get(i), points.get(i + 1));
            
            double angleDiff = Math.abs(angle2 - angle1);
            totalCurvature += angleDiff;
            count++;
        }
        
        if (count == 0) return 1.0;
        
        double avgCurvature = totalCurvature / count;
        // 曲率越小越平滑
        double smoothness = Math.max(0, 1 - avgCurvature / Math.PI);
        
        return smoothness;
    }
    
    /**
     * 评估稳定性（检测抖动）
     */
    private double evaluateStability(List<Offset> points) {
        if (points.size() < 3) return 1.0;
        
        // 计算相邻点之间的距离变化
        List<Double> distances = new ArrayList<>();
        for (int i = 1; i < points.size(); i++) {
            distances.add(euclideanDistance(points.get(i - 1), points.get(i)));
        }
        
        // 计算距离的变异系数
        double mean = distances.stream().mapToDouble(Double::doubleValue).average().orElse(0);
        double variance = distances.stream()
            .mapToDouble(d -> Math.pow(d - mean, 2))
            .average().orElse(0);
        double stdDev = Math.sqrt(variance);
        
        double cv = mean > 0 ? stdDev / mean : 0;
        
        // 变异系数越小越稳定
        double stability = Math.max(0, 1 - cv * 2);
        
        return stability;
    }
    
    /**
     * 评估笔锋（起笔和收笔的质量）
     */
    private double evaluateStrokeTip(List<Offset> points) {
        if (points.size() < 5) return 0.5;
        
        // 评估起笔（前5个点）
        double startTip = evaluateTip(points.subList(0, Math.min(5, points.size())));
        
        // 评估收笔（后5个点）
        double endTip = evaluateTip(points.subList(Math.max(0, points.size() - 5), points.size()));
        
        return (startTip + endTip) / 2;
    }
    
    private double evaluateTip(List<Offset> tipPoints) {
        if (tipPoints.size() < 3) return 0.5;
        
        // 计算前几个点的方向变化
        double angle1 = calculateAngle(tipPoints.get(0), tipPoints.get(1));
        double angle2 = calculateAngle(tipPoints.get(1), tipPoints.get(2));
        
        double angleDiff = Math.abs(angle2 - angle1);
        
        // 方向变化越小，笔锋越好
        double tipQuality = Math.max(0, 1 - angleDiff / Math.PI);
        
        return tipQuality;
    }
    
    private double calculateAngle(Offset p1, Offset p2) {
        return Math.atan2(p2.getY() - p1.getY(), p2.getX() - p1.getX());
    }
    
    private QualityLevel determineQualityLevel(double score) {
        if (score >= 0.9) return QualityLevel.EXCELLENT;
        if (score >= 0.75) return QualityLevel.GOOD;
        if (score >= 0.6) return QualityLevel.FAIR;
        return QualityLevel.NEED_IMPROVEMENT;
    }
}
```
