```dart
  notStarted,    // 未开始：从未学习过
  inProgress,    // 学习中：已查看内容，未完成首次默写
  firstDictation, // 待首次默写：已完成学习，需默写检测
```

**掌握度进度条计算逻辑：**

```dart
double calculateMasteryProgress(PoemUserProgress progress) {
  if (progress.status == PoemMasteryStatus.notStarted) return 0.0;
  
  double score = 0.0;
  
  // 基础分：已学习内容
  if (progress.hasStudiedContent) score += 0.2;
  
  // 首次默写得分贡献
  if (progress.bestDictationScore != null) {
    score += (progress.bestDictationScore! / 100) * 0.4;
  }
  
  // 复习次数贡献（最多0.2）
  score += min(progress.reviewCount / 5, 1.0) * 0.2;
  
  // 最近表现贡献
  if (progress.lastThreeScoresAverage >= 95) {
    score += 0.2;
  } else if (progress.lastThreeScoresAverage >= 80) {
    score += 0.1;
  }
  
  return min(score, 1.0);
}
```

#### 3.4.1 默写评判引擎

```java
@Service
public class DictationEvaluationService {
    
    /**
     * 逐字评判默写内容
     */
    public DictationResult evaluate(DictationSubmitRequest request, Poem poem) {
        List<ContentLine> expectedLines = poem.getContentLines().getLines();
        List<LineAnswer> answers = request.getAnswers();
        
        int totalChars = 0;
        int correctChars = 0;
        List<CharError> errors = new ArrayList<>();
        
        for (int i = 0; i < expectedLines.size(); i++) {
            ContentLine expected = expectedLines.get(i);
            String expectedText = expected.getText().replaceAll("[，。！？、；：""''（）]", "");
            String actualText = "";
            if (i < answers.size()) {
                actualText = answers.get(i).getText().replaceAll("[，。！？、；：""''（）]", "");
            }
            
            totalChars += expectedText.length();
            
            // 逐字对齐比对
            List<CharError> lineErrors = compareChars(
                expectedText, actualText, expected.getLineNumber()
            );
            
            correctChars += expectedText.length() - lineErrors.size();
            errors.addAll(lineErrors);
        }
        
        // 计算得分
        double score = totalChars > 0 ? (correctChars * 100.0 / totalChars) : 0;
        score = Math.round(score * 10.0) / 10.0; // 保留1位小数
        
        // 错误分类统计
        Map<ErrorType, Long> errorTypeCount = errors.stream()
            .collect(Collectors.groupingBy(CharError::getErrorType, Collectors.counting()));
        
        // 生成学习建议
        String suggestion = generateSuggestion(score, errors, errorTypeCount);
        
        // 识别薄弱行
        Set<Integer> weakLines = identifyWeakLines(errors, expectedLines);
        
        return DictationResult.builder()
            .score(score)
            .totalChars(totalChars)
            .correctChars(correctChars)
            .errors(errors)
            .errorTypeDistribution(errorTypeCount)
            .weakLines(weakLines)
            .suggestion(suggestion)
            .build();
    }
    
    /**
     * 逐字比对算法
     * 处理：同音字、形近字、缺字、多字、顺序颠倒
     */
    private List<CharError> compareChars(String expected, String actual, int lineNumber) {
        List<CharError> errors = new ArrayList<>();
        int maxLen = Math.max(expected.length(), actual.length());
        
        for (int i = 0; i < expected.length(); i++) {
            char expectedChar = expected.charAt(i);
            
            if (i >= actual.length()) {
                // 缺字
                errors.add(CharError.builder()
                    .lineNumber(lineNumber)
                    .position(i)
                    .expectedChar(expectedChar)
                    .actualChar(null)
                    .errorType(ErrorType.MISSING)
                    .hint(generateHint(expectedChar, ErrorType.MISSING))
                    .build());
                continue;
            }
            
            char actualChar = actual.charAt(i);
            
            if (expectedChar == actualChar) {
                continue; // 正确
            }
            
            // 判断错误类型
            ErrorType errorType = classifyError(expectedChar, actualChar);
            errors.add(CharError.builder()
                .lineNumber(lineNumber)
                .position(i)
                .expectedChar(expectedChar)
                .actualChar(actualChar)
                .errorType(errorType)
                .hint(generateHint(expectedChar, errorType))
                .build());
        }
        
        // 检查多出的字
        for (int i = expected.length(); i < actual.length(); i++) {
            errors.add(CharError.builder()
                .lineNumber(lineNumber)
                .position(i)
                .expectedChar(null)
                .actualChar(actual.charAt(i))
                .errorType(ErrorType.EXTRA)
                .hint(null)
                .build());
        }
        
        return errors;
    }
    
    /**
     * 错误分类
     */
    private ErrorType classifyError(char expected, char actual) {
        // 1. 同音字检测
        if (isHomophone(expected, actual)) {
            return ErrorType.HOMOPHONE;
        }
        // 2. 形近字检测
        if (isSimilarShape(expected, actual)) {
            return ErrorType.SIMILAR_SHAPE;
        }
        // 3. 默认：错字
        return ErrorType.WRONG_CHAR;
    }
}
```

```java
@Component
public class DictationCompletedHandler {
    
    @EventListener
    @Async("learningEventExecutor")
    public void onDictationCompleted(DictationCompletedEvent event) {
        Long userId = event.getUserId();
        Long poemId = event.getPoemId();
        DictationResult result = event.getResult();
        
        // 1. 更新篇目掌握度
        updatePoemMastery(userId, poemId, result);
        
        // 2. 将错误字加入错题本
        if (!result.getErrors().isEmpty()) {
            syncToErrorBook(userId, poemId, result.getErrors());
        }
        
        // 3. 触发间隔复习调度
        scheduleReview(userId, poemId, result.getScore());
        
        // 4. 更新学情分析数据
        updateLearningAnalytics(userId, poemId, result);
        
        // 5. 检查成就触发
        checkAchievement(userId, poemId, result);
        
        // 6. 更新知识点掌握度（意象、表达技巧等）
        updateKnowledgePointMastery(userId, poemId, result.getScore());
    }
    
    private void updatePoemMastery(Long userId, Long poemId, DictationResult result) {
        PoemUserProgress progress = progressRepository
            .findByUserIdAndPoemId(userId, poemId)
            .orElse(PoemUserProgress.createInitial(userId, poemId));
        
        // 更新统计数据
        progress.setDictationCount(progress.getDictationCount() + 1);
        progress.setBestDictationScore(
            Math.max(progress.getBestDictationScore(), result.getScore())
        );
        progress.setLastDictationScore(result.getScore());
        progress.setLastStudyTime(Instant.now());
        
        // 更新掌握状态
        PoemMasteryStatus newStatus = determineMasteryStatus(progress, result);
        progress.setStatus(newStatus);
        
        // 记录薄弱行
        if (!result.getWeakLines().isEmpty()) {
            progress.setWeakLines(result.getWeakLines());
        }
        
        // 判断是否达到「已掌握」标准
        if (progress.getConsecutiveHighScores() >= 3 
            && progress.getReviewIntervalDays() >= 30) {
            progress.setStatus(PoemMasteryStatus.MASTERED);
            progress.setMasteredAt(Instant.now());
        }
        
        progressRepository.save(progress);
    }
    
    /**
     * 掌握度状态判断
     */
    private PoemMasteryStatus determineMasteryStatus(
            PoemUserProgress progress, DictationResult result) {
        
        if (result.getScore() >= 95) {
            int consecutive = progress.getConsecutiveHighScores() + 1;
            progress.setConsecutiveHighScores(consecutive);
            if (consecutive >= 3) {
                return PoemMasteryStatus.MASTERED;
            }
            return PoemMasteryStatus.REVIEWING;
        }
        
        if (result.getScore() >= 70) {
            progress.setConsecutiveHighScores(0);
            return PoemMasteryStatus.REVIEWING;
        }
        
        progress.setConsecutiveHighScores(0);
        return PoememMasteryStatus.WEAK;
    }

    /**
     * 错误字同步至错题本（易错字训练队列）
     * 与《古诗文专项学习系统》§3.13 weak_word_drill 表对齐：
     * upsert 冲突键 uk_user_char(user_id, poem_id, expected_char, line_number, char_position)
     * 同一字多处错误合并为一条，保留首次出错的行号/字位
     */
    private void syncToErrorBook(Long userId, Long poemId, List<CharError> errors) {
        for (CharError error : errors) {
            if (error.getExpectedChar() == null) {
                continue; // 多字（EXTRA）无预期字，不进入训练队列
            }
            WeakWordDrill drill = drillRepository
                .findByUserAndChar(userId, poemId, error.getExpectedChar(),
                                   error.getLineNumber(), error.getPosition())
                .orElseGet(() -> WeakWordDrill.create(userId, poemId,
                        error.getExpectedChar(), error.getLineNumber(), error.getPosition()));

            if (drill.getStatus() == DrillStatus.CONQUERED) {
                // 已攻克的字再次出错 → 重激活（consecutive_correct 清零，wrong_count 续加）
                drill.setStatus(DrillStatus.REACTIVATED);
                drill.setConsecutiveCorrect(0);
            }
            drill.setWrongCount(drill.getWrongCount() + 1);
            drill.setLastErrorType(error.getErrorType().name());
            drill.setLastWrongAt(Instant.now());
            // 训练间隔：错后 +1 天起（连对1次+3天、连对2次+7天，见 §7.3）
            drill.setNextDrillAt(Instant.now().plus(Duration.ofDays(1)));
            drillRepository.save(drill);
        }
    }

    /**
     * SM-2 间隔复习调度
     * quality 映射（上层文档 §6.3）：≥95→easy(4)；≥80→good(3)；≥60→hard(2)；<60→again(0)
     * 注意：SM-2 参数更新与篇目状态流转必须在同一事务内执行（见 §7.1 守卫规则 G3）
     */
    private void scheduleReview(Long userId, Long poemId, double score) {
        int quality;
        if (score >= 95)      quality = 4;  // easy
        else if (score >= 80) quality = 3;  // good
        else if (score >= 60) quality = 2;  // hard
        else                  quality = 0;  // again

        PoemUserProgress progress = progressRepository
            .findByUserIdAndPoemId(userId, poemId)
            .orElseThrow(() -> new ProgressNotFoundException(userId, poemId));

        // 同一事务内：SM-2 更新 + 状态流转 + 掌握度重算
        ReviewOutcome outcome = sm2Service.applySm2(progress, quality);
        progress.setReviewIntervalDays(outcome.getIntervalDays());
        progress.setNextReviewAt(outcome.getNextReviewAt()); // 已含 0-12h 随机抖动

        double mastery = masteryCalculator.recalculate(progress);
        progress.setMastery(mastery);

        progressRepository.save(progress);

        // 发布进度变更事件（Outbox 同事务写入）
        eventPublisher.publish(PoemEventBuilder.progressChanged(
            userId, poemId, progress.getStatus(), mastery,
            outcome.getNextReviewAt().toEpochMilli()));
    }

    /**
     * 更新学情分析数据：透传默写完成事件至行为流
     * 总线事件名 poem.dictation.completed（本处理器领域事件
     * DictationCompletedEvent 的出站形态），eventId = "dict:{recordId}" 保证幂等
     */
    private void updateLearningAnalytics(Long userId, Long poemId, DictationResult result) {
        eventPublisher.publish(PoemEventBuilder.dictationCompleted(
            userId, poemId, result.getRecordId(), result.getType(),
            result.getScore(), result.getErrors().size(),
            result.getWeakLines().size(), result.getDurationSec()));
    }

    /**
     * 成就触发检查（消费方：成就系统 achievementUnlocked）
     */
    private void checkAchievement(Long userId, Long poemId, DictationResult result) {
        List<String> unlocked = new ArrayList<>();
        // 一气呵成：全篇默写 100 分
        if (result.getType() == DictationType.FULL && result.getScore() >= 100) {
            unlocked.add("POEM_PERFECT_DICTATION");
        }
        // 温故知新：已掌握篇目数达 10/30/60（阶梯成就）
        int masteredCount = progressRepository.countByUserIdAndStatus(
            userId, PoemMasteryStatus.MASTERED);
        for (int tier : new int[]{10, 30, 60}) {
            if (masteredCount == tier) unlocked.add("POEM_MASTERED_" + tier);
        }
        // 错字克星：累计攻克易错字 20/50
        int conquered = drillRepository.countByUserIdAndStatus(userId, DrillStatus.CONQUERED);
        for (int tier : new int[]{20, 50}) {
            if (conquered == tier) unlocked.add("POEM_CHAR_CONQUERER_" + tier);
        }
        if (!unlocked.isEmpty()) {
            eventPublisher.publish(PoemEventBuilder.achievementsUnlocked(userId, unlocked));
        }
    }

    /**
     * 更新知识点掌握度（意象理解、表达技巧等）
     * 通过 poem_knowledge_ref 取篇目挂载的知识点，按得分增量修正
     */
    private void updateKnowledgePointMastery(Long userId, Long poemId, double score) {
        List<Long> kpIds = knowledgeRefRepository.findKnowledgePointIdsByPoemId(poemId);
        double delta = score >= 80 ? 0.05 : (score < 60 ? -0.05 : 0);
        if (delta == 0) return;
        for (Long kpId : kpIds) {
            knowledgeMasteryRepository.adjustMastery(userId, kpId, delta, /*clamp*/0.0, 1.0);
        }
    }
}
```

#### 与上层文档的职责边界说明

本节 `DictationCompletedHandler` 代码是**编排视角**的实现参考。篇目学习状态的**唯一真源**是 `user_poem_progress.status`，其状态机定义以《古诗文专项学习系统》§6.1 为准（`not_started → learning → reciting → mastered → decaying`）；本节代码中的 `determineMasteryStatus` 仅计算展示层加速字段（`consecutive_high_scores` 等），最终状态写入必须委托 `scheduleReview` 内的 SM-2 同事务更新，两者冲突时以状态机流转表（本文档 §7.1）为准。

#### 客户端展示状态与服务端状态的映射

```text
客户端 PoemMasteryStatus（3.3.1）          服务端 user_poem_progress.status
─────────────────────────────────────────────────────────────────────
notStarted（未开始）                 ←→   not_started
inProgress（学习中）                 ←→   learning 且 explainCoverage < 0.8
firstDictation（待首次默写）         ←→   learning 且 explainCoverage ≥ 0.8
reviewing（复习中）                  ←→   reciting 且 last_dictation_score ≥ 70
weak（薄弱）                         ←→   decaying（或 reciting 且 last_dictation_score < 70）
mastered（已掌握）                   ←→   mastered
```

映射为**读时派生**（BFF 聚合层计算），客户端不落库展示状态；服务端不做反向存储，避免双写不一致。

### 3.5 弱网与离线默写补交

默写作答支持离线（《古诗文专项学习系统》§9 性能要求）。本地暂存答案，网络恢复后按行号幂等补交。

**本地草稿结构：**

```dart
class OfflineDictationDraft {
  final String clientRequestId;   // UUID，幂等键（补交与在线提交共用）
  final int poemId;
  final String dictationType;     // full / key_line / fill_blank / error_word
  final List<OfflineLineAnswer> answers; // 按 lineNumber 排序
  final DateTime startedAt;
  final int localVersion;         // 乐观锁，多端草稿合并用
}

class OfflineLineAnswer {
  final int lineNumber;
  final String text;
  final bool skipped;
}
```

**补交流程与冲突规则：**

```text
[网络恢复] → 逐条补交队列（FIFO，按 startedAt）
   │
   ├─ 服务端未见 clientRequestId → 正常评判，返回 recordId
   ├─ 服务端已见 clientRequestId → 幂等返回首次结果（不重复计分、
   │                                  不重复触发 SM-2 更新，对应错误码 REVIEW_CONFLICT）
   └─ 篇目内容已修订（content_version 不一致）→ 409 DICTATION_CONTENT_STALE
        → 客户端提示「篇目内容已更新」，丢弃草稿并重新生成题目
```

- 补交请求与在线提交走同一接口 `POST /api/v1/poems/{poemId}/dictation/submit`，请求体统一携带 `clientRequestId` 与 `contentVersion`。
- 离线期间不扣会员额度（默写评判为本地/服务端纯计算，非 LLM 调用）；仅 AI 讲解受额度管控。

### 4.3 组卷策略

|------|------|----------|------|
| Q1-Q2 | 理解性默写（key_line） | 高频篇目 content_lines + annotation | 给出释义/情境，要求写出原句 |

```java
public ExamPaper assemblePaper(Long userId, AssembleRequest req) {
    // 1. 高频篇目池：考频权重采样，优先未测篇目（防背题）
    List<PoemBrief> pool = poemClient.examFocused(req.getGrade(), req.getStageTag());
    List<PoemBrief> picked = weightedSample(pool, 4,
        p -> p.getExamFrequency() * (p.getUserTested() ? 0.4 : 1.0));

    List<QuestionSlot> slots = new ArrayList<>();
    // 2. 并行取题（CompletableFuture，超时 800ms/题，失败槽位降级为 fill_blank）
    for (int i = 0; i < 2; i++) slots.add(slot(picked.get(i), "key_line"));
    for (int i = 2; i < 4; i++) slots.add(slot(picked.get(i), "fill_blank"));

    // 3. 易错字槽位：个人队列优先，不足时平台错字榜补齐
    List<WeakWordDrill> drills = drillClient.dueItems(userId, 8);
    slots.add(drills.isEmpty()
        ? platformHotWrongCharSlot(req.getGrade())
        : errorWordSlot(drills));

    // 4. 鉴赏题槽位
    slots.add(imageryChoiceSlot(picked.get(0)));

    String paperId = "EP" + UUID.randomUUID().toString().replace("-", "").substring(0, 12);
    redis.setex("poem:paper:" + paperId, Duration.ofHours(24), serialize(slots));
    return ExamPaper.of(paperId, slots); // 题目不含答案，答案仅存 submit 评判侧
}
```

### 4.4 能力报告计算

```java
public ExamAbilityReport buildReport(Long userId, String paperId) {
    List<QuestionResult> results = paperResults(paperId); // 6 题评判结果

    double memoryAccuracy = results.stream()               // 记忆准确率：默写字准率均值
        .filter(r -> r.isDictationType())
        .mapToInt(QuestionResult::getCharAccuracy)
        .average().orElse(0);

    Map<ErrorType, Long> writing = mergeCharErrors(results); // 书写规范：错字类型分布
    int imageryCorrect = countCorrect(results, "imagery");   // 意象理解
    int appreciationCorrect = countCorrect(results, "appreciation"); // 鉴达表达

    return ExamAbilityReport.builder()
        .memoryAccuracy(memoryAccuracy)
        .writingErrorDist(writing)                    // homophone/similar_shape/missing 加权提示
        .imageryUnderstanding(imageryCorrect)
        .appreciationAbility(appreciationCorrect)
        .suggestion(buildSuggestion(memoryAccuracy, writing)) // 薄弱维度→对应回补动作
        .remediation(buildRemediation(results))       // 错字→训练队列、错句→薄弱行标记
        .build();
}
```

**回补动作映射：**

| 薄弱维度 | 判定 | 回补动作 |
|----------|------|----------|
| 记忆准确率 < 80 | Q1-Q4 字准率 | 推荐对应篇目进入「今日复习」，走场景一链路 |
| 同音字错误占比 > 50% | writing 分布 | 生成同音字对比训练（weak_word_drill error_word 模式） |
| 意象理解 0 分 | Q6 错误 | 推送意象卡片学习（imagery 相关篇目聚合） |

- 讲解类回答优先命中 `poem_explain_cache`（按 `uk_cache` 五元组）：命中则直接注入缓存讲解文本，不调 LLM、不扣额度；未命中走《AI-Prompt编排与场景模板系统》`poem_chat` 场景模板 + SSE 流式输出。

## 7. 状态机守卫总表

### 7.1 篇目学习状态机（唯一真源：`user_poem_progress.status`）

```text
not_started ──(首次学习行为：详情页阅读/AI讲解)──▶ learning
learning    ──(explainCoverage ≥ 0.8 且完成任一模式检测)──▶ reciting
reciting    ──(full 模式得分 ≥ 80)──▶ mastered
mastered    ──(到期复习得分 < 80)──▶ decaying
mastered    ──(到期复习得分 ≥ 80)──▶ mastered（仅更新 SM-2 参数）
decaying    ──(full 模式得分 ≥ 80)──▶ mastered
decaying    ──(full 模式得分 < 80)──▶ decaying（lapse_count+1，间隔重置）
```

| 守卫 | 规则 | 违反时响应 |
|------|------|-----------|
| G1 | `not_started` 之前无状态，任何状态可回 `decaying`/`mastered` 之外的前态仅由降分触发 | — |
| G2 | `learning → reciting` 需同时满足 explainCoverage 与检测完成双条件，缺一不流转 | 状态不变，仅记录检测成绩 |
| G3 | 状态流转与 SM-2 参数更新必须同事务（上层文档 §6.3 约束），禁止两段提交 | 事务回滚，返回 500，客户端提示重试 |
| G4 | 并发重复提交复习结果：以先到者为准 | 后到者 409 `REVIEW_CONFLICT` + 幂等返回首次结果体 |
| G5 | 篇目下架后状态冻结（不参与流转），仅可查看 | `POEM_NOT_FOUND` 404 |

### 7.2 默写记录生命周期

```text
客户端: [editing] ──提交──▶ [submitted] ──评判──▶ [evaluated(终态)]
离线态: [local_draft] ──补交──▶ [synced]（复用 submitted→evaluated）
```

- `dictation_record` 评判后**不可变**：纠错不修改记录，走 `poem.explain.feedback.created`；
- 同一 `clientRequestId` 重复提交 → 返回首次 `recordId` 与结果体（G4）；
- `contentVersion` 不匹配 → 409 `DICTATION_CONTENT_STALE`，草稿作废。

### 7.3 易错字训练状态机（`weak_word_drill.status`）

```text
new/training/reactivated ──选对──▶ consecutive_correct+1 ──达3──▶ conquered
conquered ──任意检测再错──▶ reactivated（wrong_count 续加，consecutive_correct 清零）
conquered 出队：不进日常队列，仅作「曾经易错」标记
next_drill_at：错后 +1天 → 连对1次 +3天 → 连对2次 +7天
```

| 守卫 | 规则 | 违反时响应 |
|------|------|-----------|
| D1 | upsert 冲突键 `uk_user_char`，同一字多处错误合并一条（保留首错行号/字位） | — |
| D2 | `conquered` 后的练习请求不消耗训练配额 | 直接返回空队列 |
| D3 | 训练提交携带 drillId + clientRequestId 双幂等 | 重复提交幂等返回 |

## 8. 异常处理与降级矩阵

| # | 异常场景 | 检测方式 | 处理策略 | 用户感知 |
|---|----------|----------|----------|----------|
| 1 | AI 讲解生成失败（两次重试后） | 统一 AI 管线熔断信号 | 返回静态结构化数据（原文+注释+译文）+ 降级提示 | 「AI 讲解暂时不可用，已展示基础注释」 |
| 2 | 讲解缓存失效且待复核 | cache status=0 | 隐藏 AI 讲解入口，仅静态内容 | 不展示入口，无报错 |
| 3 | 非会员额度耗尽触发讲解 | 额度管控返回 `MEMBER_LIMIT_REACHED` | 弹权益引导卡片 | 引导弹窗，可关闭继续背诵 |
| 4 | 默写并发重复提交 | clientRequestId 去重表 | 幂等返回首次结果 | 无感知（看到首次成绩） |
| 5 | 离线补交时篇目已修订 | contentVersion 比对 | 409 `DICTATION_CONTENT_STALE`，草稿作废 | 「篇目内容已更新，请重新开始」 |
| 6 | 提交行数不符 | requiredLines 校验 | 400 `DICTATION_LINE_COUNT_MISMATCH`（`allowPartial=true` 时缺行按 missing 批改） | 行内提示补全缺失行 |
| 7 | 答案含非中文字符/行号越界 | 正则与边界校验 | 400 `DICTATION_ANSWER_MALFORMED`，不产生记录 | 输入框标红提示 |
| 8 | 篇目下架/删除 | status 检查 | 404 `POEM_NOT_FOUND` | 空态页 + 返回列表引导 |
| 9 | 事件投递失败 | Outbox 重试计数 | 指数退避 5 次 → 死信 → 延迟任务兜底重投 | 无感知（学习主流程不阻塞） |
| 10 | SM-2 数据异常（EF 越界/间隔为负） | applySm2 前置校验 | clamp（EF∈[1.3,2.8]）+ 告警日志 | 无感知 |
| 11 | 组卷取题部分超时（场景二） | 800ms/题超时 | 失败槽位降级为 fill_blank | 卷面题量不变 |
| 12 | paperId 过期（>24h） | Redis TTL | 提示重新组卷 | 「测试已过期，重新开始」 |

## 9. 埋点与监控

### 9.1 转化漏斗（复用上层文档 §10 埋点事件）

```text
poem_detail_view → poem_explain_view → dictation_submit → score≥80
     (曝光)            (学习深度)          (检测转化)        (通过率)
```

### 9.2 关键 SQL

```sql
-- 近 7 日默写参与率与通过率（按学段）
SELECT stage_tag,
       COUNT(DISTINCT CASE WHEN event='poem_detail_view' THEN user_id END) AS viewed,
       COUNT(DISTINCT CASE WHEN event='dictation_submit' THEN user_id END) AS submitted,
       ROUND(AVG(CASE WHEN event='dictation_submit' THEN prop_float.score END), 1) AS avg_score
FROM dwd_event_daily
WHERE dt >= CURRENT_DATE - INTERVAL 7 DAY
  AND event IN ('poem_detail_view','dictation_submit')
GROUP BY stage_tag;

-- AI 讲解命中率（缓存命中不调 LLM，成本核心指标）
SELECT dt,
       SUM(CASE WHEN prop.from_cache = 1 THEN 1 ELSE 0 END) / COUNT(*) AS cache_hit_rate
FROM dwd_event_daily
WHERE event = 'poem_explain_view' AND dt >= CURRENT_DATE - INTERVAL 7 DAY
GROUP BY dt ORDER BY dt DESC;

-- 到期复习积压（次日 05:00 批任务后巡检）
SELECT COUNT(*) AS due_backlog
FROM user_poem_progress p
WHERE p.status IN ('reciting','decaying','mastered')
  AND p.next_review_at < NOW() - INTERVAL 2 DAY;
```

### 9.3 指标与告警

| 指标 | 阈值 | 级别 |
|------|------|------|
| 默写提交接口错误率 | > 2%（5min 窗口） | P1 |
| 到期复习积压 due_backlog | > 5000 或连续 3 天增长 | P2（复核调度参数） |
| AI 讲解缓存命中率 | < 90%（教材同步篇目，日粒度） | P2（成本告警） |
| `poem.dictation.completed` 事件消费延迟 | > 10min | P2 |
| 组卷降级率（场景二槽位降级占比） | > 15% | P3 |
