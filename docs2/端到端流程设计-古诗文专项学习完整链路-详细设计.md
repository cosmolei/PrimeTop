  notStarted,    // 未开始：从未学习过
  inProgress,    // 学习中：已查看内容，未完成首次默写
  firstDictation, // 待首次默写：已完成学习，需默写检测

**掌握度进度条计算逻辑：**

```dart
double calculateMasteryProgress(PoemUserProgress progress) {
  if (progress.status == PoemMasteryStatus.notStarted) return 0.0;
  
  double score = 0.0;
  
  // 基础分：已学习内容
  if (progress.hasStudiedContent) score += 0.2;
  
  // 首次默写得分贡献
  if (progress.bestDictationScore != null) {
    score += (progress.bestDictationScore! / 100) * 0.4;
  }
  
  // 复习次数贡献（最多0.2）
  score += min(progress.reviewCount / 5, 1.0) * 0.2;
  
  // 最近表现贡献
  if (progress.lastThreeScoresAverage >= 95) {
    score += 0.2;
  } else if (progress.lastThreeScoresAverage >= 80) {
    score += 0.1;
  }
  
  return min(score, 1.0);
}
```

#### 3.4.1 默写评判引擎

```java
@Service
public class DictationEvaluationService {
    
    /**
     * 逐字评判默写内容
     */
    public DictationResult evaluate(DictationSubmitRequest request, Poem poem) {
        List<ContentLine> expectedLines = poem.getContentLines().getLines();
        List<LineAnswer> answers = request.getAnswers();
        
        int totalChars = 0;
        int correctChars = 0;
        List<CharError> errors = new ArrayList<>();
        
        for (int i = 0; i < expectedLines.size(); i++) {
            ContentLine expected = expectedLines.get(i);
            String expectedText = expected.getText().replaceAll("[，。！？、；：""''（）]", "");
            String actualText = "";
            if (i < answers.size()) {
                actualText = answers.get(i).getText().replaceAll("[，。！？、；：""''（）]", "");
            }
            
            totalChars += expectedText.length();
            
            // 逐字对齐比对
            List<CharError> lineErrors = compareChars(
                expectedText, actualText, expected.getLineNumber()
            );
            
            correctChars += expectedText.length() - lineErrors.size();
            errors.addAll(lineErrors);
        }
        
        // 计算得分
        double score = totalChars > 0 ? (correctChars * 100.0 / totalChars) : 0;
        score = Math.round(score * 10.0) / 10.0; // 保留1位小数
        
        // 错误分类统计
        Map<ErrorType, Long> errorTypeCount = errors.stream()
            .collect(Collectors.groupingBy(CharError::getErrorType, Collectors.counting()));
        
        // 生成学习建议
        String suggestion = generateSuggestion(score, errors, errorTypeCount);
        
        // 识别薄弱行
        Set<Integer> weakLines = identifyWeakLines(errors, expectedLines);
        
        return DictationResult.builder()
            .score(score)
            .totalChars(totalChars)
            .correctChars(correctChars)
            .errors(errors)
            .errorTypeDistribution(errorTypeCount)
            .weakLines(weakLines)
            .suggestion(suggestion)
            .build();
    }
    
    /**
     * 逐字比对算法
     * 处理：同音字、形近字、缺字、多字、顺序颠倒
     */
    private List<CharError> compareChars(String expected, String actual, int lineNumber) {
        List<CharError> errors = new ArrayList<>();
        int maxLen = Math.max(expected.length(), actual.length());
        
        for (int i = 0; i < expected.length(); i++) {
            char expectedChar = expected.charAt(i);
            
            if (i >= actual.length()) {
                // 缺字
                errors.add(CharError.builder()
                    .lineNumber(lineNumber)
                    .position(i)
                    .expectedChar(expectedChar)
                    .actualChar(null)
                    .errorType(ErrorType.MISSING)
                    .hint(generateHint(expectedChar, ErrorType.MISSING))
                    .build());
                continue;
            }
            
            char actualChar = actual.charAt(i);
            
            if (expectedChar == actualChar) {
                continue; // 正确
            }
            
            // 判断错误类型
            ErrorType errorType = classifyError(expectedChar, actualChar);
            errors.add(CharError.builder()
                .lineNumber(lineNumber)
                .position(i)
                .expectedChar(expectedChar)
                .actualChar(actualChar)
                .errorType(errorType)
                .hint(generateHint(expectedChar, errorType))
                .build());
        }
        
        // 检查多出的字
        for (int i = expected.length(); i < actual.length(); i++) {
            errors.add(CharError.builder()
                .lineNumber(lineNumber)
                .position(i)
                .expectedChar(null)
                .actualChar(actual.charAt(i))
                .errorType(ErrorType.EXTRA)
                .hint(null)
                .build());
        }
        
        return errors;
    }
    
    /**
     * 错误分类
     */
    private ErrorType classifyError(char expected, char actual) {
        // 1. 同音字检测
        if (isHomophone(expected, actual)) {
            return ErrorType.HOMOPHONE;
        }
        // 2. 形近字检测
        if (isSimilarShape(expected, actual)) {
            return ErrorType.SIMILAR_SHAPE;
        }
        // 3. 默认：错字
        return ErrorType.WRONG_CHAR;
    }
}
```

@Component
public class DictationCompletedHandler {
    
    @EventListener
    @Async("learningEventExecutor")
    public void onDictationCompleted(DictationCompletedEvent event) {
        Long userId = event.getUserId();
        Long poemId = event.getPoemId();
        DictationResult result = event.getResult();
        
        // 1. 更新篇目掌握度
        updatePoemMastery(userId, poemId, result);
        
        // 2. 将错误字加入错题本
        if (!result.getErrors().isEmpty()) {
            syncToErrorBook(userId, poemId, result.getErrors());
        }
        
        // 3. 触发间隔复习调度
        scheduleReview(userId, poemId, result.getScore());
        
        // 4. 更新学情分析数据
        updateLearningAnalytics(userId, poemId, result);
        
        // 5. 检查成就触发
        checkAchievement(userId, poemId, result);
        
        // 6. 更新知识点掌握度（意象、表达技巧等）
        updateKnowledgePointMastery(userId, poemId, result.getScore());
    }
    
    private void updatePoemMastery(Long userId, Long poemId, DictationResult result) {
        PoemUserProgress progress = progressRepository
            .findByUserIdAndPoemId(userId, poemId)
            .orElse(PoemUserProgress.createInitial(userId, poemId));
        
        // 更新统计数据
        progress.setDictationCount(progress.getDictationCount() + 1);
        progress.setBestDictationScore(
            Math.max(progress.getBestDictationScore(), result.getScore())
        );
        progress.setLastDictationScore(result.getScore());
        progress.setLastStudyTime(Instant.now());
        
        // 更新掌握状态
        PoemMasteryStatus newStatus = determineMasteryStatus(progress, result);
        progress.setStatus(newStatus);
        
        // 记录薄弱行
        if (!result.getWeakLines().isEmpty()) {
            progress.setWeakLines(result.getWeakLines());
        }
        
        // 判断是否达到「已掌握」标准
        if (progress.getConsecutiveHighScores() >= 3 
            && progress.getReviewIntervalDays() >= 30) {
            progress.setStatus(PoemMasteryStatus.MASTERED);
            progress.setMasteredAt(Instant.now());
        }
        
        progressRepository.save(progress);
    }
    
    /**
     * 掌握度状态判断
