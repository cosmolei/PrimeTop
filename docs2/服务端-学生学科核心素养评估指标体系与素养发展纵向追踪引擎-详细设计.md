# 服务端-学生学科核心素养评估指标体系与素养发展纵向追踪引擎-详细设计

## 1. 概述

### 1.1 功能定位

本引擎负责建立覆盖全学科、对齐教育部《课程标准》中"核心素养"定义的评估指标体系，持续追踪学生在各学科核心素养维度上的发展轨迹，输出素养评估结果、发展曲线、薄弱素养诊断和靶向提升建议。

与已有的"学科能力维度评估"（侧重计算/阅读/推理等通用能力）和"学科核心素养评估与新高考选科推荐"（侧重选科决策）不同，本引擎聚焦于：

1. **指标体系建模**：将各学科课程标准中定义的核心素养条目（如数学的"数学抽象""逻辑推理""数学建模"等）结构化为可量化评估的指标树。
2. **素养发展追踪**：以时间序列方式持续记录每个素养维度的评估值变化，生成发展曲线与趋势预测。
3. **薄弱素养诊断**：识别长期低水平的素养维度，生成靶向提升建议与学习内容推荐。
4. **课标对齐校验**：确保素养评估结果与课标条目保持一致，支持教育监管合规要求。

### 1.2 设计目标

| 目标 | 描述 |
|------|------|
| 指标覆盖 | 覆盖义务教育+高中阶段9大学科的核心素养指标 |
| 评估精度 | 核心素养评估值与教师人工评估的相关系数 ≥ 0.75 |
| 追踪连续性 | 支持按周/月/学期/学年多粒度的素养发展快照 |
| 实时性 | 每次学习行为发生后 ≤ 5分钟内更新对应素养评估值 |
| 可解释性 | 每个素养评分提供证据链溯源（关联的学习行为和答题记录） |

### 1.3 适用范围

- 学段：小学三年级至高中三年级（核心素养评估需要一定的答题数据基础）
- 学科：语文、数学、英语、物理、化学、生物、历史、地理、政治
- 用户角色：学生（评估对象）、家长（查看报告）、教师（班级素养分析）、管理后台（指标配置）

### 4.2 指标得分计算

每个评估指标从学习行为中采集证据，计算指标原始分：

```python
import math
from datetime import datetime, timedelta
from typing import List, Dict

class IndicatorScorer:
    """核心素养指标得分计算器"""
    
    # 滑动窗口大小（天）
    WINDOW_DAYS = 90
    # 时间衰减半衰期（天）
    DECAY_HALF_LIFE = 30.0
    
    def calculate_indicator_score(
        self,
        student_id: int,
        indicator_code: str,
        evidence_records: List[Dict],
        reference_time: datetime = None
    ) -> Dict:
        """
        计算指标得分
        
        Args:
            student_id: 学生ID
            indicator_code: 指标编码
            evidence_records: 证据记录列表
            reference_time: 参考时间（默认当前时间）
        
        Returns:
            {
                "indicator_code": str,
                "raw_score": float,      # 0-100
                "confidence": float,     # 0-1
                "effective_evidence_count": int,
                "weighted_avg": float
            }
        """
        if not reference_time:
            reference_time = datetime.now()
        
        # 1. 过滤窗口内证据
        window_start = reference_time - timedelta(days=self.WINDOW_DAYS)
        valid_evidence = [
            e for e in evidence_records
            if e["recorded_at"] >= window_start
        ]
        
        if not valid_evidence:
            return {
                "indicator_code": indicator_code,
                "raw_score": 0.0,
                "confidence": 0.0,
                "effective_evidence_count": 0,
                "weighted_avg": 0.0
            }
        
        # 2. 计算时间衰减权重
        total_weight = 0.0
        weighted_score_sum = 0.0
        
        for evidence in valid_evidence:
            days_ago = (reference_time - evidence["recorded_at"]).days
            # 指数衰减： w = 0.5^(days / half_life)
            decay_weight = math.pow(0.5, days_ago / self.DECAY_HALF_LIFE)
            
            # 难度加权：难题高分权重更大
            difficulty_weight = self._get_difficulty_weight(evidence)
            
            # 综合权重
            weight = decay_weight * difficulty_weight
            weighted_score_sum += evidence["source_score"] * weight
            total_weight += weight
        
        weighted_avg = weighted_score_sum / total_weight if total_weight > 0 else 0.0
        
        # 3. 计算置信度（基于有效证据数量）
        confidence = self._calculate_confidence(len(valid_evidence))
        
        return {
            "indicator_code": indicator_code,
            "raw_score": round(weighted_avg, 2),
            "confidence": round(confidence, 3),
            "effective_evidence_count": len(valid_evidence),
            "weighted_avg": round(weighted_avg, 2)
        }
    
    def _get_difficulty_weight(self, evidence: Dict) -> float:
        """根据题目难度计算权重"""
        difficulty = evidence.get("metadata", {}).get("difficulty", 0.5)
        # 难度映射：简单=0.8, 中等=1.0, 困难=1.3
        if difficulty < 0.3:
            return 0.8
        elif difficulty < 0.7:
            return 1.0
        else:
            return 1.3
    
    def _calculate_confidence(self, evidence_count: int) -> float:
        """基于证据数量计算置信度
        
        使用对数曲线：更多的证据带来更高的置信度，但边际递减
        置信度 = 1 - exp(-count / 10)
        10条证据 → ~63%, 20条 → ~86%, 30条 → ~95%
        """
        return min(1.0, 1.0 - math.exp(-evidence_count / 10.0))
```

### 4.3 素养分合成

将多个指标的得分合成为核心素养总分：

```python
class CompetencyScoreAggregator:
    """核心素养得分合成器"""
    
    def aggregate(
        self,
        competency_code: str,
        indicator_scores: List[Dict],
        indicator_weights: Dict[str, float]
    ) -> Dict:
        """
        合成核心素养总分
        
        Args:
            competency_code: 素养编码
            indicator_scores: 各指标得分列表
            indicator_weights: 指标权重字典 {indicator_code: weight}
        
        Returns:
            {
                "competency_code": str,
                "total_score": float,       # 0-100
                "confidence": float,        # 0-1
                "level": int,               # 1-5
                "indicator_details": list
            }
        """
        total_weight = 0.0
        weighted_sum = 0.0
        total_confidence_weight = 0.0
        confidence_sum = 0.0
        
        indicator_details = []
        
        for indicator in indicator_scores:
            code = indicator["indicator_code"]
            weight = indicator_weights.get(code, 0.1)
            
            # 只计算有证据的指标
            if indicator["effective_evidence_count"] > 0:
                weighted_sum += indicator["raw_score"] * weight
                total_weight += weight
                
                # 置信度加权
                confidence_sum += indicator["confidence"] * weight
                total_confidence_weight += weight
            
            indicator_details.append({
                "indicator_code": code,
                "score": indicator["raw_score"],
                "confidence": indicator["confidence"],
                "evidence_count": indicator["effective_evidence_count"],
                "weight": weight,
                "has_data": indicator["effective_evidence_count"] > 0
            })
        
        if total_weight == 0:
            return {
                "competency_code": competency_code,
                "total_score": 0.0,
                "confidence": 0.0,
                "level": 1,
                "indicator_details": indicator_details
            }
        
        total_score = weighted_sum / total_weight
        overall_confidence = confidence_sum / total_confidence_weight if total_confidence_weight > 0 else 0.0
        
        # 置信度衰减：如果总体置信度低，适当降低分数（保守估计）
        adjusted_score = self._apply_confidence_penalty(total_score, overall_confidence)
        
        level = self._score_to_level(competency_code, adjusted_score)
        
        return {
            "competency_code": competency_code,
            "total_score": round(adjusted_score, 2),
            "raw_score": round(total_score, 2),
            "confidence": round(overall_confidence, 3),
            "level": level,
            "indicator_details": indicator_details
        }
    
    def _apply_confidence_penalty(self, score: float, confidence: float) -> float:
        """低置信度时进行保守调整
        
        当置信度 < 0.5 时，将分数向中位数（50）拉近
        调整公式：adjusted = score * confidence + 50 * (1 - confidence)
        """
        if confidence >= 0.5:
            return score
        return score * confidence + 50.0 * (1.0 - confidence)
    
    def _score_to_level(self, competency_code: str, score: float) -> int:
        """分值映射到素养等级"""
        # 默认映射，实际应从 core_competency_definition.competency_level_defs 读取
        if score >= 81:
            return 5
        elif score >= 61:
            return 4
        elif score >= 41:
            return 3
        elif score >= 21:
            return 2
        else:
            return 1
```

### 4.4 素养发展曲线计算

```python
from typing import List, Tuple

class CompetencyTrendAnalyzer:
    """核心素养发展趋势分析器"""
    
    def analyze_trend(
        self,
        snapshots: List[Dict],
        current_score: float
    ) -> Dict:
        """
        分析素养发展趋势
        
        Args:
            snapshots: 历史快照列表（按时间升序）
            current_score: 当前素养分
        
        Returns:
            {
                "trend_direction": "UP" | "DOWN" | "STABLE",
                "trend_velocity": float,       # 变化速率（分/月）
                "predicted_next_score": float,  # 预测下一个月分数
                "milestone_progress": dict,     # 等级突破进度
                "consistency_score": float,     # 稳定性评分
                "comparison": dict              # 同期对比
            }
        """
        if len(snapshots) < 2:
            return self._default_trend(current_score)
        
        # 提取分数序列
        scores = [(s["period_key"], s["score"]) for s in snapshots]
        scores.append(("current", current_score))
        
        # 1. 计算变化速率（线性回归斜率）
        velocity = self._calculate_velocity(scores)
        
        # 2. 判断趋势方向
        if velocity > 1.5:
            direction = "UP"
        elif velocity < -1.5:
            direction = "DOWN"
        else:
            direction = "STABLE"
        
        # 3. 预测下一周期分数（加权移动平均 + 趋势外推）
        predicted = self._predict_next_score(scores, velocity)
        
        # 4. 等级突破进度
        milestone = self._calculate_milestone(current_score)
        
        # 5. 稳定性评分（分数波动程度）
        consistency = self._calculate_consistency([s[1] for s in scores])
        
        return {
            "trend_direction": direction,
            "trend_velocity": round(velocity, 2),
            "predicted_next_score": round(predicted, 2),
            "milestone_progress": milestone,
            "consistency_score": round(consistency, 3),
            "history_points": scores[-12:]  # 最近12个数据点
        }
    
    def _calculate_velocity(self, scores: List[Tuple[str, float]]) -> float:
        """使用线性回归计算变化速率"""
        n = len(scores)
        if n < 2:
            return 0.0
        
        x_values = list(range(n))
        y_values = [s[1] for s in scores]
        
        x_mean = sum(x_values) / n
        y_mean = sum(y_values) / n
        
        numerator = sum((x - x_mean) * (y - y_mean) for x, y in zip(x_values, y_values))
        denominator = sum((x - x_mean) ** 2 for x in x_values)
        
        return numerator / denominator if denominator != 0 else 0.0
    
    def _predict_next_score(self, scores, velocity):
        """预测下一周期分数"""
        recent_scores = [s[1] for s in scores[-4:]]  # 最近4个数据点
        avg_recent = sum(recent_scores) / len(recent_scores)
        # 加权预测：70% 最近平均 + 30% 趋势外推
        predicted = 0.7 * avg_recent + 0.3 * (scores[-1][1] + velocity)
        return max(0.0, min(100.0, predicted))
    
    def _calculate_milestone(self, score: float) -> Dict:
        """计算等级突破进度"""
        levels = [(20, "初步感知"), (40, "基本理解"), (60, "熟练运用"), (80, "灵活迁移"), (100, "创新思辨")]
        
        for i, (threshold, name) in enumerate(levels):
            if score < threshold:
                lower = levels[i - 1][0] if i > 0 else 0
                progress = (score - lower) / (threshold - lower) * 100
                return {
                    "current_level_name": levels[i - 1][1] if i > 0 else "起步阶段",
                    "next_level_name": name,
                    "next_level_threshold": threshold,
                    "progress_to_next": round(progress, 1),
                    "points_needed": round(threshold - score, 1)
                }
        
        return {
            "current_level_name": "创新思辨",
            "next_level_name": None,
            "next_level_threshold": None,
            "progress_to_next": 100.0,
            "points_needed": 0.0
        }
    
    def _calculate_consistency(self, scores: List[float]) -> float:
        """计算分数稳定性（变异系数的逆指标）"""
        if len(scores) < 2:
            return 1.0
        mean_score = sum(scores) / len(scores)
        if mean_score == 0:
            return 0.0
        variance = sum((s - mean_score) ** 2 for s in scores) / len(scores)
        std_dev = variance ** 0.5
        cv = std_dev / mean_score  # 变异系数
        # 转换为稳定性评分：CV越小越稳定
        return max(0.0, min(1.0, 1.0 - cv))
    
    def _default_trend(self, score: float) -> Dict:
        return {
            "trend_direction": "STABLE",
            "trend_velocity": 0.0,
            "predicted_next_score": score,
            "milestone_progress": self._calculate_milestone(score),
            "consistency_score": 1.0,
            "history_points": []
        }
```

### 6.1 事件订阅架构

素养评估引擎订阅以下业务事件，实现自动化的证据采集：

```python
from enum import Enum

class EvidenceEventType(Enum):
    """素养评估证据事件类型"""
    ANSWER_SUBMITTED = "answer.submitted"           # 答题提交
    HOMEWORK_COMPLETED = "homework.completed"        # 作业完成
    EXAM_QUESTION_ANSWERED = "exam.question.answered" # 考试答题
    AI_DIALOGUE_COMPLETED = "ai_dialogue.completed"   # AI对话完成
    LEARNING_SESSION_ENDED = "learning_session.ended" # 学习会话结束


class CompetencyEvidenceListener:
    """核心素养证据事件监听器"""
    
    # 事件队列 Topic
    TOPIC = "competency.evidence.collect"
    
    async def on_answer_submitted(self, event: Dict):
        """答题提交事件处理"""
        student_id = event["student_id"]
        question_id = event["question_id"]
        is_correct = event["is_correct"]
        knowledge_points = event.get("knowledge_points", [])
        difficulty = event.get("difficulty", 0.5)
        time_spent = event.get("time_spent_seconds", 0)
        
        # 1. 通过知识点反查关联的素养指标
        indicators = await self._lookup_indicators_by_knowledge(knowledge_points)
        
        if not indicators:
            return
        
        # 2. 计算该答题对每个指标的得分贡献
        source_score = self._calculate_answer_score(
            is_correct=is_correct,
            difficulty=difficulty,
            time_spent=time_spent,
            question_type=event.get("question_type")
        )
        
        # 3. 为每个关联指标创建证据记录
        for indicator in indicators:
            await self._create_evidence(
                student_id=student_id,
                competency_code=indicator["competency_code"],
                indicator_code=indicator["indicator_code"],
                source_type="QUESTION_ANSWER",
                source_id=event["answer_record_id"],
                source_score=source_score,
                metadata={
                    "question_id": question_id,
                    "difficulty": difficulty,
                    "knowledge_points": knowledge_points,
                    "is_correct": is_correct,
                    "time_spent": time_spent,
                    "question_type": event.get("question_type")
                }
            )
        
        # 4. 触发素养分增量更新
        for indicator in indicators:
            await self._trigger_competency_update(
                student_id, 
                indicator["competency_code"]
            )
    
    def _calculate_answer_score(
        self, 
        is_correct: bool, 
        difficulty: float,
        time_spent: int,
        question_type: str
    ) -> float:
        """计算单次答题的素养指标得分"""
        # 基础分：正确=60-100，错误=0-40
        if is_correct:
            base = 60 + difficulty * 40  # 难题正确得分更高
            # 时间效率加成
            expected_time = self._get_expected_time(question_type, difficulty)
            if time_spent <= expected_time:
                time_bonus = min(10, (expected_time - time_spent) / expected_time * 10)
            else:
                time_bonus = 0
            score = base + time_bonus
        else:
            # 错误也能获得部分分数（基于难度，难题即使错了也有价值）
            base = difficulty * 30
            score = base
        
        return min(100.0, max(0.0, score))
    
    def _get_expected_time(self, question_type: str, difficulty: float) -> int:
        """获取预期答题时间（秒）"""
        base_times = {
            "选择题": 60, "填空题": 90, "解答题": 300,
            "作文": 1800, "计算题": 120
        }
        base = base_times.get(question_type, 120)
        return int(base * (0.7 + difficulty * 0.6))
```

class DialogueCompetencyExtractor:
    """从AI对话中提取素养评估证据"""
    
    async def extract_from_dialogue(self, dialogue_record: Dict) -> List[Dict]:
        """
        从AI辅导对话记录中提取素养评估证据
        
        Args:
            dialogue_record: AI对话完整记录
        
        Returns:
            证据列表
        """
        evidences = []
        
        messages = dialogue_record.get("messages", [])
        if not messages:
            return evidences
        
        # 1. 提取对话中的学科和知识点
        subject = dialogue_record.get("subject_code")
        knowledge_points = dialogue_record.get("knowledge_points", [])
        
        # 2. 分析学生提问质量（反映思维品质）
        student_messages = [m for m in messages if m["role"] == "student"]
        if student_messages:
            thinking_score = self._evaluate_thinking_quality(student_messages)
            
            # 映射到对应学科的思维类素养
            thinking_competency = self._get_thinking_competency(subject)
            if thinking_competency:
                evidences.append({
                    "competency_code": thinking_competency,
                    "indicator_code": f"{thinking_competency}-DIA",
                    "source_type": "AI_DIALOGUE",
                    "source_id": dialogue_record["dialogue_id"],
                    "source_score": thinking_score,
                    "metadata": {
                        "topic": dialogue_record.get("topic"),
                        "interaction_rounds": len(student_messages),
                        "understanding_score": thinking_score
                    }
                })
        
        # 3. 分析对话中的建模/应用意识
        application_score = self._evaluate_application_awareness(messages)
        if application_score is not None:
            application_competency = self._get_application_competency(subject)
            if application_competency:
                evidences.append({
                    "competency_code": application_competency,
                    "source_type": "AI_DIALOGUE",
                    "source_id": dialogue_record["dialogue_id"],
                    "source_score": application_score,
                    "metadata": {
                        "analysis_type": "application_awareness"
                    }
                })
        
        return evidences
    
    def _evaluate_thinking_quality(self, student_messages: List[Dict]) -> float:
        """评估学生提问和回答中体现的思维品质"""
        score = 50.0  # 基础分
        
        for msg in student_messages:
            content = msg.get("content", "")
            
            # 追问深度：追问体现深入思考
            if len(content) > 50:
                score += 3
            
            # 包含"为什么""怎么"等探究性问题
            inquiry_words = ["为什么", "怎么", "如何", "如果", "能不能换"]
            if any(w in content for w in inquiry_words):
                score += 5
            
            # 包含自己的思路或尝试
            attempt_words = ["我觉得", "我试了", "是不是", "我的想法"]
            if any(w in content for w in attempt_words):
                score += 5
            
            # 纠正之前的错误理解
            if "哦我明白了" in content or "原来如此" in content:
                score += 3
        
        return min(100.0, score)

## 8. 错误处理与降级策略

### 8.1 错误码定义

| 错误码 | 含义 | 处理方式 |
|--------|------|----------|
| CC_001 | 素养定义不存在 | 返回404，提示联系管理员 |
| CC_002 | 学生无足够评估数据 | 返回空结果+引导提示 |
| CC_003 | 指标权重配置异常 | 使用默认权重0.1并记录告警 |
| CC_004 | 证据采集服务不可用 | 降级为定时批量采集 |
| CC_005 | 素养分计算溢出 | 钳位到[0,100]并记录日志 |
| CC_006 | 课标映射缺失 | 跳过该证据，不影响其他评估 |

### 8.2 降级策略

```python
class CompetencyEvaluationFallback:
    """核心素养评估降级策略"""
    
    async def evaluate_with_fallback(
        self, 
        student_id: int, 
        competency_code: str
    ) -> Dict:
        """带降级的素养评估"""
        try:
            # 正常流程：实时计算
            return await self._realtime_evaluate(student_id, competency_code)
        
        except EvidenceServiceUnavailable:
            # 降级1：使用最近快照
            latest = await self._get_latest_snapshot(student_id, competency_code)
            if latest:
                latest["degraded"] = True
                latest["degrade_reason"] = "证据服务不可用，使用缓存快照"
                return latest
            else:
                return self._empty_result(competency_code, "暂无评估数据")
        
        except CalculationTimeout:
            # 降级2：简化计算（减少窗口、跳过权重优化）
            simplified = await self._simplified_evaluate(student_id, competency_code)
            simplified["degraded"] = True
            simplified["degrade_reason"] = "计算超时，使用简化模型"
            return simplified
        
        except Exception as e:
            # 最终降级：返回默认结果
            return self._empty_result(competency_code, f"评估异常: {str(e)}")
    
    async def _simplified_evaluate(self, student_id: int, competency_code: str) -> Dict:
        """简化评估：只用最近7天的数据，均分计算"""
        recent_evidence = await self._get_recent_evidence(
            student_id, competency_code, days=7
        )
        if not recent_evidence:
            return self._empty_result(competency_code)
        
        avg_score = sum(e["source_score"] for e in recent_evidence) / len(recent_evidence)
        return {
            "competency_code": competency_code,
            "total_score": round(avg_score, 2),
            "confidence": min(0.5, len(recent_evidence) / 20),
            "level": self._score_to_level(avg_score),
            "simplified": True
        }
```

### 9.2 批量更新优化

```python
class CompetencyBatchUpdater:
    """素养评估批量更新器（非实时场景使用）"""
    
    BATCH_SIZE = 200  # 每批处理学生数
    
    async def batch_update_snapshot(
        self, 
        competency_code: str = None,
        period_key: str = None
    ):
        """批量更新素养快照"""
        
        # 获取需要更新的学生列表
        student_ids = await self._get_students_needing_update(competency_code)
        
        for i in range(0, len(student_ids), self.BATCH_SIZE):
            batch = student_ids[i:i + self.BATCH_SIZE]
            
            # 并行计算
            tasks = [
                self._update_single_student(
                    sid, competency_code, period_key
                ) 
                for sid in batch
            ]
            await asyncio.gather(*tasks, return_exceptions=True)
            
            # 批量写入（减少DB交互）
            await self._bulk_insert_snapshots(batch, competency_code, period_key)
    
    async def _get_students_needing_update(self, competency_code: str = None):
        """获取需要更新的学生：最近7天有新证据但快照超过24小时"""
        sql = """
            SELECT DISTINCT e.student_id
            FROM competency_evidence_record e
            LEFT JOIN student_competency_snapshot s 
                ON e.student_id = s.student_id 
                AND e.competency_code = s.competency_code
                AND s.snapshot_type = 'REALTIME'
            WHERE e.recorded_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
                AND (s.created_at IS NULL 
                     OR s.created_at < DATE_SUB(NOW(), INTERVAL 24 HOUR))
                AND (%s IS NULL OR e.competency_code = %s)
            LIMIT 10000
        """
        return await self.db.fetch_all(sql, (competency_code, competency_code))
```
