# 客户端 - 学习报告详情与分享页面架构与交互设计

## 1. 概述

### 1.1 文档范围

本文档细化客户端侧「学习报告」全链路页面架构与交互设计，涵盖：

- 报告列表页（日报 / 周报 / 月报 / 专项报告）
- 报告详情页（多维度数据、AI 洞察、图表、建议）
- 多角色差异化视图（学生端 / 家长端）
- 报告分享与导出（PDF / 长图 / 系统分享）
- 报告历史与趋势对比
- 报告推送入口与通知跳转
- 离线缓存与降级策略

> 服务端报告生成逻辑请参考 `学习报告生成与交付服务-详细设计.md`。
> 学情分析仪表盘请参考 `客户端-学情分析页面架构与报告展示交互-详细设计.md`。

### 1.2 与相邻模块关系

| 模块 | 关系 |
|------|------|
| 学情分析页面 | 学情分析是实时仪表盘；报告是周期性生成的结构化文档 |
| 学习报告生成与交付服务 | 服务端负责数据聚合、AI 摘要生成、PDF 渲染 |
| 数据可视化与图表渲染引擎 | 报告页面复用 VizChart 组件库 |
| 通知中心与站内消息系统 | 报告生成后推送通知，跳转至报告详情 |
| 收藏与笔记系统 | 报告可收藏 |
| 分享与社交裂变系统 | 报告分享使用统一分享 SDK |

```dart
// lib/features/report/providers/report_providers.dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

/// 报告列表 Provider
@riverpod
class ReportList extends _$ReportList {
  @override
  FutureOr<List<ReportListItem>> build(ReportType type) {
    return _loadFromCache(type);
  }

  Future<List<ReportListItem>> _loadFromCache(ReportType type) async {
    final box = Hive.box<ReportListCache>('report_lists');
    final key = type.name;
    final cached = box.get(key);
    if (cached != null &&
        cached.cachedAt.isAfter(DateTime.now().subtract(Duration(hours: 1)))) {
      // 缓存有效，先返回缓存 ID 列表，后台刷新
      final detailBox = Hive.box<ReportDetailCache>('report_details');
      return cached.reportIds
          .map((id) => detailBox.get(id))
          .whereType<ReportDetailCache>()
          .map((c) => ReportListItem.fromJson(c.detailJson))
          .toList();
    }
    return _fetchFromServer(type);
  }

  Future<List<ReportListItem>> _fetchFromServer(ReportType type) async {
    final api = ref.read(apiProvider);
    final result = await api.getReports(type: type.name);
    final items = result.items;

    // 写入缓存
    final box = Hive.box<ReportListCache>('report_lists');
    await box.put(
      type.name,
      ReportListCache(
        reportType: type.name,
        reportIds: items.map((e) => e.reportId).toList(),
        cachedAt: DateTime.now(),
        totalCount: items.length,
      ),
    );

    return items;
  }

  /// 下拉刷新
  Future<void> refresh() async {
    state = const AsyncLoading();
    state = AsyncData(await _fetchFromServer(type));
  }

  /// 加载更多
  Future<void> loadMore() async {
    final current = state.valueOrNull ?? [];
    if (current.isEmpty) return;

    final api = ref.read(apiProvider);
    final lastId = current.last.reportId;
    final result = await api.getReports(
      type: type.name,
      cursor: lastId,
    );

    if (result.items.isNotEmpty) {
      state = AsyncData([...current, ...result.items]);
    }
  }
}

/// 报告详情 Provider
@riverpod
class ReportDetailNotifier extends _$ReportDetailNotifier {
  @override
  FutureOr<ReportDetail> build(String reportId, {ReportRole role = ReportRole.student}) {
    return _load(reportId, role);
  }

  Future<ReportDetail> _load(String reportId, ReportRole role) async {
    // 先尝试本地缓存
    final box = Hive.box<ReportDetailCache>('report_details');
    final cached = box.get(reportId);
    if (cached != null &&
        cached.cachedAt.isAfter(DateTime.now().subtract(Duration(hours: 2)))) {
      final detail = ReportDetail.fromJson(cached.detailJson);
      if (detail.viewRole == role) return detail;
    }

    // 网络请求
    final api = ref.read(apiProvider);
    final detail = await api.getReportDetail(
      reportId: reportId,
      role: role,
    );

    // 写入缓存
    await box.put(
      reportId,
      ReportDetailCache(
        reportId: reportId,
        detailJson: detail.toJson(),
        cachedAt: DateTime.now(),
      ),
    );

    // 标记已读
    unawaited(api.markReportRead(reportId));

    return detail;
  }

  /// 切换角色视图
  Future<void> switchRole(ReportRole role) async {
    state = const AsyncLoading();
    state = AsyncData(await _load(reportId, role));
  }

  /// 收藏/取消收藏
  Future<void> toggleBookmark() async {
    final current = state.valueOrNull;
    if (current == null) return;

    final api = ref.read(apiProvider);
    final newVal = !current.isBookmarked;
    await api.bookmarkReport(reportId, newVal);

    state = AsyncData(current.copyWith(isBookmarked: newVal));
  }
}

  Ref ref,

  final api = ref.read(apiProvider);
```

```dart
class _ReportTypeList extends ConsumerWidget {
  final ReportType type;
  const _ReportTypeList({required this.type});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final reportsAsync = ref.watch(reportListProvider(type));

    return reportsAsync.when(
      data: (reports) {
        if (reports.isEmpty) {
          return _EmptyReportState(type: type);
        }
        return RefreshIndicator(
          onRefresh: () => ref.read(reportListProvider(type).notifier).refresh(),
          child: PagedListWidget(
            items: reports,
            onLoadMore: () =>
                ref.read(reportListProvider(type).notifier).loadMore(),
            itemBuilder: (context, item, index) =>
                _ReportCard(item: item),
          ),
        );
      },
      loading: () => const _ReportListSkeleton(),
      error: (e, _) => ErrorBoundary(
        error: e,
        onRetry: () => ref.invalidate(reportListProvider(type)),
      ),
    );
  }
}
```

### 8.3 PDF 导出流程

```
用户点击 "导出 PDF"
       │
       ▼
检查本地是否已缓存 PDF ──是──→ 打开 PDF 预览页
       │ 否
       ▼
调用 GET /reports/:id/pdf
       │
       ├── 200 → 获取下载 URL → 下载 PDF → 缓存本地 → 打开预览
       │
       └── 202 → 返回 task_id → 显示进度 → 轮询/SSE 等待 → 下载 → 预览
```

```dart
class ReportPdfService {
  final ApiClient _api;

  Future<String> getPdfPath(String reportId) async {
    // 检查本地缓存
    final box = Hive.box<ReportDetailCache>('report_details');
    final cached = box.get(reportId);
    if (cached?.pdfLocalPath != null) {
      final file = File(cached!.pdfLocalPath!);
      if (await file.exists()) return cached.pdfLocalPath!;
    }

    // 请求服务端
    final result = await _api.getReportPdf(reportId);

    if (result.status == 'ready') {
      // 下载 PDF
      final localPath = await _downloadFile(result.pdfUrl!, reportId);
      // 缓存路径
      if (cached != null) {
        await box.put(
          reportId,
          ReportDetailCache(
            reportId: reportId,
            detailJson: cached.detailJson,
            cachedAt: cached.cachedAt,
            pdfLocalPath: localPath,
          ),
        );
      }
      return localPath;
    }

    // 正在渲染，轮询等待
    return _waitForPdf(reportId, result.taskId!);
  }

  Future<String> _waitForPdf(String reportId, String taskId) async {
    for (var i = 0; i < 30; i++) {
      await Future.delayed(const Duration(seconds: 2));
      final status = await _api.getPdfStatus(reportId, taskId);
      if (status.status == 'ready') {
        return _downloadFile(status.pdfUrl!, reportId);
      }
      if (status.status == 'failed') {
        throw ReportException('PDF 生成失败，请稍后重试');
      }
    }
    throw ReportException('PDF 生成超时');
  }

  Future<String> _downloadFile(String url, String reportId) async {
    final dir = await getApplicationDocumentsDirectory();
    final path = '${dir.path}/reports/${reportId}.pdf';
    await Dio().download(url, path);
    return path;
  }
}
```
