# 题目相似度计算与智能去重服务 - 详细设计

## 1. 模块定位

### 1.1 功能定位

题目相似度计算与智能去重服务是题库质量保障与同类题推荐的核心基础设施，负责：
- **题库去重**：识别并标记重复或高度相似的题目，保持题库质量
- **同类题推荐**：为"练一道同类题"等场景实时检索相似题目
- **AI生成题目去重**：在AI题目生成引擎产出新题后，与已有题库比对，避免重复入库
- **搜题匹配增强**：在OCR搜题场景中，基于语义相似度提升匹配精度

### 1.2 与其他模块的关系

| 关联模块 | 交互方式 | 说明 |
|----------|----------|------|
| 题目与题库服务 | 同步调用 | 题目CRUD时触发相似度检查 |
| AI题目智能生成引擎 | 异步事件 | 新题生成后触发去重检查 |
| 拍照搜题与习题答疑 | 同步调用 | 搜题时进行语义匹配增强 |
| 练习与测评系统 | 同步调用 | "同类题推荐"时检索相似题目 |
| 错题整理 | 同步调用 | 错题复习时推荐相似变式题 |
| 搜索服务与全文检索 | 互补关系 | ES负责关键词检索，本服务负责语义相似度 |
| RAG与知识库系统 | 依赖 | 复用Embedding模型和向量数据库 |
| 多模型调度与成本治理 | 依赖 | Embedding调用通过模型调度层 |

### 1.3 核心挑战

1. **多模态题目表示**：题目包含文本、公式、图表、代码等多种内容，需要统一的向量化方案
2. **学科差异**：理科题目侧重公式/数字/图形，文科题目侧重语义/逻辑，相似度计算策略不同
3. **实时性要求**：同类题推荐需 <500ms 响应，去重检查可异步但需在入库前完成
4. **题库规模**：预期 100 万+ 题目，向量检索需高效 ANN 算法
5. **误判成本**：误判为重复导致好题被丢弃，误判为不同导致题库冗余

## 3. 核心算法设计

### 3.1 题目向量化管线

```python
from dataclasses import dataclass
from typing import Optional
import hashlib
import re

@dataclass
class QuestionEmbeddingInput:
    """题目向量化输入"""
    question_id: str
    subject_code: str
    stage_code: str
    question_type: str
    difficulty: float
    knowledge_point_ids: list[str]
    stem_text: str              # 题干纯文本
    options_text: str           # 选项文本（选择题）
    answer_text: str            # 答案文本
    formulas: list[str]         # 提取的公式列表
    has_image: bool             # 是否含图
    has_table: bool             # 是否含表


class QuestionVectorizer:
    """题目向量化器"""

    def __init__(self, embedding_client, config: dict):
        self.embedding_client = embedding_client  # 模型调度层客户端
        self.config = config

    def build_text_for_embedding(self, q: QuestionEmbeddingInput) -> str:
        """
        构造用于Embedding的文本表示
        策略：题干 + 选项 + 知识点标签 + 难度区间
        """
        parts = []

        # 1. 学科和题型前缀，帮助模型区分上下文
        parts.append(f"[{q.subject_code}][{q.question_type}]")

        # 2. 知识点标签
        if q.knowledge_point_ids:
            parts.append("知识点:" + " ".join(q.knowledge_point_ids[:5]))

        # 3. 题干（核心内容）
        stem = self._normalize_text(q.stem_text)
        parts.append(stem)

        # 4. 选项（选择题/判断题）
        if q.options_text:
            parts.append(self._normalize_text(q.options_text))

        # 5. 答案摘要（用于区分相同题干不同答案的变式题）
        if q.answer_text:
            answer_digest = self._extract_answer_digest(q.answer_text)
            parts.append(f"答案摘要:{answer_digest}")

        return " ".join(parts)

    def _normalize_text(self, text: str) -> str:
        """文本归一化：去除多余空白、统一标点、数字归一化"""
        if not text:
            return ""
        # 统一全角/半角
        text = text.replace("（", "(").replace("）", ")")
        text = text.replace("，", ",").replace("。", ".")
        text = text.replace("＝", "=").replace("＋", "+").replace("－", "-")
        # 压缩空白
        text = re.sub(r'\s+', ' ', text).strip()
        return text

    def _extract_answer_digest(self, answer: str) -> str:
        """提取答案摘要：选择题取选项字母，计算题取最终数值"""
        answer = answer.strip()
        # 选择题答案：A/B/C/D
        if re.match(r'^[A-Da-d]$', answer):
            return answer.upper()
        # 判断题
        if answer in ('对', '正确', '√', 'True'):
            return 'T'
        if answer in ('错', '错误', '×', 'False'):
            return 'F'
        # 提取最终数值
        nums = re.findall(r'[-+]?\d*\.?\d+', answer)
        if nums:
            return nums[-1]
        # 取前30字符
        return answer[:30]

    async def compute_text_vector(self, q: QuestionEmbeddingInput) -> list[float]:
        """计算文本语义向量"""
        text = self.build_text_for_embedding(q)
        response = await self.embedding_client.embed(
            texts=[text],
            model=self.config.get('embedding.model', 'bge-large-zh-v1.5'),
            normalize=True
        )
        return response.embeddings[0]

    def compute_structural_vector(self, q: QuestionEmbeddingInput) -> list[float]:
        """
        计算结构特征向量（无需AI调用，基于元数据手工构造）
        维度(256):
        - [0-19]   学科 one-hot (20维)
        - [20-30]  题型 one-hot (11维)
        - [31-35]  学段 one-hot (5维)
        - [36-44]  难度分桶 one-hot (9桶)
        - [45-245] 知识点哈希特征 (200维, SimHash风格)
        - [246-255] 辅助特征 (10维: 含图/含表/含公式/文本长度桶等)
        """
        import numpy as np
        vec = np.zeros(256, dtype=np.float32)

        # 学科 one-hot
        subject_map = {
            'MATH': 0, 'CHIN': 1, 'ENG': 2, 'PHYS': 3, 'CHEM': 4,
            'BIO': 5, 'HIST': 6, 'GEO': 7, 'POL': 8, 'SCI': 9,
            # ... 扩展至20
        }
        idx = subject_map.get(q.subject_code, 19)
        vec[idx] = 1.0

        # 题型 one-hot
        type_map = {
            'SINGLE_CHOICE': 0, 'MULTI_CHOICE': 1, 'FILL': 2,
            'SHORT_ANSWER': 3, 'ESSAY': 4, 'TRUE_FALSE': 5,
            'CALCULATION': 6, 'PROOF': 7, 'READING_COMP': 8,
            'CLOZE': 9, 'TRANSLATION': 10
        }
        idx = 20 + type_map.get(q.question_type, 10)
        vec[idx] = 1.0

        # 学段 one-hot
        stage_map = {'PRIMARY': 0, 'JUNIOR': 1, 'SENIOR': 2, 'KINDERGARTEN': 3, 'ALL': 4}
        idx = 31 + stage_map.get(q.stage_code, 4)
        vec[idx] = 1.0

        # 难度分桶 (0.0-1.0 → 9个桶)
        bucket = min(int(q.difficulty * 9), 8)
        vec[36 + bucket] = 1.0

        # 知识点 SimHash 特征
        for kp_id in q.knowledge_point_ids[:20]:
            h = int(hashlib.md5(kp_id.encode()).hexdigest(), 16)
            for bit in range(200):
                vec[45 + bit] += 1.0 if (h >> (bit % 64)) & 1 else -1.0
        if q.knowledge_point_ids:
            vec[45:245] /= len(q.knowledge_point_ids[:20])

        # 辅助特征
        vec[246] = 1.0 if q.has_image else 0.0
        vec[247] = 1.0 if q.has_table else 0.0
        vec[248] = 1.0 if q.formulas else 0.0
        vec[249] = min(len(q.stem_text) / 500.0, 1.0)  # 题干长度归一化
        vec[250] = min(len(q.formulas) / 5.0, 1.0)     # 公式数量归一化
        vec[251] = min(len(q.knowledge_point_ids) / 5.0, 1.0)
        vec[252] = 1.0 if q.options_text else 0.0       # 有选项
        vec[253] = 1.0 if len(q.stem_text) < 50 else 0.0  # 极短题目
        vec[254] = q.difficulty
        vec[255] = 0.0  # reserved

        # L2 归一化
        norm = np.linalg.norm(vec)
        if norm > 0:
            vec /= norm

        return vec.tolist()

    async def vectorize(self, q: QuestionEmbeddingInput) -> dict:
        """完整向量化流程"""
        text_vector = await self.compute_text_vector(q)
        structural_vector = self.compute_structural_vector(q)

        return {
            "question_id": q.question_id,
            "text_vector": text_vector,
            "structural_vector": structural_vector,
            "subject_code": q.subject_code,
            "stage_code": q.stage_code,
            "knowledge_point_ids": q.knowledge_point_ids,
            "difficulty": q.difficulty,
            "question_type": q.question_type,
        }
```

### 3.2 综合相似度计算

```python
from enum import Enum

class SimilarityLevel(Enum):
    EXACT_DUPLICATE = "EXACT_DUPLICATE"   # >= 0.95
    HIGH_SIMILAR = "HIGH_SIMILAR"         # >= 0.85
    SIMILAR = "SIMILAR"                   # >= 0.70
    WEAK_SIMILAR = "WEAK_SIMILAR"         # >= 0.50
    DIFFERENT = "DIFFERENT"               # < 0.50


class SimilarityCalculator:
    """综合相似度计算器"""

    def __init__(self, config: dict):
        self.config = config
        # 权重配置（可按学科覆盖）
        self.default_weights = {
            'text': config.get('weights.text', 0.50),
            'structural': config.get('weights.structural', 0.25),
            'keyword': config.get('weights.keyword', 0.15),
            'formula': config.get('weights.formula', 0.10),
        }

    def get_weights(self, subject_code: str) -> dict:
        """获取学科适配的权重"""
        weights = self.default_weights.copy()

        # 理科：公式权重更高，文本权重略降
        if subject_code in ('MATH', 'PHYS', 'CHEM', 'BIO'):
            weights['formula'] = 0.20
            weights['text'] = 0.40
        # 文科：文本权重更高，公式权重降为0
        elif subject_code in ('CHIN', 'ENG', 'HIST', 'GEO', 'POL'):
            weights['formula'] = 0.0
            weights['text'] = 0.55
            weights['keyword'] = 0.20

        # 归一化
        total = sum(weights.values())
        return {k: v / total for k, v in weights.items()}

    async def compute(
        self,
        query: QuestionEmbeddingInput,
        candidate: QuestionEmbeddingInput,
        query_vectors: dict,
        candidate_vectors: dict,
    ) -> dict:
        """
        计算两道题目的综合相似度

        Args:
            query: 查询题目
            candidate: 候选题目
            query_vectors: 查询题目的预计算向量
            candidate_vectors: 候选题目的预计算向量

        Returns:
            {
                "overall_score": float,
                "text_score": float,
                "structural_score": float,
                "keyword_score": float,
                "formula_score": float | None,
                "similarity_level": SimilarityLevel,
                "detail": dict
            }
        """
        weights = self.get_weights(query.subject_code)

        # 1. 文本语义相似度 (cosine)
        text_score = self._cosine_similarity(
            query_vectors['text_vector'],
            candidate_vectors['text_vector']
        )

        # 2. 结构特征相似度
        structural_score = self._cosine_similarity(
            query_vectors['structural_vector'],
            candidate_vectors['structural_vector']
        )

        # 3. 关键词重叠度 (Jaccard)
        keyword_score = self._keyword_overlap(
            query.stem_text, candidate.stem_text
        )

        # 4. 公式相似度
        formula_score = None
        if weights['formula'] > 0 and query.formulas and candidate.formulas:
            formula_score = self._formula_similarity(
                query.formulas, candidate.formulas
            )
        elif weights['formula'] > 0:
            formula_score = 0.0

        # 5. 加权综合
        overall = (
            weights['text'] * text_score
            + weights['structural'] * structural_score
            + weights['keyword'] * keyword_score
            + (weights['formula'] * formula_score if formula_score is not None else 0)
        )
        # 重新归一化（formula可能为0）
        used_weight_sum = sum(v for k, v in weights.items()
                            if k != 'formula' or formula_score is not None)
        if used_weight_sum > 0:
            overall /= used_weight_sum

        # 6. 辅助校验
        detail = self._compute_detail(query, candidate, text_score)

        # 7. 强制规则覆盖（答案不同的选择题不判定为重复）
        if detail.get('same_answer') is False and query.question_type == 'SINGLE_CHOICE':
            overall = min(overall, 0.89)  # 选择题答案不同，最多 HIGH_SIMILAR

        # 8. 判定等级
        level = self._classify_level(overall)

        return {
            "overall_score": round(overall, 4),
            "text_score": round(text_score, 4),
            "structural_score": round(structural_score, 4),
            "keyword_score": round(keyword_score, 4),
            "formula_score": round(formula_score, 4) if formula_score is not None else None,
            "similarity_level": level,
            "detail": detail,
        }

    def _cosine_similarity(self, vec_a: list[float], vec_b: list[float]) -> float:
        """余弦相似度"""
        import numpy as np
        a = np.array(vec_a)
        b = np.array(vec_b)
        dot = np.dot(a, b)
        norm_a = np.linalg.norm(a)
        norm_b = np.linalg.norm(b)
        if norm_a == 0 or norm_b == 0:
            return 0.0
        return float(np.clip(dot / (norm_a * norm_b), 0.0, 1.0))

    def _keyword_overlap(self, text_a: str, text_b: str) -> float:
        """基于 jieba 分词的 Jaccard 相似度"""
        import jieba
        # 去除停用词后分词
        stopwords = {'的', '了', '在', '是', '和', '与', '及', '等', '中', '为',
                      '对', '把', '被', '从', '到', '由', '以', '将', '其', '这', '那'}
        words_a = set(jieba.lcut(text_a)) - stopwords
        words_b = set(jieba.lcut(text_b)) - stopwords
        if not words_a or not words_b:
            return 0.0
        intersection = words_a & words_b
        union = words_a | words_b
        return len(intersection) / len(union)

    def _formula_similarity(self, formulas_a: list[str], formulas_b: list[str]) -> float:
        """
        公式相似度：基于 LaTeX 字符串的编辑距离
        对每对公式计算相似度，取最大匹配的平均值
        """
        if not formulas_a or not formulas_b:
            return 0.0

        def normalized_edit_distance(s1: str, s2: str) -> float:
            """归一化编辑距离 → 相似度"""
            m, n = len(s1), len(s2)
            if m == 0 and n == 0:
                return 1.0
            # 快速路径：完全相同
            if s1 == s2:
                return 1.0
            # Levenshtein
            import numpy as np
            dp = np.zeros((m + 1, n + 1), dtype=np.int32)
            for i in range(m + 1): dp[i][0] = i
            for j in range(n + 1): dp[0][j] = j
            for i in range(1, m + 1):
                for j in range(1, n + 1):
                    cost = 0 if s1[i-1] == s2[j-1] else 1
                    dp[i][j] = min(dp[i-1][j] + 1, dp[i][j-1] + 1, dp[i-1][j-1] + cost)
            max_len = max(m, n)
            return 1.0 - dp[m][n] / max_len if max_len > 0 else 1.0

        # 贪心最大匹配
        total_sim = 0.0
        used_b = set()
        for fa in formulas_a:
            best_sim = 0.0
            best_j = -1
            for j, fb in enumerate(formulas_b):
                if j in used_b:
                    continue
                sim = normalized_edit_distance(fa, fb)
                if sim > best_sim:
                    best_sim = sim
                    best_j = j
            if best_j >= 0:
                used_b.add(best_j)
                total_sim += best_sim

        return total_sim / max(len(formulas_a), len(formulas_b))

    def _compute_detail(self, query, candidate, text_score) -> dict:
        """辅助校验详情"""
        detail = {}

        # 答案是否相同
        if query.answer_text and candidate.answer_text:
            detail['same_answer'] = (
                query._extract_answer_digest(query.answer_text)
                == candidate._extract_answer_digest(candidate.answer_text)
            )

        # 知识点重叠
        kp_overlap = set(query.knowledge_point_ids) & set(candidate.knowledge_point_ids)
        detail['knowledge_point_overlap_count'] = len(kp_overlap)
        detail['knowledge_point_overlap_ratio'] = (
            len(kp_overlap) / max(len(set(query.knowledge_point_ids) | set(candidate.knowledge_point_ids)), 1)
        )

        # 题型是否一致
        detail['same_type'] = query.question_type == candidate.question_type

        # 难度差异
        detail['difficulty_diff'] = abs(query.difficulty - candidate.difficulty)

        # 文本长度比
        len_a, len_b = len(query.stem_text), len(candidate.stem_text)
        detail['length_ratio'] = min(len_a, len_b) / max(len_a, len_b, 1)

        return detail

    def _classify_level(self, score: float) -> SimilarityLevel:
        if score >= self.config.get('thresholds.exact_duplicate', 0.95):
            return SimilarityLevel.EXACT_DUPLICATE
        elif score >= self.config.get('thresholds.high_similar', 0.85):
            return SimilarityLevel.HIGH_SIMILAR
        elif score >= self.config.get('thresholds.similar', 0.70):
            return SimilarityLevel.SIMILAR
        elif score >= self.config.get('thresholds.weak_similar', 0.50):
            return SimilarityLevel.WEAK_SIMILAR
        else:
            return SimilarityLevel.DIFFERENT
```

### 3.3 题目指纹（快速精确去重）

在向量检索之前，先用轻量级指纹进行精确去重，避免重复调用 Embedding：

```python
import hashlib

class QuestionFingerprinter:
    """题目指纹生成器 — 用于O(1)精确去重"""

    @staticmethod
    def compute(question: QuestionEmbeddingInput) -> str:
        """
        生成题目指纹 (SHA-256)
        基于归一化后的题干+选项+答案
        """
        normalizer = QuestionVectorizer(None, {})
        parts = [
            normalizer._normalize_text(question.stem_text),
            normalizer._normalize_text(question.options_text or ''),
            normalizer._normalize_text(question.answer_text or ''),
            question.subject_code,
            question.question_type,
        ]
        raw = "|".join(parts)
        return hashlib.sha256(raw.encode('utf-8')).hexdigest()

    @staticmethod
    def compute_stem_only(question: QuestionEmbeddingInput) -> str:
        """仅基于题干的指纹（检测改了答案的变式题）"""
        normalizer = QuestionVectorizer(None, {})
        raw = normalizer._normalize_text(question.stem_text) + "|" + question.subject_code
        return hashlib.sha256(raw.encode('utf-8')).hexdigest()


# Redis 存储指纹集合
# Key: fingerprint:{stem_hash} → question_id
# Key: fingerprint:full:{full_hash} → question_id
# 用于 O(1) 精确查重
```

```python
class SimilarityServiceDegradation:
    """相似度服务降级策略"""

    LEVEL_FULL = 0        # 全功能：向量+指纹+关键词
    LEVEL_NO_VECTOR = 1   # 降级：仅指纹+关键词（向量服务不可用）
    LEVEL_FINGERPRINT = 2 # 最低：仅指纹检查（Embedding和向量均不可用）
    LEVEL_OFF = 3         # 关闭：跳过所有检查

    def get_degradation_level(self) -> int:
        """根据当前系统状态判断降级级别"""
        if not self.milvus_health_check():
            if not self.embedding_health_check():
                return self.LEVEL_FINGERPRINT
            return self.LEVEL_NO_VECTOR
        return self.LEVEL_FULL

    async def search_similar(self, question, top_k, **kwargs):
        level = self.get_degradation_level()

        if level == self.LEVEL_FULL:
            return await self._full_search(question, top_k, **kwargs)

        elif level == self.LEVEL_NO_VECTOR:
            # 降级：使用ES全文检索代替向量检索
            return await self._es_keyword_search(question, top_k, **kwargs)

        elif level == self.LEVEL_FINGERPRINT:
            # 最低降级：仅返回精确匹配结果
            exact = await self._fingerprint_search(question)
            return exact if exact else []

        else:
            return []

    async def dedup_check(self, question):
        level = self.get_degradation_level()

        if level <= self.LEVEL_NO_VECTOR:
            # 指纹检查仍然可以做
            return await self._fingerprint_dedup(question)

        if level == self.LEVEL_FULL:
            return await self._full_dedup(question)

        # 完全降级：放行，异步延迟检查
        dedup_check_single.apply_async(args=[question.id], countdown=300)
        return {"is_duplicate": False, "recommendation": "DEFERRED"}
```

## 10. 性能设计

### 10.1 性能目标

| 操作 | 目标延迟 | QPS目标 | 说明 |
|------|----------|---------|------|
| 同类题推荐 | < 500ms (p99) | 200 | 用户实时请求 |
| 单题去重检查 | < 1s (p99) | 50 | 入库前检查 |
| 全量扫描(10万题) | < 2h | - | 后台批处理 |
| 向量构建(单题) | < 2s | 50 (batch) | 含Embedding调用 |
| 指纹精确查重 | < 10ms | 1000 | Redis O(1) |

### 10.2 优化手段

1. **向量预计算**：题目入库时异步计算并缓存向量，推荐时直接使用
2. **分片检索**：按学科分Collection，减少单次检索的向量空间
3. **多级过滤**：先指纹(O(1)) → 再向量检索(ANN) → 最后精确计算
4. **批量Embedding**：批量调用Embedding API，减少网络开销
5. **结果缓存**：同类题推荐结果缓存10分钟，相同查询直接返回
6. **增量索引**：新增题目增量插入Milvus，避免全量重建

### 10.3 容量估算

```text
假设题库规模：100万题
向量维度：1024维 (float32) + 256维 (float32)
单题向量存储：1280 × 4 bytes = 5.12 KB
总向量存储：100万 × 5.12 KB ≈ 4.88 GB

Milvus 内存需求（IVF_FLAT, nlist=4096）：
  - 索引额外开销约 20% → ≈ 6 GB
  - 建议 8 GB 以上内存

Redis 指纹存储：
  - 100万 × 2指纹 × 100 bytes ≈ 200 MB

MySQL 相似度记录：
  - 假设5%为相似对 → 5万条 × 1KB ≈ 50 MB
```

### 12.1 题目与题库服务集成

```python
# 在题目服务中调用相似度服务

class QuestionService:
    def __init__(self, similarity_client: SimilarityClient):
        self.similarity = similarity_client

    async def create_question(self, data: dict) -> dict:
        """创建题目 — 集成去重检查"""
        # 1. 构建题目
        question = self._build_question(data)

        # 2. 去重检查
        dedup_result = await self.similarity.dedup_check(question.id)
        if dedup_result['recommendation'] == 'REJECT':
            raise BusinessException(
                code='QUESTION_DUPLICATE',
                message=f'题目与已有题目高度重复 (相似度: {dedup_result["max_similarity"]})',
                detail=dedup_result['duplicates']
            )

        # 3. 入库
        saved = await self.repo.save(question)

        # 4. 触发异步向量化（事件总线）
        await self.event_bus.publish('question.created', {
            'question_id': saved.id
        })

        return saved
```

### 12.2 练习与测评系统集成

```python
class PracticeService:
    def __init__(self, similarity_client: SimilarityClient):
        self.similarity = similarity_client

    async def get_similar_questions(
        self,
        question_id: str,
        user_id: str,
        count: int = 3,
    ) -> list[dict]:
        """获取同类题（用于"练一道同类题"功能）"""
        result = await self.similarity.search(
            question_id=question_id,
            top_k=count * 3,  # 多取一些用于过滤
            min_similarity=0.70,
            max_similarity=0.95,
        )

        # 过滤用户已做过的题目
        done_ids = await self.get_user_done_question_ids(user_id)
        candidates = [
            r for r in result['results']
            if r['question_id'] not in done_ids
        ]

        # 多样性采样：确保不完全集中在同一知识点
        selected = self._diversity_sample(candidates, count)

        return selected

    def _diversity_sample(self, candidates: list, count: int) -> list:
        """多样性采样：避免连续推荐相同知识点的题目"""
        selected = []
        used_kps = set()

        for c in candidates:
            if len(selected) >= count:
                break
            kps = set(c.get('knowledge_points', []))
            overlap = len(kps & used_kps)
            # 如果知识点重叠度太高（>80%），跳过
            if kps and overlap / len(kps) > 0.8 and len(selected) > 0:
                continue
            selected.append(c)
            used_kps.update(kps)

        return selected
```

### 12.3 AI题目生成引擎集成

```python
class AIGenerationService:
    def __init__(self, similarity_client: SimilarityClient):
        self.similarity = similarity_client

    async def generate_questions(self, params: dict) -> list[dict]:
        """AI生成题目 — 集成去重"""
        # 1. AI生成
        generated = await self._call_ai_generate(params)

        results = []
        for q in generated:
            # 2. 每道生成题进行去重检查
            dedup = await self.similarity.dedup_check(q.id)

            if dedup['is_duplicate']:
                q['status'] = 'DUPLICATE_REJECTED'
                q['duplicate_of'] = dedup['duplicates'][0]['question_id']
                q['similarity_score'] = dedup['max_similarity']
            else:
                q['status'] = 'PASSED_DEDUP'

            results.append(q)

        return results
```

## 14. 数据安全与隐私

- 题目内容向量存储在 Milvus，不存储原始题目文本（仅存ID引用）
- 相似度记录中的题目ID关联需通过题目服务获取详情，遵循题目服务的权限控制
- 管理后台审核接口需 `CONTENT_ADMIN` 角色
- 向量重建等批量操作需记录审计日志
- Embedding调用通过模型调度层统一管理API密钥
