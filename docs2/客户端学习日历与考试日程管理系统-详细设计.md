# 客户端学习日历与考试日程管理系统 - 详细设计

## 1. 概述

### 1.1 功能定位

学习日历与考试日程管理系统是 PrimeTop 客户端的**时间维度学习管理中心**，为学生、家长提供可视化的学习时间线和考试日程管理能力。本系统将分散在各个模块中的时间相关信息（学习记录、计划任务、考试日期、校历事件、打卡数据）聚合为统一的日历视图，帮助用户直观了解学习节奏、规划备考策略、保持学习习惯。

### 1.2 核心价值

| 用户 | 价值 |
|------|------|
| 学生 | 直观查看每日学习情况，管理考试日程，保持学习节奏 |
| 家长 | 了解孩子学习频率和强度，掌握考试时间节点 |
| 系统 | 为推荐引擎提供时间维度特征，驱动学习提醒和计划调整 |

### 1.3 与现有模块的边界

| 模块 | 职责边界 | 与本系统的关系 |
|------|----------|----------------|
| 学期与学年管理服务 | 管理学校级学期、校历事件（开学/放假/考试周） | 本系统消费其数据，叠加用户个人事件 |
| 学习规划 | 生成学习计划与任务 | 本系统展示计划时间线，反向提供日历冲突检测 |
| 学习记录与进度追踪 | 采集原始学习行为 | 本系统聚合展示为日历热力图 |
| 学习提醒与智能通知 | 推送提醒消息 | 本系统为其提供考试日程等触发源 |
| 学习习惯养成与打卡 | 打卡数据和连续天数 | 本系统展示打卡日历视图 |
| 学习数据可视化 | 通用图表组件 | 本系统使用其 HeatmapChart 组件 |

### 1.4 设计原则

1. **聚合而非重复**：本系统不产生原始学习数据，仅聚合展示各模块数据
2. **轻量优先**：日历页面需快速加载，大量数据采用预聚合 + 按需加载
3. **分龄适配**：幼儿/小学侧重趣味性，初高中侧重效率和考试导向
4. **离线可用**：核心日历数据本地缓存，弱网环境下可查看基本视图

def calc_intensity(study_duration: int) -> str:
    """根据学习时长计算强度等级"""
    minutes = study_duration / 60
    if minutes == 0:
        return 'none'
    elif minutes < 15:
        return 'light'       # 轻度：不到15分钟
    elif minutes < 30:
        return 'low'         # 较低：15-30分钟
    elif minutes < 60:
        return 'medium'      # 中等：30-60分钟
    elif minutes < 120:
        return 'high'        # 较高：1-2小时
    else:
        return 'intensive'   # 高强度：2小时以上

## 5. 核心算法与逻辑

### 5.1 学习连续天数计算

```python
def calculate_streak(user_id: int, reference_date: date = None) -> dict:
    """
    计算学习连续天数。
    
    规则：
    1. 当天有学习记录（≥5分钟）计为1天
    2. 从reference_date往前连续计算
    3. 如果今天还没学习，从昨天开始算
    4. 允许1天"豁免"（用于生病等特殊情况，需家长授权）
    """
    if reference_date is None:
        reference_date = date.today()
    
    # 从Redis获取最近60天热力图数据
    heatmap = get_study_heatmap(user_id, reference_date, days=60)
    
    current_streak = 0
    longest_streak = 0
    temp_streak = 0
    check_date = reference_date
    
    # 如果今天还没学习，从昨天开始算
    if heatmap.get(reference_date, 0) < 300:  # 小于5分钟不算
        check_date = reference_date - timedelta(days=1)
    
    # 计算当前连续天数
    while check_date in heatmap and heatmap[check_date] >= 300:
        current_streak += 1
        check_date -= timedelta(days=1)
    
    # 计算最长连续天数
    for d, duration in sorted(heatmap.items()):
        if duration >= 300:
            temp_streak += 1
            longest_streak = max(longest_streak, temp_streak)
        else:
            temp_streak = 0
    
    return {
        "current_streak": current_streak,
        "longest_streak": longest_streak,
        "total_study_days": sum(1 for d in heatmap.values() if d >= 300),
    }
```

### 5.2 日历冲突检测

```python
def detect_calendar_conflicts(
    user_id: int,
    exam_date: date,
    subject_code: str = None
) -> list[dict]:
    """
    检测考试日程冲突。
    
    冲突类型：
    1. 同日已有考试
    2. 同周密集考试（≥3场）
    3. 与校历事件冲突（如节假日）
    4. 与学习计划里程碑冲突
    """
    conflicts = []
    
    # 1. 同日考试冲突
    same_day_exams = query_exams(user_id, exam_date=exam_date, status='active')
    if same_day_exams:
        conflicts.append({
            "type": "same_day_exam",
            "level": "warning",
            "message": f"当天已有{len(same_day_exams)}场考试",
            "details": [{"id": e.id, "title": e.title} for e in same_day_exams]
        })
    
    # 2. 同周密集考试
    week_start = exam_date - timedelta(days=exam_date.weekday())
    week_end = week_start + timedelta(days=6)
    week_exams = query_exams(user_id, from_date=week_start, to_date=week_end, status='active')
    if len(week_exams) >= 3:
        conflicts.append({
            "type": "dense_week",
            "level": "info",
            "message": f"本周共有{len(week_exams)}场考试，注意合理安排复习",
            "details": [{"id": e.id, "title": e.title, "date": str(e.exam_date)} for e in week_exams]
        })
    
    # 3. 校历事件冲突
    school_events = query_calendar_events(exam_date)
    for event in school_events:
        if event.event_type in ('holiday', 'sports_day'):
            conflicts.append({
                "type": "school_event",
                "level": "warning",
                "message": f"当天为校历事件：{event.name}",
                "suggest": "请确认考试日期是否正确"
            })
    
    return conflicts

    - 默认提前 7天、3天、1天 提醒

    """
    
    def __init__(self, db, redis, learning_client, checkin_client, 
                 plan_client, academic_client):
        self.db = db
        self.redis = redis
        self.learning = learning_client
        self.checkin = checkin_client
        self.plan = plan_client
        self.academic = academic_client
    
    async def get_monthly_data(
        self, user_id: int, month: str, include_events: bool = False
    ) -> dict:
        """获取月度日历聚合数据"""
        
        # 1. 尝试从Redis缓存获取
        cache_key = f"calendar:month:{user_id}:{month}"
        cached = await self.redis.hgetall(cache_key)
        if cached and not include_events:
            return json.loads(cached.get("data", "{}"))
        
        # 2. 解析月份范围
        year, m = map(int, month.split("-"))
        start_date = date(year, m, 1)
        end_date = date(year, m, calendar.monthrange(year, m)[1])
        
        # 3. 并行聚合各模块数据
        results = await asyncio.gather(
            self._get_study_daily_stats(user_id, start_date, end_date),
            self._get_checkin_status(user_id, start_date, end_date),
            self._get_exam_count(user_id, start_date, end_date),
            self._get_plan_stats(user_id, start_date, end_date),
            self._get_streak(user_id, start_date),
        )
        
        study_stats = results[0]   # Dict[date, StudyDayStats]
        checkins = results[1]      # Set[date]
        exam_counts = results[2]   # Dict[date, int]
        plan_stats = results[3]    # Dict[date, PlanDayStats]
        streak_info = results[4]   # StreakInfo
        
        # 4. 合并为日历数据
        days = {}
        current = start_date
        while current <= end_date:
            day_str = current.isoformat()
            study = study_stats.get(current)
            
            days[day_str] = {
                "studyDuration": study.duration if study else 0,
                "questionCount": study.question_count if study else 0,
                "correctRate": study.correct_rate if study else 0,
                "checkedIn": current in checkins,
                "examCount": exam_counts.get(current, 0),
                "planCompleted": plan_stats.get(current, PlanDayStats()).completed,
                "planTotal": plan_stats.get(current, PlanDayStats()).total,
                "subjects": study.subjects if study else [],
                "intensity": calc_intensity(study.duration if study else 0),
            }
            current += timedelta(days=1)
        
        # 5. 生成月度摘要
        total_study_days = sum(
            1 for d in days.values() if d["studyDuration"] >= 300
        )
        total_duration = sum(d["studyDuration"] for d in days.values())
        total_questions = sum(d["questionCount"] for d in days.values())
        rates = [d["correctRate"] for d in days.values() if d["correctRate"] > 0]
        
        summary = {
            "totalStudyDays": total_study_days,
            "totalStudyDuration": total_duration,
            "totalQuestions": total_questions,
            "avgCorrectRate": sum(rates) / len(rates) if rates else 0,
            "checkinDays": sum(1 for d in days.values() if d["checkedIn"]),
            "currentStreak": streak_info.current_streak,
            "longestStreak": streak_info.longest_streak,
        }
        
        # 6. 缓存（5分钟TTL）
        result = {"month": month, "days": days, "summary": summary}
        await self.redis.hset(cache_key, "data", json.dumps(result))
        await self.redis.expire(cache_key, 300)
        
        return result
    
    async def _get_study_daily_stats(
        self, user_id: int, start: date, end: date
    ) -> dict[date, StudyDayStats]:
        """调用学习记录服务获取每日学习统计"""
        return await self.learning.get_daily_stats(
            user_id=user_id, start_date=start, end_date=end
        )
    
    async def _get_checkin_status(
        self, user_id: int, start: date, end: date
    ) -> set[date]:
        """调用打卡系统获取打卡日期集"""
        return await self.checkin.get_checkin_dates(
            user_id=user_id, start_date=start, end_date=end
        )
    
    async def _get_exam_count(
        self, user_id: int, start: date, end: date
    ) -> dict[date, int]:
        """获取每日考试数量"""
        exams = await self.db.query(
            """SELECT exam_date, COUNT(*) as cnt 
               FROM user_exam_schedules 
               WHERE user_id = :uid 
               AND exam_date BETWEEN :start AND :end 
               AND status = 'active' 
               GROUP BY exam_date""",
            {"uid": user_id, "start": start, "end": end}
        )
        return {row.exam_date: row.cnt for row in exams}
```

## 10. 错误处理

### 10.1 错误码体系

| 错误码 | HTTP状态码 | 说明 | 客户端处理 |
|--------|-----------|------|------------|
| 70001 | 400 | 月份格式错误 | 提示用户 |
| 70002 | 400 | 考试日期无效（如过去日期） | 提示修改日期 |
| 70003 | 404 | 考试日程不存在 | 刷新列表 |
| 70004 | 409 | 同日同科考试冲突 | 展示冲突详情 |
| 70005 | 403 | 无权限（非本人的考试） | 返回列表 |
| 70006 | 429 | 日历请求频率超限 | 使用缓存数据 |
| 70007 | 400 | 考试数量超出限制（最多50条active） | 提示整理 |

### 10.2 降级策略

| 场景 | 策略 | 用户体验 |
|------|------|----------|
| 服务端不可用 | 使用本地Hive缓存数据 | 显示"上次同步时间"标记 |
| 学习记录服务超时 | 仅展示考试和打卡数据 | 学习时长区域灰显 |
| Redis缓存未命中 | 直接查询数据库 | 增加loading时间 |
| 年度热力图数据量大 | 仅返回最近180天 | 提示"查看更早数据"
