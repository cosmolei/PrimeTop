# 客户端-编程练习页面架构与代码编辑器交互组件 详细设计

## 1. 概述

### 1.1 模块定位

本模块是 PrimeTop 信息技术学科在**客户端侧**的核心承载页面与交互组件体系。服务端已有「信息技术学科智能辅导与编程实践自动评测引擎（IT-Engine）」和「教育AI代码执行沙箱（ECS）」两份详细设计，本设计作为其客户端 counterpart，填补信息技术学科在学生端的页面架构与代码编辑器交互空白。

核心职责：

| 职责 | 说明 |
| --- | --- |
| 编程练习答题页 | 学生在 App 内直接编写代码、提交评测、查看结果的完整流程 |
| 代码编辑器组件 | 支持多语言语法高亮、自动缩进、括号匹配、代码补全的轻量编辑器 |
| 评测结果展示 | 测试用例通过/失败状态、执行耗时/内存、AI 反馈的结构化呈现 |
| 算法可视化播放器 | 将排序、递归等算法的执行过程以动画步骤形式逐步播放 |
| Scratch 图形化编程 | 面向小学低年级的拖拽式积木编程界面（WebView 封装） |
| 编程学习路径页 | 按知识点组织的编程练习目录与进度追踪 |

### 1.3 设计原则

1. **移动优先**：编辑器针对触屏 + 软键盘场景深度优化，桌面 Web 端兼容但非首要目标
2. **渐进复杂度**：小学 Scratch → 初中 Python → 高中 C++，交互复杂度逐级提升
3. **离线可用**：题目描述、模板代码、历史提交可离线浏览；代码编写不需要网络
4. **安全隔离**：Scratch 编程通过 WebView 沙箱隔离；Python/C++ 代码不在端侧执行
5. **复用优先**：题目描述复用 Markdown 渲染引擎，AI 辅导复用对话组件，不重复造轮子

### 3.3 CodeBuffer — 文本模型

基于行列表的文本缓冲区，支持高效的插入/删除和撤销/重做。

```dart
/// 代码缓冲区
class CodeBuffer {
  final List<String> _lines = [''];
  final List<EditOperation> _undoStack = [];
  final List<EditOperation> _redoStack = [];
  static const int maxUndoSteps = 100;

  String get text => _lines.join('\n');
  int get lineCount => _lines.length;
  String lineAt(int index) =>
      (index >= 0 && index < _lines.length) ? _lines[index] : '';

  /// 在指定行列位置插入文本
  void insert({required int line, required int col, required String text}) {
    final op = EditOperation(
      type: EditType.insert,
      lineIndex: line,
      charIndex: col,
      text: text,
    );
    _applyInsert(op);
    _pushUndo(op);
  }

  /// 删除指定范围文本
  void delete({required int startLine, required int startCol,
      required int endLine, required int endCol}) {
    final deleted = _extractRange(startLine, startCol, endLine, endCol);
    final op = EditOperation(
      type: EditType.delete,
      lineIndex: startLine,
      charIndex: startCol,
      text: deleted,
    );
    _applyDelete(op);
    _pushUndo(op);
  }

  /// 替换整行
  void replaceLine(int lineIndex, String newContent) {
    final old = lineAt(lineIndex);
    delete(startLine: lineIndex, startCol: 0,
           endLine: lineIndex, endCol: old.length);
    insert(line: lineIndex, col: 0, text: newContent);
  }

  /// 撤销
  bool undo() {
    if (_undoStack.isEmpty) return false;
    final op = _undoStack.removeLast();
    if (op.type == EditType.insert) {
      _applyDelete(op);
    } else {
      _applyInsert(op);
    }
    _redoStack.add(op);
    return true;
  }

  /// 重做
  bool redo() {
    if (_redoStack.isEmpty) return false;
    final op = _redoStack.removeLast();
    if (op.type == EditType.insert) {
      _applyInsert(op);
    } else {
      _applyDelete(op);
    }
    _undoStack.add(op);
    return true;
  }

  void _applyInsert(EditOperation op) {
    final line = _lines[op.lineIndex];
    final newText = line.substring(0, op.charIndex) + op.text +
        line.substring(op.charIndex);
    final parts = newText.split('\n');
    _lines[op.lineIndex] = parts[0];
    for (int i = 1; i < parts.length; i++) {
      _lines.insert(op.lineIndex + i, parts[i]);
    }
  }

  void _applyDelete(EditOperation op) {
    // 根据 op.text 还原文本（删除操作的逆操作是插入）
    _applyInsert(op);
  }

  void _pushUndo(EditOperation op) {
    _undoStack.add(op);
    if (_undoStack.length > maxUndoSteps) _undoStack.removeAt(0);
    _redoStack.clear();
  }

  String _extractRange(int sLine, int sCol, int eLine, int eCol) {
    if (sLine == eLine) {
      return _lines[sLine].substring(sCol, eCol);
    }
    final buf = StringBuffer(_lines[sLine].substring(sCol));
    for (int i = sLine + 1; i < eLine; i++) {
      buf.write('\n');
      buf.write(_lines[i]);
    }
    buf.write('\n');
    buf.write(_lines[eLine].substring(0, eCol));
    return buf.toString();
  }

  factory CodeBuffer.fromString(String text) {
    final buf = CodeBuffer();
    buf._lines.clear();
    buf._lines.addAll(text.isEmpty ? [''] : text.split('\n'));
    return buf;
  }
}

@immutable
class EditOperation {
  final EditType type;
  final int lineIndex;
  final int charIndex;
  final String text;
  const EditOperation({
    required this.type,
    required this.lineIndex,
    required this.charIndex,
    required this.text,
  });
}

enum EditType { insert, delete }
```

### 3.4 语法高亮引擎

基于正则规则的轻量高亮器，支持 Python3、C++、Java。

```dart
abstract class SyntaxHighlighter {
  List<HighlightSpan> highlight(String line, {HighlightContext? context});
  String get languageId;
}

@immutable
class HighlightSpan {
  final int start;
  final int end;
  final HighlightType type;
  const HighlightSpan(this.start, this.end, this.type);
}

enum HighlightType {
  keyword,      // if/else/for/def/class
  string,       // "..." / '...'
  number,       // 42 / 3.14
  comment,      // # ... / // ...
  function,     // 函数名
  builtin,      // print/len/range
  operator,     // + - * / =
  punctuation,  // ( ) [ ] { } ,
  constant,     // True/False/None
  variable,     // 普通标识符
}

/// Python 高亮器实现
class PythonHighlighter implements SyntaxHighlighter {
  @override
  String get languageId => 'python3';

  static const _keywords = {
    'def','class','if','elif','else','for','while','try','except','finally',
    'with','as','import','from','return','yield','raise','pass','break',
    'continue','global','nonlocal','lambda','assert','del','in','is','not',
    'and','or','async','await',
  };

  static const _builtins = {
    'print','len','range','enumerate','zip','map','filter','sorted','reversed',
    'sum','min','max','abs','round','type','isinstance','input','open',
    'list','dict','set','tuple','str','int','float','bool','format',
  };

  static const _constants = {'True','False','None'};

  static final _regex = RegExp(
    r'(?<comment>#.*)'
    r'|(?<string>"[^"\n]*"|\'[^\n\']*\'|"""[\s\S]*?"""|\'\'\'[\s\S]*?\'\'\')'
    r'|(?<number>\b\d+\.?\d*(?:[eE][+-]?\d+)?\b|0[xX][0-9a-fA-F]+\b)'
    r'|(?<ident>[A-Za-z_]\w*)'
    r'|(?<op>[+\-*/%=<>!&|^~]{1,2})'
    r'|(?<punct>[(){}\[\],:.;@])',
  );

  @override
  List<HighlightSpan> highlight(String line, {HighlightContext? context}) {
    final spans = <HighlightSpan>[];
    for (final m in _regex.allMatches(line)) {
      spans.add(HighlightSpan(m.start, m.end, _classify(m)));
    }
    return spans;
  }

  HighlightType _classify(Match m) {
    if (m.namedGroup('comment') != null) return HighlightType.comment;
    if (m.namedGroup('string') != null) return HighlightType.string;
    if (m.namedGroup('number') != null) return HighlightType.number;
    final ident = m.namedGroup('ident');
    if (ident != null) {
      if (_keywords.contains(ident)) return HighlightType.keyword;
      if (_constants.contains(ident)) return HighlightType.constant;
      if (_builtins.contains(ident)) return HighlightType.builtin;
      return HighlightType.variable;
    }
    if (m.namedGroup('op') != null) return HighlightType.operator;
    if (m.namedGroup('punct') != null) return HighlightType.punctuation;
    return HighlightType.variable;
  }
}

/// 高亮器工厂
class SyntaxHighlighterFactory {
  static SyntaxHighlighter? create(String languageId) {
    switch (languageId) {
      case 'python3': case 'python': return PythonHighlighter();
      case 'cpp': case 'c++': return CppHighlighter();
      case 'java': return JavaHighlighter();
      default: return null;
    }
  }
}
```

### 3.6 代码补全

针对教育场景的轻量补全，不依赖 LSP，基于关键词表 + 上下文规则。

```dart
class CodeCompletionProvider {
  final String language;

  static const _pythonKeywords = [
    'def', 'class', 'if', 'elif', 'else', 'for', 'while', 'try', 'except',
    'return', 'import', 'from', 'with', 'as', 'lambda', 'pass', 'break',
    'continue', 'global', 'yield', 'raise', 'assert',
  ];

  static const _pythonBuiltins = [
    'print(', 'len(', 'range(', 'enumerate(', 'zip(', 'sorted(',
    'reversed(', 'sum(', 'min(', 'max(', 'abs(', 'round(', 'input(',
    'list(', 'dict(', 'set(', 'tuple(', 'str(', 'int(', 'float(',
    'isinstance(', 'map(', 'filter(', 'open(',
  ];

  List<CompletionSuggestion> getSuggestions(String prefix, String context) {
    if (prefix.isEmpty) return [];

    final pool = language == 'python3'
        ? [..._pythonKeywords, ..._pythonBuiltins]
        : const [];

    return pool
        .where((item) => item.startsWith(prefix))
        .take(8)
        .map((item) => CompletionSuggestion(
              text: item,
              displayText: item,
              type: _pythonBuiltins.contains(item)
                  ? CompletionType.function
                  : CompletionType.keyword,
            ))
        .toList();
  }
}

@immutable
class CompletionSuggestion {
  final String text;       // 插入文本
  final String displayText; // 显示文本
  final CompletionType type;
  const CompletionSuggestion({
    required this.text,
    required this.displayText,
    required this.type,
  });
}
```

```dart
class VisualizationPlayerController extends ChangeNotifier {
  final AlgorithmVisualization visualization;
  int _currentStep = 0;
  bool _isPlaying = false;
  double _speed = 1.0;
  Timer? _timer;

  int get currentStep => _currentStep;
  int get totalSteps => visualization.steps.length;
  bool get isPlaying => _isPlaying;
  double get speed => _speed;
  bool get canStepForward => _currentStep < totalSteps - 1;
  bool get canStepBackward => _currentStep > 0;

  VisualizationStep get currentVisualizationStep =>
      visualization.steps[_currentStep];

  void play() {
    if (_isPlaying) return;
    _isPlaying = true;
    notifyListeners();
    _scheduleNextTick();
  }

  void pause() {
    _isPlaying = false;
    _timer?.cancel();
    notifyListeners();
  }

  void stepForward() {
    pause();
    if (canStepForward) {
      _currentStep++;
      notifyListeners();
    }
  }

  void stepBackward() {
    pause();
    if (canStepBackward) {
      _currentStep--;
      notifyListeners();
    }
  }

  void jumpTo(int step) {
    pause();
    _currentStep = step.clamp(0, totalSteps - 1);
    notifyListeners();
  }

  void setSpeed(double speed) {
    _speed = speed;
    if (_isPlaying) {
      _timer?.cancel();
      _scheduleNextTick();
    }
    notifyListeners();
  }

  void reset() {
    pause();
    _currentStep = 0;
    notifyListeners();
  }

  void _scheduleNextTick() {
    if (!_isPlaying || !canStepForward) {
      pause();
      return;
    }
    final baseDelayMs = 800; // 基础间隔 800ms
    _timer = Timer(
      Duration(milliseconds: (baseDelayMs / _speed).round()),
      () {
        _currentStep++;
        notifyListeners();
        if (canStepForward) {
          _scheduleNextTick();
        } else {
          pause();
        }
      },
    );
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }
}
```

### 6.2 状态机实现

```dart
class SubmissionStateMachine {
  SubmissionStatus _status = SubmissionStatus.idle;
  SubmissionStatus get status => _status;
  String? _errorMessage;
  String? get errorMessage => _errorMessage;

  final void Function(SubmissionStatus)? onStatusChanged;

  /// 触发提交流程
  Future<void> submit({
    required int taskId,
    required String language,
    required String sourceCode,
    required ProgrammingRepository repository,
  }) async {
    // 1. 前端校验
    _transition(SubmissionStatus.validating);
    final validation = _validateCode(sourceCode, language);
    if (!validation.isValid) {
      _errorMessage = validation.error;
      _transition(SubmissionStatus.idle);
      return;
    }

    // 2. 提交到服务端
    _transition(SubmissionStatus.submitting);
    try {
      final response = await repository.submitCode(
        taskId: taskId,
        language: language,
        sourceCode: sourceCode,
      );
      _transition(SubmissionStatus.queued);

      // 3. 轮询评测结果
      await _pollEvaluationResult(
        repository: repository,
        submissionId: response.submissionId,
      );
    } on NetworkException catch (e) {
      _errorMessage = '网络错误：${e.message}，请检查网络后重试';
      _transition(SubmissionStatus.networkError);
      _transition(SubmissionStatus.idle);
    }
  }

  /// 前端代码校验
  _ValidationResult _validateCode(String code, String language) {
    if (code.trim().isEmpty) {
      return _ValidationResult(isValid: false, error: '代码不能为空');
    }
    if (code.length > 50000) {
      return _ValidationResult(isValid: false, error: '代码超过最大长度限制(50000字符)');
    }
    if (language == 'python3') {
      // 检查危险的模块导入
      final dangerousPatterns = [
        RegExp(r'\bimport\s+os\b'),
        RegExp(r'\bimport\s+subprocess\b'),
        RegExp(r'\bimport\s+socket\b'),
        RegExp(r'\b__import__\s*\('),
      ];
      for (final pattern in dangerousPatterns) {
        if (pattern.hasMatch(code)) {
          return _ValidationResult(
            isValid: false,
            error: '代码包含不允许的模块导入',
          );
        }
      }
    }
    return _ValidationResult(isValid: true);
  }

  /// 轮询评测结果（指数退避）
  Future<void> _pollEvaluationResult({
    required ProgrammingRepository repository,
    required String submissionId,
  }) async {
    _transition(SubmissionStatus.evaluating);

    const maxAttempts = 30;
    const initialDelay = Duration(seconds: 1);
    const maxDelay = Duration(seconds: 5);

    var delay = initialDelay;
    for (int attempt = 0; attempt < maxAttempts; attempt++) {
      await Future.delayed(delay);

      final result = await repository.getEvaluationResult(submissionId);
      if (result.status == 'success' || result.status == 'failed') {
        // 评测完成，根据结果切换状态
        if (result.passedCount == result.totalCount) {
          _transition(SubmissionStatus.success);
        } else if (result.passedCount > 0) {
          _transition(SubmissionStatus.partialPass);
        } else {
          // 根据具体失败原因细分
          final firstFailure = result.cases.firstWhere(
            (c) => c.status != 'passed',
            orElse: () => result.cases.first,
          );
          _transition(_mapFailureStatus(firstFailure.status));
        }
        return;
      }

      // 指数退避
      delay = Duration(
        milliseconds: (delay.inMilliseconds * 1.5).round().clamp(
          initialDelay.inMilliseconds,
          maxDelay.inMilliseconds,
        ),
      );
    }

    // 超时未获得结果
    _errorMessage = '评测超时，请稍后在提交历史中查看结果';
    _transition(SubmissionStatus.idle);
  }

  SubmissionStatus _mapFailureStatus(String serverStatus) {
    switch (serverStatus) {
      case 'wrong_answer': return SubmissionStatus.failed;
      case 'time_limit_exceeded': return SubmissionStatus.timeLimitExceeded;
      case 'memory_limit_exceeded': return SubmissionStatus.memoryLimitExceeded;
      case 'runtime_error': return SubmissionStatus.runtimeError;
      case 'compilation_error': return SubmissionStatus.compileError;
      default: return SubmissionStatus.failed;
    }
  }

  void _transition(SubmissionStatus newStatus) {
    _status = newStatus;
    onStatusChanged?.call(newStatus);
  }
}

@immutable
class _ValidationResult {
  final bool isValid;
  final String? error;
  const _ValidationResult({required this.isValid, this.error});
}
```
