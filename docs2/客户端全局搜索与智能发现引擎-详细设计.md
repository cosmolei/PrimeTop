## 1. 模块概述

### 1.1 功能目标

为学生提供统一的全局搜索入口，支持跨模块内容检索（题目、知识点、教材章节、错题、笔记、学科工具、课程资源等），结合本地离线索引与远程搜索服务的双层架构，实现快速、智能、个性化的内容发现体验。

### 1.2 设计原则

| 原则 | 说明 |
|------|------|
| **统一入口** | 全局搜索入口贯穿 APP，任何页面均可触发搜索 |
| **多源融合** | 本地索引 + 远程服务双层检索，自动选择最优源 |
| **即时响应** | 搜索建议 100ms 内响应，搜索结果 500ms 内首屏展示 |
| **分龄适配** | 搜索交互和结果展示根据学段自动调整 |
| **离线可用** | 本地索引保底，无网络时仍可搜索已缓存内容 |
| **隐私安全** | 搜索历史仅本地存储，不上报原始搜索内容 |

### 1.3 搜索范围

| 内容类型 | 数据源 | 离线可用 | 说明 |
|----------|--------|----------|------|
| 题目 | 本地 + 远程 | ✅ | 题库、已做题目、拍题历史 |
| 知识点 | 本地 + 远程 | ✅ | 知识点名称、解释、关联章节 |
| 教材章节 | 本地 + 远程 | ✅ | 教材目录、章节名称 |
| 错题 | 本地 | ✅ | 个人错题本 |
| 笔记 | 本地 | ✅ | 个人笔记 |
| 学科工具 | 本地 | ✅ | 公式表、常数表、学科参考资料 |
| 课程资源 | 远程 | ❌ | 视频课、专题课、学习路径 |
| AI 对话历史 | 本地 | ✅ | 历史对话记录 |
| 考点 | 远程 | ❌ | 考点清单、高频考点 |
| 教师资源 | 远程 | ❌ | 教师布置的作业和资源 |

### 1.4 关键指标

| 指标 | 目标值 | 说明 |
|------|--------|------|
| 搜索建议响应时间 | < 100ms | 本地索引查询 |
| 搜索结果首屏时间 | < 500ms | 包含本地+远程聚合 |
| 离线搜索可用率 | 100% | 本地索引覆盖 |
| 搜索结果点击率 | > 30% | 搜索结果被点击的比率 |
| 搜索无结果率 | < 10% | 搜索后无任何结果的比例 |

```dart
/// 搜索历史记录
class SearchHistoryEntry {
  final String id;
  final String query;                // 搜索关键词
  final DateTime searchedAt;         // 搜索时间
  final int resultCount;             // 结果数量
  final List<SearchContentType> clickedTypes; // 点击过的内容类型
  final String? clickedItemId;       // 最后点击的结果 ID

  const SearchHistoryEntry({
    required this.id,
    required this.query,
    required this.searchedAt,
    this.resultCount = 0,
    this.clickedTypes = const [],
    this.clickedItemId,
  });

  /// 转换为 Hive 存储格式
  Map<String, dynamic> toJson() => {
    'id': id,
    'query': query,
    'searchedAt': searchedAt.toIso8601String(),
    'resultCount': resultCount,
    'clickedTypes': clickedTypes.map((t) => t.name).toList(),
    'clickedItemId': clickedItemId,
  };

  factory SearchHistoryEntry.fromJson(Map<String, dynamic> json) =>
    SearchHistoryEntry(
      id: json['id'],
      query: json['query'],
      searchedAt: DateTime.parse(json['searchedAt']),
      resultCount: json['resultCount'] ?? 0,
      clickedTypes: (json['clickedTypes'] as List?)
        ?.map((t) => SearchContentType.values.byName(t))
        .toList() ?? [],
      clickedItemId: json['clickedItemId'],
    );
```

### 4.2 建议引擎实现

```dart
/// 搜索建议引擎
class SearchSuggestionEngine {
  final SearchHistoryService _historyService;
  final LocalSearchService _localSearch;
  final SearchHotQueriesCache _hotCache;

  /// 获取搜索建议（综合多源）
  Future<List<SearchSuggestion>> getSuggestions({
    required String query,
    required AgeGroup ageGroup,
    int limit = 10,
  }) async {
    if (query.isEmpty) {
      // 空查询 → 返回历史 + 热门
      return _getEmptyQuerySuggestions(ageGroup, limit);
    }

    final results = <SearchSuggestion>[];

    // 1. 搜索历史（精确前缀匹配，最多 3 条）
    final history = await _historyService.search(query, limit: 3);
    results.addAll(history.map((h) => SearchSuggestion(
      text: h.query,
      type: SearchSuggestionType.history,
      frequency: h.resultCount,
    )));

    // 2. 本地索引自动补全（最多 5 条）
    if (results.length < limit) {
      final completions = await _localSearch.autocomplete(
        query,
        limit: min(5, limit - results.length),
      );
      results.addAll(completions.map((c) => SearchSuggestion(
        text: c.text,
        type: SearchSuggestionType.autocomplete,
        contentType: c.contentType,
        highlight: _highlightMatch(c.text, query),
      )));
    }

    // 3. 最近浏览的匹配内容（最多 2 条）
    if (results.length < limit) {
      final recent = await _localSearch.searchRecentViews(
        query,
        limit: min(2, limit - results.length),
      );
      results.addAll(recent.map((r) => SearchSuggestion(
        text: r.title,
        type: SearchSuggestionType.recentView,
        contentType: r.contentType,
      )));
    }

    return results;
  }

  /// 空查询时的建议（展示历史 + 热门 + 发现）
  Future<List<SearchSuggestion>> _getEmptyQuerySuggestions(
    AgeGroup ageGroup, int limit,
  ) async {
    final results = <SearchSuggestion>[];

    // 最近搜索（最多 5 条）
    final recent = await _historyService.getRecent(limit: 5);
    results.addAll(recent.map((h) => SearchSuggestion(
      text: h.query,
      type: SearchSuggestionType.history,
    )));

    // 热门搜索（最多 5 条）
    final hot = await _hotCache.getHotQueries(
      grade: _currentGrade,
      limit: min(5, limit - results.length),
    );
    results.addAll(hot.map((h) => SearchSuggestion(
      text: h.query,
      type: SearchSuggestionType.hotQuery,
      frequency: h.frequency,
    )));

    return results.take(limit).toList();
  }

  String _highlightMatch(String text, String query) {
    final index = text.toLowerCase().indexOf(query.toLowerCase());
    if (index == -1) return text;
    return '<b>${text.substring(index, index + query.length)}</b>${text.substring(index + query.length)}';
  }
}
```

```dart
  final LocalSearchService _localSearch;
  final RemoteSearchService _remoteSearch;
  final SearchDeduplicator _deduplicator;
  final SearchRanker _ranker;

  /// 执行搜索（核心方法）
  Future<SearchResponse> search(SearchRequest request) async {
    final searchId = _generateSearchId();
    final stopwatch = Stopwatch()..start();

    // 1. 本地搜索（始终执行）
    final localFuture = _localSearch.search(request);
    
    // 2. 远程搜索（条件执行）
    final remoteFuture = _shouldSearchRemote(request)
      ? _remoteSearch.search(request).timeout(
          const Duration(milliseconds: 500),
          onTimeout: () => RemoteSearchResult.empty(),
        )
      : Future.value(RemoteSearchResult.empty());

    // 3. 并行执行，本地先返回
    List<SearchItem> localItems = [];
    List<SearchItem> remoteItems = [];
    bool localOk = false;
    bool remoteOk = false;

    // 等待本地结果（快速）
    try {
      final localResult = await localFuture;
      localItems = localResult.items;
      localOk = true;
    } catch (e) {
      _logger.warning('本地搜索失败: $e');
    }

    // 异步等待远程结果
    remoteFuture.then((result) {
      remoteItems = result.items;
      remoteOk = true;
    }).catchError((e) {
      _logger.warning('远程搜索失败: $e');
    });

    // 等待远程（最多 500ms）
    await remoteFuture.timeout(
      const Duration(milliseconds: 500),
      onTimeout: () => RemoteSearchResult.empty(),
    );

    stopwatch.stop();

    // 4. 合并去重
    final merged = _deduplicator.deduplicate(
      localItems: localItems,
      remoteItems: remoteItems,
    );

    // 5. 排序
    final ranked = _ranker.rank(merged, query: request.query);

    // 6. 分面计数
    final facets = _computeFacets(ranked);

    // 7. 构建响应
    return SearchResponse(
      query: request.query,
      items: ranked.take(request.pageSize).toList(),
      facets: facets,
      totalCount: ranked.length,
      hasMore: ranked.length > request.pageSize,
      primarySource: localOk && remoteOk
        ? SearchSource.merged
        : localOk ? SearchSource.local : SearchSource.remote,
      elapsed: stopwatch.elapsed,
      metadata: SearchMetadata(
        localSucceeded: localOk,
        remoteSucceeded: remoteOk,
        localResultCount: localItems.length,
        remoteResultCount: remoteItems.length,
        dedupCount: (localItems.length + remoteItems.length) - ranked.length,
        searchId: searchId,
      ),
    );
  }

  bool _shouldSearchRemote(SearchRequest request) {
    // 离线模式或即时搜索模式不查远程
    if (request.offlineOnly || request.mode == SearchMode.instant) {
      return false;
    }
    // 检查网络状态
    return _connectivityService.isOnline;
  }

  String _generateSearchId() {
    return 'srch_${DateTime.now().millisecondsSinceEpoch}_${Random().nextInt(9999)}';
  }
}
```

### 5.2 去重策略

```dart
/// 搜索结果去重器
class SearchDeduplicator {
  /// 合并本地和远程结果，去重
  List<SearchItem> deduplicate({
    required List<SearchItem> localItems,
    required List<SearchItem> remoteItems,
  }) {
    final seen = <String, SearchItem>{};

    // 先加入本地结果
    for (final item in localItems) {
      final key = _dedupeKey(item);
      seen[key] = item;
    }

    // 远程结果覆盖本地（远程数据更丰富）
    for (final item in remoteItems) {
      final key = _dedupeKey(item);
      if (seen.containsKey(key)) {
        // 合并：保留远程的高亮信息，保留本地的 source 标记
        seen[key] = _mergeItems(seen[key]!, item);
      } else {
        seen[key] = item;
      }
    }

    return seen.values.toList();
  }

  /// 去重键：类型 + ID
  String _dedupeKey(SearchItem item) => '${item.contentType.name}:${item.id}';

  /// 合并本地和远程结果
  SearchItem _mergeItems(SearchItem local, SearchItem remote) {
    return SearchItem(
      id: remote.id,
      contentType: remote.contentType,
      title: remote.highlightedTitle.isNotEmpty ? remote.title : local.title,
      highlightedTitle: remote.highlightedTitle.isNotEmpty
        ? remote.highlightedTitle
        : local.highlightedTitle,
      subtitle: remote.subtitle ?? local.subtitle,
      highlightedSubtitle: remote.highlightedSubtitle.isNotEmpty
        ? remote.highlightedSubtitle
        : local.highlightedSubtitle,
      snippet: remote.snippet ?? local.snippet,
      highlightedSnippet: remote.highlightedSnippet.isNotEmpty
        ? remote.highlightedSnippet
        : local.highlightedSnippet,
      coverUrl: remote.coverUrl ?? local.coverUrl,
      source: SearchSource.merged,
      relevanceScore: max(local.relevanceScore, remote.relevanceScore),
      metadata: {...local.metadata, ...remote.metadata},
      quickAction: remote.quickAction ?? local.quickAction,
    );
  }
}
```

### 5.3 排序策略

```dart
/// 搜索结果排序器
class SearchRanker {
  /// 综合排序
  List<SearchItem> rank(List<SearchItem> items, {required String query}) {
    final scored = items.map((item) {
      final score = _computeScore(item, query);
      return MapEntry(item, score);
    }).toList();

    // 按评分降序
    scored.sort((a, b) => b.value.compareTo(a.value));

    // 类型多样性保证：连续 3 条不能是同一类型
    return _ensureDiversity(scored.map((e) => e.key).toList());
  }

  double _computeScore(SearchItem item, String query) {
    double score = 0.0;

    // 1. 基础相关度（由搜索引擎给出）
    score += item.relevanceScore * 0.4;

    // 2. 标题匹配度
    final titleMatch = _textMatchScore(item.title, query);
    score += titleMatch * 0.3;

    // 3. 内容类型权重
    score += _typeWeight(item.contentType) * 0.1;

    // 4. 数据新鲜度
    score += _freshnessScore(item) * 0.1;

    // 5. 个人偏好权重（基于历史点击）
    score += _personalizationScore(item.contentType) * 0.1;

    return score.clamp(0.0, 1.0);
  }

  /// 类型多样性保证
  List<SearchItem> _ensureDiversity(List<SearchItem> items) {
    final result = <SearchItem>[];
    var consecutiveCount = 0;
    SearchContentType? lastType;

    for (final item in items) {
      if (item.contentType == lastType) {
        consecutiveCount++;
        if (consecutiveCount >= 3) {
          // 尝试找下一个不同类型的插入
          final swapIdx = items.indexWhere(
            (i) => i.contentType != lastType && !result.contains(i),
            items.indexOf(item) + 1,
          );
          if (swapIdx != -1) {
            result.add(items[swapIdx]);
            continue;
          }
        }
      } else {
        consecutiveCount = 0;
      }
      result.add(item);
      lastType = item.contentType;
    }

    return result;
  }

  double _typeWeight(SearchContentType type) => switch (type) {
    SearchContentType.knowledgePoint => 1.0,
    SearchContentType.question => 0.9,
    SearchContentType.examPoint => 0.85,
    SearchContentType.chapter => 0.8,
    SearchContentType.mistake => 0.7,
    SearchContentType.note => 0.6,
    SearchContentType.course => 0.5,
    SearchContentType.tool => 0.4,
    SearchContentType.aiConversation => 0.3,
    SearchContentType.teacherResource => 0.2,
  };

  double _textMatchScore(String text, String query) {
    if (text.toLowerCase().contains(query.toLowerCase())) return 1.0;
    // 模糊匹配：检查查询中的每个字是否在文本中
    final chars = query.split('');
    final matched = chars.where((c) => text.contains(c)).length;
    return matched / chars.length;
  }

  double _freshnessScore(SearchItem item) {
    // 基于元数据中的创建时间
    final createdAt = item.metadata['createdAt'] as String?;
    if (createdAt == null) return 0.5;
    final age = DateTime.now().difference(DateTime.parse(createdAt));
    if (age.inDays < 7) return 1.0;
    if (age.inDays < 30) return 0.8;
    if (age.inDays < 90) return 0.6;
    return 0.4;
  }

  double _personalizationScore(SearchContentType type) {
    // 基于 UserDefaults 中存储的点击偏好
    // 这里返回默认值，实际实现会查询偏好数据
    return 0.5;
  }
}
```
