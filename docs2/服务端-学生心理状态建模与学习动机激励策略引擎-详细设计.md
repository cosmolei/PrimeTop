## 1. 模块概述

### 1.1 定位

学生心理状态建模与学习动机激励策略引擎（以下简称"心理状态引擎"或 PSM Engine）是 PrimeTop 学习闭环中连接**学习行为分析**与**个性化教学策略**的核心中间服务。

与现有模块的分工差异：

| 模块 | 关注粒度 | 核心职责 |
|------|----------|----------|
| AI辅导对话情感感知引擎 | **单次对话级** | 识别当前对话中学生的情绪（困惑/焦虑/兴奋） |
| 学习动力量化与学习倦怠预警引擎 | **统计指标级** | 基于时长、频率等行为数据量化学习动力、预警倦怠 |
| **本引擎** | **跨会话心理画像级** | 构建学生长期心理状态模型，追踪心理状态演变轨迹，生成针对性激励策略 |

### 1.2 核心价值

1. **防流失**：在学生产生放弃念头之前，系统已识别到动机下降趋势并介入
2. **提效能**：在学生处于"心流"状态时，加大学习强度；在低落时适当减负
3. **护心理**：识别过度焦虑、自我效能感过低等负面心理状态，避免教育产品加重心理负担
4. **个性化**：不同学生对同一激励方式反应不同——有人受排行榜驱动，有人受鼓励话语激励，有人受知识解锁驱动

### 1.3 核心能力

| 能力 | 说明 |
|------|------|
| 心理状态推断 | 基于多源行为数据推断学生当前心理状态（动机水平、自我效能感、焦虑度、专注度） |
| 状态轨迹追踪 | 记录心理状态随时间的演变轨迹，识别状态拐点和趋势 |
| 激励策略匹配 | 根据学生心理画像匹配最优激励策略（鼓励型/挑战型/社交型/成就型） |
| 激励时机选择 | 判断何时介入激励（不是每次都需要干预，过度激励适得其反） |
| 策略效果反馈 | 追踪激励策略的实际效果，形成闭环优化 |
| 异常状态告警 | 识别需要家长/教师关注的心理异常信号 |
| 激励偏好学习 | 通过长期观察学习每个学生对不同激励方式的响应模式 |

### 1.4 适用场景

| 场景 | 触发条件 | 输出 |
|------|----------|------|
| 首页推荐策略调整 | 学生连续3天学习动力下降 | 调整首页卡片顺序，减少挑战性任务，增加趣味内容 |
| AI对话风格适配 | 学生自我效能感低 | AI回复增加肯定性语言，降低题目难度 |
| 练习难度调节 | 学生焦虑度偏高 | 降低下一组练习的难度，增加成就感 |
| 学习计划微调 | 学生进入倦怠期 | 插入休息日、减少每日任务量、加入趣味挑战 |
| 排行榜可见性控制 | 学生竞争焦虑度高 | 隐藏排行榜或仅展示"超越自我"型数据 |
| 家长推送调整 | 学生心理状态异常 | 向家长发送关怀建议而非成绩报告 |
| 学习打卡提醒 | 学生连续未完成计划 | 调整提醒语气（从催促变为鼓励） |
| 成就系统触发 | 学生达成里程碑 | 选择最匹配学生激励偏好的庆祝方式 |

## 2. 心理状态模型设计

### 2.1 心理状态维度定义

基于教育心理学中的**自我决定理论（SDT）**和**成就目标理论**，定义四个核心心理维度：

```typescript
/**
 * 学生心理状态快照
 * 每个维度取值 0.0 ~ 1.0
 */
interface PsychologicalState {
  /** 快照时间戳 */
  timestamp: string; // ISO 8601

  /** 动机水平 (Motivation Level)
   *  0.0 = 完全无动力  1.0 = 极高动力
   *  推断依据：学习频率、主动发起学习比例、任务完成率
   */
  motivationLevel: number;

  /** 自我效能感 (Self-Efficacy)
   *  0.0 = 极度不自信  1.0 = 高度自信
   *  推断依据：正确率趋势、错题订正主动性、挑战题尝试率
   */
  selfEfficacy: number;

  /** 学术焦虑度 (Academic Anxiety)
   *  0.0 = 完全放松  1.0 = 极度焦虑
   *  推断依据：答题速度突变、反复擦改、跳题率、深夜学习比例
   */
  anxietyLevel: number;

  /** 心流体验度 (Flow State)
   *  0.0 = 严重分心  1.0 = 深度沉浸
   *  推断依据：单次学习时长、连续答题数、跳出频率
   */
  flowState: number;

  /** 综合心理健康指数 (综合评分) */
  overallWellbeing: number;

  /** 数据来源置信度 0.0 ~ 1.0 */
  confidence: number;

  /** 推断依据（可解释性） */
  evidence: PsychStateEvidence;
}

interface PsychStateEvidence {
  /** 参与推断的行为指标列表 */
  indicators: PsychIndicator[];
  /** 主要影响因素（前3个） */
  topFactors: string[];
  /** 数据时间窗口 */
  analysisWindow: {
    start: string;
    end: string;
    dataPoints: number;
  };
}

interface PsychIndicator {
  name: string;
  value: number;
  weight: number;
  contribution: string; // "正向" | "负向" | "中性"
  description: string;  // 人类可读的解释
}
```

### 2.2 心理状态分类（标签体系）

```typescript
/**
 * 心理状态分类标签
 * 每个学生在任意时刻被标记为一个主状态 + 可选的子状态
 */
enum PsychStateLabel {
  // === 正面状态 ===
  FLOW = 'flow',                     // 心流沉浸：高度专注、高效学习
  GROWTH_MINDSET = 'growth_mindset',  // 成长心态：主动挑战、不怕犯错
  STEADY_PROGRESS = 'steady_progress', // 稳步前进：持续稳定的学习节奏

  // === 中性状态 ===
  ROUTINE = 'routine',               // 常规执行：按部就班、缺乏激情但不消极
  RECOVERY = 'recovery',             // 恢复中：从低谷回升

  // === 负面状态 ===
  FRUSTRATION = 'frustration',       // 挫败感：反复失败后的沮丧
  BOREDOM = 'boredom',               // 无聊感：内容太简单或重复
  ANXIETY = 'anxiety',               // 焦虑感：考试压力或自我要求过高
  BURNOUT = 'burnout',               // 倦怠期：长期高强度后的疲惫
  DISENGAGEMENT = 'disengagement',   // 疏离感：有离开产品的倾向
  LOW_SELF_EFFICACY = 'low_self_efficacy', // 低自我效能：认为自己学不会
}

/**
 * 心理状态分类规则
 * 基于四个维度的组合判断
 */
const STATE_CLASSIFICATION_RULES: StateRule[] = [
  {

  {

  },
  {
    label: PsychStateLabel.FRUSTRATION,
    conditions: {
      motivationLevel: { min: 0.3 }, // 还有动力
      selfEfficacy: { max: 0.35 },
      anxietyLevel: { min: 0.5 },
    },
    description: '学生有学习意愿但遇到挫败',
    intervention: 'reduce_difficulty_and_encourage',
  },
  {
    label: PsychStateLabel.BOREDOM,
    conditions: {
      motivationLevel: { max: 0.35 },
      flowState: { max: 0.3 },
      anxietyLevel: { max: 0.3 }, // 低焦虑 + 低动力 = 无聊
    },
    description: '学生对当前内容缺乏兴趣',
    intervention: 'increase_challenge_or_switch_topic',
  },
  {
    label: PsychStateLabel.ANXIETY,
    conditions: {
      anxietyLevel: { min: 0.7 },
      motivationLevel: { min: 0.5 }, // 高焦虑但还有动力
    },

    intervention: 'calm_and_simplify',
  },
  {
    label: PsychStateLabel.BURNOUT,
    conditions: {
      motivationLevel: { max: 0.25 },
      flowState: { max: 0.2 },

    intervention: 'break_and_light_activity',
  },
  {
    label: PsychStateLabel.DISENGAGEMENT,
    conditions: {
      motivationLevel: { max: 0.15 },
      flowState: { max: 0.1 },
      // 连续7天无学习行为 或 日均学习时长骤降70%+
    },
    description: '学生有流失风险',
    intervention: 're_engagement_campaign',
  },
];
```

### 2.3 激励偏好模型

```typescript
/**
 * 学生激励偏好画像
 * 描述该学生对不同激励类型的响应敏感度
 */
interface MotivationProfile {
  studentId: string;

  /** 激励类型偏好评分 (0.0 ~ 1.0) */
  preferences: {
    /** 成就驱动：徽章、等级、解锁新内容 */
    achievement: number;

    /** 社交驱动：排行榜、组队挑战、分享成绩 */
    social: number;

    /** 探索驱动：新知识点解锁、隐藏彩蛋、冒险任务 */
    exploration: number;

    /** 竞争驱动：限时挑战、超越排名、对战胜负 */
    competition: number;

    /** 自主驱动：自由选择学习路径、自定义学习计划 */
    autonomy: number;

    /** 情感驱动：鼓励话语、成长故事、老师/家长表扬 */
    emotional: number;
  };

  /** 最有效的激励策略（Top 3） */
  topStrategies: MotivationStrategy[];

  /** 禁忌策略（可能产生反效果的策略） */
  avoidStrategies: string[];

  /** 偏好学习来源 */
  inferredFrom: 'explicit_survey' | 'behavior_observation' | 'hybrid';

  /** 最后更新时间 */
  updatedAt: string;
}

/**
 * 激励策略
 */
interface MotivationStrategy {
  id: string;
  name: string;
  type: string; // 对应 preferences 中的 key
  description: string;
  /** 实施方式 */
  implementation: StrategyImplementation;
  /** 历史效果评分 0.0~1.0 */
  effectivenessScore: number;
  /** 适用心理状态 */
  applicableStates: PsychStateLabel[];
  /** 不适用状态（使用后可能产生反效果） */
  contraindicatedStates: PsychStateLabel[];
}

interface StrategyImplementation {
  /** 策略触达渠道 */
  channel: 'in_app_popup' | 'notification' | 'ai_dialogue' | 'home_card' | 'achievement';
  /** 策略内容模板（支持变量替换） */
  contentTemplate: string;
  /** 策略参数 */
  parameters: Record<string, unknown>;
  /** 冷却时间（避免重复触发），单位：小时 */
  cooldownHours: number;
}
```

## 3. 数据结构设计

### 3.1 心理状态快照表

```sql
CREATE TABLE student_psych_snapshot (
    id                  BIGSERIAL PRIMARY KEY,
    student_id          BIGINT NOT NULL,
    snapshot_time       TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- 四维心理指标
    motivation_level    DECIMAL(4,3) NOT NULL CHECK (motivation_level BETWEEN 0 AND 1),
    self_efficacy       DECIMAL(4,3) NOT NULL CHECK (self_efficacy BETWEEN 0 AND 1),
    anxiety_level       DECIMAL(4,3) NOT NULL CHECK (anxiety_level BETWEEN 0 AND 1),
    flow_state          DECIMAL(4,3) NOT NULL CHECK (flow_state BETWEEN 0 AND 1),
    overall_wellbeing   DECIMAL(4,3) NOT NULL CHECK (overall_wellbeing BETWEEN 0 AND 1),

    -- 分类标签
    state_label         VARCHAR(32) NOT NULL,

    -- 数据置信度
    confidence          DECIMAL(4,3) NOT NULL DEFAULT 0.5,

    -- 推断依据
    evidence            JSONB NOT NULL,
    -- evidence 结构:
    -- {
    --   "indicators": [
    --     {"name": "daily_study_duration_7d", "value": 45.2, "weight": 0.15,
    --      "contribution": "positive", "description": "近7天日均学习45.2分钟"}
    --   ],
    --   "topFactors": ["correct_rate_trend", "active_session_ratio"],
    --   "analysisWindow": {"start": "2026-06-02T00:00:00Z", "end": "2026-06-09T00:00:00Z", "dataPoints": 127}
    -- }

    -- 关联数据源
    source_event_count  INT NOT NULL DEFAULT 0,     -- 分析期内行为事件数
    analysis_window_hours INT NOT NULL DEFAULT 168,  -- 分析窗口（小时），默认168=7天

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 核心查询索引：按学生+时间查最新快照
CREATE UNIQUE INDEX idx_sps_student_time ON student_psych_snapshot(student_id, snapshot_time DESC);
-- 按状态标签查询（运营分析）
CREATE INDEX idx_sps_label ON student_psych_snapshot(state_label, snapshot_time DESC);
-- 按时间分区（按月）
CREATE INDEX idx_sps_created ON student_psych_snapshot(created_at);
```

```sql
CREATE TABLE student_psych_trajectory (
    id                  BIGSERIAL PRIMARY KEY,
    student_id          BIGINT NOT NULL,
    trajectory_date     DATE NOT NULL,                -- 轨迹点日期

    -- 日维度聚合指标
    avg_motivation      DECIMAL(4,3),
    avg_self_efficacy   DECIMAL(4,3),
    avg_anxiety         DECIMAL(4,3),
    avg_flow            DECIMAL(4,3),
    avg_wellbeing       DECIMAL(4,3),

    -- 日维度主导状态（出现次数最多的标签）
    dominant_state      VARCHAR(32) NOT NULL,

    -- 状态转换
    state_transitions   JSONB,
    -- state_transitions 结构:
    -- [{"from": "steady_progress", "to": "frustration", "timestamp": "...", "trigger": "consecutive_wrong_answers"}]

    -- 关键事件标记
    key_events          JSONB,
    -- key_events 结构:
    -- [{"type": "exam_completed", "score": 85, "impact": "positive"},
    --  {"type": "streak_broken", "detail": "7天打卡中断"}]

    -- 学习行为摘要
    study_duration_min  INT DEFAULT 0,               -- 当日学习时长（分钟）
    questions_attempted INT DEFAULT 0,               -- 当日答题数
    correct_rate        DECIMAL(4,3),                 -- 当日正确率
    sessions_count      INT DEFAULT 0,                -- 当日学习会话数
    active_subjects     JSONB,                         -- 当日学习的学科列表

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE(student_id, trajectory_date)
);

CREATE INDEX idx_spt_student_date ON student_psych_trajectory(student_id, trajectory_date DESC);
```

### 3.3 激励策略执行记录表

```sql
CREATE TABLE motivation_strategy_execution (
    id                  BIGSERIAL PRIMARY KEY,
    student_id          BIGINT NOT NULL,

    -- 策略信息
    strategy_id         VARCHAR(64) NOT NULL,         -- 策略模板ID

    interacted_at       TIMESTAMPTZ,                   -- 互动时间
    effectiveness_score DECIMAL(4,3),                   -- 效果评分（事后计算）

    -- 效果评估依据
    effect_measurement  JSONB,
    -- effect_measurement 结构:
    -- {
    --   "studyDurationAfter24h": 52,     // 策略后24h学习时长(分钟)

    --   "nextLoginWithin": "2h",         // 下次登录间隔
    --   "subject": "overall"             // 影响范围
    -- }

    executed_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    evaluated_at        TIMESTAMPTZ                    -- 效果评估时间
);

    id                  BIGSERIAL PRIMARY KEY,
    student_id          BIGINT NOT NULL UNIQUE,

    -- 六维激励偏好评分
    pref_achievement    DECIMAL(4,3) NOT NULL DEFAULT 0.5,
    pref_social         DECIMAL(4,3) NOT NULL DEFAULT 0.5,
    pref_exploration    DECIMAL(4,3) NOT NULL DEFAULT 0.5,
    pref_competition    DECIMAL(4,3) NOT NULL DEFAULT 0.5,
    pref_autonomy       DECIMAL(4,3) NOT NULL DEFAULT 0.5,
    pref_emotional      DECIMAL(4,3) NOT NULL DEFAULT 0.5,

    -- 偏好推断来源
    inference_source    VARCHAR(16) NOT NULL DEFAULT 'behavior_observation',
    -- 'explicit_survey' = 问卷填写, 'behavior_observation' = 行为观察, 'hybrid' = 混合

    -- 冷却策略：记录最近一次各类型策略的执行时间，避免过度激励
    last_achievement_at TIMESTAMPTZ,
    last_social_at      TIMESTAMPTZ,
    last_exploration_at TIMESTAMPTZ,
    last_competition_at TIMESTAMPTZ,
    last_autonomy_at    TIMESTAMPTZ,
    last_emotional_at   TIMESTAMPTZ,

    -- 禁忌策略（对某学生产生反效果的策略ID列表）
    avoid_strategy_ids  JSONB DEFAULT '[]',

    -- 统计
    total_strategies_executed INT DEFAULT 0,
    avg_effectiveness   DECIMAL(4,3),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_smp_student ON student_motivation_profile(student_id);
```

### 3.5 心理异常告警表

```sql
CREATE TABLE psych_alert (
    id                  BIGSERIAL PRIMARY KEY,
    student_id          BIGINT NOT NULL,
    alert_type          VARCHAR(32) NOT NULL,
    -- alert_type 取值:
    --   burnout_warning        倦怠预警
    --   disengagement_warning  疏离预警
    --   high_anxiety           高焦虑
    --   severe_frustration     严重挫败
    --   motivation_crash       动力骤降
    --   abnormal_pattern       异常行为模式

    severity            VARCHAR(16) NOT NULL,         -- low / medium / high / critical
    state_label         VARCHAR(32) NOT NULL,         -- 触发时状态标签
    snapshot_id         BIGINT,                        -- 关联快照

    -- 告警详情
    description         TEXT NOT NULL,
    evidence            JSONB NOT NULL,                -- 触发依据
    suggested_actions   JSONB NOT NULL,                -- 建议操作列表

    -- 处理状态
    status              VARCHAR(16) NOT NULL DEFAULT 'pending',
    -- pending / notified_parent / notified_teacher / resolved / dismissed
    resolved_by         VARCHAR(64),                   -- 处理人
    resolved_at         TIMESTAMPTZ,
    resolution_note     TEXT,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_pa_student ON psych_alert(student_id, created_at DESC);
CREATE INDEX idx_pa_status ON psych_alert(status, severity, created_at DESC);
CREATE INDEX idx_pa_type ON psych_alert(alert_type, created_at DESC);
```
