# 服务端分布式链路追踪与全链路可观测体系 - 详细设计

## 1. 文档概述

### 1.1 目的

本文档详细设计 PrimeTop 服务端的分布式链路追踪（Distributed Tracing）与全链路可观测（Observability）体系。在微服务/模块化架构下，一次用户请求可能跨越网关、多个业务服务、AI 能力层、数据库、缓存和消息队列等多个组件。本设计确保每一笔关键请求的完整调用链路可追踪、可还原、可诊断。

### 1.2 范围

- 分布式链路追踪基础设施选型与部署
- Trace/Span 数据模型与传播机制
- 各层组件的埋点接入方案
- AI 辅导全链路（用户请求 → 网关 → AI 服务 → 模型调用 → 响应）专项追踪
- Metrics 指标体系（RED + USE 方法论）
- Log-Trace-Metric 三维关联
- 采样策略与存储降本
- 告警规则与根因分析辅助
- 客户端侧基础可观测接入

### 1.3 与已有文档的关系

| 已有文档 | 关系 |

### 2.2 技术选型

| 组件 | 选型 | 理由 |
|------|------|------|
| **采集 SDK** | OpenTelemetry (OTel) | 语言无关、社区活跃、统一 Trace/Metric/Log 三大信号 |
| **链路追踪后端** | Jaeger (初期) / Grafana Tempo (中期) | Jaeger 开箱即用；Tempo 与 Grafana 深度整合，存储成本更低 |
| **指标存储** | Prometheus (初期) / VictoriaMetrics (规模增长后) | Prometheus 生态成熟；VM 更适合高基数场景 |
| **日志存储** | Loki (轻量) / Elasticsearch (全文检索) | Loki 与 Grafana 天然集成，成本低；ES 适合复杂检索 |
| **可视化** | Grafana | 统一 Dashboard，支持 Trace/Metric/Log 联合查询 |
| **采集网关** | OpenTelemetry Collector | 统一接收、处理、路由，解耦应用与后端 |
| **传播格式** | W3C Trace Context + B3 (兼容) | W3C 为标准；B3 兼容旧系统 |

## 3. 数据模型

### 3.1 Trace 数据模型

#### 3.1.1 Trace

```
Trace:
  - trace_id: string          // 128-bit 全局唯一，十六进制编码 (32 chars)
  - trace_state: string       // W3C trace-state，供应商特定数据
  - start_time: timestamp     // 链路开始时间 (Unix nano)
  - end_time: timestamp       // 链路结束时间
  - duration_ms: float        // 总耗时
  - service_count: int        // 涉及服务数
  - span_count: int           // 总 Span 数
  - root_span_name: string    // 根 Span 操作名
  - status: enum              // OK / ERROR / UNSET
```

#### 3.1.2 Span

```
Span:
  // 身份标识
  - trace_id: string          // 所属 Trace
  - span_id: string           // 64-bit 唯一 (16 chars)
  - parent_span_id: string    // 父 Span (根 Span 为空)
  - trace_state: string

  // 时间
  - start_time: timestamp
  - end_time: timestamp

  // 描述
  - name: string              // 操作名，如 "POST /api/v1/ai/chat"
  - kind: enum                // INTERNAL / SERVER / CLIENT / PRODUCER / CONSUMER

  // 上下文
  - resource: map             // 资源属性 (service.name, host, pod)
  - attributes: map           // 业务属性 (见 3.2)
  - events: []TimedEvent      // 事件 (日志、异常)
  - links: []Link             // 关联 (异步场景)

  // 状态
  - status: SpanStatus
    - code: enum              // OK / ERROR / UNSET
    - message: string         // 错误描述
```

#### 4.1.3 服务间传播（Go 服务示例）

```go
package middleware

import (
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/propagation"
    "go.opentelemetry.io/otel/trace"
)

// TraceMiddleware 为 HTTP handler 注入链路追踪
func TraceMiddleware(serviceName string) func(http.Handler) http.Handler {
    tracer := otel.Tracer(serviceName)
    propagator := otel.GetTextMapPropagator()

    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            // 从请求头提取上游传播的 SpanContext
            ctx := propagator.Extract(r.Context(), propagation.HeaderCarrier(r.Header))

            // 创建当前服务的 Server Span
            spanName := r.Method + " " + r.URL.Path
            ctx, span := tracer.Start(ctx, spanName,
                trace.WithSpanKind(trace.SpanKindServer),
                trace.WithAttributes(
                    attribute.String("http.method", r.Method),
                    attribute.String("http.url", r.URL.String()),
                    attribute.String("http.scheme", r.URL.Scheme),
                    attribute.String("http.host", r.Host),
                    attribute.String("net.peer.ip", r.RemoteAddr),
                ),
            )
            defer span.End()

            // 注入业务属性
            if userID := r.Header.Get("X-User-ID"); userID != "" {
                span.SetAttributes(attribute.String(AttrUserID, userID))
            }
            if grade := r.Header.Get("X-Student-Grade"); grade != "" {
                span.SetAttributes(attribute.String(AttrStudentGrade, grade))
            }

            // 包装 ResponseWriter 以捕获状态码
            ww := &responseWriterWrapper{ResponseWriter: w, statusCode: 200}
            next.ServeHTTP(ww, r.WithContext(ctx))

            span.SetAttributes(attribute.Int("http.status_code", ww.statusCode))
            if ww.statusCode >= 400 {
                span.SetStatus(codes.Error, http.StatusText(ww.statusCode))
            }
        })
    }
}

type responseWriterWrapper struct {
    http.ResponseWriter
    statusCode int
}

func (w *responseWriterWrapper) WriteHeader(code int) {
    w.statusCode = code
    w.ResponseWriter.WriteHeader(code)
}
```

### 4.2 异步消息传播

#### 4.2.1 消息队列 Trace 传播（Kafka / RabbitMQ）

```go
package mq

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/propagation"
    "go.opentelemetry.io/otel/trace"
)

// Producer 在消息 Header 中注入 TraceContext
func ProduceWithTrace(ctx context.Context, topic string, msg []byte, headers map[string]string) error {
    tracer := otel.Tracer("mq-producer")
    propagator := otel.GetTextMapPropagator()

    ctx, span := tracer.Start(ctx, "MQ.Produce "+topic,
        trace.WithSpanKind(trace.SpanKindProducer),
    )
    defer span.End()

    // 将 TraceContext 注入消息 Header
    carrier := propagation.MapCarrier{}
    propagator.Inject(ctx, carrier)
    for k, v := range carrier {
        headers[k] = v
    }

    // 记录消息属性
    span.SetAttributes(
        attribute.String("messaging.system", "kafka"),
        attribute.String("messaging.destination", topic),
        attribute.Int("messaging.message.payload_size_bytes", len(msg)),
    )

    // 发送消息（实际 Kafka/RabbitMQ 发送逻辑）
    return kafkaProduce(topic, msg, headers)
}

// Consumer 从消息 Header 中提取 TraceContext
func ConsumeWithTrace(ctx context.Context, topic string, handler func(context.Context, []byte) error) {
    propagator := otel.GetTextMapPropagator()
    tracer := otel.Tracer("mq-consumer")

    for msg := range kafkaConsume(topic) {
        // 从消息 Header 提取上游 SpanContext
        carrier := propagation.MapCarrier(msg.Headers)
        ctx := propagator.Extract(context.Background(), carrier)

        // 创建 Consumer Span，Link 到 Producer Span
        links := []trace.Link{}
        if sc := trace.SpanContextFromContext(ctx); sc.IsValid() {
            links = append(links, trace.Link{SpanContext: sc})
        }

        ctx, span := tracer.Start(ctx, "MQ.Consume "+topic,
            trace.WithSpanKind(trace.SpanKindConsumer),
            trace.WithLinks(links...),
        )

        if err := handler(ctx, msg.Value); err != nil {
            span.RecordError(err)
            span.SetStatus(codes.Error, err.Error())
        }
        span.End()
    }
}
```

#### 4.2.2 定时任务 Trace 传播

```go
// 定时任务每次执行创建新的根 Trace
func ScheduledJobWithTrace(jobName string, fn func(context.Context) error) {
    tracer := otel.Tracer("scheduler")
    ctx, span := tracer.Start(context.Background(), "Cron.Execute "+jobName,
        trace.WithSpanKind(trace.SpanKindInternal),
        trace.WithAttributes(
            attribute.String("job.name", jobName),
            attribute.String("job.trigger", "cron"),
        ),
    )
    defer span.End()

    start := time.Now()
    if err := fn(ctx); err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
    }
    span.SetAttributes(attribute.Float64("job.duration_ms", float64(time.Since(start).Milliseconds())))
}
```

### 4.3 跨线程/协程传播

```go
// Go: 使用 context.Context 自动传播
// 确保所有 goroutine 都从父 ctx 派生

func ProcessAITask(ctx context.Context, task Task) {
    tracer := otel.Tracer("ai-service")
    ctx, span := tracer.Start(ctx, "AI.ProcessTask")
    defer span.End()

    // ❌ 错误：不传播 ctx
    // go func() { callModel(task) }()

    // ✅ 正确：传播 ctx
    g, gCtx := errgroup.WithContext(ctx)
    g.Go(func() error {
        return callModel(gCtx, task)
    })
    if err := g.Wait(); err != nil {
        span.RecordError(err)
    }
}
```

### 5.3 AI 能力层

AI 服务是 PrimeTop 最核心也最耗时的链路，需要精细追踪：

```go
package ai

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/trace"
)

type AIClient struct {
    tracer trace.Tracer
    // ...
}

func (c *AIClient) ChatCompletion(ctx context.Context, req *ChatRequest) (*ChatResponse, error) {
    ctx, span := c.tracer.Start(ctx, "AI.ChatCompletion "+req.Model+" "+req.PromptTemplate,
        trace.WithSpanKind(trace.SpanKindClient),
        trace.WithAttributes(
            attribute.String(AttrAIModel, req.Model),
            attribute.String(AttrAIPromptTpl, req.PromptTemplate),
            attribute.Int(AttrAITokenIn, len(req.Messages)),
            attribute.String(AttrStudentGrade, req.StudentGrade),
            attribute.String(AttrSubject, req.Subject),
        ),
    )
    defer span.End()

    start := time.Now()

    // 记录 Prompt 构建阶段
    ctx, buildSpan := c.tracer.Start(ctx, "AI.BuildPrompt")
    prompt, ragResults, err := c.buildPrompt(ctx, req)
    if err != nil {
        buildSpan.RecordError(err)
        buildSpan.End()
        return nil, err
    }
    buildSpan.SetAttributes(
        attribute.Int("rag.result_count", len(ragResults)),
        attribute.Float64("rag.search_ms", ragResults.SearchDurationMs),
    )
    buildSpan.End()

    // 记录模型调用阶段
    ctx, modelSpan := c.tracer.Start(ctx, "AI.ModelCall "+req.Model,
        trace.WithAttributes(
            attribute.String("http.url", req.Endpoint),
        ),
    )
    resp, err := c.callModelAPI(ctx, req.Endpoint, prompt)
    callDuration := time.Since(start)
    modelSpan.End()

    if err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
        span.SetAttributes(
            attribute.Float64(AttrAILatencyMs, float64(callDuration.Milliseconds())),
        )
        return nil, err
    }

    // 记录响应指标
    tokenOut := resp.Usage.CompletionTokens
    costCents := c.calculateCost(req.Model, resp.Usage.TotalTokens)
    span.SetAttributes(
        attribute.Float64(AttrAILatencyMs, float64(callDuration.Milliseconds())),
        attribute.Int(AttrAITokenOut, tokenOut),
        attribute.Int("ai.token.total", resp.Usage.TotalTokens),
        attribute.Float64(AttrAICostCents, costCents),
        attribute.String("ai.finish_reason", resp.FinishReason),
    )

    return resp, nil
}
```

### 5.4 客户端侧基础追踪

客户端不需要完整的链路追踪 SDK，但应记录关键信息并关联服务端 Trace：

```dart
// Flutter 客户端：轻量追踪
class ClientTracer {
  static String generateTraceId() {
    // 生成 128-bit trace ID，格式与 W3C 一致
    final random = Random.secure();
    final bytes = List<int>.generate(16, (_) => random.nextInt(256));
    return bytes.map((b) => b.toRadixString(16).padLeft(2, '0')).join();
  }

  /// 发起请求时生成 TraceID 并注入 Header
  static Future<Response> tracedRequest(
    String method,
    String url,
    Map<String, dynamic>? body,
  ) async {
    final traceId = generateTraceId();
    final spanId = _generateSpanId(); // 64-bit

    final headers = {
      'traceparent': '00-$traceId-$spanId-01',
      'X-Client-Version': Platform.isIOS ? iosVersion : androidVersion,
      'X-Device-ID': deviceId,
    };

    final stopwatch = Stopwatch()..start();

    try {
      final response = await http.post(
        Uri.parse(url),
        headers: headers,
        body: jsonEncode(body),
      );

      stopwatch.stop();

      // 记录客户端性能事件
      AnalyticsService.logNetworkEvent(
        endpoint: url,
        method: method,
        statusCode: response.statusCode,
        durationMs: stopwatch.elapsedMilliseconds,
        traceId: traceId,
      );

      return response;
    } catch (e) {
      stopwatch.stop();
      AnalyticsService.logNetworkError(
        endpoint: url,
        error: e.toString(),
        durationMs: stopwatch.elapsedMilliseconds,
        traceId: traceId,
      );
      rethrow;
    }
  }
}
```

## 8. 采样策略

### 8.1 采样层级

```
入口 (Gateway) → 采样决策 → 传播到下游所有服务
```

使用 OTel Collector 的 **Tail-Based Sampling** 实现智能采样：

```yaml
# OTel Collector 配置
processors:
  tail_sampling:
    decision_wait: 10s          # 等待 10s 让完整链路到达
    num_traces: 100000          # 内存中缓存的 Trace 数
    expected_new_traces_per_sec: 1000
    policies:
      # 1. 错误链路：100% 保留
      - name: error-traces
        type: status_code
        status_code:
          status_codes:
            - ERROR

      # 2. 慢请求：100% 保留（>3s）
      - name: slow-traces
        type: latency
        latency:
          threshold_ms: 3000

      # 3. AI 请求：100% 保留（高成本链路）
      - name: ai-traces
        type: string_attribute
        string_attribute:
          key: service.name
          values:
            - ai-service
          enabled_regex_match: false

      # 4. 支付请求：100% 保留
      - name: payment-traces
        type: string_attribute
        string_attribute:
          key: http.route
          values:
            - /api/v1/payment/*
            - /api/v1/subscription/*
          enabled_regex_match: true

      # 5. 新用户请求：100% 保留（注册后 24h 内）
      - name: new-user-traces
        type: string_attribute
        string_attribute:
          key: app.user.is_new
          values:
            - "true"

      # 6. 正常流量：采样 5%
      - name: normal-traces
        type: probabilistic
        probabilistic:
          sampling_percentage: 5

      # 7. 健康检查等：完全忽略
      - name: drop-health
        type: and
        and:
          rules:
            - string_attribute:
                key: http.route
                values:
                  - /healthz
                  - /readyz
                  - /metrics
          action: drop
```

### 8.2 采样比例建议

| 请求类型 | 采样率 | 理由 |
|---------|--------|------|
| AI 辅导请求 | 100% | 高成本、核心链路，需完整审计 |
| 支付/订阅请求 | 100% | 资金相关，必须可追溯 |
| 错误请求 | 100% | 排障必需 |
| 慢请求 (>3s) | 100% | 性能分析必需 |
| 新用户请求 (24h内) | 100% | 新用户体验分析 |
| 拍题 OCR | 50% | 中等成本，适度采样 |
| 同步课堂浏览 | 10% | 低风险、高频 |
| 健康检查/指标 | 0% | 无业务价值 |
| 其他正常请求 | 5% | 基线采样 |

### 11.1 自动根因标记

```go
func RecordErrorWithClassification(ctx context.Context, err error) {
    span := trace.SpanFromContext(ctx)

    // 分类错误

    span.SetAttributes(

// 错误分类表
func classifyError(err error) (code, category string) {

    case isAIModelError(err):
        return "AI_MODEL_ERROR", "ai"

        return "RATE_LIMITED", "throttling"
    case isAuthError(err):
        return "AUTH_FAILED", "security"
    case isValidationError(err):
        return "VALIDATION_ERROR", "business"
    default:
        return "UNKNOWN", "unknown"
    }
}
```

## 13. 性能开销与降级策略

### 13.1 开销预估

| 操作 | 额外延迟 | 内存开销 | 说明 |
|------|---------|---------|------|
| Span 创建 | ~1μs | ~1KB/Span | Go OTel SDK |
| Context 传播 | ~0.1μs | ~100B | W3C Trace Context |
| HTTP Header 注入 | ~5μs | ~200B | propagator.Inject |
| 日志 TraceID 注入 | ~0.5μs | ~50B | 从 ctx 提取 |
| Span Export (batch) | ~0.1ms/batch | 批量缓冲 | 异步导出 |

**总体影响**：每笔请求增加约 **0.5-2ms** 延迟（<1%），对用户体验无感知。

### 13.2 降级策略

```go
package tracing

import (
    "context"
    "sync/atomic"
)

// CircuitBreaker: 当 Collector 不可达时自动降级
type TracingCircuitBreaker struct {
    failCount    atomic.Int64
    isOpen       atomic.Bool
    threshold    int64
    resetTimeout time.Duration
}

func NewTracingCircuitBreaker() *TracingCircuitBreaker {
    cb := &TracingCircuitBreaker{
        threshold:    10,
        resetTimeout: 30 * time.Second,
    }
    // 后台恢复检测
    go cb.resetLoop()
    return cb
}

func (cb *TracingCircuitBreaker) RecordFailure() {
    if cb.failCount.Add(1) >= cb.threshold {
        cb.isOpen.Store(true)
    }
}

func (cb *TracingCircuitBreaker) RecordSuccess() {
    cb.failCount.Store(0)
    cb.isOpen.Store(false)
}

func (cb *TracingCircuitBreaker) IsOpen() bool {
    return cb.isOpen.Load()
}

// 降级后：Span 仅写入内存 buffer，不导出
// 应用层无需任何修改，OTel SDK 内置此行为
```

## 14. 安全与隐私

### 14.1 敏感数据脱敏

```go
package tracing

import (
    "go.opentelemetry.io/otel/attribute"
)

// SanitizeUserID 对用户 ID 做脱敏处理
// 规则：保留前4位和后2位，中间用 * 替换
func SanitizeUserID(userID string) string {
    if len(userID) <= 6 {
        return "****"
    }
    return userID[:4] + strings.Repeat("*", len(userID)-6) + userID[len(userID)-2:]
}

// SanitizePhone 手机号脱敏
func SanitizePhone(phone string) string {
    if len(phone) != 11 {
        return "***"
    }
    return phone[:3] + "****" + phone[7:]
}

// ❌ 禁止在 Span 中记录的内容
// - 完整手机号
// - 密码 / Token / API Key
// - 作文原文（版权保护）
// - 支付卡号
// - 身份证号

// ✅ 可以记录的内容
// - 脱敏用户 ID
// - 错误码（不含错误详情中的敏感信息）
// - 模型名称、Prompt 模板名（不含完整 Prompt 内容）
// - 学段、年级、学科
// - Token 数量、延迟、成本
```

### 15.2 初始化代码模板

```go
package tracing

import (
    "context"
    "fmt"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/exporters/otlp/otlptrace/otlptracegrpc"
    "go.opentelemetry.io/otel/sdk/resource"
    sdktrace "go.opentelemetry.io/otel/sdk/trace"
    semconv "go.opentelemetry.io/otel/semconv/v1.24.0"
)

func InitTracerProvider(serviceName, serviceVersion, collectorEndpoint string) (*sdktrace.TracerProvider, error) {
    // 创建 OTLP gRPC Exporter
    exporter, err := otlptracegrpc.New(context.Background(),
        otlptracegrpc.WithEndpoint(collectorEndpoint),
        otlptracegrpc.WithInsecure(),
    )
    if err != nil {
        return nil, fmt.Errorf("failed to create OTLP exporter: %w", err)
    }

    // 创建 Resource（标识当前服务）
    res, err := resource.New(context.Background(),
        resource.WithAttributes(
            semconv.ServiceNameKey.String(serviceName),
            semconv.ServiceVersionKey.String(serviceVersion),
            semconv.DeploymentEnvironmentKey.String(getEnv("DEPLOYMENT_ENV", "development")),
        ),
    )
    if err != nil {
        return nil, fmt.Errorf("failed to create resource: %w", err)
    }

    // 创建 TracerProvider
    tp := sdktrace.NewTracerProvider(
        sdktrace.WithBatcher(exporter,
            sdktrace.WithBatchTimeout(5*time.Second),
            sdktrace.WithMaxExportBatchSize(512),
        ),
        sdktrace.WithResource(res),
        // 生产环境使用 ParentBased + TraceIDRatio 采样
        sdktrace.WithSampler(sdktrace.ParentBased(
            sdktrace.TraceIDRatioBased(1.0), // 入口 100%，由 Collector 做最终采样
        )),
    )

    // 注册为全局 TracerProvider
    otel.SetTracerProvider(tp)
    otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
        propagation.TraceContext{},
        propagation.Baggage{},
    ))

    return tp, nil
}

// 优雅关闭
func ShutdownTracerProvider(tp *sdktrace.TracerProvider) {
    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()
    tp.Shutdown(ctx)
}
```
