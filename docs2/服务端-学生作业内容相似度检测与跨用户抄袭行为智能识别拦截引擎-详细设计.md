# 服务端-学生作业内容相似度检测与跨用户抄袭行为智能识别拦截引擎-详细设计

## 1. 概述

### 1.1 功能定位

本引擎负责在学生提交作业、练习答案、作文等学习内容时，实时/准实时地检测该提交与平台内其他学生历史提交内容的相似度，智能识别抄袭、搬运、协作过度等学术不诚信行为，并触发分级处置策略（标记、预警、拦截、上报教师/家长）。

### 1.2 与现有系统边界

| 现有系统 | 覆盖范围 | 本引擎差异 |
| --- | --- | --- |
| 作文抄袭检测与学术诚信度评估引擎 | 仅针对**作文文体**的文本抄袭，侧重段落级语义相似度和网络来源比对 | 本引擎覆盖**全题型作业**（主观题、填空题、简答题、编程题、数学解答题等），侧重**平台内跨学生**比对 |
| 在线测评防作弊检测与考试诚信度评估引擎 | 针对**在线考试场景**的实时作弊检测（切屏、搜索、多人摄像头异常等） | 本引擎针对**课后作业/练习提交场景**，无实时监控条件，依赖内容比对和行为序列分析 |
| AI生成内容检测引擎 | 检测作业是否由AI生成 | 本引擎检测学生之间的**互相抄袭**，而非AI代写 |

### 1.3 适用范围

- **触发场景**：学生提交作业、提交练习答案、提交作文、提交编程练习
- **检测范围**：同班级、同年级、同教材版本、全平台（逐级扩大搜索窗口）
- **覆盖题型**：主观题、简答题、填空题、作文、编程题、数学解答题、实验报告
- **不含**：客观题（选择题/判断题，因答案唯一不具备抄袭检测意义）

### 1.4 核心目标

1. **高召回**：不遗漏明显的抄袭行为，尤其是整段搬运和改写抄袭
2. **低误报**：避免将"标准答案式回答"误判为抄袭（如数学公式推导，正确步骤天然相似）
3. **实时性**：主观题/简答题在提交后 3 秒内返回检测结果
4. **可解释**：每次抄袭判定都能输出具体的相似片段、比对来源和置信度
5. **可申诉**：学生和教师可对判定结果发起申诉，进入人工复核流程

### 2.1 核心流程

1. 学生通过作业服务提交答案
2. 作业服务异步调用相似度检测API
3. 三路并行召回候选集：文本指纹（SimHash）、语义向量（Embedding ANN）、结构特征
4. 对召回的候选项进行精确比对
5. 多维度融合判定：综合文本相似度、语义相似度、结构相似度、行为特征
6. 根据融合置信度执行分级处置策略
7. 检测结果写入数据库，触发后续通知/标记流程

## 5. 核心算法设计

### 5.1 文本指纹提取 (SimHash)

```python
import hashlib
import re
from collections import Counter

class SimHashExtractor:
    """提取文本的SimHash指纹"""

    def __init__(self, hash_bits=64, window_size=4):
        self.hash_bits = hash_bits
        self.window_size = window_size  # 滑动窗口大小（词组数量）

    def extract(self, text: str) -> int:
        """
        提取SimHash指纹
        :param text: 输入文本
        :return: 64位SimHash值
        """
        # 1. 文本预处理：去除多余空白、标点归一化
        cleaned = self._preprocess(text)

        # 2. 分词（中文使用jieba，英文按空格）
        tokens = self._tokenize(cleaned)
        if len(tokens) < 2:
            # 文本太短，使用字符级hash
            return self._direct_hash(text)

        # 3. 滑动窗口生成特征
        features = self._generate_features(tokens)

        # 4. 每个特征hash后加权
        weights = [1.0] * len(features)
        vector = [0] * self.hash_bits

        for feature, weight in zip(features, weights):
            feature_hash = int(hashlib.md5(feature.encode('utf-8')).hexdigest(), 16)
            for i in range(self.hash_bits):
                bit = (feature_hash >> i) & 1
                vector[i] += weight if bit else -weight

        # 5. 生成最终SimHash
        simhash = 0
        for i in range(self.hash_bits):
            if vector[i] > 0:
                simhash |= (1 << i)

        return simhash

    def _preprocess(self, text: str) -> str:
        """文本预处理"""
        # 去除LaTeX公式标记但保留公式内容
        text = re.sub(r'\$([^$]+)\$', r'\1', text)
        # 统一标点
        text = re.sub(r'[，。！？；：、""'']', ' ', text)
        # 压缩空白
        text = re.sub(r'\s+', ' ', text).strip()
        return text.lower()

    def _tokenize(self, text: str) -> list:
        """分词"""
        try:
            import jieba
            return list(jieba.cut(text))
        except ImportError:
            return text.split()

    def _generate_features(self, tokens: list) -> list:
        """滑动窗口生成特征词组"""
        features = []
        for i in range(len(tokens) - self.window_size + 1):
            window = tokens[i:i + self.window_size]
            features.append(' '.join(window))
        return features

    def _direct_hash(self, text: str) -> int:
        """短文本直接hash"""
        h = hashlib.md5(text.encode('utf-8')).hexdigest()
        return int(h[:16], 16)  # 取前64位


def hamming_distance(hash1: int, hash2: int, bits: int = 64) -> int:
    """计算两个SimHash的海明距离"""
    x = hash1 ^ hash2
    count = 0
    while x and count < bits:
        count += x & 1
        x >>= 1
    return count


def simhash_similarity(hash1: int, hash2: int, bits: int = 64) -> float:
    """SimHash相似度 = 1 - hamming_distance / bits"""
    return 1.0 - hamming_distance(hash1, hash2, bits) / bits
```

### 5.2 MinHash 精确召回 (LSH)

```python
import hashlib
import struct

class MinHashExtractor:
    """MinHash提取，用于精确的集合相似度计算"""

    def __init__(self, num_perm=128):
        self.num_perm = num_perm
        # 生成num_perm个不同的hash种子
        self.seeds = [(i * 0x9e3779b9 + 0x85ebca6b) & 0xFFFFFFFF for i in range(num_perm)]

    def extract(self, text: str, k=5) -> list:
        """
        提取MinHash签名
        :param text: 输入文本
        :param k: shingle大小
        :return: MinHash签名数组 [h1, h2, ..., h128]
        """
        # 1. 生成k-shingle集合
        shingles = self._get_shingles(text, k)
        if not shingles:
            return [0] * self.num_perm

        # 2. 对每个种子计算最小hash
        signature = []
        for seed in self.seeds:
            min_hash = float('inf')
            for shingle in shingles:
                h = self._hash_with_seed(shingle, seed)
                if h < min_hash:
                    min_hash = h
            signature.append(min_hash)

        return signature

    def _get_shingles(self, text: str, k: int) -> set:
        """生成k-shingle集合"""
        cleaned = ''.join(text.split())
        if len(cleaned) < k:
            return {cleaned}
        return {cleaned[i:i+k] for i in range(len(cleaned) - k + 1)}

    def _hash_with_seed(self, s: str, seed: int) -> int:
        """带种子的hash"""
        data = struct.pack('I', seed) + s.encode('utf-8')
        return int(hashlib.sha256(data).hexdigest()[:16], 16)


def estimate_jaccard(sig1: list, sig2: list) -> float:
    """通过MinHash签名估算Jaccard相似度"""
    if len(sig1) != len(sig2):
        raise ValueError("签名长度不一致")
    matches = sum(1 for a, b in zip(sig1, sig2) if a == b)
    return matches / len(sig1)
```

```python
import re
import json

class StructureFeatureExtractor:
    """提取答案的结构特征，用于区分标准答案和抄袭"""

    def extract(self, text: str, question_type: str) -> dict:
        """
        提取结构特征签名
        """
        features = {
            "sentence_count": self._count_sentences(text),
            "avg_sentence_length": self._avg_sentence_length(text),
            "total_length": len(text),
            "formula_count": self._count_formulas(text),
            "formula_ratio": self._formula_ratio(text),
            "step_indicators": self._count_step_indicators(text),
            "keyword_density": self._keyword_density(text),
            "punctuation_pattern": self._punctuation_pattern(text),
            "paragraph_count": text.count('\n') + 1,
            "digit_ratio": sum(c.isdigit() for c in text) / max(len(text), 1),
            "chinese_ratio": sum('\u4e00' <= c <= '\u9fff' for c in text) / max(len(text), 1),
            "unique_char_ratio": len(set(text)) / max(len(text), 1),
            "question_type": question_type,
        }

        # 生成结构签名（用于快速检索）
        features["structure_hash"] = self._generate_structure_hash(features)
        return features

    def _count_sentences(self, text: str) -> int:
        return len(re.split(r'[。！？.!?；;]', text))

    def _avg_sentence_length(self, text: str) -> float:
        sentences = [s for s in re.split(r'[。！？.!?；;]', text) if s.strip()]
        if not sentences:
            return 0.0
        return sum(len(s) for s in sentences) / len(sentences)

    def _count_formulas(self, text: str) -> int:
        # 匹配 $...$, $$...$$, 以及常见数学表达式
        inline = len(re.findall(r'\$[^$]+\$', text))
        block = len(re.findall(r'\$\$[^$]+\$\$', text))
        plain = len(re.findall(r'[a-z]\s*=\s*[\daxy]', text))
        return inline + block + plain

    def _formula_ratio(self, text: str) -> float:
        if not text:
            return 0.0
        formula_chars = sum(len(m) for m in re.findall(r'\$[^$]+\$', text))
        return formula_chars / len(text)

    def _count_step_indicators(self, text: str) -> int:
        """计算步骤指示词数量（解、设、因为、所以、由题意、综上等）"""
        indicators = ['解：', '解:', '设', '因为', '所以', '由题意', '可知', '综上',
                       '即', '故', '则', '因此', '由于', '步骤', 'Step', 'step']
        return sum(text.count(ind) for ind in indicators)

    def _keyword_density(self, text: str) -> dict:
        """关键词密度"""
        words = list(jieba.cut(text))
        word_count = len(words)
        if word_count == 0:
            return {}
        freq = {}
        for w in words:
            if len(w) >= 2:
                freq[w] = freq.get(w, 0) + 1
        # 返回top-5关键词密度
        sorted_freq = sorted(freq.items(), key=lambda x: -x[1])[:5]
        return {w: c / word_count for w, c in sorted_freq}

    def _punctuation_pattern(self, text: str) -> str:
        """标点模式（用于快速结构匹配）"""
        pattern = re.sub(r'[^，。！？；：、,.!?;:\s]', '', text)
        return hashlib.md5(pattern.encode()).hexdigest()[:8]

    def _generate_structure_hash(self, features: dict) -> str:
        """生成结构特征哈希"""
        sig_keys = ["sentence_count", "formula_count", "step_indicators",
                    "paragraph_count", "question_type"]
        sig_values = {k: features.get(k) for k in sig_keys}
        return hashlib.md5(json.dumps(sig_values, sort_keys=True).encode()).hexdigest()
```

### 5.5 多维度融合判定引擎

```python
from dataclasses import dataclass
from typing import List, Optional

@dataclass
class MatchCandidate:
    """匹配候选"""
    target_submission_id: int
    target_student_id: int
    simhash_similarity: float       # SimHash相似度 0-1
    minhash_jaccard: float          # MinHash估算的Jaccard相似度 0-1
    semantic_cosine_sim: float      # 语义余弦相似度 0-1
    structure_similarity: float     # 结构相似度 0-1
    text_exact_ratio: float         # 精确文本重合比例 0-1
    behavior_risk_score: float      # 行为风险评分 0-1
    matched_fragments: list         # 匹配片段详情


@dataclass
class DetectionResult:
    """检测结果"""
    risk_level: str          # none/low/medium/high/critical
    confidence: float        # 综合置信度 0-1
    max_similarity: float    # 最高综合相似度
    action: str              # pass/flag/warn/block/report
    top_matches: list        # Top匹配列表
    decision_detail: dict    # 判定详情


class FusionDecisionEngine:
    """多维度融合判定引擎"""

    # 各维度权重（可配置）
    DEFAULT_WEIGHTS = {
        "simhash": 0.15,
        "minhash": 0.15,
        "semantic": 0.30,
        "structure": 0.10,
        "exact_text": 0.15,
        "behavior": 0.15,
    }

    # 风险等级阈值
    RISK_THRESHOLDS = {
        "none":     0.00,
        "low":      0.35,  # 0.35-0.50
        "medium":   0.50,  # 0.50-0.70
        "high":     0.70,  # 0.70-0.85
        "critical": 0.85,  # 0.85+
    }

    # 处置动作映射
    ACTION_MAP = {
        "none":     "pass",
        "low":      "flag",     # 仅标记，不通知
        "medium":   "warn",     # 通知学生
        "high":     "block",    # 拦截提交，要求重做
        "critical": "report",   # 上报教师和家长
    }

    def __init__(self, weights=None, thresholds=None):
        self.weights = weights or self.DEFAULT_WEIGHTS
        self.thresholds = thresholds or self.RISK_THRESHOLDS

    def evaluate(self, candidates: List[MatchCandidate],
                 question_type: str,
                 is_whitelisted: bool = False) -> DetectionResult:
        """
        对所有候选进行融合评估，返回最高风险结果
        """
        if is_whitelisted:
            return DetectionResult(
                risk_level="none",
                confidence=0.0,
                max_similarity=0.0,
                action="pass",
                top_matches=[],
                decision_detail={"reason": "whitelisted"}
            )

        # 对每个候选项计算综合相似度
        scored = []
        for candidate in candidates:
            score = self._compute_fusion_score(candidate, question_type)
            scored.append((candidate, score))

        # 按综合相似度排序
        scored.sort(key=lambda x: -x[1])

        if not scored:
            return DetectionResult("none", 0.0, 0.0, "pass", [], {})

        top_candidate, top_score = scored[0]
        risk_level = self._score_to_risk(top_score)
        action = self._risk_to_action(risk_level)

        # 特殊题型修正：数学推导等标准过程天然相似
        if question_type == "math_solution" and top_score < 0.75:
            # 检查是否为标准解题模式
            if self._is_standard_solution_pattern(top_candidate):
                risk_level = self._downgrade_risk(risk_level)
                action = self._risk_to_action(risk_level)

        # 构建top matches
        top_matches = []
        for candidate, score in scored[:5]:
            top_matches.append({
                "target_submission_id": candidate.target_submission_id,
                "target_student_id": candidate.target_student_id,
                "similarity": round(score, 4),
                "match_type": self._dominant_dimension(candidate),
                "matched_fragments": candidate.matched_fragments[:3]
            })

        return DetectionResult(
            risk_level=risk_level,
            confidence=round(top_score, 4),
            max_similarity=round(top_score, 4),
            action=action,
            top_matches=top_matches,
            decision_detail={
                "text_simhash_similarity": round(top_candidate.simhash_similarity, 4),
                "minhash_jaccard": round(top_candidate.minhash_jaccard, 4),
                "semantic_cosine_sim": round(top_candidate.semantic_cosine_sim, 4),
                "structure_similarity": round(top_candidate.structure_similarity, 4),
                "exact_text_ratio": round(top_candidate.text_exact_ratio, 4),
                "behavior_risk_score": round(top_candidate.behavior_risk_score, 4),
                "fusion_weights": self.weights
            }
        )

    def _compute_fusion_score(self, candidate: MatchCandidate, question_type: str) -> float:
        """计算多维度融合得分"""
        w = self._get_adjusted_weights(question_type)

        score = (
            candidate.simhash_similarity * w["simhash"] +
            candidate.minhash_jaccard * w["minhash"] +
            candidate.semantic_cosine_sim * w["semantic"] +
            candidate.structure_similarity * w["structure"] +
            candidate.text_exact_ratio * w["exact_text"] +
            candidate.behavior_risk_score * w["behavior"]
        )

        # 如果精确文本重合度极高，直接提升风险
        if candidate.text_exact_ratio > 0.8:
            score = max(score, 0.85)

        return min(score, 1.0)

    def _get_adjusted_weights(self, question_type: str) -> dict:
        """根据题型调整权重"""
        adjusted = dict(self.weights)

        if question_type == "math_solution":
            # 数学题标准推导步骤天然相似，降低文本权重，提高行为权重
            adjusted["simhash"] = 0.10
            adjusted["exact_text"] = 0.10
            adjusted["behavior"] = 0.25
            adjusted["semantic"] = 0.25
        elif question_type == "essay":
            # 作文重视语义和精确文本
            adjusted["semantic"] = 0.40
            adjusted["exact_text"] = 0.20
            adjusted["structure"] = 0.05
        elif question_type == "code":
            # 编程题重视结构和行为
            adjusted["structure"] = 0.25
            adjusted["behavior"] = 0.25
            adjusted["semantic"] = 0.15

        return adjusted

    def _score_to_risk(self, score: float) -> str:
        """分数转风险等级"""
        if score >= self.thresholds["critical"]:
            return "critical"
        elif score >= self.thresholds["high"]:
            return "high"
        elif score >= self.thresholds["medium"]:
            return "medium"
        elif score >= self.thresholds["low"]:
            return "low"
        return "none"

    def _risk_to_action(self, risk: str) -> str:
        return self.ACTION_MAP.get(risk, "pass")

    def _is_standard_solution_pattern(self, candidate: MatchCandidate) -> bool:
        """判断是否为标准解题模式（高结构相似但低行为风险）"""
        return (candidate.structure_similarity > 0.7 and
                candidate.behavior_risk_score < 0.3 and
                candidate.semantic_cosine_sim < 0.6)

    def _downgrade_risk(self, risk: str) -> str:
        """风险降级"""
        downgrade_map = {"critical": "medium", "high": "low", "medium": "low"}
        return downgrade_map.get(risk, risk)

    def _dominant_dimension(self, candidate: MatchCandidate) -> str:
        """判断主导匹配维度"""
        dims = {
            "text": candidate.text_exact_ratio,
            "semantic": candidate.semantic_cosine_sim,
            "structure": candidate.structure_similarity,
            "behavior": candidate.behavior_risk_score,
        }
        return max(dims, key=dims.get)
```

### 5.6 行为风险分析

```python
from datetime import datetime, timedelta

class BehaviorRiskAnalyzer:
    """分析提交行为中的风险信号"""

    def analyze(self, submission_data: dict, reference_data: dict = None) -> dict:
        """
        分析行为风险
        :param submission_data: 当前提交的行为数据
        :param reference_data: 被比对的参考提交行为数据
        :return: {risk_score: float, factors: dict}
        """
        factors = {}
        risk_signals = []

        # 1. 提交时间间隔分析
        if reference_data:
            time_gap = abs(
                (submission_data["submit_time"] - reference_data["submit_time"]).total_seconds() / 60
            )
            factors["submit_time_gap_minutes"] = round(time_gap, 1)
            if time_gap < 5:
                risk_signals.append(("time_proximity", 0.3))
            elif time_gap < 30:
                risk_signals.append(("time_proximity", 0.15))

        # 2. 编辑日志相似度（如果客户端有记录）
        if "edit_log" in submission_data and "edit_log" in reference_data:
            edit_sim = self._edit_log_similarity(
                submission_data["edit_log"],
                reference_data["edit_log"]
            )
            factors["edit_log_similarity"] = round(edit_sim, 3)
            if edit_sim > 0.8:
                risk_signals.append(("edit_pattern", 0.35))

        # 3. 是否存在同IP/同设备
        if submission_data.get("device_id") == reference_data.get("device_id"):
            factors["same_device"] = True
            risk_signals.append(("same_device", 0.4))

        if submission_data.get("ip_address") == reference_data.get("ip_address"):
            factors["same_ip"] = True
            risk_signals.append(("same_ip", 0.2))

        # 4. 提交速度异常（过快完成作业）
        expected_duration = submission_data.get("expected_duration_minutes", 0)
        actual_duration = submission_data.get("actual_duration_minutes", 0)
        if expected_duration > 0 and actual_duration > 0:
            speed_ratio = actual_duration / expected_duration
            factors["completion_speed_ratio"] = round(speed_ratio, 2)
            if speed_ratio < 0.2:
                risk_signals.append(("too_fast", 0.25))

        # 5. 剪贴板粘贴比例
        paste_ratio = submission_data.get("paste_ratio", 0)
        factors["paste_ratio"] = paste_ratio
        if paste_ratio > 0.7:
            risk_signals.append(("high_paste", 0.3))

        # 6. 活跃会话检测（是否与其他学生同时在线）
        concurrent_sessions = submission_data.get("concurrent_classmate_sessions", 0)
        factors["concurrent_sessions"] = concurrent_sessions
        if concurrent_sessions > 0:
            risk_signals.append(("concurrent_activity", 0.15))

        # 聚合风险评分
        total_risk = sum(weight for _, weight in risk_signals)
        risk_score = min(total_risk, 1.0)

        return {
            "risk_score": risk_score,
            "factors": factors,
            "signals": [sig for sig, _ in risk_signals]
        }

    def _edit_log_similarity(self, log1: list, log2: list) -> float:
        """计算编辑日志相似度（编辑序列模式比对）"""
        if not log1 or not log2:
            return 0.0
        # 将编辑动作序列化
        seq1 = self._serialize_edit_log(log1)
        seq2 = self._serialize_edit_log(log2)
        # 使用编辑距离计算相似度
        max_len = max(len(seq1), len(seq2))
        if max_len == 0:
            return 0.0
        distance = self._levenshtein(seq1, seq2)
        return 1.0 - distance / max_len

    def _serialize_edit_log(self, log: list) -> str:
        """将编辑日志序列化为动作序列字符串"""
        actions = []
        for entry in log:
            action_type = entry.get("type", "")
            if action_type == "insert":
                actions.append(f"I:{entry.get('length', 0)}")
            elif action_type == "delete":
                actions.append(f"D:{entry.get('length', 0)}")
            elif action_type == "paste":
                actions.append(f"P:{entry.get('length', 0)}")
            elif action_type == "replace":
                actions.append(f"R:{entry.get('length', 0)}")
        return "|".join(actions)

    @staticmethod
    def _levenshtein(s1: str, s2: str) -> int:
        """计算Levenshtein编辑距离"""
        if len(s1) < len(s2):
            return BehaviorRiskAnalyzer._levenshtein(s2, s1)
        if len(s2) == 0:
            return len(s1)
        previous_row = range(len(s2) + 1)
        for i, c1 in enumerate(s1):
            current_row = [i + 1]
            for j, c2 in enumerate(s2):
                insertions = previous_row[j + 1] + 1
                deletions = current_row[j] + 1
                substitutions = previous_row[j] + (c1 != c2)
                current_row.append(min(insertions, deletions, substitutions))
            previous_row = current_row
        return previous_row[-1]
```

## 7. 召回策略设计

### 7.1 LSH 分桶召回 (SimHash)

```python
class SimHashLSHIndex:
    """SimHash LSH分桶索引"""

    BAND_COUNT = 8       # 8个band
    BAND_BITS = 8        # 每band 8 bits

    def __init__(self, redis_client):
        self.redis = redis_client

    def insert(self, submission_id: int, simhash: int):
        """插入SimHash到LSH索引"""
        bands = self._split_bands(simhash)
        pipe = self.redis.pipeline()
        for i, band_value in enumerate(bands):
            key = f"simhash:band:{i}:{band_value}"
            pipe.sadd(key, submission_id)
            pipe.expire(key, 90 * 24 * 3600)  # 90天TTL
        pipe.execute()

    def query(self, simhash: int, max_distance: int = 3) -> set:
        """
        查询海明距离<=max_distance的候选
        """
        bands = self._split_bands(simhash)
        candidates = set()

        for i, band_value in enumerate(bands):
            # 精确匹配当前band
            key = f"simhash:band:{i}:{band_value}"
            bucket = self.redis.smembers(key)
            candidates.update(int(x) for x in bucket)

            # 如果允许距离>1，还需搜索邻近band值
            if max_distance >= 2:
                for delta in range(1, max_distance + 1):
                    for adj in [band_value + delta, band_value - delta]:
                        if 0 <= adj < (1 << self.BAND_BITS):
                            adj_key = f"simhash:band:{i}:{adj}"
                            adj_bucket = self.redis.smembers(adj_key)
                            candidates.update(int(x) for x in adj_bucket)

        return candidates

    def _split_bands(self, simhash: int) -> list:
        """将64位SimHash分割为8个8位band"""
        bands = []
        mask = (1 << self.BAND_BITS) - 1
        for i in range(self.BAND_COUNT):
            shift = i * self.BAND_BITS
            bands.append((simhash >> shift) & mask)
        return bands

    def remove(self, submission_id: int, simhash: int):
        """从索引中移除"""
        bands = self._split_bands(simhash)
        pipe = self.redis.pipeline()
        for i, band_value in enumerate(bands):
            key = f"simhash:band:{i}:{band_value}"
            pipe.srem(key, submission_id)
        pipe.execute()
```

```python
class BatchDetectionOptimizer:
    """批量检测优化器"""

    async def batch_detect_assignment(self, assignment_id: int, question_id: int):
        """
        对同一作业题目的所有提交进行批量交叉检测
        优化点：
        1. 批量加载指纹，避免逐条查询
        2. 向量批量检索
        3. 本地计算SimHash两两距离
        """
        # 1. 批量加载所有提交指纹
        fingerprints = await self.fp_repo.batch_get_by_question(assignment_id, question_id)

        # 2. 本地SimHash两两比对（避免多次网络请求）
        pairs = self._find_simhash_pairs(fingerprints, max_distance=5)

        # 3. 对候选对进行精确比对
        detailed_results = await asyncio.gather(*[
            self._detailed_compare(fp1, fp2) for fp1, fp2 in pairs
        ])

        # 4. 批量写入检测结果
        await self.record_repo.batch_insert(detailed_results)

    def _find_simhash_pairs(self, fingerprints: list, max_distance: int) -> list:
        """本地SimHash两两距离计算，返回距离<=max_distance的对"""
        pairs = []
        n = len(fingerprints)
        for i in range(n):
            for j in range(i + 1, n):
                dist = hamming_distance(
                    fingerprints[i].simhash_value,
                    fingerprints[j].simhash_value
                )
                if dist <= max_distance:
                    pairs.append((fingerprints[i], fingerprints[j]))
        return pairs
```

```python
import ast
import tokenize
import io

class CodeSimilarityDetector:
    """编程题代码相似度检测"""

    def detect(self, code1: str, code2: str, language: str = "python") -> dict:
        """
        代码相似度检测
        """
        # 1. AST结构相似度
        ast_sim = self._ast_similarity(code1, code2, language)

        # 2. Token序列相似度
        token_sim = self._token_sequence_similarity(code1, code2, language)

        # 3. 代码指纹比对（去除变量名后）
        normalized_sim = self._normalized_code_similarity(code1, code2, language)

        # 4. 注释/变量名重合度
        comment_sim = self._comment_similarity(code1, code2, language)

        # 融合得分
        fusion_score = (
            ast_sim * 0.35 +
            token_sim * 0.25 +
            normalized_sim * 0.30 +
            comment_sim * 0.10
        )

        return {
            "overall_similarity": round(fusion_score, 4),
            "ast_similarity": round(ast_sim, 4),
            "token_similarity": round(token_sim, 4),
            "normalized_similarity": round(normalized_sim, 4),
            "comment_similarity": round(comment_sim, 4),
            "language": language
        }

    def _ast_similarity(self, code1: str, code2: str, language: str) -> float:
        """AST结构相似度"""
        try:
            if language == "python":
                tree1 = ast.parse(code1)
                tree2 = ast.parse(code2)
            else:
                # 其他语言通过API转换为统一AST
                tree1 = self._parse_universal(code1, language)
                tree2 = self._parse_universal(code2, language)

            # 将AST序列化并计算树编辑距离
            seq1 = self._ast_to_sequence(tree1)
            seq2 = self._ast_to_sequence(tree2)

            return self._tree_similarity(seq1, seq2)
        except SyntaxError:
            return 0.0

    def _normalized_code_similarity(self, code1: str, code2: str, language: str) -> float:
        """归一化代码相似度（去除变量名、注释后比对）"""
        norm1 = self._normalize_code(code1, language)
        norm2 = self._normalize_code(code2, language)
        return self._ratio_similarity(norm1, norm2)

    def _normalize_code(self, code: str, language: str) -> str:
        """代码归一化：统一变量名、去除注释、统一格式"""
        # 去除注释
        if language == "python":
            code = "\n".join(
                line.split("#")[0] for line in code.split("\n")
            )
        elif language in ("java", "cpp", "javascript"):
            code = __import__("re").sub(r'//.*', '', code)
            code = __import__("re").sub(r'/\*.*?\*/', '', code, flags=__import__("re").DOTALL)

        # 统一变量名（替换为VAR_1, VAR_2...）
        if language == "python":
            try:
                tree = ast.parse(code)
                renamer = VariableRenamer()
                tree = renamer.visit(tree)
                code = ast.unparse(tree)
            except:
                pass

        # 压缩空白
        code = __import__("re").sub(r'\s+', ' ', code).strip()
        return code

    def _ast_to_sequence(self, tree) -> list:
        """将AST转为节点类型序列"""
        sequence = []
        for node in ast.walk(tree):
            sequence.append(type(node).__name__)
        return sequence

    def _tree_similarity(self, seq1: list, seq2: list) -> float:
        """基于序列的树相似度"""
        from difflib import SequenceMatcher
        return SequenceMatcher(None, seq1, seq2).ratio()

    def _token_sequence_similarity(self, code1: str, code2: str, language: str) -> float:
        """Token序列相似度"""
        tokens1 = self._extract_tokens(code1, language)
        tokens2 = self._extract_tokens(code2, language)
        from difflib import SequenceMatcher
        return SequenceMatcher(None, tokens1, tokens2).ratio()

    def _extract_tokens(self, code: str, language: str) -> list:
        """提取token类型序列"""
        if language == "python":
            try:
                tokens = tokenize.generate_tokens(io.StringIO(code).readline)
                return [tok.type for tok in tokens if tok.type != tokenize.ENDMARKER]
            except:
                return []
        return list(code)

    def _comment_similarity(self, code1: str, code2: str, language: str) -> float:
        """注释相似度"""
        comments1 = self._extract_comments(code1, language)
        comments2 = self._extract_comments(code2, language)
        if not comments1 and not comments2:
            return 1.0  # 都没有注释，视为相似
        from difflib import SequenceMatcher
        return SequenceMatcher(None, comments1, comments2).ratio()

    def _extract_comments(self, code: str, language: str) -> str:
        """提取注释内容"""

            return "\n".join(re.findall(r'#\s*(.+)', code))

class VariableRenamer(ast.NodeTransformer):
    """AST变量名统一重命名"""

    def __init__(self):
        self.var_map = {}
        self.counter = 0

    def visit_Name(self, node):
        if node.id not in self.var_map:
            self.counter += 1
            self.var_map[node.id] = f"VAR_{self.counter}"
        node.id = self.var_map[node.id]
```

```python
class MathSolutionPatternMatcher:
    """数学解答题标准模式识别"""

    # 常见标准解题模式库
    STANDARD_PATTERNS = {
        "quadratic_vertex": {
            "description": "二次函数顶点公式",
            "formulas": ["x=-b/(2a)", "y=f(-b/(2a))"],
            "step_keywords": ["顶点", "对称轴", "配方"]
        },
        "vieta_theorem": {
            "description": "韦达定理",
            "formulas": ["x1+x2=-b/a", "x1*x2=c/a"],
            "step_keywords": ["韦达", "两根之和", "两根之积"]
        },
        "derivative_monotonicity": {
            "description": "导数判断单调性",
            "formulas": ["f'(x)>0", "f'(x)<0"],
            "step_keywords": ["求导", "单调递增", "单调递减"]
        },
        # ... 更多标准模式
    }

    def is_standard_pattern(self, content: str, similarity: float) -> bool:
        """
        判断答案是否匹配标准解题模式
        """
        for pattern_id, pattern in self.STANDARD_PATTERNS.items():
            if self._match_pattern(content, pattern):
                return True
        return False

    def _match_pattern(self, content: str, pattern: dict) -> bool:
        """检查内容是否匹配特定模式"""
        formula_matches = sum(
            1 for f in pattern["formulas"]
            if f.replace(" ", "") in content.replace(" ", "")
        )
        keyword_matches = sum(
            1 for kw in pattern["step_keywords"]
            if kw in content
        )

        total_criteria = len(pattern["formulas"]) + len(pattern["step_keywords"])
        match_count = formula_matches + keyword_matches
        return match_count / total_criteria >= 0.5
```

| 数据泄露 | 检测记录暴露学生隐私 | API返回脱敏数据，仅审核流程中展示完整信息 |

### 13.3 合规要求

- 抄袭判定结果不对外公开，仅通知当事人和教师/家长
- 申诉流程必须在 7 天内响应
- 抄袭记录不纳入学生永久档案，每学期自动清零
- 遵守《未成年人保护法》和《个人信息保护法》相关要求
