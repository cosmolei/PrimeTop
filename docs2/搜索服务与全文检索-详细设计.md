# 搜索服务与全文检索 - 详细设计

## 1. 模块概述

### 1.1 定位

搜索服务是 PrimeTop 平台的基础能力层，为题目检索、知识点查找、教材内容搜索、错题搜索等场景提供统一的全文本检索能力。本模块基于 Elasticsearch（以下简称 ES）构建，向上游业务模块提供标准化搜索 API，支持中文分词、多字段权重排序、聚合过滤和拼音模糊匹配。

### 1.2 与 RAG 向量检索的关系

| 维度 | 全文检索（本模块） | 向量检索（RAG 模块） |
|------|-------------------|---------------------|
| 引擎 | Elasticsearch | Milvus / pgvector |
| 原理 | BM25 / TF-IDF 词频匹配 | 语义向量相似度 |
| 适用场景 | 精确关键词、题目编号、公式片段 | 语义相似、模糊意图 |
| 延迟 | <50ms | 50-200ms |
| 典型调用 | 题库搜索、知识点搜索 | AI 辅导知识增强 |

RAG 模块中的多路召回（BM25 通道）也使用 ES，但由 RAG 内部管理索引和查询。本模块管理的是**面向用户和业务**的搜索索引。

### 1.3 搜索场景矩阵

| 搜索场景 | 主要索引 | 触发入口 | 调用频率 |
|----------|---------|----------|---------|
| 题目搜索 | `question_v1` | 拍题结果关联、题库浏览、类题推荐 | 高 |
| 知识点搜索 | `knowledge_point_v1` | 知识点查询、薄弱点展示 | 中 |
| 教材章节搜索 | `chapter_v1` | 同步课堂章节定位 | 中 |
| 错题搜索 | `mistake_v1` | 错题本搜索、错因分析 | 中 |
| 全局搜索 | 跨索引 | APP 搜索框 | 低 |

## 4. API 接口设计

### 4.1 统一搜索请求结构

```python
# 搜索请求基类
class SearchRequest(BaseModel):
    keyword: str = Field(..., min_length=1, max_length=200, description="搜索关键词")
    page: int = Field(1, ge=1, description="页码")
    page_size: int = Field(20, ge=1, le=50, description="每页数量")
    filters: Optional[Dict[str, Any]] = Field(None, description="过滤条件")
    sort_by: Optional[str] = Field(None, description="排序字段")
    sort_order: Optional[str] = Field("desc", pattern="^(asc|desc)$")
    highlight: bool = Field(True, description="是否高亮关键词")
```

```python
# 统一搜索响应
class SearchResponse(BaseModel):
    total: int = Field(..., description="总命中数")
    page: int = Field(..., description="当前页码")
    page_size: int = Field(..., description="每页数量")
    items: List[Dict[str, Any]] = Field(..., description="搜索结果列表")
    highlights: Optional[List[Dict[str, str]]] = Field(None, description="高亮片段")
    aggregations: Optional[Dict[str, Any]] = Field(None, description="聚合结果")
    suggestions: Optional[List[str]] = Field(None, description="搜索建议")
    took_ms: int = Field(..., description="查询耗时(ms)")
```

## 5. 核心查询构建逻辑

### 5.1 题目搜索查询构建器

```python
class QuestionQueryBuilder:
    """题目搜索查询构建器"""

    # 字段权重配置（用于 multi_match 查询）
    FIELD_BOOSTS = {
        "stem": 10.0,           # 题干权重最高
        "stem.pinyin": 3.0,     # 拼音匹配辅助
        "options": 5.0,         # 选项内容
        "image_ocr_text": 2.0,  # 图片 OCR 文本
        "answer_summary": 1.0,  # 答案摘要
        "formula_latex": 1.5,   # 公式
        "tags": 3.0,            # 标签精确匹配
    }

    def build(self, req: QuestionSearchRequest) -> dict:
        must_clauses = []
        filter_clauses = []

        # 1. 关键词查询 - multi_match
        if req.keyword:
            must_clauses.append({
                "multi_match": {
                    "query": req.keyword,
                    "fields": [f"{k}^{v}" for k, v in self.FIELD_BOOSTS.items()],
                    "type": "best_fields",
                    "fuzziness": "AUTO",
                    "prefix_length": 2,
                    "max_expansions": 50,
                    "minimum_should_match": "75%"
                }
            })

            # 如果关键词包含英文/数字，增加标准分词器查询
            if self._has_english_or_number(req.keyword):
                must_clauses.append({
                    "multi_match": {
                        "query": req.keyword,
                        "fields": ["stem^5", "formula_latex^3"],
                        "analyzer": "standard",
                        "type": "best_fields",
                        "boost": 0.5
                    }
                })

        # 2. 过滤条件
        f = req.filters or {}
        
        if f.get("subject_code"):
            filter_clauses.append({"term": {"subject_code": f["subject_code"]}})
        if f.get("stage"):
            filter_clauses.append({"term": {"stage": f["stage"]}})
        if f.get("grade"):
            grades = f["grade"] if isinstance(f["grade"], list) else [f["grade"]]
            filter_clauses.append({"terms": {"grade": grades}})
        if f.get("question_type"):
            types = f["question_type"] if isinstance(f["question_type"], list) else [f["question_type"]]
            filter_clauses.append({"terms": {"question_type": types}})
        if f.get("difficulty"):
            diff = f["difficulty"]
            range_q = {}
            if "gte" in diff: range_q["gte"] = diff["gte"]
            if "lte" in diff: range_q["lte"] = diff["lte"]
            if range_q:
                filter_clauses.append({"range": {"difficulty": range_q}})
        if f.get("knowledge_point_ids"):
            kp_ids = f["knowledge_point_ids"] if isinstance(f["knowledge_point_ids"], list) else [f["knowledge_point_ids"]]
            filter_clauses.append({"terms": {"knowledge_point_ids": kp_ids}})
        if f.get("textbook_id"):
            filter_clauses.append({"term": {"textbook_id": f["textbook_id"]}})
        if f.get("has_image") is not None:
            filter_clauses.append({"term": {"has_image": f["has_image"]}})
        if f.get("tags"):
            tags = f["tags"] if isinstance(f["tags"], list) else [f["tags"]]
            filter_clauses.append({"terms": {"tags": tags}})

        # 3. 组装 bool 查询
        bool_query = {"must": must_clauses, "filter": filter_clauses}

        # 4. 排序
        sort = self._build_sort(req.sort_by, req.sort_order)

        # 5. 高亮
        highlight = self._build_highlight() if req.highlight else None

        # 6. 聚合
        aggs = self._build_aggregations()

        # 7. 组装完整查询
        query = {
            "query": {"bool": bool_query},
            "from": (req.page - 1) * req.page_size,
            "size": req.page_size,
            "sort": sort,
            "_source": self._source_fields(),
        }
        if highlight:
            query["highlight"] = highlight
        if aggs:
            query["aggs"] = aggs

        return query

    def _build_sort(self, sort_by: Optional[str], sort_order: str) -> list:
        if sort_by == "relevance" or sort_by is None:
            return ["_score", {"usage_count": "desc"}]
        elif sort_by == "difficulty_asc":
            return [{"difficulty": "asc"}, "_score"]
        elif sort_by == "difficulty_desc":
            return [{"difficulty": "desc"}, "_score"]
        elif sort_by == "usage_desc":
            return [{"usage_count": "desc"}, "_score"]
        elif sort_by == "correct_rate_asc":
            return [{"correct_rate": "asc"}, "_score"]
        elif sort_by == "created_desc":
            return [{"created_at": "desc"}, "_score"]
        elif sort_by == "random":
            return [{"_script": {
                "type": "number",
                "script": "Math.random()",
                "order": "asc"
            }}]
        return ["_score"]

    def _build_highlight(self) -> dict:
        return {
            "fields": {
                "stem": {
                    "pre_tags": ["<em>"],
                    "post_tags": ["</em>"],
                    "fragment_size": 200,
                    "number_of_fragments": 1,
                    "no_match_size": 150
                },
                "options": {
                    "pre_tags": ["<em>"],
                    "post_tags": ["</em>"],
                    "fragment_size": 100,
                    "number_of_fragments": 1
                }
            }
        }

    def _build_aggregations(self) -> dict:
        return {
            "by_difficulty": {
                "terms": {"field": "difficulty", "size": 5}
            },
            "by_question_type": {
                "terms": {"field": "question_type", "size": 10}
            },
            "by_grade": {
                "terms": {"field": "grade", "size": 12}
            }
        }

    @staticmethod
    def _source_fields() -> list:
        return [
            "question_id", "subject_code", "stage", "grade", "question_type",
            "difficulty", "stem", "knowledge_point_ids", "tags",
            "correct_rate", "usage_count", "has_image"
        ]

    @staticmethod
    def _has_english_or_number(text: str) -> bool:
        import re
        return bool(re.search(r'[a-zA-Z0-9]', text))
```

```python
class ESSyncConsumer:
    """ES 同步消费者 - 监听 Kafka 消息并同步到 ES"""

    BATCH_SIZE = 100
    FLUSH_INTERVAL_MS = 2000

    def __init__(self, es_client: AsyncElasticsearch, kafka_consumer: AIOKafkaConsumer):
        self.es = es_client
        self.consumer = kafka_consumer
        self.buffer: Dict[str, list] = {}  # index -> [actions]

    async def run(self):
        async for msg in self.consumer:
            topic = msg.topic
            payload = json.loads(msg.value)

            # 解析 Canal 消息格式
            table = payload["table"]
            action = payload["type"]  # INSERT / UPDATE / DELETE
            data = payload["data"]
            pk = payload["pk"]

            # 路由到对应索引
            index_name = self._get_index_name(table)
            if not index_name:
                continue

            # 转换数据格式
            es_action = self._build_action(index_name, action, pk, data)

            # 加入缓冲区
            if index_name not in self.buffer:
                self.buffer[index_name] = []
            self.buffer[index_name].append(es_action)

            # 批量刷新
            if len(self.buffer[index_name]) >= self.BATCH_SIZE:
                await self._flush(index_name)

    async def _flush(self, index_name: str):
        actions = self.buffer.pop(index_name, [])
        if not actions:
            return

        # 构建 bulk 请求体
        bulk_body = []
        for action in actions:
            bulk_body.append(action["meta"])
            if "data" in action:
                bulk_body.append(action["data"])

        try:
            result = await self.es.bulk(body=bulk_body, index=index_name, refresh=False)
            if result.get("errors"):
                for item in result["items"]:
                    if item.get("error"):
                        logger.error(f"ES bulk error: {item['error']}")
        except Exception as e:
            logger.error(f"ES flush failed for {index_name}: {e}")
            # 写入死信队列
            await self._send_to_dlq(index_name, actions, str(e))

    def _build_action(self, index: str, action: str, pk: str, data: dict) -> dict:
        if action == "DELETE":
            return {
                "meta": {"delete": {"_index": index, "_id": pk}}
            }
        else:
            # INSERT 或 UPDATE
            es_doc = self._transform_data(index, data)
            return {
                "meta": {"index": {"_index": index, "_id": pk}},
                "data": es_doc
            }

    def _transform_data(self, index: str, data: dict) -> dict:
        """将 MySQL 数据转换为 ES 文档格式"""
        transformers = {
            "question_v1": self._transform_question,
            "knowledge_point_v1": self._transform_kp,
            "chapter_v1": self._transform_chapter,
        }
        transformer = transformers.get(index)
        if transformer:
            return transformer(data)
        return data

    @staticmethod
    def _transform_question(data: dict) -> dict:
        return {
            "question_id": data["id"],
            "subject_code": data["subject_code"],
            "stage": data["stage"],
            "grade": data.get("grade", "").split(","),
            "semester": data.get("semester"),
            "textbook_id": data.get("textbook_id"),
            "chapter_ids": data.get("chapter_ids", "").split(",") if data.get("chapter_ids") else [],
            "knowledge_point_ids": data.get("kp_ids", "").split(",") if data.get("kp_ids") else [],
            "question_type": data["question_type"],
            "difficulty": int(data.get("difficulty", 3)),
            "stem": data["stem"],
            "options": data.get("options"),
            "answer_summary": data.get("answer_summary"),
            "tags": data.get("tags", "").split(",") if data.get("tags") else [],
            "exam_type": data.get("exam_type"),
            "year": data.get("year"),
            "source": data.get("source"),
            "has_image": bool(data.get("image_urls")),
            "image_ocr_text": data.get("ocr_text"),
            "formula_latex": data.get("formula_latex"),
            "usage_count": int(data.get("usage_count", 0)),
            "correct_rate": float(data.get("correct_rate", 0.0)),
            "updated_at": data.get("updated_at"),
        }
```

```python
class ESReindexJob:
    """ES 全量修复定时任务 - 每日凌晨执行"""

    INDEX_TABLE_MAP = {
        "question_v1": ("questions", "updated_at"),
        "knowledge_point_v1": ("knowledge_points", "updated_at"),
        "chapter_v1": ("chapters", "updated_at"),
    }

    async def run(self, index_name: str):
        """全量重建指定索引（使用 reindex 策略避免停机）"""
        alias = index_name
        new_index = f"{index_name}__{datetime.now().strftime('%Y%m%d%H%M')}"

        # 1. 创建新索引（使用最新 mapping）
        await self.es.indices.create(index=new_index, body=self._get_mapping(index_name))

        # 2. 从 MySQL 分页读取并写入新索引
        table, _ = self.INDEX_TABLE_MAP[index_name]
        offset = 0
        batch_size = 500
        while True:
            rows = await self.db.fetch_all(
                f"SELECT * FROM {table} ORDER BY id LIMIT :limit OFFSET :offset",
                {"limit": batch_size, "offset": offset}
            )
            if not rows:
                break

            bulk_body = []
            for row in rows:
                doc = self._transform_row(index_name, dict(row))
                bulk_body.append({"index": {"_index": new_index, "_id": str(row["id"])}})
                bulk_body.append(doc)

            await self.es.bulk(body=bulk_body, refresh=False)
            offset += batch_size

        # 3. 原子切换别名
        old_indices = await self.es.indices.get(alias=alias)
        actions = [
            {"add": {"index": new_index, "alias": alias}},
        ]
        for old_idx in old_indices:
            actions.append({"remove": {"index": old_idx, "alias": alias}})

        await self.es.indices.update_aliases({"actions": actions})

        # 4. 删除旧索引
        for old_idx in old_indices:
            if old_idx != new_index:
                await self.es.indices.delete(index=old_idx)

        logger.info(f"Reindex completed: {alias} -> {new_index}")
```

```python
        "anonymous": {"rpm": 5, "burst": 10},
        "free_user": {"rpm": 10, "burst": 20},
        "premium_user": {"rpm": 30, "burst": 60},
        "admin": {"rpm": 100, "burst": 200},
    },
    "suggest": {
        "anonymous": {"rpm": 10, "burst": 20},
        "free_user": {"rpm": 20, "burst": 40},
        "premium_user": {"rpm": 60, "burst": 120},
    }
}

# Redis 限流实现
class SearchRateLimiter:
    async def check(self, user_id: str, endpoint: str, tier: str) -> bool:
        config = SEARCH_RATE_LIMITS.get(endpoint, {}).get(tier)
        if not config:
            return True

        key = f"rate_limit:search:{endpoint}:{user_id}"
        current = await self.redis.incr(key)
        if current == 1:
            await self.redis.expire(key, 60)

        if current > config["rpm"]:
            raise SearchError("SEARCH_010", "搜索频率过高，请稍后再试")

        return True
```

```python
class PersonalizedSearchService:
    """基于用户画像的个性化搜索排序"""

    async def search_questions(
        self,
        req: QuestionSearchRequest,
        user_profile: StudentProfile
    ) -> SearchResponse:
        query = self.query_builder.build(req)

        # 注入个性化排序因子
        query["query"] = {
            "function_score": {
                "query": query["query"],
                "functions": self._build_personalized_functions(user_profile),
                "score_mode": "sum",
                "boost_mode": "multiply",
                "max_boost": 2.0
            }
        }

        result = await self.es.search(body=query)
        return self._parse_response(result)

    def _build_personalized_functions(self, profile: StudentProfile) -> list:
        functions = []

        # 1. 难度匹配：推荐略高于当前水平的题目
        target_difficulty = profile.ability_level + 1  # 维果茨基最近发展区
        functions.append({
            "gauss": {
                "difficulty": {
                    "origin": target_difficulty,
                    "scale": 1.0,
                    "decay": 0.5
                }
            },
            "weight": 1.5
        })

        # 2. 薄弱知识点题目提权
        if profile.weak_kp_ids:
            functions.append({
                "filter": {"terms": {"knowledge_point_ids": profile.weak_kp_ids}},
                "weight": 2.0
            })

        # 3. 当前教材版本提权
        if profile.textbook_id:
            functions.append({
                "filter": {"term": {"textbook_id": profile.textbook_id}},
                "weight": 1.2
            })

        # 4. 排除已做过的题目（去重）
        if profile.recently_done_question_ids:
            functions.append({
                "filter": {
                    "bool": {
                        "must_not": [
                            {"terms": {"question_id": profile.recently_done_question_ids}}
                        ]
                    }
                },
                "weight": 1.0
            })

        return functions
```
