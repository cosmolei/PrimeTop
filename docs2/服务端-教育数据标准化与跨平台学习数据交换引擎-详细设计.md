# 服务端-教育数据标准化与跨平台学习数据交换引擎-详细设计

## 1. 模块概述

### 1.1 设计目标

构建统一的教育数据标准化与跨平台数据交换引擎，使 PrimeTop 能够：

1. **标准化输出**：将内部学习行为数据转换为国际通用的教育数据标准格式（xAPI/CALIPER/OneRoster），支持与第三方教育平台、学校 SIS 系统、政府教育数据平台的数据互通。
2. **标准化输入**：从外部系统（学校学生信息系统、教务系统、其他教育产品）导入学生、班级、课程、成绩等数据，并自动映射为 PrimeTop 内部实体。
3. **B 端数据服务**：为合作学校、教育机构、区域教育平台提供标准化数据接口，支撑 B 端商业合作场景。
4. **合规性保障**：在数据交换全链路中落实 PIPL（个人信息保护法）、《未成年人保护法》、教育部《教育移动互联网应用程序备案管理办法》等法规要求。

### 1.2 设计原则

| 原则 | 说明 |
| --- | --- |
| 最小化披露 | 仅交换必要字段，敏感信息脱敏后输出 |
| 双向可审计 | 所有数据交换操作留存完整审计日志 |
| 格式可扩展 | 标准适配器模式，支持后续新增数据标准 |
| 异步解耦 | 数据导出/导入通过消息队列异步处理，不影响主链路 |
| 版本兼容 | 标准格式版本变更时保持向后兼容 |

/**

 * 所有具体标准适配器实现此接口
 */
public interface StandardAdapter<T> {
    
    /**
     * 获取支持的标准的编码
     */
    String getStandardCode();
    
    /**
     * 获取支持的实体类型列表
     */
    List<String> getSupportedEntityTypes();
    
    /**
     * 将内部实体转换为目标标准格式
     */
    List<T> toStandard(List<InternalEntity> internalEntities, MappingContext context);
    
    /**
     * 将标准格式数据转换为内部实体
     */
    List<InternalEntity> fromStandard(List<T> standardEntities, MappingContext context);
    
    /**
     * 校验标准格式数据是否符合规范
     */
    ValidationResult validate(T standardEntity);
    
    /**
     * 获取标准格式的 JSON Schema
     */
    JsonSchema getJsonSchema(String entityType);
}

     */
    @Override
    public List<OneRosterEntity> toStandard(List<InternalEntity> internalEntities,
                                             MappingContext context) {
        List<OneRosterEntity> results = new ArrayList<>();
        
        for (InternalEntity internal : internalEntities) {
            String entityType = internal.getEntityType();
            List<StandardMappingRule> rules = mappingRuleService
                .getRules(STANDARD_CODE, entityType, context.getPartnerId());
            
            OneRosterEntity standardEntity = new OneRosterEntity();
            standardEntity.setEntityType(entityType);
            
            for (StandardMappingRule rule : rules) {
                Object internalValue = extractFieldValue(internal, rule.getInternalFieldPath());
                
                // 应用转换器
                if (rule.getTransformerClass() != null) {
                    FieldTransformer transformer = transformerRegistry
                        .getTransformer(rule.getTransformerClass());
                    internalValue = transformer.transform(internalValue, 
                        rule.getTransformerParams());
                }
                
                // 敏感字段脱敏
                if (Boolean.TRUE.equals(rule.getSensitive()) 
                    && Boolean.TRUE.equals(context.getRedactSensitive())) {
                    internalValue = redactor.redact(internalValue, 
                        RedactStrategy.valueOf(rule.getRedactStrategy()));
                }
                
                // 设置默认值
                if (internalValue == null && rule.getDefaultValue() != null) {
                    internalValue = rule.getDefaultValue();
                }
                
                standardEntity.setField(rule.getStandardFieldPath(), internalValue);
            }
            
            // 设置元数据
            standardEntity.setSourcedId(internal.getEntityId());
            standardEntity.setStatus(StatusType.ACTIVE);
            standardEntity.setDateLastModified(internal.getUpdatedAt());
            
            results.add(standardEntity);
        }
        
        return results;
    }
    
    /**
     * OneRoster 标准格式 → 内部实体
     */
    @Override
    public List<InternalEntity> fromStandard(List<OneRosterEntity> standardEntities,
                                              MappingContext context) {
        List<InternalEntity> results = new ArrayList<>();
        
        for (OneRosterEntity stdEntity : standardEntities) {
            String entityType = stdEntity.getEntityType();
            List<StandardMappingRule> rules = mappingRuleService
                .getRules(STANDARD_CODE, entityType, context.getPartnerId());
            
            InternalEntity internal = new InternalEntity();
            internal.setEntityType(mapToInternalEntityType(entityType));
            
            for (StandardMappingRule rule : rules) {
                Object standardValue = stdEntity.getField(rule.getStandardFieldPath());
                
                // 反向转换
                if (rule.getTransformerClass() != null) {
                    FieldTransformer transformer = transformerRegistry
                        .getTransformer(rule.getTransformerClass());
                    standardValue = transformer.reverse(standardValue,
                        rule.getTransformerParams());
                }
                
                internal.setField(rule.getInternalFieldPath(), standardValue);
            }
            
            // 保留外部 sourcedId 用于去重和关联
            internal.setExternalId(stdEntity.getSourcedId());
            internal.setSourceSystem(context.getSourceSystem());
            
            results.add(internal);
        }
        
        return results;
    }
    
    /**
     * 校验 OneRoster 实体是否符合规范
     */
    @Override
    public ValidationResult validate(OneRosterEntity entity) {
        ValidationResult result = new ValidationResult();
        
        // sourcedId 必填且为 UUID 格式
        if (StringUtils.isBlank(entity.getSourcedId())) {
            result.addError("sourcedId", "sourcedId is required");
        } else if (!isValidUUID(entity.getSourcedId())) {
            result.addError("sourcedId", "sourcedId must be a valid UUID");
        }
        
        // status 必填
        if (entity.getStatus() == null) {
            result.addError("status", "status is required (active|tobedeleted)");
        }
        
        // 按实体类型校验必填字段
        switch (entity.getEntityType()) {
            case "USER":
                validateUser(entity, result);
                break;
            case "CLASS":
                validateClass(entity, result);
                break;
            case "ENROLLMENT":
                validateEnrollment(entity, result);
                break;
            case "RESULT":
                validateResult(entity, result);
                break;
        }
        
        return result;
    }
    
    private void validateUser(OneRosterEntity entity, ValidationResult result) {
        requireField(entity, "username", result);
        requireField(entity, "givenName", result);
        requireField(entity, "familyName", result);
        requireField(entity, "role", result);
        
        String role = (String) entity.getField("role");
        if (role != null && !Set.of("student", "teacher", "parent", "aide")
            .contains(role.toLowerCase())) {
            result.addError("role", "Invalid role: " + role);
        }
    }
    
    private void validateClass(OneRosterEntity entity, ValidationResult result) {
        requireField(entity, "title", result);
        requireField(entity, "classType", result);
        requireField(entity, "schoolSourcedId", result);
        requireField(entity, "courseSourcedId", result);
        
        String classType = (String) entity.getField("classType");
        if (classType != null && !Set.of("homeroom", "scheduled").contains(classType)) {
            result.addError("classType", "Invalid classType: " + classType);
        }
    }
    
    private void validateEnrollment(OneRosterEntity entity, ValidationResult result) {
        requireField(entity, "userSourcedId", result);
        requireField(entity, "classSourcedId", result);
        requireField(entity, "role", result);
    }
    
    private void validateResult(OneRosterEntity entity, ValidationResult result) {
        requireField(entity, "lineItemSourcedId", result);
        requireField(entity, "studentSourcedId", result);
        requireField(entity, "scoreStatus", result);
        
        String scoreStatus = (String) entity.getField("scoreStatus");
        if (scoreStatus != null && !Set.of("submitted", "graded", "pending", 
            "incomplete", "exempt").contains(scoreStatus)) {
            result.addError("scoreStatus", "Invalid scoreStatus: " + scoreStatus);
        }
    }
    
    private void requireField(OneRosterEntity entity, String field, 
                               ValidationResult result) {
        if (entity.getField(field) == null) {
            result.addError(field, field + " is required");
        }
    }
    
    @Override
    public JsonSchema getJsonSchema(String entityType) {
        // 返回 OneRoster 官方 JSON Schema
        return OneRosterSchemaRegistry.getSchema(entityType);
    }
    
    private Object extractFieldValue(InternalEntity entity, String fieldPath) {
        return BeanUtils.getProperty(entity, fieldPath);
    }
    
    private boolean isValidUUID(String value) {
        try {
            UUID.fromString(value);
            return true;
        } catch (IllegalArgumentException e) {
            return false;
        }
    }
}

@Component
@Slf4j
public class XApi103Adapter implements StandardAdapter<XApiStatement> {

    /**
     * PrimeTop 事件类型到 xAPI Verb 的映射
     */
    private static final Map<String, VerbMapping> VERB_MAP = Map.ofEntries(
        Map.entry("QUESTION_ANSWERED", VerbMapping.of(

    /**
     * 将内部学习行为事件转为 xAPI Statement
     */
    @Override
    public List<XApiStatement> toStandard(List<InternalEntity> events,
                                          MappingContext context) {
        List<XApiStatement> statements = new ArrayList<>();
        
        for (InternalEntity event : events) {
            String eventType = event.getEntityType();
            VerbMapping verbMapping = VERB_MAP.get(eventType);
            
            if (verbMapping == null) {
                log.warn("No xAPI verb mapping for event type: {}, skipping", eventType);
                continue;
            }
            
            XApiStatement statement = buildStatement(event, verbMapping, context);
            statements.add(statement);
        }

    private XApiStatement buildStatement(InternalEntity event,
                                          VerbMapping verbMapping,
                                          MappingContext context) {
        XApiStatement stmt = new XApiStatement();
        
        // Statement ID
        stmt.setId(UUID.randomUUID());
        
        // Timestamp (ISO 8601)
        stmt.setTimestamp(event.getCreatedAt().atZone(ZoneId.of("Asia/Shanghai"))
            .toOffsetDateTime().toString());
        
        // Actor (学生)
        XApiActor actor = XApiActor.builder()
            .objectType("Agent")
            .name(event.get("userName"))
            .account(XApiAccount.builder()
                .homePage("https://primetop.cn")
                .name(event.getUserId())
                .build())
            .build();
        stmt.setActor(actor);
        
        // Verb
        XApiVerb verb = XApiVerb.builder()
            .id(verbMapping.getVerbId())
            .display(Map.of("zh-CN", verbMapping.getVerbDisplay(), 
                            "en-US", verbMapping.getVerbDisplay()))
            .build();
        stmt.setVerb(verb);
        
        // Object
        XApiObject obj = XApiObject.builder()
            .objectType("Activity")
            .id(verbMapping.getActivityBase() + "/" + event.getEntityId())
            .definition(XApiDefinition.builder()
                .name(Map.of("zh-CN", (String) event.getOrDefault("title", event.getEntityId())))
                .description(Map.of("zh-CN", (String) event.getOrDefault("description", "")))
                .type(verbMapping.getActivityType())
                .build())
            .build();
        stmt.setObject(obj);
        
        // Result (答题类事件包含结果)
        if (event.hasField("score") || event.hasField("correct")) {
            XApiResult result = XApiResult.builder()
                .success((Boolean) event.get("correct"))
                .score(XApiScore.builder()
                    .raw(((Number) event.getOrDefault("score", 0)).doubleValue())
                    .min(0.0)
                    .max(((Number) event.getOrDefault("maxScore", 100)).doubleValue())
                    .scaled(((Number) event.getOrDefault("score", 0)).doubleValue() /
                            ((Number) event.getOrDefault("maxScore", 100)).doubleValue())
                    .build())
                .completion(true)
                .duration(formatDuration(event.get("durationMs")))
                .response((String) event.get("studentAnswer"))
                .build();
            stmt.setResult(result);
        }
        
        // Context (附加上下文信息)
        XApiContext xContext = XApiContext.builder()
            .registration(UUID.randomUUID())
            .contextActivities(XApiContextActivities.builder()
                .parent(buildParentActivities(event))
                .category(buildCategoryActivities(event))
                .build())
            .extensions(Map.of(
                "https://primetop.cn/xapi/extensions/grade",
                event.getOrDefault("grade", ""),
                "https://primetop.cn/xapi/extensions/subject",
                event.getOrDefault("subject", ""),
                "https://primetop.cn/xapi/extensions/textbook-version",
                event.getOrDefault("textbookVersion", ""),
                "https://primetop.cn/xapi/extensions/knowledge-points",
                event.getOrDefault("knowledgePoints", List.of())
            ))
            .build();
        stmt.setContext(xContext);
        
        // Authority
        stmt.setAuthority(XApiActor.builder()
            .objectType("Agent")
            .name("PrimeTop Learning Platform")
            .account(XApiAccount.builder()
                .homePage("https://primetop.cn")
                .name("system")
                .build())
            .build());
        
        return stmt;
    }
    
    /**
     * 构建父级活动（章节→单元→学科层级）
     */
    private List<XApiActivityRef> buildParentActivities(InternalEntity event) {
        List<XApiActivityRef> parents = new ArrayList<>();
        
        String chapterId = (String) event.get("chapterId");
        if (chapterId != null) {
            parents.add(XApiActivityRef.builder()
                .objectType("Activity")
                .id(ACTIVITY_BASE + "chapter/" + chapterId)
                .build());
        }
        
        String subjectId = (String) event.get("subjectId");
        if (subjectId != null) {
            parents.add(XApiActivityRef.builder()
                .objectType("Activity")
                .id(ACTIVITY_BASE + "subject/" + subjectId)
                .build());
        }
        
        return parents;
    }
    
    private String formatDuration(Object durationMs) {
        if (durationMs == null) return null;
        long ms = ((Number) durationMs).longValue();
        return "PT" + (ms / 1000) + "S";
    }
    
    @Override
    public List<InternalEntity> fromStandard(List<XApiStatement> statements,
                                              MappingContext context) {
        // xAPI 主要用于输出（导出学习行为给外部系统）
        // 导入场景较少，支持从 LRS 导入历史学习记录
        List<InternalEntity> results = new ArrayList<>();
        
        for (XApiStatement stmt : statements) {
            InternalEntity entity = new InternalEntity();
            entity.setEntityType("LEARNING_EVENT");
            entity.setEntityId(stmt.getId().toString());
            entity.setExternalId(stmt.getId().toString());
            entity.setSourceSystem(context.getSourceSystem());
            
            // 反向解析 Verb
            String verbId = stmt.getVerb().getId();
            String eventType = reverseMapVerb(verbId);
            entity.set("eventType", eventType);
            entity.set("userId", stmt.getActor().getAccount().getName());
            entity.set("timestamp", stmt.getTimestamp());
            
            // 解析 Result
            if (stmt.getResult() != null) {
                XApiScore score = stmt.getResult().getScore();
                if (score != null) {
                    entity.set("score", score.getRaw());
                    entity.set("maxScore", score.getMax());
                }
                entity.set("correct", stmt.getResult().getSuccess());
                entity.set("studentAnswer", stmt.getResult().getResponse());
            }
            
            results.add(entity);
        }
        
        return results;
    }
    
    private String reverseMapVerb(String verbId) {
        return VERB_MAP.entrySet().stream()
            .filter(e -> e.getValue().getVerbId().equals(verbId))
            .map(Map.Entry::getKey)
            .findFirst()
            .orElse("UNKNOWN_EVENT");
    }
    
    @Override

        if (statement.getObject() == null || statement.getObject().getId() == null) {
            result.addError("object", "Object with ID is required");
        }
        if (statement.getTimestamp() == null) {
            result.addError("timestamp", "Timestamp is required");
        }
        
        return result;
    }
    
    @Override

/**
 * 年级编码转换器
 * 内部: "7" (七年级) ↔ OneRoster grade: "G07"
 */
public class GradeLevelTransformer implements FieldTransformer {
    
    private static final Map<String, String> INTERNAL_TO_STANDARD = Map.of(
        "1", "G01", "2", "G02", "3", "G03",
        "4", "G04", "5", "G05", "6", "G06",
        "7", "G07", "8", "G08", "9", "G09",
        "10", "G10", "11", "G11", "12", "G12"
    );
    
    @Override
    public Object transform(Object value, String params) {
        if (value == null) return null;
        return INTERNAL_TO_STANDARD.getOrDefault(value.toString(), value);
    }
    
    @Override
    public Object reverse(Object value, String params) {
        if (value == null) return null;
        for (Map.Entry<String, String> e : INTERNAL_TO_STANDARD.entrySet()) {
            if (e.getValue().equals(value.toString())) {
                return e.getKey();
            }
        }
        return value;
    }
}

/**
 * 日期时间 ISO 8601 转换器
 */
public class DateTimeISO8601Transformer implements FieldTransformer {
    
    private static final DateTimeFormatter ISO_FORMATTER =
        DateTimeFormatter.ISO_OFFSET_DATE_TIME;
    private static final ZoneId DEFAULT_ZONE = ZoneId.of("Asia/Shanghai");
    
    @Override
    public Object transform(Object value, String params) {
        if (value == null) return null;
        if (value instanceof LocalDateTime) {
            return ((LocalDateTime) value).atZone(DEFAULT_ZONE)
                .format(ISO_FORMATTER);
        }
        if (value instanceof LocalDate) {
            return ((LocalDate) value).format(DateTimeFormatter.ISO_DATE);
        }
        return value;
    }
    
    @Override
    public Object reverse(Object value, String params) {
        if (value == null) return null;
        String str = value.toString();
        try {
            return OffsetDateTime.parse(str, ISO_FORMATTER)
                .atZoneSameInstant(DEFAULT_ZONE)
                .toLocalDateTime();
        } catch (DateTimeParseException e) {
            try {
                return LocalDate.parse(str);
            } catch (DateTimeParseException e2) {
                return null;
            }
        }
    }
}

/**
 * 手机号脱敏转换器

public class PhoneRedactTransformer implements FieldTransformer {
    
    @Override
    public Object transform(Object value, String params) {
        if (value == null) return null;
        String phone = value.toString();
        if (phone.length() >= 11) {
            return phone.substring(0, 3) + "****" + phone.substring(7);
        }
        return "***";
    }
    
    @Override
    public Object reverse(Object value, String params) {
        // 脱敏不可逆
        return value;
    }

    public Object transform(Object value, String params) {
        if (value == null) return null;
        return CODE_MAP.getOrDefault(value.toString(), value);
    }
    
    @Override
    public Object reverse(Object value, String params) {
        if (value == null) return null;
        return CODE_MAP.entrySet().stream()
            .filter(e -> e.getValue().equals(value.toString()))
            .map(Map.Entry::getKey)
            .findFirst()
            .orElse(value);
    }
}

/**
 * 授权校验处理器

@Component
@Order(1)
public class ConsentValidationHandler implements PipelineHandler {
    
    @Autowired
    private DataUsageConsentService consentService;
    
    @Override
    public String getStageName() {
        return "CONSENT_VALIDATION";
    }
    
    @Override
    public PipelineContext handle(PipelineContext context) {
        long start = System.currentTimeMillis();
        DataExchangeJob job = context.getJob();
        
        // 导出任务需要授权校验
        if ("EXPORT".equals(job.getJobType()) || 
            "BIDIRECTIONAL".equals(job.getDirection())) {
            
            String consentId = job.getConsentId();
            if (consentId == null) {
                context.getErrors().add("No consent ID provided for export job");
                return context;
            }
            
            DataUsageConsent consent = consentService.getById(consentId);
            if (consent == null) {
                context.getErrors().add("Consent not found: " + consentId);
                return context;
            }
            
            if (!"GRANTED".equals(consent.getStatus())) {
                context.getErrors().add(
                    "Consent is not active, current status: " + consent.getStatus());
                return context;
            }
            
            if (consent.getExpiresAt() != null && 
                consent.getExpiresAt().isBefore(LocalDateTime.now())) {
                context.getErrors().add("Consent has expired");
                return context;
            }
            
            // 校验授权范围是否覆盖请求的数据类型
            List<String> consentScopes = JsonUtils.parseArray(consent.getConsentScopes());
            List<String> requestedScopes = extractRequestedScopes(job);
            
            for (String scope : requestedScopes) {
                if (!consentScopes.contains(scope)) {
                    context.getErrors().add(
                        "Consent does not cover scope: " + scope);
                }
            }
            
            context.getMappingContext().setConsentId(consentId);
            context.getMappingContext().setRedactSensitive(
                Boolean.TRUE.equals(job.getRedactSensitive()));
        }
        
        context.recordStageTime(getStageName(), 
            System.currentTimeMillis() - start);
        return context;
    }
    
    @Override
    public boolean supports(DataExchangeJob job) {
        return true; // 所有任务都需要经过授权校验
    }
}

@Component
@Order(2)
public class DataExtractionHandler implements PipelineHandler {
    
    @Autowired
    private DataExtractionService extractionService;
    
    @Override
    public String getStageName() {
        return "DATA_EXTRACTION";
    }
    
    @Override
    public PipelineContext handle(PipelineContext context) {
        long start = System.currentTimeMillis();
        DataExchangeJob job = context.getJob();
        
        // 分页拉取数据，避免内存溢出
        int pageSize = 500;
        int pageNum = 0;
        List<InternalEntity> allEntities = new ArrayList<>();
        
        while (true) {
            Page<InternalEntity> page = extractionService.extractData(
                job.getFilterCriteria(),
                job.getScopeType(),
                job.getSinceTime(),
                job.getUntilTime(),
                pageNum, pageSize);
            
            allEntities.addAll(page.getRecords());
            
            // 更新进度
            job.setTotalRecords((long) allEntities.size());
            
            if (!page.hasNext()) break;
            pageNum++;
        }
        
        context.setInternalEntities(allEntities);
        context.recordStageTime(getStageName(),
            System.currentTimeMillis() - start);
        
        return context;
    }
    
    @Override
    public boolean supports(DataExchangeJob job) {
        return "EXPORT".equals(job.getJobType()) || 
               "SYNC".equals(job.getJobType());
    }
}
