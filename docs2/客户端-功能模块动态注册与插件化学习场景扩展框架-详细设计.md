# 客户端-功能模块动态注册与插件化学习场景扩展框架 详细设计

## 1. 概述

### 1.1 模块定位

PrimeTop 作为覆盖幼儿至高中全学段、全学科的 AI 辅助学习 APP，需要持续迭代新增学习场景（如新学科工具、新练习类型、新专题模块、运营活动页面等）。传统硬编码方式导致每新增一个功能就需要修改导航配置、首页布局、路由表、埋点注册等多处代码，开发效率低、耦合风险高。

本框架提供一套 **插件化的功能模块注册与扩展机制**，使每个学习功能模块可以：

1. **声明式注册**：通过模块清单（Manifest）声明自身提供的页面、路由、首页卡片、导航入口、快捷操作等。
2. **动态发现与装配**：主框架在启动时扫描已注册模块，自动构建导航结构、首页 Feed、快捷入口面板。
3. **隔离与降级**：模块间通过明确的接口契约通信，单个模块崩溃不影响其他模块。
4. **按需加载**：支持 Flutter Deferred Components / iOS On-Demand Resources / Android Dynamic Feature Modules 实现模块级动态分发。

### 1.2 核心职责

| 职责 | 说明 |
|------|------|
| 模块注册中心 | 统一管理所有功能模块的生命周期 |
| 导航贡献 | 模块声明导航 Tab、路由、深链接 |
| 首页 Feed 贡献 | 模块向首页学习工作台提供内容卡片 |
| 快捷入口贡献 | 模块向首页快捷面板提供入口 |
| 跨模块通信 | 定义模块间松耦合的调用契约 |
| 动态加载 | 支持按需下载和加载模块资源 |
| 运行时管控 | 配合灰度/特性开关控制模块可见性 |

```dart
/// 所有功能模块必须实现的抽象接口
abstract class LearningModule {
  /// 模块清单
  ModuleManifest get manifest;

  /// 模块初始化（异步）
  /// 在此方法中完成：数据库初始化、服务注册、数据预加载等
  Future<void> initialize(ModuleContext context);

  /// 模块激活（变为可见可交互）
  /// 在此方法中完成：注册事件监听、启动定时器、刷新数据等
  Future<void> activate(ModuleContext context);

  /// 模块停用（变为不可见）
  /// 在此方法中完成：取消事件监听、停止定时器、保存状态等
  Future<void> deactivate(ModuleContext context);

  /// 模块卸载（释放所有资源）
  Future<void> dispose(ModuleContext context);

  /// 处理来自其他模块的消息（可选）
  Future<dynamic> handleModuleMessage(ModuleMessage message) async => null;
}
```

```dart
/// 跨模块事件总线 —— 模块间松耦合通信的核心
class ModuleEventBus {
  static final ModuleEventBus _instance = ModuleEventBus._();
  factory ModuleEventBus() => _instance;
  ModuleEventBus._();

  final _controller = StreamController<ModuleEvent>.broadcast();

  /// 发布事件
  void emit(ModuleEvent event) {
    _controller.add(event);
  }

  /// 订阅特定类型的事件
  StreamSubscription<ModuleEvent> on(String eventType, Function(ModuleEvent) handler) {
    return _controller.stream
        .where((e) => e.type == eventType)
        .listen(handler as Function(ModuleEvent));
  }

  /// 订阅所有事件（调试用）
  Stream<ModuleEvent> get allEvents => _controller.stream;
}

/// 跨模块事件
class ModuleEvent {
  final String type;              // 事件类型
  final ModuleId sourceModule;    // 发送方模块ID
  final Map<String, dynamic> payload; // 事件数据
  final DateTime timestamp;       // 发送时间

  ModuleEvent({
    required this.type,
    required this.sourceModule,
    this.payload = const {},
    DateTime? timestamp,
  }) : timestamp = timestamp ?? DateTime.now();
}
```

### 3.4 模块间直接调用

```dart
/// 模块间通信服务 —— 用于需要同步获取结果的跨模块调用
class ModuleCommunication {
  final Map<ModuleId, LearningModule> _modules;

  /// 向指定模块发送消息并获取响应
  Future<T?> sendMessage<T>(
    ModuleId targetModule,
    String action, {
    Map<String, dynamic>? params,
  }) async {
    final module = _modules[targetModule];
    if (module == null) {
      throw ModuleNotFoundException(targetModule);
    }
    final message = ModuleMessage(
      action: action,
      params: params ?? {},
    );
    final result = await module.handleModuleMessage(message);
    return result as T?;
  }

  /// 检查模块是否可用
  bool isModuleAvailable(ModuleId moduleId) {
    return _modules[moduleId] != null;
  }
}

/// 模块消息
class ModuleMessage {
  final String action;            // 操作名称
  final Map<String, dynamic> params; // 参数

  ModuleMessage({required this.action, this.params = const {}});
}
```

### 4.2 模块注册与加载流程

```dart
/// 模块注册中心核心实现
class ModuleRegistry {
  // ... (字段定义见 2.2)

  /// 注册模块
  Future<void> register(LearningModule module) async {
    final manifest = module.manifest;
    final moduleId = manifest.id;

    if (_modules.containsKey(moduleId)) {
      _log.warning('Module $moduleId already registered, skipping.');
      return;
    }

    // 1. 检查特性开关
    if (!await _featureFlags.isEnabled('module.$moduleId')) {
      _log.info('Module $moduleId disabled by feature flag.');
      return;
    }

    // 2. 检查学段适配
    final userStage = _userContext.currentStage;
    if (!manifest.supportedStages.contains(userStage)) {
      _log.info('Module $moduleId not supported for stage $userStage.');
      return;
    }

    // 3. 检查依赖
    final deps = manifest.dependencies;
    if (deps != null) {
      for (final reqId in deps.required) {
        if (!_modules.containsKey(reqId) || _modules[reqId]!.status != ModuleStatus.ready) {
          throw ModuleDependencyException(
            'Module $moduleId requires $reqId which is not ready.'
          );
        }
      }
    }

    // 4. 创建注册记录
    final registered = RegisteredModule(
      manifest: manifest,
      status: ModuleStatus.registered,
      registeredAt: DateTime.now(),
      moduleInstance: module,
    );
    _modules[moduleId] = registered;
    _categoryIndex.putIfAbsent(manifest.category, () => []).add(moduleId);

    // 5. 根据加载策略决定是否立即初始化
    if (manifest.loadStrategy == ModuleLoadStrategy.eager) {
      await _initializeModule(registered);
    }
  }

  /// 初始化模块
  Future<void> _initializeModule(RegisteredModule module) async {
    final moduleId = module.manifest.id;

    try {
      module.status = ModuleStatus.initializing;
      _notifyLoadEvent(ModuleLoadEvent(moduleId, ModuleStatus.initializing));

      final context = _buildModuleContext(module);
      await (module.moduleInstance as LearningModule).initialize(context);

      module.status = ModuleStatus.ready;
      module.loadedAt = DateTime.now();
      _notifyLoadEvent(ModuleLoadEvent(moduleId, ModuleStatus.ready));

      _log.info('Module $moduleId initialized successfully.');

    } catch (e, stackTrace) {
      module.status = ModuleStatus.error;
      _notifyLoadEvent(ModuleLoadEvent(
        moduleId,
        ModuleStatus.error,
        error: e.toString(),
      ));

      _log.error('Module $moduleId initialization failed.', e, stackTrace);

      // 降级处理：记录错误，但不影响其他模块
      _errorReporter.report(
        ModuleInitError(moduleId: moduleId, error: e, stackTrace: stackTrace),
      );
    }
  }

  /// 激活模块
  Future<void> activateModule(ModuleId moduleId) async {
    final module = _modules[moduleId];
    if (module == null) {
      throw ModuleNotFoundException(moduleId);
    }

    if (module.status == ModuleStatus.ready ||
        module.status == ModuleStatus.deactivated) {
      final context = _buildModuleContext(module);
      await (module.moduleInstance as LearningModule).activate(context);
      module.status = ModuleStatus.activated;

      _eventBus.emit(ModuleEvent(
        type: ModuleEventTypes.moduleActivated,
        sourceModule: moduleId,
        payload: {'version': module.manifest.version},
      ));
    }
  }

  /// 停用模块
  Future<void> deactivateModule(ModuleId moduleId) async {
    final module = _modules[moduleId];
    if (module == null) return;

    if (module.status == ModuleStatus.activated) {
      final context = _buildModuleContext(module);
      await (module.moduleInstance as LearningModule).deactivate(context);
      module.status = ModuleStatus.deactivated;

      _eventBus.emit(ModuleEvent(
        type: ModuleEventTypes.moduleDeactivated,
        sourceModule: moduleId,
      ));
    }
  }

  /// 卸载模块（释放资源）
  Future<void> unloadModule(ModuleId moduleId) async {
    final module = _modules[moduleId];
    if (module == null) return;

    if (module.status == ModuleStatus.activated) {
      await deactivateModule(moduleId);
    }

    final context = _buildModuleContext(module);
    await (module.moduleInstance as LearningModule).dispose(context);
    module.status = ModuleStatus.unloaded;

    _modules.remove(moduleId);
    _categoryIndex[module.manifest.category]?.remove(moduleId);
  }
}
```

```dart
/// 首页学习工作台根据已激活模块动态组装内容卡片
class HomeFeedAssembler {
  final ModuleRegistry _registry;
  final RecommendationEngine _recommender;

  HomeFeedAssembler(this._registry, this._recommender);

  /// 组装首页内容卡片流
  Future<List<HomeCard>> assemble({
    required UserContext user,
    int maxCards = 10,
  }) async {
    final candidates = <HomeCardCandidate>[];

    // 1. 收集所有模块贡献的首页卡片
    for (final module in _registry.activatedModules) {
      final cardContributions = module.manifest.homeCards;
      if (cardContributions == null) continue;

      for (final contribution in cardContributions) {
        // 检查可见性
        if (!_checkVisibility(contribution.visibilityRule, user)) {
          continue;
        }

        // 获取卡片数据
        try {
          final data = await contribution.dataProvider(user);
          if (data == null) continue; // 无数据则跳过

          candidates.add(HomeCardCandidate(
            contribution: contribution,
            data: data,
            module: module.manifest.id,
          ));
        } catch (e) {
          _log.error('Card data provider failed for ${contribution.cardId}', e);
        }
      }
    }

    // 2. 排序：按优先级 + 推荐引擎打分
    candidates.sort((a, b) {
      final scoreA = a.contribution.defaultPriority + _recommender.score(a, user);
      final scoreB = b.contribution.defaultPriority + _recommender.score(b, user);
      return scoreB.compareTo(scoreA);
    });

    // 3. 去重与多样性保障
    final result = _ensureDiversity(candidates.take(maxCards).toList());

    return result.map((c) => HomeCard(
      id: c.contribution.cardId,
      type: c.contribution.type,
      data: c.data,
      builder: c.contribution.widgetBuilder,
      module: c.module,
    )).toList();
  }

  /// 多样性保障：同一类型卡片不超过 maxPerType 张
  List<HomeCardCandidate> _ensureDiversity(
    List<HomeCardCandidate> cards, {
    int maxPerType = 2,
  }) {
    final typeCount = <HomeCardType, int>{};
    final result = <HomeCardCandidate>[];

    for (final card in cards) {
      final type = card.contribution.type;
      final count = typeCount[type] ?? 0;
      if (count < maxPerType) {
        result.add(card);
        typeCount[type] = count + 1;
      }
    }

    return result;
  }
}
```

```dart
/// 延迟加载模块的下载管理
class DeferredModuleLoader {
  final ResourcePackManager _packManager;
  final ModuleRegistry _registry;

  /// 确保模块资源已下载
  Future<bool> ensureDownloaded(ModuleId moduleId) async {
    final module = _registry.getModule(moduleId);
    if (module == null) return false;
    if (module.manifest.loadStrategy != ModuleLoadStrategy.deferred) {
      return true; // 非延迟模块无需下载
    }

    // 检查本地缓存
    final cached = await _readCache(moduleId);
    if (cached?.downloaded == true && cached?.version == module.manifest.version) {
      return true; // 已下载且版本匹配
    }

    // 下载资源包
    try {
      final packUrl = await _resolveModulePackUrl(moduleId, module.manifest.version);
      await _packManager.downloadAndUnpack(
        url: packUrl,
        targetPath: '/modules/$moduleId',
        version: module.manifest.version,
        onProgress: (received, total) {
          _registry.notifyDownloadProgress(moduleId, received, total);
        },
      );

      // 更新缓存
      await _updateCache(moduleId, ModuleStateCache(
        moduleId: moduleId,
        enabled: true,
        downloaded: true,
        version: module.manifest.version,
        launchCount: cached?.launchCount ?? 0,
        lastUsedAt: DateTime.now(),
        config: {},
      ));

      return true;
    } catch (e) {
      _log.error('Failed to download module $moduleId', e);
      return false;
    }
  }

  /// 预下载模块（在WiFi + 充电时）
  Future<void> preloadModules(List<ModuleId> moduleIds) async {
    final connectivity = await _connectivityChecker.check();
    if (!connectivity.isWifi || !connectivity.isCharging) return;

    for (final id in moduleIds) {
      await ensureDownloaded(id);
    }
  }
}
```

### 6.2 错误处理策略

```dart
/// 模块错误处理器
class ModuleErrorHandler {
  final Logger _log;
  final CrashReporter _crashReporter;
  final FeatureFlags _flags;

  /// 处理模块错误
  Future<void> handleError(ModuleException error) async {
    switch (error.runtimeType) {
      case ModuleNotFoundException:
        _log.warning(error.message);
        // 静默降级：不展示该模块的入口
        break;

      case ModuleDependencyException:
        _log.error(error.message);
        // 尝试延迟重试：将模块加入待初始化队列
        _scheduleRetry(error.moduleId, delay: const Duration(minutes: 1));
        break;

      case ModuleInitError:
        _log.error('Module init failed', error);
        await _crashReporter.report(error);
        // 标记模块为错误状态，通知用户
        _notifyModuleError(error.moduleId!, '模块初始化失败，请稍后重试');
        break;

      case ModuleDownloadException:
        _log.warning(error.message);
        // 提示用户网络问题
        _notifyModuleError(
          error.moduleId!,
          '资源包下载失败，请检查网络后重试',
          actionable: true,
        );
        break;

      case ModuleVersionException:
        _log.error(error.message);
        // 强制更新模块资源包
        await _forceModuleUpdate(error.moduleId!);
        break;
    }
  }

  /// 调度重试
  Timer _scheduleRetry(ModuleId? moduleId, {required Duration delay}) {
    return Timer(delay, () {
      if (moduleId != null) {
        _registry.retryInitialization(moduleId);
      }
    });
  }
}
```

### 6.3 降级方案

| 异常场景 | 降级策略 | 用户感知 |
|---------|---------|---------|
| 模块初始化失败 | 隐藏模块入口，显示通用提示 | "该功能暂时不可用" |
| 模块资源包下载失败 | 显示下载重试按钮 | "资源下载失败，点击重试" |
| 模块页面渲染崩溃 | 显示模块级 Error Boundary 降级页 | "该功能出了点问题" + 重试按钮 |
| 模块依赖未就绪 | 延迟激活，日志记录 | 无感（入口暂不显示） |
| 模块间通信超时 | 返回默认值或空值，调用方自行处理 | 可能看到"暂无数据" |
| 特性开关关闭模块 | 模块不注册，入口不展示 | 无感 |

### 7.2 缓存与预加载策略

| 加载策略 | 适用场景 | 预加载条件 | 缓存策略 |
|---------|---------|-----------|---------|
| `eager` | P0 核心模块（AI辅导、拍题、用户） | APP启动时立即初始化 | 常驻内存 |
| `lazy` | P1 重要模块（学情、错题、规划） | 首次导航到该模块时 | LRU（最少最近使用） |
| `deferred` | P2 增强模块（实验模拟、专题） | 用户首次点击 + 资源包已下载 | 可卸载释放内存 |

```dart
/// 首页卡片并行数据加载
class ParallelCardLoader {
  /// 并行加载所有卡片数据，超时自动降级
  Future<List<HomeCard>> loadCards(
    List<HomeCardContribution> contributions, {
    required UserContext user,
    Duration timeout = const Duration(seconds: 2),
    int maxConcurrency = 5,
  }) async {
    // 使用池化并行加载
    final pool = ThreadPool(maxConcurrency: maxConcurrency);
    final futures = contributions.map((c) {
      return pool.submit(() => c.dataProvider(user).timeout(
        timeout,
        onTimeout: () => null, // 超时返回null，跳过该卡片
      ));
    });

    final results = await Future.wait(futures);

    // 过滤掉无数据或超时的卡片
    return List.generate(contributions.length, (i) {
      final data = results[i];
      if (data == null) return null;
      return HomeCard(
        id: contributions[i].cardId,
        type: contributions[i].type,
        data: data,
        builder: contributions[i].widgetBuilder,
      );
    }).whereType<HomeCard>().toList();
  }
}
```

### 7.4 内存管理

```dart
/// 内存压力响应：在低内存时卸载非活跃模块
class MemoryAwareModuleManager {
  final ModuleRegistry _registry;
  final MemoryInfoProvider _memoryInfo;

  /// 注册内存压力回调
  void setup() {
    _memoryInfo.onMemoryWarning.listen((level) {
      switch (level) {
        case MemoryWarningLevel.moderate:
          // 中度内存压力：卸载所有deactivated的deferred模块
          _unloadInactiveDeferredModules();
          break;

        case MemoryWarningLevel.critical:
          // 严重内存压力：卸载所有非核心模块
          _unloadAllNonCoreModules();
          // 清理图片缓存
          ImageCacheProvider.clear();
          break;

        case MemoryWarningLevel.severe:
          // 极端内存压力：仅保留当前可见模块
          _unloadAllExceptActive();
          break;
      }
    });
  }

  void _unloadInactiveDeferredModules() {
    for (final module in _registry.modules.values) {
      if (module.status == ModuleStatus.deactivated &&
          module.manifest.loadStrategy == ModuleLoadStrategy.deferred) {
        _registry.unloadModule(module.manifest.id);
      }
    }
  }
}
```

```dart
/// 延迟加载模块资源包的完整性校验
class ModulePackIntegrity {
  /// 校验资源包
  Future<bool> verify(ModuleId moduleId, String packPath) async {
    // 1. 校验文件哈希
    final expectedHash = await _fetchExpectedHash(moduleId);
    final actualHash = await _computeFileHash(packPath);
    if (expectedHash != actualHash) {
      _log.error('Module $moduleId pack hash mismatch.');
      return false;
    }

    // 2. 校验签名（防止篡改）
    final signature = await _fetchSignature(moduleId);
    final isValid = await _verifySignature(packPath, signature);
    if (!isValid) {
      _log.error('Module $moduleId pack signature invalid.');
      return false;
    }

    return true;
  }

  Future<String> _computeFileHash(String path) async {
    final bytes = await File(path).readAsBytes();
    return sha256.convert(bytes).toString();
  }
}
```
