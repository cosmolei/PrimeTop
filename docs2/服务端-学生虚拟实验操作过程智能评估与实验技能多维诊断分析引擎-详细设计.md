# 服务端-学生虚拟实验操作过程智能评估与实验技能多维诊断分析引擎-详细设计

## 1. 概述

### 1.1 功能定位

本引擎是 PrimeTop 教育平台虚拟实验系统的**核心评估中枢**，负责对学生在虚拟实验环境中的操作过程进行实时跟踪、智能评估、多维诊断和个性化反馈。

与现有系统的关系：
- **虚拟实验模拟操作与数据记录 E2E**：负责实验操作的前端交互和数据记录管线
- **实验数据管理与虚拟实验参数配置服务**：负责实验参数、器材、药品等基础数据管理
- **学科实验报告生成与数据分析辅助服务**：负责实验后的报告生成和数据分析

本引擎专注于**操作过程的智能评估**这一核心环节，填补从"操作记录"到"报告生成"之间的评估空白，构建完整的虚拟实验教学闭环。

### 1.2 设计目标

| 目标 | 描述 |
| --- | --- |
| 实时评估 | 学生操作虚拟实验时，对每一步操作进行毫秒级合理性判断 |
| 多维诊断 | 从规范性、安全性、准确性、效率、理解度5个维度评估实验技能 |
| 智能反馈 | 在关键节点提供适时提示、纠错建议和安全警告 |
| 画像积累 | 长期跟踪学生实验能力成长轨迹，形成实验素养画像 |
| 个性化推荐 | 基于诊断结果推荐针对性练习和改进方案 |

### 1.3 适用范围

- 学段：小学科学、初中物化生、高中物化生
- 实验类型：演示实验、操作实验、探究实验、分组实验
- 调用方：客户端虚拟实验组件、教师端实验管理系统、学情分析系统

```typescript
/** 实验操作事件 */
interface ExperimentActionEvent {
  eventId: string;                  // 事件唯一ID
  sessionId: string;                // 实验会话ID
  experimentId: string;             // 实验模板ID
  studentId: string;                // 学生ID
  timestamp: number;                // 事件时间戳(ms)
  actionType: ActionType;           // 操作类型
  apparatusId: string;              // 操作涉及的器材ID
  targetApparatusId?: string;       // 操作目标器材ID（如倒入烧杯）

}

type ActionType =
  | 'PICK_UP'          // 取用器材
  | 'PLACE_DOWN'       // 放置器材
  | 'POUR'             // 倾倒（液体转移）
  | 'MEASURE'          // 测量/读数
  | 'HEAT'             // 加热
  | 'STIR'             // 搅拌
  | 'CONNECT'          // 连接器材（如导管连接）
  | 'DISCONNECT'       // 断开连接
  | 'OBSERVE'          // 观察记录
  | 'ADJUST'           // 调节参数（温度、电压等）
  | 'ADD_REAGENT'      // 添加试剂
  | 'REMOVE'           // 移除器材/废弃物
  | 'RECORD'           // 记录数据
  | 'CLEAN'            // 清洗器材
  | 'OTHER';           // 其他操作

interface ActionParameter {
  // 通用参数
  duration?: number;       // 操作持续时间(ms)
  quantity?: number;       // 操作量(ml, g, 滴数等)
  temperature?: number;    // 温度(℃)
  // 倾倒参数

  pourAngle?: number;      // 倾倒角度(°)
  // 加热参数

  // 测量参数
  measuredValue?: number;  // 测量读数

}

/** 器材状态快照 */
interface ApparatusSnapshot {

  unit?: string;           // 单位

  state?: 'SOLID' | 'LIQUID' | 'GAS' | 'MIXTURE';
  color?: string;          // 颜色描述
  isActive: boolean;       // 是否在工作状态
}

  | 'BEAKER'           // 烧杯
  | 'TEST_TUBE'        // 试管
  | 'FLASK'            // 锥形瓶/圆底烧瓶
  | 'GRADUATED_CYLINDER' // 量筒
  | 'PIPET'            // 移液管
  | 'BURET'            // 滴定管
  | 'BALANCE'          // 天平
  | 'THERMOMETER'      // 温度计
  | 'BURNER'           // 酒精灯/本生灯
  | 'STAND'            // 铁架台
  | 'WIRE'             // 导线
  | 'METER'            // 电表
  | 'SWITCH'           // 开关
  | 'LENS'             // 透镜

/** 操作评估结果 */
interface ActionEvaluation {
  eventId: string;
  experimentSessionId: string;
  actionType: ActionType;
  // 合规性评估
  isCorrect: boolean;                 // 操作是否正确
  correctnessScore: number;           // 正确度评分 0-100
  // 安全评估
  safetyLevel: 'SAFE' | 'CAUTION' | 'WARNING' | 'DANGER';
  safetyIssues: SafetyIssue[];        // 安全问题列表
  // 规范性评估
  regulationCompliance: number;       // 规范符合度 0-100
  regulationIssues: string[];         // 规范性问题描述
  // 效率评估
  efficiencyScore: number;            // 效率评分 0-100
  isRedundant: boolean;               // 是否为冗余操作
  // 时机评估
  isOptimalTiming: boolean;           // 操作时机是否最优
  // 反馈
  immediateFeedback?: FeedbackMessage;// 即时反馈消息
  // 知识点关联
  relatedKnowledgePoints: string[];   // 关联知识点ID
  // 错误分类
  errorType?: ExperimentErrorType;    // 错误类型（如有）

type ExperimentErrorType =
  | 'WRONG_ORDER'              // 操作顺序错误
  | 'WRONG_APPARATUS'          // 器材选择错误

  | 'EXCESSIVE_QUANTITY'       // 用量过多
  | 'INSUFFICIENT_QUANTITY'    // 用量不足
  | 'UNSAFE_OPERATION'         // 不安全操作
  | 'MISSING_STEP'             // 缺少必要步骤
  | 'REDUNDANT_STEP'           // 冗余操作
  | 'WRONG_PARAMETER'          // 参数错误（温度过高等）
  | 'OBSERVATION_ERROR'        // 观察记录错误
  | 'MEASUREMENT_ERROR'        // 测量读数错误

interface SafetyIssue {
  level: 'CAUTION' | 'WARNING' | 'DANGER';

  description: string;
  correctAction: string;
  knowledgePointId?: string;
}

type SafetyCategory =
  | 'CHEMICAL_HAZARD'    // 化学品危险
  | 'HEAT_HAZARD'        // 热危险
  | 'ELECTRICAL_HAZARD'  // 电危险
  | 'GLASS_HAZARD'       // 玻璃器材危险
  | 'EYE_SAFETY'         // 眼部安全
  | 'GAS_HAZARD'         // 气体危险
  | 'WASTE_DISPOSAL';    // 废物处理

interface FeedbackMessage {
  type: 'PRAISE' | 'HINT' | 'WARNING' | 'ERROR' | 'INFO';
  content: string;
  visualCue?: 'GREEN_GLOW' | 'YELLOW_FLASH' | 'RED_FLASH' | 'SHAKE';
  audioCue?: 'SUCCESS' | 'WARNING_BEEP' | 'ERROR_BUZZ';
  autoDismissMs?: number;    // 自动消失时间
  actionableHint?: string;   // 可执行的纠正建议
}

/** 实验技能多维评估报告 */
interface ExperimentSkillReport {
  reportId: string;
  sessionId: string;
  studentId: string;
  experimentId: string;
  // 综合评分
  overallScore: number;              // 综合得分 0-100
  grade: 'A' | 'B' | 'C' | 'D' | 'F';
  // 五维评估
  dimensionScores: SkillDimensions;
  // 操作统计
  totalActions: number;
  correctActions: number;
  errorActions: number;
  redundantActions: number;
  duration: number;                  // 实验总时长(ms)
  // 诊断结果
  strengths: string[];               // 优势项
  weaknesses: string[];              // 薄弱项
  // 改进建议
  recommendations: ExperimentRecommendation[];
  // 知识点掌握度影响
  knowledgePointImpacts: KnowledgePointImpact[];
  generatedAt: string;               // 报告生成时间ISO
}

interface SkillDimensions {
  correctness: number;       // 准确性：操作步骤和结果是否正确
  safety: number;            // 安全性：操作过程的安全意识
  regulation: number;        // 规范性：是否符合实验操作规范
  efficiency: number;        // 效率：操作简洁度与时间管理
  understanding: number;     // 理解度：对实验原理的理解程度
}

interface ExperimentRecommendation {
  type: 'RETRY_EXPERIMENT' | 'WATCH_DEMO' | 'REVIEW_KNOWLEDGE' | 'PRACTICE_SKILL' | 'ADVANCED_CHALLENGE';
  title: string;
  description: string;
  targetUrl?: string;
  knowledgePointIds?: string[];
  priority: number;          // 1-5, 1为最高
}

interface KnowledgePointImpact {
  knowledgePointId: string;
  knowledgePointName: string;
  subject: string;
  beforeMastery: number;     // 实验前掌握度 0-1
  afterMastery: number;      // 实验后掌握度 0-1
  evidenceActions: string[]; // 支撑掌握度变化的事件ID
}

/** 实验标准操作协议 (Standard Operating Procedure) */
interface ExperimentSOP {
  sopId: string;
  experimentId: string;
  experimentName: string;
  subject: 'PHYSICS' | 'CHEMISTRY' | 'BIOLOGY' | 'SCIENCE';
  // 标准步骤序列
  standardSteps: StandardStep[];
  // 关键节点（这些节点的正确执行对实验成功至关重要）
  criticalCheckpoints: string[];     // stepId列表
  // 禁止操作
  forbiddenActions: ForbiddenAction[];
  // 安全要求
  safetyRequirements: SafetyRequirement[];
  // 评分权重
  scoringWeights: ScoringWeights;
  // 弹性配置
  allowRedo: boolean;                // 是否允许重做
  allowHint: boolean;                // 是否允许提示
  timeLimit?: number;                // 时间限制(ms)
}

interface StandardStep {
  stepId: string;
  stepNumber: number;
  description: string;
  expectedActionType: ActionType;
  expectedParameters: ActionParameter;
  // 可接受的参数范围（用于容错判断）
  parameterTolerance?: ParameterTolerance;
  // 前置步骤（必须完成后才能执行此步骤）
  prerequisites: string[];
  // 关键知识点
  knowledgePointIds: string[];
  // 常见错误
  commonErrors: CommonError[];
  // 提示
  hint: string;
  // 是否为关键安全步骤
  isSafetyCritical: boolean;
}

interface ParameterTolerance {
  quantityTolerance?: number;     // 用量误差范围(±)
  temperatureTolerance?: number;  // 温度误差范围(±)
  durationTolerance?: number;     // 时间误差范围(±)
  measuredValueTolerance?: number;// 读数误差范围(±)
}

interface CommonError {
  errorType: ExperimentErrorType;
  condition: string;              // 错误发生的条件描述
  feedback: string;               // 给学生的反馈
  severity: 'LOW' | 'MEDIUM' | 'HIGH';
}

interface ForbiddenAction {
  actionType: ActionType;
  condition: string;              // 禁止条件描述
  reason: string;                 // 禁止原因
  safetyLevel: 'WARNING' | 'DANGER';
  interventionMessage: string;    // 干预消息
}

interface SafetyRequirement {
  category: SafetyCategory;
  requirement: string;
  checkMethod: 'PRE_CHECK' | 'REALTIME' | 'POST_CHECK';
  apparatusIds?: string[];
}

interface ScoringWeights {
  correctness: number;     // 权重，总和为1.0
  safety: number;
  regulation: number;
  efficiency: number;
  understanding: number;
}
```

### 2.2 状态流转

实验评估会话状态机：

```
┌─────────────┐     开始实验      ┌──────────────┐
│  CREATED    │ ───────────────▶ │  IN_PROGRESS │
└─────────────┘                  └──────┬───────┘
                                        │
                          ┌─────────────┼─────────────┐
                          │             │             │
                    正常完成       用户中断      危险操作拦截
                          │             │             │
                          ▼             ▼             ▼
                   ┌──────────┐  ┌──────────┐  ┌───────────┐
                   │EVALUATING│  │ABORTED   │  │INTERCEPTED│
                   └─────┬────┘  └──────────┘  └─────┬─────┘
                         │                           │
                    评估完成                      用户纠正后
                         │                           │
                         ▼                           │
                   ┌──────────┐                      │
                   │ COMPLETED│◀─────────────────────┘
                   └──────────┘
```

```typescript
enum ExperimentSessionStatus {
  CREATED = 'CREATED',                // 会话已创建
  IN_PROGRESS = 'IN_PROGRESS',        // 实验进行中
  EVALUATING = 'EVALUATING',          // 正在生成评估报告
  COMPLETED = 'COMPLETED',            // 实验正常完成
  ABORTED = 'ABORTED',                // 用户主动中断
  INTERCEPTED = 'INTERCEPTED',        // 危险操作被拦截，等待纠正
  TIMEOUT = 'TIMEOUT',                // 超时
}

// 状态转换规则
const sessionTransitions: Record<ExperimentSessionStatus, ExperimentSessionStatus[]> = {
  CREATED: ['IN_PROGRESS'],
  IN_PROGRESS: ['EVALUATING', 'ABORTED', 'INTERCEPTED', 'TIMEOUT'],
  INTERCEPTED: ['IN_PROGRESS', 'ABORTED'],  // 纠正后继续，或放弃
  EVALUATING: ['COMPLETED'],
  COMPLETED: [],
  ABORTED: [],
  TIMEOUT: [],
};
```

## 5. 核心算法设计

### 5.1 操作评估引擎（实时）

```java
/**
 * 操作评估核心处理器
 * <p>
 * 对每个提交的操作事件进行多维度实时评估：
 * 1. 操作顺序校验（是否按SOP步骤执行）
 * 2. 参数合理性校验（用量、温度、时间等）
 * 3. 安全合规校验（是否违反安全规则）
 * 4. 规范性校验（操作手法是否符合实验规范）
 * 5. 效率评估（是否存在冗余操作）
 */
@Service
public class ActionEvaluationEngine {

    @Autowired
    private ExperimentSopCacheService sopCacheService;

    @Autowired
    private ExperimentSafetyRuleEngine safetyRuleEngine;

    @Autowired
    private ExperimentFeedbackGenerator feedbackGenerator;

    /**
     * 评估单个操作事件
     *
     * @param event    操作事件
     * @param session  实验会话上下文
     * @return 操作评估结果
     */
    public ActionEvaluation evaluate(ExperimentActionEvent event, ExperimentSessionContext session) {

        ExperimentSOP sop = sopCacheService.getSop(session.getExperimentId());
        StandardStep currentStep = findCurrentStep(sop, session);

        ActionEvaluation.ActionEvaluationBuilder builder = ActionEvaluation.builder()
            .eventId(event.getEventId())
            .actionType(event.getActionType());

        // === 1. 操作顺序与步骤匹配校验 ===
        StepMatchResult stepMatch = matchStep(event, currentStep, sop, session);
        builder.isCorrect(stepMatch.isCorrect())
              .correctnessScore(stepMatch.getScore())
              .errorType(stepMatch.getErrorType())
              .errorDescription(stepMatch.getErrorDescription());

        // === 2. 安全合规校验 ===
        SafetyCheckResult safetyResult = safetyRuleEngine.check(event, session, sop);
        builder.safetyLevel(safetyResult.getLevel())
              .safetyIssues(safetyResult.getIssues());

        // === 3. 规范性校验 ===
        RegulationResult regResult = checkRegulation(event, currentStep, sop);
        builder.regulationCompliance(regResult.getScore())
              .regulationIssues(regResult.getIssues());

        // === 4. 效率评估 ===
        EfficiencyResult effResult = assessEfficiency(event, session);
        builder.efficiencyScore(effResult.getScore())
              .isRedundant(effResult.isRedundant())
              .isOptimalTiming(effResult.isOptimalTiming());

        // === 5. 知识点关联 ===
        List<String> knowledgePoints = extractKnowledgePoints(event, currentStep, sop);
        builder.relatedKnowledgePoints(knowledgePoints);

        // === 6. 生成即时反馈 ===
        FeedbackMessage feedback = feedbackGenerator.generate(
            stepMatch, safetyResult, regResult, effResult, event, session
        );
        builder.immediateFeedback(feedback);

        // === 7. 危险操作拦截逻辑 ===
        if (safetyResult.getLevel() == SafetyLevel.DANGER) {
            handleDangerAction(event, session, safetyResult);
        }

        // 更新会话上下文
        session.advance(event, builder.build());

        return builder.build();
    }

    /**
     * 操作-步骤匹配：判断操作是否匹配当前期望步骤
     */
    private StepMatchResult matchStep(
            ExperimentActionEvent event,
            StandardStep expectedStep,
            ExperimentSOP sop,
            ExperimentSessionContext session) {

        // 情况1：操作完全匹配当前步骤
        if (isActionMatchStep(event, expectedStep)) {
            // 进一步校验参数容差
            ParameterCheckResult paramResult = checkParameters(event, expectedStep);
            if (paramResult.isWithinTolerance()) {
                return StepMatchResult.correct(expectedStep, 95 + paramResult.getBonus());
            } else {
                return StepMatchResult.partiallyCorrect(
                    expectedStep,
                    paramResult.getScore(),
                    ExperimentErrorType.WRONG_PARAMETER,
                    paramResult.getDescription()
                );
            }
        }

        // 情况2：操作匹配后续步骤（跳步）
        for (int i = session.getCurrentStepIndex() + 1; i < sop.getStandardSteps().size(); i++) {
            StandardStep futureStep = sop.getStandardSteps().get(i);
            if (isActionMatchStep(event, futureStep)) {
                return StepMatchResult.error(
                    ExperimentErrorType.MISSING_STEP,
                    String.format("跳过了步骤%d：%s。请先完成：%s",
                        futureStep.getStepNumber() - 1,
                        sop.getStandardSteps().get(i - 1).getDescription(),
                        sop.getStandardSteps().get(i - 1).getDescription()),
                    30
                );
            }
        }

        // 情况3：操作匹配已完成步骤（重复操作）
        for (int i = 0; i < session.getCurrentStepIndex(); i++) {
            StandardStep pastStep = sop.getStandardSteps().get(i);
            if (isActionMatchStep(event, pastStep)) {
                return StepMatchResult.error(
                    ExperimentErrorType.REDUNDANT_STEP,
                    String.format("该步骤（%s）已完成，无需重复操作", pastStep.getDescription()),
                    50
                );
            }
        }

        // 情况4：操作不属于任何标准步骤
        // 检查是否为禁止操作
        ForbiddenAction forbidden = checkForbidden(event, sop);
        if (forbidden != null) {
            return StepMatchResult.error(
                ExperimentErrorType.UNSAFE_OPERATION,
                forbidden.getInterventionMessage(),
                0
            );
        }

        // 情况5：合理但不必要的操作
        if (isReasonableExploration(event, session)) {
            return StepMatchResult.partiallyCorrect(null, 60, null, "该操作非必需但不影响实验", 60);
        }

        // 默认：无关操作
        return StepMatchResult.error(
            ExperimentErrorType.WRONG_APPARATUS,
            "该操作不在当前实验步骤中",
            20
        );
    }

    /**
     * 参数容差校验
     */
    private ParameterCheckResult checkParameters(
            ExperimentActionEvent event, StandardStep step) {

        if (step.getParameterTolerance() == null) {
            return ParameterCheckResult.fullScore();
        }

        ParameterTolerance tolerance = step.getParameterTolerance();
        ActionParameter actual = event.getParameters();
        ActionParameter expected = step.getExpectedParameters();
        List<String> deviations = new ArrayList<>();
        double score = 100;

        // 用量校验
        if (tolerance.getQuantityTolerance() != null
                && expected.getQuantity() != null
                && actual.getQuantity() != null) {
            double diff = Math.abs(actual.getQuantity() - expected.getQuantity());
            double allowed = tolerance.getQuantityTolerance();
            if (diff > allowed) {
                double penalty = Math.min(50, (diff / allowed - 1) * 30);
                score -= penalty;
                deviations.add(String.format(
                    "用量偏差%.1f（允许±%.1f）", diff, allowed
                ));
            }
        }

        // 温度校验
        if (tolerance.getTemperatureTolerance() != null
                && expected.getTemperature() != null
                && actual.getTemperature() != null) {
            double diff = Math.abs(actual.getTemperature() - expected.getTemperature());
            double allowed = tolerance.getTemperatureTolerance();
            if (diff > allowed) {
                score -= Math.min(40, (diff / allowed - 1) * 20);
                deviations.add(String.format("温度偏差%.1f℃", diff));
            }
        }

        // 读数校验
        if (tolerance.getMeasuredValueTolerance() != null
                && expected.getMeasuredValue() != null
                && actual.getMeasuredValue() != null) {
            double diff = Math.abs(actual.getMeasuredValue() - expected.getMeasuredValue());
            double allowed = tolerance.getMeasuredValueTolerance();
            if (diff > allowed) {
                score -= Math.min(30, (diff / allowed - 1) * 15);
                deviations.add(String.format("读数偏差%.2f", diff));
            }
        }

        return ParameterCheckResult.builder()
            .withinTolerance(score >= 60)
            .score(score)
            .description(String.join("；", deviations))
            .build();
    }
}
```

```java
 * 维护化学、物理、生物实验的安全规则库，

            .ruleId("CHEM_SAFETY_001")

            .category(SafetyCategory.CHEMICAL_HAZARD)
            .triggerCondition(
                // 条件：操作对象是水，目标是含浓硫酸的容器

                if (voltage > maxVoltage) {
                    return SafetyCheckResult.danger(
                        String.format("电压%.1fV超过安全限制%.1fV！", voltage, maxVoltage),
                        "请立即降低电压到安全范围"
                    );
                }
                return SafetyCheckResult.safe();
            })
            .build()
    );

    /**
     * 执行安全检查
     */
    public SafetyCheckResult check(
            ExperimentActionEvent event,
            ExperimentSessionContext session,
            ExperimentSOP sop) {

        List<SafetyRule> applicableRules = getApplicableRules(sop.getSubject());

        List<SafetyCheckResult> results = new ArrayList<>();
        for (SafetyRule rule : applicableRules) {
            if (rule.matches(event, session)) {
                SafetyCheckResult result = rule.evaluate(event, session, sop);
                if (result.getLevel() != SafetyLevel.SAFE) {
                    results.add(result);
                }
            }
        }

        // 同时检查 SOP 中的自定义安全要求
        for (SafetyRequirement req : sop.getSafetyRequirements()) {
            if (req.getCheckMethod() == SafetyCheckMethod.REALTIME) {
                SafetyCheckResult reqResult = checkSafetyRequirement(req, event, session);
                if (reqResult.getLevel() != SafetyLevel.SAFE) {
                    results.add(reqResult);
                }
            }
        }

        // 返回最高级别安全问题
        return results.stream()
            .max(Comparator.comparingInt(SafetyCheckResult::getSeverityRank))
            .orElse(SafetyCheckResult.safe());
    }
}
```

### 5.3 实验技能评分算法

```java
/**
 * 实验技能多维评分引擎
 * <p>
 * 实验完成后，基于全部操作事件序列计算五维评分。
 */
@Service
public class ExperimentSkillScoringEngine {

    /**
     * 计算综合评分
     */
    public ExperimentSkillReport generateReport(
            ExperimentSessionContext session,
            List<ActionEvaluation> evaluations,
            ExperimentSOP sop) {

        SkillDimensions dimensions = calculateDimensions(evaluations, session, sop);

        // 加权综合分
        ScoringWeights weights = sop.getScoringWeights();
        double overall = dimensions.getCorrectness() * weights.getCorrectness()
                       + dimensions.getSafety() * weights.getSafety()
                       + dimensions.getRegulation() * weights.getRegulation()
                       + dimensions.getEfficiency() * weights.getEfficiency()
                       + dimensions.getUnderstanding() * weights.getUnderstanding();

        String grade = calculateGrade(overall);

        // 生成诊断分析
        DiagnosticAnalysis diagnosis = diagnose(dimensions, evaluations, sop);

        // 生成改进建议
        List<ExperimentRecommendation> recommendations = generateRecommendations(
            dimensions, diagnosis, session, sop
        );

        // 知识点掌握度影响计算
        List<KnowledgePointImpact> kpImpacts = calculateKnowledgePointImpacts(
            evaluations, session
        );

        return ExperimentSkillReport.builder()
            .sessionId(session.getSessionId())
            .studentId(session.getStudentId())
            .experimentId(sop.getExperimentId())
            .overallScore(Math.round(overall * 100) / 100.0)
            .grade(grade)
            .dimensionScores(dimensions)
            .totalActions(evaluations.size())
            .correctActions((int) evaluations.stream().filter(ActionEvaluation::isCorrect).count())
            .errorActions((int) evaluations.stream().filter(e -> !e.isCorrect()).count())
            .redundantActions((int) evaluations.stream().filter(ActionEvaluation::isRedundant).count())
            .duration(session.getDuration())
            .strengths(diagnosis.getStrengths())
            .weaknesses(diagnosis.getWeaknesses())
            .recommendations(recommendations)
            .knowledgePointImpacts(kpImpacts)
            .build();
    }

    /**
     * 五维评分计算
     */
    private SkillDimensions calculateDimensions(
            List<ActionEvaluation> evals,
            ExperimentSessionContext session,
            ExperimentSOP sop) {

        // 1. 准确性（40%来自步骤正确率，30%参数精确度，30%关键节点表现）
        double stepAccuracy = percentage(evals, ActionEvaluation::isCorrect);
        double avgCorrectnessScore = evals.stream()
            .mapToDouble(ActionEvaluation::getCorrectnessScore)
            .average().orElse(0);
        double criticalCheckpointScore = calculateCriticalCheckpointScore(evals, sop);
        double correctness = stepAccuracy * 0.4 + avgCorrectnessScore * 0.3 + criticalCheckpointScore * 0.3;

        // 2. 安全性（基于安全事件扣分）
        long dangerCount = evals.stream()
            .mapToLong(e -> e.getSafetyLevel() == SafetyLevel.DANGER ? 1 : 0)
            .sum();
        long warningCount = evals.stream()
            .mapToLong(e -> e.getSafetyLevel() == SafetyLevel.WARNING ? 1 : 0)
            .sum();
        long cautionCount = evals.stream()
            .mapToLong(e -> e.getSafetyLevel() == SafetyLevel.CAUTION ? 1 : 0)
            .sum();
        double safety = Math.max(0, 100 - dangerCount * 25 - warningCount * 10 - cautionCount * 3);

        // 3. 规范性（操作规范符合度的平均值）
        double regulation = evals.stream()
            .mapToDouble(ActionEvaluation::getRegulationCompliance)
            .average().orElse(100);

        // 4. 效率（冗余操作扣分 + 时间效率）
        long redundantCount = evals.stream()
            .mapToLong(e -> e.isRedundant() ? 1 : 0)
            .sum();
        double redundancyPenalty = redundantCount * 5;
        double timeEfficiency = calculateTimeEfficiency(session, sop);
        double efficiency = Math.max(0, (100 - redundancyPenalty) * 0.5 + timeEfficiency * 0.5);

        // 5. 理解度（基于观察记录质量、知识点关联操作的表现）
        double observationScore = calculateObservationScore(evals, session);
        double knowledgeScore = calculateKnowledgeScore(evals, session);
        double understanding = observationScore * 0.5 + knowledgeScore * 0.5;

        return SkillDimensions.builder()
            .correctness(round(correctness))
            .safety(round(safety))
            .regulation(round(regulation))
            .efficiency(round(efficiency))
            .understanding(round(understanding))
            .build();
    }

    /**
     * 时间效率评分
     * <p>
     * 对比学生实际耗时与SOP建议时间：
     * - 在建议时间的60%-120%范围内为满分
     * - 过快（<60%）可能操作草率，适当扣分
     * - 过慢（>120%）效率不足
     */
    private double calculateTimeEfficiency(ExperimentSessionContext session, ExperimentSOP sop) {
        if (sop.getTimeLimit() == null) return 100;

        long actualMs = session.getDuration();
        long expectedMs = sop.getEstimatedTime();
        double ratio = (double) actualMs / expectedMs;

        if (ratio >= 0.6 && ratio <= 1.2) return 100;
        if (ratio < 0.6) return Math.max(60, 100 - (0.6 - ratio) * 200);
        return Math.max(40, 100 - (ratio - 1.2) * 50);
    }

    /**
     * 关键节点表现评分
     */
    private double calculateCriticalCheckpointScore(
            List<ActionEvaluation> evals, ExperimentSOP sop) {

        Set<String> criticalSteps = new HashSet<>(sop.getCriticalCheckpoints());
        if (criticalSteps.isEmpty()) return 100;

        List<ActionEvaluation> checkpointEvals = evals.stream()
            .filter(e -> e.getStepId() != null && criticalSteps.contains(e.getStepId()))
            .toList();

        if (checkpointEvals.isEmpty()) return 0;

        return checkpointEvals.stream()
            .mapToDouble(ActionEvaluation::getCorrectnessScore)
            .average().orElse(0);
    }

    private String calculateGrade(double score) {
        if (score >= 90) return "A";
        if (score >= 80) return "B";
        if (score >= 70) return "C";
        if (score >= 60) return "D";
        return "F";
    }
}
```

```java
     */

        if (dims.getSafety() >= threshold)

        if (dims.getRegulation() >= threshold)
            strengths.add("实验操作符合规范要求");
        if (dims.getEfficiency() >= threshold)
            strengths.add("操作简洁高效，时间管理合理");
        if (dims.getUnderstanding() >= threshold)
            strengths.add("对实验原理理解深入，观察记录准确");

        return strengths;
    }

    /**
     * 识别薄弱维度
     */
    private List<String> identifyWeaknesses(SkillDimensions dims) {
        List<String> weaknesses = new ArrayList<>();
        double threshold = 70;

        if (dims.getCorrectness() < threshold)
            weaknesses.add("部分操作步骤存在错误，需加强实验流程记忆");
        if (dims.getSafety() < threshold)
            weaknesses.add("安全操作意识需提升，注意遵守安全规程");
        if (dims.getRegulation() < threshold)
            weaknesses.add("实验操作规范性有待提高");
        if (dims.getEfficiency() < threshold)
            weaknesses.add("存在冗余操作，建议优化实验流程");
        if (dims.getUnderstanding() < threshold)
            weaknesses.add("对实验原理的理解和观察记录能力需加强");

        return weaknesses;
    }

    /**
     * 检测错误模式
     */
    private List<ErrorPattern> detectErrorPatterns(List<ActionEvaluation> evals) {
        Map<ExperimentErrorType, Long> errorCounts = evals.stream()
            .filter(e -> e.getErrorType() != null)
            .collect(Collectors.groupingBy(
                ActionEvaluation::getErrorType,
                Collectors.counting()
            ));

        List<ErrorPattern> patterns = new ArrayList<>();
        for (Map.Entry<ExperimentErrorType, Long> entry : errorCounts.entrySet()) {
            if (entry.getValue() >= 2) {
                patterns.add(ErrorPattern.builder()
                    .errorType(entry.getKey())
                    .frequency(entry.getValue())
                    .description(getErrorPatternDescription(entry.getKey(), entry.getValue()))
                    .severity(entry.getValue() >= 3 ? "HIGH" : "MEDIUM")
                    .build());
            }
        }
        return patterns;
    }

    /**
     * 生成改进建议
     */
    private List<ExperimentRecommendation> generateRecommendations(
            SkillDimensions dims,
            DiagnosticAnalysis diagnosis,
            ExperimentSessionContext session,
            ExperimentSOP sop) {

        List<ExperimentRecommendation> recs = new ArrayList<>();

        // 针对最低维度生成针对性建议
        SkillDimension weakest = findWeakestDimension(dims);

        switch (weakest) {
            case CORRECTNESS ->
                recs.add(ExperimentRecommendation.builder()
                    .type("RETRY_EXPERIMENT")
                    .title("重新进行实验练习")
                    .description("建议在引导模式下重新完成实验，重点关注操作步骤顺序")
                    .knowledgePointIds(sop.getAllKnowledgePointIds())
                    .priority(1)
                    .build());

            case SAFETY ->
                recs.add(ExperimentRecommendation.builder()
                    .type("WATCH_DEMO")
                    .title("观看安全操作演示视频")
                    .description("重点学习实验安全操作要点和应急处理方法")
                    .knowledgePointIds(sop.getSafetyKnowledgePointIds())
                    .priority(1)
                    .build());

            case EFFICIENCY ->
                recs.add(ExperimentRecommendation.builder()
                    .type("PRACTICE_SKILL")
                    .title("精简实验流程练习")
                    .description("尝试在不看提示的情况下连续完成实验，减少不必要的操作")
                    .priority(2)
                    .build());

            case UNDERSTANDING ->
                recs.add(ExperimentRecommendation.builder()
                    .type("REVIEW_KNOWLEDGE")
                    .title("复习实验相关知识点")
                    .description("建议复习本实验涉及的核心知识点和原理")
                    .knowledgePointIds(sop.getAllKnowledgePointIds())
                    .priority(1)
                    .build());

        for (ErrorPattern pattern : diagnosis.getErrorPatterns()) {

                .priority(3)
                .build());

        return recs.stream()
            .sorted(Comparator.comparingInt(ExperimentRecommendation::getPriority))
            .limit(5)
            .toList();
    }
```

```java
/**
 * SOP模板多级缓存

@Service
public class ExperimentSopCacheService {

    // L1: 本地内存缓存（Caffeine, 最热数据）
    private final Cache<String, ExperimentSOP> l1Cache = Caffeine.newBuilder()
        .maximumSize(500)
        .expireAfterWrite(Duration.ofHours(1))
        .recordStats()
        .build();

    // L2: Redis分布式缓存
    @Autowired
    private RedisTemplate<String, ExperimentSOP> redisTemplate;

    // L3: 数据库
    @Autowired
    private ExperimentSopRepository repository;

    public ExperimentSOP getSop(String experimentId) {
        // L1
        ExperimentSOP sop = l1Cache.getIfPresent(experimentId);
        if (sop != null) return sop;

        // L2
        sop = redisTemplate.opsForValue().get("exp:sop:" + experimentId);
        if (sop != null) {
            l1Cache.put(experimentId, sop);
            return sop;
        }

        // L3
        sop = repository.findActiveByExperimentId(experimentId);
        if (sop != null) {
            redisTemplate.opsForValue().set(
                "exp:sop:" + experimentId, sop, Duration.ofHours(24)
            );
            l1Cache.put(experimentId, sop);
        }
        return sop;
    }

    /**
     * SOP更新时清除缓存
     */
    public void evictSop(String experimentId) {
        l1Cache.invalidate(experimentId);
        redisTemplate.delete("exp:sop:" + experimentId);
    }
}
```

### 8.3 操作事件批量写入

```java
/**
 * 操作事件异步批量写入
 * <p>
 * 操作事件先写入Redis（保证实时性），
 * 再通过批量消费者异步持久化到数据库。
 */
@Service
public class ActionEventBuffer {

    @Autowired
    private StringRedisTemplate redisTemplate;

    private static final int BATCH_SIZE = 200;
    private static final long FLUSH_INTERVAL_MS = 5000;

    /**
     * 写入操作事件到Redis缓冲区
     */
    public void buffer(ExperimentActionEvent event) {
        String key = "exp:actions:" + event.getSessionId();
        redisTemplate.opsForList().rightPush(key, serialize(event));
        redisTemplate.expire(key, Duration.ofHours(2));
    }

    /**
     * 定时批量持久化到数据库
     */
    @Scheduled(fixedDelay = 5000)
    public void flushToDatabase() {
        // 扫描待持久化的会话
        Set<String> sessionKeys = redisTemplate.keys("exp:actions:*");
        if (sessionKeys == null || sessionKeys.isEmpty()) return;

        for (String key : sessionKeys) {
            List<String> rawData = redisTemplate.opsForList()
                .range(key, 0, BATCH_SIZE - 1);
            if (rawData == null || rawData.isEmpty()) continue;

            List<ExperimentActionEvent> events = rawData.stream()
                .map(this::deserialize)
                .toList();

            // 批量插入
            eventRepository.batchInsert(events);

            // 从Redis移除已持久化的数据
            redisTemplate.opsForList().trim(key, rawData.size(), -1);
        }
    }
}
```

```dart
// Flutter 示例：虚拟实验操作事件提交
class ExperimentActionReporter {
  final String sessionId;
  final Dio httpClient;
  final Queue<Map<String, dynamic>> _offlineQueue = Queue();
  final StreamController<SafetyAlert> _safetyAlertController;

  /// 提交单个操作事件
  Future<ActionEvaluation> reportAction(ExperimentAction action) async {
    final event = {
      'actionType': action.type.name,
      'apparatusId': action.apparatusId,
      'targetApparatusId': action.targetApparatusId,
      'parameters': _serializeParameters(action.parameters),
      'resultSnapshot': _serializeSnapshot(action.resultSnapshot),
      'clientTimestamp': DateTime.now().millisecondsSinceEpoch,
    };

    try {
      final response = await httpClient.post(
        '/api/v1/experiment-eval/sessions/$sessionId/actions',
        data: event,
      );

      final evaluation = ActionEvaluation.fromJson(response.data['data']);

      // 处理安全警告
      if (evaluation.safetyLevel == 'DANGER') {
        _safetyAlertController.add(SafetyAlert(
          level: SafetyLevel.danger,
          message: evaluation.immediateFeedback?.content ?? '危险操作',
          correctAction: evaluation.immediateFeedback?.actionableHint,
        ));
      }

      return evaluation;
    } on DioException catch (e) {
      // 弱网降级：放入离线队列
      if (e.type == DioExceptionType.connectionTimeout ||
          e.type == DioExceptionType.networkError) {
        _offlineQueue.add({
          'sessionId': sessionId,
          ...event,
        });
        // 返回默认评估（乐观模式）
        return ActionEvaluation.defaultSafe();
      }
      rethrow;
    }
  }

  /// 网络恢复后批量提交
  Future<void> flushOfflineQueue() async {
    while (_offlineQueue.isNotEmpty) {
      final batch = <Map<String, dynamic>>[];
      for (int i = 0; i < 50 && _offlineQueue.isNotEmpty; i++) {
        batch.add(_offlineQueue.removeFirst());
      }

      await httpClient.post(
        '/api/v1/experiment-eval/sessions/$sessionId/actions/batch',
        data: {'actions': batch},
      );
    }
  }
}
```

## 10. 安全合规与审计

### 10.1 数据安全

- 学生操作数据属于未成年人学习行为数据，按《未成年人保护法》和《个人信息保护法》要求处理
- 操作事件中的设备信息（设备型号、App版本）仅用于技术分析，不做用户画像
- 实验报告数据保留期限：活跃期间永久保留；账号注销后30天内删除

### 10.2 安全事件上报

当检测到 DANGER 级别安全事件时：
1. **实时通知**：通过 WebSocket / 推送通知教师端（如有绑定）
2. **家长通知**：通过家长端消息推送
3. **事件归档**：写入安全事件表，用于后续分析和改进
4. **自动拦截**：在客户端阻止危险操作继续执行
