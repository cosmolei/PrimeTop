# 客户端 - 横屏模式与大屏设备适配设计

> **模块定位**：PrimeTop 客户端多屏幕形态适配系统，覆盖手机竖屏/横屏、平板、折叠屏、Chrome OS / 桌面模式等场景的布局策略、组件复用与交互差异化方案。
> **覆盖学段**：全学段（通过分龄布局策略差异化）
> **优先级**：P1（V1.0 阶段需支持 iPad / Android 平板基础适配，V1.5 阶段支持折叠屏和横屏模式）
> **技术栈**：Flutter 3.x + Dart 3.x + MediaQuery + LayoutBuilder + MediaQuery_flutter

---

## 1. 设计目标

| 目标 | 说明 |
|------|------|
| 多设备覆盖 | 手机(4.7"-6.7")、平板(7"-13")、折叠屏(展开/折叠)、桌面(Chrome OS/macOS) |
| 布局自适应 | 同一套 Widget 代码根据可用宽度自动切换紧凑/中等/展开布局 |
| 横屏体验 | 关键学习场景（AI对话、同步课堂、练习、错题本）提供优化的横屏布局 |
| 性能一致 | 大屏设备不因布局复杂度增加而降低帧率（目标稳定 60fps） |
| 分龄差异 | 幼儿/小学大屏侧重内容放大和亲子并排，初中/高中大屏侧重信息密度和分屏 |
| 开发效率 | 布局断点统一，响应式组件开箱即用，新页面无需从零适配 |

/// 屏幕信息 - 全局可用的屏幕状态
@immutable
class ScreenInfo {
  final ScreenType type;
  final LayoutMode layoutMode;
  final Orientation orientation;
  final Size screenSize;
  final double widthDp;
  final double heightDp;
  final EdgeInsets safeAreaPadding;
  final EdgeInsets viewInsets; // 键盘
  final double pixelRatio;
  final bool isFoldable;
  final FoldableState foldableState;
  final double hingeWidth; // 折叠屏铰链宽度

  const ScreenInfo({
    required this.type,
    required this.layoutMode,
    required this.orientation,
    required this.screenSize,
    required this.widthDp,
    required this.heightDp,
    required this.safeAreaPadding,
    required this.viewInsets,
    required this.pixelRatio,
    this.isFoldable = false,
    this.foldableState = FoldableState.flat,
    this.hingeWidth = 0,
  });

  /// 是否为平板或更大设备
  bool get isTablet => type.index >= ScreenType.medium.index;

  /// 是否为桌面级设备
  bool get isDesktop => type == ScreenType.extraLarge;

  /// 是否为横屏
  bool get isLandscape => orientation == Orientation.landscape;

  /// 是否为折叠屏展开态
  bool get isFoldedOut => isFoldable && foldableState == FoldableState.open;

  /// 推荐的字体缩放因子（大屏适当放大）
  double get fontScaleFactor {
    switch (type) {
      case ScreenType.compact:
        return 1.0;
      case ScreenType.medium:
        return 1.05;
      case ScreenType.expanded:
        return 1.1;
      case ScreenType.extraLarge:
        return 1.15;
    }
  }

  /// 推荐的内容最大宽度（防止内容在大屏上过度拉伸）
  double get contentMaxWidth {
    switch (type) {
      case ScreenType.compact:
        return double.infinity;
      case ScreenType.medium:
        return 720;
      case ScreenType.expanded:
        return 840;
      case ScreenType.extraLarge:
        return 960;
    }
  }
}

/// 折叠屏状态
enum FoldableState {
  flat,    // 普通设备 / 未折叠
  closed,  // 折叠态（小屏）
  half,    // 半折叠（笔记本模式）
  open,    // 展开态（大屏）
}

/// 屏幕信息 Provider

### 3.3 ScreenInfoBuilder（顶层注入）

```dart
/// lib/core/widgets/screen_info_builder.dart

/// 顶层 Widget：监听 MediaQuery 变化，注入 ScreenInfo
class ScreenInfoBuilder extends ConsumerWidget {
  final Widget child;

  const ScreenInfoBuilder({super.key, required this.child});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final mediaQuery = MediaQuery.of(context);
    final size = mediaQuery.size;
    final width = size.width;
    final orientation = mediaQuery.orientation;

    // 检测折叠屏
    final displayFeatures = MediaQuery.of(context).displayFeatures;
    final foldInfo = _detectFoldable(displayFeatures, size);

    // 确定屏幕类型
    final screenType = _resolveScreenType(width);

    // 确定布局模式
    final layoutMode = _resolveLayoutMode(screenType, orientation, foldInfo);

    final screenInfo = ScreenInfo(
      type: screenType,
      layoutMode: layoutMode,
      orientation: orientation,
      screenSize: size,
      widthDp: width,
      heightDp: size.height,
      safeAreaPadding: mediaQuery.padding,
      viewInsets: mediaQuery.viewInsets,
      pixelRatio: mediaQuery.devicePixelRatio,
      isFoldable: foldInfo.isFoldable,
      foldableState: foldInfo.state,
      hingeWidth: foldInfo.hingeWidth,
    );

    // 覆盖 Provider
    ref.watch(screenInfoProvider.notifier);
    // 通过 ProviderScope override 注入
    return ProviderScope(
      overrides: [
        screenInfoProvider.overrideWithValue(screenInfo),
      ],
      child: child,
    );
  }

  ScreenType _resolveScreenType(double width) {
    if (width < AppBreakpoints.compact) return ScreenType.compact;
    if (width < AppBreakpoints.medium) return ScreenType.medium;
    if (width < AppBreakpoints.extraLarge) return ScreenType.expanded;
    return ScreenType.extraLarge;
  }

  LayoutMode _resolveLayoutMode(
    ScreenType type,
    Orientation orientation,
    _FoldInfo fold,
  ) {
    // 折叠屏半开态 → 双栏（上下）
    if (fold.state == FoldableState.half) {
      return LayoutMode.dualPane;
    }
    switch (type) {
      case ScreenType.compact:
        return LayoutMode.singlePane;
      case ScreenType.medium:
        return LayoutMode.dualPane;
      case ScreenType.expanded:
        return orientation == Orientation.landscape
            ? LayoutMode.triplePane
            : LayoutMode.dualPane;
      case ScreenType.extraLarge:
        return LayoutMode.triplePane;
    }
  }

  _FoldInfo _detectFoldable(List<DisplayFeature> features, Size size) {
    // 查找 DisplayFeatureType.hinge / fold
    for (final feature in features) {
      if (feature.type == DisplayFeatureType.hinge ||
          feature.type == DisplayFeatureType.fold) {
        return _FoldInfo(
          isFoldable: true,
          state: _inferFoldState(feature, size),
          hingeWidth: feature.bounds.width,
        );
      }
    }
    return _FoldInfo(isFoldable: false);
  }

  FoldableState _inferFoldState(DisplayFeature feature, Size size) {
    // 简单推断：如果 feature 在屏幕中间垂直方向 → 展开态
    final bounds = feature.bounds;
    final centerX = bounds.left + bounds.width / 2;
    final screenCenterX = size.width / 2;
    if ((centerX - screenCenterX).abs() < 50) {
      return FoldableState.open;
    }
    return FoldableState.flat;
  }
}

class _FoldInfo {
  final bool isFoldable;
  final FoldableState state;
  final double hingeWidth;
  _FoldInfo({
    this.isFoldable = false,
    this.state = FoldableState.flat,
    this.hingeWidth = 0,
  });
}

### 4.1 AdaptiveScaffold（自适应骨架屏）

```dart
/// lib/core/widgets/adaptive_scaffold.dart

/// 自适应骨架屏：根据 ScreenInfo 自动切换单栏/双栏/三栏布局
///
/// 用法：
/// ```dart
/// AdaptiveScaffold(
///   destinations: navDestinations,
///   selectedIndex: 0,
///   onDestinationSelected: (i) => ...,
///   body: (_) => const ChatPage(),
///   secondaryBody: (_) => const KnowledgePanel(),
///   tertiaryBody: (_) => const ChatHistoryPanel(),
/// )
/// ```

typedef PaneBuilder = Widget Function(BuildContext context);

class AdaptiveScaffold extends ConsumerStatefulWidget {
  final List<AdaptiveDestination> destinations;
  final int selectedIndex;
  final ValueChanged<int> onDestinationSelected;
  final PaneBuilder body;
  final PaneBuilder? secondaryBody;
  final PaneBuilder? tertiaryBody;
  final Widget? floatingActionButton;

  const AdaptiveScaffold({
    super.key,
    required this.destinations,
    required this.selectedIndex,
    required this.onDestinationSelected,
    required this.body,
    this.secondaryBody,
    this.tertiaryBody,
    this.floatingActionButton,
  });

  @override
  ConsumerState<AdaptiveScaffold> createState() => _AdaptiveScaffoldState();
}

class _AdaptiveScaffoldState extends ConsumerState<AdaptiveScaffold> {
  // 次级面板是否显示（小屏下通过 FAB 或手势控制）
  bool _secondaryVisible = false;
  bool _tertiaryVisible = false;

  @override
  Widget build(BuildContext context) {
    final screen = ref.watch(screenInfoProvider);

    return switch (screen.layoutMode) {
      LayoutMode.singlePane => _buildSinglePane(context, screen),
      LayoutMode.dualPane => _buildDualPane(context, screen),
      LayoutMode.triplePane => _buildTriplePane(context, screen),
      LayoutMode.adaptiveGrid => _buildSinglePane(context, screen),
    };
  }

  /// 单栏：底部导航 + 全屏 body
  Widget _buildSinglePane(BuildContext context, ScreenInfo screen) {
    return Scaffold(
      body: widget.body(context),
      bottomNavigationBar: _BottomNav(
        destinations: widget.destinations,
        selectedIndex: widget.selectedIndex,
        onSelected: widget.onDestinationSelected,
      ),
      floatingActionButton: widget.floatingActionButton,
    );
  }

  /// 双栏：侧边导航/抽屉 + body + secondaryBody
  Widget _buildDualPane(BuildContext context, ScreenInfo screen) {
    return Scaffold(
      body: Row(
        children: [
          // 侧边导航栏
          _SideNavigationRail(
            destinations: widget.destinations,
            selectedIndex: widget.selectedIndex,
            onSelected: widget.onDestinationSelected,
            extended: screen.type == ScreenType.expanded,
          ),
          // 主内容区
          Expanded(
            flex: 3,
            child: widget.body(context),
          ),
          // 次级面板
          if (widget.secondaryBody != null && _secondaryVisible)
            _SecondaryPane(
              flex: 2,
              onClose: () => setState(() => _secondaryVisible = false),
              child: widget.secondaryBody!(context),
            ),
        ],
      ),
      floatingActionButton: widget.floatingActionButton ??
          (_secondaryVisible
              ? null
              : _buildShowSecondaryFab(screen)),
    );
  }

  /// 三栏：侧边导航 + body + secondary + tertiary
  Widget _buildTriplePane(BuildContext context, ScreenInfo screen) {
    return Scaffold(
      body: Row(
        children: [
          // 展开的侧边导航
          _SideNavigationRail(
            destinations: widget.destinations,
            selectedIndex: widget.selectedIndex,
            onSelected: widget.onDestinationSelected,
            extended: true,
          ),
          // 三级面板（最左侧，如历史记录）
          if (widget.tertiaryBody != null)
            SizedBox(
              width: 280,
              child: widget.tertiaryBody!(context),
            ),
          // 主内容区
          Expanded(
            flex: 3,
            child: widget.body(context),
          ),
          // 次级面板
          if (widget.secondaryBody != null)
            Expanded(
              flex: 2,
              child: widget.secondaryBody!(context),
            ),
        ],
      ),
    );
  }

  Widget? _buildShowSecondaryFab(ScreenInfo screen) {
    if (widget.secondaryBody == null) return null;
    return FloatingActionButton(
      mini: true,
      child: const Icon(Icons.info_outline),
      onPressed: () => setState(() => _secondaryVisible = true),
    );
  }
}

/// 通用双栏视图控制器
class DualPaneController extends ChangeNotifier {
  bool _detailVisible = false;
  String? _selectedId;

  bool get detailVisible => _detailVisible;
  String? get selectedId => _selectedId;

  void select(String id) {
    _selectedId = id;
    _detailVisible = true;
    notifyListeners();
  }

  void clear() {
    _selectedId = null;
    _detailVisible = false;
    notifyListeners();
  }
}

/// 双栏视图：列表+详情
/// 大屏自动并排显示，小屏自动切换（Master-Detail 模式）
class DualPaneView extends ConsumerStatefulWidget {
  final Widget listPane;
  final Widget Function(String? selectedId) detailPane;
  final DualPaneController? controller;
  final double listPaneWidth; // 大屏下列表固定宽度
  final Widget? emptyDetailPlaceholder;

  const DualPaneView({
    super.key,
    required this.listPane,
    required this.detailPane,
    this.controller,
    this.listPaneWidth = 360,
    this.emptyDetailPlaceholder,
  });

  @override
  ConsumerState<DualPaneView> createState() => _DualPaneViewState();
}

class _DualPaneViewState extends ConsumerState<DualPaneView> {
  late DualPaneController _controller;

  @override
  void initState() {
    super.initState();
    _controller = widget.controller ?? DualPaneController();
    _controller.addListener(() => setState(() {}));
  }

  @override
  void dispose() {
    if (widget.controller == null) _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final screen = ref.watch(screenInfoProvider);

    if (screen.layoutMode == LayoutMode.singlePane) {
      // 小屏：切换显示
      if (_controller.detailVisible && _controller.selectedId != null) {
        return Column(
          children: [
            // 返回按钮条
            _BackBar(onBack: _controller.clear),
            Expanded(child: widget.detailPane(_controller.selectedId)),
          ],
        );
      }
      return widget.listPane;
    }

    // 大屏：并排显示
    return Row(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        // 列表面板
        SizedBox(
          width: widget.listPaneWidth,
          child: widget.listPane,
        ),
        // 分割线
        const VerticalDivider(width: 1),
        // 详情面板
        Expanded(
          child: _controller.selectedId != null
              ? widget.detailPane(_controller.selectedId)
              : (widget.emptyDetailPlaceholder ?? const _EmptyDetailPlaceholder()),
        ),
      ],
    );
  }
}

### 4.3 ResponsiveGrid（自适应网格）

```dart
/// lib/core/widgets/responsive_grid.dart

/// 自适应网格布局
class ResponsiveGrid extends ConsumerWidget {
  final List<Widget> children;
  final double spacing;
  final double runSpacing;
  final double minItemWidth;

  const ResponsiveGrid({
    super.key,
    required this.children,
    this.spacing = 12,
    this.runSpacing = 12,
    this.minItemWidth = 160,
  });

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final screen = ref.watch(screenInfoProvider);
    final availableWidth = screen.widthDp - screen.safeAreaPadding.horizontal;

    // 根据可用宽度自动计算列数
    final crossAxisCount = (availableWidth / (minItemWidth + spacing)).floor();
    final actualColumns = crossAxisCount.clamp(1, children.length);

    return Wrap(
      spacing: spacing,
      runSpacing: runSpacing,
      children: children.map((child) {
        final itemWidth =
            (availableWidth - (actualColumns - 1) * spacing) / actualColumns;
        return SizedBox(width: itemWidth, child: child);
      }).toList(),
    );
  }
}
```

/// lib/core/widgets/constrained_content.dart

/// 内容宽度约束：防止大屏上内容过度拉伸
/// 用于 AI 对话气泡、文章正文等场景
class ConstrainedContent extends ConsumerWidget {
  final Widget child;
  final double? maxWidth;
  final bool centerAlign;

  const ConstrainedContent({
    super.key,
    required this.child,
    this.maxWidth,
    this.centerAlign = true,
  });

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final screen = ref.watch(screenInfoProvider);
    final effectiveMaxWidth = maxWidth ?? screen.contentMaxWidth;

    return Center(
      child: ConstrainedBox(
        constraints: BoxConstraints(maxWidth: effectiveMaxWidth),
        child: child,
      ),
    );
  }
}

/// lib/core/services/foldable_service.dart

class FoldableService {
  /// 检测当前设备是否为折叠屏
  static bool isFoldableDevice(List<DisplayFeature> features) {
    return features.any((f) =>
        f.type == DisplayFeatureType.hinge ||
        f.type == DisplayFeatureType.fold);
  }

  /// 获取铰链位置
  static Rect? getHingeBounds(List<DisplayFeature> features) {
    for (final f in features) {
      if (f.type == DisplayFeatureType.hinge) {
        return f.bounds;
      }
    }
    return null;
  }

  /// 铰链是否为垂直方向（即设备横折还是竖折）
  static bool isVerticalHinge(Rect hinge, Size screenSize) {
    return hinge.height > hinge.width;
  }
}

/// lib/core/services/orientation_manager.dart

class OrientationManager {
  /// 各页面的横屏策略
  static const Map<String, OrientationPolicy> _pagePolicies = {
    '/': OrientationPolicy.auto,               // 首页：自动
    '/chat': OrientationPolicy.auto,            // AI 对话：自动
    '/classroom': OrientationPolicy.auto,       // 同步课堂：自动
    '/photo-solve': OrientationPolicy.portrait,  // 拍题：强制竖屏
    '/practice': OrientationPolicy.auto,        // 练习：自动
    '/practice/timed': OrientationPolicy.landscape, // 限时考试：建议横屏
    '/phonics': OrientationPolicy.portrait,      // 拼音识字：强制竖屏
    '/parent': OrientationPolicy.portrait,       // 家长中心：强制竖屏
  };

  static OrientationPolicy getPolicy(String route) {
    return _pagePolicies[route] ?? OrientationPolicy.auto;
  }

  static void applyPolicy(OrientationPolicy policy) {
    switch (policy) {
      case OrientationPolicy.portrait:
        SystemChrome.setPreferredOrientations([
          DeviceOrientation.portraitUp,
          DeviceOrientation.portraitDown,
        ]);
      case OrientationPolicy.landscape:
        SystemChrome.setPreferredOrientations([
          DeviceOrientation.landscapeLeft,
          DeviceOrientation.landscapeRight,
        ]);
      case OrientationPolicy.auto:
        SystemChrome.setPreferredOrientations([]); // 恢复自动
    }
  }
}

enum OrientationPolicy {
  portrait,  // 强制竖屏
  landscape, // 强制横屏
  auto,      // 自动旋转
}

          _DismissIntent: _CallbackAction(() => _handleDismiss(context)),
        },
        child: child,
      ),
    );
  }
}

class _CallbackAction extends Action<Intent> {
  final VoidCallback callback;
  _CallbackAction(this.callback);

  @override
  Object? invoke(Intent intent) {
    callback();
    return null;
  }
}

## 11. 错误处理与降级策略

### 11.1 布局降级矩阵

| 异常场景 | 降级策略 |
|----------|----------|
| MediaQuery 获取失败 | 降级为 compact 单栏布局 |
| DisplayFeature 检测异常 | 忽略折叠屏特性，使用标准布局 |
| 布局切换时动画卡顿 | 跳过动画，直接切换（低端设备） |

| 横竖屏切换时状态丢失 | 通过 Provider 持久化选中状态 |
| 键盘弹出导致布局溢出 | 强制 ResizeToAvoidBottomInset + 滚动包裹 |

class LayoutErrorBoundary extends StatefulWidget {
  final Widget child;
  final Widget Function(BuildContext, FlutterErrorDetails)? errorBuilder;

  const LayoutErrorBoundary({
    super.key,
    required this.child,
    this.errorBuilder,
  });

  @override
  State<LayoutErrorBoundary> createState() => _LayoutErrorBoundaryState();
}

class _LayoutErrorBoundaryState extends State<LayoutErrorBoundary> {
  bool _hasError = false;

  @override
  Widget build(BuildContext context) {
    if (_hasError) {
      // 降级为最基础的单栏布局
      return Scaffold(
        body: SafeArea(child: widget.child),
      );
    }
    return widget.child;
  }

  @override
  void didUpdateWidget(LayoutErrorBoundary oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (_hasError) {
      // 重建时重置错误状态
      setState(() => _hasError = false);
    }
  }
}

### 12.2 布局切换动画优化

```dart
/// 布局模式切换动画
class AnimatedLayoutSwitcher extends StatelessWidget {
  final LayoutMode layoutMode;
  final Map<LayoutMode, Widget> layouts;

  const AnimatedLayoutSwitcher({
    super.key,
    required this.layoutMode,
    required this.layouts,
  });

  @override
  Widget build(BuildContext context) {
    return AnimatedSwitcher(
      duration: const Duration(milliseconds: 300),
      switchInCurve: Curves.easeOutCubic,
      switchOutCurve: Curves.easeInCubic,
      layoutBuilder: (Widget? currentChild, List<Widget> previousChildren) {
        // 使用 Stack 实现平滑过渡，避免闪烁
        return Stack(
          alignment: Alignment.centerLeft,
          children: <Widget>[
            ...previousChildren,
            if (currentChild != null) currentChild,
          ],
        );
      },
      child: KeyedSubtree(
        key: ValueKey(layoutMode),
        child: layouts[layoutMode] ?? const SizedBox.shrink(),
      ),
    );
  }
}
