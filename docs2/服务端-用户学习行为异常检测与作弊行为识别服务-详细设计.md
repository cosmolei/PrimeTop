# 服务端 - 用户学习行为异常检测与作弊行为识别服务 - 详细设计

> **版本**: v1.0  
> **日期**: 2026-06-07  
> **模块**: 安全风控 / 学习行为分析  
> **优先级**: P1  
> **依赖**: 设备指纹与反作弊风控引擎、用户服务、练习测评会话服务、答题记录服务

---

## 1. 模块概述

### 1.1 功能定位

本服务面向 PrimeTop 学习场景，通过实时与离线结合的方式，检测用户在学习过程中的异常行为模式，识别作弊、滥用、自动化攻击等违规行为，并执行相应的风控措施。

与 `设备指纹与反作弊风控引擎` 不同，后者聚焦于设备级别的身份识别和通用反作弊，本服务专注于 **学习业务层面的行为异常分析**，包括答题作弊检测、AI 接口滥用识别、多账号协同作弊等。

### 1.2 设计目标

1. **实时检测**：在答题、AI 提问等关键操作时实时评估风险，毫秒级响应
2. **离线分析**：定期批量分析历史行为，发现慢速模式和团伙作弊
3. **精准识别**：基于多维特征和行为建模，减少误判
4. **渐进处置**：根据风险等级执行从警告到封禁的梯度处罚
5. **可解释性**：每次风控决策都有清晰的证据链，支持人工复核

### 1.3 检测场景

| 场景 | 描述 | 危害等级 |
|------|------|----------|
| 答题秒答作弊 | 极短时间内连续答对大量题目，未阅读题干 | 高 |
| 答案泄露/共享 | 多个用户对同一套题答案高度一致且错误模式相同 | 高 |
| AI 接口滥用 | 使用自动化脚本高频调用 AI 辅导获取非学习内容 | 高 |
| 拍照搜题刷量 | 批量拍照搜题但不查看解析，仅为消耗资源或测试接口 | 中 |
| 多账号协作作弊 | 多个账号交替使用，规避单账号使用限制 | 中 |
| 模拟替考 | 非本人操作考试模拟，替他人完成 | 中 |
| 虚假学习时长 | 挂机刷时长，未进行有效学习行为 | 低 |
| 积分/成就刷量 | 通过异常模式快速获取积分、成就或打卡记录 | 中 |

## 4. 检测规则详细设计

### 4.1 答题秒答作弊检测

#### 4.1.1 检测逻辑

```
规则编码: SPEED_CHEATING_V1
触发条件: 用户提交答案

检测维度:
1. 答题耗时 < 题目字数 × 最小阅读时间系数
2. 连续 N 道题答题时间均低于阈值
3. 答题时间分布异常均匀（标准差极小）
4. 正确率远超用户历史基线

参数:
  min_read_speed: 3 字/秒      -- 最小阅读速度
  consecutive_threshold: 5     -- 连续异常题数阈值
  accuracy_spike: 0.3          -- 正确率偏离基线上限
  time_uniformity_cv: 0.1      -- 时间变异系数阈值
  window_minutes: 30           -- 滑动窗口(分钟)

风险加分:
  连续5题: +15
  连续10题: +30
  连续20题: +50
```

#### 4.1.2 核心算法

```java
/**
 * 答题速度异常检测器
 */
@Component
public class SpeedCheatingDetector {
    
    @Autowired
    private RedisTemplate<String, String> redis;
    
    @Autowired
    private UserRiskProfileService profileService;
    
    private static final String ANSWER_SPEED_KEY = "risk:speed:%d";
    private static final int WINDOW_SECONDS = 1800; // 30分钟
    private static final int MIN_READ_SPEED = 3;    // 3字/秒
    
    public DetectionResult detect(Long userId, AnswerSubmitEvent event) {
        DetectionResult result = new DetectionResult();
        
        // 1. 计算本题最小合理答题时间
        int charCount = event.getQuestionText().length();
        int minTimeMs = (charCount / MIN_READ_SPEED) * 1000;
        
        // 题型修正: 选择题增加选项阅读时间, 计算题增加计算时间
        minTimeMs += getTypeAdjustmentMs(event.getQuestionType());
        
        boolean isFast = event.getTimeSpentMs() < minTimeMs;
        
        // 2. 记录到滑动窗口
        String key = String.format(ANSWER_SPEED_KEY, userId);
        long now = System.currentTimeMillis();
        
        String entry = String.format("%d:%s:%d",
            event.getTimeSpentMs(),
            event.isCorrect() ? "1" : "0",
            now);
        
        // Redis sorted set, score = timestamp
        redis.opsForZSet().add(key, entry, now);
        // 清理窗口外数据
        redis.opsForZSet().removeRangeByScore(key, 0, now - WINDOW_SECONDS * 1000L);
        
        // 3. 分析窗口内的答题模式
        Set<String> recent = redis.opsForZSet().rangeByScore(key, 
            now - WINDOW_SECONDS * 1000L, now);
        
        if (recent == null || recent.size() < 5) {
            return result.clean();
        }
        
        List<Long> times = recent.stream()
            .map(e -> Long.parseLong(e.split(":")[0]))
            .collect(Collectors.toList());
        
        long fastCount = times.stream()
            .filter(t -> t < minTimeMs)
            .count();
        
        // 4. 连续异常检测
        int consecutive = countConsecutive(times, minTimeMs);
        
        // 5. 时间均匀性检测 (CV = std/mean)
        double cv = calculateCV(times);
        boolean isUniform = cv < 0.1 && times.size() >= 10;
        
        // 6. 正确率偏离检测
        long correctCount = recent.stream()
            .filter(e -> e.split(":")[1].equals("1"))
            .count();
        double currentAccuracy = (double) correctCount / recent.size();
        
        double baselineAccuracy = profileService.getBaselineAccuracy(userId);
        boolean accuracySpike = (currentAccuracy - baselineAccuracy) > 0.3;
        
        // 7. 综合判定
        double score = 0;
        List<String> hitRules = new ArrayList<>();
        
        if (consecutive >= 5) {
            score += 15;
            hitRules.add("CONSECUTIVE_FAST_" + consecutive);
        }
        if (consecutive >= 10) {
            score += 15; // 额外加分
        }
        if (consecutive >= 20) {
            score += 20;
        }
        if (isUniform) {
            score += 20;
            hitRules.add("UNIFORM_TIMING");
        }
        if (accuracySpike) {
            score += 15;
            hitRules.add("ACCURACY_SPIKE");
        }
        
        if (score > 0) {
            result.setRiskScore(Math.min(score, 100));
            result.setHitRules(hitRules);
            result.setEvidence(Map.of(
                "consecutiveFast", consecutive,
                "timeCV", cv,
                "currentAccuracy", currentAccuracy,
                "baselineAccuracy", baselineAccuracy,
                "windowSize", recent.size(),
                "questionId", event.getQuestionId(),
                "timeSpentMs", event.getTimeSpentMs(),
                "minExpectedMs", minTimeMs
            ));
        }
        
        return result;
    }
    
    private int getTypeAdjustmentMs(String questionType) {
        return switch (questionType) {
            case "SINGLE_CHOICE", "MULTI_CHOICE" -> 3000; // 选项阅读时间
            case "FILL_BLANK" -> 5000;                     // 输入时间
            case "CALCULATION", "PROOF" -> 15000;          // 计算推导时间
            case "ESSAY" -> 30000;                         // 作答时间
            default -> 2000;
        };
    }
    
    private int countConsecutive(List<Long> times, long threshold) {
        int max = 0, current = 0;
        for (Long t : times) {
            if (t < threshold) {
                current++;
                max = Math.max(max, current);
            } else {
                current = 0;
            }
        }
        return max;
    }
    
    private double calculateCV(List<Long> values) {
        double mean = values.stream().mapToLong(Long::longValue).average().orElse(1.0);
        double variance = values.stream()
            .mapToDouble(v -> Math.pow(v - mean, 2))
            .average().orElse(0.0);
        double std = Math.sqrt(variance);
        return mean > 0 ? std / mean : 0;
    }
}
```

### 4.2 答案泄露/共享检测

#### 4.2.1 检测逻辑

```
规则编码: ANSWER_LEAKAGE_V1
触发条件: 练习/考试提交后离线分析

检测维度:
1. 答案序列相似度: 两组答案序列完全一致或高度相似
2. 错误模式一致性: 答错的题目和选错的选项完全相同
3. 答题时间相关性: 两个用户答题节奏高度相关
4. 时间窗口: 两份答题在相近时间段完成
5. 地理位置/IP: 同一网络环境

参数:
  answer_similarity_threshold: 0.95   -- 答案相似度阈值
  error_pattern_match: 0.9            -- 错误模式匹配阈值

/**
 * 答案泄露离线分析器
 * 每日定时执行，分析前一天的所有练习/考试记录
 */
@Component
public class AnswerLeakageAnalyzer {
    
    @Scheduled(cron = "0 0 3 * * *") // 每天凌晨3点执行
    public void analyzeDaily() {
        LocalDate yesterday = LocalDate.now().minusDays(1);
        
        // 1. 获取所有练习集/考试记录
        List<PracticeSession> sessions = practiceRepo
            .findByDateAndMinQuestions(yesterday, 10);
        
        // 2. 按题目集合分组 (相同题目集才可能泄露)
        Map<String, List<PracticeSession>> byQuestionSet = sessions.stream()
            .collect(Collectors.groupingBy(PracticeSession::getQuestionSetHash));
        
        List<RiskEvent> detectedEvents = new ArrayList<>();
        
        for (Map.Entry<String, List<PracticeSession>> entry : byQuestionSet.entrySet()) {
            List<PracticeSession> group = entry.getValue();
            if (group.size() < 2) continue;
            
            // 3. 两两比较
            for (int i = 0; i < group.size(); i++) {
                for (int j = i + 1; j < group.size(); j++) {
                    LeakageResult result = compareSessions(group.get(i), group.get(j));
                    if (result.isSuspicious()) {
                        detectedEvents.add(buildRiskEvent(result));
                    }
                }
            }
        }
        
        // 4. 批量写入风险事件
        if (!detectedEvents.isEmpty()) {
            riskEventRepo.batchInsert(detectedEvents);
            log.warn("检测到 {} 起疑似答案泄露事件", detectedEvents.size());
        }
    }
    
    private LeakageResult compareSessions(PracticeSession a, PracticeSession b) {
        LeakageResult result = new LeakageResult(a.getUserId(), b.getUserId());
        
        // 答案序列
        List<String> answersA = a.getAnswers();
        List<String> answersB = b.getAnswers();
        
        // 只比较相同的题目
        List<Integer> commonIndices = findCommonQuestions(a, b);
        if (commonIndices.size() < 10) {
            return result.notSuspicious();
        }
        
        // 1. 答案相似度
        int matchCount = 0;
        int errorMatchCount = 0;
        int totalErrors = 0;
        
        for (int idx : commonIndices) {
            if (answersA.get(idx).equals(answersB.get(idx))) {
                matchCount++;
            }
            // 错误模式一致
            boolean aWrong = !a.isCorrect(idx);
            boolean bWrong = !b.isCorrect(idx);
            if (aWrong && bWrong) {
                totalErrors++;
                if (answersA.get(idx).equals(answersB.get(idx))) {
                    errorMatchCount++;
                }
            }
        }
        
        double answerSimilarity = (double) matchCount / commonIndices.size();
        double errorPatternMatch = totalErrors > 0 
            ? (double) errorMatchCount / totalErrors : 0.0;
        
        // 2. 时间相关性 (Pearson)
        List<Long> timesA = commonIndices.stream()
            .map(idx -> a.getTimeSpentMs(idx)).collect(Collectors.toList());
        List<Long> timesB = commonIndices.stream()
            .map(idx -> b.getTimeSpentMs(idx)).collect(Collectors.toList());
        double timeCorrelation = pearsonCorrelation(timesA, timesB);
        
        // 3. 综合判定
        boolean suspicious = answerSimilarity >= 0.95 
            && errorPatternMatch >= 0.9
            && timeCorrelation >= 0.85;
        
        // 时间间隔检查
        long timeGapHours = Math.abs(Duration.between(
            a.getCompletedAt(), b.getCompletedAt()).toHours());
        suspicious = suspicious && timeGapHours <= 48;
        
        if (suspicious) {
            result.setSuspicious(true);
            result.setAnswerSimilarity(answerSimilarity);
            result.setErrorPatternMatch(errorPatternMatch);
            result.setTimeCorrelation(timeCorrelation);
            result.setTimeGapHours(timeGapHours);
            result.setCommonQuestionCount(commonIndices.size());
            result.setRiskScore(calculateLeakageScore(
                answerSimilarity, errorPatternMatch, timeCorrelation));
        }
        
        return result;
    }
    
    private double calculateLeakageScore(double sim, double error, double time) {
        // 加权评分
        return sim * 40 + error * 30 + time * 30;
    }
}

```java
/**
 * AI 接口滥用实时检测
 * 在 AI 服务调用前置拦截
 */
@Component
public class AiAbuseDetector {
    
    @Autowired
    private RedisTemplate<String, String> redis;
    
    private static final String RATE_KEY = "risk:ai_rate:%d:%s";
    
    public DetectionResult checkBeforeCall(Long userId, String question, 
                                            String conversationId) {
        DetectionResult result = new DetectionResult();
        
        // 1. 频率检查
        long minuteCount = incrementAndGet(RATE_KEY, userId, "min", 60);
        long hourCount = incrementAndGet(RATE_KEY, userId, "hour", 3600);
        long dayCount = incrementAndGet(RATE_KEY, userId, "day", 86400);
        
        if (minuteCount > 10) {
            return result.flag("RATE_MINUTE", 30, 
                Map.of("count", minuteCount, "limit", 10));
        }
        if (hourCount > 60) {
            return result.flag("RATE_HOUR", 40,
                Map.of("count", hourCount, "limit", 60));
        }
        if (dayCount > 200) {
            return result.flag("RATE_DAY", 60,
                Map.of("count", dayCount, "limit", 200));
        }
        
        // 2. 相似问题检测 (与最近3个问题比较)
        String recentKey = String.format("risk:ai_recent:%d", userId);
        List<String> recentQuestions = redis.opsForList().range(recentKey, 0, 2);
        
        if (recentQuestions != null) {
            for (String recent : recentQuestions) {
                double similarity = textSimilarity(question, recent);
                if (similarity > 0.85) {
                    return result.flag("SIMILAR_QUESTION", 25,
                        Map.of("similarity", similarity, "recentQuestion", recent));
                }
            }
        }
        
        // 3. 非学习内容检测 (快速关键词匹配)
        if (isNonEducational(question)) {
            return result.flag("NON_EDUCATIONAL", 20,
                Map.of("question", question));
        }
        
        // 记录本次问题
        redis.opsForList().leftPush(recentKey, question);
        redis.opsForList().trim(recentKey, 0, 4);
        redis.expire(recentKey, Duration.ofHours(1));
        
        return result.clean();
    }
    
    private long incrementAndGet(String pattern, Long userId, String window, int seconds) {
        String key = String.format(pattern, userId, window);
        Long count = redis.opsForValue().increment(key);
        if (count != null && count == 1) {
            redis.expire(key, Duration.ofSeconds(seconds));
        }
        return count != null ? count : 0;
    }
    
    /**
     * 简单的文本相似度 (Jaccard + 编辑距离混合)
     */
    private double textSimilarity(String a, String b) {
        if (a.equals(b)) return 1.0;
        
        Set<String> setA = Arrays.stream(a.split(""))
            .filter(s -> s.trim().length() > 0)
            .collect(Collectors.toSet());
        Set<String> setB = Arrays.stream(b.split(""))
            .filter(s -> s.trim().length() > 0)
            .collect(Collectors.toSet());
        
        Set<String> intersection = new HashSet<>(setA);
        intersection.retainAll(setB);
        
        Set<String> union = new HashSet<>(setA);
        union.addAll(setB);
        
        double jaccard = union.isEmpty() ? 0 : (double) intersection.size() / union.size();
        
        // 长度相似度
        int maxLen = Math.max(a.length(), b.length());
        if (maxLen == 0) return 1.0;
        int editDist = levenshteinDistance(a, b);
        double lengthSim = 1.0 - (double) editDist / maxLen;
        
        return jaccard * 0.4 + lengthSim * 0.6;
    }
}
