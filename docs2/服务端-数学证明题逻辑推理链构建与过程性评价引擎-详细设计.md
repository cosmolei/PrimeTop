## 1. 概述

### 1.1 功能目标

针对初高中数学中的**证明题与逻辑推理题**，提供从题目理解、推理链构建、学生作答过程评价到反馈生成的完整能力。与通用"理科解题"引擎不同，本引擎聚焦于：

1. **形式逻辑推理**：处理"因为…所以…"（∵/∴）结构化的逻辑推理过程，而非纯计算。
2. **证明链完整性校验**：检测推理链中是否存在逻辑跳跃、循环论证或条件遗漏。
3. **学生证明过程分步评价**：对学生的每一步推理进行正确性判定和充分性分析。
4. **证明思路引导**：在学生卡住时，给出下一步方向提示而非直接告知结论。

### 1.2 适用题型范围

| 题型分类 | 典型题目 | 学段 |
|---------|---------|------|
| 平面几何证明 | 全等三角形、相似三角形、圆的性质 | 初中 |
| 立体几何证明 | 线面平行/垂直、二面角 | 高中 |
| 代数证明 | 不等式证明、数列归纳法、函数性质 | 高中 |
| 解析几何证明 | 轨迹方程、定点定值问题 | 高中 |
| 逻辑推理题 | 充分必要条件、命题等价 | 初中-高中 |
| 组合与数论证明 | 组合恒等式、整除性、存在性 | 高中竞赛/拓展 |

### 1.3 核心挑战

1. **非结构化输入**：学生证明过程为自然语言+数学符号混合体，难以用固定模板解析。
2. **逻辑等价判定**：同一步骤可有多种等价表述（如"SAS"与"边角边"），需语义级理解。
3. **充分性判定**：某一步"正确"不等于"充分"，需要判定该步是否足以支撑后续推理。
4. **多解路径**：同一证明题可有多条有效路径，不能用单一标准答案硬匹配。

### 2.2 内部模块职责

| 模块 | 职责 |
|------|------|
| 题目理解与条件提取器 | 解析证明题的已知条件、求证目标、图形信息 |
| 推理链构建器 | 将学生证明过程或 AI 生成的证明拆解为有向推理步骤链 |
| 证明模式知识库 | 存储常见证明模式（SAS/ASA/SSS、韦达定理法、放缩法等） |
| 参考推理链生成器 | 基于 AI + 知识库生成标准推理链（用于评分参考） |
| 过程评价器 | 对比学生推理链与参考链，逐步判定正确性与充分性 |
| 反馈生成器 | 生成适龄化反馈文本（提示、纠错、表扬） |

## 3. 数据结构定义

### 3.1 证明题结构化表示

```typescript
/**
 * 证明题结构化模型
 */
interface ProofQuestion {
  id: string;                          // 题目唯一ID
  questionType: ProofType;             // 证明类型
  subject: 'math';                     // 学科
  stage: 'junior' | 'senior';         // 初中/高中
  grade: number;                       // 年级
  
  // 题目内容
  stimulus: string;                    // 题干文本（含 LaTeX）
  givenConditions: ProofCondition[];   // 已知条件（结构化）
  toProve: string;                     // 求证目标
  figureUrl?: string;                  // 图形图片URL（如有）
  figureStructured?: GeoFigure;        // 结构化图形（如有）
  
  // 元数据
  knowledgePoints: string[];           // 关联知识点ID
  proofPatterns: string[];             // 预期可能用到的证明模式
  difficulty: 1 | 2 | 3 | 4 | 5;      // 难度等级
  fullScore: number;                   // 满分
  
  // 参考证明（题库提供或AI生成）
  referenceProofs: ReferenceProof[];   // 参考证明（可有多条路径）
  
  // 评分配置
  scoringRubric: ProofScoringRubric;   // 评分量规
}

type ProofType = 
  | 'geometric_congruence'     // 全等证明
  | 'geometric_similarity'     // 相似证明
  | 'geometric_circle'         // 圆的性质
  | 'geometric_solid'          // 立体几何
  | 'analytic_geometry'        // 解析几何
  | 'algebraic_inequality'     // 不等式证明
  | 'algebraic_sequence'       // 数列证明
  | 'function_property'        // 函数性质
  | 'mathematical_induction'   // 数学归纳法
  | 'logical_reasoning'        // 逻辑推理
  | 'combinatorial'            // 组合证明
  | 'number_theory';           // 数论证明

/**
 * 结构化条件
 */
interface ProofCondition {
  id: string;                          // 条件标识，如 "C1", "C2"
  rawText: string;                     // 原始文本
  structured: ConditionAtom;           // 结构化表示
  type: 'explicit' | 'implicit' | 'derived';  // 显式给定/隐含/推导得到
  scope?: string;                      // 适用范围（如"在△ABC中"）
}

/**
 * 条件原子——结构化的数学命题
 */
interface ConditionAtom {
  predicate: string;                   // 谓词，如 "equal", "parallel", "perpendicular", "congruent"
  subject: MathEntity;                 // 主体
  object: MathEntity;                  // 客体
  qualifier?: string;                  // 限定词，如 "at point P"
  negated: boolean;                    // 是否否定
}

/**
 * 数学实体
 */
interface MathEntity {
  kind: 'point' | 'line' | 'segment' | 'angle' | 'triangle' 
      | 'circle' | 'arc' | 'polygon' | 'function' | 'variable' 
      | 'expression' | 'set';
  label: string;                       // 标签，如 "AB", "∠ABC", "△ABC"
  properties?: Record<string, string>; // 属性，如 { length: "5", unit: "cm" }
}

/**
 * 参考证明
 */
interface ReferenceProof {
  id: string;
  proofPath: ProofStep[];              // 推理步骤序列
  totalSteps: number;
  patternsUsed: string[];              // 使用的证明模式
  keyInsight: string;                  // 关键突破点描述
  estimatedDifficulty: number;         // 推理难度
}

/**
 * 单个推理步骤
 */
interface ProofStep {
  stepNumber: number;                  // 步骤序号
  premises: string[];                  // 前置条件ID列表（依赖哪些已知条件或前序结论）
  reasoning: string;                   // 推理依据文本，如"因为AB=AC，∠BAC=∠DAE"
  conclusion: string;                  // 本步结论，如"所以△ABC≅△ADE (SAS)"
  justification: ProofJustification;   // 结构化依据
  subSteps?: ProofStep[];              // 子步骤（嵌套证明，如分类讨论）
}

/**
 * 推理依据
 */
interface ProofJustification {
  type: JustificationType;
  pattern?: string;                    // 应用的证明模式/定理名
  theorem?: string;                    // 引用的定理
  definition?: string;                 // 引用的定义
  axiom?: string;                      // 引用的公理
  detail: string;                      // 详细说明
}

type JustificationType = 
  | 'given'                    // 已知条件
  | 'theorem'                  // 定理引用
  | 'definition'               // 定义引用
  | 'axiom'                    // 公理引用
  | 'previously_proven'        // 前序结论
  | 'algebraic_manipulation'   // 代数变形
  | 'substitution'             // 等量代换
  | 'contradiction'            // 反证法
  | 'induction'                // 数学归纳法
  | 'construction';            // 辅助线/辅助函数构造
```

### 3.3 评价结果

```typescript
/**
 * 证明过程评价结果
 */
interface ProofEvaluationResult {
  evaluationId: string;
  answerId: string;
  
  overallScore: number;                // 总分
  maxScore: number;
  scoreBreakdown: ScoreBreakdown;      // 分项得分
  
  stepEvaluations: StepEvaluation[];   // 逐步评价
  chainAnalysis: ProofChainAnalysis;   // 推理链整体分析
  
  logicalGaps: LogicalGap[];           // 逻辑漏洞
  circularReasoning: boolean;          // 是否存在循环论证
  missingConditions: string[];         // 未使用的已知条件
  extraAssumptions: string[];          // 额外假设（使用了未给定的条件）
  
  correctnessVerdict: 'correct' | 'partially_correct' | 'incorrect' | 'incomplete';
  
  feedback: ProofFeedback;             // 反馈内容
  knowledgePointsGap: string[];        // 暴露的知识薄弱点
}

/**
 * 分项得分
 */
interface ScoreBreakdown {
  logicCorrectness: number;            // 逻辑正确性（每步推理是否正确）
  completeness: number;                // 完整性（是否完成证明）
  rigor: number;                       // 严谨性（理由是否充分）
  presentation: number;                // 规范性（格式、符号使用）
  method: number;                      // 方法合理性（证明路径选择）
}

/**
 * 单步评价
 */
interface StepEvaluation {
  stepIndex: number;                   // 对应学生步骤序号
  
  isCorrect: boolean;                  // 该步推理是否正确
  isSufficient: boolean;               // 该步理由是否充分
  isNecessary: boolean;                // 该步是否必要（非冗余）
  
  correctnessScore: number;            // 正确性得分 (0-1)
  sufficiencyScore: number;            // 充分性得分 (0-1)
  
  matchedReferenceStep?: number;       // 匹配到的参考步骤序号
  alternativePath: boolean;            // 是否属于另解路径
  
  issues: StepIssue[];                 // 该步存在的问题
  feedback: string;                    // 该步反馈文本
  
  // 详细判定
  premiseValidation: PremiseCheck[];   // 前提条件检查
  conclusionValidation: ConclusionCheck; // 结论检查
  justificationValidation?: JustificationCheck; // 依据检查
}

/**
 * 前提条件检查
 */
interface PremiseCheck {
  premise: string;                     // 前提内容
  status: 'valid' | 'invalid' | 'unverified' | 'misused';
  detail: string;                      // 说明
  sourceStep?: number;                 // 前提来源步骤
}

/**
 * 结论检查
 */
interface ConclusionCheck {
  statement: string;                   // 结论内容
  followsFromPremises: boolean;        // 是否能从前提推出
  counterexampleExists: boolean;       // 是否存在反例
  detail: string;
}

/**
 * 步骤问题
 */
interface StepIssue {
  type: IssueType;
  severity: 'error' | 'warning' | 'suggestion';
  description: string;
  correction?: string;                 // 建议修正
}

type IssueType = 
  | 'logical_leap'             // 逻辑跳跃
  | 'missing_justification'    // 缺少依据
  | 'wrong_justification'      // 依据错误
  | 'unjustified_assumption'   // 未证自明（额外假设）
  | 'circular_reasoning'       // 循环论证
  | 'notation_error'           // 符号错误
  | 'calculation_error'        // 计算错误
  | 'redundant_step'           // 冗余步骤
  | 'incomplete_case'          // 情况遗漏（如分类讨论不全）
  | 'wrong_direction';         // 方向错误（证充分vs证必要）

/**
 * 推理链整体分析
 */
interface ProofChainAnalysis {
  chainLength: number;                 // 推理链长度
  branchPoints: number;                // 分支点数量（分类讨论）
  longestPath: number;                 // 最长路径
  
  chainCompleteness: number;           // 链完整性 0-1
  chainCoherence: number;             // 链连贯性 0-1（步骤间逻辑衔接）
  
  structure: 'linear' | 'branched' | 'cyclic' | 'fragmented';
  
  // 从已知到结论的覆盖度
  conditionCoverage: number;           // 已用条件比例
  goalReached: boolean;                // 是否到达求证目标
  
  // 与参考证明的比较
  pathSimilarity: number;              // 路径相似度 0-1
  isAlternativePath: boolean;          // 是否属于有效另解
}
```

## 4. API 接口设计

### 4.1 证明题结构化解析接口

将非结构化的证明题文本解析为结构化模型。

```
POST /api/v1/proof/parse-question
```

**请求体：**
```json
{
  "rawText": "如图，在△ABC中，AB=AC，D是BC的中点，DE⊥AB于E，DF⊥AC于F。求证：DE=DF。",
  "figureUrl": "https://cdn.primetop.com/figures/q12345.png",
  "stage": "junior",
  "grade": 8
}
```

**响应体：**
```json
{
  "code": 0,
  "data": {
    "questionId": "pq_temp_20260622001",
```

### 4.2 参考推理链生成接口

基于结构化题目，生成一条或多条参考证明路径。

```
POST /api/v1/proof/generate-reference
```

**请求体：**
```json
{
  "questionId": "pq_12345",
  "maxPaths": 3,
  "preferMethod": "congruent_triangle",  // 偏好方法，可选
  "detail": "detailed"                   // detailed | concise
}
```

**响应体（摘要）：**
```json

  "code": 0,

    "referenceProofs": [
```
