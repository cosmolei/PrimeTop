# 客户端-直播课堂观看与实时互动学习页面架构与交互设计

## 1. 概述

### 1.1 模块定位

直播课堂观看与互动学习页面是 PrimeTop 客户端中学生在移动端参与教师实时授课的核心交互入口。该页面承载直播视频流播放、实时消息互动、课堂答题、举手提问、课件同步查看等核心功能，是连接服务端直播流媒体服务与课堂实时互动引擎的客户端枢纽。

### 1.2 核心职责

| 职责 | 说明 |
| --- | --- |
| 直播流拉取与渲染 | 支持 RTMP / HLS / WebRTC 拉流，低延迟播放教师音视频 |
| 实时消息通道 | 基于 WebSocket 的双向通信，承载弹幕、提问、答题等交互信令 |
| 课堂互动功能 | 答题卡片弹出、举手连麦、表情回应、随堂测验提交 |
| 课件同步浏览 | 教师端翻页/标注/白板操作的实时同步与渲染 |
| 学习行为采集 | 观看时长、互动频率、答题记录等行为数据埋点上报 |
| 异常恢复 | 断网重连、弱网降级、直播流切换与状态恢复 |

- `服务端-教育直播课堂与实时互动教学流媒体服务` — 提供直播推拉流地址、房间信令

- `客户端-学情分析页面架构与报告展示交互` — 课堂参与数据回写
- `客户端-首页与学习工作台页面架构与交互设计` — 直播入口卡片
- `客户端-AI对话页面交互与组件架构` — 课后 AI 答疑衔接

### 4.2 Riverpod 状态管理实现

```dart
/// 直播间状态Provider
@riverpod
class LiveClassRoom extends Notifier<LiveRoomState> {
  late LiveLessonInfo _lessonInfo;
  LiveSignalClient? _signalClient;
  LiveVideoController? _videoController;

  @override
  LiveRoomState build() {
    return LiveRoomState.initial();
  }

  /// 初始化直播间
  Future<void> init(String lessonId) async {
    state = state.copyWith(phase: LiveRoomPhase.loading);

    try {
      // 1. 获取课程信息
      _lessonInfo = await ref.read(liveLessonRepositoryProvider)
          .getLessonInfo(lessonId);

      // 2. 判断课程状态
      switch (_lessonInfo.status) {
        case LiveLessonStatus.scheduled:
        case LiveLessonStatus.aboutToStart:
          state = state.copyWith(
            phase: LiveRoomPhase.preview,
            lessonInfo: _lessonInfo,
          );
          _scheduleLiveStartWatcher();
          break;

        case LiveLessonStatus.live:
          await _joinLiveRoom();
          break;

        case LiveLessonStatus.replay:
          state = state.copyWith(
            phase: LiveRoomPhase.replay,
            lessonInfo: _lessonInfo,
          );
          break;

        default:
          state = state.copyWith(
            phase: LiveRoomPhase.error,
            errorMessage: '课程状态异常: ${_lessonInfo.status}',
          );
      }
    } catch (e) {
      state = state.copyWith(
        phase: LiveRoomPhase.error,
        errorMessage: '加载失败: $e',
      );
    }
  }

  /// 加入直播间
  Future<void> _joinLiveRoom() async {
    state = state.copyWith(phase: LiveRoomPhase.connecting);

    // 1. 建立信令通道
    _signalClient = LiveSignalClient(
      lessonId: _lessonInfo.lessonId,
      userId: ref.read(currentUserIdProvider),
      onMessage: _handleSignalMessage,
      onConnectionChange: _handleConnectionChange,
    );
    await _signalClient!.connect();

    // 2. 初始化视频播放器
    _videoController = LiveVideoController(
      streamConfig: _lessonInfo.streamConfig,
      onStateChanged: _handleVideoStateChange,
    );
    await _videoController!.initialize();

    // 3. 发送进入房间信令
    await _signalClient!.sendEnterRoom(
      deviceId: await _getDeviceId(),
      clientInfo: await _getClientInfo(),
    );

    // 4. 启动心跳
    _startHeartbeat();

    state = state.copyWith(
      phase: LiveRoomPhase.live,
      lessonInfo: _lessonInfo,
      connectionState: LiveConnectionState.initial(),
      enteredAt: DateTime.now(),
    );

    // 5. 埋点上报
    ref.read(analyticsProvider).logEvent(
      'live_class_enter',
      params: {
        'lesson_id': _lessonInfo.lessonId,
        'subject': _lessonInfo.subject,
        'entry_point': state.entryPoint?.name,
      },
    );
  }

  /// 处理信令消息
  void _handleSignalMessage(LiveMessage message) {
    switch (message.type) {
      case LiveMessageType.pollStart:
        final poll = message as PollMessage;
        state = state.copyWith(
          activePolls: {...state.activePolls, poll.pollId: poll},
        );
        break;

      case LiveMessageType.pollEnd:
        final poll = message as PollMessage;
        state = state.copyWith(
          activePolls: state.activePolls..remove(poll.pollId),
          recentPolls: [poll, ...state.recentPolls].take(20).toList(),
        );
        break;

      case LiveMessageType.system:
      case LiveMessageType.text:
      case LiveMessageType.question:
      case LiveMessageType.answer:
        state = state.copyWith(
          messages: [...state.messages, message],
        );
        break;

      case LiveMessageType.notice:
        final notice = message as TextMessage;
        state = state.copyWith(pinnedNotice: notice.content);
        break;

      case LiveMessageType.reward:
        _showRewardAnimation(message);
        break;

      default:
        break;
    }
  }

  /// 处理连接状态变化
  void _handleConnectionChange(ConnectionStatus status) {
    switch (status) {
      case ConnectionStatus.reconnecting:
        state = state.copyWith(
          connectionState: state.connectionState.copyWith(
            signalStatus: ConnectionStatus.reconnecting,
          ),
        );
        break;

      case ConnectionStatus.connected:
        state = state.copyWith(
          connectionState: state.connectionState.copyWith(
            signalStatus: ConnectionStatus.connected,
            lastHeartbeat: DateTime.now(),
          ),
        );
        break;

      case ConnectionStatus.failed:
        _enterDegradedMode();
        break;

      default:
        break;
    }
  }

  /// 进入降级模式
  void _enterDegradedMode() {
    state = state.copyWith(
      phase: LiveRoomPhase.degraded,
      connectionState: state.connectionState.copyWith(
        videoStatus: ConnectionStatus.failed,
      ),
    );

    // 尝试切换到纯音频模式
    _videoController?.switchToAudioOnly();

    // 展示降级提示
    ref.read(toastProvider).show(
      '网络较差，已切换为音频模式',
      duration: const Duration(seconds: 3),
    );
  }

  /// 提交答题
  Future<void> submitPollAnswer(String pollId, String selectedKey) async {
    final poll = state.activePolls[pollId];
    if (poll == null || poll.status != PollStatus.waiting) return;

    await _signalClient!.submitPollAnswer(
      pollId: pollId,
      selectedKey: selectedKey,
    );

    // 更新本地状态
    state = state.copyWith(
      activePolls: {
        ...state.activePolls,
        pollId: poll.copyWith(status: PollStatus.submitted),
      },
    );

    // 埋点
    ref.read(analyticsProvider).logEvent(
      'live_poll_submit',
      params: {
        'lesson_id': _lessonInfo.lessonId,
        'poll_id': pollId,
        'selected': selectedKey,
        'time_taken_sec': DateTime.now()
            .difference(poll.timestamp).inSeconds,
      },
    );
  }

  /// 发送消息
  Future<void> sendMessage(String content) async {
    if (content.trim().isEmpty) return;

    // 内容安全检查(本地预检)
    final filterResult = await ref.read(
      contentFilterProvider,
    ).quickCheck(content);
    if (!filterResult.passed) {
      ref.read(toastProvider).show('消息包含不适宜内容，请修改后发送');
      return;
    }

    final message = TextMessage(
      messageId: const Uuid().v4(),
      senderId: ref.read(currentUserIdProvider),
      senderName: ref.read(currentUserNameProvider),
      timestamp: DateTime.now(),
      type: LiveMessageType.text,
      role: MessageRole.student,
      content: content,
      mentionedUsers: const [],
      isPinned: false,
    );

    await _signalClient!.sendMessage(message);
    state = state.copyWith(messages: [...state.messages, message]);
  }

  /// 举手
  Future<void> raiseHand() async {
    await _signalClient!.sendRaiseHand();
    state = state.copyWith(hasRaisedHand: true);
  }

  /// 退出直播间
  Future<void> leave() async {
    final duration = state.enteredAt != null
```

```dart
/// 白板笔迹渲染器
class WhiteboardPainter extends CustomPainter {
  final List<WhiteboardStroke> strokes;

  WhiteboardPainter({required this.strokes});

  @override
  void paint(Canvas canvas, Size size) {
    for (final stroke in strokes) {
      final paint = Paint()
        ..color = Color(stroke.color)
        ..strokeWidth = stroke.strokeWidth
        ..style = PaintingStyle.stroke
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round;

      if (stroke.tool == StrokeTool.highlighter) {
        paint.blendMode = BlendMode.multiply;
        paint.color = Color(stroke.color).withValues(alpha: 0.4);
      }

      if (stroke.points.length < 2) continue;

      final path = Path();
      path.moveTo(stroke.points.first.dx, stroke.points.first.dy);
      for (var i = 1; i < stroke.points.length; i++) {
        // 使用贝塞尔曲线平滑笔迹
        final p0 = stroke.points[i - 1];
        final p1 = stroke.points[i];
        final midPoint = Offset(
          (p0.dx + p1.dx) / 2,
          (p0.dy + p1.dy) / 2,
        );
        path.quadraticBezierTo(p0.dx, p0.dy, midPoint.dx, midPoint.dy);
      }
      canvas.drawPath(path, paint);
    }
  }

  @override
  bool shouldRepaint(covariant WhiteboardPainter oldDelegate) {
    return oldDelegate.strokes != strokes;
  }
}
```

```dart
/// 直播信令客户端 - 管理与服务的实时双向通信
abstract class LiveSignalClient {
  /// 建立连接
  Future<void> connect();

  /// 断开连接
  Future<void> disconnect();

  /// 发送进入房间
  Future<void> sendEnterRoom({
    required String deviceId,
    required ClientInfo clientInfo,
  });

  /// 发送消息
  Future<void> sendMessage(LiveMessage message);

  /// 提交答题
  Future<void> submitPollAnswer({
    required String pollId,
    required String selectedKey,
  });

  /// 举手
  Future<void> sendRaiseHand();

  /// 发送心跳
  Future<void> sendHeartbeat();

  /// 释放资源
  void dispose();
}

/// WebSocket 实现
class WebSocketSignalClient implements LiveSignalClient {
  final String lessonId;
  final String userId;
  final void Function(LiveMessage) onMessage;
  final void Function(ConnectionStatus) onConnectionChange;

  WebSocketChannel? _channel;
  Timer? _heartbeatTimer;
  Timer? _reconnectTimer;
  int _reconnectAttempts = 0;
  bool _disposed = false;

  static const _heartbeatInterval = Duration(seconds: 15);
  static const _maxReconnectAttempts = 5;
  static const _reconnectDelays = [
    Duration(seconds: 1),
    Duration(seconds: 2),
    Duration(seconds: 4),
    Duration(seconds: 8),
    Duration(seconds: 16),
  ];

  WebSocketSignalClient({
    required this.lessonId,
    required this.userId,
    required this.onMessage,
    required this.onConnectionChange,
  });

  @override
  Future<void> connect() async {
    onConnectionChange(ConnectionStatus.connecting);

    final wsUrl = Uri.parse(
      '${Environment.wsBaseUrl}/live/$lessonId?uid=$userId',
    );

    try {
      _channel = WebSocketChannel.connect(wsUrl);
      await _channel!.ready;

      _channel!.stream.listen(
        _onData,
        onError: (error) => _onError(error),
        onDone: () => _onDone(),
      );

      _reconnectAttempts = 0;
      onConnectionChange(ConnectionStatus.connected);
      _startHeartbeat();
    } catch (e) {
      onConnectionChange(ConnectionStatus.failed);
      _scheduleReconnect();
    }
  }

  void _onData(dynamic data) {
    try {
      final json = jsonDecode(data as String) as Map<String, dynamic>;
      final message = LiveMessageDeserializer.deserialize(json);
      onMessage(message);
    } catch (e) {
      debugPrint('Signal message parse error: $e');
    }
  }

  void _onError(dynamic error) {
    debugPrint('WebSocket error: $error');
    _scheduleReconnect();
  }

  void _onDone() {
    if (_disposed) return;
    _scheduleReconnect();
  }

  void _startHeartbeat() {
    _heartbeatTimer?.cancel();
    _heartbeatTimer = Timer.periodic(_heartbeatInterval, (_) {
      _sendRaw({'type': 'heartbeat', 'ts': DateTime.now().millisecondsSinceEpoch});
    });
  }

  void _scheduleReconnect() {
    _heartbeatTimer?.cancel();

    if (_disposed) return;
    if (_reconnectAttempts >= _maxReconnectAttempts) {
      onConnectionChange(ConnectionStatus.failed);
      return;
    }

    onConnectionChange(ConnectionStatus.reconnecting);

    final delay = _reconnectDelays[
      _reconnectAttempts.clamp(0, _reconnectDelays.length - 1)
    ];
    _reconnectAttempts++;

    _reconnectTimer?.cancel();
    _reconnectTimer = Timer(delay, () => connect());
  }

  void _sendRaw(Map<String, dynamic> data) {
    if (_channel != null && _channel!.closeCode == null) {
      _channel!.sink.add(jsonEncode(data));
    }
  }

  @override
  Future<void> sendEnterRoom({
    required String deviceId,
    required ClientInfo clientInfo,
  }) async {
    _sendRaw({
      'type': 'enter_room',
      'device_id': deviceId,
      'client_info': clientInfo.toJson(),
    });
  }

  @override
  Future<void> sendMessage(LiveMessage message) async {
    _sendRaw({
      'type': 'message',
      'payload': message.toJson(),
    });
  }

  @override
  Future<void> submitPollAnswer({
    required String pollId,
    required String selectedKey,
  }) async {
    _sendRaw({
      'type': 'poll_answer',
      'poll_id': pollId,
      'selected': selectedKey,
    });
  }

  @override
  Future<void> sendRaiseHand() async {
    _sendRaw({'type': 'raise_hand'});
  }

  @override
  Future<void> sendHeartbeat() async {
    _sendRaw({'type': 'heartbeat', 'ts': DateTime.now().millisecondsSinceEpoch});
  }

  @override
  void dispose() {
    _disposed = true;
    _heartbeatTimer?.cancel();
    _reconnectTimer?.cancel();
    _channel?.sink.close();
  }

  @override
  Future<void> disconnect() async {
    _heartbeatTimer?.cancel();
    _reconnectTimer?.cancel();
    await _channel?.sink.close();
  }
}
```

### 8.1 异常场景与处理

| 场景 | 原因 | 处理策略 | 用户提示 |

### 8.2 降级策略实现

```dart
/// 降级管理器
class LiveDegradationManager {
  final LiveVideoController videoController;
  final LiveSignalClient signalClient;

  VideoQuality _currentQuality = VideoQuality.auto;
  bool _audioOnly = false;
  int _bufferEvents = 0;
  DateTime _lastBufferTime = DateTime.now();

  /// 记录缓冲事件，触发降级
  void onBuffering() {
    final now = DateTime.now();

    // 30秒内发生3次以上缓冲则降级
    if (now.difference(_lastBufferTime).inSeconds < 30) {
      _bufferEvents++;
    } else {
      _bufferEvents = 1;
    }
    _lastBufferTime = now;

    if (_bufferEvents >= 3 && !_audioOnly) {
      _degrade();
    }
  }

  void _degrade() {
    // 降级路径: auto → hd → sd → audio_only
    switch (_currentQuality) {
      case VideoQuality.auto:
        _currentQuality = VideoQuality.hd;
        videoController.switchQuality(VideoQuality.hd);
        break;
      case VideoQuality.hd:
        _currentQuality = VideoQuality.sd;
        videoController.switchQuality(VideoQuality.sd);
        break;
      case VideoQuality.sd:
        _enterAudioOnlyMode();
        break;
      default:
        _enterAudioOnlyMode();
    }
  }

  void _enterAudioOnlyMode() {
    _audioOnly = true;
    videoController.switchToAudioOnly();
  }

  /// 恢复策略: 网络好转时尝试提升
  void onNetworkRecovered() {
    if (_audioOnly) {
      // 尝试恢复视频
      videoController.tryRecoverVideo();
      _audioOnly = false;
      _currentQuality = VideoQuality.sd;
    } else if (_currentQuality != VideoQuality.auto) {
      _currentQuality = VideoQuality.auto;
      videoController.switchQuality(VideoQuality.auto);
    }
  }
}
```

```dart
    }
```

## 14. 开发注意事项

1. **Flutter WebRTC SDK选型**: 建议使用 `flutter_webrtc` 或集成声网/腾讯云RTC SDK，需评估包体积影响
2. **WebSocket连接复用**: 直播间WebSocket应与全局WebSocket连接(如有)做连接复用或隔离设计
3. **iOS后台音频**: 需配置 `UIBackgroundModes` 的 `audio` 权限，确保后台时音频不中断
4. **Android省电策略**: 后台时关闭视频解码器，前台时快速恢复；使用 `WorkManager` 管理定时心跳
5. **课件资源预加载**: 课程开始前5分钟预加载课件图片到本地缓存
6. **消息序列化**: 使用 `json_serializable` 或 `protobuf` 进行消息序列化，避免手动JSON解析
7. **内存管理**: 直播退出时必须释放视频解码器、WebSocket连接、定时器等资源，防止内存泄漏
