### 2.3 TimerNotifier（核心状态管理器）

```dart
// lib/features/timer/state/timer_notifier.dart

@riverpod
class TimerNotifier extends _$TimerNotifier {
  late MonotonicTimer _monotonicTimer;
  Timer? _uiUpdateTimer;

  @override
  TimerState build() {
    // 启动时从本地存储恢复未完成的会话
    _restoreFromLocal();
    // 监听 APP 生命周期
    ref.listen(appLifecycleProvider, (_, lifecycle) {
      if (lifecycle == AppLifecycleState.paused) {
        _onAppBackground();
      } else if (lifecycle == AppLifecycleState.resumed) {
        _onAppForeground();
      }
    });
    return const TimerState();
  }

  /// 开始新会话
  Future<void> startSession({
    required TimerMode mode,
    String? presetId,
    String? subjectId,
    String? chapterId,
    String? taskId,
    String? taskType,
    int? customFocusMinutes,
    int? customShortBreakMinutes,
    int? customLongBreakMinutes,
    int? customCyclesBeforeLongBreak,
  }) async {
    // 解析预设参数
    final preset = _resolvePreset(
      presetId: presetId,
      customFocusMinutes: customFocusMinutes,
      customShortBreakMinutes: customShortBreakMinutes,
      customLongBreakMinutes: customLongBreakMinutes,
      customCyclesBeforeLongBreak: customCyclesBeforeLongBreak,
    );

    // 乐观创建本地会话
    final sessionId = const Uuid().v7();
    final session = ActiveSession(
      id: sessionId,
      userId: ref.read(currentUserProvider).id,
      startTime: DateTime.now(),
      mode: mode,
      subjectId: subjectId,
      chapterId: chapterId,
      taskId: taskId,
      taskType: taskType,
      focusDurationSeconds: preset.focusMinutes * 60,
      shortBreakSeconds: preset.shortBreakMinutes * 60,
      longBreakSeconds: preset.longBreakMinutes * 60,
      cyclesBeforeLongBreak: preset.cyclesBeforeLongBreak,
      autoStartBreak: preset.autoStartBreak,
      autoStartNextCycle: preset.autoStartNextCycle,
    );

    // 创建第一个周期
    final cycle = PomodoroCycle(
      id: const Uuid().v7(),
      sessionId: sessionId,
      cycleIndex: 1,
      focusDurationSeconds: session.focusDurationSeconds,
      focusStartTime: DateTime.now(),
      focusStatus: PeriodStatus.active,
      breakDurationSeconds: session.shortBreakSeconds,
      breakType: BreakType.shortBreak,
    );

    _monotonicTimer = MonotonicTimer()..start();

    state = state.copyWith(
      activeSession: session,
      currentCycle: cycle,
      phase: TimerPhase.focus,
      remainingSeconds: session.focusDurationSeconds,
      totalPhaseSeconds: session.focusDurationSeconds,
      isRunning: true,
      syncStatus: SyncStatus.pending,
    );

    // 持久化到本地
    await _persistToLocal();
    // 异步同步到服务端
    _syncCreateSession(session);
    // 启动 UI 更新定时器
    _startUITimer();
    // 保持屏幕常亮
    _keepScreenAwake(true);
    // 触发埋点
    ref.read(analyticsProvider).logEvent('timer_session_start', params: {
      'mode': mode.name,
      'subject_id': subjectId,
      'focus_minutes': preset.focusMinutes,
    });
  }

  /// 暂停
  void pause() {
    if (!state.isRunning || state.phase == TimerPhase.idle) return;
    _monotonicTimer.pause();
    _uiUpdateTimer?.cancel();
    state = state.copyWith(isRunning: false);
    _recordInterruption(InterruptionReason.userPaused, InterruptionSource.userManual);
  }

  /// 恢复
  void resume() {
    if (state.isRunning || state.phase == TimerPhase.idle) return;
    _monotonicTimer.resume();
    _startUITimer();
    state = state.copyWith(isRunning: true);
    _closeCurrentInterruption();
  }

  /// 跳过当前阶段（跳过休息/跳过专注）
  void skipPhase() {
    if (state.phase == TimerPhase.focus) {
      // 跳过专注 → 直接进入休息
      _completeFocusPeriod(skipped: true);
    } else if (state.phase == TimerPhase.shortBreak || state.phase == TimerPhase.longBreak) {
      // 跳过休息 → 进入下一轮
      _completeBreakPeriod(skipped: true);
    }
  }

  /// 结束会话
  Future<void> endSession({required EndReason reason}) async {
    _uiUpdateTimer?.cancel();
    _monotonicTimer.stop();
    _keepScreenAwake(false);

    final session = state.activeSession!;
    final focusSeconds = state.totalFocusSeconds;

    // 计算专注得分
    final focusScore = _calculateFocusScore(session, focusSeconds);

    // 判断完成还是放弃
    final status = focusSeconds < session.focusDurationSeconds * 0.5
        ? SessionStatus.abandoned
        : SessionStatus.completed;

    // 更新本地状态
    state = state.copyWith(
      phase: TimerPhase.idle,
      isRunning: false,
      activeSession: null,
      currentCycle: null,
    );

    // 清除本地持久化
    await _clearLocalSession();

    // 同步到服务端
    await _syncEndSession(
      session: session,
      status: status,
      focusScore: focusScore,
      reason: reason,
    );

    // 触发跨模块事件
    ref.read(eventBusProvider).fire(TimerSessionEndedEvent(
      sessionId: session.id,
      focusSeconds: focusSeconds,
      focusScore: focusScore,
      subjectId: session.subjectId,
      status: status,
    ));

    // 停止白噪音
    ref.read(ambientSoundProvider.notifier).stop();
  }

  // ============================================================
  // 内部方法
  // ============================================================

  /// 专注阶段倒计时结束
  void _onFocusPhaseComplete() {
    _completeFocusPeriod(skipped: false);
  }

  /// 休息阶段倒计时结束
  void _onBreakPhaseComplete() {
    _completeBreakPeriod(skipped: false);
  }

  void _completeFocusPeriod({required bool skipped}) {
    final cycle = state.currentCycle!;
    final session = state.activeSession!;

    // 更新周期专注阶段
    final updatedCycle = cycle.copyWith(
      focusEndTime: DateTime.now(),
      focusStatus: skipped ? PeriodStatus.skipped : PeriodStatus.completed,
      actualFocusSeconds: skipped ? _monotonicTimer.elapsedSeconds : cycle.focusDurationSeconds,
    );

    // 更新会话统计
    final newCompletedCycles = skipped ? session.completedCycles : session.completedCycles + 1;
    final newFocusSeconds = session.totalFocusSeconds +
        (skipped ? 0 : updatedCycle.actualFocusSeconds);

    final updatedSession = session.copyWith(
      completedCycles: newCompletedCycles,
      totalFocusSeconds: newFocusSeconds,
    );

    // 缓存完成的周期
    state = state.copyWith(
      pendingCycles: [...state.pendingCycles, updatedCycle],
      activeSession: updatedSession,
    );

    // 发送本地通知（专注结束）
    ref.read(timerNotifProvider).showFocusEndNotification(
      cycleIndex: cycle.cycleIndex,
    );

    // 判断是否进入休息
    final isLastCycle = cycle.cycleIndex >= session.cyclesBeforeLongBreak;
    // 这里简化：根据已完成周期判断下一阶段
    final shouldTakeLongBreak = newCompletedCycles > 0 &&
        newCompletedCycles % session.cyclesBeforeLongBreak == 0;

    if (session.autoStartBreak) {
      _startBreakPhase(
        isLongBreak: shouldTakeLongBreak,
        cycle: updatedCycle,
        session: updatedSession,
      );
    } else {
      // 等待用户手动开始休息
      state = state.copyWith(
        phase: shouldTakeLongBreak ? TimerPhase.longBreak : TimerPhase.shortBreak,
        isRunning: false,
        remainingSeconds: shouldTakeLongBreak
            ? session.longBreakSeconds
            : session.shortBreakSeconds,
        totalPhaseSeconds: shouldTakeLongBreak
            ? session.longBreakSeconds
            : session.shortBreakSeconds,
      );
    }

    _persistToLocal();
  }

  void _startBreakPhase({
    required bool isLongBreak,
    required PomodoroCycle cycle,
    required ActiveSession session,
  }) {
    final breakSeconds = isLongBreak ? session.longBreakSeconds : session.shortBreakSeconds;
    final updatedCycle = cycle.copyWith(
      breakDurationSeconds: breakSeconds,
      breakType: isLongBreak ? BreakType.longBreak : BreakType.shortBreak,
      breakStartTime: DateTime.now(),
      breakStatus: PeriodStatus.active,
    );

    _monotonicTimer = MonotonicTimer()..start();

    state = state.copyWith(
      currentCycle: updatedCycle,
      phase: isLongBreak ? TimerPhase.longBreak : TimerPhase.shortBreak,
      remainingSeconds: breakSeconds,
      totalPhaseSeconds: breakSeconds,
      isRunning: true,
    );

    _startUITimer();
    // 休息时不需要保持屏幕常亮
    _keepScreenAwake(false);
  }

  void _completeBreakPeriod({required bool skipped}) {
    final cycle = state.currentCycle!;
    final session = state.activeSession!;

    final updatedCycle = cycle.copyWith(
      breakEndTime: DateTime.now(),
      breakStatus: skipped ? PeriodStatus.skipped : PeriodStatus.completed,
    );

    final updatedSession = session.copyWith(
      totalBreakSeconds: session.totalBreakSeconds +
          (skipped ? 0 : cycle.breakDurationSeconds),
    );

    state = state.copyWith(
      pendingCycles: [...state.pendingCycles, updatedCycle],
      activeSession: updatedSession,
    );

    // 发送本地通知（休息结束）
    ref.read(timerNotifProvider).showBreakEndNotification();

    // 开始下一轮专注
    final nextCycleIndex = cycle.cycleIndex + 1;
    final nextCycle = PomodoroCycle(
      id: const Uuid().v7(),
      sessionId: session.id,
      cycleIndex: nextCycleIndex,
      focusDurationSeconds: session.focusDurationSeconds,
      focusStartTime: DateTime.now(),
      focusStatus: PeriodStatus.active,
      breakDurationSeconds: session.shortBreakSeconds,
      breakType: BreakType.shortBreak,
    );

    _monotonicTimer = MonotonicTimer()..start();

    state = state.copyWith(
      currentCycle: nextCycle,
      phase: TimerPhase.focus,
      remainingSeconds: session.focusDurationSeconds,
      totalPhaseSeconds: session.focusDurationSeconds,
      isRunning: true,
    );

    _startUITimer();
    _keepScreenAwake(true);
    _persistToLocal();
  }

  /// UI 定时器：每 200ms 刷新一次倒计时
  void _startUITimer() {
    _uiUpdateTimer?.cancel();
    _uiUpdateTimer = Timer.periodic(const Duration(milliseconds: 200), (_) {
      final elapsed = _monotonicTimer.elapsedSeconds;
      final total = state.totalPhaseSeconds;
      final remaining = total - elapsed;
      if (remaining <= 0) {
        _uiUpdateTimer?.cancel();
        state = state.copyWith(remainingSeconds: 0);
        // 倒计时结束回调
        if (state.phase == TimerPhase.focus) {
          _onFocusPhaseComplete();
        } else {
          _onBreakPhaseComplete();
        }
      } else {
        state = state.copyWith(remainingSeconds: remaining);
      }
    });
  }

  void _recordInterruption(InterruptionReason reason, InterruptionSource source) {
    final session = state.activeSession;
    if (session == null) return;
    final cycle = state.currentCycle;
    if (cycle == null) return;

    state = state.copyWith(
      currentInterruption: Interruption(
        id: const Uuid().v7(),
        sessionId: session.id,
        cycleId: cycle.id,
        startTime: DateTime.now(),
        reason: reason,
        source: source,
      ),
    );
  }

  void _closeCurrentInterruption() {
    final interruption = state.currentInterruption;
    if (interruption == null) return;

    final closed = interruption.copyWith(
      endTime: DateTime.now(),
      durationSeconds: DateTime.now().difference(interruption.startTime).inSeconds,
    );

    state = state.copyWith(
      currentInterruption: null,
      pendingInterruptions: [...state.pendingInterruptions, closed],
      activeSession: state.activeSession?.copyWith(
        interruptionCount: state.activeSession!.interruptionCount + 1,
        totalInterruptionSeconds: state.activeSession!.totalInterruptionSeconds +
            closed.durationSeconds,
      ),
    );
  }

  void _onAppBackground() {
    if (!state.isRunning) return;
    _uiUpdateTimer?.cancel();
    _recordInterruption(
      InterruptionReason.appBackground,
      InterruptionSource.clientAuto,
    );
    state = state.copyWith(
      backgroundTimestamp: DateTime.now().millisecondsSinceEpoch,
    );
    // 发送离开提醒通知
    Future.delayed(const Duration(seconds: 30), () {
      if (state.backgroundTimestamp != null && state.activeSession != null) {
        ref.read(timerNotifProvider).showAwayReminderNotification();
      }
    });
  }

  void _onAppForeground() {
    final bgTs = state.backgroundTimestamp;
    if (bgTs == null || state.activeSession == null) return;

    final awaySeconds = DateTime.now().millisecondsSinceEpoch - bgTs;
    final awaySecondsRounded = (awaySeconds / 1000).round();
    final focusDuration = state.currentCycle?.focusDurationSeconds ?? 0;

    _closeCurrentInterruption();
    state = state.copyWith(backgroundTimestamp: null);

    if (awaySecondsRounded > focusDuration * 0.3) {
      // 离开太久，弹中断对话框
      state = state.copyWith(isRunning: false);
      _showInterruptionDialog(awaySecondsRounded);
    } else {
      // 短暂离开，恢复
      _startUITimer();
      state = state.copyWith(isRunning: true);
    }
  }

  int _calculateFocusScore(ActiveSession session, int focusSeconds) {
    int baseScore = (session.completedCycles * 30).clamp(0, 100);
    int interruptionPenalty = (session.interruptionCount * 5).clamp(0, 30);
    if (focusSeconds > 0) {
      final ratio = session.totalInterruptionSeconds /
          (focusSeconds + session.totalInterruptionSeconds);
      if (ratio > 0.2) interruptionPenalty += 10;
    }
    if (session.interruptionCount > 0 &&
        session.totalInterruptionSeconds / session.interruptionCount > 120) {
      interruptionPenalty += 5;
    }
    return (baseScore - interruptionPenalty).clamp(0, 100);
  }

  void _keepScreenAwake(bool enable) {
    ref.read(platformChannelProvider).setScreenAwake(enable);
  }
}

enum EndReason { completed, stoppedEarly, timeUp }
```

```dart
@HiveType(typeId: 41)

class ActiveSessionHive with _$ActiveSessionHive {

  @HiveField(4) final String? subjectId;
```

### 3.2 恢复策略

```dart
// lib/features/timer/local/session_recovery.dart

/// APP 启动时恢复未完成的计时器会话
class SessionRecovery {
  final HiveInterface _hive;

  /// 检查并恢复本地存储的活跃会话
  /// 返回 null 表示无待恢复会话
  Future<RecoveryResult?> recover() async {
    final box = await _hive.openBox('timer_active_session');
    final data = box.get('active');
    if (data == null) return null;

    final session = ActiveSessionHive.fromJson(data);
    final cycleBox = await _hive.openBox('timer_current_cycle');
    final cycleData = cycleBox.get('current');

    // 计算已经过去的时间
    final elapsedSincePersist = DateTime.now().millisecondsSinceEpoch -
        session.monotonicStartTimestamp;
    final totalElapsed = session.monotonicElapsedSeconds +
        (elapsedSincePersist ~/ 1000);

    // 判断是否已经超时太久（超过 1 小时视为放弃）
    if (totalElapsed > 3600) {
      await _clearLocal();
      return RecoveryResult.abandoned(
        sessionId: session.id,
        elapsedSeconds: totalElapsed,
      );
    }

    return RecoveryResult.active(
      session: session,
      cycle: cycleData != null ? PomodoroCycleHive.fromJson(cycleData) : null,
      elapsedSeconds: totalElapsed,
    );
  }

  Future<void> _clearLocal() async {
    final box = await _hive.openBox('timer_active_session');
    await box.clear();
    final cycleBox = await _hive.openBox('timer_current_cycle');
    await cycleBox.clear();
  }
}

@freezed
class RecoveryResult with _$RecoveryResult {
  const factory RecoveryResult.active({
    required ActiveSessionHive session,
    required PomodoroCycleHive? cycle,
    required int elapsedSeconds,
  }) = ActiveRecovery;
  const factory RecoveryResult.abandoned({
    required String sessionId,
    required int elapsedSeconds,
  }) = AbandonedRecovery;
}
```

```dart
/// 学习吉祥物动画
/// 根据专注进度和阶段变化表情/动作
/// 
/// 状态矩阵：
/// ├── 专注开始 (>75%)  → 精神饱满 😊
/// ├── 专注中 (50-75%)  → 认真思考 🤔
/// ├── 专注中 (25-50%)  → 略显疲惫 😅
/// ├── 专注快结束 (<25%) → 鼓励加油 💪
/// ├── 休息中            → 开心放松 🎉
/// └── 暂停中            → 等待疑惑 😶
class AnimatedMascot extends StatefulWidget {
  final double progress;   // 1.0 → 0.0
  final TimerPhase phase;
  final double size;

  @override
  State<AnimatedMascot> createState() => _AnimatedMascotState();

class _AnimatedMascotState extends State<AnimatedMascot>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _bounceAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1500),
    )..repeat(reverse: true);
    _bounceAnimation = Tween<double>(begin: -4, end: 4).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
  }

  @override
  Widget build(BuildContext context) {
    final emoji = _selectEmoji(widget.progress, widget.phase);
    return AnimatedBuilder(
      animation: _bounceAnimation,
      builder: (_, __) => Transform.translate(
        offset: Offset(0, _bounceAnimation.value),
        child: Text(emoji, style: TextStyle(fontSize: widget.size)),
      ),
    );
  }

  String _selectEmoji(double progress, TimerPhase phase) {
    if (phase != TimerPhase.focus) return '🎉';
    if (progress > 0.75) return '😊';
    if (progress > 0.50) return '🤔';
    if (progress > 0.25) return '😅';
    return '💪';
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }
}
```

```dart
    if (ratio >= 0.3) return const Color(0xFFFFC107); // 少量: 黄色
```

### 5.5 AmbientSoundPlayer 封装

```dart
// lib/features/timer/services/ambient_sound_player.dart

/// 白噪音播放器
/// 封装 audioplayers 包，支持多音轨混合、渐入渐出、循环播放
class AmbientSoundPlayer {
  final AudioPlayer _player = AudioPlayer();
  String? _currentSoundId;
  bool _isPlaying = false;

  /// 播放指定白噪音
  Future<void> play(String soundId, {double volume = 0.5}) async {
    if (_currentSoundId == soundId && _isPlaying) return;
    await stop();
    _currentSoundId = soundId;
    await _player.setAsset('assets/ambient/$soundId.mp3');
    await _player.setLoopMode(LoopMode.all);
    await _player.setVolume(0.0);
    await _player.play();
    // 渐入 2 秒
    await _fadeIn(duration: const Duration(seconds: 2), targetVolume: volume);
    _isPlaying = true;
  }

  /// 停止播放（渐出）
  Future<void> stop() async {
    if (!_isPlaying) return;
    await _fadeOut(duration: const Duration(seconds: 1));
    await _player.stop();
    _currentSoundId = null;
    _isPlaying = false;
  }

  /// 设置音量
  Future<void> setVolume(double volume) async {
    await _player.setVolume(volume.clamp(0.0, 1.0));
  }

  Future<void> _fadeIn({
    required Duration duration,
    required double targetVolume,
  }) async {
    const steps = 20;
    final stepDuration = duration.milliseconds ~/ steps;
    for (int i = 0; i <= steps; i++) {
      await _player.setVolume(targetVolume * (i / steps));
      await Future.delayed(Duration(milliseconds: stepDuration));
    }
  }

  Future<void> _fadeOut({required Duration duration}) async {
    const steps = 10;
    final stepDuration = duration.milliseconds ~/ steps;
    final currentVolume = await _player.volume;
    for (int i = steps; i >= 0; i--) {
      await _player.setVolume(currentVolume * (i / steps));
      await Future.delayed(Duration(milliseconds: stepDuration));
    }
  }
}
```

## 11. 降级策略

### 11.1 降级矩阵

| 降级场景 | 处理方式 | 用户感知 |
|---------|---------|---------|
| 网络不可用 | 100% 本地运行，数据缓存在 pendingQueue | 无感知（计时器完全离线可用） |
| 服务端 5xx | 重试 3 次，之后转离线模式 | 无感知 |
| 白噪音资源未下载 | 使用系统默认铃声或不播放 | 提示"背景音不可用" |
| Hive 本地存储损坏 | 清除并从服务端恢复最近数据 | 可能丢失当前进行中会话 |
| 设备性能差（低端机） | 降低动画帧率，关闭粒子效果 | 略有视觉降级 |
| 后台被系统杀死 | 下次启动恢复会话，提示"学习被打断" | 弹恢复对话框 |
| 屏幕常亮获取失败 | 正常计时但不保持常亮 | 可能自动息屏 |

### 11.2 安全模式

```
当 APP 连续崩溃 > 3 次时进入安全模式：
1. 禁用所有动画（_AnimatedBackground / Confetti / Mascot）
2. 禁用白噪音
3. 使用最简单的 Text 显示倒计时
4. 计时核心逻辑不受影响（纯 Dart，无平台依赖）
```
