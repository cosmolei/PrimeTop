## 1. 概述

### 1.1 模块定位

考试模拟与真题练习是面向初中、高中学生的核心备考功能模块。区别于日常练习测评，本模块强调：

- **仿真考试体验**：严格模拟真实考试的限时、题型分布、评分规则
- **真题权威性**：收录历年中考、高考、期末真题及名校模拟卷
- **备考闭环**：从"选择试卷 → 限时答题 → 交卷 → 成绩分析 → 薄弱点复习"形成完整链路
- **分级适配**：根据用户年级、地区自动匹配相应试卷

### 1.2 核心用户场景

| 场景 | 用户画像 | 典型路径 |
|------|---------|---------|
| 周末自测 | 高二学生，期中考前2周 | 首页推荐 → 选择近期期中模拟卷 → 限时90分钟答题 → 查看成绩报告 → 薄弱知识点复习 |
| 真题演练 | 初三学生，中考冲刺 | 考试入口 → 选择本省去年中考真题 → 全真模拟 → AI批改 → 详细解析 |
| 专项突破 | 高一学生，数学函数薄弱 | 考试入口 → 智能组卷(函数专项) → 30分钟限时 → 针对性提升建议 |
| 碎片练习 | 初二学生，课间10分钟 | 考试入口 → 快速模式(10题) → 即时反馈 → 错题自动收录 |

### 1.3 设计原则

1. **沉浸优先**：答题界面零干扰，顶部仅显示计时与进度
2. **渐进压力**：考前倒计时、剩余5分钟预警、自动交卷等机制模拟真实压力
3. **快速恢复**：支持断点续答，中断后可回到上次位置
4. **结果导向**：交卷后提供多维度的成绩分析与提升建议

```dart
  final String examId;
  final String attemptId;
  final ExamPaperInfo paperInfo;
  final List<ExamQuestion> questions;       // 题目列表
  final Map<int, UserAnswer> answers;       // 题号 → 用户答案
  final Set<int> markedQuestions;           // 标记的题目编号

// 题目数据
class ExamQuestion {
  final int questionNo;                    // 题号(1-based)

  final List<String>? options;             // 选项(选择题)

  final String? imageUrl;                  // 附图

class UserAnswer {
```

```dart
class ExamTimerController {
  Timer? _timer;
  int _remainingSeconds;
  final void Function(int) onTick;
  final void Function() onFiveMinuteWarning;
  final void Function() onTimeUp;

  void start() {
    _timer = Timer.periodic(Duration(seconds: 1), (_) {
      _remainingSeconds--;
      onTick(_remainingSeconds);

      // 剩余5分钟预警
      if (_remainingSeconds == 300) {
        onFiveMinuteWarning();
      }

      // 时间到自动交卷
      if (_remainingSeconds <= 0) {
        _timer?.cancel();
        onTimeUp();
      }
    });
  }

  // 剩余时间格式化
  String formatTime(int seconds) {
    final h = seconds ~/ 3600;
    final m = (seconds % 3600) ~/ 60;
    final s = seconds % 60;
    if (h > 0) return '${h.toString().padLeft(2, '0')}:${m.toString().padLeft(2, '0')}:${s.toString().padLeft(2, '0')}';
    return '${m.toString().padLeft(2, '0')}:${s.toString().padLeft(2, '0')}';
  }
}
```

```dart
          isMarked: state.answers[questionNo]?.isMarked ?? false,

        ...state.answers,
        questionNo: UserAnswer(
          questionNo: questionNo,
          essayAnswer: answer,
          answeredAt: DateTime.now(),
          isMarked: state.answers[questionNo]?.isMarked ?? false,
        ),
      },
    );
    _debouncedSave();
  }

  // 标记题目
  void toggleMark(int questionNo) {
    final newMarked = Set<int>.from(state.markedQuestions);
    if (newMarked.contains(questionNo)) {
      newMarked.remove(questionNo);
    } else {
      newMarked.add(questionNo);
    }
    state = state.copyWith(markedQuestions: newMarked);
    _debouncedSave();
  }

  // 切换题目
  void goToQuestion(int index) {
    state = state.copyWith(currentIndex: index);
  }

  // 上一题/下一题
  void previousQuestion() {
    if (state.currentIndex > 0) {
      state = state.copyWith(currentIndex: state.currentIndex - 1);
    }
  }

  void nextQuestion() {
    if (state.currentIndex < state.questions.length - 1) {
      state = state.copyWith(currentIndex: state.currentIndex + 1);
    }
  }

  // 更新计时器
  void updateTimer(int remainingSeconds) {
    state = state.copyWith(
      timerState: state.timerState.copyWith(remainingSeconds: remainingSeconds),
    );
  }

  // 防抖保存
  Timer? _saveDebounce;
  void _debouncedSave() {
    _saveDebounce?.cancel();
    _saveDebounce = Timer(Duration(seconds: 2), () {
      _saveProgress();
    });
  }

  Future<void> _saveProgress() async {
    // 1. 本地保存
    await ExamAnswerPersistence().saveProgress(state);
    // 2. 服务端心跳上报
    await _examApi.saveProgress(
      attemptId: state.attemptId,
      answers: state.answers,
      markedQuestions: state.markedQuestions.toList(),
      currentIndex: state.currentIndex,
      remainingSeconds: state.timerState.remainingSeconds,
    );
  }
}
```

```dart
class ExamSubmittingController {
  // 交卷后轮询批改状态
  Stream<ExamGradingStatus> pollGradingStatus(String attemptId) async* {
    while (true) {
      final status = await _examApi.getGradingStatus(attemptId);
      yield status;

      if (status.status == 'completed') {
        break;
      }

      // 根据进度动态调整轮询间隔
      final delay = status.gradedQuestions / status.totalQuestions > 0.8
          ? Duration(seconds: 1)   // 接近完成，快速轮询
          : Duration(seconds: 3);  // 正常轮询
      await Future.delayed(delay);
    }
  }
}
```

### 5.2 答题页生命周期

```dart
// 答题页生命周期管理
class ExamAnswerScreen extends ConsumerStatefulWidget {
  // ...
}

class _ExamAnswerScreenState extends ConsumerState<ExamAnswerScreen>
    with WidgetsBindingObserver {

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _initExamSession();
  }

  Future<void> _initExamSession() async {
    // 1. 检查本地缓存(断点续答)
    final cached = await ExamAnswerPersistence()
        .loadProgress(widget.examId, widget.attemptId);

    if (cached != null) {
      // 恢复断点
      ref.read(examAnswerProvider(widget.examId, widget.attemptId).notifier)
          .restoreFromCache(cached);
    } else {
      // 开始新答题
      final session = await _examApi.startAttempt(widget.examId);
      ref.read(examAnswerProvider(widget.examId, widget.attemptId).notifier)
          .initSession(session);
    }

    // 启动计时器
    _startTimer();

    // 启动自动保存定时器
    _startAutoSave();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    switch (state) {
      case AppLifecycleState.paused:
        // 切后台：立即保存进度
        _saveProgressImmediately();
        // 暂停计时器(如果规则不允许后台计时)
        _pauseTimer();
        break;
      case AppLifecycleState.resumed:
        // 回前台：恢复计时器
        _resumeTimer();
        break;
      default:
        break;
    }
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    _timer?.cancel();
    _autoSaveTimer?.cancel();
    super.dispose();
  }

  // 退出确认（物理返回键 / 退出按钮）
  Future<bool> _onWillPop() async {
    final result = await showDialog<ExamExitAction>(
      context: context,
      builder: (ctx) => ExamExitConfirmDialog(
        answeredCount: state.answers.length,
        totalCount: state.questions.length,
      ),
    );

    switch (result) {
      case ExamExitAction.saveAndExit:
        await _saveProgressImmediately();
        return true;
      case ExamExitAction.submitAndExit:
        await _submitExam();
        return true;
      case ExamExitAction.cancel:
      default:
        return false;
    }
  }
}
```

## 6. 错误处理

### 6.1 网络异常处理

```dart
class ExamErrorHandler {
  // 答题过程中网络断开
  static void handleNetworkError(BuildContext context, ExamNetworkError error) {
    switch (error.type) {
      case ExamNetworkErrorType.saveFailed:
        // 保存失败：本地已保存，显示提示
        showSnackBar('网络异常，答案已保存到本地，恢复网络后自动同步');
        break;

      case ExamNetworkErrorType.submitFailed:
        // 交卷失败：重试对话框
        showDialog(
          context: context,
          builder: (ctx) => AlertDialog(
            title: Text('交卷失败'),
            content: Text('网络异常，请检查网络后重试。你的答案已保存在本地。'),
            actions: [
              TextButton(onPressed: () => Navigator.pop(ctx), child: Text('取消')),
              TextButton(onPressed: () => _retrySubmit(), child: Text('重试')),
            ],
          ),
        );
        break;

      case ExamNetworkErrorType.loadFailed:
        // 加载失败：重试或离线
        showDialog(
          context: context,
          builder: (ctx) => AlertDialog(
            title: Text('加载失败'),
            content: Text('无法连接服务器，请检查网络。'),
            actions: [
              TextButton(onPressed: () => Navigator.pop(ctx), child: Text('返回')),
              TextButton(onPressed: () => _retry(), child: Text('重试')),
            ],
          ),
        );
        break;
    }
  }
}
```

### 6.2 异常场景处理

| 异常场景 | 处理策略 |
|---------|---------|
| 答题中APP崩溃 | 本地缓存恢复，从断点续答 |
| 答题中切后台超时 | 规则允许暂停：累计后台时长；不允许：继续计时 |
| 交卷时服务端超时 | 本地标记为"待提交"，后台自动重试 |
| 批改服务异常 | 选择题/填空题立即出分，解答题标注"AI批改中"稍后通知 |
| 真题资源已下架 | 列表页标记"已下架"，已作答的正常展示 |
| 会员到期 | 当前进行中的考试允许完成，新考试引导续费 |

```dart
class ExamAnswerSaveStrategy {
  // 保存优先级
  // 1. 本地 Hive 缓存（即时写入）
  // 2. 服务端心跳上报（每30秒/答案变更后2秒）
  // 3. 交卷时全量同步（关键操作，必须成功）

  static const localFirstPolicy = SavePolicy(
    localSave: SaveStrategy.immediate,      // 立即本地保存
    remoteSave: SaveStrategy.debounce(2000), // 2秒防抖远程保存
    heartbeatInterval: 30000,                // 30秒心跳
    retryPolicy: RetryPolicy(
      maxRetries: 3,
      backoff: [2, 4, 8],                   // 指数退避(秒)
    ),
  );
}
```

### 7.1 题目预加载

```dart
class ExamQuestionPreloader {
  // 答题时预加载前后题目资源（图片、公式等）
  static void preloadAdjacentQuestions(
    List<ExamQuestion> questions,
    int currentIndex,
  ) {
    final preloadRange = [
      currentIndex - 1, // 前一题
      currentIndex + 1, // 后一题
      currentIndex + 2, // 后两题
    ];

    for (final index in preloadRange) {
      if (index >= 0 && index < questions.length) {
        final q = questions[index];
        // 预加载图片
        if (q.imageUrl != null) {
          precacheImage(NetworkImage(q.imageUrl!), context);
        }
        // 预渲染公式（MathJax/KaTeX缓存）
        FormulaCache.preRender(q.formulas);
      }
    }
  }
}
```
