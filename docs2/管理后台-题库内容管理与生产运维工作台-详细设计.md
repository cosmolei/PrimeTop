# 管理后台 - 题库内容管理与生产运维工作台 详细设计

## 1. 概述

### 1.1 功能定位

题库内容管理与生产运维工作台是 PrimeTop 管理后台的核心模块之一，面向内容运营团队、教研团队和质量审核团队，提供题目内容的全生命周期管理能力，包括题目创建、编辑、审核、发布、标签维护、难度标定、批量导入导出、质量监控等完整功能链路。

### 1.2 设计目标

| 目标 | 说明 |
| --- | --- |
| 统一管理入口 | 为题库管理人员提供一站式工作台，覆盖题目生产到下线的完整流程 |
| 高效生产工具 | 提供富文本编辑、公式录入、图形上传、批量操作等生产工具链 |
| 质量保障流程 | 内置多级审核机制，支持教研初审、复审、终审的分级审核管线 |
| 知识体系对齐 | 与知识点体系、课标条目、教材章节深度关联，确保内容可被精准检索和推荐 |
| 数据驱动迭代 | 提供题库质量指标看板，基于用户反馈和使用数据驱动内容持续优化 |

### 1.3 适用范围

- 内容运营人员：日常题目录入、编辑、标签维护
- 教研人员：题目质量审核、难度标定、解析校验
- 题库管理员：题库策略配置、批量操作、生命周期管理
- 平台管理员：全局数据查看、权限管理

```typescript
interface QuestionListItem {
```

#### 4.3.4 AI 辅助标注

编辑页内嵌 AI 辅助功能，帮助录入员提高效率：

| AI 功能 | 触发方式 | 输入 | 输出 |
| --- | --- | --- | --- |
| 知识点自动推荐 | 题干输入完成后点击"AI标注" | 题干文本 | 推荐知识点列表(带置信度) |
| 难度自动评估 | 同上 | 题干+答案 | 难度评级(1-5) + 评估依据 |
| 解析自动生成 | 点击"AI生成解析" | 题干+题型+学科 | 结构化解析(草稿，需人工校验) |
| 课标自动对齐 | 同"AI标注" | 题干+知识点 | 课标条目ID列表 |
| 同类题推荐 | 点击"查找同类题" | 题干+知识点 | 相似题目列表 |

```typescript
// AI辅助标注请求
POST /api/admin/questions/ai-assist
interface AiAssistRequest {
  action: 'TAG_KNOWLEDGE_POINTS' | 'EVALUATE_DIFFICULTY' | 'GENERATE_SOLUTION' | 'ALIGN_CURRICULUM' | 'FIND_SIMILAR';
  questionContent: string;       // 题干文本
  questionType: string;
  subjectCode: string;
  stageCode: string;
  gradeCode: string;
  answerContent?: string;        // 答案（生成解析时需要）
}

interface AiAssistResponse {
  code: number;
  data: {
    // TAG_KNOWLEDGE_POINTS
    knowledgePoints?: { id: string; name: string; confidence: number }[];
    // EVALUATE_DIFFICULTY
    difficulty?: { level: number; reasoning: string };
    // GENERATE_SOLUTION
    solution?: { steps: string[]; keyPoints: string[]; commonMistakes: string[] };
```

```typescript
async function batchReview(questionIds: string[], action: ReviewAction): Promise<BatchResult> {
  const BATCH_SIZE = 50;
  const results: BatchResult[] = [];

  for (let i = 0; i < questionIds.length; i += BATCH_SIZE) {
    const batch = questionIds.slice(i, i + BATCH_SIZE);
    const result = await reviewBatch(batch, action);
    results.push(result);
    // 更新进度
    updateProgress(i + batch.length, questionIds.length);
  }

  return aggregateResults(results);
}
```

```typescript
function useAutoSave(content: QuestionContent, questionId: string) {
  const [saveStatus, setSaveStatus] = useState<'SAVED' | 'SAVING' | 'ERROR'>('SAVED');

  const debouncedSave = useMemo(
    () => debounce(async (content: QuestionContent) => {
      setSaveStatus('SAVING');
      try {
        await api.saveDraft(questionId, content);
        setSaveStatus('SAVED');
      } catch {
        setSaveStatus('ERROR');
        // 保存到本地存储作为备份
        localStorage.setItem(`question_draft_${questionId}`, JSON.stringify(content));
      }
    }, 3000),
    [questionId]
  );

  useEffect(() => {
    debouncedSave(content);
  }, [content, debouncedSave]);

  return saveStatus;
}
```
