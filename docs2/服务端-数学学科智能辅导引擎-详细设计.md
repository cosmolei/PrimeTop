# 服务端-数学学科智能辅导引擎-详细设计

## 1. 概述

### 1.1 模块定位

数学学科智能辅导引擎是 PrimeTop 全学段 AI 辅助学习平台的核心学科服务之一，面向小学、初中、高中全学段学生，提供数学概念理解、问题求解、思路启发、错因诊断和能力评估的一体化辅导能力。

与其他理科学科（物理、化学、生物）的辅导引擎并列，数学引擎因其贯穿全学段、知识点密集、逻辑链路长、抽象程度跨度大等特点，在题型覆盖、解题模型库、知识图谱关联和分步推导引擎等方面具有独特设计。

### 1.2 设计目标

| 目标 | 描述 |
| --- | --- |
| 全学段覆盖 | 覆盖小学（数与代数、图形与几何、统计）、初中（代数、几何、函数、统计概率）、高中（函数、几何、代数、概率统计、微积分基础） |
| 分步推导 | 对数学问题进行严格的分步推导，展示完整的逻辑链路，而非直接给答案 |
| 启发式引导 | 优先提供思路提示和方向引导，支持渐进式揭示答案 |
| 错因诊断 | 识别学生常见错误模式，定位概念误区和计算盲区 |
| 多表征呈现 | 同一数学概念支持符号、图形、数值、文字四种表征方式的转换 |
| 适龄化表达 | 根据学段自动调整讲解深度、术语使用和示例选取 |

### 1.3 适用范围

- **服务消费方**：AI 对话引擎、拍照搜题服务、同步课堂学习服务、错题订正服务、练习测评服务、考点梳理服务
- **学段范围**：小学 1-6 年级、初中 7-9 年级、高中 10-12 年级
- **教材覆盖**：人教版、北师大版、苏教版、沪教版等主流教材的数学知识体系

```python
from enum import Enum

class MathProblemType(Enum):
```

### 3.2 解题模型注册表

```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional

@dataclass
class SolutionModel:
    """解题模型定义"""
    model_id: str                              # 模型唯一标识
    name: str                                  # 模型名称
    description: str                           # 模型描述
    applicable_types: List[MathProblemType]    # 适用题型
    grade_range: tuple[int, int]               # 适用年级范围 (1,12)
    prerequisite_knowledge: List[str]          # 前置知识点ID列表

    common_pitfalls: List[str]                 # 常见陷阱
    difficulty_level: int                      # 模型本身难度 1-5
    estimated_steps: tuple[int, int]           # 预期解题步数范围

@dataclass
class MathProblemContext:
    """数学问题上下文"""
    problem_id: Optional[str]                  # 关联题库ID
    raw_input: str                             # 原始输入（文本/LaTeX混合）
    parsed_latex: str                          # 解析后的LaTeX表达式
    problem_type: MathProblemType              # 题型分类
    grade_level: int                           # 年级
    textbook_version: str                      # 教材版本
    chapter_id: Optional[str]                  # 关联章节
    knowledge_points: List[str]                # 关联知识点ID
    conditions: List[Dict]                     # 提取的已知条件
    unknowns: List[str]                        # 求解目标
    figures: List[Dict]                        # 关联图形描述
    difficulty: int                            # 题目难度 1-5
    is_multi_step: bool                        # 是否为多步骤综合题
    student_input: Optional[str]              # 学生的作答内容
    student_grade: int                         # 学生当前年级
```

### 3.3 解题步骤数据结构

```python
@dataclass
class SolutionStep:
    """单步解题步骤"""
    step_index: int                            # 步骤序号 (从1开始)
    step_type: StepType                        # 步骤类型
    description: str                           # 文字描述
    math_expression: str                       # LaTeX 数学表达式
    explanation: str                           # 为什么这样做（理由说明）
    referenced_theorem: Optional[str]          # 使用的定理/公式ID
    referenced_knowledge: List[str]            # 关联知识点ID
    sub_steps: List['SolutionStep']            # 子步骤（用于嵌套展开）
    hint_level: int                            # 提示级别 0=完整, 1=思路, 2=方向

class StepType(Enum):
    GIVEN = "given"                # 已知条件
    ASSUMPTION = "assumption"      # 假设/设
    TRANSFORMATION = "transform"   # 变形/化简
    SUBSTITUTION = "substitute"    # 代入
    CALCULATION = "calculate"      # 计算
    THEOREM_APPLY = "theorem"      # 应用定理
    CONCLUSION = "conclusion"      # 结论
    VERIFICATION = "verify"        # 验证/检验

@dataclass
class SolutionResult:
    """完整解题结果"""
    problem_context: MathProblemContext
    steps: List[SolutionStep]                  # 完整步骤列表
    final_answer: str                          # 最终答案 (LaTeX)
    answer_format: str                         # 答案格式说明
    key_method: str                            # 核心方法总结
    alternative_methods: List[str]             # 其他解法ID
    common_mistakes: List[Dict]                # 常见错误提醒
    knowledge_chain: List[str]                 # 知识链路
    difficulty_assessment: int                 # 实际评估难度
    solving_time_estimate: int                 # 预估解题时间(秒)
```

### 3.4 错因诊断数据结构

```python
@dataclass
```

### 5.2 解题模型匹配算法

```python
class SolutionRouter:
    """解题策略路由器"""

    def __init__(self, model_registry, knowledge_service):
        self.model_registry = model_registry      # 解题模型注册表
        self.knowledge_service = knowledge_service # 知识图谱服务

    def route(self, context: MathProblemContext) -> List[SolutionModel]:
        """
        为给定问题匹配最优解题模型，返回排序后的候选列表。
        """
        candidates = []

        # Step 1: 按题型过滤
        type_matches = self.model_registry.find_by_type(
            context.problem_type,
            context.grade_level
        )

        for model in type_matches:
            score = self._compute_match_score(model, context)
            candidates.append((model, score))

        # Step 2: 按匹配分排序
        candidates.sort(key=lambda x: x[1], reverse=True)

        # Step 3: 返回 Top-3 候选
        return [model for model, _ in candidates[:3]]

    def _compute_match_score(self, model: SolutionModel,
                              context: MathProblemContext) -> float:
        """计算模型与问题的匹配度"""
        score = 0.0

        # 题型匹配权重 40%
        if context.problem_type in model.applicable_types:
            score += 40

        # 年级匹配权重 20%
        if model.grade_range[0] <= context.grade_level <= model.grade_range[1]:
            score += 20

        # 知识点重叠度 25%
        overlap = len(
            set(model.prerequisite_knowledge) & set(context.knowledge_points)
        )
        total_kp = max(len(context.knowledge_points), 1)
        score += 25 * (overlap / total_kp)

        # 难度适配 15%
        difficulty_diff = abs(model.difficulty_level - context.difficulty)
        score += 15 * max(0, 1 - difficulty_diff / 5)

        return score


    def execute(self, model: SolutionModel,
                context: MathProblemContext,
                hint_level: int = 0) -> SolutionResult:
        """
        使用选定模型执行解题推导。
        hint_level: 0=完整, 1=仅思路, 2=仅方向
        """
        # 构建 Prompt
        prompt = self._build_solve_prompt(model, context, hint_level)

        # 调用大模型
        raw_response = self.llm_adapter.generate(
            prompt=prompt,
            model=self._select_llm(context),
            temperature=0.1,     # 数学推理需要低随机性
            max_tokens=4096
        )

        # 结构化后处理
        result = self._parse_solution(raw_response, context, model)

        # 知识点关联
        result.knowledge_chain = self.knowledge_service.get_chain(
            context.knowledge_points
        )

        return result
```

## 7. 错因诊断引擎

### 7.1 诊断流程

```
学生答案 + 正确答案 + 解题过程
        │
        ├── 答案正确？──→ 是 ──→ 记录正确、更新掌握度、推荐提升题
        │
        ├── 答案错误
        │     │
        │     ├── 步骤级对比：哪一步开始出错？
        │     │     │
        │     │     ├── 符号错误？──→ 匹配 sign_error 模式库
        │     │     ├── 计算错误？──→ 匹配 calculation_error 模式库
        │     │     ├── 公式错误？──→ 匹配 formula_misapplication 模式库
        │     │     ├── 方法错误？──→ 匹配 method_selection 模式库
        │     │     └── 逻辑错误？──→ 匹配 logic_error 模式库
        │     │
        │     ├── 概念层面：错因是否指向特定知识点？
        │     │     └── 关联 knowledge_points 更新掌握度
        │     │
        │     └── 生成纠正建议 + 推荐练习
        │
        └── 未作答 ──→ 分析卡点，提供分步提示
```

```python
class MathErrorDiagnoser:
    """数学错因诊断器"""

    def __init__(self, error_pattern_repo, knowledge_service, llm_adapter):
        self.error_repo = error_pattern_repo
        self.knowledge_service = knowledge_service
        self.llm = llm_adapter

    async def diagnose(self, context: MathProblemContext,
                       student_answer: str,
                       solving_process: str = None) -> Dict:
        """诊断学生答案的错误原因"""

        # Step 1: 直接判断是否正确
        is_correct = self._check_answer(student_answer, context)

        if is_correct:
            return await self._handle_correct(context)

        # Step 2: AI 驱动的深度诊断
        diagnosis = await self._ai_diagnose(
            context, student_answer, solving_process
        )

        # Step 3: 匹配已知错误模式
        patterns = self._match_error_patterns(diagnosis)

        # Step 4: 定位知识盲区
        knowledge_gaps = self._identify_knowledge_gaps(
            patterns, context.knowledge_points
        )

        # Step 5: 生成纠正方案
        remediation = self._generate_remediation(
            patterns, knowledge_gaps, context
        )

        # Step 6: 记录到学生档案
        await self._update_student_profile(
            context.student_id, patterns, knowledge_gaps
        )

        return {
            "is_correct": False,
            "diagnosis": diagnosis,
            "error_patterns": patterns,
            "knowledge_gaps": knowledge_gaps,
            "remediation": remediation
        }

    async def _ai_diagnose(self, context, student_answer, solving_process):

{{
  "error_category": "concept_misunderstanding|calculation_error|sign_error|formula_misapplication|logic_error|incomplete_discussion|range_error|method_selection|carelessness",
  "error_step": "出错的步骤编号（如能识别）",

  "specific_error": "具体错在哪里",

1. 优先判断是否存在概念性错误，而非简单的计算失误

        response = await self.llm.generate(
```

### 8.1 四种表征及转换

```python
class RepresentationConverter:
    """数学概念多表征转换器"""

    REPRESENTATIONS = {
        "symbolic":  "符号表征 - 数学公式、表达式",
        "graphical": "图形表征 - 函数图像、几何图形",
        "numerical": "数值表征 - 数据表格、坐标点",
        "verbal":    "语言表征 - 文字描述、自然语言"
    }

    async def convert(self, concept_id: str,
                      from_rep: str, to_rep: str,
                      content: Dict, grade_level: int) -> Dict:
        """将数学概念从一种表征转换为另一种"""

        conversion_key = f"{from_rep}_to_{to_rep}"

        converter_map = {
            "symbolic_to_graphical": self._symbolic_to_graph,
            "symbolic_to_numerical": self._symbolic_to_table,
            "symbolic_to_verbal":    self._symbolic_to_words,
            "graphical_to_symbolic": self._graph_to_symbolic,
            "numerical_to_symbolic": self._table_to_symbolic,
            "verbal_to_symbolic":    self._words_to_symbolic,
            "graphical_to_verbal":   self._graph_to_words,
            "verbal_to_graphical":   self._words_to_graph,
        }

        converter = converter_map.get(conversion_key)
        if converter:
            return await converter(content, grade_level)
        else:
            return await self._general_conversion(
                concept_id, from_rep, to_rep, content, grade_level
            )

    async def _symbolic_to_graph(self, expr: Dict, grade: int) -> Dict:
        """符号 → 图形：将函数表达式转换为绘图指令"""
        return {
            "type": "function_plot",
            "function": expr["latex"],
            "x_range": self._auto_x_range(expr, grade),
            "y_range": "auto",
            "key_points": await self._calculate_key_points(expr),
            "annotations": await self._generate_graph_annotations(expr, grade)
        }

    async def _symbolic_to_table(self, expr: Dict, grade: int) -> Dict:
        """符号 → 数值：将函数表达式转换为数据表"""
        import numpy as np

        x_vals = np.linspace(-5, 5, 11)
        # 安全求值（不使用 eval）
        y_vals = [self._safe_eval(expr["latex"], x) for x in x_vals]

        return {
            "type": "data_table",
            "headers": ["x", "y"],
            "rows": [[round(x, 2), round(y, 2)] for x, y in zip(x_vals, y_vals)],
            "pattern_note": self._detect_pattern(y_vals)
        }

    async def _symbolic_to_words(self, expr: Dict, grade: int) -> Dict:
        """符号 → 语言：将数学表达式翻译为自然语言描述"""
        prompt = f"""将以下数学表达式转换为{self._grade_desc(grade)}学生能理解的自然语言描述。

表达式：{expr["latex"]}

要求：
1. 用通俗的语言解释这个表达式表示什么
2. 给出一个生活中的例子
3. 控制在 3-4 句话

输出 JSON：{{
  "description": "自然语言描述",
  "example": "生活化例子",
  "key_insight": "一个关键理解点"
}}"""
        response = await self.llm.generate(prompt=prompt, temperature=0.2)
        return self._parse_json(response)
```

### 11.1 错误码定义

| --- | --- | --- |

| `MATH_PARSE_002` | LaTeX 语法错误 | 尝试自动修复，修复失败则降级为纯文本处理 |
| `MATH_TYPE_001` | 无法识别题型 | 降级为通用推理，不使用专项解题模型 |

| `MATH_SOLVE_002` | 推理结果格式异常 | 尝试重新生成，最多重试 2 次 |
| `MATH_VERIFY_001` | 答案验证失败（结果可能不正确） | 返回结果但附加「请核验」提示 |
| `MATH_GRAPH_001` | 图形指令生成失败 | 跳过图形，仅返回文字和 LaTeX |

```python
class MathOutputValidator:
    """数学输出质量校验器"""

    async def validate(self, solution: SolutionResult) -> Dict:
        checks = {
            "latex_valid": self._validate_latex(solution),
            "answer_format": self._validate_answer_format(solution),
            "step_completeness": self._validate_steps(solution),
            "knowledge_mapped": self._validate_knowledge(solution),
            "verification_passed": self._validate_verification(solution),
        }

        passed = sum(checks.values())
        total = len(checks)

        return {
            "quality_score": passed / total,
            "checks_detail": checks,
            "needs_review": passed / total < 0.8,
            "quality_level": self._quality_level(passed / total)

        elif score >= 0.8:
            return "B"    # 合格，可输出
        elif score >= 0.6:
            return "C"    # 需补充，触发二次生成
        else:
            return "D"    # 不合格，降级处理
```
