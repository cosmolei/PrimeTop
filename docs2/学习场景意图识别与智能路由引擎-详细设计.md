# 学习场景意图识别与智能路由引擎 - 详细设计

> **模块定位**：客户端/服务端协同模块，负责识别用户当前学习意图并将其路由至最合适的功能模块、AI Prompt 模板和业务处理管线。
> **优先级**：P0（MVP 核心链路）
> **依赖**：AI 对话引擎、Prompt 编排系统、用户画像、知识点体系、多模态输入引擎

---

## 1. 模块概述

### 1.1 背景与问题

PrimeTop 覆盖多个学习场景（AI 问答、拍题答疑、同步课堂、错题复习、作文辅导、背诵检测、考点梳理等），用户进入 APP 后的核心行为路径各异。如果没有统一的意图识别与路由机制：

1. **用户迷路**：不知道该用哪个功能，放弃使用
2. **功能孤岛**：各模块各自为政，缺少跨模块协同
3. **AI 回答偏差**：同一问题在不同入口得到不一致的讲解深度
4. **转化漏斗断裂**：用户拍题后没有引导至错题本/同类练习

### 1.2 目标

构建统一的「意图识别 → 场景判定 → 路由分发 → 上下文装配」管线：

| 目标 | 衡量标准 |
| --- | --- |
| 用户输入准确路由到对应功能 | 路由准确率 ≥ 95% |
| 意图识别延迟对用户透明 | 额外延迟 ≤ 100ms（缓存命中 ≤ 5ms） |
| 支持跨模块跳转引导 | 跨模块引导点击率 ≥ 30% |
| 分龄差异化路由 | 小学/初中/高中路由策略可配置 |

### 1.3 设计原则

1. **快速判定优先**：能通过规则判定就不调用大模型
2. **优雅降级**：识别失败时兜底到通用 AI 问答
3. **可观测**：每次路由决策留痕，支持离线分析
4. **渐进细化**：MVP 用规则+分类器，后续引入 LLM 意图理解

### 3.2 三级识别管线

系统采用三级漏斗式识别，优先使用低成本方案：

```
输入 → [L1: 本地规则] ──高置信──→ 路由执行
              │ 低置信/无法判定
              ▼
       [L2: 服务端分类器] ──高置信──→ 路由执行
              │ 低置信/歧义
              ▼
       [L3: LLM意图理解] ──→ 路由执行
              │ 仍不确定
              ▼
       [兜底: GENERAL_QA] → 通用AI问答
```

## 4. 详细设计

### 4.1 L1: 本地规则引擎（客户端）

#### 4.1.1 设计思路

在客户端执行零延迟的意图预判，覆盖以下场景：

| 规则类型 | 覆盖场景 | 示例 |
| --- | --- | --- |
| 输入模态直判 | 拍照→PHOTO_SEARCH，语音→VOICE_INPUT | 用户点击拍照按钮 |
| 关键词匹配 | 包含特定关键词直接映射 | "帮我背课文"→MEMORIZATION |
| 入口直判 | 从特定功能页进入时直接确定意图 | 错题本页点"复习"→MISTAKE_REVIEW |
| 时间规则 | 考试季/周末自动调整默认意图 | 期末季首页突出EXAM_PREP |
| 上下文推断 | 基于当前页面和最近操作序列 | 在同步课堂页提问→CONCEPT_LEARN |

#### 4.1.4 客户端预路由执行流程

```typescript
/**
 * 客户端意图预路由器
 */
export class LocalIntentRouter {
  private keywordRules: KeywordRule[] = [];
  private configVersion: string = '';

  constructor(
    private readonly userContext: UserContext,
    private readonly configService: ConfigService,
  ) {}

  /**
   * 加载/更新关键词规则
   */
  async syncRules(): Promise<void> {
    const config = await this.configService.getIntentRules();
    if (config.version !== this.configVersion) {
      // 按优先级降序排列
      this.keywordRules = config.rules
        .filter(r => r.enabled)
        .sort((a, b) => b.priority - a.priority);
      this.configVersion = config.version;
    }
  }

  /**
   * 执行本地预路由
   * @returns RoutingDecision 或 null（需要服务端判定）
   */
  resolve(input: UserInput, entryRoute?: string): RoutingDecision | null {
    const startTime = Date.now();

    // Step 1: 输入模态直判
    const modalityDecision = this.resolveByModality(input);
    if (modalityDecision) return modalityDecision;

    // Step 2: 入口上下文直判
    if (entryRoute) {
      const entryDecision = this.resolveByEntry(input, entryRoute);
      if (entryDecision) return entryDecision;
    }

    // Step 3: 关键词匹配
    if (input.text) {
      const kwDecision = this.resolveByKeywords(input.text);
      if (kwDecision && kwDecision.confidence >= 0.85) {
        return kwDecision;
      }
      // 低置信度关键词匹配结果暂存，传给服务端辅助
    }

    return null; // 需要服务端继续判定
  }

  private resolveByModality(input: UserInput): RoutingDecision | null {
    const modalityMap: Partial<Record<InputModality, LearningIntent>> = {
      [InputModality.IMAGE]:       LearningIntent.PHOTO_SEARCH,
      [InputModality.VOICE]:       LearningIntent.VOICE_INPUT,
      [InputModality.FORMULA]:     LearningIntent.FORMULA_INPUT,
      [InputModality.HANDWRITING]: LearningIntent.PHOTO_SEARCH,
    };

    const intent = modalityMap[input.modality];
    if (!intent) return null;

    const mapping = INTENT_ROUTING_MAP[intent];
    return {
      primaryIntent: intent,
      confidence: 0.99,
      modality: input.modality,
      targetModule: mapping.module,
      targetRoute: mapping.route,
      promptTemplateId: mapping.promptTemplate || undefined,
      decisionSource: 'rule',
      latencyMs: Date.now() - Date.now(), // ≈ 0
      timestamp: Date.now(),
    };
  }

  private resolveByEntry(input: UserInput, entryRoute: string): RoutingDecision | null {
    // 精确匹配或前缀匹配
    const entry = ENTRY_INTENT_MAP[entryRoute]
      ?? Object.entries(ENTRY_INTENT_MAP)
          .filter(([route]) => entryRoute.startsWith(route))
          .sort(([a], [b]) => b.length - a.length)[0]?.[1];

    if (!entry) return null;

    // 如果用户在同步课堂页输入的是作文相关，仍走作文路由
    // 关键词规则优先于入口规则
    const kwDecision = input.text ? this.resolveByKeywords(input.text) : null;
    if (kwDecision && kwDecision.confidence >= 0.85) {
      return kwDecision;
    }

    const mapping = INTENT_ROUTING_MAP[entry.defaultIntent];
    return {
      primaryIntent: entry.defaultIntent,
      ...entry.contextOverride,
      confidence: 0.90,
      modality: input.modality,
      targetModule: mapping.module,
      targetRoute: mapping.route,
      promptTemplateId: mapping.promptTemplate || undefined,
      decisionSource: 'rule',
      latencyMs: 1,
      timestamp: Date.now(),
    };
  }

  private resolveByKeywords(text: string): RoutingDecision | null {
    const normalizedText = text.toLowerCase().trim();

    for (const rule of this.keywordRules) {
      // 学段过滤
      if (rule.gradeLevel && !rule.gradeLevel.includes(this.userContext.gradeLevel)) {
        continue;
      }

      const matched = rule.keywords.some(kw => normalizedText.includes(kw));
      if (!matched) continue;

      // 计算置信度：基于关键词在文本中的位置和比重
      const confidence = this.calcKeywordConfidence(normalizedText, rule.keywords);

      const mapping = INTENT_ROUTING_MAP[rule.intent];
      return {
        primaryIntent: rule.intent,
        subIntent: rule.subIntent,
        confidence,
        modality: InputModality.TEXT,
        targetModule: mapping.module,
        targetRoute: mapping.route,
        promptTemplateId: mapping.promptTemplate,
        decisionSource: 'rule',
        latencyMs: 1,
        timestamp: Date.now(),
      };
    }

    return null;
  }

  /**
   * 关键词置信度计算
   * - 关键词出现在句首：+0.15
   * - 多个关键词命中：+0.05 每个
   * - 关键词占比高：+0.10
   */
  private calcKeywordConfidence(text: string, keywords: string[]): number {
    let score = 0.75; // 基础分

    const matchCount = keywords.filter(kw => text.includes(kw)).length;
    const matchRatio = matchCount / keywords.length;

    score += Math.min(matchRatio * 0.15, 0.15);

    // 关键词在句首
    const startsWithKw = keywords.some(kw => text.startsWith(kw));
    if (startsWithKw) score += 0.10;

    return Math.min(score, 0.98);
  }
}
```

```typescript
   * 对用户输入进行意图分类

export interface ClassifyRequest {
  text: string;                     // 用户输入文本（ASR 转写后的文本也走此路径）
  modality: InputModality;
  userContext: UserContext;
  entryRoute?: string;              // 客户端当前页面路由

    keywordMatch?: {
      intent: LearningIntent;
      confidence: number;
    };
  };
}

export interface ClassifyResponse {
  candidates: Array<{
    intent: LearningIntent;
    subIntent?: SubIntent;
    confidence: number;
  }>;
  modelVersion: string;
  latencyMs: number;
}

/**
 * FastText 分类器实现
 */
export class FastTextIntentClassifier implements IntentClassifier {
  private model: any; // FastText 模型实例
  private readonly MODEL_PATH = '/models/intent_classifier.ftz';
  private readonly CONFIDENCE_THRESHOLD = 0.75;

  async classify(request: ClassifyRequest): Promise<ClassifyResponse> {
    const startTime = Date.now();

    // 特征预处理
    const features = this.preprocess(request);

    // FastText 预测
    const predictions = this.model.predict(features, 3, this.CONFIDENCE_THRESHOLD);
    // predictions: [{label: 'homework_help', score: 0.89}, ...]

    // 子意图二次分类（仅对高置信主意图执行）
    const candidates = await Promise.all(
      predictions.map(async (pred) => {
        const subIntent = pred.score > 0.85
          ? await this.classifySubIntent(request.text, pred.label)
          : undefined;

        return {
          intent: pred.label as LearningIntent,
          subIntent,
          confidence: pred.score,
        };
      })
    );

    return {
      candidates,
      modelVersion: 'ft-v1.3',
      latencyMs: Date.now() - startTime,
    };
  }

  /**
   * 特征预处理：拼接文本特征和上下文特征
   */
  private preprocess(request: ClassifyRequest): string {
    const ctx = request.userContext;
    const parts = [
      request.text,
      `__grade__${ctx.gradeLevel}`,
      `__semester__${ctx.semester}`,
    ];

    if (request.entryRoute) {
      parts.push(`__entry__${request.entryRoute.split('/').filter(Boolean)[0]}`);
    }

    return parts.join(' ');
  }
}
```

```typescript
/**
 * 路由决策组装器
 * 整合三级识别结果 + 用户上下文 → 最终路由决策
 */
export class RoutingDecisionAssembler {
  constructor(
    private readonly intentMap: typeof INTENT_ROUTING_MAP,
    private readonly contextAssembler: ContextAssembler,
  ) {}

  assemble(
    classifyResult: ClassifyResponse,
    userContext: UserContext,
    input: UserInput,
  ): RoutingDecision {
    const topCandidate = classifyResult.candidates[0];
    const intent = topCandidate.intent;
    const mapping = this.intentMap[intent];

    // 学段过滤：如果意图不支持当前学段，降级
    if (mapping.gradeLevelFilter && !mapping.gradeLevelFilter.includes(userContext.gradeLevel)) {
      return this.buildFallbackDecision(userContext, input, 'grade_filtered');
    }

    // 置信度检查
    if (topCandidate.confidence < mapping.minConfidence) {
      return this.buildFallbackDecision(userContext, input, 'low_confidence');
    }

    // 会员权限检查
    const accessCheck = this.checkAccess(intent, userContext.membershipTier);
    if (!accessCheck.allowed) {
      return {
        ...this.buildBasicDecision(intent, mapping, input, userContext),
        targetRoute: `/paywall?feature=${intent}&returnTo=${mapping.route}`,
        suggestedNextActions: [],
        confidence: topCandidate.confidence,
        decisionSource: 'rule', // 权限路由是规则决策
      };
    }

    // 组装最终决策
    const decision: RoutingDecision = {
      primaryIntent: intent,
      subIntent: topCandidate.subIntent,
      confidence: topCandidate.confidence,
      modality: input.modality,
      targetModule: mapping.module,
      targetRoute: this.enrichRoute(mapping.route, userContext, topCandidate),
      promptTemplateId: mapping.promptTemplate || undefined,
      contextOverrides: this.buildContextOverrides(intent, userContext),
      suggestedNextActions: this.generateSuggestions(intent, userContext),
      decisionSource: classifyResult.candidates.length > 0
        ? (classifyResult.modelVersion.startsWith('ft') ? 'classifier' : 'llm')
        : 'fallback',
      latencyMs: 0, // 由调用方填充
      timestamp: Date.now(),
    };

    return decision;
  }

  /**
   * 根据意图和用户上下文生成跨模块引导建议
   */
  private generateSuggestions(
    intent: LearningIntent,
    ctx: UserContext,
  ): SuggestedAction[] {
    const suggestions: SuggestedAction[] = [];

    switch (intent) {
      case LearningIntent.HOMEWORK_HELP: {
        // 解题后推荐加入错题本
        suggestions.push({
          intent: LearningIntent.MISTAKE_REVIEW,
          label: '加入错题本',
          route: '/mistake-book/add',
          icon: 'bookmark',
          priority: 10,
        });
        // 推荐同类题
        suggestions.push({
          intent: LearningIntent.EXAM_PREP,
          label: '练一道同类题',
          route: '/exam-prep/similar',
          icon: 'dumbbell',
          priority: 8,
        });
        break;
      }

      case LearningIntent.CONCEPT_LEARN: {
        // 概念学习后推荐练习
        suggestions.push({
          intent: LearningIntent.EXAM_PREP,
          label: '做几道练习题',
          route: '/exam-prep/practice',
          icon: 'pen',
          priority: 9,
        });
        break;
      }

      case LearningIntent.MISTAKE_REVIEW: {
        // 复习错题后推荐概念回顾
        suggestions.push({
          intent: LearningIntent.CONCEPT_LEARN,
          label: '回顾相关知识点',
          route: '/sync-classroom/review',
          icon: 'lightbulb',
          priority: 8,
        });
        break;
      }

      case LearningIntent.ESSAY_HELP: {
        // 作文辅导后推荐背诵素材
        suggestions.push({
          intent: LearningIntent.MEMORIZATION,
          label: '积累好词好句',
          route: '/memorization/material?type=essay',
          icon: 'sparkles',
          priority: 7,
        });
        break;
      }
    }

    // 如果有待复习项目，始终推荐
    if (ctx.dueReviews && ctx.dueReviews.length > 0 && intent !== LearningIntent.MISTAKE_REVIEW) {
      suggestions.push({
        intent: LearningIntent.MISTAKE_REVIEW,
        label: `你有 ${ctx.dueReviews.length} 道错题待复习`,
        route: '/mistake-book/review',
        icon: 'alert-circle',
        priority: 15, // 高优先级
      });
    }

    return suggestions.sort((a, b) => b.priority - a.priority).slice(0, 3);
  }

  /**
   * 会员权限检查
   */
  private checkAccess(
    intent: LearningIntent,
    tier: string,
  ): { allowed: boolean; reason?: string } {
    // 免费用户限制
    const freeUserRestrictions: Partial<Record<LearningIntent, {
      dailyLimit: number;
      message: string;
    }>> = {
      [LearningIntent.EXAM_PREP]: {
        dailyLimit: 3,
        message: '免费用户每日限 3 次考试练习，升级会员解锁',
      },
      [LearningIntent.ESSAY_HELP]: {
        dailyLimit: 2,
        message: '免费用户每日限 2 次作文辅导，升级会员解锁',
      },
    };

    if (tier === 'free') {
      const restriction = freeUserRestrictions[intent];
      if (restriction) {
        // 实际限额检查需要查 Redis 计数器
        return { allowed: false, reason: restriction.message };
      }
    }

    return { allowed: true };
  }

  /**
   * 丰富路由参数：注入学科、章节等上下文参数
   */
  private enrichRoute(
    baseRoute: string,
    ctx: UserContext,
    candidate: { subIntent?: SubIntent },
  ): string {
    const params = new URLSearchParams();

    if (ctx.activeSessionId) {
      params.set('sessionId', ctx.activeSessionId);
    }
    if (candidate.subIntent) {
      params.set('subIntent', candidate.subIntent);
    }

    const queryString = params.toString();
    return queryString ? `${baseRoute}?${queryString}` : baseRoute;
  }
}
```

```typescript
export class ContextAssembler {
  constructor(

        const weakPoints = await this.learningRecordRepo

        return {

            errorType: m.errorType,
          })),
          weakPoints: weakPoints.slice(0, 5),

        };

          : subIntent === SubIntent.POLISH
            ? { task: 'polish', focusAreas: ['language', 'structure'] }
            : { task: 'review' };

        return {
          ...base,
          ...essayContext,
          targetWordCount: this.getTargetWordCount(userContext.gradeLevel),
          writingStyle: this.getWritingStyleGuide(userContext.gradeLevel),
        };
      }

      case LearningIntent.EXAM_PREP: {
        const examInfo = await this.getUpcomingExamInfo(userContext);

        return {
          ...base,
          examType: examInfo?.type,       // 'midterm' | 'final' | 'zhongkao' | 'gaokao'
          daysUntilExam: examInfo?.daysUntil,
          weakChapters: await this.learningRecordRepo.getWeakChapters(userContext.userId),
        };
      }

      default:
        return base;
    }
  }

  private getExplanationDepth(level: GradeLevel): 'simple' | 'moderate' | 'detailed' {
    switch (level) {
      case GradeLevel.PRESCHOOL:
      case GradeLevel.PRIMARY:
        return 'simple';
      case GradeLevel.JUNIOR:
        return 'moderate';
      case GradeLevel.SENIOR:
        return 'detailed';

    switch (level) {
      case GradeLevel.PRIMARY:   return 300;
      case GradeLevel.JUNIOR:    return 600;
      case GradeLevel.SENIOR:    return 800;
      default:                   return 300;
    }
```

```typescript
/**
 * 客户端意图路由调用入口
 * 在 AI 输入框提交时触发
 */
export async function handleUserInput(
  input: UserInput,
  navigation: NavigationService,
): Promise<void> {
  const userContext = await getUserContext();
  const entryRoute = navigation.getCurrentRoute();

  // Step 1: 本地预路由
  const localRouter = new LocalIntentRouter(userContext, configService);
  await localRouter.syncRules();

  const localDecision = localRouter.resolve(input, entryRoute);

  if (localDecision && localDecision.confidence >= 0.85) {
    // 本地路由命中，直接执行
    executeRouting(localDecision, navigation);
    return;
  }

  // Step 2: 服务端识别
  const serverDecision = await intentApi.resolve({
    input,
    context: {
      entryRoute,
      clientKeywordHint: localDecision
        ? { intent: localDecision.primaryIntent, confidence: localDecision.confidence }
        : undefined,
    },
  });

  executeRouting(serverDecision, navigation);
}

/**
 * 执行路由跳转
 */
function executeRouting(decision: RoutingDecision, navigation: NavigationService): void {
  // 1. 记录路由日志
  logRoutingDecision(decision);

  // 2. 执行页面跳转
  navigation.push(decision.targetRoute, {
    // 传递上下文参数
    intent: decision.primaryIntent,
    subIntent: decision.subIntent,
    promptTemplateId: decision.promptTemplateId,
    contextOverrides: decision.contextOverrides,
    suggestedActions: decision.suggestedNextActions,
  });

  // 3. 展示引导建议（非阻塞式 Toast/BottomSheet）
  if (decision.suggestedNextActions && decision.suggestedNextActions.length > 0) {
    showSuggestedActions(decision.suggestedNextActions);
  }
}
```

### 7.2 服务端完整管线

```python
# server/intent/router.py

class IntentRouterService:
    """意图路由服务 - 服务端主入口"""

    def __init__(self):
        self.local_rules = KeywordRuleMatcher()      # L1 (服务端也有，用于日志对比)
        self.classifier = FastTextClassifier()         # L2
        self.llm_resolver = LLMIntentResolver()        # L3
        self.assembler = RoutingDecisionAssembler()
        self.logger = RoutingLogger()

    async def resolve(self, request: IntentResolveRequest) -> RoutingDecision:
        """主路由入口"""
        start_time = time.time()

        # 获取用户上下文
        user_ctx = await self._get_user_context(request.user_id)

        # L2: 分类器
        classify_result = await self.classifier.classify(
            text=request.input.text,
            modality=request.input.modality,
            user_context=user_ctx,
            entry_route=request.context.entry_route,
            client_hint=request.context.client_keyword_hint,
        )

        # 判断是否需要 L3
        need_llm = self._should_use_llm(classify_result, request)
        if need_llm:
            llm_result = await self.llm_resolver.resolve(
                text=request.input.text,
                user_context=user_ctx,
                timeout_ms=2000,
            )
            # LLM 结果覆盖分类器结果（如果 LLM 置信度更高）
            if llm_result.candidates[0].confidence > classify_result.candidates[0].confidence:
                classify_result = llm_result

        # 组装路由决策
        decision = self.assembler.assemble(
            classify_result=classify_result,
            user_context=user_ctx,
            input=request.input,
        )

        # 填充延迟
        decision.latency_ms = int((time.time() - start_time) * 1000)

        # 异步记录日志
        asyncio.create_task(
            self.logger.log(decision, request, user_ctx)
        )

        return decision

    def _should_use_llm(self, result: ClassifyResponse, request: IntentResolveRequest) -> bool:
        """判断是否需要 LLM 兜底"""
        if not result.candidates:
            return True

        top = result.candidates[0]

        # 置信度过低
        if top.confidence < 0.6:
            return True

        # Top1 和 Top2 差距过小（歧义）
        if len(result.candidates) >= 2:
            gap = top.confidence - result.candidates[1].confidence
            if gap < 0.15:
                return True

        # 长文本复杂描述
        if len(request.input.text) > 200:
            return True

        return False
```

## 8. 错误处理

### 8.1 错误场景与策略

| 错误场景 | 检测方式 | 处理策略 |
| --- | --- | --- |
| 服务端不可用 | HTTP 超时/5xx | 客户端本地规则兜底 + GENERAL_QA |
| 分类器模型加载失败 | 模型文件校验 | 降级为纯规则匹配 |
| LLM 超时 | 2s 超时 | 使用分类器结果（即使低置信） |
| LLM 输出解析失败 | JSON 解析异常 | 降级 GENERAL_QA |
| 用户上下文获取失败 | 缓存/DB 异常 | 使用默认上下文（小学/语文/GUEST） |
| 路由目标模块不存在 | 路由表查空 | 降级 GENERAL_QA + 记录异常日志 |
| 意图不在学段支持范围 | gradeLevelFilter | 重定向到通用问答 + 提示该功能适用学段 |

### 8.2 降级策略

```typescript
/**
 * 分级降级策略
 */
export class DegradationStrategy {
  /**
   * 根据错误类型选择降级方案
   */
  static handle(error: RoutingError): RoutingDecision {
    // 记录错误指标
    metrics.increment('intent.routing.error', { type: error.type });

    switch (error.type) {
      case 'server_unavailable':
        // 完全依赖本地规则
        return this.localFallback(error.input, error.localHint);

      case 'classifier_failure':
        // 跳过分类器，尝试 LLM 或直接兜底
        return error.localHint?.confidence > 0.6
          ? this.useLocalHint(error.localHint, error.input)
          : this.generalQA(error.input);

      case 'llm_timeout':
        // 使用分类器结果（即使低置信）
        return error.classifierResult?.candidates[0]
          ? this.fromClassifyResult(error.classifierResult, error.input)
          : this.generalQA(error.input);

      case 'all_failed':
      default:
        return this.generalQA(error.input);
    }
  }

  private static generalQA(input: UserInput): RoutingDecision {
    return {
      primaryIntent: LearningIntent.GENERAL_QA,
      confidence: 0.1,
      modality: input.modality,
      targetModule: 'ai_tutor',
      targetRoute: '/ai-tutor/session?type=general',
      promptTemplateId: 'general_qa_v1',
      decisionSource: 'fallback',
      latencyMs: 0,
      timestamp: Date.now(),
    };
  }
}
```

### 9.2 缓存策略

```typescript
/**
 * 路由结果缓存（减少重复识别）
 */
export class RoutingCache {
  private readonly cache: LRUCache<string, RoutingDecision>;
  private readonly TTL_MS = 5 * 60 * 1000; // 5 分钟
  private readonly MAX_ENTRIES = 1000;

  constructor() {
    this.cache = new LRUCache({ maxSize: this.MAX_ENTRIES });
  }

  /**
   * 生成缓存键
   * 组合：用户ID + 文本哈希 + 入口路由 + 学段
   */
  private cacheKey(userId: string, text: string, entryRoute: string, gradeLevel: string): string {
    const hash = murmurhash3(text.trim().toLowerCase());
    return `${userId}:${hash}:${entryRoute}:${gradeLevel}`;
  }

  get(key: string): RoutingDecision | null {
    const entry = this.cache.get(key);
    if (!entry) return null;
    if (Date.now() - entry.timestamp > this.TTL_MS) {
      this.cache.delete(key);
      return null;
    }
    return entry;
  }
}
```
