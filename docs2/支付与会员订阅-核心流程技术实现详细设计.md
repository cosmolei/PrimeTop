# 支付与会员订阅 - 核心流程技术实现详细设计

## 1. 概述

### 1.1 设计目标
为 PrimeTop 项目提供完整的支付与会员订阅技术实现方案，确保支付流程的安全性、可靠性和一致性，实现会员权益的精确管控和灵活配置。

### 1.2 技术栈
- 后端框架：SpringBoot 2.7+
- 数据库：MySQL 8.0+
- 缓存：Redis 6.0+
- 支付渠道：微信支付、支付宝支付
- 任务调度：Spring Task
- 监控：Prometheus + Grafana

## 4. 核心业务流程实现

### 4.1 订单创建流程
```java
@Service
@Transactional
public class OrderServiceImpl implements OrderService {
    
    @Autowired
    private OrderMapper orderMapper;
    
    @Autowired
    private MembershipPlanMapper membershipPlanMapper;
    
    @Autowired
    private OrderLockService orderLockService; // 分布式锁服务
    
    @Override
    public CreateOrderResponse createOrder(CreateOrderRequest request) {
        Long userId = request.getUserId();
        
        // 1. 校验套餐状态
        MembershipPlan plan = membershipPlanMapper.selectByType(
            request.getMembershipType());
        if (plan == null || !plan.getStatus().equals("ACTIVE")) {
            throw new BusinessException("订阅套餐不可用");
        }
        
        // 2. 防止重复下单：检查用户是否有待支付订单
        String lockKey = "order:create:" + userId;
        OrderLock lock = orderLockService.acquireLock(lockKey, 5);
        try {
            Order pendingOrder = orderMapper.selectPendingOrder(userId);
            if (pendingOrder != null) {
                // 如果待支付订单未过期，直接返回
                if (pendingOrder.getExpireTime().after(new Date())) {
                    return buildResponse(pendingOrder);
                } else {
                    // 过期则取消
                    cancelOrder(pendingOrder.getOrderNo());
                }
            }
            
            // 3. 生成订单号
            String orderNo = generateOrderNo(userId);
            
            // 4. 创建订单记录
            Order order = new Order();
            order.setOrderNo(orderNo);
            order.setUserId(userId);
            order.setMembershipType(request.getMembershipType().name());
            order.setPlanId(plan.getId());
            order.setAmount(plan.getSalePrice());
            order.setPayChannel(request.getPayChannel().name());
            order.setPayMethod(request.getPayMethod() != null ? 
                request.getPayMethod().name() : "APP");
            order.setOrderStatus("PENDING");
            order.setExpireTime(calculateExpireTime()); // 15分钟后过期
            order.setClientIp(request.getClientIp());
            order.setDeviceInfo(request.getDeviceInfo());
            
            orderMapper.insert(order);
            
            // 5. 调用支付接口获取支付参数
            Map<String, Object> paymentInfo = paymentService.createPayment(
                orderNo, plan.getSalePrice(), request.getPayChannel(), 
                request.getPayMethod());
            
            // 6. 组装返回结果
            CreateOrderResponse response = new CreateOrderResponse();
            response.setOrderNo(orderNo);
            response.setAmount(plan.getSalePrice());
            response.setExpireTime(order.getExpireTime().getTime());
            response.setPaymentInfo(paymentInfo);
            
            return response;
            
        } finally {
            lock.release();
        }
    }
    
    private String generateOrderNo(Long userId) {
        // 生成规则：PT + 时间戳 + 用户ID后4位 + 随机数
        String timestamp = String.valueOf(System.currentTimeMillis());
        String userIdSuffix = String.format("%04d", userId % 10000);
        String random = String.format("%04d", new Random().nextInt(10000));
        return "PT" + timestamp + userIdSuffix + random;
    }
    
    private Date calculateExpireTime() {
        Calendar calendar = Calendar.getInstance();
        calendar.add(Calendar.MINUTE, 15); // 15分钟后过期
        return calendar.getTime();
    }
}
```

### 4.2 支付回调处理流程
```java
@Service
public class PaymentCallbackServiceImpl implements PaymentCallbackService {
    
    @Autowired
    private OrderMapper orderMapper;
    
    @Autowired
    private MembershipMapper membershipMapper;
    
    @Autowired
    private PaymentCallbackMapper paymentCallbackMapper;
    
    @Override
    public void handleWechatCallback(String body, String signature, 
            String timestamp, String nonce, String serial) {
        
        // 1. 验证签名
        if (!verifyWechatSignature(body, signature, timestamp, nonce, serial)) {
            throw new SecurityException("签名验证失败");
        }
        
        // 2. 解析回调数据
        WechatPayCallback callback = JsonUtil.fromJson(body, WechatPayCallback.class);
        String orderNo = callback.getOutTradeNo();
        
        // 3. 幂等性处理：记录回调日志
        String callbackId = saveCallbackLog(orderNo, "WECHAT", body);
        
        // 4. 查询订单
        Order order = orderMapper.selectByOrderNo(orderNo);
        if (order == null) {
            log.error("订单不存在：{}", orderNo);
            return;
        }
        
        // 5. 幂等性校验：检查订单状态
        if ("PAID".equals(order.getOrderStatus())) {
            log.info("订单已支付，无需重复处理：{}", orderNo);
            updateCallbackStatus(callbackId, "SUCCESS", "订单已支付");
            return;
        }
        
        // 6. 验证金额
        if (!callback.getAmount().equals(order.getAmount())) {
            log.error("支付金额不匹配：订单金额={}, 支付金额={}", 
                order.getAmount(), callback.getAmount());
            updateCallbackStatus(callbackId, "FAILED", "金额不匹配");
            return;
        }
        
        // 7. 更新订单状态
        updateOrderStatus(order, callback);
        
        // 8. 生效会员权益
        activateMembership(order);
        
        // 9. 更新回调状态
        updateCallbackStatus(callbackId, "SUCCESS", "处理成功");
    }
    
    private void updateOrderStatus(Order order, WechatPayCallback callback) {
        order.setOrderStatus("PAID");
        order.setThirdPartyOrderNo(callback.getTransactionId());
        order.setPaidAmount(callback.getAmount());
        order.setPaidTime(new Date(callback.getTimeEnd()));
        order.setNotifyTime(new Date());
        orderMapper.updateById(order);
    }
    
    private void activateMembership(Order order) {
        Long userId = order.getUserId();
        String membershipType = order.getMembershipType();
        
        // 1. 查询现有会员
        Membership existingMembership = membershipMapper.selectByUserId(userId);
        
        // 2. 计算会员有效期
        MembershipPlan plan = membershipPlanMapper.selectById(order.getPlanId());
        Date startTime = new Date();
        Date endTime;
        
        if (existingMembership != null && 
            existingMembership.getEndTime().after(startTime)) {
            // 延续现有会员
            startTime = existingMembership.getEndTime();
            endTime = calculateEndTime(startTime, plan.getDurationDays());
        } else {
            // 新建会员
            endTime = calculateEndTime(startTime, plan.getDurationDays());
        }
        
        // 3. 更新或创建会员记录
        Membership membership = new Membership();
        membership.setUserId(userId);
        membership.setMembershipType(membershipType);
        membership.setStartTime(startTime);
        membership.setEndTime(endTime);
        membership.setAutoRenew(false);
        
        if (existingMembership != null) {
            membership.setId(existingMembership.getId());
            membershipMapper.updateById(membership);
        } else {
            membershipMapper.insert(membership);
        }
        
        // 4. 清除用户缓存
        clearUserMembershipCache(userId);
    }
    
    private Date calculateEndTime(Date startTime, int durationDays) {
        Calendar calendar = Calendar.getInstance();
        calendar.setTime(startTime);
        calendar.add(Calendar.DAY_OF_MONTH, durationDays);
        return calendar.getTime();
    }
}
```

### 4.3 会员权益校验流程
```java
@Service
public class MembershipServiceImpl implements MembershipService {
    
    @Autowired
    private MembershipMapper membershipMapper;
    
    @Autowired
    private MembershipPlanMapper membershipPlanMapper;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private static final String MEMBERSHIP_CACHE_KEY = "membership:user:";
    private static final long CACHE_EXPIRE_SECONDS = 300; // 5分钟
    
    @Override
    public BenefitCheckResponse checkBenefit(Long userId, BenefitType benefitType) {
        // 1. 获取会员信息（先查缓存）
        Membership membership = getMembershipWithCache(userId);
        
        // 2. 校验会员状态
        boolean hasBenefit = false;
        Long remainingQuota = 0L;
        Long expireTime = 0L;
        String description = "";
        
        if (membership != null && 
            membership.getEndTime().after(new Date())) {
            
            // 获取会员套餐配置
            MembershipPlan plan = membershipPlanMapper.selectByType(
                membership.getMembershipType());
            
            if (plan != null && plan.getStatus().equals("ACTIVE")) {
                // 解析权益配置
                Map<String, Object> benefits = JsonUtil.fromJson(
                    plan.getBenefits(), Map.class);
                
                // 根据权益类型判断
                switch (benefitType) {
                    case AI_QUESTION:
                        hasBenefit = true;
                        remainingQuota = getDailyQuota(userId, "AI_QUESTION", 
                            (Integer) benefits.get("aiQuestionDailyLimit"));
                        description = "会员可无限次提问";
                        break;
                    case PHOTO_SEARCH:
                        hasBenefit = true;
                        remainingQuota = getDailyQuota(userId, "PHOTO_SEARCH", 
                            (Integer) benefits.get("photoSearchDailyLimit"));
                        description = "会员可无限次拍题";
                        break;
                    case SYNCHRONIZED_CLASSROOM:
                        hasBenefit = true;
                        description = "会员可访问全部同步课程";
                        break;
                    case MISTAKE_BOOK:
                        hasBenefit = true;
                        description = "会员可使用完整错题本功能";
                        break;
                }
                
                expireTime = membership.getEndTime().getTime();
            }
        }
        
        // 3. 组装返回结果
        BenefitCheckResponse response = new BenefitCheckResponse();
        response.setHasBenefit(hasBenefit);
        response.setRemainingQuota(remainingQuota);
        response.setExpireTime(expireTime);
        response.setDescription(description);
        
        return response;
    }
    
    private Membership getMembershipWithCache(Long userId) {
        String cacheKey = MEMBERSHIP_CACHE_KEY + userId;
        
        // 先从缓存获取
        Membership cached = (Membership) redisTemplate.opsForValue().get(cacheKey);
        if (cached != null) {
            return cached;
        }
        
        // 从数据库查询
        Membership membership = membershipMapper.selectByUserId(userId);
        if (membership == null || 
            membership.getEndTime().before(new Date())) {
            return null; // 无效会员
        }
        
        // 写入缓存
        redisTemplate.opsForValue().set(cacheKey, membership, 
            CACHE_EXPIRE_SECONDS, TimeUnit.SECONDS);
        
        return membership;
    }
    
    private Long getDailyQuota(Long userId, String quotaType, Integer dailyLimit) {
        if (dailyLimit == null || dailyLimit <= 0) {
            return Long.MAX_VALUE; // 无限制
        }
        
        // 获取今日已使用次数
        String key = "quota:" + quotaType + ":" + userId + ":" + 
            LocalDate.now().toString();
        Long usedCount = redisTemplate.opsForValue().increment(key);
        
        if (usedCount == 1) {
            // 第一次访问，设置过期时间为今日结束
            redisTemplate.expireAt(key, getEndOfDay());
        }
        
        return Math.max(0, dailyLimit - usedCount);
    }
    
    private Date getEndOfDay() {
        LocalDateTime endOfDay = LocalDate.now().atTime(23, 59, 59);
        return Date.from(endOfDay.atZone(ZoneId.systemDefault()).toInstant());
    }
    
    public void clearUserMembershipCache(Long userId) {
        String cacheKey = MEMBERSHIP_CACHE_KEY + userId;
        redisTemplate.delete(cacheKey);
    }
}
```

```java
@Override
@Transactional
public void cancelOrder(String orderNo) {
    // 1. 查询订单
    Order order = orderMapper.selectByOrderNo(orderNo);
    if (order == null) {
        throw new BusinessException("订单不存在");
    }
    
    // 2. 校验订单状态
    if (!"PENDING".equals(order.getOrderStatus())) {
        throw new BusinessException("订单状态不允许取消");
    }
    
    // 3. 更新订单状态
    order.setOrderStatus("CANCELLED");
    orderMapper.updateById(order);
    
    // 4. 调用支付渠道关闭订单（如果已创建支付单）
    try {
        paymentService.closeOrder(orderNo, order.getPayChannel());
    } catch (Exception e) {
        log.error("关闭支付订单失败：{}", orderNo, e);
        // 不影响订单取消流程
    }
}
```

## 5. 状态机与错误处理

### 5.1 订单状态流转
```java
public enum OrderStatus {
    PENDING("待支付"),
    PAID("已支付"),
    FAILED("支付失败"),
    CANCELLED("已取消"),
    REFUNDING("退款中"),
    REFUNDED("已退款");
    
    private String description;
    
    // 状态流转校验
    public boolean canTransitionTo(OrderStatus target) {
        switch (this) {
            case PENDING:
                return target == PAID || target == FAILED || target == CANCELLED;
            case PAID:
                return target == REFUNDING || target == REFUNDED;
            case REFUNDING:
                return target == REFUNDED || target == PAID;
            default:
                return false;
        }
    }
}
```

```java
@Component
public class PaymentCallbackRetryTask {
    
    @Autowired
    private PaymentCallbackMapper paymentCallbackMapper;
    
    @Autowired
    private PaymentCallbackService paymentCallbackService;
    
    @Scheduled(cron = "0 */1 * * * ?") // 每分钟执行一次
    public void retryFailedCallbacks() {
        // 查询失败的回调记录（重试次数 < 3）
        List<PaymentCallback> failedCallbacks = 
            paymentCallbackMapper.selectFailedCallbacks(3);
        
        for (PaymentCallback callback : failedCallbacks) {
            try {
                if ("WECHAT".equals(callback.getPayChannel())) {
                    // 重新处理微信回调
                    Map<String, String> callbackData = JsonUtil.fromJson(
                        callback.getCallbackData(), Map.class);
                    paymentCallbackService.handleWechatCallback(
                        callback.getCallbackData(),
                        callbackData.get("signature"),
                        callbackData.get("timestamp"),
                        callbackData.get("nonce"),
                        callbackData.get("serial")
                    );
                } else if ("ALIPAY".equals(callback.getPayChannel())) {
                    // 重新处理支付宝回调
                    Map<String, String> params = JsonUtil.fromJson(
                        callback.getCallbackData(), Map.class);
                    paymentCallbackService.handleAlipayCallback(params);
                }
                
                log.info("回调重试成功：{}", callback.getOrderNo());
            } catch (Exception e) {
                log.error("回调重试失败：{}", callback.getOrderNo(), e);
                
                // 更新重试次数
                callback.setRetryCount(callback.getRetryCount() + 1);
                paymentCallbackMapper.updateById(callback);
            }
        }
    }
}
```

```java
@Component
public class SensitiveDataFilter {
    
    /**
     * 日志脱敏：手机号、订单号等
     */
    public static String maskSensitiveData(String data, String type) {
        if (StringUtils.isBlank(data)) {
            return data;
        }
        
        switch (type) {
            case "phone":
                return data.replaceAll("(\\d{3})\\d{4}(\\d{4})", "$1****$2");
            case "order_no":
                return data.substring(0, 8) + "****" + 
                    data.substring(data.length() - 4);
            default:
                return data;
        }
    }
}
```

```java
@Component
public class PaymentRateLimiter {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    public boolean tryAcquire(Long userId, String operation) {
        String key = "rate:limit:" + operation + ":" + userId;
        
        // 限制：每分钟最多5次下单
        Long count = redisTemplate.opsForValue().increment(key);
        if (count == 1) {
            redisTemplate.expire(key, 1, TimeUnit.MINUTES);
        }
        
        return count <= 5;
    }
```
