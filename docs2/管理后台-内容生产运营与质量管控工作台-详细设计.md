### 6.3 状态机代码实现

```typescript
// 任务状态机定义
const TASK_STATE_MACHINE = {
  PENDING: {
    transitions: {
      START: 'IN_PROGRESS',
      CANCEL: 'CANCELLED',
    }
  },
  IN_PROGRESS: {
    transitions: {
      SUBMIT_REVIEW: 'IN_REVIEW',
      REVERT: 'PENDING',
    }
  },
  IN_REVIEW: {
    transitions: {
      APPROVE: 'APPROVED',
      REJECT: 'REVISION_REQUIRED',
      CONDITIONAL_PASS: 'REVISION_REQUIRED',
    }
  },
  REVISION_REQUIRED: {
    transitions: {
      START_REVISION: 'IN_PROGRESS',
      RESUBMIT: 'IN_REVIEW',
    }
  },
  APPROVED: {
    transitions: {
      PUBLISH: 'PUBLISHED',
      ARCHIVE: 'ARCHIVED',
    }
  },
  PUBLISHED: {
    transitions: {
      UNPUBLISH: 'REVISION_REQUIRED',
      ARCHIVE: 'ARCHIVED',
    }
  },
  ARCHIVED: {
    transitions: {} // 终态
  },
  CANCELLED: {
    transitions: {} // 终态
  }
};

// 状态流转校验
function validateTransition(
  currentStatus: string,
  action: string,
  userRole: string
): { valid: boolean; newStatus?: string; error?: string } {
  const state = TASK_STATE_MACHINE[currentStatus];
  if (!state) {
    return { valid: false, error: `未知状态: ${currentStatus}` };
  }

  const newStatus = state.transitions[action];
  if (!newStatus) {
    return { valid: false, error: `当前状态 ${currentStatus} 不支持操作 ${action}` };
  }

  // 角色权限校验
  const rolePermissions: Record<string, string[]> = {
    'CONTENT_OPS_MANAGER': ['CANCEL', 'PUBLISH', 'ARCHIVE'],
    'CONTENT_PROD_ADMIN': ['START', 'REVERT', 'CANCEL'],
    'QUALITY_LEAD': ['APPROVE', 'REJECT', 'CONDITIONAL_PASS', 'UNPUBLISH'],
    'SUBJECT_EDITOR': ['START', 'SUBMIT_REVIEW', 'START_REVISION', 'RESUBMIT'],
  };

  if (!rolePermissions[userRole]?.includes(action)) {
    return { valid: false, error: `角色 ${userRole} 无权执行操作 ${action}` };
  }

  return { valid: true, newStatus };
}
```

### 7.2 AI辅助审核流程

```python
# AI 预检管线 - 在提交人工审核前自动执行

class ContentAIPreCheckPipeline:
    """内容AI预检管线"""
    
    CHECK_STAGES = [
        "factual_accuracy",    # 事实准确性检查
        "knowledge_alignment", # 知识点对齐检查
        "age_appropriateness", # 适龄化检查
        "content_safety",      # 内容安全检查
        "format_validation",   # 格式规范检查
        "plagiarism_check",    # 查重检查
    ]
    
    async def run(self, task_id: int, content: str, metadata: dict) -> dict:
        """
        执行AI预检
        返回: {
            "overall_score": 85.5,
            "passed": True,
            "stages": {...},
            "issues": [...],
            "suggestions": [...]
        }
        """
        results = {}
        issues = []
        
        for stage in self.CHECK_STAGES:
            checker = self.get_checker(stage)
            result = await checker.check(content, metadata)
            results[stage] = result
            
            if not result.passed:
                issues.extend(result.issues)
        
        overall_score = self._calculate_score(results)
        
        return {
            "overall_score": overall_score,
            "passed": overall_score >= self.threshold,
            "stages": results,
            "issues": issues,
            "ai_pre_check": results,  # 存入 content_quality_reviews.ai_pre_check
        }
    
    def _calculate_score(self, results: dict) -> float:
        """各阶段加权评分"""
        weights = {
            "factual_accuracy": 30,
            "knowledge_alignment": 20,
            "age_appropriateness": 15,
            "content_safety": 20,  # 必须通过，否则直接0分
            "format_validation": 10,
            "plagiarism_check": 5,
        }
        
        # 内容安全一票否决
        if not results["content_safety"].passed:
            return 0.0
        
        score = sum(
            weights[stage] * results[stage].score / 100
            for stage in self.CHECK_STAGES
        )
        return round(score, 2)
```

```java
// 批量创建任务 - 使用批量插入
@Transactional
public BatchCreateResult batchCreateTasks(List<CreateTaskDTO> tasks, Long operatorId) {
    final int BATCH_SIZE = 100;
    
    if (tasks.size() > BATCH_SIZE) {
        throw new ContentProdException("CONTENT_PROD_006", 
            "批量创建上限为 " + BATCH_SIZE + " 条");
    }
    
    // 生成任务编号（批量获取序号）
    List<String> taskNos = idGenerator.batchGenerate("CPT", tasks.size());
    
    List<ContentProductionTask> entities = tasks.stream()
        .map(dto -> buildEntity(dto, taskNos, operatorId))
        .collect(Collectors.toList());
    
    // MyBatis 批量插入
    taskMapper.batchInsert(entities);
    
    // 异步创建关联记录（审计日志等）
    eventPublisher.publishEvent(new TasksBatchCreatedEvent(entities));
    
    return BatchCreateResult.from(entities);
}
```

```java
// 内容生产关键事件 - 通过消息总线通知其他系统

@Component
public class ContentProdEventPublisher {
    
    @Autowired
    private RocketMQTemplate mqTemplate;
    
    // 内容发布事件 → 推荐系统更新推荐池
    public void onContentPublished(ContentProductionTask task) {
        mqTemplate.asyncSend(
            "content-published-topic",
            ContentPublishedEvent.builder()
                .contentId(task.getOutputContentId())
                .contentType(task.getContentType())
                .subjectCode(task.getSubjectCode())
                .knowledgePointIds(task.getKnowledgePointIds())
                .build(),
            new SendCallback() {
                @Override public void onSuccess(SendResult r) {}
                @Override public void onException(Throwable e) {
                    log.error("通知推荐系统失败", e);
                }
            }
        );
    }
    
    // 缺口发现事件 → 触发供应商系统自动询价
    public void onGapDetected(ContentGapAnalysis gap) {
        if (gap.getDemandLevel() == DemandLevel.CRITICAL) {
            mqTemplate.send("content-gap-critical-topic", 
                GapDetectedEvent.from(gap));
        }
    }
    
    // 质量预警事件 → 通知运营经理
    public void onQualityAlert(QualityAlert alert) {
        mqTemplate.send("quality-alert-topic", alert);
    }
}
```
