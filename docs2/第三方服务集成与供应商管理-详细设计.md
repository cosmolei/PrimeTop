## 1. 模块概述

### 1.1 定位

第三方服务集成与供应商管理是 PrimeTop 的基础设施层横切关注点。本文档统一梳理项目涉及的所有第三方服务（大模型 API、短信、推送、支付、OCR、ASR/TTS、云存储、内容审核、日志监控等），定义统一的抽象层接口、供应商切换策略、密钥管理规范、成本监控体系和集成测试方案。

**本文档不替代各模块的详细设计**，而是提供跨模块的供应商管理总览和集成规范。各模块具体业务逻辑仍以各自文档为准。

### 1.2 解决的核心问题

| 问题 | 说明 |
|------|------|
| 供应商锁定风险 | 单一供应商停服/涨价/降质时缺乏替代方案 |
| 密钥散落管理 | 各模块各自管理API密钥，无统一轮转和审计 |
| 成本不可控 | 多个第三方服务调用成本缺乏统一监控和预算管控 |
| 环境隔离混乱 | 开发/测试/生产环境的第三方服务配置混用 |
| 集成测试困难 | 第三方服务Mock不统一，测试环境搭建复杂 |
| 供应商评估缺失 | 无标准化的供应商选型和评估流程 |

#### 2.2.3 抽象接口

```python
from abc import ABC, abstractmethod
from typing import AsyncIterator, Optional
from dataclasses import dataclass

@dataclass
class LLMMessage:
    role: str          # system / user / assistant
    content: str
    name: Optional[str] = None

@dataclass
class LLMResponse:
    content: str
    model: str
    usage: dict        # {"prompt_tokens": int, "completion_tokens": int, "total_tokens": int}
    finish_reason: str  # stop / length / content_filter
    request_id: str
    latency_ms: int

@dataclass
class LLMStreamChunk:
    content: str
    finish_reason: Optional[str]
    usage: Optional[dict]

class BaseLLMAdapter(ABC):
    """大模型统一抽象接口"""

    def __init__(self, config: dict):
        self.api_key = config["api_key"]
        self.base_url = config["base_url"]
        self.default_model = config["default_model"]
        self.timeout = config.get("timeout", 30)
        self.max_retries = config.get("max_retries", 2)

    @abstractmethod
    async def chat(
        self,
        messages: list[LLMMessage],
        model: Optional[str] = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
        **kwargs
    ) -> LLMResponse:
        """同步调用"""
        ...

    @abstractmethod
    async def chat_stream(
        self,
        messages: list[LLMMessage],
        model: Optional[str] = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
        **kwargs
    ) -> AsyncIterator[LLMStreamChunk]:
        """流式调用"""
        ...

    @abstractmethod
    async def count_tokens(self, text: str, model: Optional[str] = None) -> int:
        """Token计数"""
        ...

    def get_model_info(self) -> dict:
        """返回当前适配器支持的模型信息"""
        return {
            "provider": self.__class__.__name__,
            "default_model": self.default_model,
        }
```

#### 2.2.4 供应商适配器示例（通义千问）

```python
import httpx
import json
from typing import AsyncIterator, Optional

class QwenAdapter(BaseLLMAdapter):
    """阿里通义千问适配器"""

    SUPPORTED_MODELS = ["qwen-max", "qwen-plus", "qwen-turbo", "qwen-long"]

    async def chat(
        self,
        messages: list[LLMMessage],
        model: Optional[str] = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
        **kwargs
    ) -> LLMResponse:
        model = model or self.default_model
        payload = {
            "model": model,
            "messages": [{"role": m.role, "content": m.content} for m in messages],
            "temperature": temperature,
            "max_tokens": max_tokens,
            "result_format": "message",
        }

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            for attempt in range(self.max_retries + 1):
                try:
                    resp = await client.post(
                        f"{self.base_url}/v1/services/aigc/text-generation/generation",
                        headers={
                            "Authorization": f"Bearer {self.api_key}",
                            "Content-Type": "application/json",
                        },
                        json=payload,
                    )
                    resp.raise_for_status()
                    data = resp.json()
                    return LLMResponse(
                        content=data["output"]["choices"][0]["message"]["content"],
                        model=model,
                        usage=data.get("usage", {}),
                        finish_reason=data["output"]["choices"][0].get("finish_reason", "stop"),
                        request_id=data.get("request_id", ""),
                        latency_ms=int(resp.elapsed.total_seconds() * 1000),
                    )
                except httpx.HTTPStatusError as e:
                    if e.response.status_code == 429:
                        # 限流：等待后重试
                        await asyncio.sleep(2 ** attempt)
                        continue
                    raise LLMProviderError(f"Qwen API error: {e.response.status_code}")
                except httpx.TimeoutException:
                    if attempt == self.max_retries:
                        raise LLMProviderError("Qwen API timeout after retries")

    async def chat_stream(
        self,
        messages: list[LLMMessage],
        model: Optional[str] = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
        **kwargs
    ) -> AsyncIterator[LLMStreamChunk]:
        model = model or self.default_model
        payload = {
            "model": model,
            "messages": [{"role": m.role, "content": m.content} for m in messages],
            "temperature": temperature,
            "max_tokens": max_tokens,
            "result_format": "message",
            "stream": True,
            "incremental_output": True,
        }

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            async with client.stream(
                "POST",
                f"{self.base_url}/v1/services/aigc/text-generation/generation",
                headers={
                    "Authorization": f"Bearer {self.api_key}",
                    "Content-Type": "application/json",
                },
                json=payload,
            ) as resp:
                resp.raise_for_status()
                async for line in resp.aiter_lines():
                    if not line.startswith("data:"):
                        continue
                    data_str = line[5:].strip()
                    if data_str == "[DONE]":
                        break
                    try:
                        data = json.loads(data_str)
                        choice = data["output"]["choices"][0]
                        yield LLMStreamChunk(
                            content=choice["message"].get("content", ""),
                            finish_reason=choice.get("finish_reason"),
                            usage=data.get("usage"),
                        )
                    except json.JSONDecodeError:
                        continue

    async def count_tokens(self, text: str, model: Optional[str] = None) -> int:
        # 通义千问使用 tiktoken 兼容估算
        # 生产环境应调用官方 token 计数接口
        return len(text) // 3  # 粗略估算
```

#### 2.4.3 图片预处理管线

在调用 OCR 供应商之前，客户端和服务端均需进行图片预处理：

```
原始图片 → 方向检测与矫正 → 去阴影 → 对比度增强 → 缩放至合适尺寸 → 裁剪ROI → OCR
```

```python
class ImagePreprocessor:
    """图片预处理器（服务端实现）"""

    @staticmethod
    async def preprocess(image: bytes) -> bytes:
        """
        预处理管线：
        1. 检测图片方向，自动旋转至正向
        2. 去除阴影和噪点
        3. 增强文字对比度
        4. 缩放至合理尺寸（长边 ≤ 4096px）
        5. 转为高质量 JPEG
        """
        import cv2
        import numpy as np

        img_array = np.frombuffer(image, dtype=np.uint8)
        img = cv2.imdecode(img_array, cv2.IMREAD_COLOR)

        # 1. 缩放
        max_dim = max(img.shape[:2])
        if max_dim > 4096:
            scale = 4096 / max_dim
            img = cv2.resize(img, None, fx=scale, fy=scale, interpolation=cv2.INTER_AREA)

        # 2. 去阴影（基于自适应阈值）
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        blurred = cv2.GaussianBlur(gray, (21, 21), 0)
        diff = cv2.absdiff(gray, blurred)
        _, threshold = cv2.threshold(diff, 15, 255, cv2.THRESH_BINARY)

        # 3. 对比度增强 (CLAHE)
        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        enhanced = clahe.apply(gray)

        # 4. 编码输出
        _, encoded = cv2.imencode('.jpg', enhanced, [cv2.IMWRITE_JPEG_QUALITY, 95])
        return encoded.tobytes()
```

```python
    PHONE_INTERVAL_SECONDS = 60      # 同一号码两次发送间隔≥60秒
    PHONE_HOURLY_LIMIT = 5           # 同一号码每小时最多5条

    # 验证码
    CODE_LENGTH = 6
    CODE_EXPIRE_SECONDS = 300        # 验证码有效期5分钟
    CODE_MAX_VERIFY_ATTEMPTS = 5     # 最多验证5次

    async def check_and_send(self, phone: str, ip: str, adapter: BaseSMSAdapter) -> SMSResult:
        """带频率检查的发送"""
        # 1. 检查IP限制
        ip_daily = await redis.get(f"sms:ip:daily:{ip}")
        if ip_daily and int(ip_daily) >= self.IP_DAILY_LIMIT:
            raise RateLimitError("IP日发送次数已达上限")

        # 2. 检查手机号限制
        phone_interval = await redis.get(f"sms:phone:interval:{phone}")
        if phone_interval:
            remaining = self.PHONE_INTERVAL_SECONDS - (time.time() - float(phone_interval))
            if remaining > 0:
                raise RateLimitError(f"请{int(remaining)}秒后重试")

        # 3. 生成验证码
        code = "".join([str(random.randint(0, 9)) for _ in range(self.CODE_LENGTH)])

        # 4. 发送
        result = await adapter.send_verification_code(phone, code)

        if result.success:
            # 5. 记录限制数据到Redis
            pipe = redis.pipeline()
            pipe.setex(f"sms:phone:interval:{phone}", self.PHONE_INTERVAL_SECONDS, time.time())
            pipe.setex(f"sms:phone:code:{phone}", self.CODE_EXPIRE_SECONDS, json.dumps({
                "code": code,
                "attempts": 0,
                "max_attempts": self.CODE_MAX_VERIFY_ATTEMPTS,
            }))
            pipe.incr(f"sms:phone:daily:{phone}")
            pipe.expire(f"sms:phone:daily:{phone}", 86400)
            pipe.incr(f"sms:ip:daily:{ip}")
            pipe.expire(f"sms:ip:daily:{ip}", 86400)
            await pipe.execute()

        return result
```

```python
        "xiaomi": "mi_push",

        "vivo": "vivo_push",
    }

    def resolve_channel(self, device_info: dict) -> list[str]:
        """
        为设备确定推送通道优先级列表。
        返回通道列表，按优先级排序。
        """
        brand = device_info.get("brand", "").lower()
        os = device_info.get("os", "").lower()
        channels = []

        # 1. 厂商通道（仅Android）
        if os == "android" and brand in self.VENDOR_CHANNELS:
            channels.append(self.VENDOR_CHANNELS[brand])

        # 2. iOS 设备
        if os == "ios":
            channels.append("apns")

        # 3. 聚合通道作为后备

            channels.extend(["jpush", "getui", "fcm"])

        return channels or ["jpush"]  # 兜底
```

### 3.2 密钥管理规范

#### 3.2.1 加密与存储

```python
import os
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC

class SecretManager:
    """
    供应商密钥管理器。
    加密密钥（Master Key）从环境变量读取，永不入库。
    API密钥经AES-256加密后存储在 provider_secret 表中。
    """

    def __init__(self):
        master_key = os.environ["PROVIDER_MASTER_KEY"]  # 32字节 base64
        self._fernet = Fernet(master_key.encode())

    def encrypt(self, plain_text: str) -> str:
        """加密密钥值"""
        return self._fernet.encrypt(plain_text.encode()).decode()

    def decrypt(self, cipher_text: str) -> str:
        """解密密钥值"""
        return self._fernet.decrypt(cipher_text.encode()).decode()

    async def get_secret(self, provider_id: int, env: str) -> dict:
        """
        获取供应商在指定环境下的所有活跃密钥。
        返回 {"api_key": "...", "secret_key": "...", ...}
        """
        rows = await db.fetch_all(
            "SELECT secret_type, encrypted_value, key_version "
            "FROM provider_secret "
            "WHERE provider_id = :pid AND env = :env AND is_active = 1",
            {"pid": provider_id, "env": env},
        )
        secrets = {}
        for row in rows:
            secrets[row["secret_type"]] = self.decrypt(row["encrypted_value"])
        return secrets

    async def rotate_secret(
        self,
        provider_id: int,
        env: str,
        secret_type: str,
        new_value: str,
    ) -> int:
        """
        密钥轮转：
        1. 旧密钥标记 is_active = 0
        2. 新密钥写入并标记 is_active = 1
        3. 返回新密钥版本号
        """
        # 获取当前版本
        current = await db.fetch_one(
            "SELECT MAX(key_version) as max_ver FROM provider_secret "
            "WHERE provider_id = :pid AND env = :env AND secret_type = :st",
            {"pid": provider_id, "env": env, "st": secret_type},
        )
        new_version = (current["max_ver"] or 0) + 1

        async with db.transaction():
            # 旧密钥失效
            await db.execute(
                "UPDATE provider_secret SET is_active = 0 "
                "WHERE provider_id = :pid AND env = :env AND secret_type = :st AND is_active = 1",
                {"pid": provider_id, "env": env, "st": secret_type},
            )
            # 新密钥写入
            await db.execute(
                "INSERT INTO provider_secret (provider_id, env, secret_type, encrypted_value, key_version, last_rotated_at) "
                "VALUES (:pid, :env, :st, :enc, :ver, NOW())",
                {
                    "pid": provider_id,
                    "env": env,
                    "st": secret_type,
                    "enc": self.encrypt(new_value),
                    "ver": new_version,
                },
            )

        return new_version
```

## 4. 供应商路由与降级策略

### 4.1 路由引擎

```python
class ProviderRouter:
    """
    供应商路由引擎。
    根据服务类型、场景、健康状态和优先级自动选择最优供应商。
    """

    async def resolve(
        self,
        service_type: str,
        scene: str | None = None,
        exclude_providers: list[str] | None = None,
    ) -> dict:
        """
        解析最优供应商。
        
        Args:
            service_type: 服务类型 (llm/ocr/sms/...)
            scene: 业务场景 (math_solve/essay_review/...)
            exclude_providers: 排除的供应商列表（重试时使用）
        
        Returns:
            供应商配置字典
        """
        # 1. 查询可用供应商（按优先级排序）
        providers = await db.fetch_all(
            "SELECT * FROM provider_registry "
            "WHERE service_type = :st AND status = 'active' "
            "AND health_status IN ('healthy', 'degraded') "
            "ORDER BY priority ASC",
            {"st": service_type},
        )

        # 2. 过滤排除的供应商
        if exclude_providers:
            providers = [p for p in providers if p["provider_code"] not in exclude_providers]

        if not providers:
            raise NoAvailableProviderError(f"无可用 {service_type} 供应商")

        # 3. 如果有场景路由配置，优先使用场景指定供应商
        if scene:
            scene_config = await self._get_scene_routing(service_type, scene)
            if scene_config:
                for preferred_code in scene_config.get("preferred_providers", []):
                    for p in providers:
                        if p["provider_code"] == preferred_code:
                            return dict(p)

        # 4. 返回优先级最高的健康供应商
        return dict(providers[0])
```

### 4.2 调用重试与降级

```python
class ResilientProviderCall:
    """
    带重试和降级的供应商调用包装器。
    
    策略：
    1. 首次调用优先级最高的供应商
    2. 失败后自动切换到下一个供应商
    3. 所有供应商失败后返回降级结果或抛出异常
    """

    MAX_RETRIES_PER_PROVIDER = 2   # 每个供应商最多重试2次
    RETRYABLE_ERRORS = {"timeout", "rate_limited", "server_error"}

    async def call(
        self,
        service_type: str,
        operation: str,
        params: dict,
        scene: str | None = None,
        fallback_value=None,
    ) -> tuple[any, dict]:
        """
        执行带降级的供应商调用。
        
        Returns:
            (result, metadata) 其中 metadata 包含使用的供应商、延迟、成本等信息
        """
        router = ProviderRouter()
        tried_providers = []
        last_error = None

        while True:
            try:
                provider = await router.resolve(service_type, scene, tried_providers)
            except NoAvailableProviderError:
                break

            adapter = self._create_adapter(service_type, provider)
            method = getattr(adapter, operation)

            for attempt in range(self.MAX_RETRIES_PER_PROVIDER + 1):
                start = time.time()
                try:
                    result = await method(**params)
                    latency_ms = int((time.time() - start) * 1000)

                    # 记录成功日志
                    await self._log_call(
                        service_type=service_type,
                        provider_code=provider["provider_code"],
                        operation=operation,
                        status="success",
                        latency_ms=latency_ms,
                    )

                    return result, {
                        "provider": provider["provider_code"],
                        "latency_ms": latency_ms,
                        "attempt": attempt + 1,
                    }

                except ProviderError as e:
                    latency_ms = int((time.time() - start) * 1000)
                    last_error = e
                    error_type = e.error_type

                    # 记录失败日志
                    await self._log_call(
                        service_type=service_type,
                        provider_code=provider["provider_code"],
                        operation=operation,
                        status=error_type,
                        latency_ms=latency_ms,
                        error_message=str(e),
                    )

                    if error_type not in self.RETRYABLE_ERRORS:
                        # 非可重试错误（如认证失败），直接切换供应商
                        break

                    if attempt < self.MAX_RETRIES_PER_PROVIDER:
                        # 可重试：等待后重试
                        await asyncio.sleep(0.5 * (2 ** attempt))
                        continue

            # 当前供应商已耗尽，尝试下一个
            tried_providers.append(provider["provider_code"])

        # 所有供应商均失败
        if fallback_value is not None:
            return fallback_value, {"provider": "fallback", "error": str(last_error)}

        raise AllProvidersFailedError(
            f"所有 {service_type} 供应商调用失败。最后错误: {last_error}"
        )
```

### 4.3 降级场景与策略

| 服务类型 | 降级策略 | 降级表现 |
|---------|---------|---------|
| LLM | 切换供应商 → 切换低级模型 → 返回缓存/预设回复 | "暂时无法回答，请稍后再试" |
| OCR | 切换供应商 → 降低识别精度 → 提示用户手动输入 | "识别困难，请手动输入题目" |
| ASR | 切换供应商 → 降级为文字输入 | "语音服务暂不可用，请文字提问" |
| TTS | 切换供应商 → 不播放语音 | 静默显示文字 |
| SMS | 切换供应商 → 语音验证码 → 滑动验证 | "短信发送失败，请使用语音验证码" |
| Push | 切换供应商 → 端内消息替代 | 不推送，打开APP后可见 |
| Payment | 切换支付渠道 → 提示稍后重试 | "支付暂不可用，请稍后再试" |
| Storage | 切换供应商 → 本地临时存储后异步上传 | "上传中，请稍候" |
| Moderation | 切换供应商 → 本地规则引擎过滤 → 放行并标记待审 | 不影响用户体验，后台补审 |
| Embedding | 切换供应商 → 使用本地模型 | 检索质量可能下降 |

```python
class CostMeter:
    """
    第三方服务成本计量器。
    每次调用后记录成本，定期汇总。
    """

    # 各服务的计费单价配置（可后台动态更新）
    PRICING_CONFIG = {
        "llm": {
            "qwen-max": {"input": 0.02, "output": 0.06},         # 元/千token
            "doubao-pro": {"input": 0.0008, "output": 0.002},
            "ernie-4.0": {"input": 0.03, "output": 0.06},
            "deepseek-reasoner": {"input": 0.001, "output": 0.002},
        },
        "ocr": {
            "aliyun_ocr": 0.01,        # 元/次
            "baidu_ocr": 0.008,
        },
        "sms": {
            "aliyun_sms": 0.045,        # 元/条
            "tencent_sms": 0.055,
        },
        "asr": {
            "aliyun_asr": 0.008,        # 元/次
            "xunfei_asr": 0.005,
        },
    }

    async def record(
        self,
        service_type: str,
        provider_code: str,
        operation: str,
        usage: dict,  # {"prompt_tokens": 1000, "completion_tokens": 500} 或 {"count": 1}
    ) -> float:
        """计算并记录本次调用成本（元）"""
        pricing = self.PRICING_CONFIG.get(service_type, {}).get(provider_code)
        if not pricing:
            return 0.0

        if service_type == "llm" and isinstance(pricing, dict):
            cost = (
                usage.get("prompt_tokens", 0) / 1000 * pricing["input"]
                + usage.get("completion_tokens", 0) / 1000 * pricing["output"]
            )
        elif isinstance(pricing, (int, float)):
            cost = usage.get("count", 1) * pricing
        else:
            cost = 0.0

        # 写入 Redis 实时计数器
        today = datetime.now().strftime("%Y-%m-%d")
        pipe = redis.pipeline()
        pipe.incrbyfloat(f"cost:daily:{service_type}:{today}", cost)
        pipe.incrbyfloat(f"cost:daily:total:{today}", cost)
        pipe.incrbyfloat(f"cost:monthly:{service_type}:{today[:7]}", cost)
        pipe.incrbyfloat(f"cost:monthly:total:{today[:7]}", cost)
        await pipe.execute()

        return round(cost, 6)
```

### 10.1 错误码定义

| 错误码 | 说明 | 处理方式 |
|--------|------|---------|
| PROVIDER_001 | 无可用供应商 | 检查供应商注册表和健康状态 |
| PROVIDER_002 | 供应商认证失败 | 检查密钥是否正确、是否过期 |
| PROVIDER_003 | 供应商限流 | 自动重试或切换供应商 |
| PROVIDER_004 | 供应商超时 | 重试或切换供应商 |
| PROVIDER_005 | 供应商服务端错误 | 重试或切换供应商 |
| PROVIDER_006 | 响应解析失败 | 记录日志，重试或切换 |
| PROVIDER_007 | 所有供应商失败 | 返回降级结果或提示用户 |
| PROVIDER_008 | 密钥轮转失败 | 检查Master Key和新密钥 |
| PROVIDER_009 | 成本超预算 | 告警通知，自动限流 |

### 10.2 自定义异常类

```python
class ProviderError(Exception):
    """供应商调用基础异常"""
    def __init__(self, message: str, error_type: str = "unknown", provider: str = ""):
        super().__init__(message)
        self.error_type = error_type
        self.provider = provider

class LLMProviderError(ProviderError):
    """LLM供应商异常"""
    def __init__(self, message: str, provider: str = ""):
        super().__init__(message, "llm_error", provider)

class OCRProviderError(ProviderError):
    """OCR供应商异常"""
    def __init__(self, message: str, provider: str = ""):
        super().__init__(message, "ocr_error", provider)

class NoAvailableProviderError(ProviderError):
    """无可用供应商"""
    def __init__(self, message: str):
        super().__init__(message, "no_provider")

class AllProvidersFailedError(ProviderError):
    """所有供应商均失败"""
    def __init__(self, message: str):
        super().__init__(message, "all_failed")

class RateLimitError(ProviderError):
    """限流错误"""
    def __init__(self, message: str):
        super().__init__(message, "rate_limited")
```
