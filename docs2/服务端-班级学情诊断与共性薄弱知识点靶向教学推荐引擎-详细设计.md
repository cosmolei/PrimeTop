## 1. 概述

### 1.1 背景与目标

PrimeTop 已设计了个体学情分析、教师班级管理、学校教学进度同步等模块，但缺少一个关键的中间层服务：**将班级内所有学生的离散学习数据聚合为班级整体画像**，自动诊断共性薄弱环节，并向教师推荐靶向教学方案。

本引擎的核心目标：

| 目标 | 描述 |
|------|------|
| 班级学情聚合 | 将个体答题、错题、学习时长等数据实时聚合到班级维度 |
| 共性薄弱诊断 | 从多维度识别"班级整体薄弱"和"部分学生集中薄弱"的知识点 |
| 靶向推荐 | 基于诊断结果推荐教学策略、练习题目和课堂重点 |
| 效果追踪 | 追踪教师干预后的班级学情改善情况 |
| 教师赋能 | 为教师提供数据驱动的教学决策辅助 |

### 1.2 用户角色与使用场景

| 角色 | 场景 | 说明 |
|------|------|------|
| 教师 | 查看班级学情仪表盘 | 课前快速了解班级整体掌握情况 |
| 教师 | 共性薄弱点分析 | 识别需要重点讲解的知识点 |
| 教师 | 获取教学推荐 | 获取针对性的教学资源和练习推荐 |
| 教师 | 干预效果追踪 | 查看针对性教学后班级改善趋势 |
| 年级组长 | 跨班级横向对比 | 了解各班学情差异，协调教学资源 |
| 教研员 | 年级/学科分析 | 从宏观角度把握教学质量 |
| 系统 | 自动触发推荐 | 新数据到达后自动更新诊断并推送建议 |

### 1.3 设计原则

1. **准实时优先**：核心指标在 5 分钟内完成聚合更新
2. **多维交叉**：知识维度 × 能力维度 × 时间维度 × 学生分群
3. **可解释性**：每条推荐附带数据依据，教师可理解推荐理由
4. **渐进式增强**：数据量不足时提供基础统计，数据充足时启用高级模型
5. **保护隐私**：班级报表中不暴露个体可识别信息（除非教师有查看权限）

## 4. 数据库设计

### 4.1 PostgreSQL 表结构

#### 4.1.1 班级学情快照表 `class_diagnosis_snapshot`

```sql
CREATE TABLE class_diagnosis_snapshot (
    id              BIGSERIAL PRIMARY KEY,
    class_id        BIGINT       NOT NULL,
    subject         VARCHAR(20)  NOT NULL,
    grade           SMALLINT     NOT NULL,
    semester        VARCHAR(20)  NOT NULL,        -- 如 "2025-2026-2"
    
    -- 聚合统计
    student_count           INT          NOT NULL DEFAULT 0,
    active_student_count    INT          NOT NULL DEFAULT 0,   -- 近7天有学习行为
    avg_mastery_score       DECIMAL(5,2) NOT NULL DEFAULT 0,   -- 班级平均掌握度
    avg_correct_rate        DECIMAL(5,2) NOT NULL DEFAULT 0,   -- 班级平均正确率
    avg_study_minutes       DECIMAL(8,2) NOT NULL DEFAULT 0,   -- 日均学习时长(分钟)
    total_questions_answered BIGINT      NOT NULL DEFAULT 0,   -- 总答题数
    weakness_count          INT          NOT NULL DEFAULT 0,   -- 薄弱点数量
    critical_weakness_count INT          NOT NULL DEFAULT 0,   -- 严重薄弱点数量
    
    -- 能力维度JSON
    ability_radar   JSONB,    -- {"computation": 72.5, "logic": 68.0, "reading": 75.3, ...}
    
    -- 学生分群统计
    cluster_summary JSONB,    -- {"advanced": 8, "average": 25, "struggling": 7, ...}
    
    -- 快照元数据
    snapshot_type   VARCHAR(20)  NOT NULL,       -- REALTIME / DAILY / WEEKLY
    data_window_start TIMESTAMPTZ,                -- 数据窗口起始
    data_window_end   TIMESTAMPTZ,                -- 数据窗口结束
    computed_at     TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    
    CONSTRAINT uk_class_subject_semester_snapshot UNIQUE (class_id, subject, semester, snapshot_type, computed_at)
);

CREATE INDEX idx_snapshot_class_subject ON class_diagnosis_snapshot (class_id, subject, semester, computed_at DESC);
CREATE INDEX idx_snapshot_type ON class_diagnosis_snapshot (snapshot_type, computed_at DESC);
```

#### 4.1.2 班级知识点掌握度表 `class_knowledge_mastery`

```sql
CREATE TABLE class_knowledge_mastery (
    id                  BIGSERIAL PRIMARY KEY,
    class_id            BIGINT       NOT NULL,
    subject             VARCHAR(20)  NOT NULL,
    knowledge_point_id  BIGINT       NOT NULL,
    knowledge_point_name VARCHAR(200) NOT NULL,
    chapter_id          BIGINT,
    chapter_name        VARCHAR(200),
    semester            VARCHAR(20)  NOT NULL,
    
    -- 掌握度指标
    mastery_score       DECIMAL(5,2) NOT NULL,     -- 班级平均掌握度 0-100
    coverage_rate       DECIMAL(5,2) NOT NULL,     -- 覆盖率：已练习该知识点的学生比例
    correct_rate        DECIMAL(5,2) NOT NULL,     -- 班级平均正确率
    avg_attempts        DECIMAL(3,1) NOT NULL,     -- 平均尝试次数
    
    -- 分布指标
    mastered_count      INT  NOT NULL DEFAULT 0,   -- 掌握人数 (掌握度>80%)
    partial_count       INT  NOT NULL DEFAULT 0,   -- 部分掌握 (60%-80%)
    weak_count          INT  NOT NULL DEFAULT 0,   -- 薄弱 (<60%)
    not_attempted_count INT  NOT NULL DEFAULT 0,   -- 未练习
    
    -- 趋势
    trend_direction     VARCHAR(10)  NOT NULL DEFAULT 'STABLE',  -- UP / DOWN / STABLE
    trend_delta         DECIMAL(5,2) NOT NULL DEFAULT 0,         -- 与上期差值
    consecutive_decline INT          NOT NULL DEFAULT 0,         -- 连续下降周期数
    
    -- 时间分区
    period_type         VARCHAR(10)  NOT NULL,     -- DAILY / WEEKLY / MONTHLY
    period_start        DATE         NOT NULL,
    period_end          DATE         NOT NULL,
    
    computed_at         TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    created_at          TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

-- 核心查询索引
CREATE INDEX idx_ckm_class_subject_period ON class_knowledge_mastery 
    (class_id, subject, period_type, period_start DESC);
CREATE INDEX idx_ckm_knowledge_point ON class_knowledge_mastery 
    (knowledge_point_id, period_start DESC);
CREATE INDEX idx_ckm_mastery_score ON class_knowledge_mastery 
    (class_id, subject, mastery_score, period_start DESC)
    WHERE period_type = 'WEEKLY';
```

### 5.2 Kafka Streams 实时聚合拓扑

```java
@Component
public class ClassDiagnosisStreamTopology {

    /**
     * 实时聚合拓扑：消费答题事件和掌握度更新事件，
     * 按班级+学科+知识点维度进行增量聚合
     */
    @Bean
    public KStream<String, ClassKpAggregate> classDiagnosisPipeline(
            StreamsBuilder builder) {

        // 1. 消费答题事件流
        KStream<String, AnswerRecordEvent> answerStream = builder.stream(
            "answer-record-events",
            Consumed.with(Serdes.String(), jsonSerde(AnswerRecordEvent.class))
        );

        // 2. 按班级+学科+知识点重新分组
        KStream<String, KpAnswerAggregate> kpAnswerStream = answerStream
            .flatMapValues(event -> {
                // 一个答题事件可能关联多个知识点
                return event.getKnowledgePointIds().stream()
                    .map(kpId -> new KpAnswerAggregate(
                        event.getClassId(),
                        event.getSubject(),
                        kpId,
                        event.getStudentId(),
                        event.isCorrect(),
                        event.getAnswerDurationMs(),
                        event.getHintUsed()
                    ))
                    .toList();
            })
            .selectKey((key, val) -> 
                // 分区键: classId-subject-kpId
                val.classId() + "-" + val.subject() + "-" + val.kpId()
            );

        // 3. 按时间窗口(5分钟)聚合
        KTable<Windowed<String>, KpWindowAggregate> kpWindowTable = kpAnswerStream
            .groupByKey()
            .windowedBy(TimeWindows.ofSizeWithNoGrace(Duration.ofMinutes(5)))
            .aggregate(
                KpWindowAggregate::new,
                this::aggregateAnswer,
                Materialized.with(Serdes.String(), jsonSerde(KpWindowAggregate.class))
            );

        // 4. 每个窗口结束后输出到下游
        kpWindowTable
            .toStream()
            .filter((windowedKey, agg) -> agg.getAnswerCount() > 0)
            .to("class-kp-aggregate-windows",
                Produced.with(
                    WindowedSerdes.timeWindowedSerdeFrom(String.class),
                    jsonSerde(KpWindowAggregate.class)
                ));

        // 5. 同步处理掌握度更新事件
        KStream<String, MasteryUpdateEvent> masteryStream = builder.stream(
            "mastery-update-events",
            Consumed.with(Serdes.String(), jsonSerde(MasteryUpdateEvent.class))
        );

        masteryStream
            .selectKey((k, v) -> v.getClassId() + "-" + v.getSubject() + "-" + v.getKnowledgePointId())
            .groupByKey()
            .aggregate(
                ClassKpRealtime::new,
                this::aggregateMastery,
                Materialized.with(Serdes.String(), jsonSerde(ClassKpRealtime.class))
            )
            .toStream()
            .to("class-kp-realtime-mastery",
                Produced.with(Serdes.String(), jsonSerde(ClassKpRealtime.class)));

        return answerStream;
    }

    /**
     * 窗口聚合逻辑
     */
    private KpWindowAggregate aggregateAnswer(String key, KpAnswerEvent event, KpWindowAggregate agg) {
        agg.setClassId(event.getClassId());
        agg.setSubject(event.getSubject());
        agg.setKnowledgePointId(event.getKnowledgePointId());
        agg.setWindowEnd(Instant.now());
        
        agg.setAnswerCount(agg.getAnswerCount() + 1);
        if (event.isCorrect()) {
            agg.setCorrectCount(agg.getCorrectCount() + 1);
        }
        agg.addStudentId(event.getStudentId());
        agg.addDurationMs(event.getAnswerDurationMs());
        
        return agg;
    }
}
```

### 6.1 班级能力雷达图计算

```java
@Service
@Slf4j
public class ClassAbilityRadarCalculator {

    // 能力维度定义（按学科可配置）
    private static final Map<String, List<String>> SUBJECT_DIMENSIONS = Map.of(
        "数学", List.of("计算能力", "逻辑推理", "空间想象", "数据分析", "应用建模", "证明表达"),
        "语文", List.of("识记理解", "阅读鉴赏", "表达交流", "写作应用", "综合探究"),
        "英语", List.of("听力理解", "词汇语法", "阅读理解", "写作表达", "口语交际"),
        "物理", List.of("概念理解", "实验探究", "计算推理", "模型建构", "综合应用")
    );

    /**
     * 计算班级能力雷达图
     * 
     * @param classId 班级ID
     * @param subject 学科
     * @param dateRange 时间范围
     * @return 能力维度分值
     */
    public ClassAbilityRadar calculate(Long classId, String subject, 
                                        LocalDate startDate, LocalDate endDate) {
        
        List<String> dimensions = SUBJECT_DIMENSIONS.getOrDefault(subject, 
            List.of("基础理解", "综合应用", "拓展创新"));
        
        Map<String, BigDecimal> dimensionScores = new LinkedHashMap<>();
        
        for (String dimension : dimensions) {
            // 1. 获取该维度关联的知识点
            List<Long> kpIds = knowledgePointService.getKpIdsByAbilityDimension(subject, dimension);
            
            if (kpIds.isEmpty()) {
                dimensionScores.put(dimension, BigDecimal.ZERO);
                continue;
            }
            
            // 2. 获取该维度相关题目的答题数据
            List<AnswerStat> stats = answerRecordRepository.aggregateByDimension(
                classId, subject, kpIds, startDate, endDate);
            
            // 3. 计算维度得分
            BigDecimal score = calculateDimensionScore(stats, kpIds.size());
            dimensionScores.put(dimension, score);
        }
        
        return new ClassAbilityRadar(classId, subject, dimensionScores, 
                                      startDate, endDate, Instant.now());
    }
    
    /**
     * 维度得分计算逻辑
     * 
     * 得分 = 正确率 * 60% + 覆盖率 * 20% + 参与度 * 20%
     * 其中：
     *   正确率 = 正确答题数 / 总答题数 * 100
     *   覆盖率 = 已练习知识点数 / 该维度知识点总数 * 100
     *   参与度 = min(实际答题人数 / 班级活跃人数 * 100, 100)
     */
    private BigDecimal calculateDimensionScore(List<AnswerStat> stats, int totalKpCount) {
        if (stats.isEmpty()) return BigDecimal.ZERO;
        
        long totalAnswers = stats.stream().mapToLong(AnswerStat::getTotalCount).sum();
        long correctAnswers = stats.stream().mapToLong(AnswerStat::getCorrectCount).sum();
        int activeKpCount = stats.size();
        int uniqueStudents = stats.stream().mapToInt(AnswerStat::getStudentCount).max().orElse(0);
        int classSize = classRepository.getClassSize(stats.get(0).getClassId());
        
        BigDecimal correctRate = totalAnswers > 0 
            ? BigDecimal.valueOf(correctAnswers * 100.0 / totalAnswers) 
            : BigDecimal.ZERO;
        
        BigDecimal coverageRate = totalKpCount > 0 
            ? BigDecimal.valueOf(activeKpCount * 100.0 / totalKpCount) 
            : BigDecimal.ZERO;
        
        BigDecimal participationRate = classSize > 0 
            ? BigDecimal.valueOf(Math.min(uniqueStudents * 100.0 / classSize, 100.0)) 
            : BigDecimal.ZERO;
        
        return correctRate.multiply(new BigDecimal("0.6"))
            .add(coverageRate.multiply(new BigDecimal("0.2")))
            .add(participationRate.multiply(new BigDecimal("0.2")))
            .setScale(2, RoundingMode.HALF_UP);
    }
}
```

### 7.3 薄弱点检测器实现

```java
@Service
@Slf4j
public class WeaknessDetector {

    @Value("${class-diagnosis.weakness.thresholds.class-avg-critical}")
    private double thresholdCritical;
    
    @Value("${class-diagnosis.weakness.thresholds.class-avg-high}")
    private double thresholdHigh;
    
    @Value("${class-diagnosis.weakness.thresholds.class-avg-medium}")
    private double thresholdMedium;
    
    @Autowired
    private WeaknessClusterRepository weaknessRepo;
    
    @Autowired
    private ErrorPatternAnalyzer errorPatternAnalyzer;

    /**
     * 检测班级在指定时间范围内的所有薄弱知识点
     */
    public List<WeaknessCluster> detectWeaknesses(
            Long classId, String subject, LocalDate date) {
        
        // 1. 获取班级所有知识点的最新掌握度
        List<KpMasteryStat> stats = masteryRepository.findClassKpStats(
            classId, subject, date);
        
        if (stats.size() < 3) {
            log.debug("班级 {} 学科 {} 知识点数据不足, 跳过诊断", classId, subject);
            return List.of();
        }
        
        int classSize = classRepository.getStudentCount(classId);
        if (classSize < minStudents) {
            return List.of();
        }
        
        List<WeaknessCluster> detected = new ArrayList<>();
        
        for (KpMasteryStat stat : stats) {
            // 跳过样本不足的知识点
            if (stat.getAnsweredCount() < minAnswersPerKp) continue;
            if (stat.getCoverageRate() < minKpCoverage) continue;
            
            Optional<WeaknessCluster> weakness = evaluateWeakness(stat, classSize, date);
            weakness.ifPresent(detected::add);
        }
        
        // 2. 更新数据库 (新建或更新已有记录)
        for (WeaknessCluster weakness : detected) {
            upsertWeakness(weakness);
        }
        
        // 3. 检查是否有薄弱点已改善 → 更新状态
        checkResolvedWeaknesses(classId, subject, date, stats);
        
        return detected;
    }
    
    /**
     * 评估单个知识点是否构成薄弱点
     */
    private Optional<WeaknessCluster> evaluateWeakness(
            KpMasteryStat stat, int classSize, LocalDate date) {
        
        double masteryScore = stat.getMasteryScore().doubleValue();
        int weakCount = stat.getWeakCount();
        double weakRatio = (double) weakCount / classSize;
        
        // 判定严重程度
        Severity severity = determineSeverity(masteryScore);
        if (severity == null) return Optional.empty();  // 不构成薄弱
        
        // 判定薄弱类型
        WeaknessType type = determineWeaknessType(weakRatio);
        
        // 获取受影响学生列表
        List<Long> affectedStudentIds = masteryRepository
            .findWeakStudentIds(stat.getClassId(), stat.getSubject(), 
                stat.getKnowledgePointId(), individualWeakThreshold, date);
        
        // 错误模式分析
        List<ErrorPattern> errorPatterns = errorPatternAnalyzer
            .analyzeCommonErrors(stat.getClassId(), stat.getSubject(), 
                stat.getKnowledgePointId(), date.minusDays(7), date);
        
        WeaknessCluster weakness = WeaknessCluster.builder()
            .classId(stat.getClassId())
            .subject(stat.getSubject())
            .knowledgePointId(stat.getKnowledgePointId())
            .knowledgePointName(stat.getKnowledgePointName())
            .chapterId(stat.getChapterId())
            .chapterName(stat.getChapterName())
            .weaknessType(type)
            .severity(severity)
            .affectedStudentIds(affectedStudentIds.toArray(Long[]::new))
            .affectedCount(affectedStudentIds.size())
            .affectedRatio(BigDecimal.valueOf(weakRatio))
            .commonErrorPatterns(toJson(errorPatterns))
            .firstDetectedAt(getFirstDetectedTime(stat, date))
            .lastDetectedAt(Instant.now())
            .persistenceDays(calculatePersistenceDays(stat, date))
            .status("ACTIVE")
            .periodStart(date.minusDays(7))
            .periodEnd(date)
            .build();
        
        return Optional.of(weakness);
    }
    
    private Severity determineSeverity(double masteryScore) {
        if (masteryScore < thresholdCritical) return Severity.CRITICAL;
        if (masteryScore < thresholdHigh) return Severity.HIGH;
        if (masteryScore < thresholdMedium) return Severity.MEDIUM;
        if (masteryScore < 80.0) return Severity.LOW;
        return null;  // 掌握度良好，不构成薄弱
    }
    
    private WeaknessType determineWeaknessType(double weakRatio) {
        if (weakRatio >= 0.50) return WeaknessType.COMMON;
        if (weakRatio >= 0.20) return WeaknessType.PARTIAL;
        return WeaknessType.SCATTERED;
    }
}
```

```java
@Service
public class ErrorPatternAnalyzer {

    /**
     * 分析某个知识点下的常见错误模式
     * 
     * 算法：对错误记录按错误原因标签聚类，
     *      结合知识点类型特征，输出 Top-N 错误模式
     */
    public List<ErrorPattern> analyzeCommonErrors(
            Long classId, String subject, Long knowledgePointId,
            LocalDate startDate, LocalDate endDate) {
        
        // 1. 获取该知识点下所有错误记录
        List<MistakeRecord> mistakes = mistakeRepository.findByClassAndKp(
            classId, subject, knowledgePointId, startDate, endDate);
        
        if (mistakes.isEmpty()) return List.of();
        
        // 2. 按 error_reason 分组统计
        Map<String, Long> reasonCounts = mistakes.stream()
            .collect(Collectors.groupingBy(
                m -> m.getErrorReason() != null ? m.getErrorReason() : "未分类",
                Collectors.counting()
            ));
        
        int totalMistakes = mistakes.size();
        
        // 3. 计算每种错误模式的占比
        List<ErrorPattern> patterns = reasonCounts.entrySet().stream()
            .map(e -> new ErrorPattern(
                e.getKey(),
                e.getValue().intValue(),
                BigDecimal.valueOf(e.getValue() * 100.0 / totalMistakes)
                    .setScale(2, RoundingMode.HALF_UP)
            ))
            .sorted(Comparator.comparing(ErrorPattern::getRatio).reversed())
            .limit(5)
            .toList();
        
        return patterns;
    }
}
```

### 8.2 推荐生成器实现

```java
@Service
@Slf4j
public class TeachingRecommendationGenerator {

    @Autowired
    private QuestionRepository questionRepo;
    
    @Autowired
    private ResourceRepository resourceRepo;
    
    @Autowired
    private TeachingRecommendationRepository recRepo;
    
    @Autowired
    private NotificationService notificationService;

    /**
     * 每日推荐生成（由定时任务触发）
     */
    public void generateDailyRecommendations(LocalDate date) {
        // 1. 获取所有活跃的薄弱点
        List<WeaknessCluster> activeWeaknesses = weaknessRepo
            .findAllActiveByDate(date);
        
        // 2. 按班级+学科分组
        Map<String, List<WeaknessCluster>> grouped = activeWeaknesses.stream()
            .collect(Collectors.groupingBy(
                w -> w.getClassId() + "-" + w.getSubject()
            ));
        
        // 3. 为每组生成推荐
        grouped.forEach((key, weaknesses) -> {
            String[] parts = key.split("-");
            Long classId = Long.parseLong(parts[0]);
            String subject = parts[1];
            Long teacherId = classRepository.getHeadTeacherId(classId, subject);
            
            if (teacherId == null) return;
            
            try {
                generateRecommendationsForClass(classId, subject, teacherId, weaknesses, date);
            } catch (Exception e) {
                log.error("推荐生成失败 classId={} subject={}", classId, subject, e);
            }
        });
    }
    
    private void generateRecommendationsForClass(
            Long classId, String subject, Long teacherId,
            List<WeaknessCluster> weaknesses, LocalDate date) {
        
        // 检查今日已生成推荐数 (限制每天最多5条)
        int existingCount = recRepo.countByTeacherAndDate(teacherId, date);
        int maxNewRecommendations = MAX_DAILY_RECOMMENDATIONS - existingCount;
        
        if (maxNewRecommendations <= 0) return;
        
        // 按严重程度排序
        weaknesses.sort(Comparator.comparing(
            w -> ((WeaknessCluster) w).getSeverity().getLevel(), Comparator.reverseOrder()));
        
        int generated = 0;
        
        for (WeaknessCluster weakness : weaknesses) {
            if (generated >= maxNewRecommendations) break;
            
            // 检查是否已有同知识点的未处理推荐
            Optional<TeachingRecommendation> existing = recRepo
                .findPendingByClassAndKp(classId, subject, weakness.getKnowledgePointId());
            
            if (existing.isPresent()) {
                // 更新已有推荐
                updateExistingRecommendation(existing.get(), weakness);
                continue;
            }
            
            // 生成新推荐
            TeachingRecommendation rec = buildRecommendation(teacherId, weakness);
            recRepo.save(rec);
            generated++;
            
            // 推送通知给教师
            if (rec.getPriority() == RecommendationPriority.URGENT 
                || rec.getPriority() == RecommendationPriority.HIGH) {
                notificationService.notifyTeacher(teacherId, rec);
            }
        }
    }
    
    /**
     * 构建推荐
     */
    private TeachingRecommendation buildRecommendation(
            Long teacherId, WeaknessCluster weakness) {
        
        RecommendationPriority priority = determinePriority(weakness);
        
        // 根据薄弱类型生成不同的推荐策略
        RecommendationBuilder builder = RecommendationBuilder.forWeakness(weakness);
        
        if (weakness.getWeaknessType() == WeaknessType.COMMON 
            && weakness.getSeverity() == Severity.CRITICAL) {
            // 共性严重问题 → 全班干预
            builder.type(RecommendationType.STRATEGY)
                .title(String.format("【紧急】%s 共性薄弱点需全班复习", 
                    weakness.getKnowledgePointName()))
                .summary(buildCommonWeaknessSummary(weakness))
                .rationale(buildRationale(weakness))
                .exercises(selectExercises(weakness, 5, Difficulty.BASIC))
                .resources(selectResources(weakness, 2))
                .estimatedTime(30)
                .targetAll();
            
        } else if (weakness.getWeaknessType() == WeaknessType.PARTIAL) {
            // 部分学生薄弱 → 分组练习
            builder.type(RecommendationType.EXERCISE)
                .title(String.format("%s 部分学生需加强练习", 
                    weakness.getKnowledgePointName()))
                .summary(buildPartialWeaknessSummary(weakness))
                .rationale(buildRationale(weakness))
                .exercises(selectExercises(weakness, 8, Difficulty.PROGRESSIVE))
                .targetStudents(weakness.getAffectedStudentIds());
            
        } else {
            // 零散问题 → 个别辅导建议
            builder.type(RecommendationType.STRATEGY)
                .title(String.format("%s 个别辅导建议", 
                    weakness.getKnowledgePointName()))
                .summary(buildScatteredWeaknessSummary(weakness))
                .rationale(buildRationale(weakness))
                .exercises(selectExercises(weakness, 3, Difficulty.TARGETED))
                .targetStudents(weakness.getAffectedStudentIds());
        }
        
        return builder.build(teacherId, priority);
    }
    
    /**
     * 为薄弱点选择练习题
     * 策略：按难度递进选择，覆盖不同题型
     */
    private List<Long> selectExercises(WeaknessCluster weakness, int count, 
                                        DifficultyStrategy strategy) {
        List<Question> candidates = questionRepo.findByKnowledgePointAndDifficulty(
            weakness.getKnowledgePointId(), 
            weakness.getSubject(),
            strategy.toRange()
        );
        
        // 优先选择该知识点下高频出错的题目
        Set<Long> typicalMistakeIds = extractTypicalMistakeIds(weakness);
        
        List<Long> selected = new ArrayList<>();
        
        // 先选典型错题
        selected.addAll(candidates.stream()
            .filter(q -> typicalMistakeIds.contains(q.getId()))
            .limit(count / 2)
            .map(Question::getId)
            .toList());
        
        // 再按难度递进补充
        selected.addAll(candidates.stream()
            .filter(q -> !selected.contains(q.getId()))
            .sorted(Comparator.comparing(Question::getDifficulty))
            .limit(count - selected.size())
            .map(Question::getId)
            .toList());
        
        return selected;
    }
    
    /**
     * 生成推荐理由（自然语言）
     */
    private String buildRationale(WeaknessCluster weakness) {
        StringBuilder sb = new StringBuilder();
        
        sb.append("数据依据：\n");
        sb.append(String.format("- 班级平均掌握度：%.1f%%\n", 
            weakness.getMasteryScore()));
        sb.append(String.format("- 薄弱学生比例：%d/%d (%.1f%%)\n",
            weakness.getAffectedCount(), 
            weakness.getClassSize(),
            weakness.getAffectedRatio() * 100));
        sb.append(String.format("- 持续时间：%d 天\n", 
            weakness.getPersistenceDays()));
        
        if (weakness.getCommonErrorPatterns() != null 
            && !weakness.getCommonErrorPatterns().isEmpty()) {
            sb.append("- 主要错误模式：\n");
            for (ErrorPattern pattern : weakness.getCommonErrorPatterns()) {
                sb.append(String.format("  · %s (%.1f%%)\n", 
                    pattern.getPattern(), pattern.getRatio()));
            }
        }
        
        return sb.toString();
    }
}
```

### 8.3 学生分群算法

```java
@Service
public class StudentClusterAnalyzer {

    /**
     * 对班级学生进行多维度分群
     * 
     * 策略：
     * 1. 基础分群：按掌握度三分（拔尖/中等/薄弱）
     * 2. 趋势分群：按近期变化趋势（进步/下滑/稳定）
     * 3. 交叉分群：组合标记（如"正在下滑的拔尖学生"）
     */
    public Map<StudentClusterType, List<StudentClusterMember>> clusterStudents(
            Long classId, String subject, 
            LocalDate startDate, LocalDate endDate) {
        
        // 1. 获取每个学生在该学科的整体掌握度
        List<StudentSubjectStat> stats = studentStatRepository
            .getSubjectStats(classId, subject, startDate, endDate);
        
        // 2. 计算趋势数据（与前一周对比）
        LocalDate prevStart = startDate.minusDays(7);
        LocalDate prevEnd = startDate.minusDays(1);
        List<StudentSubjectStat> prevStats = studentStatRepository
            .getSubjectStats(classId, subject, prevStart, prevEnd);
        
        Map<Long, StudentSubjectStat> prevMap = prevStats.stream()
            .collect(Collectors.toMap(StudentSubjectStat::getStudentId, s -> s));
        
        // 3. 分类
        Map<StudentClusterType, List<StudentClusterMember>> result = new HashMap<>();
        
        for (StudentSubjectStat stat : stats) {
            StudentSubjectStat prev = prevMap.get(stat.getStudentId());
            BigDecimal mastery = stat.getMasteryScore();
            BigDecimal trendDelta = prev != null 
                ? mastery.subtract(prev.getMasteryScore()) 
                : BigDecimal.ZERO;
            
            // 基础分群
            StudentClusterType baseType;
            if (mastery.compareTo(new BigDecimal("85")) >= 0) {
                baseType = StudentClusterType.ADVANCED;
            } else if (mastery.compareTo(new BigDecimal("60")) >= 0) {
                baseType = StudentClusterType.AVERAGE;
            } else {
                baseType = StudentClusterType.STRUGGLING;
            }
            
            result.computeIfAbsent(baseType, k -> new ArrayList<>())
                .add(new StudentClusterMember(stat.getStudentId(), mastery, 
                    stat.getCorrectRate(), trendDelta, baseType));
            
            // 趋势分群（趋势变化超过 5 分才标记）
            if (trendDelta.compareTo(new BigDecimal("5")) > 0) {
                result.computeIfAbsent(StudentClusterType.IMPROVING, k -> new ArrayList<>())
                    .add(new StudentClusterMember(stat.getStudentId(), mastery, 
                        stat.getCorrectRate(), trendDelta, StudentClusterType.IMPROVING));
            } else if (trendDelta.compareTo(new BigDecimal("-5")) < 0) {
                result.computeIfAbsent(StudentClusterType.DECLINING, k -> new ArrayList<>())
                    .add(new StudentClusterMember(stat.getStudentId(), mastery, 
                        stat.getCorrectRate(), trendDelta, StudentClusterType.DECLINING));
            }
        }
        
        return result;
    }
}
```

```java
@Service
public class ClassDiagnosisCacheService {

    private final RedissonClient redisson;
    
    // 实时掌握度 - 使用 Hash 存储，由 Kafka 流更新
    private static final String REALTIME_KP_KEY = "class-diag:realtime:kp";
    
    // 班级摘要 - 使用 JSON 字符串
    private static final String SUMMARY_KEY_PREFIX = "class-diag:summary:";
    
    /**
     * 更新实时知识点掌握度（由 Kafka 消费者调用）
     */
    public void updateRealtimeKp(Long classId, String subject, 
                                  Long kpId, KpRealtimeData data) {
        String field = classId + ":" + subject + ":" + kpId;
        RMap<String, String> map = redisson.getMap(REALTIME_KP_KEY);
        map.put(field, JsonUtils.toJson(data));
    }
    
    /**
     * 批量获取班级知识点实时掌握度
     */
    public Map<Long, KpRealtimeData> batchGetRealtimeKp(
            Long classId, String subject, List<Long> kpIds) {
        RMap<String, String> map = redisson.getMap(REALTIME_KP_KEY);
        Map<String, String> entries = map.getAll(
            kpIds.stream()
                .map(kpId -> classId + ":" + subject + ":" + kpId)
                .collect(Collectors.toSet())
        );
        
        return entries.entrySet().stream()
            .collect(Collectors.toMap(
                e -> Long.parseLong(e.getKey().split(":")[2]),
                e -> JsonUtils.fromJson(e.getValue(), KpRealtimeData.class)
            ));
    }
    
    /**
     * 缓存班级摘要
     */
    public void cacheSummary(Long classId, String subject, 
                              String periodType, ClassDiagnosisSummary summary) {
        String key = SUMMARY_KEY_PREFIX + classId + ":" + subject + ":" + periodType;
        RBucket<String> bucket = redisson.getBucket(key);
        bucket.set(JsonUtils.toJson(summary), 5, TimeUnit.MINUTES);
    }
    
    /**
     * 使班级缓存失效（当有新数据写入时）
     */
    public void invalidateClassCache(Long classId, String subject) {
        // 使用模式匹配删除所有该班级该学科的缓存
        redisson.getKeys().deleteByPattern(
            "class-diag:*:" + classId + ":" + subject + ":*");
        redisson.getKeys().deleteByPattern(
            "class-diag:*:" + classId + ":" + subject);
    }
}
```

### 12.2 干预效果评估

```java
@Service
@Slf4j
public class InterventionTracker {

    /**
     * 评估已完成干预的效果
     * 
     * 策略：对比干预前后的知识点掌握度、正确率变化
     * 评估窗口：干预完成 3-7 天后
     */
    @Scheduled(cron = "0 0 4 * * ?")
    public void evaluatePendingInterventions() {
        // 1. 查找待评估的干预记录
        List<InterventionRecord> pending = interventionRepo.findPendingEvaluation(
            LocalDate.now().minusDays(7), LocalDate.now().minusDays(3));
        
        for (InterventionRecord intervention : pending) {
            try {
                evaluateIntervention(intervention);
            } catch (Exception e) {
                log.error("干预效果评估失败 interventionId={}", 
                    intervention.getId(), e);
            }
        }
    }
    
    private void evaluateIntervention(InterventionRecord intervention) {
        Long classId = intervention.getClassId();
        String subject = intervention.getSubject();
        List<Long> kpIds = Arrays.asList(intervention.getKnowledgePointIds());
        
        // 1. 获取干预前快照
        Map<Long, BigDecimal> preSnapshot = intervention.getPreInterventionSnapshot() != null
            ? intervention.getPreInterventionSnapshot()
            : Collections.emptyMap();
        
        // 2. 获取当前掌握度
        LocalDate now = LocalDate.now();
        LocalDate evalStart = now.minusDays(3);
        Map<Long, BigDecimal> currentMastery = masteryRepository
            .getClassKpMasteryMap(classId, subject, kpIds, evalStart, now);
        
        // 3. 计算改善分数
        BigDecimal totalImprovement = BigDecimal.ZERO;
        Map<Long, BigDecimal> improvements = new HashMap<>();
        
        for (Long kpId : kpIds) {
            BigDecimal before = preSnapshot.getOrDefault(kpId, BigDecimal.ZERO);
            BigDecimal after = currentMastery.getOrDefault(kpId, BigDecimal.ZERO);
            BigDecimal delta = after.subtract(before);
            improvements.put(kpId, delta);
            totalImprovement = totalImprovement.add(delta);
        }
        
        BigDecimal avgImprovement = totalImprovement.divide(
            BigDecimal.valueOf(kpIds.size()), 2, RoundingMode.HALF_UP);
        
        // 4. 更新干预记录
        intervention.setPostInterventionSnapshot(currentMastery);
        intervention.setImprovementScore(avgImprovement);
        intervention.setImprovementNote(generateEvaluationNote(avgImprovement, improvements));
        interventionRepo.save(intervention);
        
        // 5. 更新关联推荐的效果评分
        if (intervention.getRecommendationId() != null) {
            TeachingRecommendation rec = recRepo.findById(
                intervention.getRecommendationId()).orElse(null);
            if (rec != null) {
                rec.setEffectivenessScore(avgImprovement);
                recRepo.save(rec);
            }
        }
        
        // 6. 如果改善显著，标记薄弱点为已解决
        if (avgImprovement.compareTo(new BigDecimal("15")) > 0) {
            for (Long kpId : kpIds) {
                weaknessRepo.resolveByClassAndKp(classId, subject, kpId, 
                    "干预后掌握度提升 " + avgImprovement + "%");
            }
        }
        
        // 7. 发布评估完成事件
        eventPublisher.publishEvent(new InterventionEvaluatedEvent(
            intervention.getId(), intervention.getRecommendationId(),
            avgImprovement, intervention.getImprovementNote(), Instant.now()
        ));
    }
    
    private String generateEvaluationNote(BigDecimal improvement, 
                                           Map<Long, BigDecimal> details) {
        StringBuilder sb = new StringBuilder();
        if (improvement.compareTo(BigDecimal.ZERO) > 0) {
            sb.append("干预有效，平均掌握度提升 ").append(improvement).append("%。");
        } else {
            sb.append("干预效果不显著，平均掌握度变化 ").append(improvement).append("%。");
            sb.append("建议尝试其他教学策略。");
        }
        sb.append("\n各知识点变化：\n");
        details.forEach((kpId, delta) -> {
            sb.append(String.format("  - 知识点%d: %+.1f%%\n", kpId, delta));
        });
        return sb.toString();
    }
}
```

```java
public class WeaknessStateMachine {

    public enum WeaknessState {
        ACTIVE,      // 活跃薄弱
        IMPROVING,   // 正在改善（连续2次检测掌握度上升 > 5%）
        RESOLVED,    // 已解决（掌握度 > 80% 且稳定 > 7天）
        ARCHIVED     // 已归档（超过 90 天自动归档）
    }
    
    private static final Map<WeaknessState, Set<WeaknessState>> TRANSITIONS = Map.of(
        WeaknessState.ACTIVE, EnumSet.of(WeaknessState.IMPROVING, WeaknessState.RESOLVED, WeaknessState.ARCHIVED),
        WeaknessState.IMPROVING, EnumSet.of(WeaknessState.ACTIVE, WeaknessState.RESOLVED, WeaknessState.ARCHIVED),
        WeaknessState.RESOLVED, EnumSet.of(WeaknessState.ACTIVE),  // 可能复发
        WeaknessState.ARCHIVED, EnumSet.of(WeaknessState.ACTIVE)   // 可能重新激活
    );
    
    public static boolean canTransition(WeaknessState from, WeaknessState to) {
        return TRANSITIONS.getOrDefault(from, EnumSet.noneOf(WeaknessState.class)).contains(to);
    }
}
```

```java
public class DiagnosisFallbackHandler {

    /**
     * 当数据分析服务不可用时的降级策略
     */
    public ClassDiagnosisSummary fallbackSummary(Long classId, String subject) {
        // 1. 尝试从缓存返回上一次摘要
        ClassDiagnosisSummary cached = cacheService.getSummary(classId, subject, "WEEKLY");
        if (cached != null) {
            log.warn("使用缓存数据降级返回 classId={} subject={}", classId, subject);
            cached.setDegraded(true);
            cached.setDegradedReason("数据服务暂时不可用，展示最近一次计算结果");
            return cached;
        }

        ClassBasicStat basicStat = basicStatRepo.findLatest(classId, subject);

            return ClassDiagnosisSummary.minimal(basicStat, 
                "详细分析暂时不可用，仅展示基础统计数据");
        }
        
        // 3. 完全降级
        return ClassDiagnosisSummary.empty(classId, subject, 
            "班级学情数据暂时不可用，请稍后重试");
    }
    
    /**

        attempts = "3",
        backoff = @Backoff(delay = 60000, multiplier = 2),
        dltTopic = "class-kp-aggregate-windows.DLT"
    )
    public void handleAggregateWithRetry(KpWindowAggregate agg) {
        try {
            processAggregate(agg);
        } catch (DataAccessException e) {

            throw e;  // 触发重试

            "KP_AGGREGATE", JsonUtils.toJson(agg), Instant.now()
        ));
```
