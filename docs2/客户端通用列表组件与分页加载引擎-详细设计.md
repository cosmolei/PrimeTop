# 客户端通用列表组件与分页加载引擎 - 详细设计

## 1. 概述

### 1.1 背景

PrimeTop 应用中存在大量列表类页面：错题列表、练习列表、搜索结果、消息列表、课程目录、对话历史、题目列表、收藏夹、学习报告、工单列表、管理后台各类列表等。这些页面的交互模式高度相似（下拉刷新、上拉加载更多、空态、加载态、错误态、筛选排序），但实现方式分散、不一致。

本文设计一套统一的列表组件架构与分页加载引擎，覆盖所有列表场景，确保：
- 开发者用最少代码创建功能完整的列表页
- 交互体验和视觉表现全局一致
- 分页策略统一，后端接口规范对齐
- 离线/弱网场景有一致降级行为

### 1.2 适用范围

| 端 | 场景 |
|---|---|
| Flutter 客户端 | 所有列表类页面（学生端、家长端） |
| Flutter Web / 管理后台 | 列表类页面 |
| 服务端 | 分页 API 规范统一 |

### 1.3 设计目标

1. **零模板代码**：通过声明式配置创建列表页，不超过 50 行 Dart 代码
2. **6 种列表状态全覆盖**：初始加载、加载完成、空态、错误、加载更多、没有更多数据
3. **双分页模式**：Cursor-based（推荐）+ Offset-based（兼容旧接口）
4. **离线优先**：本地缓存 + 网络刷新，断网时展示缓存数据
5. **性能基准**：1000 项列表滑动 60fps，首屏渲染 < 200ms

## 3. 分页协议设计

### 3.1 统一分页请求

```dart
/// 分页请求参数
class PageRequest {
  /// Cursor 模式：上一页最后一条的游标值
  final String? cursor;

  /// Offset 模式：页码（从 1 开始）
  final int? page;

  /// 每页条数（默认 20，最大 100）
  final int pageSize;

  /// 排序字段
  final String? sortBy;

  /// 排序方向
  final SortDirection sortDirection;

  /// 筛选条件（key-value 形式）
  final Map<String, dynamic> filters;

  /// 搜索关键词
  final String? keyword;

  const PageRequest({
    this.cursor,
    this.page,
    this.pageSize = 20,
    this.sortBy,
    this.sortDirection = SortDirection.desc,
    this.filters = const {},
    this.keyword,
  });

  /// 序列化为查询参数
  Map<String, dynamic> toQueryParams() {
    final params = <String, dynamic>{
      'pageSize': pageSize,
      if (sortBy != null) 'sortBy': sortBy,
      'sortDirection': sortDirection.name,
      ...filters,
    };
    if (cursor != null) {
      params['cursor'] = cursor;
    } else if (page != null) {
      params['page'] = page;
    }
    if (keyword != null) {
      params['keyword'] = keyword;
    }
    return params;
  }
}

enum SortDirection { asc, desc }
```

### 3.2 统一分页响应

```dart
/// 分页响应数据
class PageResponse<T> {
  /// 数据列表
  final List<T> items;

  /// 是否还有下一页
  final bool hasMore;

  /// 下一页游标（Cursor 模式）
  final String? nextCursor;

  /// 总条数（Offset 模式专用，Cursor 模式可为 null）
  final int? total;

  /// 当前页码（Offset 模式专用）
  final int? currentPage;

  /// 聚合信息（可选，如分组统计）
  final Map<String, dynamic>? aggregations;

  /// 服务端时间戳
  final DateTime serverTime;

  const PageResponse({
    required this.items,
    required this.hasMore,
    this.nextCursor,
    this.total,
    this.currentPage,
    this.aggregations,
    DateTime? serverTime,
  }) : serverTime = serverTime ?? DateTime.now();

  /// 空响应
  factory PageResponse.empty() => PageResponse<T>(
        items: [],
        hasMore: false,
        serverTime: DateTime.now(),
      );

  /// 从 JSON 解析
  factory PageResponse.fromJson(
    Map<String, dynamic> json,
    T Function(Map<String, dynamic>) fromItemJson,
  ) {
    return PageResponse<T>(
      items: (json['items'] as List)
          .map((e) => fromItemJson(e as Map<String, dynamic>))
          .toList(),
      hasMore: json['hasMore'] as bool? ?? false,
      nextCursor: json['nextCursor'] as String?,
      total: json['total'] as int?,
      currentPage: json['page'] as int?,
      aggregations: json['aggregations'] as Map<String, dynamic>?,
      serverTime: json['serverTime'] != null
          ? DateTime.parse(json['serverTime'] as String)
          : DateTime.now(),
    );
  }
}
```

### 3.3 服务端 API 规范

#### Cursor 模式（推荐）

```
GET /api/v1/mistakes?cursor=eyJpZCI6MTAwfQ&pageSize=20&sortBy=createdAt&sortDirection=desc&subject=MATH&grade=G7

Response:
{
  "code": 0,
  "data": {
    "items": [
      { "id": "mistake_101", "subject": "MATH", ... },
      { "id": "mistake_100", "subject": "MATH", ... }
    ],
    "hasMore": true,
    "nextCursor": "eyJpZCI6OTl9",
    "serverTime": "2026-05-26T09:00:00Z"
  }
}
```

**Cursor 编码规则**：
- Cursor 为 Base64 编码的 JSON 字符串
- 包含排序字段值和 ID，确保分页稳定性
- 示例：`{"id":100,"createdAt":"2026-05-25T10:00:00Z"}` → `eyJpZCI6MTAwLCJjcmVhdGVkQXQiOiIyMDI2LTA1LTI1VDEwOjAwOjAwWiJ9`

#### Offset 模式（管理后台）

```
GET /api/v1/admin/questions?page=3&pageSize=50&sortBy=difficulty&sortDirection=asc&subject=PHYSICS

Response:
{
  "code": 0,
  "data": {
    "items": [...],
    "hasMore": true,
    "total": 1234,
    "page": 3,
    "pageSize": 50,
    "serverTime": "2026-05-26T09:00:00Z"

}
```

### 3.4 服务端 Cursor 生成规范（Python）

```python
import base64
import json
from dataclasses import dataclass
from typing import Any, Optional


@dataclass
class CursorValue:
    """Cursor 携带的排序锚点值"""
    id: str
    sort_field: str
    sort_value: Any

    def encode(self) -> str:
        payload = {
            "id": self.id,
            "f": self.sort_field,
            "v": self.sort_value,
        }
        return base64.urlsafe_b64encode(
            json.dumps(payload, ensure_ascii=False).encode()
        ).decode()

    @classmethod
    def decode(cls, cursor: str) -> "CursorValue":
        payload = json.loads(
            base64.urlsafe_b64decode(cursor.encode())
        )
        return cls(
            id=payload["id"],
            sort_field=payload["f"],
            sort_value=payload["v"],
        )


def build_cursor_query(
    model,
    cursor: Optional[str],
    sort_by: str,
    sort_desc: bool,
    page_size: int,
    filters: dict,
):
    """
    构建基于 Cursor 的分页查询。
    
    用法:
        query = build_cursor_query(
            model=MistakeRecord,
            cursor=request.cursor,
            sort_by="created_at",
            sort_desc=True,
            page_size=20,
            filters={"subject": "MATH"},
        )
        rows = await session.execute(query)
    """
    from sqlalchemy import select

    stmt = select(model)

    # 应用筛选条件
    for key, value in filters.items():
        if hasattr(model, key):
            stmt = stmt.where(getattr(model, key) == value)

    # Cursor 分页：基于排序字段值 + ID 的复合条件
    if cursor:
        cv = CursorValue.decode(cursor)
        sort_col = getattr(model, cv.sort_field, None)
        id_col = model.id

        if sort_col is not None:
            if sort_desc:
                # 降序：(sort_value < cursor_value) OR (sort_value == cursor_value AND id < cursor_id)
                stmt = stmt.where(
                    (sort_col < cv.sort_value) |
                    ((sort_col == cv.sort_value) & (id_col < cv.id))
                )
            else:
                # 升序：(sort_value > cursor_value) OR (sort_value == cursor_value AND id > cursor_id)
                stmt = stmt.where(
                    (sort_col > cv.sort_value) |
                    ((sort_col == cv.sort_value) & (id_col > cv.id))
                )

    # 排序
    sort_col = getattr(model, sort_by, None)
    if sort_col is not None:
        stmt = stmt.order_by(
            sort_col.desc() if sort_desc else sort_col.asc(),
            model.id.desc() if sort_desc else model.id.asc(),
        )

    # 多取一条用于判断 hasMore
    stmt = stmt.limit(page_size + 1)

    return stmt


def build_page_response(rows: list, page_size: int, last_sort_field: str):
    """从查询结果构建分页响应"""
    has_more = len(rows) > page_size
    items = rows[:page_size]

    next_cursor = None
    if has_more and items:
        last = items[-1]
        cv = CursorValue(
            id=str(last.id),
            sort_field=last_sort_field,
            sort_value=getattr(last, last_sort_field),
        )
        next_cursor = cv.encode()

    return {
        "items": [item.to_dict() for item in items],
        "hasMore": has_more,
        "nextCursor": next_cursor,
    }
```

```dart
/// 列表状态枚举
enum PagedListStatus {
  /// 初始态（尚未加载）
  initial,

  /// 首次加载中
  loading,

  /// 刷新中（下拉刷新触发）
  refreshing,

  /// 加载完成（有数据）
  loaded,

  /// 空态（首次加载无数据）
  empty,

  /// 首次加载失败
  error,

  /// 加载更多中
  loadingMore,

  /// 加载更多失败
  loadMoreError,

  /// 已加载全部数据
  noMore,
}

/// 列表完整状态
class PagedListState<T> {
  final PagedListStatus status;
  final List<T> items;
  final String? nextCursor;
  final int? totalCount;
  final String? errorMessage;
  final String? errorCode;
  final bool hasMore;
  final DateTime? lastRefreshTime;
  final Map<String, dynamic> activeFilters;
  final String? activeSortBy;
  final SortDirection? activeSortDirection;

  const PagedListState({
    this.status = PagedListStatus.initial,
    this.items = const [],
    this.nextCursor,
    this.totalCount,
    this.errorMessage,
    this.errorCode,
    this.hasMore = true,
    this.lastRefreshTime,
    this.activeFilters = const {},
    this.activeSortBy,
    this.activeSortDirection,
  });

  bool get isLoading =>
      status == PagedListStatus.loading ||
      status == PagedListStatus.refreshing;

  bool get canLoadMore =>
      hasMore &&
      status != PagedListStatus.loading &&
      status != PagedListStatus.refreshing &&
      status != PagedListStatus.loadingMore;

  bool get isEmpty =>
      status == PagedListStatus.empty ||
      (status == PagedListStatus.loaded && items.isEmpty);

  PagedListState<T> copyWith({
    PagedListStatus? status,
    List<T>? items,
    String? nextCursor,
    int? totalCount,
    String? errorMessage,
    String? errorCode,
    bool? hasMore,
    DateTime? lastRefreshTime,
    Map<String, dynamic>? activeFilters,
    String? activeSortBy,
    SortDirection? activeSortDirection,
  }) {
    return PagedListState<T>(
      status: status ?? this.status,
      items: items ?? this.items,
      nextCursor: nextCursor ?? this.nextCursor,
      totalCount: totalCount ?? this.totalCount,
      errorMessage: errorMessage ?? this.errorMessage,
      errorCode: errorCode ?? this.errorCode,
      hasMore: hasMore ?? this.hasMore,
      lastRefreshTime: lastRefreshTime ?? this.lastRefreshTime,
      activeFilters: activeFilters ?? this.activeFilters,
      activeSortBy: activeSortBy ?? this.activeSortBy,
      activeSortDirection: activeSortDirection ?? this.activeSortDirection,
    );
  }
}
```

### 4.3 数据获取抽象

```dart
/// 开发者需实现的分页数据获取接口
abstract class PageFetcher<T> {
  /// 获取首页数据
  Future<PageResponse<T>> fetchFirst(PageRequest request);

  /// 获取下一页数据
  Future<PageResponse<T>> fetchMore(PageRequest request);

  /// 获取离线缓存数据（可选覆写）
  Future<List<T>?> fetchCached(String cacheKey) async => null;

  /// 保存数据到离线缓存（可选覆写）
  Future<void> saveCached(String cacheKey, List<T> items) async {}

  /// 获取列表项唯一 ID
  String itemId(T item);

  /// 列表项去重（默认基于 ID 去重）
  List<T> deduplicate(List<T> existing, List<T> newItems) {
    final existingIds = existing.map(itemId).toSet();
    return [
      ...existing,
      ...newItems.where((item) => !existingIds.contains(itemId(item))),
    ];
  }
}

/// 基于 Dio 的通用 PageFetcher 实现
class DioPageFetcher<T> extends PageFetcher<T> {
  final Dio dio;
  final String endpoint;
  final T Function(Map<String, dynamic>) fromJson;
  final String Function(T) itemIdGetter;
  final String? cacheBoxName;
  final Duration cacheTTL;

  DioPageFetcher({
    required this.dio,
    required this.endpoint,
    required this.fromJson,
    required this.itemIdGetter,
    this.cacheBoxName,
    this.cacheTTL = const Duration(minutes: 30),
  });

  @override
  String itemId(T item) => itemIdGetter(item);

  @override
  Future<PageResponse<T>> fetchFirst(PageRequest request) async {
    final response = await dio.get(
      endpoint,
      queryParameters: request.toQueryParams(),
    );
    return PageResponse.fromJson(
      response.data['data'] as Map<String, dynamic>,
      fromJson,
    );
  }

  @override
  Future<PageResponse<T>> fetchMore(PageRequest request) => fetchFirst(request);

  @override
  Future<List<T>?> fetchCached(String cacheKey) async {
    if (cacheBoxName == null) return null;
    final box = await Hive.openBox(cacheBoxName!);
    final cached = box.get(cacheKey);
    if (cached == null) return null;

    final data = jsonDecode(cached as String) as List;
    final items = data
        .map((e) => fromJson(e as Map<String, dynamic>))
        .toList();

    // 检查缓存是否过期
    final cachedAt = box.get('${cacheKey}_time') as int?;
    if (cachedAt != null &&
        DateTime.now().millisecondsSinceEpoch - cachedAt >
            cacheTTL.inMilliseconds) {
      return null; // 缓存过期
    }
    return items;
  }

  @override
  Future<void> saveCached(String cacheKey, List<T> items) async {
    if (cacheBoxName == null) return;
    final box = await Hive.openBox(cacheBoxName!);
    await box.put(
      cacheKey,
      jsonEncode(items.map((e) => (e as dynamic).toJson()).toList()),
    );
    await box.put(
      '${cacheKey}_time',
      DateTime.now().millisecondsSinceEpoch,
    );
  }
}
```

### 4.4 列表控制器

```dart
/// 分页列表控制器
class PagedListController<T> extends ChangeNotifier {
  final PageFetcher<T> _fetcher;
  final int pageSize;
  final String cacheKey;
  final bool enableOfflineCache;

  PagedListState<T> _state = const PagedListState();
  PagedListState<T> get state => _state;

  // 防抖定时器
  Timer? _debounceTimer;
  static const _debounceDuration = Duration(milliseconds: 300);

  // 请求取消
  CancelToken? _cancelToken;

  PagedListController({
    required PageFetcher<T> fetcher,
    this.pageSize = 20,
    required this.cacheKey,
    this.enableOfflineCache = false,
  }) : _fetcher = fetcher;

  /// 初始化加载
  Future<void> initialize({
    Map<String, dynamic>? filters,
    String? sortBy,
    SortDirection? sortDirection,
  }) async {
    _state = _state.copyWith(
      activeFilters: filters ?? {},
      activeSortBy: sortBy,
      activeSortDirection: sortDirection,
    );

    // 先尝试加载离线缓存
    if (enableOfflineCache) {
      final cached = await _fetcher.fetchCached(cacheKey);
      if (cached != null && cached.isNotEmpty) {
        _state = _state.copyWith(
          status: PagedListStatus.loaded,
          items: cached,
          hasMore: true, // 缓存数据可能不完整
        );
        notifyListeners();

        // 后台静默刷新
        _silentRefresh();
        return;
      }
    }

    await _loadFirst();
  }

  /// 首次加载
  Future<void> _loadFirst() async {
    _cancelToken?.cancel();
    _cancelToken = CancelToken();

    _state = _state.copyWith(status: PagedListStatus.loading);
    notifyListeners();

    try {
      final request = PageRequest(
        pageSize: pageSize,
        sortBy: _state.activeSortBy,
        sortDirection: _state.activeSortDirection ?? SortDirection.desc,
        filters: _state.activeFilters,
      );

      final response = await _fetcher.fetchFirst(request);

      if (response.items.isEmpty) {
        _state = _state.copyWith(
          status: PagedListStatus.empty,
          hasMore: false,
          lastRefreshTime: response.serverTime,
        );
      } else {
        _state = _state.copyWith(
          status: PagedListStatus.loaded,
          items: response.items,
          nextCursor: response.nextCursor,
          hasMore: response.hasMore,
          totalCount: response.total,
          lastRefreshTime: response.serverTime,
        );

        // 保存离线缓存
        if (enableOfflineCache) {
          await _fetcher.saveCached(cacheKey, response.items);
        }
      }
    } on DioException catch (e) {
      if (e.type == DioExceptionType.cancel) return;
      _state = _state.copyWith(
        status: PagedListStatus.error,
        errorMessage: _mapDioError(e),
        errorCode: 'NETWORK_ERROR',
      );
    } catch (e) {
      _state = _state.copyWith(
        status: PagedListStatus.error,
        errorMessage: e.toString(),
        errorCode: 'UNKNOWN_ERROR',
      );
    }
    notifyListeners();
  }

  /// 下拉刷新
  Future<void> refresh() async {
    _cancelToken?.cancel();
    _cancelToken = CancelToken();

    _state = _state.copyWith(status: PagedListStatus.refreshing);
    notifyListeners();

    try {
      final request = PageRequest(
        pageSize: pageSize,
        sortBy: _state.activeSortBy,
        sortDirection: _state.activeSortDirection ?? SortDirection.desc,
        filters: _state.activeFilters,
      );

      final response = await _fetcher.fetchFirst(request);

      if (response.items.isEmpty) {
        _state = _state.copyWith(
          status: PagedListStatus.empty,
          items: [],
          hasMore: false,
          nextCursor: null,
          lastRefreshTime: response.serverTime,
        );
      } else {
        _state = _state.copyWith(
          status: PagedListStatus.loaded,
          items: response.items,
          nextCursor: response.nextCursor,
          hasMore: response.hasMore,
          totalCount: response.total,
          lastRefreshTime: response.serverTime,
        );

        if (enableOfflineCache) {
          await _fetcher.saveCached(cacheKey, response.items);
        }
      }
    } on DioException catch (e) {
      if (e.type == DioExceptionType.cancel) return;
      // 刷新失败不覆盖已有数据
      _state = _state.copyWith(status: PagedListStatus.loaded);
    } catch (_) {
      _state = _state.copyWith(status: PagedListStatus.loaded);
    }
    notifyListeners();
  }

  /// 静默刷新（不改变 loading 状态）
  Future<void> _silentRefresh() async {
    try {
      final request = PageRequest(
        pageSize: pageSize,
        sortBy: _state.activeSortBy,
        sortDirection: _state.activeSortDirection ?? SortDirection.desc,
        filters: _state.activeFilters,
      );

      final response = await _fetcher.fetchFirst(request);
      _state = _state.copyWith(
        items: response.items,
        nextCursor: response.nextCursor,
        hasMore: response.hasMore,
        totalCount: response.total,
        lastRefreshTime: response.serverTime,
      );

      if (enableOfflineCache) {
        await _fetcher.saveCached(cacheKey, response.items);
      }
      notifyListeners();
    } catch (_) {
      // 静默刷新失败，保留缓存数据
    }
  }

  /// 加载更多
  Future<void> loadMore() async {
    if (!_state.canLoadMore) return;

    _state = _state.copyWith(status: PagedListStatus.loadingMore);
    notifyListeners();

    try {
      final request = PageRequest(
        cursor: _state.nextCursor,
        pageSize: pageSize,
        sortBy: _state.activeSortBy,
        sortDirection: _state.activeSortDirection ?? SortDirection.desc,
        filters: _state.activeFilters,
      );

      final response = await _fetcher.fetchMore(request);

      final merged = _fetcher.deduplicate(_state.items, response.items);

      _state = _state.copyWith(
        status: merged.isEmpty
            ? PagedListStatus.empty
            : response.hasMore
                ? PagedListStatus.loaded
                : PagedListStatus.noMore,
        items: merged,
        nextCursor: response.nextCursor,
        hasMore: response.hasMore,
        totalCount: response.total,
      );

      // 更新离线缓存
      if (enableOfflineCache) {
        await _fetcher.saveCached(cacheKey, merged);
      }
    } on DioException catch (e) {
      if (e.type == DioExceptionType.cancel) return;
      _state = _state.copyWith(
        status: PagedListStatus.loadMoreError,
        errorMessage: _mapDioError(e),
        errorCode: 'NETWORK_ERROR',
      );
    } catch (e) {
      _state = _state.copyWith(
        status: PagedListStatus.loadMoreError,
        errorMessage: e.toString(),
        errorCode: 'UNKNOWN_ERROR',
      );
    }
    notifyListeners();
  }

  /// 更新筛选条件（防抖）
  void updateFilters(Map<String, dynamic> filters) {
    _state = _state.copyWith(activeFilters: filters);
    _debounceTimer?.cancel();
    _debounceTimer = Timer(_debounceDuration, () => _loadFirst());
  }

  /// 更新搜索关键词（防抖）
  void updateKeyword(String keyword) {
    _state = _state.copyWith(
      activeFilters: {..._state.activeFilters, 'keyword': keyword},
    );
    _debounceTimer?.cancel();
    _debounceTimer = Timer(_debounceDuration, () => _loadFirst());
  }

  /// 更新排序
  void updateSort(String sortBy, SortDirection direction) {
    _state = _state.copyWith(
      activeSortBy: sortBy,
      activeSortDirection: direction,
    );
    _loadFirst();
  }

  /// 本地更新单条数据（用于乐观更新）
  void updateItem(int index, T Function(T) updater) {
    if (index < 0 || index >= _state.items.length) return;
    final items = List<T>.from(_state.items);
    items[index] = updater(items[index]);
    _state = _state.copyWith(items: items);
    notifyListeners();
  }

  /// 本地删除单条数据
  void removeItem(int index) {
    if (index < 0 || index >= _state.items.length) return;
    final items = List<T>.from(_state.items)..removeAt(index);
    _state = _state.copyWith(
      items: items,
      totalCount: (_state.totalCount ?? 0) - 1,
    );
    if (items.isEmpty) {
      _state = _state.copyWith(status: PagedListStatus.empty);
    }
    notifyListeners();
  }

  /// 本地插入数据到顶部
  void insertAtTop(T item) {
    final items = [item, ..._state.items];
    _state = _state.copyWith(
      status: PagedListStatus.loaded,
      items: items,
      totalCount: (_state.totalCount ?? 0) + 1,
    );
    notifyListeners();
  }

  String _mapDioError(DioException e) {
    switch (e.type) {
      case DioExceptionType.connectionTimeout:
      case DioExceptionType.sendTimeout:
      case DioExceptionType.receiveTimeout:
        return '网络连接超时，请稍后重试';
      case DioExceptionType.connectionError:
        return '网络连接失败，请检查网络设置';
      default:
        return '请求失败，请稍后重试';
    }
  }

  @override
  void dispose() {
    _cancelToken?.cancel();
    _debounceTimer?.cancel();
    super.dispose();
  }
}
```

class PagedListWidget<T extends ListItem> extends StatefulWidget {
  /// 列表控制器
  final PagedListController<T> controller;

  /// 列表项构建器
  final Widget Function(BuildContext context, T item, int index) itemBuilder;

  /// 空态构建器
  final WidgetBuilder? emptyBuilder;

  /// 错误态构建器
  final Widget Function(BuildContext context, String error)? errorBuilder;

  /// 骨架屏构建器
  final WidgetBuilder? skeletonBuilder;

  /// 头部组件（筛选栏等）
  final WidgetBuilder? headerBuilder;

  /// 分隔线构建器
  final Widget Function(BuildContext context, int index)? separatorBuilder;

  /// 列表项动画
  final ListItemAnimation? animation;

  /// 布局模式
  final PagedListLayout layout;

  /// 网格模式下的列数
  final int gridCrossAxisCount;

  /// 网格模式下的间距
  final double gridSpacing;

  /// 是否启用分组（需要 item 实现 groupKey）
  final bool enableGrouping;

  /// 分组头构建器
  final Widget Function(BuildContext context, String groupKey, int count)?
      groupHeaderBuilder;

  /// 列表项预估高度（用于滚动优化）
  final double estimatedItemHeight;

  /// 是否启用下拉刷新
  final bool enablePullRefresh;

  /// 是否启用加载更多

  /// 是否启用滑动操作（左滑删除等）

  final AgeGroup? ageGroup;

  final ScrollController? scrollController;

  final EdgeInsets padding;

  const PagedListWidget({
    super.key,
    required this.controller,
    required this.itemBuilder,
    this.emptyBuilder,
    this.errorBuilder,
    this.skeletonBuilder,
    this.headerBuilder,
    this.separatorBuilder,
    this.animation,
    this.layout = PagedListLayout.list,
    this.gridCrossAxisCount = 2,
    this.gridSpacing = 12,
    this.enableGrouping = false,
    this.groupHeaderBuilder,
    this.estimatedItemHeight = 80,

    this.enableLoadMore = true,
    this.enableSwipeActions = false,
    this.swipeActions,
    this.ageGroup,

class _PagedListWidgetState<T extends ListItem>
    extends State<PagedListWidget<T>> {
  @override
  void initState() {
    super.initState();
    widget.controller.addListener(_onStateChanged);
    // 首次加载
    if (widget.controller.state.status == PagedListStatus.initial) {
      widget.controller.initialize();
    }
  }

  @override
  void dispose() {
    widget.controller.removeListener(_onStateChanged);
    super.dispose();

  void _onStateChanged() {
    if (mounted) setState(() {});
  }

  @override
  Widget build(BuildContext context) {
    final state = widget.controller.state;

    return Column(
      children: [
        // 头部（筛选栏）
        if (widget.headerBuilder != null) widget.headerBuilder!(context),

        // 主体内容
        Expanded(
          child: _buildBody(context, state),
        ),
      ],
    );
  }

  Widget _buildBody(BuildContext context, PagedListState<T> state) {
    switch (state.status) {
      case PagedListStatus.initial:
      case PagedListStatus.loading:
        return _buildSkeleton(context);

      case PagedListStatus.empty:
        return _buildEmpty(context);

      case PagedListStatus.error:
        return _buildError(context, state.errorMessage ?? '加载失败');

      case PagedListStatus.refreshing:
      case PagedListStatus.loaded:
      case PagedListStatus.noMore:
      case PagedListStatus.loadingMore:
      case PagedListStatus.loadMoreError:
        return _buildList(context, state);
    }
  }

  Widget _buildSkeleton(BuildContext context) {
    if (widget.skeletonBuilder != null) {
      return widget.skeletonBuilder!(context);
    }
    return _DefaultSkeleton(
      ageGroup: widget.ageGroup ?? AgeGroup.middleSchool,
    );
  }

  Widget _buildEmpty(BuildContext context) {
    if (widget.emptyBuilder != null) {
      return widget.emptyBuilder!(context);
    }
    return _DefaultEmptyView(ageGroup: widget.ageGroup);
  }

  Widget _buildError(BuildContext context, String error) {
    if (widget.errorBuilder != null) {
      return widget.errorBuilder!(context, error);
    }
    return _DefaultErrorView(
      message: error,
      onRetry: () => widget.controller.refresh(),
    );
  }

  Widget _buildList(BuildContext context, PagedListState<T> state) {
    Widget listWidget;

    if (widget.layout == PagedListLayout.grid) {
      listWidget = _buildGridView(context, state);
    } else if (widget.enableGrouping) {
      listWidget = _buildGroupedList(context, state);
    } else {
      listWidget = _buildListView(context, state);
    }

    // 包裹下拉刷新
    if (widget.enablePullRefresh) {
      listWidget = RefreshIndicator(
        onRefresh: widget.controller.refresh,
        child: listWidget,
      );
    }

    return listWidget;
  }

  Widget _buildListView(BuildContext context, PagedListState<T> state) {
    return ListView.separated(
      controller: widget.scrollController,
      padding: widget.padding,
      itemCount: state.items.length + 1, // +1 for footer
      separatorBuilder: widget.separatorBuilder ??
          (_, __) => const SizedBox(height: 8),
      itemBuilder: (context, index) {
        if (index >= state.items.length) {
          return _buildFooter(context, state);
        }

        final item = state.items[index];
        Widget child = widget.itemBuilder(context, item, index);

        // 滑动操作
        if (widget.enableSwipeActions && widget.swipeActions != null) {
          child = _SwipeActionWrapper(
            actions: widget.swipeActions!,
            item: item,
            child: child,
          );
        }

        // 列表项动画
        if (widget.animation != null) {
          child = _ListItemAnimationWrapper(
            animation: widget.animation!,
            index: index,
            child: child,
          );
        }

        return child;
      },
    );
  }

  Widget _buildGridView(BuildContext context, PagedListState<T> state) {
    return GridView.builder(
      controller: widget.scrollController,
      padding: widget.padding,
      gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
        crossAxisCount: widget.gridCrossAxisCount,
        mainAxisSpacing: widget.gridSpacing,
        crossAxisSpacing: widget.gridSpacing,
      ),
      itemCount: state.items.length + 1,
      itemBuilder: (context, index) {
        if (index >= state.items.length) {
          return _buildFooter(context, state);
        }
        return widget.itemBuilder(context, state.items[index], index);
      },
    );
  }

  Widget _buildGroupedList(BuildContext context, PagedListState<T> state) {
    final groups = <String, List<_GroupedItem<T>>>{};
    for (var i = 0; i < state.items.length; i++) {
      final item = state.items[i];
      final key = item.groupKey ?? 'default';
      groups.putIfAbsent(key, () => []).add(_GroupedItem(item, i));
    }

    final sections = <Widget>[];
    for (final entry in groups.entries) {
      if (widget.groupHeaderBuilder != null) {
        sections.add(
          widget.groupHeaderBuilder!(
            context,
            entry.key,
            entry.value.length,
          ),
        );
      }
      for (final grouped in entry.value) {
        sections.add(
          widget.itemBuilder(context, grouped.item, grouped.originalIndex),
        );
        if (widget.separatorBuilder != null) {
          sections.add(widget.separatorBuilder!(context, grouped.originalIndex));
        }
      }
    }
    sections.add(_buildFooter(context, state));

    return ListView(
      controller: widget.scrollController,
      padding: widget.padding,
      children: sections,
    );
  }

  Widget _buildFooter(BuildContext context, PagedListState<T> state) {
    switch (state.status) {
      case PagedListStatus.loadingMore:
        return const Padding(
          padding: EdgeInsets.symmetric(vertical: 16),
          child: Center(
            child: SizedBox(
              width: 20,
              height: 20,
              child: CircularProgressIndicator(strokeWidth: 2),
            ),
          ),
        );

      case PagedListStatus.loadMoreError:
        return Padding(
          padding: const EdgeInsets.symmetric(vertical: 16),
          child: Center(
            child: TextButton.icon(
              onPressed: widget.controller.loadMore,
              icon: const Icon(Icons.refresh, size: 16),
              label: Text(state.errorMessage ?? '加载失败，点击重试'),
            ),
          ),
        );

      case PagedListStatus.noMore:
        return Padding(
          padding: const EdgeInsets.symmetric(vertical: 16),
          child: Center(
            child: Text(
              '— 已经到底了 —',
              style: Theme.of(context).textTheme.bodySmall?.copyWith(
                    color: Theme.of(context).disabledColor,
                  ),
            ),
          ),
        );

      default:
        // 有更多数据时，自动触发加载更多
        if (state.hasMore && widget.enableLoadMore) {
          WidgetsBinding.instance.addPostFrameCallback((_) {
            widget.controller.loadMore();
          });
        }
        return const SizedBox.shrink();
    }
  }
}

class _GroupedItem<T> {
  final T item;
  final int originalIndex;
  const _GroupedItem(this.item, this.originalIndex);
}

### 7.3 乐观更新与操作反馈

```dart
/// 列表操作辅助类 — 处理乐观更新 + 服务端确认/回滚
class ListActionHelper<T extends ListItem> {
  final PagedListController<T> controller;

  ListActionHelper(this.controller);

  /// 删除操作（乐观更新）
  ///
  /// 1. 本地立即移除
  /// 2. 显示 Snackbar 提供「撤销」
  /// 3. 确认删除则调用 API
  /// 4. API 失败则回滚
  Future<void> deleteWithUndo({
    required int index,
    required Future<void> Function(T item) apiCall,
    required String undoMessage,
  }) async {
    final state = controller.state;
    if (index < 0 || index >= state.items.length) return;

    final removedItem = state.items[index];
    controller.removeItem(index);

    // 通过全局 Key 显示 Snackbar
    final scaffoldMessenger = scaffoldMessengerKey.currentState;
    scaffoldMessenger?.showSnackBar(
      SnackBar(
        content: Text(undoMessage),
        duration: const Duration(seconds: 5),
        action: SnackBarAction(
          label: '撤销',
          onPressed: () {
            controller.insertAtTop(removedItem);
          },
        ),
      ),
    );

    try {
      await apiCall(removedItem);
    } catch (e) {
      // API 失败，回滚
      controller.insertAtTop(removedItem);
      scaffoldMessenger?.showSnackBar(
        const SnackBar(content: Text('操作失败，已恢复')),
      );
    }
  }

  /// 顶部插入操作（如新建错题）
  Future<void> insertWithConfirm({
    required T item,
    required Future<void> Function(T item) apiCall,
  }) async {
    controller.insertAtTop(item);
    try {
      await apiCall(item);
    } catch (e) {
      // 插入失败，移除本地项
      final index = controller.state.items.indexOf(item);
      if (index >= 0) controller.removeItem(index);
    }
  }
}
```

```dart
/// 无限滚动触发控制器
///
/// 在距离底部 N 条记录时预加载下一页
class InfiniteScrollController {
  final PagedListController _listController;
  final int preloadThreshold; // 距底部多少条时触发预加载

  InfiniteScrollController({
    required PagedListController listController,
    this.preloadThreshold = 5,
  }) : _listController = listController;

  /// 根据滚动位置判断是否需要预加载
  void onScrolled(ScrollMetrics metrics) {
    if (!_listController.state.canLoadMore) return;

    // 方案一：基于可见项索引（需配合 itemExtent 或确切高度）
    // 这里用方案二：基于滚动百分比

    final maxScroll = metrics.maxScrollExtent;
    final currentScroll = metrics.pixels;

    if (maxScroll <= 0) return;

    // 滚动到 80% 时触发预加载
    if (currentScroll / maxScroll > 0.8) {
      _listController.loadMore();
    }
  }

  /// 绑定到 ScrollController
  void attach(ScrollController scrollController) {
    scrollController.addListener(() {
      onScrolled(scrollController.position);
    });
  }
}
```

```dart
class ListImagePreloader {

  /// 预加载即将可见的图片
  void preloadRange(
    BuildContext context,
    List<String> imageUrls,
    int visibleStart,
    int visibleEnd,
    {int buffer = 5}

      if (!precacheImage.containsKey(url)) {

          NetworkImage(url),
          context,
        );
      }
    }
  }
```

```dart
/// 离线操作队列
class OfflineActionQueue {
  final Box _box;
  static const _queueKey = 'offline_action_queue';

  OfflineActionQueue(this._box);

  /// 入队操作
  Future<void> enqueue(OfflineAction action) async {
    final queue = _getQueue()..add(action);
    await _box.put(_queueKey, jsonEncode(queue));
  }

  /// 处理队列（网络恢复时调用）
  Future<void> processQueue(ConnectivityService connectivity) async {
    if (!connectivity.isOnline) return;

    final queue = _getQueue();
    final failed = <OfflineAction>[];

    for (final action in queue) {
      try {
        await action.execute();
      } catch (e) {
        failed.add(action);
        if (_isCriticalError(e)) break; // 严重错误停止处理
      }
    }

    // 更新队列（移除成功的，保留失败的）
    await _box.put(_queueKey, jsonEncode(failed));
  }

  List<OfflineAction> _getQueue() {
    final raw = _box.get(_queueKey);
    if (raw == null) return [];
    return (jsonDecode(raw) as List)
        .map((e) => OfflineAction.fromJson(e))
        .toList();
  }
}
```
