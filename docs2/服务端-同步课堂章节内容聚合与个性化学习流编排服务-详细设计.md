# 服务端-同步课堂章节内容聚合与个性化学习流编排服务 - 详细设计

> **模块定位**: 将散落在知识点库、题库、教材资源库中的章节学习内容动态聚合，根据学生画像编排个性化学习流（预习→概念→例题→练习→复习），按需插入自适应内容（薄弱强化/拓展挑战），为客户端提供统一的章节学习内容API。
>
> **优先级**: P0 (MVP核心闭环)
> **依赖**: 知识点体系与教材映射引擎、题目与题库服务、用户学习画像与能力维度模型、学习记录与进度追踪服务、AI模型上下文管理与对话记忆引擎

### 2.1 章节内容清单 (ChapterContentManifest)

```python
class ChapterContentManifest(TypedDict):
    """章节内容清单 — 一个章节所有可聚合内容的元数据目录"""
    chapter_id: str               # 章节ID (格式: tb_{版本}_{册}_{章}_{节})
    textbook_id: str              # 教材版本ID
    subject: str                  # 学科
    grade: str                    # 年级代码
    term: Literal["上册", "下册"]
    
    # 知识点
    knowledge_points: List[KnowledgePointRef]
    
    # 学习资源
    resources: List[ContentResourceRef]
    
    # 题目池
    question_pool: QuestionPoolRef
    
    # 教学要求
    teaching_requirements: TeachingRequirements
    
    # 元数据
    total_estimated_minutes: int  # 预估学习总时长(分钟)
    difficulty_level: float       # 章节整体难度 0-1
    updated_at: str               # ISO8601


class KnowledgePointRef(TypedDict):
    """知识点引用"""
    kp_id: str
    name: str
    level: Literal["core", "important", "supplementary"]  # 核心/重要/补充
    order: int                    # 学习顺序
    prerequisites: List[str]      # 前置知识点ID
    estimated_minutes: int
    mastery_threshold: float      # 掌握度阈值 0-1


class ContentResourceRef(TypedDict):
    """学习资源引用"""
    resource_id: str
    resource_type: Literal[
        "concept_text",       # 概念讲解文本
        "example",            # 例题
        "video",              # 视频讲解
        "animation",          # 动画演示
        "image",              # 图片/图表
        "audio",              # 音频(如英语听力)
        "interactive",        # 交互式内容
        "summary",            # 知识总结
        "mind_map",           # 思维导图
    ]
    kp_ids: List[str]            # 关联知识点
    difficulty: float            # 0-1
    duration_seconds: int        # 时长(文本为阅读时长估算)
    title: str
    storage_uri: str
    format: str                  # json/markdown/mp4/webp/mp3/html


class QuestionPoolRef(TypedDict):
    """题目池引用"""
    basic_count: int              # 基础题数量
    medium_count: int             # 中等题数量
    hard_count: int               # 拔高题数量
    total_count: int
    tags: List[str]               # 题目标签


class TeachingRequirements(TypedDict):
    """教学要求"""
    cognitive_levels: List[str]   # 认知层次: 记忆/理解/应用/分析/综合/评价
    key_abilities: List[str]      # 关键能力
    exam_weight: float            # 考试占比权重
    curriculum_standard: str      # 课标要求文本
```

```python
class LearningFlowTemplate(TypedDict):
    """学习流模板 — 定义章节内学习的阶段和步骤"""
    template_id: str
    name: str
    subject: str
    grade_range: Tuple[str, str]  # 适用的年级范围
    stage_sequence: List[LearningStage]


class LearningStage(TypedDict):
    """学习阶段"""
    stage_type: Literal[
        "preview",         # 预习
        "concept",         # 概念学习
        "example",         # 例题讲解
        "guided_practice", # 引导练习
        "independent_practice",  # 独立练习
        "review",          # 复习总结
        "challenge",       # 拓展挑战
    ]
    name: str
    description: str
    estimated_minutes: int
    required: bool                # 是否必学阶段
    content_slots: List[ContentSlot]  # 内容槽位
    completion_criteria: CompletionCriteria
    order: int


class ContentSlot(TypedDict):
    """内容槽位 — 可填充具体内容的占位"""
    slot_id: str
    slot_type: Literal[
        "fixed",           # 固定内容(教材核心)
        "recommended",     # 推荐内容(基于画像)
        "adaptive",        # 自适应内容(动态插入)
        "optional",        # 可选拓展
    ]
    content_types: List[str]      # 接受的内容类型
    min_count: int
    max_count: int
    kp_filter: Optional[List[str]]  # 限定知识点范围
    difficulty_range: Tuple[float, float]


class CompletionCriteria(TypedDict):
    """阶段完成条件"""
    type: Literal[
        "time_spent",       # 学习时长达标
        "content_viewed",   # 内容全部浏览
        "practice_passed",  # 练习通过(正确率)
        "manual_confirm",   # 手动确认
    ]
```

### 3.1 章节内容聚合流程

```python
class ChapterContentAggregator:
    """章节内容聚合器 — 从多数据源聚合章节全部可学习内容"""
    
    def __init__(
        self,
        knowledge_point_service: KnowledgePointService,
        question_service: QuestionService,
        resource_service: ResourceService,
        cache: Redis,
    ):
        self.kp_service = knowledge_point_service
        self.question_service = question_service
        self.resource_service = resource_service
        self.cache = cache
    
    async def get_or_build_manifest(
        self, chapter_id: str, textbook_id: str
    ) -> ChapterContentManifest:
        """获取或构建章节内容清单(带缓存)"""
        
        # 1. 查缓存
        cache_key = f"chapter_manifest:{chapter_id}:{textbook_id}"
        cached = await self.cache.get(cache_key)
        if cached:
            manifest = json.loads(cached)
            # 校验版本哈希是否变化
            current_hash = await self._compute_version_hash(chapter_id)
            if current_hash == manifest.get("version_hash"):
                return manifest
        
        # 2. 构建清单
        manifest = await self._build_manifest(chapter_id, textbook_id)
        
        # 3. 持久化 + 缓存
        await self._save_manifest(manifest)
        await self.cache.set(
            cache_key, json.dumps(manifest, ensure_ascii=False),
            ex=3600  # 1小时缓存
        )
        
        return manifest
    
    async def _build_manifest(
        self, chapter_id: str, textbook_id: str
    ) -> ChapterContentManifest:
        """构建章节内容清单"""
        
        # 并行拉取各数据源
        kp_task = self.kp_service.get_chapter_knowledge_points(chapter_id)
        res_task = self.resource_service.get_chapter_resources(chapter_id)
        q_task = self.question_service.get_chapter_question_stats(chapter_id)
        meta_task = self.kp_service.get_chapter_metadata(chapter_id)
        
        kps, resources, question_stats, metadata = await asyncio.gather(
            kp_task, res_task, q_task, meta_task
        )
        
        # 构建知识点引用
        kp_refs = [
            KnowledgePointRef(
                kp_id=kp.id,
                name=kp.name,
                level=kp.importance_level,
                order=kp.sort_order,
                prerequisites=kp.prerequisite_ids,
                estimated_minutes=kp.estimated_learn_minutes,
                mastery_threshold=kp.mastery_threshold,
            )
            for kp in sorted(kps, key=lambda x: x.sort_order)
        ]
        
        # 构建资源引用
        res_refs = [
            ContentResourceRef(
                resource_id=r.id,
                resource_type=r.type,
                kp_ids=r.knowledge_point_ids,
                difficulty=r.difficulty,
                duration_seconds=r.duration_seconds,
                title=r.title,
                storage_uri=r.storage_uri,
                format=r.format,
            )
            for r in resources
        ]
        
        # 构建题目池引用
        pool_ref = QuestionPoolRef(
            basic_count=question_stats.get("basic", 0),
            medium_count=question_stats.get("medium", 0),
            hard_count=question_stats.get("hard", 0),
            total_count=question_stats.get("total", 0),
            tags=question_stats.get("tags", []),
        )
        
        # 计算总时长
        total_minutes = sum(kp.estimated_minutes for kp in kp_refs)
        
        return ChapterContentManifest(
            chapter_id=chapter_id,
            textbook_id=textbook_id,
            subject=metadata.subject,
            grade=metadata.grade,
            term=metadata.term,
            knowledge_points=kp_refs,
            resources=res_refs,
            question_pool=pool_ref,
            teaching_requirements=metadata.teaching_requirements,
            total_estimated_minutes=total_minutes,
            difficulty_level=metadata.difficulty_level,
            updated_at=datetime.now(timezone.utc).isoformat(),
        )
```

### 3.2 学习流编排流程

```python
class LearningFlowOrchestrator:
    """学习流编排引擎 — 根据模板和用户画像生成个性化学习流"""
    
    # 默认阶段序列
    DEFAULT_STAGE_SEQUENCE = [
        "preview",                # 预习
        "concept",                # 概念学习
        "example",                # 例题讲解
        "guided_practice",        # 引导练习
        "independent_practice",   # 独立练习
        "review",                 # 复习总结
    ]
    
    # 分龄模板策略
    AGE_STAGE_STRATEGIES = {
        "primary_low": {   # 小学低年级(1-3)
            "required_stages": ["preview", "concept", "guided_practice", "review"],
            "optional_stages": ["example", "independent_practice", "challenge"],
            "default_difficulty": 0.3,
            "max_content_per_stage": 3,
            "prefer_resource_types": ["animation", "interactive", "image", "audio"],
        },
        "primary_high": {  # 小学高年级(4-6)
            "required_stages": ["preview", "concept", "example", "guided_practice", "review"],
            "optional_stages": ["independent_practice", "challenge"],
            "default_difficulty": 0.4,
            "max_content_per_stage": 4,
            "prefer_resource_types": ["concept_text", "example", "video", "interactive"],
        },
        "junior": {        # 初中
            "required_stages": ["concept", "example", "guided_practice", "independent_practice", "review"],
            "optional_stages": ["preview", "challenge"],
            "default_difficulty": 0.5,
            "max_content_per_stage": 5,
            "prefer_resource_types": ["concept_text", "example", "summary", "mind_map"],
        },
        "senior": {        # 高中
            "required_stages": ["concept", "example", "independent_practice", "review"],
            "optional_stages": ["preview", "guided_practice", "challenge"],
            "default_difficulty": 0.6,
            "max_content_per_stage": 6,
            "prefer_resource_types": ["concept_text", "example", "summary", "mind_map"],
        },
    }
    
    def __init__(
        self,
        manifest_aggregator: ChapterContentAggregator,
        adaptive_inserter: AdaptiveInsertionEngine,
        user_profile_service: UserProfileService,
        progress_service: ProgressTrackingService,
        template_repo: LearningFlowTemplateRepository,
    ):
        self.aggregator = manifest_aggregator
        self.adaptive_inserter = adaptive_inserter
        self.profile_service = user_profile_service
        self.progress_service = progress_service
        self.template_repo = template_repo
    
    async def generate_personalized_flow(
        self,
        user_id: str,
        chapter_id: str,
        textbook_id: str,
        force_regenerate: bool = False,
    ) -> PersonalizedLearningFlow:
        """
        为用户生成个性化章节学习流
        核心入口方法
        """
        
        # 1. 检查是否已有未完成的学习流
        if not force_regenerate:
            existing = await self._get_active_flow(user_id, chapter_id)
            if existing:
                return existing
        
        # 2. 获取章节内容清单
        manifest = await self.aggregator.get_or_build_manifest(
            chapter_id, textbook_id
        )
        
        # 3. 获取用户画像和学习上下文
        profile = await self.profile_service.get_learning_profile(user_id)
        context = await self._build_learning_context(user_id, chapter_id, manifest)
        
        # 4. 选择学习流模板
        template = await self._select_template(manifest, profile)
        
        # 5. 生成基础学习流(填充内容)
        base_flow = await self._build_base_flow(template, manifest, profile, context)
        
        # 6. 自适应插入(薄弱强化/前置补救/拓展挑战)
        enhanced_flow = await self.adaptive_inserter.enhance_with_adaptives(
            base_flow, manifest, profile, context
        )
        
        # 7. 个性化排序
        sorted_flow = await self._personalized_sort(enhanced_flow, profile, context)
        
        # 8. 持久化
        flow_instance = PersonalizedLearningFlow(
            flow_id=str(uuid.uuid4()),
            user_id=user_id,
            chapter_id=chapter_id,
            template_id=template["template_id"],
            status="not_started",
            current_stage_index=0,
            current_slot_index=0,
            skipped_stages=[],
            inserted_stages=[s for s in sorted_flow["inserted_stages"]],
            difficulty_override=context.get("difficulty_override"),
            stages=sorted_flow["stages"],
            started_at=None,
            last_active_at=None,
            completed_at=None,
            total_time_spent_seconds=0,
            generated_at=datetime.now(timezone.utc).isoformat(),
            expires_at=self._compute_expiry(manifest).isoformat(),
        )
        
        await self._save_flow(flow_instance)
        return flow_instance
    
    async def _build_learning_context(
        self,
        user_id: str,
        chapter_id: str,
        manifest: ChapterContentManifest,
    ) -> dict:
        """构建学习上下文"""
        
        # 获取用户对该章节知识点的掌握度
        kp_mastery = await self.progress_service.batch_get_kp_mastery(
            user_id, [kp["kp_id"] for kp in manifest["knowledge_points"]]
        )
        
        # 获取前置章节完成情况
        prerequisite_chapters = await self._get_prerequisite_chapters(chapter_id)
        prereq_completion = await self.progress_service.batch_get_chapter_completion(
            user_id, prerequisite_chapters
        )
        
        # 获取最近学习行为(判断学习疲劳度)
        recent_activity = await self.progress_service.get_recent_learning_sessions(
            user_id, hours=2
        )
        
        # 计算知识点掌握度分布
        weak_kps = [
            kp for kp in manifest["knowledge_points"]
            if kp_mastery.get(kp["kp_id"], 0) < kp["mastery_threshold"]
        ]
        
        strong_kps = [
            kp for kp in manifest["knowledge_points"]
            if kp_mastery.get(kp["kp_id"], 0) >= 0.85
        ]
        
        # 判断是否为首次学习
        is_first_visit = all(
            kp_mastery.get(kp["kp_id"], 0) == 0
            for kp in manifest["knowledge_points"]
        )
        
        return {
            "kp_mastery": kp_mastery,
            "weak_kps": weak_kps,
            "strong_kps": strong_kps,
            "prerequisite_completion": prereq_completion,
            "has_prerequisite_gaps": any(
                v < 0.6 for v in prereq_completion.values()
            ),
            "recent_study_minutes": sum(
                s.duration_seconds for s in recent_activity
            ) / 60,
            "is_fatigued": sum(
                s.duration_seconds for s in recent_activity
            ) > 90 * 60,  # 90分钟以上视为疲劳
            "is_first_visit": is_first_visit,
            "difficulty_override": self._compute_difficulty_override(
                kp_mastery, manifest
            ),
        }
    
    def _compute_difficulty_override(
        self, kp_mastery: dict, manifest: ChapterContentManifest
    ) -> Optional[float]:
        """根据知识点掌握度计算难度覆盖值"""
        if not kp_mastery:
            return None  # 首次学习，使用默认难度
        
        avg_mastery = sum(kp_mastery.values()) / len(kp_mastery) if kp_mastery else 0
        
        # 掌握度低→降低难度, 掌握度高→提升难度
        base_difficulty = manifest["difficulty_level"]
        adjusted = base_difficulty + (avg_mastery - 0.5) * 0.3
        return max(0.1, min(0.9, adjusted))
    
    async def _build_base_flow(
        self,
        template: LearningFlowTemplate,
        manifest: ChapterContentManifest,
        profile: dict,
        context: dict,
    ) -> dict:
        """基于模板构建基础学习流并填充内容"""
        
        age_group = self._get_age_group(profile["grade"])
        strategy = self.AGE_STAGE_STRATEGIES[age_group]
        
        filled_stages = []
        
        for stage_def in template["stage_sequence"]:
            # 判断阶段是否需要跳过
            if not self._should_include_stage(stage_def, strategy, context):
                continue
            
            # 为每个内容槽位填充具体内容
            filled_contents = await self._fill_content_slots(
                stage_def["content_slots"],
                manifest,
                profile,
                context,
                strategy,
            )
            
            filled_stages.append(FilledStage(
                stage=stage_def,
                contents=filled_contents,
                progress=StageProgress(
                    status="locked" if len(filled_stages) > 0 else "available",
                    started_at=None,
                    completed_at=None,
                    time_spent_seconds=0,
                    content_viewed_count=0,
                    content_total_count=len(filled_contents),
                    practice_score=None,
                ),
                adaptive_insertions=[],
            ))
        
        # 解锁第一个阶段
        if filled_stages:
            filled_stages[0]["progress"]["status"] = "available"
        
        return {"stages": filled_stages, "inserted_stages": []}
    
    async def _fill_content_slots(
        self,
        slots: List[ContentSlot],
        manifest: ChapterContentManifest,
        profile: dict,
        context: dict,
        strategy: dict,
    ) -> List[FilledContent]:
        """为内容槽位填充具体资源"""
        
        filled = []
        difficulty = context.get("difficulty_override") or strategy["default_difficulty"]
        
        for slot in slots:
            # 筛选匹配的资源
            candidates = self._filter_resources_for_slot(
                slot, manifest["resources"], difficulty, strategy
            )
            
            # 评分排序
            scored = [
                (res, self._score_resource(res, slot, profile, context))
                for res in candidates
            ]
            scored.sort(key=lambda x: x[1], reverse=True)
            
            # 取 top-N
            count = min(slot["max_count"], strategy["max_content_per_stage"])
            for res, score in scored[:count]:
                filled.append(FilledContent(
                    slot_id=slot["slot_id"],
                    content_ref=res,
                    personalized=slot["slot_type"] in ("recommended", "adaptive"),
                    relevance_score=score,
                    order=len(filled),
                ))
        
        return filled
    
    def _score_resource(
        self,
        resource: ContentResourceRef,
        slot: ContentSlot,
        profile: dict,
        context: dict,
    ) -> float:
        """为资源打分(个性化相关性)"""
        score = 0.5  # 基础分
        
        # 偏好类型加分
        age_group = self._get_age_group(profile["grade"])
        strategy = self.AGE_STAGE_STRATEGIES[age_group]
        if resource["resource_type"] in strategy["prefer_resource_types"]:
            score += 0.2
        
        # 关联薄弱知识点加分
        weak_kp_ids = {kp["kp_id"] for kp in context.get("weak_kps", [])}
        if set(resource["kp_ids"]) & weak_kp_ids:
            score += 0.15
        
        # 难度匹配度加分
        target_diff = context.get("difficulty_override") or 0.5
        diff_distance = abs(resource["difficulty"] - target_diff)
        score += max(0, 0.15 - diff_distance * 0.3)
        
        return min(1.0, score)
```

### 3.3 自适应插入引擎

class AdaptiveInsertionEngine:
    """自适应内容插入引擎 — 根据学习画像动态插入额外学习内容"""
    
    def __init__(
        self,
        knowledge_point_service: KnowledgePointService,
        question_service: QuestionService,
        resource_service: ResourceService,
    ):
        self.kp_service = knowledge_point_service
        self.question_service = question_service
        self.resource_service = resource_service
    
    async def enhance_with_adaptives(
        self,
        base_flow: dict,
        manifest: ChapterContentManifest,
        profile: dict,
        context: dict,
    ) -> dict:
        """在基础学习流中插入自适应内容"""
        
        insertions = []
        
        # 1. 前置知识补救
        if context.get("has_prerequisite_gaps"):
            insertion = await self._create_prerequisite_remedy(
                manifest, context
            )
            if insertion:
                insertions.append(insertion)
        
        # 2. 薄弱知识点强化
        if context.get("weak_kps"):
            insertion = await self._create_weak_kp_reinforcement(
                manifest, context, profile
            )
            if insertion:
                insertions.append(insertion)
        
        # 3. 拓展挑战(针对掌握度高的学生)
        if len(context.get("strong_kps", [])) >= len(manifest["knowledge_points"]) * 0.6:
            insertion = await self._create_challenge_extension(
                manifest, context, profile
            )
            if insertion:
                insertions.append(insertion)
        
        # 将插入内容合并到学习流中
        for ins in sorted(insertions, key=lambda x: x["insert_after_stage_index"]):
            idx = ins["insert_after_stage_index"] + 1
            base_flow["stages"].insert(idx, ins["filled_stage"])
        
        base_flow["inserted_stages"] = insertions
        return base_flow
    
    async def _create_prerequisite_remedy(

    ) -> Optional[dict]:
        """创建前置知识补救阶段"""
        
        # 找出未达标的前置知识点

        for kp in manifest["knowledge_points"]:
            for prereq_id in kp["prerequisites"]:
                mastery = context["kp_mastery"].get(prereq_id, 0)
                if mastery < 0.5:
                    weak_prereqs.append(prereq_id)
        
        if not weak_prereqs:
            return None
        
        # 查找补救资源
        resources = await self.resource_service.find_remedy_resources(
            kp_ids=weak_prereqs,
            limit=5,
        )
        
        if not resources:
            return None
        
        stage = LearningStage(
            stage_type="guided_practice",
            name="前置知识复习",
            description=f"检测到 {len(weak_prereqs)} 个前置知识点需要巩固，先来复习一下吧",
            estimated_minutes=10,
            required=True,
            content_slots=[ContentSlot(
                slot_id="remedy_content",
                slot_type="adaptive",
                content_types=["concept_text", "example", "video", "interactive"],
                min_count=2,
                max_count=4,
                kp_filter=weak_prereqs,
                difficulty_range=(0.1, 0.4),
            )],
            completion_criteria=CompletionCriteria(
                type="content_viewed",
                threshold=1.0,
            ),
            order=0,
        )
        
        return {
            "reason": "prerequisite_remedy",
            "target_kp_ids": weak_prereqs,
            "insert_after_stage_index": 0,  # 在预习之后插入
            "filled_stage": FilledStage(
                stage=stage,
                contents=[

                        slot_id="remedy_content",

                progress=StageProgress(
                    status="locked",
