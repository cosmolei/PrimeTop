# 服务端-学生自主学习契约与承诺目标智能追踪引擎-详细设计

## 1. 概述

### 1.1 功能定位

学生学习契约（Learning Contract）是一种基于教育心理学"自我决定理论"（SDT）和行为经济学"承诺一致性原理"设计的自主学习激励机制。与 AI 自动生成的学习计划不同，学习契约强调**学生自主发起、自主承诺、自主追踪**，通过将学习目标转化为可量化、可追踪、可兑现的"契约"，提升学生的内在学习动机和自我管理能力。

### 1.2 核心目标

1. **激发自主学习意识**：从"要我学"到"我要学"，让学生主动设定学习承诺
2. **行为承诺约束**：利用心理学中的承诺一致性原理，通过公开承诺提高执行率
3. **数据驱动追踪**：自动从各学习模块采集进度数据，无需学生手动打卡
4. **智能预警与干预**：预测契约违约风险，提前触发鼓励性提醒
5. **家校协同透明**：将契约履行情况同步给家长，形成三方共识

### 1.3 适用范围

| 维度 | 说明 |
| --- | --- |
| 适用学段 | 小学三年级及以上（需具备基本自主意识）至高中 |
| 适用学科 | 全学科（语文、数学、英语、物理、化学、生物、历史、地理、政治） |
| 契约类型 | 日度契约、周度契约、目标契约（考前冲刺/假期计划/学期目标） |
| 关联模块 | 学习规划、每日任务、学情分析、积分商城、家长中心、通知服务 |

### 1.4 与现有模块的边界

| 模块 | 关系 | 区别 |
| --- | --- | --- |
| 学习规划服务 | 契约可作为学习规划的输入和执行约束 | 规划是 AI 生成 → 推给学生；契约是学生自主提出 → AI 辅助校准 |
| 每日任务 | 契约目标分解为每日任务条目 | 每日任务包含所有来源（系统/教师/规划），契约任务是其中的学生自驱部分 |
| 学习打卡 | 契约进度主要依靠系统数据自动采集 | 打卡是手动行为；契约进度是数据驱动的自动评估 |
| 积分商城 | 契约履行可触发积分奖励 | 积分来源多样，契约奖励是其中的"自驱型"来源 |
| 成就系统 | 契约里程碑可解锁成就徽章 | 成就系统覆盖全局；契约成就是子集 |

## 6. 智能分析与预测

### 6.1 契约合理性评估模型

创建契约时，系统基于学生历史数据评估目标合理性。

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class StudentBaseline:
    """学生基线数据"""
    avg_daily_value: float       # 过去30天日均完成量
    std_daily_value: float       # 标准差
    active_days_rate: float      # 活跃率
    best_streak_30d: int         # 近30天最长连续天数
    historical_fulfillment_rate: float  # 历史契约平均完成率
    grade_level: int             # 年级
    available_study_time: float  # 可用学习时间（分钟/日）

@dataclass
class ContractConfig:
    """契约配置"""
    target_value: float
    duration_days: int
    daily_target: float
    metric_type: str
    subjects: list

@dataclass
class FeasibilityResult:
    """可行性评估结果"""
    reasonable: bool
    confidence: float
    success_rate: float
    risk_factors: list
    suggested_target: float
    analysis: str


def evaluate_feasibility(
    baseline: StudentBaseline,
    config: ContractConfig
) -> FeasibilityResult:
    """评估契约目标可行性"""
    risk_factors = []
    
    # 1. 目标量 vs 基线水平
    ratio = config.daily_target / max(baseline.avg_daily_value, 0.1)
    if ratio > 2.0:
        risk_factors.append(f"日均目标({config.daily_target})远超当前水平({baseline.avg_daily_value:.1f})")
    elif ratio > 1.5:
        risk_factors.append(f"日均目标({config.daily_target})显著高于当前水平({baseline.avg_daily_value:.1f})")
    
    # 2. 活跃率考量
    if baseline.active_days_rate < 0.6:
        risk_factors.append(f"近期活跃率偏低({baseline.active_days_rate*100:.0f}%)，建议先提升学习频率")
    
    # 3. 可用时间
    estimated_time = config.daily_target * 3  # 假设每题3分钟
    if estimated_time > baseline.available_study_time:
        risk_factors.append(f"预计需要{estimated_time:.0f}分钟/日，超过可用学习时间")
    
    # 4. 历史完成率
    if baseline.historical_fulfillment_rate < 0.5:
        risk_factors.append("历史契约完成率较低，建议从较低目标开始")
    
    # 5. 综合成功概率预测
    success_rate = predict_success_rate(baseline, config)
    
    # 6. 推荐目标值
    suggested_target = baseline.avg_daily_value * 1.15 * config.duration_days  # 当前水平+15%
    
    reasonable = len(risk_factors) <= 1 and success_rate >= 0.5
    confidence = min(1.0, baseline.active_days_rate * 0.4 + (1 - ratio / 3) * 0.6)
    
    analysis = generate_analysis_text(baseline, config, ratio, success_rate)
    
    return FeasibilityResult(
        reasonable=reasonable,
        confidence=confidence,
        success_rate=success_rate,
        risk_factors=risk_factors,
        suggested_target=suggested_target,
        analysis=analysis
    )


def predict_success_rate(
    baseline: StudentBaseline,
    config: ContractConfig
) -> float:
    """预测契约成功概率"""
    # 基于历史数据的启发式模型
    target_ratio = config.daily_target / max(baseline.avg_daily_value, 0.1)
    
    # 目标难度系数：1.0=当前水平，1.5=有挑战，2.0=困难
    difficulty_score = max(0, 1 - (target_ratio - 1.0) / 1.5)
    
    # 活跃度系数
    activity_score = baseline.active_days_rate
    
    # 历史表现系数
    history_score = baseline.historical_fulfillment_rate if baseline.historical_fulfillment_rate > 0 else 0.6
    
    # 持续性系数（基于标准差的逆指标）
    consistency_score = max(0, 1 - baseline.std_daily_value / max(baseline.avg_daily_value, 1))
    
    # 加权融合
    success_rate = (
        difficulty_score * 0.35 +
        activity_score * 0.25 +
        history_score * 0.25 +
        consistency_score * 0.15
    )
    
    return float(np.clip(success_rate, 0.05, 0.95))


def generate_analysis_text(
    baseline: StudentBaseline,
    config: ContractConfig,
    ratio: float,
    success_rate: float
) -> str:
    """生成自然语言分析"""
    parts = []
    
    if ratio <= 1.2:
        parts.append(f"目标接近你当前的日均水平({baseline.avg_daily_value:.0f})，")
        parts.append("是比较平稳的提升计划。")
    elif ratio <= 1.5:
        parts.append(f"目标略高于当前水平({baseline.avg_daily_value:.0f})，")
        parts.append("有一定挑战性但完全可以实现。")
    else:
        parts.append(f"目标显著高于当前水平({baseline.avg_daily_value:.0f})，")
        parts.append("建议循序渐进，先从较低目标开始。")
    
    if success_rate >= 0.7:
        parts.append(f"预测成功率达{success_rate*100:.0f}%，加油！")
    elif success_rate >= 0.5:
        parts.append(f"预测成功率{success_rate*100:.0f}%，需要坚持。")
    else:
        parts.append(f"预测成功率仅{success_rate*100:.0f}%，建议调整目标。")
    
    return "".join(parts)
```

### 6.2 契约里程碑智能拆分算法

```python
def generate_milestones(
    target_value: float,
    duration_days: int,
    start_date: date,
    num_milestones: int = 4
) -> list:
    """生成里程碑拆分方案"""
    
    # 采用递增权重模型：前期目标占比低，后期递增
    # 给学生适应期，同时保留最终冲刺激情
    weights = [0.2, 0.25, 0.25, 0.3]  # 默认4个里程碑的权重
    
    if num_milestones == 3:
        weights = [0.25, 0.35, 0.4]
    elif num_milestones == 5:
        weights = [0.15, 0.2, 0.2, 0.2, 0.25]
    
    milestones = []
    cumulative = 0
    
    for i in range(num_milestones):
        cumulative += weights[i]
        milestone_target = round(target_value * cumulative, 1)
        milestone_date = start_date + timedelta(
            days=int(duration_days * cumulative) + 1
        )
        
        milestone = {
            "order": i + 1,
            "name": MILESTONE_NAMES.get(num_milestones, {}).get(i, f"阶段{i+1}"),
            "target_value": milestone_target,
            "target_date": milestone_date.isoformat(),
            "reward_points": calculate_milestone_reward(target_value, weights[i])
        }
        milestones.append(milestone)
    
    return milestones


MILESTONE_NAMES = {
    4: {0: "起步适应期", 1: "稳定提升期", 2: "深化突破期", 3: "最终冲线"},
    3: {0: "基础期", 1: "提升期", 2: "冲到期"},
    5: {0: "启动", 1: "适应", 2: "加速", 3: "巩固", 4: "冲线"}
}
```

### 6.3 违约风险实时预测模型

采用梯度提升树模型，每日为每个活跃契约计算风险分数。

**特征工程：**

| 特征 | 说明 | 计算方式 |
| --- | --- | --- |
| time_progress | 时间进度比 | daysElapsed / totalDays |
| value_progress | 价值进度比 | currentValue / targetValue |
| progress_gap | 进度差距 | value_progress - time_progress |
| recent_avg_daily | 近7天日均 | sum(last_7_days) / 7 |
| recent_active_days | 近7天活跃天数 | count(value > 0 in last_7_days) |
| current_streak | 当前连续达标天数 | 连续 is_target_met=true 的天数 |
| days_remaining | 剩余天数 | endDate - today |
| needed_daily_avg | 所需日均 | (target - current) / days_remaining |
| needed_vs_capacity | 需求vs能力 | needed_daily_avg / historical_avg_daily |
| milestone_completion | 里程碑完成率 | achieved / total |

**模型输出：** `fulfillment_probability ∈ [0, 1]`

```python
import lightgbm as lgb
import pandas as pd

class ContractRiskPredictor:
    """契约违约风险预测器"""
    
    FEATURE_COLUMNS = [
        'time_progress', 'value_progress', 'progress_gap',
        'recent_avg_daily', 'recent_active_days', 'current_streak',
        'days_remaining', 'needed_daily_avg', 'needed_vs_capacity',
        'milestone_completion'
    ]
    
    def __init__(self, model_path: str):
        self.model = lgb.Booster(model_file=model_path)
    
    def predict(self, contract_features: pd.DataFrame) -> dict:
        """预测单个或批量契约的履行概率"""
        X = contract_features[self.FEATURE_COLUMNS]
        probabilities = self.model.predict(X)
        
        results = []
        for i, prob in enumerate(probabilities):
            risk_level = self._classify_risk(prob)
            results.append({
                'contract_id': contract_features.iloc[i].get('contract_id'),
                'fulfillment_probability': float(prob),
                'risk_level': risk_level,
                'needed_daily': float(contract_features.iloc[i]['needed_daily_avg']),
                'recommendation': self._generate_recommendation(
                    risk_level, contract_features.iloc[i]
                )
            })
        return results
    
    def _classify_risk(self, prob: float) -> str:
        if prob >= 0.7:
            return 'LOW'
        elif prob >= 0.4:
            return 'MEDIUM'
        else:
            return 'HIGH'
    
    def _generate_recommendation(self, risk: str, features: pd.Series) -> str:
        if risk == 'LOW':
            return "进度良好，保持当前节奏即可！"
        elif risk == 'MEDIUM':
            gap = max(0, features['needed_daily_avg'] - features['recent_avg_daily'])
            return f"还需每日多完成{gap:.0f}题即可赶上进度，加油！"
        else:
            return f"当前进度有较大差距，建议调整目标或增加每日学习量。"
```

```java
    public void onLearningEvent(LearningEvent event, Acknowledgment ack) {
        try {
            // 1. 查询该学生所有ACTIVE状态的契约
            List<StudentContract> activeContracts = 
                contractService.findActiveContractsByStudentId(event.getStudentId());
            
            if (activeContracts.isEmpty()) {
                ack.acknowledge();
                return;
            }
            
            // 2. 对每个契约，判断事件是否匹配度量指标
            for (StudentContract contract : activeContracts) {
                BigDecimal contribution = calculateContribution(contract, event);
                if (contribution.compareTo(BigDecimal.ZERO) > 0) {
                    // 3. 更新契约进度
                    updateContractProgress(contract, event, contribution);
                    
                    // 4. 检查里程碑和日度达标
                    checkMilestones(contract);
                    checkDailyTarget(contract);
                    
                    // 5. 检查是否提前完成
                    if (isContractCompleted(contract)) {
                        handleEarlyCompletion(contract);
                    }
                }
            }
            
            ack.acknowledge();
        } catch (Exception e) {
            log.error("处理契约进度事件失败, event={}", event, e);
            // 重试机制：消息重投最多3次，超过则进入死信队列
            ack.acknowledge();
            deadLetterService.sendIfRetriesExceeded(event, e);
        }
    }
    
    /**
     * 计算事件对契约的贡献值
     */
    private BigDecimal calculateContribution(StudentContract contract, LearningEvent event) {
        MetricConfig config = contract.getMetricConfigObject();
        
        // 1. 检查事件类型是否匹配
        if (!isEventTypeMatched(config.getMetricType(), event.getEventType())) {
            return BigDecimal.ZERO;
        }
        
        // 2. 检查学科是否在范围内
        if (config.getSubjects() != null && !config.getSubjects().isEmpty()) {
            if (!config.getSubjects().contains(event.getSubject())) {
                return BigDecimal.ZERO;
            }
        }
        
        // 3. 检查排除来源
        if (config.getExcludeSources() != null 
            && config.getExcludeSources().contains(event.getSource())) {
            return BigDecimal.ZERO;
        }
        
        // 4. 根据度量类型计算贡献值
        switch (config.getMetricType()) {
            case QUESTION_COUNT:
                if (config.isCountCorrectOnly() && !event.isCorrect()) {
                    return BigDecimal.ZERO;
                }
                return BigDecimal.ONE;  // 每题贡献1
                
            case STUDY_TIME:
                return BigDecimal.valueOf(event.getDurationMinutes());
                
            case ACCURACY:
                // 正确率提升贡献，由批量任务计算
                return BigDecimal.ZERO;
                
            case MASTERY:
                // 掌握度提升贡献，由批处理任务计算
                return BigDecimal.ZERO;
                
            case MISTAKE_REVIEW:
                return BigDecimal.ONE;
                
            default:
                return BigDecimal.ZERO;
        }
    }
    
    /**
     * 更新契约进度（Redis + MySQL双写）
     */
    private void updateContractProgress(
        StudentContract contract,
        LearningEvent event,
        BigDecimal contribution
    ) {
        String progressKey = "contract:progress:" + contract.getId();
        String dailyKey = progressKey + ":daily:" + LocalDate.now();
        
        // Redis 实时更新
        redisTemplate.opsForValue().increment(
            progressKey + ":current", 
            contribution.doubleValue()
        );
        redisTemplate.opsForValue().increment(
            dailyKey,
            contribution.doubleValue()
        );
        redisTemplate.expire(dailyKey, 7, TimeUnit.DAYS);
        
        // 记录事件流水
        ContractEventLog eventLog = ContractEventLog.builder()
            .contractId(contract.getId())
            .studentId(contract.getStudentId())
            .eventType(event.getEventType())
            .eventSource(event.getSource())
            .sourceId(event.getSourceId())
            .contributionValue(contribution)
            .contributionDetail(event.getDetail())
            .eventTime(event.getEventTime())
            .build();
        eventLogMapper.insert(eventLog);
        
        // 异步更新MySQL（批量优化）
        progressBuffer.add(new ProgressUpdateTask(
            contract.getId(), 
            contract.getStudentId(), 
            LocalDate.now(), 
            contribution
        ));
    }
}
```

```java
/**
 * 单个契约结算
 */
@Transactional
public void settleContract(StudentContract contract) {
    // 1. 计算最终履行率
    BigDecimal targetValue = contract.getTargetValue();
    BigDecimal actualValue = contract.getCurrentValue();
    BigDecimal fulfillmentRate = actualValue
        .divide(targetValue, 4, RoundingMode.HALF_UP)
        .multiply(BigDecimal.valueOf(100))
        .setScale(2, RoundingMode.HALF_UP);
    
    // 2. 判断是否履行成功（>=80%视为成功）
    boolean isFulfilled = fulfillmentRate.compareTo(BigDecimal.valueOf(80)) >= 0;
    
    // 3. 计算奖励
    int rewardPoints = 0;
    if (isFulfilled) {
        rewardPoints = contract.getRewardPoints();
        
        // 超额完成奖励
        if (fulfillmentRate.compareTo(BigDecimal.valueOf(100)) > 0) {
            BigDecimal bonus = fulfillmentRate
                .subtract(BigDecimal.valueOf(100))
                .divide(BigDecimal.valueOf(100), 2, RoundingMode.HALF_UP)
                .multiply(BigDecimal.valueOf(contract.getRewardPoints()))
                .multiply(BigDecimal.valueOf(0.2))  // 超额部分按20%奖励
                .setScale(0, RoundingMode.DOWN);
            rewardPoints += bonus.intValue();
        }
        
        // 发放积分
        pointsService.grantPoints(
            contract.getStudentId(),
            rewardPoints,
            PointSource.CONTRACT_FULFILLMENT,
            String.format("契约\"%s\"履行奖励", contract.getTitle())
        );
        
        // 解锁成就
        achievementService.checkAndUnlock(
            contract.getStudentId(),
            AchievementType.CONTRACT_MASTER,
            Map.of("contractId", contract.getId(), 
                   "fulfillmentRate", fulfillmentRate)
        );
    }
    
    // 4. 生成AI评语
    String aiComment = generateCompletionComment(contract, fulfillmentRate, isFulfilled);
    
    // 5. 写入结算记录
    ContractFulfillment fulfillment = ContractFulfillment.builder()
        .contractId(contract.getId())
        .studentId(contract.getStudentId())
        .totalTarget(targetValue)
        .totalAchieved(actualValue)
        .fulfillmentRate(fulfillmentRate)
        .activeDays(countActiveDays(contract.getId()))
        .totalDays(calculateTotalDays(contract))
        .bestStreak(findBestStreak(contract.getId()))
        .milestoneAchieved(countAchievedMilestones(contract.getId()))
        .milestoneTotal(countTotalMilestones(contract.getId()))
        .rewardGranted(rewardPoints)
        .completionNote(aiComment)
        .settledAt(LocalDateTime.now())
        .build();
    fulfillmentMapper.insert(fulfillment);
    
    // 6. 更新契约状态
    contract.setStatus(ContractStatus.COMPLETED);
    contract.setFulfillmentRate(fulfillmentRate);
    contract.setCompletedAt(LocalDateTime.now());
    contractMapper.updateById(contract);
    
    // 7. 通知学生
    notificationService.send(contract.getStudentId(), Notification.builder()
        .type(isFulfilled ? "CONTRACT_FULFILLED" : "CONTRACT_ENDED")
        .title(isFulfilled ? "🎉 契约目标达成！" : "契约周期已结束")
        .content(aiComment)
        .data(Map.of("contractId", contract.getId()))
        .build());
    
    // 8. 通知家长（如果公开）
    if (contract.getIsPublic() == 1) {
        parentService.notifyParents(contract.getStudentId(), ParentNotification.builder()
            .type("CHILD_CONTRACT_SETTLED")
            .title(String.format("孩子的契约\"%s\"已结算", contract.getTitle()))
            .content(String.format("履行率：%.1f%%。%s", 
                fulfillmentRate.doubleValue(), aiComment))
            .build());
    }
}
```

```java
    @Scheduled(cron = "0 */15 * * * ?")  // 每15分钟检查一次

        int currentHour = now.getHour();

            if (todayValue.compareTo(BigDecimal.valueOf(contract.getDailyTarget())) >= 0) {
                continue;  // 今日已达标，跳过提醒
            }
            
            // 检查今日是否已发过提醒
            String reminderKey = "contract:reminder:" + contract.getId() + ":" + LocalDate.now();
            if (Boolean.TRUE.equals(redisTemplate.hasKey(reminderKey))) {
                continue;  // 今日已提醒过
            }

            String message = buildReminderMessage(contract, todayValue);
            notificationService.send(contract.getStudentId(), 
                Notification.builder()
                    .type("CONTRACT_DAILY_REMINDER")
                    .title("📋 学习契约提醒")
                    .content(message)
                    .build());
            
            // 标记今日已提醒（TTL到当天结束）
            redisTemplate.opsForValue().set(reminderKey, "1", 
                Duration.ofSeconds(getSecondsUntilMidnight()));
        }
    }
    
    private String buildReminderMessage(StudentContract contract, BigDecimal todayValue) {
        BigDecimal dailyTarget = BigDecimal.valueOf(contract.getDailyTarget());
        BigDecimal remaining = dailyTarget.subtract(todayValue);
        
        if (todayValue.compareTo(BigDecimal.ZERO) == 0) {
            return String.format("今天还没有开始「%s」哦，目标：%s题，开始吧！",
                contract.getTitle(), dailyTarget.intValue());
        } else if (remaining.compareTo(BigDecimal.ZERO) > 0) {
            return String.format("「%s」今日还差%s题就达标了，继续加油！",
                contract.getTitle(), remaining.intValue());
        } else {
            return "今日目标已完成，棒极了！";
        }
    }
```

| 契约模板管理 | 运营管理员 | 后台管理 |

### 12.2 数据隐私

- 契约数据属于学生个人学习数据，按PIPL要求保护
- 契约分享到班级圈时，仅展示标题和进度百分比，不展示具体答题数据
- 家长查看时，展示进度概览但不展示学生与AI的对话内容
- 契约数据在账号注销时按数据导出与注销流程处理

### 12.3 反作弊

```java
/**
 * 检测异常进度增长（刷数据嫌疑）
 */
@Component
public class ContractAntiCheatChecker {
    
    /**
     * 每日检查：如果某契约单日进度增量异常高
     */
    @Scheduled(cron = "0 0 2 * * ?")
    public void checkAbnormalProgress() {
        LocalDate yesterday = LocalDate.now().minusDays(1);
        List<ContractProgressDaily> records = progressMapper
            .findByDateWithAnomalies(yesterday);
        
        for (ContractProgressDaily record : records) {
            // 检查1：日增量超过历史均值的5倍
            BigDecimal avgDaily = getHistoricalAvgDaily(record.getContractId(), 30);
            if (record.getDailyValue().compareTo(avgDaily.multiply(BigDecimal.valueOf(5))) > 0) {
                flagAnomaly(record, "DAILY_VALUE_ANOMALY", 
                    String.format("日增量%s为均值%s的%.1f倍",
                        record.getDailyValue(), avgDaily,
                        record.getDailyValue().divide(avgDaily, 2, RoundingMode.HALF_UP).doubleValue()));
            }
            
            // 检查2：短时间内大量答题事件
            int eventCount = eventLogMapper.countByContractAndDate(
                record.getContractId(), yesterday);
            if (eventCount > 500) {
                flagAnomaly(record, "EVENT_COUNT_ANOMALY",
                    String.format("单日事件数%d超过阈值", eventCount));
            }
        }
    }
}
```

```dart
// Flutter 示例：契约数据本地缓存
class ContractRepository {
  final ApiClient _api;
  final CacheStore _cache;
  
  /// 获取活跃契约列表（优先缓存）
  Future<List<Contract>> getActiveContracts() async {
    final cacheKey = 'contracts_active';
    final cached = await _cache.read<List>(cacheKey);
    
    if (cached != null) {
      // 后台静默刷新
      _refreshInBackground(cacheKey);
      return cached.map((j) => Contract.fromJson(j)).toList();
    }
    
    final response = await _api.get('/contracts?status=ACTIVE');
    await _cache.write(cacheKey, response.data['list'], ttl: Duration(minutes: 10));
    return (response.data['list'] as List)
        .map((j) => Contract.fromJson(j))
        .toList();
  }
  
  /// 获取契约详情（含进度实时数据）
  Future<Contract> getContractDetail(int id) async {
    final response = await _api.get('/contracts/$id');
    return Contract.fromJson(response.data);
  }
}
```
