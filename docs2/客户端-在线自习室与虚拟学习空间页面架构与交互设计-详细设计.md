## 1. 概述

### 1.1 文档目的

本文档为 PrimeTop 客户端「在线自习室」功能提供完整的页面架构、交互流程、组件设计、状态管理和代码实现规范，使前端开发人员能够直接依据本文档进行编码实现。

### 1.2 功能定位

为学生提供「云端自习室」——多名学生以实时同步方式聚集在虚拟学习空间中，各自独立学习但互相可见学习状态，通过社交陪伴感和群体专注氛围提升学习动力。

### 1.3 核心用户流程

```
发现自习室 → 加入自习室 → 开始专注计时 → 学习中（可见他人状态）→ 结束专注 → 查看总结 → 离开/继续
     ↑                                                              ↓
     ←─────────── 创建自习室 ←── 没有合适的？ ←───────────────────────┘
```

### 1.4 页面路由总览

| 路由路径 | 页面名称 | 说明 |
| --- | --- | --- |
| `/study-room` | 自习室发现页 | 自习室列表、搜索、创建入口 |
| `/study-room/discover` | 热门自习室 | 按热门/学科/学段浏览 |
| `/study-room/create` | 创建自习室页 | 填写自习室信息表单 |

| `/study-room/my-stats` | 我的自习室统计 | 个人维度数据看板 |

### 3.3 状态管理

采用 Riverpod StateNotifier 模式：

```dart
/// 自习室发现页状态
@immutable
class StudyRoomDiscoverState {
  final AsyncValue<List<StudyRoomListItem>> rooms;
  final StudyRoomFilter filter;
  final bool hasMore;
  final int currentPage;
  final String? searchKeyword;

  const StudyRoomDiscoverState({
    required this.rooms,
    required this.filter,
    this.hasMore = true,
    this.currentPage = 1,
    this.searchKeyword,
  });

  StudyRoomDiscoverState copyWith({...});
}

/// 发现页控制器
final studyRoomDiscoverProvider =
    StateNotifierProvider.autoDispose<StudyRoomDiscoverNotifier,
        StudyRoomDiscoverState>((ref) {
  return StudyRoomDiscoverNotifier(ref.read);
});

class StudyRoomDiscoverNotifier extends StateNotifier<StudyRoomDiscoverState> {
  final Reader _read;

  StudyRoomDiscoverNotifier(this._read)
      : super(StudyRoomDiscoverState(
          rooms: const AsyncValue.loading(),
          filter: StudyRoomFilter(
            sortBy: 'hot',
            page: 1,
            size: 20,
          ),
        )) {
    _loadRooms();
  }

  /// 加载自习室列表
  Future<void> _loadRooms() async {
    try {
      final result = await _read(studyRoomRepositoryProvider)
          .discoverRooms(state.filter);
      state = state.copyWith(
        rooms: AsyncValue.data(result.items),
        hasMore: result.hasMore,
        currentPage: result.page,
      );
    } catch (e, st) {
      state = state.copyWith(rooms: AsyncValue.error(e, st));
    }
  }

  /// 切换排序
  Future<void> changeSort(String sortBy) async {
    state = state.copyWith(
      filter: state.filter.copyWith(sortBy: sortBy, page: 1),
      rooms: const AsyncValue.loading(),
    );
    _loadRooms();
  }

  /// 设置筛选
  Future<void> applyFilter(StudyRoomFilter newFilter) async {
    state = state.copyWith(
      filter: newFilter.copyWith(page: 1),
      rooms: const AsyncValue.loading(),
    );
    _loadRooms();
  }

  /// 搜索
  Future<void> search(String keyword) async {
    state = state.copyWith(
      searchKeyword: keyword,
      filter: state.filter.copyWith(page: 1),
      rooms: const AsyncValue.loading(),
    );
    _loadRooms();
  }

  /// 加载更多
  Future<void> loadMore() async {
    if (!state.hasMore) return;
    final nextFilter = state.filter.copyWith(page: state.currentPage + 1);
    try {
      final result = await _read(studyRoomRepositoryProvider)
          .discoverRooms(nextFilter);
      final current = state.rooms.valueOrNull ?? [];
      state = state.copyWith(
        rooms: AsyncValue.data([...current, ...result.items]),
        hasMore: result.hasMore,
        currentPage: result.page,
        filter: nextFilter,
      );
    } catch (e) {
      // 静默失败，保留已有数据
      state = state.copyWith(hasMore: false);
    }
  }

  /// 下拉刷新
  Future<void> refresh() async {
    state = state.copyWith(
      filter: state.filter.copyWith(page: 1),
      rooms: const AsyncValue.loading(),
    );
    _loadRooms();
  }
}
```

### 3.4 页面 Widget 实现

```dart
class SelfStudyRoomDiscoverPage extends ConsumerStatefulWidget {
  const SelfStudyRoomDiscoverPage({super.key});

  @override
  ConsumerState<SelfStudyRoomDiscoverPage> createState() =>
      _SelfStudyRoomDiscoverPageState();
}

class _SelfStudyRoomDiscoverPageState
    extends ConsumerState<SelfStudyRoomDiscoverPage>
    with SingleTickerProviderStateMixin {
  late TabController _tabController;
  final ScrollController _scrollController = ScrollController();

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: 4, vsync: this);
    _tabController.addListener(_onTabChanged);
    _scrollController.addListener(_onScroll);
  }

  void _onTabChanged() {
    if (_tabController.indexIsChanging) return;
    final sorts = ['hot', 'new', 'focus', 'official'];
    ref.read(studyRoomDiscoverProvider.notifier).changeSort(sorts[_tabController.index]);
  }

  void _onScroll() {
    if (_scrollController.position.pixels >=
        _scrollController.position.maxScrollExtent - 200) {
      ref.read(studyRoomDiscoverProvider.notifier).loadMore();
    }
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(studyRoomDiscoverProvider);

    return Scaffold(
      appBar: AppBar(
        title: const Text('自习室'),
        actions: [
          IconButton(
            icon: const Icon(Icons.search),
            onPressed: () => _showSearchDialog(context),
          ),
          IconButton(
            icon: const Icon(Icons.add_circle_outline),
            onPressed: () => context.push('/study-room/create'),
          ),
        ],
        bottom: TabBar(
          controller: _tabController,
          tabs: const [
            Tab(text: '🔥热门'),
            Tab(text: '🆕最新'),
            Tab(text: '⏱专注榜'),
            Tab(text: '🎓官方'),
          ],
        ),
      ),
      body: Column(
        children: [
          // 筛选行
          _buildFilterRow(),
          // 列表
          Expanded(
            child: state.rooms.when(
              data: (rooms) => RefreshIndicator(
                onRefresh: () =>
                    ref.read(studyRoomDiscoverProvider.notifier).refresh(),
                child: rooms.isEmpty
                    ? _buildEmptyState()
                    : ListView.builder(
                        controller: _scrollController,
                        padding: const EdgeInsets.all(12),
                        itemCount: rooms.length + (state.hasMore ? 1 : 0),
                        itemBuilder: (context, index) {
                          if (index >= rooms.length) {
                            return _buildLoadingMore();
                          }
                          return _StudyRoomCard(
                            room: rooms[index],
                            onTap: () => _joinRoom(context, rooms[index]),
                          );
                        },
                      ),
              ),
              loading: () => const Center(child: CircularProgressIndicator()),
              error: (e, _) => _buildErrorState(e),
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildFilterRow() {
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
      child: Wrap(
        spacing: 8,
        children: [
          FilterChip(
            label: Text(state.filter.subjectId != null
                ? _subjectName(state.filter.subjectId!)
                : '学科'),
            onSelected: (_) => _showSubjectPicker(),
          ),
          FilterChip(
            label: Text(state.filter.gradeLevel != null
                ? _gradeName(state.filter.gradeLevel!)
                : '学段'),
            onSelected: (_) => _showGradePicker(),
          ),
          FilterChip(
            label: Text(state.filter.focusMode != null
                ? (state.filter.focusMode == 1 ? '严格模式' : '自由模式')
                : '专注模式'),
            onSelected: (_) => _showFocusModePicker(),
          ),
        ],
      ),
    );
  }

  /// 加入自习室
  Future<void> _joinRoom(BuildContext context, StudyRoomListItem room) async {
    if (room.isLocked && room.roomType != 3) {
      // 私密房间，需要输入密码
      final password = await _showPasswordDialog(context);
      if (password == null) return;
      _navigateToRoom(context, room.roomCode, password);
    } else {
      _navigateToRoom(context, room.roomCode, null);
    }
  }

  void _navigateToRoom(BuildContext context, String roomCode, String? password) {
    context.push('/study-room/$roomCode', extra: password != null 
        ? {'password': password} : null);
  }
}
```

```dart
/// 计时器状态
@immutable
class TimerState {
  final TimerMode mode;          // stopwatch / pomodoro / countdown
  final TimerPhase phase;        // idle / focusing / resting / completed
  final int elapsedSec;          // 已经过去的秒数
  final int targetSec;           // 目标秒数
  final int totalFocusSec;       // 本次会话累计专注时长
  final int pomodoroCount;       // 完成的番茄钟数量
  final DateTime? startTime;     // 开始时间

  const TimerState({
    this.mode = TimerMode.pomodoro,
    this.phase = TimerPhase.idle,
    this.elapsedSec = 0,
    this.targetSec = 1500,
    this.totalFocusSec = 0,
    this.pomodoroCount = 0,
    this.startTime,
  });

  const TimerState.initial()
      : mode = TimerMode.pomodoro,
        phase = TimerPhase.idle,
        elapsedSec = 0,
        targetSec = 1500,
        totalFocusSec = 0,
        pomodoroCount = 0,
        startTime = null;

  /// 剩余秒数（番茄钟/倒计时模式）
  int get remainingSec =>
      (targetSec - elapsedSec).clamp(0, targetSec);

  /// 进度 (0.0 ~ 1.0)
  double get progress =>
      targetSec > 0 ? (elapsedSec / targetSec).clamp(0.0, 1.0) : 0.0;

  bool get isRunning =>
      phase == TimerPhase.focusing || phase == TimerPhase.resting;
}

enum TimerPhase { idle, focusing, resting, paused, completed }
enum StudyRoomPageMode { normal, focus, minimized }
enum ConnectionStatus { disconnected, connecting, connected, reconnecting }
```

### 5.3 自习室控制器（核心业务逻辑）

```dart
/// 自习室页面主控制器
final studyRoomControllerProvider = StateNotifierProvider.autoDispose
    .family<StudyRoomController, StudyRoomState, String>(
  (ref, roomCode) => StudyRoomController(ref.read, roomCode),
);

class StudyRoomController extends StateNotifier<StudyRoomState> {
  final Reader _read;
  final String _roomCode;

  StudyRoomConnection? _connection;
  Timer? _localTimer;          // 本地秒级计时器
  StreamSubscription? _eventSub;
  DateTime? _pausedAt;

  StudyRoomController(this._read, this._roomCode)
      : super(const StudyRoomState()) {
    _init();
  }

  // ═══════════════════════════════════════════
  //  初始化与连接
  // ═══════════════════════════════════════════

  Future<void> _init() async {
    try {
      // 1. 调用 REST API 获取 WS 连接信息
      final joinResult = await _read(studyRoomRepositoryProvider)
          .joinRoom(_roomCode);

      state = state.copyWith(
        roomInfo: joinResult.roomInfo,
        participants: joinResult.participants,
        isHost: joinResult.roomInfo.hostUserId ==
                _read(currentUserIdProvider),
        connectionStatus: ConnectionStatus.connecting,
      );

      // 2. 建立 WebSocket 连接
      _connection = StudyRoomConnection(
        wsUrl: joinResult.wsUrl,
        sessionId: joinResult.sessionId,
        onConnected: _onConnected,
        onDisconnected: _onDisconnected,
        onError: _onError,
      );

      _eventSub = _connection!.eventStream.listen(_handleServerMessage);
      await _connection!.connect();

    } catch (e) {
      state = state.copyWith(
        errorMessage: '加入自习室失败: ${_friendlyError(e)}',
        connectionStatus: ConnectionStatus.disconnected,
      );
    }
  }

  void _onConnected() {
    state = state.copyWith(connectionStatus: ConnectionStatus.connected);
  }

  void _onDisconnected() {
    state = state.copyWith(connectionStatus: ConnectionStatus.reconnecting);
    _scheduleReconnect();
  }

  void _onError(String error) {
    state = state.copyWith(errorMessage: error);
  }

  /// 指数退避重连
  int _reconnectAttempts = 0;
  Future<void> _scheduleReconnect() async {
    _reconnectAttempts++;
    final delay = Duration(seconds: 
        (1 << _reconnectAttempts).clamp(1, 16));

    await Future.delayed(delay);

    if (state.connectionStatus == ConnectionStatus.connected) return;

    try {
      await _connection?.reconnect();
      _reconnectAttempts = 0;
    } catch (_) {
      if (_reconnectAttempts < 5) _scheduleReconnect();
    }
  }

  // ═══════════════════════════════════════════
  //  服务端消息处理
  // ═══════════════════════════════════════════

  void _handleServerMessage(ServerMessage message) {
    message.when(
      joinResponse: _onJoinResponse,
      stateSync: _onStateSync,
      participantUpdate: _onParticipantUpdate,
      timerSync: _onTimerSync,
      chatBroadcast: _onChatMessage,
      roomEvent: _onRoomEvent,
      error: _onServerError,
      sessionSummary: _onSessionSummary,
    );
  }

  void _onJoinResponse(JoinRoomResponse resp) {
    state = state.copyWith(
      roomInfo: resp.roomInfo,
      participants: resp.participants,
      connectionStatus: ConnectionStatus.connected,
    );
  }

  void _onStateSync(RoomStateSync sync) {
    // 更新房间聚合状态（在线人数等）
    if (state.roomInfo != null) {
      state = state.copyWith(
        roomInfo: state.roomInfo!.copyWith(
          onlineCount: sync.onlineCount,
          focusCount: sync.focusCount,
          restCount: sync.restCount,
        ),
      );
    }
  }

  void _onParticipantUpdate(ParticipantUpdate update) {
    final participants = List<ParticipantInfo>.from(state.participants);
    final index = participants.indexWhere((p) => p.userId == update.info.userId);

    switch (update.type) {
      case UpdateType.join:
        if (index == -1) participants.add(update.info);
        break;
      case UpdateType.leave:
        participants.removeWhere((p) => p.userId == update.info.userId);
        break;
      case UpdateType.statusChange:
      case UpdateType.timerChange:
        if (index >= 0) participants[index] = update.info;
        break;
    }

    state = state.copyWith(participants: participants);
  }

  void _onTimerSync(TimerSync sync) {
    // 更新其他用户的计时器状态（在宫格中展示）
    final participants = List<ParticipantInfo>.from(state.participants);
    final index = participants.indexWhere((p) => p.userId == sync.userId);
    if (index >= 0) {
      participants[index] = participants[index].copyWith(
        timerMode: sync.mode,
        elapsedSec: sync.elapsedSec,
        targetSec: sync.targetSec,
      );
      state = state.copyWith(participants: participants);
    }

    // 如果是自己，同步本地计时器
    if (sync.userId == _read(currentUserIdProvider)) {
      _syncLocalTimer(sync);
    }
  }

  void _onChatMessage(ChatBroadcast chat) {
    final msg = ChatMessage(
      userId: chat.userId,
      nickName: chat.nickName,
      avatarUrl: chat.avatarUrl,
      content: chat.content,
      type: chat.type,
      timestamp: DateTime.fromMillisecondsSinceEpoch(chat.timestamp),
      isSystem: false,
    );
    state = state.copyWith(
      chatMessages: state.chatMessages.add(msg),
    );
  }

  void _onRoomEvent(RoomEventNotification event) {
    // 处理房间事件
    switch (event.event) {
      case EventType.roomLocked:
        state = state.copyWith(
          roomInfo: state.roomInfo?.copyWith(isLocked: true),
        );
        _addSystemMessage('房间已锁定');
        break;
      case EventType.userKicked:
        if (event.fromUser == _read(currentUserIdProvider)) {
          state = state.copyWith(errorMessage: '你已被主持人移出自习室');
          _cleanupAndExit();
        }
        break;
      case EventType.focusModeOn:
        state = state.copyWith(
          roomInfo: state.roomInfo?.copyWith(focusMode: 1),
          pageMode: StudyRoomPageMode.focus,
        );
        _addSystemMessage('主持人开启了专注模式，聊天已关闭');
        break;
      case EventType.pomodoroPhase:
        _addSystemMessage('集体休息时间到了，放松一下吧 🧘');
        break;
      case EventType.roomEnding:
        _addSystemMessage('自习室即将结束，请准备保存学习记录');
        break;
      default:
        break;
    }
  }

  void _onSessionSummary(SessionSummary summary) {
    state = state.copyWith(
      lastSummary: summary,
      showSummary: true,
    );
  }

  void _onServerError(ErrorResponse error) {
    state = state.copyWith(
      errorMessage: _mapErrorCode(error.code),
    );
  }

  // ═══════════════════════════════════════════
  //  计时器控制
  // ═══════════════════════════════════════════

  /// 开始专注
  Future<void> startFocus({int? subjectId}) async {
    final sid = subjectId ?? state.selectedSubjectId;
    final targetSec = _getTargetSec();

    // 发送 WS 指令
    _connection?.send(TimerAction(
      roomId: state.roomInfo!.roomId,
      op: TimerOp.start,
      mode: state.myTimer.mode,
      targetSec: targetSec,
      subjectId: sid,
    ));

    // 本地立即响应（乐观更新）
    state = state.copyWith(
      myTimer: state.myTimer.copyWith(
        phase: TimerPhase.focusing,
        startTime: DateTime.now(),
        targetSec: targetSec,
        elapsedSec: 0,
      ),
    );
    _startLocalTimer();
  }

  /// 暂停
  Future<void> pauseTimer() async {
    _connection?.send(TimerAction(
      roomId: state.roomInfo!.roomId,
      op: TimerOp.pause,
      mode: state.myTimer.mode,
    ));

    _stopLocalTimer();
    _pausedAt = DateTime.now();
    state = state.copyWith(
      myTimer: state.myTimer.copyWith(phase: TimerPhase.paused),
    );
  }

  /// 恢复
  Future<void> resumeTimer() async {
    _connection?.send(TimerAction(
      roomId: state.roomInfo!.roomId,
      op: TimerOp.resume,
      mode: state.myTimer.mode,
    ));

    _startLocalTimer();
    state = state.copyWith(
      myTimer: state.myTimer.copyWith(phase: TimerPhase.focusing),
    );
  }

  /// 切换计时模式
  Future<void> switchMode(TimerMode mode) async {
    _connection?.send(TimerAction(
      roomId: state.roomInfo!.roomId,
      op: TimerOp.switchMode,
      mode: mode,
    ));

    state = state.copyWith(
      myTimer: state.myTimer.copyWith(
        mode: mode,
        phase: TimerPhase.idle,
        elapsedSec: 0,
        targetSec: _defaultTarget(mode),
      ),
    );
    _stopLocalTimer();
  }

  /// 本地秒计时器（乐观更新，与服务端定时校正）
  void _startLocalTimer() {
    _localTimer?.cancel();
    _localTimer = Timer.periodic(const Duration(seconds: 1), (_) {
      final timer = state.myTimer;
      if (!timer.isRunning) {
        _localTimer?.cancel();
        return;
      }

      final newElapsed = timer.elapsedSec + 1;

      // 番茄钟/倒计时到达目标
      if (timer.mode != TimerMode.stopwatch &&
          newElapsed >= timer.targetSec) {
        _onTimerComplete();
        return;
      }

      state = state.copyWith(
        myTimer: timer.copyWith(elapsedSec: newElapsed),
      );
    });
  }

  void _stopLocalTimer() {
    _localTimer?.cancel();
    _localTimer = null;
  }

  /// 计时完成处理
  void _onTimerComplete() {
    _stopLocalTimer();
    final timer = state.myTimer;

    if (timer.mode == TimerMode.pomodoro) {
      // 番茄钟完成，进入休息阶段
      state = state.copyWith(
        myTimer: timer.copyWith(
          phase: TimerPhase.resting,
          elapsedSec: 0,
          targetSec: 300, // 5分钟休息
          pomodoroCount: timer.pomodoroCount + 1,
          totalFocusSec: timer.totalFocusSec + timer.targetSec,
        ),
      );
      _addSystemMessage('番茄钟完成！🍅 休息5分钟');
      _startLocalTimer(); // 开始休息计时
    } else {
      state = state.copyWith(
        myTimer: timer.copyWith(
          phase: TimerPhase.completed,
          totalFocusSec: timer.totalFocusSec + timer.targetSec,
        ),
      );
    }

    // 通知服务端
    _connection?.send(TimerAction(
      roomId: state.roomInfo!.roomId,
      op: TimerOp.complete,
      mode: timer.mode,
    ));
  }

  /// 服务端计时器同步（校正本地偏差）
  void _syncLocalTimer(TimerSync sync) {
    final current = state.myTimer;
    final corrected = current.copyWith(
      elapsedSec: sync.elapsedSec,
      targetSec: sync.targetSec,
    );

    // 仅在偏差超过3秒时校正
    if ((sync.elapsedSec - current.elapsedSec).abs() > 3) {
      state = state.copyWith(myTimer: corrected);
    }
  }

  int _getTargetSec() {
    final mode = state.myTimer.mode;
    return switch (mode) {
      TimerMode.pomodoro => 1500,   // 25分钟
      TimerMode.countdown => state.myTimer.targetSec,
      TimerMode.stopwatch => 0,     // 无目标
    };
  }

  int _defaultTarget(TimerMode mode) => switch (mode) {
    TimerMode.pomodoro => 1500,
    TimerMode.countdown => 1800,
    TimerMode.stopwatch => 0,
  };

  // ═══════════════════════════════════════════
  //  聊天
  // ═══════════════════════════════════════════

  Future<void> sendChat(String content) async {
    if (state.roomInfo?.focusMode == 1) {
      state = state.copyWith(errorMessage: '专注模式下无法聊天');
      return;
    }
    if (content.trim().isEmpty) return;
    if (content.length > 200) {
      state = state.copyWith(errorMessage: '消息不能超过200字');
      return;
    }

    // 敏感词过滤
    final filtered = await _read(contentSafetyProvider).filter(content);
    if (filtered.blocked) {
      state = state.copyWith(errorMessage: '消息包含不适当内容');
      return;
    }

    _connection?.send(SendChat(
      roomId: state.roomInfo!.roomId,
      content: filtered.text,
      type: ChatType.text,
    ));
  }

  // ═══════════════════════════════════════════
  //  心跳
  // ═══════════════════════════════════════════

  Timer? _heartbeatTimer;

  void _startHeartbeat() {
    _heartbeatTimer?.cancel();
    _heartbeatTimer = Timer.periodic(
      const Duration(seconds: 15),
      (_) {
        final clientState = _detectClientState();
        _connection?.send(Heartbeat(
          roomId: state.roomInfo?.roomId ?? 0,
          clientTimestamp: DateTime.now().millisecondsSinceEpoch ~/ 1000,
          state: clientState,
        ));
      },
    );
  }

  ClientState _detectClientState() {
    // 通过 WidgetsBindingObserver 生命周期回调判断
    if (_isScreenOff) return ClientState.screenOff;
    if (_isInBackground) return ClientState.background;
    return ClientState.foreground;
  }

  // ═══════════════════════════════════════════
  //  App 生命周期感知
  // ═══════════════════════════════════════════

  bool _isInBackground = false;
  bool _isScreenOff = false;
  DateTime? _backgroundedAt;

  void onAppPaused() {
    _isInBackground = true;
    _backgroundedAt = DateTime.now();
    _stopLocalTimer();  // 暂停本地计时（服务端会在120s后标记AWAY）
  }

  void onAppResumed() {
    _isInBackground = false;
    _isScreenOff = false;

    // 恢复计时器
    if (state.myTimer.phase == TimerPhase.focusing ||
        state.myTimer.phase == TimerPhase.resting) {
      _startLocalTimer();
    }

    // 如果离开超过30秒，请求全量状态同步
    if (_backgroundedAt != null &&
        DateTime.now().difference(_backgroundedAt!).inSeconds > 30) {
      _connection?.requestFullSync();
    }
    _backgroundedAt = null;
  }

  // ═══════════════════════════════════════════
  //  离开与清理
  // ═══════════════════════════════════════════

  Future<void> leaveRoom({String reason = 'manual'}) async {
    _connection?.send(LeaveRoomRequest(
      roomId: state.roomInfo?.roomId ?? 0,
      reason: reason,
    ));
    _cleanupAndExit();
  }

  void _cleanupAndExit() {
    _stopLocalTimer();
    _heartbeatTimer?.cancel();
    _eventSub?.cancel();
    _connection?.dispose();
  }

  @override
  void dispose() {
    _cleanupAndExit();
    super.dispose();
  }

  // ═══════════════════════════════════════════
  //  辅助方法
  // ═══════════════════════════════════════════

      type: ChatType.text,
      timestamp: DateTime.now(),
      isSystem: true,
    );
    state = state.copyWith(
      chatMessages: state.chatMessages.add(msg),
    );
  }

  String _friendlyError(Object e) {
    if (e is StudyRoomException) {
      return switch (e.code) {
        'SR_102' => '房间已锁定',
        'SR_103' => '密码错误',

      };
    }

  }

  String _mapErrorCode(int code) => switch (code) {

    202 => '消息包含敏感内容',
    _ => '操作失败 ($code)',
  };
```

```dart
/// 自习室 WebSocket 连接封装
class StudyRoomConnection {
  final String wsUrl;
  final String sessionId;
  final VoidCallback onConnected;
  final VoidCallback onDisconnected;
  final void Function(String) onError;

  WebSocketChannel? _channel;
  StreamController<ServerMessage> _eventController =
      StreamController.broadcast();
  Stream<ServerMessage> get eventStream => _eventController.stream;

  bool _disposed = false;

  StudyRoomConnection({
    required this.wsUrl,
    required this.sessionId,
    required this.onConnected,
    required this.onDisconnected,
    required this.onError,
  });

  Future<void> connect() async {
    try {
      _channel = WebSocketChannel.connect(Uri.parse(wsUrl));

      // 发送 Join 握手
      _channel!.sink.add(_encodeMessage(ClientMessage.joinRequest(
        JoinRoomRequest(
          roomId: _extractRoomId(wsUrl),
          sessionId: sessionId,
          deviceInfo: await _getDeviceInfo(),
        ),
      )));

      // 监听消息
      _channel!.stream.listen(
        _onData,
        onError: (e) => onError(e.toString()),
        onDone: () {
          if (!_disposed) onDisconnected();
        },
      );
    } catch (e) {
      onError('连接失败: $e');
    }
  }

  Future<void> reconnect() async {
    await _channel?.sink.close();
    await connect();
  }

  void send(ClientMessage message) {
    if (_channel != null && !_disposed) {
      _channel!.sink.add(_encodeMessage(message));
    }
  }

  void requestFullSync() {
    send(ClientMessage.heartbeat(
      Heartbeat(
        roomId: 0, // 全量同步标记
        clientTimestamp: DateTime.now().millisecondsSinceEpoch ~/ 1000,
        state: ClientState.foreground,
      ),
    ));
  }

  void _onData(dynamic data) {
    try {
      final message = _decodeMessage(data as List<int>);
      _eventController.add(message);

      if (message is JoinRoomResponse && message.success) {
        onConnected();
      }
    } catch (e) {
      // 忽略解析错误
    }
  }

  List<int> _encodeMessage(ClientMessage message) {
    // Protobuf 编码
    return message.writeToBuffer();
  }

  ServerMessage _decodeMessage(List<int> bytes) {
    // Protobuf 解码
    return ServerMessage.fromBuffer(bytes);
  }

  int _extractRoomId(String url) {
    // 从 ws://.../study-room/{roomId}?token=... 提取
    final uri = Uri.parse(url);
    final segments = uri.pathSegments;
    return int.tryParse(segments.last) ?? 0;
  }

  Future<String> _getDeviceInfo() async {
    final info = await DeviceInfoPlugin().deviceInfo;
    if (info is IosDeviceInfo) {
      return '${info.utsname.machine} / iOS ${info.systemVersion}';
    } else if (info is AndroidDeviceInfo) {
      return '${info.model} / Android ${info.version.release}';
    }
    return 'Unknown Device';
  }

  void dispose() {
    _disposed = true;
    _channel?.sink.close();
    _eventController.close();
  }
}
```

```dart
class _TimerRingPainter extends CustomPainter {
  final double progress;      // 0.0 ~ 1.0
  final Color phaseColor;
  final Animation<double> pulseAnimation;

  _TimerRingPainter({
    required this.progress,
    required this.phaseColor,
    required this.pulseAnimation,
  }) : super(repaint: pulseAnimation);

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.width / 2 - 8;

    // 背景环
    final bgPaint = Paint()
      ..color = phaseColor.withOpacity(0.1)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 6;
    canvas.drawCircle(center, radius, bgPaint);

    // 进度环
    final progressPaint = Paint()
      ..color = phaseColor
      ..style = PaintingStyle.stroke
      ..strokeWidth = 6
      ..strokeCap = StrokeCap.round;
    canvas.drawArc(
      Rect.fromCircle(center: center, radius: radius),
      -pi / 2, // 从顶部开始
      2 * pi * progress,
      false,
      progressPaint,
    );

    // 脉冲光晕（专注中）
    final pulseValue = pulseAnimation.value;
    final glowPaint = Paint()
      ..color = phaseColor.withOpacity(0.15 * (1 - pulseValue))
      ..style = PaintingStyle.stroke
      ..strokeWidth = 6 + 12 * pulseValue
      ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8);
    canvas.drawCircle(center, radius, glowPaint);
  }

  @override
  bool shouldRepaint(_TimerRingPainter old) =>
      progress != old.progress || phaseColor != old.phaseColor;
}
```

```dart
/// 自习室相关深链接路由
class StudyRoomDeepLinks {
  static const _scheme = 'primetop';
  static const _host = 'study-room';

  /// 处理深链接
  static bool handleDeepLink(BuildContext context, Uri uri) {
    if (uri.scheme != _scheme) return false;
    if (uri.host != _host) return false;

    // primetop://study-room/:roomCode
    final segments = uri.pathSegments;
    if (segments.isEmpty) return false;

    final roomCode = segments[0];
    final password = uri.queryParameters['pwd'];

    context.push('/study-room/$roomCode',
        extra: password != null ? {'password': password} : null);
    return true;
  }

  /// 生成分享链接
  static String buildShareUrl(String roomCode, {String? password}) {
    final params = password != null ? '?pwd=$password' : '';
    return 'https://share.primetop.com/sr/$roomCode$params';
  }

  /// 生成分享文案
  static String buildShareText(String title, int onlineCount,
      String shareUrl) {
    return '🚀 一起来自习室学习吧！\n'
        '📚 $title\n'
        '👥 当前 $onlineCount 人正在专注学习\n'
        '🔗 $shareUrl';

}
```

### 12.2 WebSocket 消息频率控制

```dart
/// 客户端消息发送节流
class MessageThrottle {
  final Duration interval;
  final int maxPerBurst;
  final Queue<DateTime> _timestamps = Queue();

  MessageThrottle({
    this.interval = const Duration(seconds: 1),
    this.maxPerBurst = 5,
  });

  bool canSend() {
    final now = DateTime.now();
    while (_timestamps.isNotEmpty &&
        now.difference(_timestamps.first) > interval) {
      _timestamps.removeFirst();
    }
    if (_timestamps.length < maxPerBurst) {
      _timestamps.add(now);
      return true;
    }
    return false;
  }
}
```
