### 3.3 Notifier 示例

```dart
// lib/features/poem/notifiers/poem_detail_notifier.dart

@riverpod
class PoemDetailNotifier extends _$PoemDetailNotifier {
  @override
  Future<PoemDetail?> build(int poemId) async {
    // 先查本地缓存
    final cached = await ref.read(poemCacheProvider).getPoem(poemId);
    if (cached != null) {
      // 后台刷新
      _refreshInBackground(poemId);
      return cached;
    }
    // 缓存未命中，网络加载
    return _fetchFromRemote(poemId);
  }

  Future<PoemDetail> _fetchFromRemote(int poemId) async {
    final response = await ref.read(poemApiProvider).getPoemDetail(poemId);
    final detail = PoemDetail.fromJson(response.data);
    // 写入缓存
    await ref.read(poemCacheProvider).putPoem(detail);
    return detail;
  }

  void _refreshInBackground(int poemId) {
    // 静默刷新，不触发 UI 重建
    _fetchFromRemote(poemId).then((detail) {
      state = AsyncData(detail);
    });
  }

  /// 切换收藏
  Future<void> toggleBookmark() async {
    final current = state.valueOrNull;
    if (current == null) return;

    final newStatus = !current.userProgress!.isBookmarked;
    // 乐观更新
    state = AsyncData(current.copyWith(
      userProgress: current.userProgress!.copyWith(isBookmarked: newStatus),
    ));

    try {
      await ref.read(poemApiProvider).toggleBookmark(
        poemId: poemId,
        bookmarked: newStatus,
      );
    } catch (e) {
      // 回滚
      state = AsyncData(current);
      ref.read(snackbarProvider).show('操作失败，请重试');
    }
  }
}
```

```dart
class _PoemDetailPageState extends ConsumerState<PoemDetailPage>
    with SingleTickerProviderStateMixin {
  late final TabController _tabController;

  @override
  void initState() {
    super.initState();
    _tabController = TabController(
      length: PoemTab.values.length,
      vsync: this,
    );
  }

  @override
  void dispose() {
    _tabController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final detailAsync = ref.watch(poemDetailNotifierProvider(widget.poemId));

    return detailAsync.when(
      loading: () => const Scaffold(
        body: PoemDetailSkeleton(),
      ),
      error: (error, _) => Scaffold(
        body: ErrorStateWidget(
          error: error,
          onRetry: () => ref.invalidate(poemDetailNotifierProvider(widget.poemId)),
        ),
      ),
      data: (detail) {
        if (detail == null) return const NotFoundWidget();
        return _PoemDetailContent(
          detail: detail,
          tabController: _tabController,
        );
      },
    );
  }
}
```

```dart
class _PoemDictationPageState extends ConsumerState<PoemDictationPage> {
  final _scrollController = ScrollController();
  final _focusNodes = <int, FocusNode>{};

  @override
  void initState() {
    super.initState();
    // 启动计时器
    ref.read(dictationProvider(widget.poemId).notifier).startTimer();
  }

  @override
  void dispose() {
    _scrollController.dispose();
    for (final fn in _focusNodes.values) {
      fn.dispose();
    }
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(dictationProvider(widget.poemId));

    return PopScope(
      canPop: false,
      onPopInvoked: (didPop) => _handleBack(didPop),
      child: Scaffold(
        appBar: AppBar(
          title: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text('默写检测'),
              Text(state.poemTitle,
                style: Theme.of(context).textTheme.bodySmall),
            ],
          ),
          actions: [
            // 计时器
            Padding(
              padding: const EdgeInsets.symmetric(horizontal: 12),
              child: _TimerDisplay(elapsedSeconds: state.elapsedSeconds),
            ),
          ],
        ),
        body: state.isLoading
            ? const Center(child: CircularProgressIndicator())
            : _buildBody(context, state),
        bottomNavigationBar: _DictationBottomBar(
          progress: _calculateProgress(state),
          isPaused: state.isPaused,
          onPause: () => ref.read(dictationProvider(widget.poemId).notifier).togglePause(),
          onSubmit: () => _handleSubmit(state),
        ),
      ),
    );
  }

  Widget _buildBody(BuildContext context, DictationState state) {
    switch (state.type) {
      case DictationType.fillBlank:
        return _FillBlankBody(
          blanks: state.blanks,
          answers: state.answers,
          onAnswerChanged: (idx, text) {
            ref.read(dictationProvider(widget.poemId).notifier)
                .updateAnswer(idx, text);
          },
          focusNodes: _focusNodes,
        );
      case DictationType.full:
        return _FullDictationBody(
          lines: state.displayLines,
          answers: state.answers,
          onAnswerChanged: (lineNum, text) {
            ref.read(dictationProvider(widget.poemId).notifier)
                .updateLineAnswer(lineNum, text);
          },
          focusNodes: _focusNodes,
        );
      case DictationType.keyLine:
        return _KeyLineBody(
          keyLines: state.keyLines,
          answers: state.answers,
          onAnswerChanged: (lineNum, text) {
            ref.read(dictationProvider(widget.poemId).notifier)
                .updateLineAnswer(lineNum, text);
          },
        );
      case DictationType.errorWord:
        return _ErrorWordBody(
          errorWords: state.errorWords,
          answers: state.answers,
          onAnswerChanged: (idx, text) {
            ref.read(dictationProvider(widget.poemId).notifier)
                .updateAnswer(idx, text);
          },
        );
    }
  }

  Future<void> _handleSubmit(DictationState state) async {
    // 确认对话框
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('确认提交'),
        content: Text('已完成 ${_calculateProgress(state)}%，确定提交吗？'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(_, false), child: const Text('再看看')),
          FilledButton(onPressed: () => Navigator.pop(_, true), child: const Text('提交')),
        ],
      ),
    );

    if (confirmed != true) return;

    // 提交到服务端
    final result = await ref.read(dictationProvider(widget.poemId).notifier)
        .submit();

    if (result != null && mounted) {
      context.goReplacement('/poem/dictation-result/${result.recordId}');
    }
  }

  void _handleBack(bool didPop) {
    if (didPop) return;
    showDialog(context: context, builder: (_) => AlertDialog(
      title: const Text('确定退出？'),
      content: const Text('当前默写进度不会保存'),
      actions: [
        TextButton(onPressed: () => Navigator.pop(_), child: const Text('继续默写')),
        FilledButton(
          onPressed: () { Navigator.pop(_); Navigator.pop(context); },
          child: const Text('退出'),
        ),
      ],
    ));
  }

  double _calculateProgress(DictationState state) {
    if (state.type == DictationType.fillBlank) {
      final filled = state.answers.length;
      return state.blanks.isEmpty ? 0 : filled / state.blanks.length;
    }
    // other types...
    return state.answers.isNotEmpty ? 1.0 : 0.0;
  }
}
```

```dart
class _FillBlankBody extends StatelessWidget {

  final Map<int, String> answers;
  final void Function(int blankIndex, String text) onAnswerChanged;
  final Map<int, FocusNode> focusNodes;

    required this.answers,
    required this.onAnswerChanged,
    required this.focusNodes,
  });

  @override
  Widget build(BuildContext context) {
    return ListView.separated(
      padding: const EdgeInsets.all(16),
      itemCount: blanks.length,

      itemBuilder: (_, index) {
        final blank = blanks[index];
        final focusNode = focusNodes.putIfAbsent(index, () => FocusNode());

        return Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            // 上下文提示
            Text.rich(
              TextSpan(children: [
                TextSpan(text: blank.beforeContext),
                const TextSpan(
                  text: '____',
                  style: TextStyle(
                    decoration: TextDecoration.underline,
                    color: Colors.grey,
                  ),
                ),
                TextSpan(text: blank.afterContext),
              ]),
              style: TextStyle(
                fontSize: 16,
                height: 1.8,
                color: Theme.of(context).colorScheme.outline,
              ),
            ),
            const SizedBox(height: 8),
            // 输入框
            TextField(
              focusNode: focusNode,
              decoration: InputDecoration(
                hintText: blank.hint != null ? '拼音提示：${blank.hint}' : '请填写',
                border: const OutlineInputBorder(),
                focusedBorder: OutlineInputBorder(
                  borderSide: BorderSide(
                    color: Theme.of(context).colorScheme.primary,
                    width: 2,
                  ),
                ),
              ),
              onChanged: (text) => onAnswerChanged(index, text),
              onSubmitted: (_) {
                // 自动跳到下一个空
                if (index < blanks.length - 1) {
                  focusNodes[index + 1]?.requestFocus();
                }
              },
            ),
            // 行号标签
            Padding(
              padding: const EdgeInsets.only(top: 4),
              child: Text('第 ${blank.lineNumber} 行',
                style: Theme.of(context).textTheme.labelSmall?.copyWith(

            ),
```

```dart
// lib/features/poem/services/poem_cache_service.dart

class PoemCacheService {
  final Box _poemBox;
  final Box _progressBox;

  /// 缓存篇目详情
  Future<void> cachePoem(PoemDetail detail) async {
    await _poemBox.put('poem_${detail.poemId}', detail.toJson());
  }

  /// 获取缓存的篇目
  PoemDetail? getCachedPoem(int poemId) {
    final json = _poemBox.get('poem_$poemId');
    if (json == null) return null;
    return PoemDetail.fromJson(Map<String, dynamic>.from(json));
  }

  /// 缓存用户进度
  Future<void> cacheProgress(int poemId, PoemUserProgress progress) async {
    await _progressBox.put('progress_$poemId', progress.toJson());
  }

  /// 预缓存教材同步篇目（WiFi环境下）
  Future<void> precacheTextbookPoems(int textbookId, String grade) async {
    final connectivity = await Connectivity().checkConnectivity();
    if (connectivity == ConnectivityResult.wifi) {
      // 批量缓存
      final poems = await _api.getTextbookSyncPoems(textbookId, grade);
      for (final poem in poems) {
        await cachePoem(poem);
      }
    }
  }

  /// 离线默写检测
  /// 本地比对答案，不依赖网络
  DictationResult checkOfflineDictation(
    int poemId,
    DictationType type,
    Map<int, String> answers,
    int durationSec,
  ) {
    final cached = getCachedPoem(poemId);
    if (cached == null) {
      throw Exception('篇目未缓存，无法离线默写');
    }

    final errors = <DictationError>[];
    var correctChars = 0;
    var totalChars = 0;

    for (final line in cached.contentLines.lines) {
      final userAnswer = answers[line.lineNumber] ?? '';
      totalChars += line.text.replaceAll(RegExp(r'[，。、；：！？]'), '').length;

      // 逐字比对
      final expectedChars = line.text.characters.toList();
      final actualChars = userAnswer.characters.toList();

      for (var i = 0; i < expectedChars.length; i++) {
        final expected = expectedChars[i];
        if (RegExp(r'[，。、；：！？\s]').hasMatch(expected)) continue;

        if (i < actualChars.length) {
          if (actualChars[i] == expected) {
            correctChars++;
          } else {
            errors.add(DictationError(
              lineNumber: line.lineNumber,
              position: i,
              expectedChar: expected,
              actualChar: actualChars[i],
              errorType: _classifyError(expected, actualChars[i]),
              hint: _generateHint(expected),
            ));
          }
        } else {
          errors.add(DictationError(
            lineNumber: line.lineNumber,
            position: i,
            expectedChar: expected,
            actualChar: null,
            errorType: ErrorType.missing,
            hint: _generateHint(expected),
          ));
        }
      }
    }

    return DictationResult(
      recordId: -1, // 离线记录标记
      poemId: poemId,
      score: totalChars > 0 ? (correctChars / totalChars * 100) : 0,
      totalChars: totalChars,
      correctChars: correctChars,
      errors: errors,
      weakLines: errors.map((e) => e.lineNumber).toSet().toList(),
    );
  }

  ErrorType _classifyError(String expected, String actual) {
    // 同音字判断（简化版，实际应查拼音表）
    // 形近字判断
    // 这里返回默认类型
    return ErrorType.homophone;
  }

  String _generateHint(String char) {
    // 返回字形记忆提示
    return '$char 的书写提示';
  }
}
```

```dart
// 统一错误处理
extension PoemErrorHandler on WidgetRef {
  void handlePoemError(Object error, StackTrace stack) {
    if (error is DioException) {
      switch (error.type) {
        case DioExceptionType.connectionTimeout:
        case DioExceptionType.receiveTimeout:
          read(snackbarProvider).show('网络超时，请检查网络后重试');
        case DioExceptionType.connectionError:
          read(snackbarProvider).show('网络连接失败，已切换到离线模式');
        default:
          read(snackbarProvider).show('加载失败，请重试');

    } else {

    }
  }
}
```
