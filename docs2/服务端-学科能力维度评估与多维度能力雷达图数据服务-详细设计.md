# 服务端 - 学科能力维度评估与多维度能力雷达图数据服务

## 1. 文档概述

### 1.1 目的

本文档详细设计 PrimeTop 的**学科能力维度评估服务**，该服务负责将学生各学科的学习行为数据（答题记录、AI辅导交互、练习结果等）映射到标准化的能力维度模型上，持续评估各维度的掌握水平，并生成雷达图数据供客户端渲染。

### 1.2 范围

- 各学科能力维度体系定义与维护
- 维度掌握度增量计算引擎
- 雷达图数据聚合与缓存
- 维度变化趋势追踪
- 跨学科通用能力提取
- 管理后台维度配置
- 客户端 API 接口

### 1.3 依赖

| 依赖模块 | 说明 |
|----------|------|
| 知识点体系与教材映射引擎 | 提供知识点树、知识点→维度映射 |
| 练习答题实时批改服务 | 提供答题结果事件 |
| 用户学习画像 | 提供能力维度快照读写 |

## 3. 核心算法设计

### 3.1 维度掌握度增量更新算法

采用**贝叶斯增量更新**策略，每次答题事件触发对应维度的掌握度调整。

#### 3.1.1 算法原理

使用 Beta 分布建模掌握度：
- 先验：`Beta(α_prior, β_prior)`，均值 = `α / (α + β)`
- 观察到正确：`α_posterior = α_prior + evidence_weight`
- 观察到错误：`β_posterior = β_prior + evidence_weight`
- 后验均值即为更新后的掌握度

#### 3.1.2 证据权重计算

```python
def calculate_evidence_weight(
    difficulty: float,      # 题目难度 0~1
    is_correct: bool,       # 是否正确
    current_mastery: float, # 当前掌握度
    question_type: str,     # 题目类型
    time_spent_ratio: float # 实际用时/预期用时
) -> float:
    """
    计算单次答题对维度掌握度的证据权重。
    难度越大、答对/答错与预期越不一致，权重越大。
    """
    # 基础权重：难度因子
    base_weight = 0.5 + difficulty * 0.5  # 难题权重更大

    # 惊喜因子：结果越出乎意料，学习信号越强
    expected_prob = current_mastery  # 当前掌握度即期望正确率
    if is_correct:
        surprise = 1.0 - expected_prob  # 掌握度低但答对了 → 强信号
    else:
        surprise = expected_prob  # 掌握度高但答错了 → 强信号

    # 用时因子
    time_factor = 1.0
    if time_spent_ratio > 2.0:
        time_factor = 0.7  # 超时完成 → 降低信号强度
    elif time_spent_ratio < 0.3:
        time_factor = 0.8  # 过快完成（可能是蒙的）

    # 题目类型因子
    type_factor = {
        'CHOICE': 0.8,
        'FILL': 1.0,
        'SHORT_ANSWER': 1.2,
        'CALCULATION': 1.1,
        'ESSAY': 1.3,
        'PROOF': 1.3,
    }.get(question_type, 1.0)

    return base_weight * (0.5 + surprise) * time_factor * type_factor
```

#### 3.1.3 增量更新主流程

```python
class DimensionMasteryUpdater:
    """维度掌握度增量更新器"""

    # 初始先验参数（冷启动）
    DEFAULT_ALPHA = 2.0
    DEFAULT_BETA = 2.0  # 均值 0.5

    async def update_on_answer(
        self,
        user_id: int,
        subject_code: str,
        knowledge_point_ids: list[int],
        is_correct: bool,
        difficulty: float,
        question_type: str,
        time_spent_seconds: int,
        expected_time_seconds: int,
        answer_record_id: str,
    ) -> list[DimensionUpdateResult]:
        """
        答题事件触发维度掌握度更新。

        Returns: 各受影响维度的更新结果
        """
        # 1. 查找受影响的维度
        affected_dimensions = await self._get_affected_dimensions(
            subject_code, knowledge_point_ids
        )
        if not affected_dimensions:
            return []

        # 2. 获取当前掌握度
        current_masteries = await self._get_current_masteries(
            user_id, subject_code, [d.dimension_code for d in affected_dimensions]
        )

        results = []
        for dim in affected_dimensions:
            current = current_masteries.get(dim.dimension_code)
            old_score = current.mastery_score if current else 0.5
            confidence = current.confidence if current else 0.0

            # 3. 计算证据权重
            time_ratio = (time_spent_seconds / expected_time_seconds
                         if expected_time_seconds > 0 else 1.0)
            evidence = calculate_evidence_weight(
                difficulty=difficulty,
                is_correct=is_correct,
                current_mastery=old_score,
                question_type=question_type,
                time_spent_ratio=time_ratio,
            )

            # 4. 贝叶斯更新
            alpha, beta = self._mastery_to_beta(old_score, confidence)
            if is_correct:
                alpha += evidence * dim.relevance_weight
            else:
                beta += evidence * dim.relevance_weight
            new_score = alpha / (alpha + beta)
            new_score = max(0.0, min(1.0, new_score))

            # 5. 更新置信度
            new_confidence = min(1.0, confidence + evidence * 0.05)

            # 6. 持久化
            await self._persist_update(
                user_id=user_id,
                subject_code=subject_code,
                dimension_code=dim.dimension_code,
                old_score=old_score,
                new_score=new_score,
                trigger_type='ANSWER_CORRECT' if is_correct else 'ANSWER_WRONG',
                trigger_id=answer_record_id,
                evidence_weight=evidence,
                confidence=new_confidence,
            )

            results.append(DimensionUpdateResult(
                dimension_code=dim.dimension_code,
                dimension_name=dim.name,
                old_score=old_score,
                new_score=new_score,
                delta=new_score - old_score,
            ))

        # 7. 发送维度变更事件
        await self._publish_dimension_changed_event(
            user_id, subject_code, results
        )

        return results

    def _mastery_to_beta(self, mastery: float, confidence: float) -> tuple[float, float]:
        """将掌握度和置信度转换为 Beta 分布参数"""
        # 置信度越高，等效样本量越大，分布越集中
        equivalent_n = max(4, confidence * 100)
        alpha = mastery * equivalent_n
        beta = (1 - mastery) * equivalent_n
        return max(0.1, alpha), max(0.1, beta)
```

### 3.2 AI 辅导会话维度更新

AI 对话不直接产生对/错信号，而是通过对话质量间接影响维度：

```python
class AISessionDimensionUpdater:
    """AI辅导会话的维度更新"""

    async def update_on_session_end(self, session: AISession) -> list[DimensionUpdateResult]:
        """
        AI辅导会话结束后，根据会话摘要更新维度。

        understanding_indicators = await self._analyze_understanding(session)

        results = []
        for dim_code, indicators in understanding_indicators.items():
            # 基于AI对话的信号：追问次数（多→可能不理解）、
            # 是否请求换一种讲法、是否成功解决原问题等

                subject_code=session.subject_code,
                dimension_code=dim_code,
                old_score=current.mastery_score,
                new_score=new_score,
                trigger_type='AI_SESSION',
                trigger_id=session.id,
                evidence_weight=0.5,  # AI会话信号弱于直接答题
                confidence=current.confidence + 0.02,
            )
            results.append(...)

        return results

    def _compute_session_delta(self, indicators: dict) -> float:
        """
        基于对话指标计算掌握度变化量。
        """
        delta = 0.0

        # 正面信号
        if indicators.get('problem_resolved'):
            delta += 0.03
        if indicators.get('student_explained_back'):  # 学生复述理解
            delta += 0.02
        if indicators.get('asked_similar_correct'):  # 同类题答对
            delta += 0.04

        # 负面信号
        follow_up_count = indicators.get('follow_up_count', 0)
        if follow_up_count > 5:
            delta -= 0.02  # 大量追问 → 可能没理解
        if indicators.get('requested_simpler_explanation'):
            delta -= 0.01
        if indicators.get('asked_similar_wrong'):  # 同类题答错
            delta -= 0.04

        return delta
```

### 3.3 综合学科评分计算

各维度加权求和得到学科综合能力评分：

```python
def compute_subject_composite_score(
    dimension_scores: dict[str, float],
    dimension_weights: dict[str, float],
) -> float:
    """
    计算学科综合能力评分（加权平均）。
    dimension_scores: {dimension_code: mastery_score}
    dimension_weights: {dimension_code: weight}
    """
    total_weight = 0.0
    weighted_sum = 0.0

    for dim_code, score in dimension_scores.items():
        weight = dimension_weights.get(dim_code, 0.2)
        weighted_sum += score * weight
        total_weight += weight

    return weighted_sum / total_weight if total_weight > 0 else 0.0
```

```python
def extract_cross_subject_abilities(
    user_masteries: dict[str, float],  # {dimension_code: score}
) -> dict[str, float]:
    """
    从学科维度掌握度中提取跨学科通用能力评分。
    """
    result = {}
    for ability_code, config in CROSS_SUBJECT_ABILITIES.items():
        total_weight = 0.0
        weighted_sum = 0.0
        for dim_code, weight in config['weights'].items():
            if dim_code in user_masteries:
                weighted_sum += user_masteries[dim_code] * weight
                total_weight += weight
        if total_weight > 0:
            result[ability_code] = weighted_sum / total_weight
    return result
```

## 5. 事件驱动集成

### 5.1 订阅的事件

| 事件 | 来源 | 触发动作 |
|------|------|----------|
| `answer.submitted` | 练习批改服务 | 答题后增量更新维度掌握度 |
| `ai.session.ended` | AI对话引擎 | 会话结束后分析理解度更新维度 |
| `mistake.review.completed` | 错题复习服务 | 错题复习成功后更新维度 |
| `user.grade.changed` | 学期管理服务 | 升年级后重置/衰减维度置信度 |
| `knowledge.point.mapping.updated` | 知识点服务 | 维度-知识点映射变更时触发重算 |

### 5.2 发出的事件

| 事件 | 触发条件 | 消费者 |
|------|----------|--------|
| `dimension.mastery.updated` | 维度掌握度变化 | 学习画像服务、首页推荐服务 |
| `dimension.significant_change` | 单次变化>0.05 或连续3次同向变化 | 学情报告服务、家长通知服务 |

### 5.3 事件处理器注册

```python
# dimension_event_handlers.py

from framework.event_bus import EventHandler, event_handler

class DimensionEventHandlers(EventHandler):

    @event_handler("answer.submitted", channel="standard")
    async def on_answer_submitted(self, event: EventEnvelope):
        """答题事件 → 维度掌握度更新"""
        payload = AnswerSubmittedPayload(**event.payload)
        await self.updater.update_on_answer(
            user_id=payload.user_id,
            subject_code=payload.subject_code,
            knowledge_point_ids=payload.knowledge_point_ids,
            is_correct=payload.is_correct,
            difficulty=payload.difficulty,
            question_type=payload.question_type,
            time_spent_seconds=payload.time_spent_seconds,
            expected_time_seconds=payload.expected_time_seconds,
            answer_record_id=payload.answer_record_id,
        )

    @event_handler("ai.session.ended", channel="standard")
    async def on_ai_session_ended(self, event: EventEnvelope):
        """AI会话结束 → 维度掌握度更新"""
        payload = AISessionEndedPayload(**event.payload)
        session = await self.session_repo.get(payload.session_id)
        if session and session.subject_code:
            await self.ai_updater.update_on_session_end(session)

    @event_handler("mistake.review.completed", channel="standard")
    async def on_mistake_review_completed(self, event: EventEnvelope):
        """错题复习完成 → 维度掌握度更新"""
        payload = MistakeReviewPayload(**event.payload)
        if payload.is_redeemed:  # 订正成功
            await self.updater.update_on_answer(
                user_id=payload.user_id,
                subject_code=payload.subject_code,
                knowledge_point_ids=payload.knowledge_point_ids,
                is_correct=True,
                difficulty=payload.difficulty,
                question_type=payload.question_type,
                time_spent_seconds=payload.time_spent_seconds,
                expected_time_seconds=payload.expected_time_seconds,
                answer_record_id=payload.mistake_id,
            )

    @event_handler("user.grade.changed", channel="standard")
    async def on_grade_changed(self, event: EventEnvelope):
        """升年级 → 维度置信度衰减"""
        payload = UserGradeChangedPayload(**event.payload)
        await self._decay_confidence_on_grade_up(
            payload.user_id, payload.new_grade_code
        )
```

### 5.4 升年级置信度衰减

```python
async def _decay_confidence_on_grade_up(self, user_id: int, new_grade: str):
    """
    升年级时降低维度掌握度的置信度，
    因为高年级的同一维度要求更高，历史数据需要打折。
    """
    # 按升学距离决定衰减系数
    decay_factor = 0.7  # 统一衰减到 70%

    masteries = await self.repo.get_all_user_masteries(user_id)
    for m in masteries:
        new_confidence = m.confidence * decay_factor
        await self.repo.update_confidence(
            user_id=user_id,
            subject_code=m.subject_code,
            dimension_code=m.dimension_code,
            confidence=new_confidence,
        )
```

```python
@register_task(
    name="dimension.recalculate_percentiles",
    schedule=schedules.crontab(hour=3, minute=0),  # 每天凌晨 3:00
)
def recalculate_percentiles():
    """
    重算各维度的同龄百分位分布。
    使用近似百分位算法避免全表扫描。
    """
    for subject_dim in get_all_active_dimensions():
        # 按年级分组计算百分位
        for grade_code in get_active_grades():
            scores = get_dimension_scores_for_grade(
                subject_dim.subject_code,
                subject_dim.dimension_code,
                grade_code,
            )
            if not scores:
                continue

            # 计算 P10, P25, P50, P75, P90
            percentiles = calculate_percentiles(scores, [10, 25, 50, 75, 90])

            # 缓存到 Redis
            cache_key = f"dimension:percentile:{subject_dim.dimension_code}:{grade_code}"
            redis.set(cache_key, json.dumps(percentiles), ex=86400 * 2)
```

```python
async def invalidate_user_dimension_cache(user_id: int, subject_code: str):
    """维度掌握度变更后失效相关缓存"""
    pipe = redis.pipeline()
    pipe.delete(f"dim:mastery:{user_id}:{subject_code}")
    pipe.delete(f"dim:radar:{user_id}:{subject_code}")
    pipe.delete(f"dim:trend:{user_id}:{subject_code}:*")
    # 通配符删除需要 SCAN
    async for key in redis.scan_iter(f"dim:trend:{user_id}:{subject_code}:*"):
        pipe.delete(key)
    await pipe.execute()
```

| `DIM_006` | 503 | 百分位数据不可用（缓存未就绪） |
| `DIM_010` | 500 | 维度更新内部错误 |

### 8.2 降级策略

| 场景 | 降级方案 |
|------|----------|
| Redis 不可用 | 直接查 MySQL，跳过缓存层 |
| 百分位缓存未命中 | 返回无百分位数据的雷达图，不阻塞 |
| 维度更新失败 | 写入重试队列，3次重试后进 DLQ，记录告警 |
| 知识点→维度映射缺失 | 跳过该维度更新，记录 warning 日志 |
| 趋势数据缺失 | 返回空趋势，前端显示"数据不足" |

### 8.3 乐观锁冲突处理

```python
async def _persist_update(self, ..., version: int):
    """
    带乐观锁的维度更新持久化。
    """
    rows = await db.execute(
        update(user_dimension_mastery)
        .where(
            (user_dimension_mastery.c.user_id == user_id) &
            (user_dimension_mastery.c.subject_code == subject_code) &
            (user_dimension_mastery.c.dimension_code == dimension_code) &
            (user_dimension_mastery.c.version == version)
        )
        .values(
            mastery_score=new_score,
            confidence=new_confidence,
            sample_count=user_dimension_mastery.c.sample_count + 1,
            last_updated_at=utcnow(),
            version=version + 1,
        )
    )
    if rows == 0:
        # 乐观锁冲突，重新读取并重试
        raise OptimisticLockError(
            f"Concurrent update on dimension {dimension_code} for user {user_id}"
        )
```

### 9.3 雷达图渲染组件

```dart
// lib/features/dimension/widgets/dimension_radar_chart.dart

class DimensionRadarChart extends StatelessWidget {
  final List<DimensionItem> dimensions;
  final Map<String, double>? compareData;
  final bool showLabels;
  final bool showPercentile;

  const DimensionRadarChart({
    super.key,
    required this.dimensions,
    this.compareData,
    this.showLabels = true,
    this.showPercentile = true,
  });

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    return CustomPaint(
      size: const Size(280, 280),
      painter: _RadarChartPainter(
        dimensions: dimensions,
        compareData: compareData,
        primaryColor: theme.colorScheme.primary,
        compareColor: theme.colorScheme.outline.withOpacity(0.3),
        gridColor: theme.colorScheme.outlineVariant.withOpacity(0.3),
        labelStyle: theme.textTheme.bodySmall!,
      ),
    );
  }
}

class _RadarChartPainter extends CustomPainter {
  final List<DimensionItem> dimensions;
  final Map<String, double>? compareData;
  final Color primaryColor;
  final Color compareColor;
  final Color gridColor;
  final TextStyle labelStyle;

  _RadarChartPainter({
    required this.dimensions,
    this.compareData,
    required this.primaryColor,
    required this.compareColor,
    required this.gridColor,
    required this.labelStyle,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = min(size.width, size.height) / 2 - 40;
    final n = dimensions.length;
    if (n < 3) return;

    final angleStep = 2 * pi / n;

    // 1. 绘制背景网格（5层）
    for (var level = 1; level <= 5; level++) {
      final r = radius * level / 5;
      final path = Path();
      for (var i = 0; i < n; i++) {
        final angle = -pi / 2 + i * angleStep;
        final point = Offset(
          center.dx + r * cos(angle),
          center.dy + r * sin(angle),
        );
        if (i == 0) path.moveTo(point.dx, point.dy);
        else path.lineTo(point.dx, point.dy);
      }
      path.close();
      canvas.drawPath(
        path,
        Paint()..color = gridColor..style = PaintingStyle.stroke..strokeWidth = 0.5,
      );
    }

    // 2. 绘制轴线
    for (var i = 0; i < n; i++) {
      final angle = -pi / 2 + i * angleStep;
      canvas.drawLine(
        center,
        Offset(center.dx + radius * cos(angle), center.dy + radius * sin(angle)),
        Paint()..color = gridColor..strokeWidth = 0.5,
      );
    }

    // 3. 绘制对比数据（如有）
    if (compareData != null) {
      _drawDataArea(
        canvas, center, radius, n, angleStep,
        (i) => compareData![dimensions[i].dimensionCode] ?? 0,
        compareColor,
        PaintingStyle.fill,
      );
    }

    // 4. 绘制用户数据
    _drawDataArea(
      canvas, center, radius, n, angleStep,
      (i) => dimensions[i].score,
      primaryColor.withOpacity(0.3),
      PaintingStyle.fill,
    );
    // 边线
    _drawDataArea(
      canvas, center, radius, n, angleStep,
      (i) => dimensions[i].score,
      primaryColor,
      PaintingStyle.stroke,
    );

    // 5. 绘制标签
    for (var i = 0; i < n; i++) {
      final angle = -pi / 2 + i * angleStep;
      final labelR = radius + 24;
      final labelPoint = Offset(
        center.dx + labelR * cos(angle),
        center.dy + labelR * sin(angle),
      );
      final textSpan = TextSpan(
        text: '${dimensions[i].name}\n${(dimensions[i].score * 100).toInt()}',
        style: labelStyle,
      );
      final tp = TextPainter(
        text: textSpan,
        textAlign: TextAlign.center,
        textDirection: TextDirection.ltr,
      )..layout();
      tp.paint(canvas, Offset(
        labelPoint.dx - tp.width / 2,
        labelPoint.dy - tp.height / 2,
      ));
    }
  }

  void _drawDataArea(
    Canvas canvas, Offset center, double radius, int n, double angleStep,
    double Function(int) getValue, Color color, PaintingStyle style,
  ) {
    final path = Path();
    for (var i = 0; i < n; i++) {
      final angle = -pi / 2 + i * angleStep;
      final r = radius * getValue(i);
      final point = Offset(
        center.dx + r * cos(angle),
        center.dy + r * sin(angle),
      );
      if (i == 0) path.moveTo(point.dx, point.dy);
      else path.lineTo(point.dx, point.dy);
    }
    path.close();
    canvas.drawPath(
      path,
      Paint()
        ..color = color
        ..style = style
        ..strokeWidth = 2.0,
    );
  }

  @override
  bool shouldRepaint(covariant _RadarChartPainter old) {
    return old.dimensions != dimensions || old.compareData != compareData;
  }
}
```
