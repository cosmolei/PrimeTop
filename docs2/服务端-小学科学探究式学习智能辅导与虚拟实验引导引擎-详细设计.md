# 服务端-小学科学探究式学习智能辅导与虚拟实验引导引擎 详细设计

## 1. 概述

### 1.1 模块定位

小学科学探究式学习智能辅导引擎是 PrimeTop 全学科智能辅导体系的重要组成部分，专门服务于小学 3-6 年级科学学科的 AI 辅助学习场景。与语文、数学、英语等学科智能辅导引擎平行，但具有独特定位——**科学学科是小学阶段唯一以"探究式学习"为核心教学方法的学科**，本引擎需深度适配这一教学特征。

根据《义务教育科学课程标准（2022年版）》，小学科学课程涵盖物质科学、生命科学、地球与宇宙科学、技术与工程四大领域，旨在培养学生的科学探究能力、科学思维和科学态度。本引擎将 AI 辅导能力与科学探究教学法深度融合，为学生提供：
- 科学概念的可视化、生活化讲解
- 虚拟实验引导与探究过程辅助
- 科学观察项目的智能指导
- 跨学科知识关联与拓展

### 1.2 核心职责

| 职责 | 说明 |
| --- | --- |
| **科学问题智能解答** | 接收学生的科学问题，以适龄、探究式的方式引导理解 |
| **虚拟实验引导** | 引导学生通过虚拟实验模拟科学现象，理解科学原理 |
| **科学观察项目管理** | 管理长期观察任务（如植物生长观察、天气记录），提供智能提醒与指导 |
| **探究过程辅助** | 在"提出问题→猜想假设→实验验证→得出结论"全流程中提供 AI 辅助 |
| **跨领域知识关联** | 建立四大科学领域间及与初中物理/化学/生物/地理的衔接关系 |
| **科学素养评估** | 多维度评估学生科学探究能力、概念理解度和科学思维水平 |
| **生活化场景关联** | 将抽象科学概念与学生日常生活经验关联，增强理解 |

### 1.4 设计原则

1. **探究优先于结论**：不直接告诉学生"答案"，而是通过提问、引导、实验模拟等方式帮助学生自主发现科学规律。
2. **生活化驱动**：所有抽象概念均从学生可感知的日常生活现象引入。
3. **安全第一**：虚拟实验引导中严格遵守实验安全规范，对涉及火、电、化学品的实验给出明确安全提示。
4. **跨学科贯通**：建立小学科学与初中物理/化学/生物/地理的知识衔接桥梁，为升学做好准备。
5. **多模态适配**：科学学习高度依赖视觉理解（图表、实验装置、自然现象），引擎需支持图片输入理解和图解输出。

### 2.2 知识点数据结构

```typescript
/**
 * 科学知识点实体
 */
interface ScienceKnowledgePoint {
  /** 知识点唯一标识 */
  id: string;
  
  /** 知识点名称，如"水的三态变化" */
  name: string;
  
  /** 所属领域 */
  domain: ScienceDomain;
  
  /** 适用年级范围 */
  gradeRange: {
    minGrade: number;  // 3-6
    maxGrade: number;
  };
  
  /** 课标要求描述 */
  curriculumStandard: string;
  
  /** 核心概念列表 */
  coreConcepts: ScienceConcept[];
  
  /** 前置知识点（学习该知识点前需先掌握的内容） */
  prerequisites: string[];
  
  /** 后续知识点（该知识点为后续哪些知识打基础） */
  leadsTo: string[];
  
  /** 初中衔接知识点（该知识点对应初中哪个具体知识点） */
  middleSchoolBridge?: {
    subject: 'physics' | 'chemistry' | 'biology' | 'geography';
    knowledgePointIds: string[];
  };
  
  /** 关联日常生活场景 */
  lifeScenarios: string[];
  
  /** 可做的虚拟实验列表 */
  virtualExperiments: string[];
  
  /** 难度等级 1-5 */
  difficultyLevel: number;
  
  /** 课标认知层次 */
  cognitiveLevel: '了解' | '理解' | '应用' | '分析';
}

/**
 * 科学概念定义
 */
interface ScienceConcept {
  /** 概念名称 */
  name: string;
  
  /** 适龄化定义（小学语言） */
  childFriendlyDefinition: string;
  
  /** 标准学术定义 */
  formalDefinition: string;
  
  /** 生活实例 */
  lifeExamples: string[];
  
  /** 常见 misconceptions（学生易误解的点） */
  commonMisconceptions: Misconception[];
  
  /** 可视化辅助说明（图解描述，供渲染引擎使用） */
  visualAidDescription?: string;
}

/**
 * 科学误解记录
 */
interface Misconception {
  /** 误解描述 */
  misconception: string;
  
  /** 正确理解 */
  correctUnderstanding: string;
  
  /** 纠正引导策略 */
  correctionStrategy: string;
  
  /** 验证实验（如有） */
  verificationExperiment?: string;
}
```

### 2.4 数据库表结构

```sql
-- 科学知识点主表
CREATE TABLE science_knowledge_point (
    id              VARCHAR(32) PRIMARY KEY,
    name            VARCHAR(128) NOT NULL COMMENT '知识点名称',
    domain          VARCHAR(32) NOT NULL COMMENT '所属领域: material/life/earth_space/tech_eng',
    grade_min       TINYINT NOT NULL COMMENT '最低适用年级 3-6',
    grade_max       TINYINT NOT NULL COMMENT '最高适用年级 3-6',
    curriculum_std  TEXT NOT NULL COMMENT '课标要求描述',
    difficulty      TINYINT DEFAULT 3 COMMENT '难度 1-5',
    cognitive_level VARCHAR(16) DEFAULT '理解' COMMENT '认知层次',
    textbook_chapter_id VARCHAR(32) COMMENT '关联教材章节ID',
    status          TINYINT DEFAULT 1 COMMENT '1=启用 0=禁用',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_domain_grade (domain, grade_min, grade_max),
    INDEX idx_chapter (textbook_chapter_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='科学知识点主表';

-- 科学概念表
CREATE TABLE science_concept (
    id                VARCHAR(32) PRIMARY KEY,
    knowledge_point_id VARCHAR(32) NOT NULL COMMENT '所属知识点',
    name              VARCHAR(128) NOT NULL COMMENT '概念名称',
    child_definition  TEXT NOT NULL COMMENT '适龄化定义',
    formal_definition TEXT NOT NULL COMMENT '标准学术定义',
    life_examples     JSON COMMENT '生活实例列表',
    visual_aid_desc   TEXT COMMENT '可视化辅助描述',
    sort_order        INT DEFAULT 0,
    FOREIGN KEY (knowledge_point_id) REFERENCES science_knowledge_point(id),
    INDEX idx_kp (knowledge_point_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='科学概念表';

-- 科学误解表
CREATE TABLE science_misconception (
    id                   VARCHAR(32) PRIMARY KEY,
    concept_id           VARCHAR(32) NOT NULL COMMENT '关联概念',
    misconception_text   TEXT NOT NULL COMMENT '误解描述',
    correct_understanding TEXT NOT NULL COMMENT '正确理解',
    correction_strategy  TEXT NOT NULL COMMENT '纠正引导策略',
    verification_exp     VARCHAR(64) COMMENT '验证实验ID',
    FOREIGN KEY (concept_id) REFERENCES science_concept(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='科学常见误解表';

-- 虚拟实验定义表
CREATE TABLE science_virtual_experiment (
    id                VARCHAR(32) PRIMARY KEY,
    name              VARCHAR(128) NOT NULL COMMENT '实验名称',
    knowledge_point_id VARCHAR(32) NOT NULL COMMENT '关联知识点',
    domain            VARCHAR(32) NOT NULL COMMENT '所属领域',
    grade_level       TINYINT NOT NULL COMMENT '适用年级',
    description       TEXT NOT NULL COMMENT '实验描述',
    objective         TEXT NOT NULL COMMENT '学习目标',
    materials         JSON COMMENT '虚拟实验器材列表',
    steps             JSON NOT NULL COMMENT '实验步骤(结构化)',
    safety_warnings   JSON COMMENT '安全提示列表',
    expected_result   TEXT NOT NULL COMMENT '预期结果',
    explanation       TEXT NOT NULL COMMENT '科学原理说明',
    duration_min      INT DEFAULT 15 COMMENT '预估时长(分钟)',
    difficulty        TINYINT DEFAULT 2 COMMENT '难度 1-5',
    component_type    VARCHAR(64) COMMENT '对应客户端组件类型',
    status            TINYINT DEFAULT 1,
    created_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (knowledge_point_id) REFERENCES science_knowledge_point(id),
    INDEX idx_kp_grade (knowledge_point_id, grade_level)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='虚拟实验定义表';

-- 科学观察项目表
CREATE TABLE science_observation_project (
    id              VARCHAR(32) PRIMARY KEY,
    name            VARCHAR(128) NOT NULL COMMENT '观察项目名称',
    description     TEXT NOT NULL COMMENT '项目描述',
    domain          VARCHAR(32) NOT NULL COMMENT '所属领域',
    grade_level     TINYINT NOT NULL COMMENT '适用年级',
    duration_days   INT NOT NULL COMMENT '持续天数',
    observation_fields JSON NOT NULL COMMENT '观察记录字段定义',
    reminder_cron   VARCHAR(64) COMMENT '观察提醒cron表达式',
    guidance_prompts JSON COMMENT '阶段性引导问题列表',
    report_template TEXT COMMENT '观察报告模板',
    status          TINYINT DEFAULT 1,
```
