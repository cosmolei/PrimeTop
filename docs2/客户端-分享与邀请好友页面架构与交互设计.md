# 客户端-分享与邀请好友页面架构与交互设计

> 版本：v1.0 | 日期：2026-06-06 | 关联：分享与社交裂变系统、邀请裂变与用户增长服务

## 1. 概述

### 1.1 文档目的

本文档详细定义"分享邀请"功能的客户端页面架构、交互流程、组件设计、状态管理和错误处理策略，使开发人员可直接依据本文档进行编码实现。

### 1.2 范围

覆盖以下页面与功能：

| 页面/组件 | 说明 |
|-----------|------|
| 分享入口与弹窗 | 各页面触发分享的统一入口 |
| 分享卡片生成页 | 生成含海报/链接/口令的分享内容 |
| 邀请好友主页 | 展示邀请规则、进度、奖励 |
| 邀请记录列表 | 已邀请好友及奖励发放状态 |
| 新用户落地页 | 被邀请人打开链接后的承接页 |
| 邀请码/口令输入页 | 新用户填写邀请码的入口 |

### 1.3 设计原则

1. **轻量触达**：分享操作不超过 2 步完成
2. **适龄适配**：主分享面向家长端，学生端仅保留学习成果分享
3. **性能优先**：海报图片本地渲染，避免网络等待
4. **离线容错**：核心展示数据本地缓存，网络失败可降级展示

```dart
/// lib/widgets/share/share_bottom_sheet.dart

/// 分享弹窗组件
class ShareBottomSheet extends StatefulWidget {
  /// 分享业务类型
  final ShareBusinessType businessType;

  /// 业务数据 ID（如报告ID、成就ID）
  final String? businessId;

  /// 预加载的分享内容（可选，用于离线场景）
  final ShareContent? preloadedContent;

  const ShareBottomSheet({
    super.key,
    required this.businessType,
    this.businessId,
    this.preloadedContent,
  });

  /// 便捷调用方法
  static Future<ShareResult?> show(
    BuildContext context, {
    required ShareBusinessType businessType,
    String? businessId,
    ShareContent? preloadedContent,
  }) {
    return showModalBottomSheet<ShareResult>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (_) => ShareBottomSheet(
        businessType: businessType,
        businessId: businessId,
        preloadedContent: preloadedContent,
      ),
    );
  }

  @override
  State<ShareBottomSheet> createState() => _ShareBottomSheetState();
}

class _ShareBottomSheetState extends State<ShareBottomSheet> {
  ShareMediaType _selectedMedia = ShareMediaType.poster;
  ShareContent? _content;
  bool _isLoading = true;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _loadShareContent();
  }

  /// 加载分享内容
  /// 优先使用预加载内容，否则从服务端获取
  Future<void> _loadShareContent() async {
    if (widget.preloadedContent != null) {
      setState(() {
        _content = widget.preloadedContent;
        _isLoading = false;
      });
      return;
    }

    try {
      final content = await _fetchShareContent();
      setState(() {
        _content = content;
        _isLoading = false;
      });
    } catch (e) {
      setState(() {
        _errorMessage = _mapErrorMessage(e);
        _isLoading = false;
      });
    }
  }

  /// 从服务端获取分享内容
  Future<ShareContent> _fetchShareContent() async {
    final api = context.read<ShareApiService>();
    final response = await api.getShareContent(
      businessType: widget.businessType,
      businessId: widget.businessId,
    );

    // 如果是海报模式，本地渲染海报图片
    if (_selectedMedia == ShareMediaType.poster) {
      final posterPath = await _renderPoster(response);
      return response.copyWith(posterPath: posterPath);
    }

    return response;
  }

  /// 处理渠道点击
  Future<void> _handleChannelTap(ShareChannel channel) async {
    if (_content == null) return;

    final result = await _executeShare(channel, _content!);

    // 埋点上报
    _reportShareEvent(channel, result);

    if (mounted) {
      Navigator.of(context).pop(result);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      constraints: BoxConstraints(
        maxHeight: MediaQuery.of(context).size.height * 0.7,
      ),
      decoration: const BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          _buildHeader(),
          if (_isLoading)
            const Expanded(child: Center(child: LoadingIndicator()))
          else if (_errorMessage != null)
            Expanded(child: _buildErrorState())
          else
            ...[
            _buildPreview(),
            _buildMediaTypeTabs(),
            _buildChannelGrid(),
          ],
          _buildCancelButton(),
        ],
      ),
    );
  }
}
```

#### 3.3.2 API 请求

```dart
/// lib/services/invite_api_service.dart

class InviteApiService {
  final Dio _dio;

  InviteApiService(this._dio);

  /// 获取邀请进度
  Future<InviteProgress> getInviteProgress() async {
    final response = await _dio.get('/api/v1/invite/progress');
    return InviteProgress.fromJson(response.data['data']);
  }

  /// 获取最近邀请记录
  Future<List<InviteRecord>> getRecentRecords({int limit = 5}) async {
    final response = await _dio.get(
      '/api/v1/invite/records',
      queryParameters: {'limit': limit},
    );
    return (response.data['data'] as List)
        .map((e) => InviteRecord.fromJson(e))
        .toList();
  }

  /// 分页获取邀请记录
  Future<PageResult<InviteRecord>> getInviteRecords({
    int page = 1,
    int pageSize = 20,
    InviteRecordStatus? status,
  }) async {
    final response = await _dio.get(
      '/api/v1/invite/records',
      queryParameters: {
        'page': page,
        'pageSize': pageSize,
        if (status != null) 'status': status.name,
      },
    );
    return PageResult.fromJson(
      response.data,
      fromJsonT: (e) => InviteRecord.fromJson(e as Map<String, dynamic>),
    );
  }

  /// 获取分享内容
  Future<ShareContent> getShareContent({
    required ShareBusinessType businessType,
    String? businessId,
  }) async {
    final response = await _dio.post(
      '/api/v1/share/content',
      data: {
        'businessType': businessType.name,
        if (businessId != null) 'businessId': businessId,
      },
    );
    return ShareContent.fromJson(response.data['data']);
  }
}
```

```dart
/// lib/pages/share/invite_records_page.dart

class InviteRecordsPage extends StatefulWidget {
  const InviteRecordsPage({super.key});

  @override
  State<InviteRecordsPage> createState() => _InviteRecordsPageState();
}

class _InviteRecordsPageState extends State<InviteRecordsPage> {
  final PagingController<int, InviteRecord> _pagingController =
      PagingController(firstPageKey: 1);

  InviteRecordStatus? _statusFilter;

  @override
  void initState() {
    super.initState();
    _pagingController.addPageRequestListener(_fetchPage);
  }

  Future<void> _fetchPage(int pageKey) async {
    try {
      final result = await context.read<InviteApiService>().getInviteRecords(
            page: pageKey,
            pageSize: 20,
            status: _statusFilter,
          );

      if (result.hasMore) {
        _pagingController.appendPage(result.items, pageKey + 1);
      } else {
        _pagingController.appendLastPage(result.items);
      }
    } catch (e) {
      _pagingController.error = e;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('邀请记录')),
      body: Column(
        children: [
          _buildStatusFilter(),
          Expanded(
            child: RefreshIndicator(
              onRefresh: () => Future.sync(() => _pagingController.refresh()),
              child: PagedListView<int, InviteRecord>(
                pagingController: _pagingController,
```

```dart
/// lib/services/deep_link_handler.dart

class DeepLinkHandler {
  final NavigationService _navigation;
  final AuthService _auth;
  final InviteCodeService _inviteCode;

  /// 处理深链接
  /// uri 格式: primetop://share/invite/landing?code=ABC123
  Future<void> handleDeepLink(Uri uri) async {
    if (uri.path == '/share/invite/landing') {
      final code = uri.queryParameters['code'];
      if (code == null || code.isEmpty) {
        _navigation.navigateTo('/home');
        return;
      }
      await _handleInviteLanding(code);
    }
  }

  /// 处理邀请落地
  Future<void> _handleInviteLanding(String code) async {
    // 缓存邀请码（无论是否登录都先缓存）
    await _inviteCode.cachePendingInviteCode(code);

    if (_auth.isLoggedIn) {
      // 已登录：尝试绑定邀请码
      final success = await _inviteCode.bindInviteCode(code);
      if (success) {
        _navigation.navigateTo(
          '/share/invite/code',
          arguments: {'code': code, 'autoBind': true},
        );
      } else {
        // 绑定失败（可能已绑定过）
        _navigation.navigateTo('/home');
      }
    } else {
      // 未登录：跳转注册页，携带邀请码
      _navigation.navigateTo(
        '/auth/register',
        arguments: {'inviteCode': code},
      );
    }
  }
}
```

```dart
/// lib/services/invite_code_service.dart

class InviteCodeService {
  final SharedPreferences _prefs;
  final InviteApiService _api;
  final AuthService _auth;

  static const _codeKey = 'pending_invite_code';
  static const _codeTimeKey = 'pending_invite_code_time';

  /// 邀请码有效期：7天
  static const _codeValidityMs = 7 * 24 * 60 * 60 * 1000;

  /// 获取缓存的邀请码
  String? getCachedCode() {
    final code = _prefs.getString(_codeKey);
    if (code == null) return null;

    final cacheTime = _prefs.getInt(_codeTimeKey) ?? 0;
    final now = DateTime.now().millisecondsSinceEpoch;

    // 过期清除
    if (now - cacheTime > _codeValidityMs) {
      _prefs.remove(_codeKey);
      _prefs.remove(_codeTimeKey);
      return null;
    }

    return code;
  }

  /// 绑定邀请码
  Future<InviteBindResult> bindInviteCode(String code) async {
    final result = await _api.bindInviteCode(code);

    if (result.success) {
      // 绑定成功，清除缓存
      await _prefs.remove(_codeKey);
      await _prefs.remove(_codeTimeKey);
    }

    return result;
  }

  /// 检查是否有待处理邀请码，在注册成功后调用
  Future<void> checkAndBindPendingCode() async {
    if (!_auth.isLoggedIn) return;

    final code = getCachedCode();
    if (code == null) return;

    try {
      await bindInviteCode(code);
    } catch (_) {
      // 静默失败，不阻断注册流程
    }
  }
}
```

```dart
class ShareDispatcher {
  final WechatService _wechat;
  final QQService _qq;

  /// 执行分享
  Future<ShareResult> execute(
    ShareChannel channel,
    ShareContent content,
  ) async {
    try {
      final result = await switch (channel) {
        ShareChannel.wechat => _shareToWechat(content),
        ShareChannel.wechatMoments => _shareToWechatMoments(content),
        ShareChannel.qq => _shareToQQ(content),
        ShareChannel.sms => _shareViaSMS(content),
        ShareChannel.copyLink => _copyLink(content),
        ShareChannel.saveImage => _saveImage(content),
        ShareChannel.more => _shareViaSystemSheet(content),
      };
      return result;
    } on PlatformException catch (e) {
      return ShareResult(
        success: false,
        channel: channel,
        errorMessage: _mapPlatformError(e),
      );
    }
  }

  Future<ShareResult> _shareToWechat(ShareContent content) async {
    // 检查微信是否安装
    if (!await _wechat.isInstalled()) {
      return ShareResult(
        success: false,
        channel: ShareChannel.wechat,
        errorMessage: '请先安装微信',
      );
    }

    if (content.mediaType == ShareMediaType.poster && content.posterPath != null) {
      await _wechat.shareImage(
        imagePath: content.posterPath!,
        scene: WechatScene.session,
      );
    } else {
      await _wechat.shareWebPage(
        url: content.shareUrl!,
        title: content.title,
        description: content.description,
        thumbnailUrl: content.thumbnailUrl,
        scene: WechatScene.session,
      );
    }

    return ShareResult(success: true, channel: ShareChannel.wechat);
  }

  Future<ShareResult> _copyLink(ShareContent content) async {
    final text = content.mediaType == ShareMediaType.code
        ? '${content.title}\n邀请码：${content.inviteCode}\n${content.shareUrl}'
        : '${content.title}\n${content.shareUrl}';

    await Clipboard.setData(ClipboardData(text: text));
    return ShareResult(success: true, channel: ShareChannel.copyLink);
  }

  Future<ShareResult> _saveImage(ShareContent content) async {
    if (content.posterPath == null) {
      return ShareResult(
        success: false,
        channel: ShareChannel.saveImage,
        errorMessage: '当前分享内容无海报图片',
      );
    }

    final saved = await Gal.putImage(content.posterPath!);
    return ShareResult(
      success: saved,
      channel: ShareChannel.saveImage,
      errorMessage: saved ? null : '保存失败',
    );
  }

  Future<ShareResult> _shareViaSystemSheet(ShareContent content) async {
    final text = '${content.title}\n${content.shareUrl ?? ''}';
    await Share.share(text);
    return ShareResult(success: true, channel: ShareChannel.more);
  }
```

```dart
/// lib/widgets/share/share_error_handler.dart

class ShareErrorHandler {
  /// 将异常映射为用户可见的错误消息
  static String mapErrorMessage(Object error) {
    if (error is DioException) {
      final code = error.response?.data?['code'];
      return switch (code) {
        'INVALID_INVITE_CODE' => '邀请码不存在，请检查后重试',
        'CODE_ALREADY_BOUND' => '你已绑定过邀请码',
        'CODE_EXPIRED' => '该邀请码已过期',
        'SELF_INVITE' => '不能使用自己的邀请码',
        'INVITE_NOT_ACTIVE' => '邀请活动暂未开始',
        _ => '网络异常，请稍后重试',
      };
    }

    if (error is PlatformException) {
      return switch (error.code) {
        'WECHAT_NOT_INSTALLED' => '请先安装微信客户端',
        'SHARE_CANCELLED' => '',
        'SAVE_DENIED' => '需要相册权限才能保存图片',
        _ => '分享失败，请重试',
      };
    }

    return '操作失败，请稍后重试';
  }

  /// 是否需要权限引导
  static bool needsPermissionGuide(Object error) {
    if (error is PlatformException) {
      return error.code == 'SAVE_DENIED';
    }
    return false;
  }
}
```

## 8. 性能优化

### 8.1 海报渲染优化

```dart
/// lib/widgets/share/poster_cache.dart

/// 海报渲染缓存
class PosterCache {
  static final _cache = <String, String>{};

  /// 获取缓存的海报路径
  static String? get(String key) => _cache[key];

  /// 缓存海报路径
  static void put(String key, String path) {
    // 清理旧缓存（保留最近 5 张）
    if (_cache.length >= 5) {
      final oldest = _cache.keys.first;
      final oldPath = _cache.remove(oldest);
      if (oldPath != null) {
        File(oldPath).deleteSync(ignoreDotDotDot: true);
      }
    }
    _cache[key] = path;
  }

  /// 清理所有缓存
  static void clear() {
    for (final path in _cache.values) {
      File(path).deleteSync(ignoreDotDotDot: true);
    }
    _cache.clear();
  }
}
```

```dart
/// lib/services/share_content_filter.dart

class ShareContentFilter {
  /// 过滤分享内容中的敏感信息
  /// 避免在分享海报/链接中暴露学生真实姓名、学校等
  static String filterStudentInfo(String text, StudentProfile profile) {
    return text
        .replaceAll(profile.realName, profile.displayName)
        .replaceAll(profile.schoolName ?? '', '');
  }

  /// 海报中只展示脱敏昵称
  static String maskNickname(String nickname) {

  }
}
```
