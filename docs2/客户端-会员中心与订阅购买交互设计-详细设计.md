# 客户端-会员中心与订阅购买交互设计 - 详细设计文档

## 1. 模块概述

### 1.1 定位

会员中心与订阅购买是 PrimeTop 客户端的核心商业转化模块，负责会员套餐展示与对比、支付购买流程、订阅生命周期管理、权益展示与引导。本模块是连接「用户价值感知」与「付费行为」的关键桥梁，直接影响会员转化率和续费率。

> **核心原则**（来自《盈利模式与定价策略》）：
> - 教育优先：不利用学习焦虑做营销
> - 价格展示仅限家长端，学生端仅展示权益与升级引导
> - 所有支付必须经过家长验证（密码/生物识别）
> - 透明公平，无隐藏消费

```dart
/// 会员相关路由守卫
class SubscriptionGuard extends RouteGuard {
  @override
  Future<GuardResult> check(RouteData route) async {
    // 1. 登录检查
    if (!authProvider.isLoggedIn) {
      return GuardResult.redirect('/login', args: {'redirect': route.uri.toString()});
    }

    // 2. 支付页面需家长验证
    if (route.uri.path == '/subscription/payment') {
      final isMinor = authProvider.isMinor;
      if (isMinor) {
        final verified = await _showParentVerification();
        if (!verified) {
          return GuardResult.block('需要家长验证才能进行支付');
        }
      }
    }

    // 3. 订阅管理页需有订阅记录
    if (route.uri.path == '/profile/membership') {
      // 允许访问，但页面内部处理无订阅状态
    }

    return GuardResult.pass();
  }
}
```

```dart
/// 全局会员状态 Provider
@riverpod
class MembershipNotifier extends _$MembershipNotifier {
  @override
  Future<UserMembership?> build() async {
    // 从本地缓存先返回
    final cached = _localCache.getMembership();
    if (cached != null) {
      // 后台静默刷新
      _refreshInBackground();
      return cached;
    }
    // 缓存为空，同步请求
    return _fetchMembership();
  }

  Future<UserMembership?> _fetchMembership() async {
    final result = await _api.getMembershipStatus();
    _localCache.saveMembership(result);
    return result;
  }

  void _refreshInBackground() {
    _api.getMembershipStatus().then((result) {
      _localCache.saveMembership(result);
      state = AsyncData(result);
    });
  }

  /// 支付成功后主动刷新
  Future<void> refreshAfterPayment() async {
    state = const AsyncLoading();
    state = AsyncData(await _fetchMembership());
  }

  /// 监听会员变更事件（WebSocket / 轮询）
  void onMembershipChanged(MembershipChangedEvent event) {
    refreshAfterPayment();
  }
}
```

### 5.2 家长验证拦截器

```dart
/// 支付前家长验证
class PaymentParentVerifier {
  /// 检查是否需要家长验证
  /// 规则：未满 18 岁的用户必须通过家长验证
  static Future<bool> verify(BuildContext context) async {
    final auth = context.read(authProvider);
    if (!auth.isMinor) return true; // 成年人直接通过

    // 弹出家长验证弹窗
    final result = await showDialog<ParentVerifyResult>(
      context: context,
      barrierDismissible: false,
      builder: (_) => const ParentVerifyDialog(),
    );

    if (result == ParentVerifyResult.success) return true;

    // 验证失败
    if (result == ParentVerifyResult.failed) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('家长验证未通过，请让家长协助操作')),
      );
    }
    return false;
  }
}

/// 家长验证弹窗
class ParentVerifyDialog extends StatefulWidget {
  @override
  State<ParentVerifyDialog> createState() => _ParentVerifyDialogState();
}

class _ParentVerifyDialogState extends State<ParentVerifyDialog> {
  final _passwordController = TextEditingController();
  bool _isBiometric = false;
  int _attemptCount = 0;

  static const _maxAttempts = 5;

  @override
  void initState() {
    super.initState();
    _checkBiometric();
  }

  Future<void> _checkBiometric() async {
    final available = await BiometricUtil.isAvailable();
    if (available && mounted) {
      setState(() => _isBiometric = true);
      _authenticateWithBiometric();
    }
  }

  Future<void> _authenticateWithBiometric() async {
    final result = await BiometricUtil.authenticate(
      reason: '请家长验证身份以确认支付',
    );
    if (mounted) {
      Navigator.pop(context, result
          ? ParentVerifyResult.success
          : ParentVerifyResult.failed);
    }
  }

  Future<void> _verifyPassword() async {
    if (_attemptCount >= _maxAttempts) {
      Navigator.pop(context, ParentVerifyResult.locked);
      return;
    }

    final password = _passwordController.text;
    final result = await context.read(parentVerifyProvider.notifier)
        .verifyPassword(password);

    if (result) {
      Navigator.pop(context, ParentVerifyResult.success);
    } else {
      _attemptCount++;
      setState(() {});
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('密码错误，还可尝试 ${_maxAttempts - _attemptCount} 次')),
      );
    }
```

```dart
/// 支付流程 Provider
@riverpod
class PaymentFlow extends _$PaymentFlow {
  @override
  PaymentFlowState build() => const PaymentIdle();

  static const _maxPollCount = 10;  // 最大轮询次数
  static const _pollInterval = Duration(seconds: 2);

  Future<void> startPayment({
    required String planCode,
    required String channel,
  }) async {
    // 阶段1：创建订单
    state = const PaymentCreatingOrder();

    try {
      final order = await _api.createOrder(
        planCode: planCode,
        channel: channel,
      );

      state = PaymentWaitingSdk(orderNo: order.orderNo, channel: channel);

      // 阶段2：调用支付 SDK
      final sdkResult = await _invokePaymentSdk(
        order: order,
        channel: channel,
      );

      // 阶段3：确认支付结果
      state = PaymentConfirming(orderNo: order.orderNo);

      if (sdkResult.isCompleted) {
        // SDK 报告成功，轮询服务端确认
        await _pollOrderStatus(order.orderNo);
      } else if (sdkResult.isCancelled) {
        // 用户取消
        state = PaymentFailed(
          orderNo: order.orderNo,
          errorMessage: '支付已取消',
          canRetry: true,
        );
      } else {
        // SDK 报告失败或未知，仍轮询兜底
        await _pollOrderStatus(order.orderNo);
      }
    } catch (e) {
      state = PaymentFailed(
        orderNo: '',
        errorMessage: '创建订单失败：$e',
        canRetry: true,
      );
    }
  }

  /// 调用对应平台支付 SDK
  Future<PaymentSdkResult> _invokePaymentSdk({
    required OrderCreationResult order,
    required String channel,
  }) async {
    switch (channel) {
      case 'wechat_pay':
        return _wechatPaySdk.pay(
          partnerId: order.partnerId!,
          prepayId: order.prepayId!,
          nonceStr: order.nonceStr!,
          timeStamp: order.timeStamp!,
          packageValue: order.packageValue!,
          sign: order.sign!,
        );
      case 'alipay':
        return _alipaySdk.pay(order.payInfo!);
      case 'apple_iap':
        return _appleIapSdk.purchase(order.productId!);
      default:
        throw UnsupportedError('不支持的支付渠道: $channel');
    }
  }

  /// 轮询订单状态
  Future<void> _pollOrderStatus(String orderNo) async {
    for (var i = 0; i < _maxPollCount; i++) {
      state = PaymentConfirming(orderNo: orderNo, pollingCount: i);

      await Future.delayed(_pollInterval);

      final orderStatus = await _api.getOrderStatus(orderNo);

      switch (orderStatus.status) {
        case 'paid':
          // 支付成功！
          await ref.read(membershipNotifierProvider.notifier).refreshAfterPayment();
          state = PaymentSuccess(
            orderNo: orderNo,
            planName: orderStatus.planName!,
            expiresAt: orderStatus.expiresAt!,
          );
          _trackPaymentSuccess(orderNo, orderStatus);
          return;
        case 'closed':
          state = PaymentFailed(
            orderNo: orderNo,
            errorMessage: '订单已关闭',
            canRetry: true,
          );
          return;
        case 'pending':
          continue; // 继续轮询
      }
    }

    // 超时
    state = PaymentTimeout(orderNo: orderNo);
  }

  void _trackPaymentSuccess(String orderNo, OrderStatusResult result) {
    AnalyticsService.track('biz_pay_result', properties: {
      'order_id': orderNo,
      'result': 'success',
      'plan_type': result.planCode,
      'amount_cent': result.amountCents,
      'pay_channel': result.paymentChannel,
    });
  }
}
```

```dart
class _RestorePurchasePageState extends ConsumerState<RestorePurchasePage> {
  bool _isRestoring = false;
  String? _resultMessage;

  Future<void> _restore() async {
    setState(() {
      _isRestoring = true;
      _resultMessage = null;
    });

    try {
      // 1. 调用 StoreKit 恢复
      final transactions = await _appleIapSdk.restorePurchases();

      if (transactions.isEmpty) {
        setState(() {
          _resultMessage = '未找到可恢复的购买记录';
          _isRestoring = false;
        });
        return;

      for (final tx in transactions) {
        await _api.verifyAppleReceipt(
          receiptData: tx.receiptData,
          transactionId: tx.transactionId,
        );

      // 3. 刷新会员状态

        _isRestoring = false;

      if (mounted) context.go('/profile/membership');
```

```dart
class PaymentErrorHandler {

    if (error is TimeoutException) {
      _showPaymentConfirmDialog(context);
      return;
```

| 支付结果伪造 | 不信任客户端 SDK 回调，以服务端支付回调为准 |
| 订单篡改 | 订单号由服务端生成，客户端不可修改金额 |
| 截屏录屏 | 支付确认页禁止截屏（Android FlagSecure / iOS 屏幕录制保护） |
| 中间人攻击 | 支付相关接口强制 HTTPS + Certificate Pinning |
| 会员状态缓存过期 | 本地缓存 TTL 30分钟 + 服务端事件推送主动失效 |
| Apple IAP 越狱破解 | StoreKit 2 服务端验证 + 设备完整性检测 |
| 重复支付 | 订单创建前检查是否有 pending 订单 |

### 15.2 平台特殊处理

```dart
class PlatformPaymentConfig {
  /// iOS 特殊规则
  /// 1. 必须使用 Apple IAP，不能提供第三方支付

  /// 3. 审核期间不能展示价格比较/折扣信息
  static bool get forceAppleIap => Platform.isIOS;

  /// Android 规则
  /// 1. 支持微信支付、支付宝
  /// 2. 如果上架 Google Play，需额外支持 Google Play Billing
  static List<PaymentMethod> get androidMethods => [
    PaymentMethod.wechatPay,
    PaymentMethod.alipay,
  ];

  /// Apple 审核模式
  /// 审核期间隐藏折扣标签和价格对比
  static bool get isReviewMode {
    // 从服务端配置读取
    return false;
  }
}
```
