## 1. 模块概述

### 1.1 功能定位

本服务负责从跨模块的学习行为事件流中，挖掘学生学习活动序列模式（Learning Sequence Patterns），分析不同学习路径与学习效果之间的因果关系，为自适应推荐引擎和学习规划引擎提供数据驱动的路径优化建议。

**核心区别**：

| 已有服务 | 关注点 | 本服务关注点 |
|---------|--------|-------------|
| 练习答题行为数据采集 | 采集单次答题行为 | 分析跨模块行为序列模式 |
| 学习行为事件流引擎 | 事件传递与级联处理 | 从事件流中挖掘模式 |
| 学生学习状态建模 | 当前状态快照评估 | 行为轨迹的时间序列分析 |
| 自适应推荐引擎 | 推荐什么内容 | 以什么顺序学习最有效 |
| 学习规划服务 | 生成学习计划 | 验证哪些计划模式真正有效 |

### 1.2 核心能力

1. **行为序列构建**：将离散学习事件按时间组织为有序学习路径
2. **模式挖掘**：使用序列模式挖掘算法发现高频学习路径
3. **效果归因**：关联学习路径与知识掌握度变化，量化路径效果
4. **群体对比**：按学段、学科、能力层次对比不同群体路径差异
5. **个性化建议**：基于学生画像匹配最优学习序列，输出路径建议
6. **异常路径检测**：识别低效学习路径（如反复跳过讲解直接看答案）

### 1.3 业务价值

- **优化推荐策略**：从"推荐正确内容"升级到"推荐正确顺序"
- **指导计划生成**：让学习规划引擎基于真实效果数据生成路径
- **预警低效行为**：及时检测并干预无效学习模式
- **评估教学设计**：验证教材编排顺序和课程设计是否合理

## 4. 核心算法设计

### 4.1 行为序列构建算法

### 4.1.1 会话切分策略

```python
"""
学习会话切分算法
将连续的学习行为按时间和语义边界切分为有意义的会话单元
"""

from datetime import timedelta
from enum import Enum

class SessionSplitStrategy(Enum):
    TIME_GAP = "time_gap"            # 按时间间隔切分
    SUBJECT_CHANGE = "subject_change"  # 按学科切换切分
    KP_CHANGE = "kp_change"          # 按知识点切换切分
    LONG_PAUSE = "long_pause"        # 长时间暂停后切分


def split_into_sessions(
    behaviors: list[dict],
    max_session_gap_seconds: int = 1800,  # 30分钟无操作 → 新会话
    max_session_duration_seconds: int = 7200,  # 单次会话上限2小时
    subject_change_splits: bool = True,
) -> list[list[dict]]:
    """
    将行为列表切分为学习会话
    
    Args:
        behaviors: 按时间排序的学习行为列表
        max_session_gap_seconds: 最大允许间隔(秒)
        max_session_duration_seconds: 单次会话最大时长(秒)
        subject_change_splits: 学科切换是否强制切分
    
    Returns:
        会话列表，每个会话是一个行为列表
    """
    if not behaviors:
        return []
    
    sessions = []
    current_session = [behaviors[0]]
    session_start_ts = behaviors[0]["start_ts"]
    last_ts = behaviors[0]["start_ts"]
    last_subject = behaviors[0].get("meta", {}).get("subject_code")
    
    for behavior in behaviors[1:]:
        current_ts = behavior["start_ts"]
        time_gap = (current_ts - last_ts).total_seconds()
        current_subject = behavior.get("meta", {}).get("subject_code")
        
        # 切分条件判断
        should_split = False
        split_reason = None
        
        # 条件1: 时间间隔超过阈值
        if time_gap > max_session_gap_seconds:
            should_split = True
            split_reason = SessionSplitStrategy.TIME_GAP
        
        # 条件2: 会话时长超过上限
        elif (current_ts - session_start_ts).total_seconds() > max_session_duration_seconds:
            should_split = True
            split_reason = SessionSplitStrategy.LONG_PAUSE
        
        # 条件3: 学科切换(可选)
        elif subject_change_splits and last_subject and current_subject and last_subject != current_subject:
            should_split = True
            split_reason = SessionSplitStrategy.SUBJECT_CHANGE
        
        if should_split:
            sessions.append(current_session)
            current_session = [behavior]
            session_start_ts = current_ts
        else:
            current_session.append(behavior)
        
        last_ts = current_ts
        last_subject = current_subject
    
    if current_session:
        sessions.append(current_session)
    
    return sessions
```

```python
BEHAVIOR_TO_CATEGORY = {
    "VIEW_CHAPTER": "INPUT",
    "READ_KNOWLEDGE_POINT": "INPUT",
    "WATCH_VIDEO": "INPUT",
    "AI_ASK": "INTERACTION",
    "AI_FOLLOWUP": "INTERACTION",
    "PHOTO_SEARCH": "INTERACTION",
    "PRACTICE_START": "PRACTICE",
    "PRACTICE_ANSWER": "PRACTICE",
    "PRACTICE_REVIEW": "REVIEW",
    "ERROR_ADD": "REVIEW",
    "ERROR_REVIEW": "REVIEW",
    "ERROR_REDO": "PRACTICE",
    "PLAN_TASK_COMPLETE": "PRACTICE",
    "EXAM_START": "PRACTICE",
    "EXAM_ANSWER": "PRACTICE",
    "EXAM_REVIEW": "REVIEW",
    "MEMORIZE_PRACTICE": "OUTPUT",
    "WRITING_SUBMIT": "OUTPUT",
    "KNOWLEDGE_MAP_BROWSE": "NAVIGATION",
    "LISTEN_PRACTICE": "INPUT",
    "ORAL_PRACTICE": "OUTPUT",
}


def normalize_sequence(
    session_behaviors: list[dict],
    granularity: str = "category",  # "category" | "type" | "hybrid"
) -> list[str]:
    """
    将行为会话标准化为符号序列
    
    Args:
        session_behaviors: 单个会话的行为列表
        granularity: 序列粒度
            - "category": 按行为大类 ["INPUT", "PRACTICE", "REVIEW"]
            - "type": 按具体行为类型 ["AI_ASK", "PRACTICE_ANSWER"]
            - "hybrid": 混合表示，保留大类但附加修饰符
    
    Returns:
        标准化序列
    """
    if granularity == "category":
        # 合并连续相同大类
        sequence = []
        for b in session_behaviors:
            cat = BEHAVIOR_TO_CATEGORY.get(b["type"], "NAVIGATION")
            if not sequence or sequence[-1] != cat:
                sequence.append(cat)
        return sequence
    
    elif granularity == "type":
        return [b["type"] for b in session_behaviors]
    
    elif granularity == "hybrid":
        # 混合表示：大类+子类修饰
        sequence = []
        for b in session_behaviors:
            cat = BEHAVIOR_TO_CATEGORY.get(b["type"], "NAVIGATION")
            # 添加修饰信息：是否涉及AI、是否首次学习
            modifiers = []
            if "AI" in b["type"] or b["type"] in ("PHOTO_SEARCH",):
                modifiers.append("AI")
            if b.get("meta", {}).get("first_attempt", False):
                modifiers.append("NEW")
            
            symbol = f"{cat}[{'|'.join(modifiers)}]" if modifiers else cat
            if not sequence or sequence[-1] != symbol:
                sequence.append(symbol)
        return sequence
    
    return []
```

### 4.2 序列模式挖掘

#### 4.2.1 PrefixSpan 算法封装

```python
"""
PrefixSpan 序列模式挖掘封装
从标准化学习序列中挖掘频繁子序列
"""

from collections import defaultdict
from typing import Sequence


class PrefixSpanMiner:
    """PrefixSpan 频繁序列模式挖掘"""
    
    def __init__(
        self,
        min_support_rate: float = 0.05,   # 最小支持率(占总学生数)
        max_pattern_length: int = 8,       # 最大模式长度
        min_pattern_length: int = 2,       # 最小模式长度
    ):
        self.min_support_rate = min_support_rate
        self.max_pattern_length = max_pattern_length
        self.min_pattern_length = min_pattern_length
    
    def mine(
        self,
        sequences: list[list[str]],   # 标准化序列列表
        student_count: int = None,
    ) -> list[dict]:
        """
        挖掘频繁序列模式
        
        Returns:
            模式列表 [{"sequence": [...], "support": int, "support_rate": float}]
        """
        if not sequences:
            return []
        
        total = student_count or len(sequences)
        min_support = max(3, int(total * self.min_support_rate))
        
        # 使用投影数据库的 PrefixSpan 算法
        patterns = self._prefix_span(
            sequences, 
            prefix=[], 
            min_support=min_support,
            max_len=self.max_pattern_length,
        )
        
        # 过滤过短的模式
        patterns = [p for p in patterns if len(p["sequence"]) >= self.min_pattern_length]
        
        # 按支持度降序排列
        patterns.sort(key=lambda x: (-x["support"], len(x["sequence"])))
        
        return patterns
    
    def _prefix_span(
        self,
        database: list[list[str]],
        prefix: list[str],
        min_support: int,
        max_len: int,
    ) -> list[dict]:
        """PrefixSpan 递归实现"""
        if len(prefix) >= max_len:
            return []
        
        results = []
        
        # 统计所有可能的下一个项的支持度
        item_support = defaultdict(int)
        for sequence in database:
            # 每个序列中每个项只计一次
            unique_items = set(sequence)
            for item in unique_items:
                item_support[item] += 1
        
        # 筛选频繁项
        frequent_items = {
            item for item, count in item_support.items() if count >= min_support
        }
        
        for item in sorted(frequent_items):
            new_prefix = prefix + [item]
            new_support = item_support[item]
            
            if len(new_prefix) >= self.min_pattern_length:
                results.append({
                    "sequence": new_prefix,
                    "support": new_support,
                    "support_rate": new_support / len(database),
                })
            
            # 构建投影数据库
            projected_db = self._build_projected_db(database, item)
            
            if projected_db:
                sub_patterns = self._prefix_span(
                    projected_db, new_prefix, min_support, max_len
                )
                results.extend(sub_patterns)
        
        return results
    
    def _build_projected_db(
        self, 
        database: list[list[str]], 
        item: str,
    ) -> list[list[str]]:
        """构建投影数据库：对每个序列，找到item第一次出现位置后的子序列"""
        projected = []
        for sequence in database:
            try:
                idx = sequence.index(item)
                suffix = sequence[idx + 1:]
                if suffix:  # 只保留非空后缀
                    projected.append(suffix)
            except ValueError:
                continue
        return projected


# 使用示例
def mine_patterns_for_cohort(
    sequences: list[list[str]],
    cohort_label: str,
) -> list[dict]:
    """为特定群体挖掘模式"""
    miner = PrefixSpanMiner(
        min_support_rate=0.05,
        max_pattern_length=6,
        min_pattern_length=2,
    )
    patterns = miner.mine(sequences)
    
    for p in patterns:
        p["cohort"] = cohort_label
        p["pattern_label"] = label_pattern(p["sequence"])
    
    return patterns


def label_pattern(sequence: list[str]) -> str:
    """为模式生成人类可读标签"""
    label_map = {
        ("INPUT", "PRACTICE"): "学后练",
        ("INPUT", "INTERACTION", "PRACTICE"): "学问练",
        ("INPUT", "PRACTICE", "REVIEW"): "学练复",
        ("PRACTICE", "REVIEW", "PRACTICE"): "练复练",
        ("INPUT", "INTERACTION"): "学中问",
        ("PRACTICE", "INTERACTION"): "练中问",
        ("INTERACTION", "PRACTICE"): "问后练",
        ("INPUT", "PRACTICE", "INPUT"): "学练学",
        ("INTERACTION", "REVIEW"): "问后复",
    }
    return label_map.get(tuple(sequence), "-".join(sequence))
```

#### 4.2.2 N-Gram 补充分析

```python
"""
N-Gram 短序列模式分析
作为 PrefixSpan 的补充，专注于2-gram和3-gram转移分析
"""

from collections import Counter


class NGramAnalyzer:
    """学习行为 N-Gram 分析器"""
    
    @staticmethod
    def extract_ngrams(
        sequences: list[list[str]],
        n: int = 2,
    ) -> dict[tuple[str, ...], int]:
        """提取n-gram频率"""
        counter = Counter()
        for seq in sequences:
            for i in range(len(seq) - n + 1):
                ngram = tuple(seq[i:i + n])
                counter[ngram] += 1
        return dict(counter)
    
    @staticmethod
    def calc_transition_matrix(
        sequences: list[list[str]],
    ) -> dict[str, dict[str, float]]:
        """
        计算2-gram转移概率矩阵
        返回: {"INPUT": {"PRACTICE": 0.45, "INTERACTION": 0.30, ...}, ...}
        """
        transitions = Counter()  # (from, to) → count
        from_counts = Counter()  # from → total
        
        for seq in sequences:
            for i in range(len(seq) - 1):
                transitions[(seq[i], seq[i + 1])] += 1
                from_counts[seq[i]] += 1
        
        matrix = {}
        for (src, dst), count in transitions.items():
            if src not in matrix:
                matrix[src] = {}
            matrix[src][dst] = count / from_counts[src]
        
        return matrix
    
    @staticmethod
    def find_surprising_transitions(
        observed: dict[str, dict[str, float]],
        baseline: dict[str, dict[str, float]],
        threshold: float = 0.1,
    ) -> list[dict]:
        """
        找出与基线相比显著偏离的转移
        
        用于对比高效果群体 vs 整体的行为差异
        """
        surprising = []
        for src in observed:
            for dst, prob in observed[src].items():
                base_prob = baseline.get(src, {}).get(dst, 0)
                diff = prob - base_prob
                if abs(diff) >= threshold:
                    surprising.append({
                        "from": src,
                        "to": dst,
                        "observed_rate": round(prob, 4),
                        "baseline_rate": round(base_prob, 4),
                        "difference": round(diff, 4),
                        "direction": "higher" if diff > 0 else "lower",
                    })
        
        surprising.sort(key=lambda x: -abs(x["difference"]))
        return surprising
```

### 4.3 效果归因分析

```python
"""
学习路径效果归因分析
使用倾向性评分匹配(PSM)控制混杂变量，量化路径的真实因果效应
"""

import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import NearestNeighbors


class PathEffectivenessAnalyzer:
    """学习路径效果分析器"""
    
    # 混杂变量（影响路径选择和学习效果的共同因素）
    COVARIATES = [
        "prior_mastery",          # 先前掌握度
        "grade_level",            # 年级
        "stage_code",             # 学段
        "study_days_total",       # 累计学习天数
        "avg_daily_duration",     # 日均学习时长
        "prior_accuracy_rate",    # 先前正确率
        "total_questions_done",   # 累计答题数
        "membership_level",       # 会员等级
    ]
    
    def __init__(self, min_sample_size: int = 30):
        self.min_sample_size = min_sample_size
    
    def analyze_pattern_effect(
        self,
        pattern_sequence: list[str],
        all_sequences: list[dict],  # [{"student_id":..., "sequence":[...], "mastery_gain":..., "covariates":{...}}]
    ) -> dict:
        """
        分析特定学习模式的因果效应
        
        Args:
            pattern_sequence: 目标模式 (如 ["INPUT", "PRACTICE", "REVIEW"])
            all_sequences: 全量学生序列数据(包含效果指标和协变量)
        
        Returns:
            {"psm_effect": float, "ci_low": float, "ci_high": float, "sample_treatment": int, ...}
        """
        # 1. 划分处理组和对照组
        treatment = []
        control = []
        
        for record in all_sequences:
            if self._contains_pattern(record["sequence"], pattern_sequence):
                treatment.append(record)
            else:
                control.append(record)
        
        if len(treatment) < self.min_sample_size or len(control) < self.min_sample_size:
            return {
                "psm_effect": None,
                "reason": "insufficient_sample",
                "sample_treatment": len(treatment),
                "sample_control": len(control),
            }
        
        # 2. 提取协变量矩阵
        X_t = np.array([[r["covariates"][c] for c in self.COVARIATES] for r in treatment])
        X_c = np.array([[r["covariates"][c] for c in self.COVARIATES] for r in control])
        
        # 3. 估计倾向性评分
        X_all = np.vstack([X_t, X_c])
        y_all = np.concatenate([np.ones(len(treatment)), np.zeros(len(control))])
        
        lr = LogisticRegression(max_iter=1000, random_state=42)
        lr.fit(X_all, y_all)
        
        ps_t = lr.predict_proba(X_t)[:, 1]
        ps_c = lr.predict_proba(X_c)[:, 1]
        
        # 4. 最近邻匹配
        nn = NearestNeighbors(n_neighbors=1, metric="euclidean")
        nn.fit(ps_c.reshape(-1, 1))
        _, indices = nn.kneighbors(ps_t.reshape(-1, 1))
        
        matched_control_indices = indices.flatten()
        
        # 5. 计算平均处理效应(ATT)
        y_t = np.array([r["mastery_gain"] for r in treatment])
        y_c_matched = np.array([control[i]["mastery_gain"] for i in matched_control_indices])
        
        att = float(np.mean(y_t - y_c_matched))
        
        # 6. Bootstrap 置信区间
        ci_low, ci_high = self._bootstrap_ci(y_t, y_c_matched, n_boot=1000)
        
        # 7. 补充描述性统计
        avg_time_t = np.mean([r.get("duration_seconds", 0) for r in treatment])
        avg_time_c = np.mean([r.get("duration_seconds", 0) for r in control])
        
        return {
            "psm_effect": round(att, 4),
            "ci_low": round(ci_low, 4),
            "ci_high": round(ci_high, 4),
            "significant": ci_low > 0 or ci_high < 0,
            "sample_treatment": len(treatment),
            "sample_control": len(control),
            "avg_mastery_gain_treatment": round(float(np.mean(y_t)), 4),
            "avg_mastery_gain_control": round(float(np.mean(y_c_matched)), 4),
            "avg_time_treatment_seconds": int(avg_time_t),
            "avg_time_control_seconds": int(avg_time_c),
            "time_efficiency": round(att / (avg_time_t / 3600), 4) if avg_time_t > 0 else None,
        }
    
    def _contains_pattern(self, sequence: list[str], pattern: list[str]) -> bool:
        """检查序列是否包含指定模式（子序列匹配）"""
        if len(pattern) > len(sequence):
            return False
        for i in range(len(sequence) - len(pattern) + 1):
            if sequence[i:i + len(pattern)] == pattern:
                return True
        return False
    
    def _bootstrap_ci(
        self, 
        y_t: np.ndarray, 
        y_c: np.ndarray, 
        n_boot: int = 1000,
        confidence: float = 0.95,
    ) -> tuple[float, float]:
        """Bootstrap 置信区间"""
        n = len(y_t)
        effects = []
        for _ in range(n_boot):
            idx = np.random.choice(n, size=n, replace=True)
            eff = np.mean(y_t[idx] - y_c[idx])
            effects.append(eff)
        
        alpha = (1 - confidence) / 2
        ci_low = float(np.percentile(effects, alpha * 100))
        ci_high = float(np.percentile(effects, (1 - alpha) * 100))
        return ci_low, ci_high
```

### 4.4 异常路径检测

```python
"""
异常学习路径检测
基于规则引擎和统计模型双重检测
"""

from dataclasses import dataclass, field
from datetime import timedelta


@dataclass
class AnomalyDetectionRule:
    """异常检测规则定义"""
    rule_id: str
    anomaly_type: str
    severity: str
    description: str
    check_fn: callable
    window_size: int = 10  # 检测窗口大小(行为数)


class AnomalyPathDetector:
    """异常学习路径检测器"""
    
    def __init__(self):
        self.rules = self._init_rules()
    
    def _init_rules(self) -> list[AnomalyDetectionRule]:
        """初始化检测规则"""
        return [
            AnomalyDetectionRule(
                rule_id="R001",
                anomaly_type="SKIP_EXPLANATION",
                severity="MEDIUM",
                description="反复跳过讲解直接查看答案",
                check_fn=self._check_skip_explanation,
                window_size=8,
            ),
            AnomalyDetectionRule(
                rule_id="R002",
                anomaly_type="RANDOM_JUMP",
                severity="LOW",
                description="无逻辑频繁跳转，行为切换频率过高",
                check_fn=self._check_random_jump,
                window_size=15,
            ),
            AnomalyDetectionRule(
                rule_id="R003",
                anomaly_type="STUCK_LOOP",
                severity="HIGH",
                description="同一知识点反复学习但掌握度无提升",
                check_fn=self._check_stuck_loop,
                window_size=20,
            ),
            AnomalyDetectionRule(
                rule_id="R004",
                anomaly_type="SURFACE_SKIM",
                severity="MEDIUM",
                description="浅层浏览，极低停留时长高切换率",
                check_fn=self._check_surface_skim,
                window_size=12,
            ),
            AnomalyDetectionRule(
                rule_id="R005",
                anomaly_type="GUESS_PATTERN",
                severity="HIGH",
                description="猜测式答题，极短时间大量作答",
                check_fn=self._check_guess_pattern,
                window_size=10,
            ),
            AnomalyDetectionRule(
                rule_id="R006",
                anomaly_type="HELP_ABUSE",
                severity="MEDIUM",
                description="过度依赖AI提问，缺乏自主思考",
                check_fn=self._check_help_abuse,
                window_size=10,
            ),
        ]
    
    def detect(
        self, 
        behaviors: list[dict],
        mastery_history: list[dict] = None,
    ) -> list[dict]:
        """
        对行为序列执行全部规则检测
        
        Returns:
            检测到的异常列表
        """
        anomalies = []
        for rule in self.rules:
            window = behaviors[-rule.window_size:] if len(behaviors) > rule.window_size else behaviors
            result = rule.check_fn(window, mastery_history or [])
            if result:
                anomalies.append({
                    "rule_id": rule.rule_id,
                    "anomaly_type": rule.anomaly_type,
                    "severity": rule.severity,
                    "description": rule.description,
                    "evidence": result,
                    "behavior_count": len(window),
                })
        return anomalies
    
    def _check_skip_explanation(self, behaviors: list[dict], _) -> dict | None:
        """检测：连续3次以上跳过讲解直接答题/看答案"""
        skip_streak = 0
        max_streak = 0
        skip_events = []
        
        for b in behaviors:
            if b["type"] == "PRACTICE_ANSWER":
                meta = b.get("meta", {})
                if meta.get("viewed_explanation") is False and meta.get("correct") is False:
                    skip_streak += 1
                    skip_events.append({
                        "seq": b["seq"],
                        "question_id": meta.get("question_id"),
                        "time_spent": b.get("duration", 0),
                    })
                    max_streak = max(max_streak, skip_streak)
                else:
                    skip_streak = 0
            else:
                skip_streak = 0
        
        if max_streak >= 3:
            return {"max_skip_streak": max_streak, "skip_events": skip_events[-3:]}
        return None
    
    def _check_random_jump(self, behaviors: list[dict], _) -> dict | None:
        """检测：行为切换频率异常高"""
        if len(behaviors) < 5:
            return None
        
        transitions = 0
        kp_changes = 0
        prev_type = None
        prev_kp = None
        
        for b in behaviors:
            if prev_type and b["type"] != prev_type:
                transitions += 1
            curr_kp = b.get("kp_id")
            if prev_kp and curr_kp and curr_kp != prev_kp:
                kp_changes += 1
            prev_type = b["type"]
            prev_kp = curr_kp
        
        transition_rate = transitions / (len(behaviors) - 1)
        kp_change_rate = kp_changes / (len(behaviors) - 1)
        
        # 阈值：切换率 > 0.7 且 知识点切换率 > 0.5
        if transition_rate > 0.7 and kp_change_rate > 0.5:
            return {
                "transition_rate": round(transition_rate, 3),
                "kp_change_rate": round(kp_change_rate, 3),
                "total_transitions": transitions,
            }
        return None
    
    def _check_stuck_loop(self, behaviors: list[dict], mastery_history: list[dict]) -> dict | None:
        """检测：同一知识点反复学习但掌握度停滞"""
        if not mastery_history:
            return None
        
        kp_activity = {}
        for b in behaviors:
            kp = b.get("kp_id")
            if kp:
                kp_activity[kp] = kp_activity.get(kp, 0) + 1
        
        for kp, count in kp_activity.items():
            if count >= 5:
                # 检查该知识点的掌握度历史
                kp_mastery = [m for m in mastery_history if m.get("kp_id") == kp]
                if len(kp_mastery) >= 3:
                    recent_mastery = [m["mastery"] for m in kp_mastery[-3:]]
                    mastery_change = recent_mastery[-1] - recent_mastery[0]
                    if abs(mastery_change) < 0.05:  # 掌握度几乎没变
                        return {
                            "kp_id": kp,
                            "activity_count": count,
                            "mastery_values": recent_mastery,
                            "mastery_change": round(mastery_change, 4),
                        }
        return None
    
    def _check_surface_skim(self, behaviors: list[dict], _) -> dict | None:
        """检测：浅层浏览模式"""
        reading_behaviors = [b for b in behaviors if b["type"] in ("VIEW_CHAPTER", "READ_KNOWLEDGE_POINT")]
        if len(reading_behaviors) < 3:
            return None
        
        avg_duration = sum(b.get("duration", 0) for b in reading_behaviors) / len(reading_behaviors)
        unique_kps = len(set(b.get("kp_id") for b in reading_behaviors if b.get("kp_id")))
        
        # 阈值：平均停留 < 30秒 且 涉及知识点 > 3个
        if avg_duration < 30 and unique_kps > 3:
            return {
                "avg_duration_seconds": round(avg_duration, 1),
                "unique_kps_browsed": unique_kps,
                "reading_count": len(reading_behaviors),
            }
        return None
    
    def _check_guess_pattern(self, behaviors: list[dict], _) -> dict | None:
        """检测：猜测式答题"""
        answers = [b for b in behaviors if b["type"] == "PRACTICE_ANSWER"]
        if len(answers) < 5:
            return None
        
        fast_wrong = 0
        for a in answers:
            meta = a.get("meta", {})
            duration = a.get("duration", 0)
            if duration < 10 and meta.get("correct") is False:  # <10秒且答错
                fast_wrong += 1
        
        fast_wrong_rate = fast_wrong / len(answers)
        avg_duration = sum(a.get("duration", 0) for a in answers) / len(answers)
        
        # 阈值：快速错误率 > 50% 且 平均答题 < 15秒
        if fast_wrong_rate > 0.5 and avg_duration < 15:
            return {
                "fast_wrong_rate": round(fast_wrong_rate, 3),
                "avg_answer_duration": round(avg_duration, 1),
                "total_answers": len(answers),
                "fast_wrong_count": fast_wrong,
            }
        return None
    
    def _check_help_abuse(self, behaviors: list[dict], _) -> dict | None:
        """检测：过度依赖AI，不自主尝试"""
        ai_asks = [b for b in behaviors if b["type"] in ("AI_ASK", "AI_FOLLOWUP")]
        practice = [b for b in behaviors if b["type"] == "PRACTICE_ANSWER"]
        
        if len(behaviors) < 5:
            return None
        
        ai_ratio = len(ai_asks) / len(behaviors)
        practice_ratio = len(practice) / len(behaviors)
        
        # 阈值：AI提问占比 > 50% 且 练习占比 < 15%
        if ai_ratio > 0.5 and practice_ratio < 0.15:
            return {
                "ai_ask_count": len(ai_asks),
                "practice_count": len(practice),
                "ai_ratio": round(ai_ratio, 3),
                "practice_ratio": round(practice_ratio, 3),
            }
        return None
```

```python
"""
每日学习序列构建任务
02:00 执行，处理前一天的全部学习行为
"""

def daily_sequence_build_job(target_date: str):
    """
    1. 从事件流和业务表中拉取前一天的全量学习行为
    2. 按学生+学科维度组织
    3. 会话切分与序列标准化
    4. 关联掌握度数据
    5. 批量写入 learning_sequence 表
    """
    log.info(f"开始构建 {target_date} 学习序列")
    
    # Step 1: 拉取全量学习行为
    behaviors = fetch_all_behaviors(target_date)
    log.info(f"拉取到 {len(behaviors)} 条行为记录")
    
    # Step 2: 按学生+学科分组
    grouped = group_by_student_subject(behaviors)
    
    # Step 3: 逐组构建序列
    sequences_to_write = []
    for (student_id, subject_code), raw_behaviors in grouped.items():
        # 按时间排序
        raw_behaviors.sort(key=lambda b: b["start_ts"])
        
        # 会话切分
        sessions = split_into_sessions(raw_behaviors)
        
        # 对每个会话标准化
        all_behaviors_in_day = []
        for session in sessions:
            normalized = normalize_sequence(session, granularity="category")
            all_behaviors_in_day.extend(session)  # 保留原始行为
        
        # 计算行为大类分布
        category_dist = calculate_category_distribution(all_behaviors_in_day)
        
        # 获取掌握度数据(从掌握度计算引擎)
        student_profile = get_student_profile(student_id)
        mastery_before = get_mastery_snapshot(student_id, subject_code, target_date, "BEFORE")
        mastery_after = get_mastery_snapshot(student_id, subject_code, target_date, "AFTER")
        
        # 计算效果指标
        answers = [b for b in all_behaviors_in_day if b["type"] in ("PRACTICE_ANSWER", "EXAM_ANSWER")]
        accuracy = sum(1 for a in answers if a.get("meta", {}).get("correct")) / max(len(answers), 1)
        
        sequences_to_write.append({
            "student_id": student_id,
            "sequence_date": target_date,
            "subject_code": subject_code,
            "behaviors": serialize_behaviors(all_behaviors_in_day),
            "session_count": len(sessions),
            "total_duration": sum(b.get("duration", 0) for b in all_behaviors_in_day),
            "category_distribution": category_dist,
            "transition_count": count_transitions(all_behaviors_in_day),
            "mastery_before": mastery_before,
            "mastery_after": mastery_after,
            "mastery_delta": (mastery_after - mastery_before) if mastery_before and mastery_after else None,
            "accuracy_rate": accuracy,
            "question_count": len(answers),
            "grade_level": student_profile["grade_level"],
            "stage_code": student_profile["stage_code"],
        })
    
    # Step 4: 批量写入
    batch_upsert("learning_sequence", sequences_to_write, 
                 conflict_fields=["student_id", "sequence_date", "subject_code"])
    
    log.info(f"学习序列构建完成，写入 {len(sequences_to_write)} 条记录")
```

```python
"""
每周模式挖掘与效果评估任务
周日 03:00 执行
"""

def weekly_pattern_mining_job():
    """每周模式挖掘主流程"""
    
    # Step 1: 拉取近30天的全量学习序列
    sequences = fetch_sequences_from_db(days=30)
    
    # Step 2: 按学段+学科分组
    cohorts = group_by_stage_subject(sequences)
    
    all_patterns = []
    for (stage, subject), cohort_sequences in cohorts.items():
        if len(cohort_sequences) < 50:
            continue  # 样本不足，跳过
        
        # Step 3: 标准化序列
        normalized = [normalize_sequence(s["behaviors"], "category") for s in cohort_sequences]
        
        # Step 4: PrefixSpan 挖掘
        miner = PrefixSpanMiner(min_support_rate=0.05, max_pattern_length=6)
        patterns = miner.mine(normalized, student_count=len(cohort_sequences))
        
        # Step 5: 为每个模式附加标签
        for p in patterns:
            p["pattern_label"] = label_pattern(p["sequence"])
            p["stage_code"] = stage
            p["subject_code"] = subject
        
        all_patterns.extend(patterns)
    
    # Step 6: 效果归因分析(对Top模式执行PSM)
    effectiveness_analyzer = PathEffectivenessAnalyzer(min_sample_size=30)
    
    for pattern in all_patterns[:200]:  # 只对Top 200模式做详细分析
        cohort_with_meta = prepare_psm_data(cohort_sequences, pattern["sequence"])
        effect = effectiveness_analyzer.analyze_pattern_effect(
            pattern["sequence"],
            cohort_with_meta,
        )
        pattern["effectiveness"] = effect
        pattern["effectiveness_score"] = compute_effectiveness_score(effect)
    
    # Step 7: 持久化
    upsert_patterns(all_patterns)
    upsert_path_effectiveness(all_patterns)
    
    log.info(f"模式挖掘完成，发现 {len(all_patterns)} 个模式，详细分析 {len(all_patterns[:200])} 个")


def compute_effectiveness_score(effect: dict) -> float:
    """
    综合效果评分计算
    结合：PSM因果效应、样本量、时间效率
    """
    if effect.get("psm_effect") is None:
        return 0.0
    
    psm_effect = effect["psm_effect"]          # 因果效应(掌握度提升)
    sample = effect["sample_treatment"]
    time_eff = effect.get("time_efficiency", 0) # 提升度/小时
    
    # 归一化各维度
    effect_score = min(psm_effect / 0.2, 1.0)         # 掌握度提升0.2为满分
    confidence = min(sample / 100, 1.0)                 # 100样本为满分置信
    efficiency_score = min(time_eff / 0.5, 1.0)         # 0.5/小时为满分效率
    
    # 加权
    return round(0.5 * effect_score + 0.3 * efficiency_score + 0.2 * confidence, 3)
```

```python
    available_time: int,
) -> dict | None:
    """
    为学生匹配最优学习路径模式
    
    匹配逻辑：
    1. 按学段+学科+年级筛选候选模式
    2. 按当前掌握度区间筛选(初始/进阶/巩固)
    3. 按效果评分排序
    4. 考虑时间约束
    """
    student = get_student_profile(student_id)
    
    candidates = query_patterns(
        stage_code=student["stage_code"],
        subject_code=subject_code,
        grade_level=student["grade_level"],
        min_effectiveness=0.3,
    )
    
    # 按掌握度区间过滤
    if current_mastery < 0.3:
        mastery_band = "INITIAL"
    elif current_mastery < 0.7:
        mastery_band = "PROGRESSING"
    else:
        mastery_band = "CONSOLIDATING"
    
    # 对候选模式计算适配度
    scored = []
    for pattern in candidates:
        fit_score = compute_fit_score(pattern, student, current_mastery, available_time)
        scored.append((pattern, fit_score))
    
    scored.sort(key=lambda x: -x[1])
    
    return scored[0][0] if scored and scored[0][1] > 0.5 else None


def compute_fit_score(
    pattern: dict,
    student: dict,
    current_mastery: float,
    available_time: int,
) -> float:
    """
    计算模式对学生的适配度评分(0-1)
    
    因素：
    - 效果评分(40%)
    - 时间匹配度(25%)
    - 掌握度阶段匹配(20%)
    - 学习风格匹配(15%)
    """
    # 效果评分
    eff_score = pattern.get("effectiveness_score", 0)
    
    # 时间匹配度
    pattern_time = pattern.get("avg_duration", 0) // 60  # 分钟
    if pattern_time <= available_time:
        time_score = pattern_time / max(available_time, 1)
    else:
        time_score = max(0, 1 - (pattern_time - available_time) / available_time * 0.5)
    
    # 掌握度阶段匹配
    pattern_pre_mastery = pattern.get("avg_pre_mastery", 0.5)
    mastery_diff = abs(pattern_pre_mastery - current_mastery)
    mastery_score = max(0, 1 - mastery_diff * 2)
    
    # 学习风格匹配(基于学生学习风格画像)
    style_match = check_style_match(pattern, student.get("learning_style"))
    
    return (
        0.40 * eff_score +
        0.25 * time_score +
        0.20 * mastery_score +
        0.15 * style_match
    )
```
