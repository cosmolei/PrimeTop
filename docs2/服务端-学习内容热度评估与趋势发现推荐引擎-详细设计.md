# 服务端-学习内容热度评估与趋势发现推荐引擎-详细设计

## 1. 概述

### 1.1 功能定位

本引擎负责实时追踪平台内所有学习内容（微课视频、知识点讲解、练习题组、专题文章、试卷等）的学生消费行为，计算多维热度分数，识别上升趋势内容，并为首页、学科页面、个性化推荐等场景提供热门内容数据源。

### 1.2 与现有模块的边界

| 模块 | 职责 | 与本引擎的关系 |
| --- | --- | --- |
| 教育热点追踪与考试命题趋势预测分析引擎 | 追踪外部教育政策、考试命题趋势 | 本引擎消费其产出的"外部热点标签"，为内容热度加权 |
| 教育内容触达转化漏斗与学习内容消费深度多维分析引擎 | 分析内容推荐→点击→完成各环节转化率 | 本引擎向其提供内容热度维度数据用于漏斗对照 |
| 首页内容卡片推荐与聚合编排服务 | 首页卡片拼装与展示 | 本引擎为其提供"热门榜单"数据源 |
| 统一埋点数据治理与事件质量保障引擎 | 埋点事件质量管理 | 本引擎消费其清洗后的标准行为事件流 |
| 推荐内容多样性保障与信息茧房破壁引擎 | 推荐结果去重与多样性 | 本引擎的 trending feed 作为多样性补充召回源 |

### 1.3 设计目标

1. **实时性**：内容热度分数延迟不超过 5 分钟
2. **可扩展**：支持千万级内容条目、百万级 DAU 的热度计算
3. **多维排序**：支持按学科、学段、年级、区域等维度切片
4. **个性化**：相同热点事件下为不同学生生成差异化热门 Feed
5. **抗刷**：识别并过滤非正常浏览行为对热度的干扰

## 3. 核心算法设计

### 3.1 热度分数计算模型

综合热度分数采用加权多因子模型，覆盖"广度×深度×互动"三个维度：

```
heat_score = W_view × f_view(UV_24h)
           + W_complete × f_complete(complete_count_24h)
           + W_engage × f_engage(like + share + comment)
           + W_trend × momentum_bonus
           + W_external × external_hot_boost
           + W_seasonal × seasonal_boost
```

**默认权重配置（可通过管理后台调整）：**

| 因子 | 权重 | 说明 |
| --- | --- | --- |
| W_view | 1.0 | 浏览广度 |
| W_complete | 2.5 | 完成深度（权重最高，防止标题党） |
| W_engage | 1.5 | 互动信号 |
| W_trend | 1.2 | 上升动量加成 |
| W_external | 0.8 | 外部热点关联 |
| W_seasonal | 0.5 | 季节性加成 |

**子函数使用对数压缩，避免头部内容霸榜：**

```python
import math

def f_view(uv_count: int) -> float:
    """浏览因子：对数压缩，防止超高UV内容垄断榜首"""
    if uv_count <= 0:
        return 0.0
    return math.log1p(uv_count)  # ln(1 + uv)

def f_complete(complete_count: int) -> float:
    """完成因子：完成学习是最强质量信号"""
    if complete_count <= 0:
        return 0.0
    return math.log1p(complete_count) * 1.5  # 额外加权

def f_engage(likes: int, shares: int, comments: int) -> float:
    """互动因子：不同互动行为权重不同"""
    # 分享 > 评论 > 点赞，因为分享代表高认可度
    weighted = likes * 1.0 + shares * 3.0 + comments * 2.0
    if weighted <= 0:
        return 0.0
    return math.log1p(weighted)

def momentum_bonus(score_now: float, score_prev: float) -> float:
    """趋势动量加成：环比增长越快，加成越高"""
    if score_prev <= 0:
        # 新内容首次上榜，给予初始动量
        return score_now * 0.15
    ratio = (score_now - score_prev) / score_prev
    # 动量上限为当前分的 30%，防止暴增
    return max(0.0, min(ratio * score_now * 0.3, score_now * 0.3))
```

### 3.2 趋势方向判定

```python
from enum import Enum

class TrendDirection(Enum):
    RISING = "RISING"
    STABLE = "STABLE"
    DECLINING = "DECLINING"
    NEW = "NEW"

def calculate_trend(score_now: float, score_prev: float, 
                    has_prev: bool) -> tuple[TrendDirection, float]:
    """
    判定内容趋势方向与动量值。
    
    Returns:
        (direction, momentum): 趋势方向和动量值
    """
    if not has_prev or score_prev == 0:
        return TrendDirection.NEW, score_now * 0.15
    
    change_ratio = (score_now - score_prev) / score_prev
    
    if change_ratio > 0.20:
        return TrendDirection.RISING, change_ratio
    elif change_ratio < -0.20:
        return TrendDirection.DECLINING, change_ratio
    else:
        return TrendDirection.STABLE, change_ratio
```

### 3.3 滑动窗口热度聚合

基于 Redis 时间桶实现滑动窗口聚合，避免全量扫描。

```python
from datetime import datetime, timedelta
from typing import List

class HeatWindowAggregator:
    """滑动窗口热度聚合器"""
    
    def __init__(self, redis_client):
        self.redis = redis_client
    
    def aggregate_24h(self, content_id: int) -> dict:
        """聚合最近24小时的热度数据"""
        now = datetime.utcnow()
        total_views = 0
        total_completes = 0
        total_likes = 0
        total_shares = 0
        total_comments = 0
        
        # 遍历24个小时桶
        for i in range(24):
            bucket_time = now - timedelta(hours=i)
            bucket_key = f"heat:realtime:{content_id}:{bucket_time.strftime('%Y%m%d%H')}"
            
            data = self.redis.hgetall(bucket_key)
            if data:
                total_views += int(data.get(b'views', 0))
                total_completes += int(data.get(b'completes', 0))
                total_likes += int(data.get(b'likes', 0))
                total_shares += int(data.get(b'shares', 0))
                total_comments += int(data.get(b'comments', 0))

            bucket_time = now - timedelta(hours=i)
            uv_keys.append(f"heat:uv:{content_id}:{bucket_time.strftime('%Y%m%d%H')}")
        
        uv_24h = self.redis.pfcount(*uv_keys) if uv_keys else 0
        
        return {
```

### 3.4 反作弊过滤

```python
class HeatAntiSpamFilter:
    """热度反作弊过滤器"""
    
    # 单用户单内容每小时行为上限
    MAX_VIEWS_PER_HOUR = 5
    # 单用户单内容每天行为上限
    MAX_VIEWS_PER_DAY = 20
    # 同一IP每分钟对同一内容的行为上限
    MAX_SAME_IP_PER_MIN = 10
    
    def is_valid_event(self, event: dict) -> bool:
        """判断事件是否为正常行为"""
        user_id = event.get('user_id')
        content_id = event.get('content_id')
        event_type = event.get('event_type')
        
        # 1. 检查用户-内容 频次
        user_content_key = f"heat:antispam:user:{user_id}:{content_id}:{event_type}"
        current_hour_key = f"{user_content_key}:{datetime.utcnow().strftime('%Y%m%d%H')}"
        
        count = self.redis.incr(current_hour_key)
        if count == 1:
            self.redis.expire(current_hour_key, 3600)
        if count > self.MAX_VIEWS_PER_HOUR:
            return False
        
        # 2. 检查 IP 频次
        ip = event.get('client_ip')
        if ip:
            ip_key = f"heat:antispam:ip:{ip}:{content_id}:{datetime.utcnow().strftime('%Y%m%d%H%M')}"
            ip_count = self.redis.incr(ip_key)
            if ip_count == 1:
                self.redis.expire(ip_key, 60)
            if ip_count > self.MAX_SAME_IP_PER_MIN:
                return False
        
        # 3. 检查设备指纹（如有）
        device_id = event.get('device_id')
        if device_id:
            # 类似检查逻辑
            pass
        
        return True
```

### 5.2 事件处理主流程

```python
import json
from kafka import KafkaConsumer
from concurrent.futures import ThreadPoolExecutor

class HeatEventConsumer:
    """热度事件消费者：从Kafka消费学习行为事件，写入Redis和MySQL"""
    
    def __init__(self, redis_client, mysql_pool, kafka_config):
        self.redis = redis_client
        self.mysql = mysql_pool
        self.consumer = KafkaConsumer(
            'learning-behavior-events',
            bootstrap_servers=kafka_config['bootstrap_servers'],
            group_id='heat-engine-group',
            value_deserializer=lambda m: json.loads(m.decode('utf-8')),
            enable_auto_commit=False,
            max_poll_records=500,
        )
        self.executor = ThreadPoolExecutor(max_workers=8)
        self.antispam = HeatAntiSpamFilter(redis_client)
    
    def run(self):
        """主消费循环"""
        for message in self.consumer:
            try:
                event = message.value
                self._process_event(event)
                self.consumer.commit()
            except Exception as e:
                # 记录错误，不提交offset，让Kafka重试
                logger.error(f"Failed to process heat event: {event}, error: {e}")
    
    def _process_event(self, event: dict):
        """处理单个热度事件"""
        content_id = event.get('content_id')
        user_id = event.get('user_id')
        event_type = event.get('event_type')
        event_time_str = event.get('event_time')
        
        if not content_id or not event_type:
            return
        
        # 映射事件类型到热度字段
        heat_field_map = {
            'VIEW': 'views',
            'COMPLETE': 'completes',
            'LIKE': 'likes',
            'SHARE': 'shares',
            'COMMENT': 'comments',
        }
        
        field = heat_field_map.get(event_type)
        if not field:
            return  # 非热度相关事件，忽略
        
        # 反作弊检查
        if not self.antispam.is_valid_event(event):
            logger.warning(f"Antispam blocked event: user={user_id}, "
                          f"content={content_id}, type={event_type}")
            return
        
        # 事件去重（event_id 唯一性保证）
        event_id = event.get('event_id')
        if event_id and not self._check_and_mark_event(event_id):
            return  # 重复事件
        
        # 写入 Redis 实时计数
        bucket = self._get_time_bucket(event_time_str)
        redis_key = f"heat:realtime:{content_id}:{bucket}"
        self.redis.hincrby(redis_key, field, 1)
        self.redis.expire(redis_key, 172800)  # 48h TTL
        
        # 写入 UV（HyperLogLog）
        uv_key = f"heat:uv:{content_id}:{bucket}"
        self.redis.pfadd(uv_key, str(user_id))
        self.redis.expire(uv_key, 172800)
        
        # 异步写入 MySQL 事件流水
        self.executor.submit(self._write_event_log, event)
    
    def _get_time_bucket(self, event_time_str: str) -> str:
        """将时间转换为小时桶 key"""
        from datetime import datetime
        dt = datetime.fromisoformat(event_time_str.replace('Z', '+00:00'))
        return dt.strftime('%Y%m%d%H')
    
    def _check_and_mark_event(self, event_id: str) -> bool:
        """事件去重检查（Redis SET + TTL）"""
        key = f"heat:event:dedup:{event_id}"
        result = self.redis.set(key, '1', nx=True, ex=86400)
        return result is not None
    
    def _write_event_log(self, event: dict):
        """异步写入事件流水表"""
        sql = """
            INSERT INTO content_heat_event_log 
                (event_id, content_id, content_type, user_id, user_grade, 
                 user_school_id, event_type, event_time, event_date, extra)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
            ON DUPLICATE KEY UPDATE event_id = event_id
        """
        params = (
            event.get('event_id'),
            event.get('content_id'),
            event.get('content_type'),
            event.get('user_id'),
            event.get('user_grade'),
            event.get('user_school_id'),
            event.get('event_type'),
            event.get('event_time'),
            event.get('event_time', '')[:10],
            json.dumps(event.get('extra', {}), ensure_ascii=False),
        )
        with self.mysql.get_connection() as conn:
            conn.execute(sql, params)
```

定时任务每小时执行一次全量热度计算。

```python
from datetime import datetime, timedelta

class HeatScoreCalculator:
    """热度分数定时计算器"""
    
    BATCH_SIZE = 500  # 每批处理500个内容
    
    def __init__(self, redis_client, mysql_pool, config):
        self.redis = redis_client
        self.mysql = mysql_pool
        self.config = config
        self.aggregator = HeatWindowAggregator(redis_client)
    
    def run_hourly(self):
        """每小时执行一次热度计算"""
        logger.info("Starting hourly heat score calculation")
        start_time = datetime.utcnow()
        
        # 1. 获取需要计算的内容ID列表
        active_content_ids = self._get_active_content_ids()
        logger.info(f"Found {len(active_content_ids)} active content items")
        
        # 2. 分批计算热度
        snapshot_time = start_time.replace(minute=0, second=0, microsecond=0)
        
        for i in range(0, len(active_content_ids), self.BATCH_SIZE):
            batch = active_content_ids[i:i + self.BATCH_SIZE]
            self._calculate_batch(batch, snapshot_time)
        
        # 3. 更新榜单
        self._update_rankings(snapshot_time)
        
        # 4. 清理过期数据
        self._cleanup_expired_data()
        
        elapsed = (datetime.utcnow() - start_time).total_seconds()
        logger.info(f"Heat calculation completed in {elapsed:.1f}s")
    
    def _get_active_content_ids(self) -> list:
        """获取最近24小时有行为的内容ID"""
        # 方法1: 从 Redis 扫描有热度的内容（通过索引ZSet）
        recent_active = set()
        
        # 扫描最近24小时有行为的内容
        now = datetime.utcnow()
        for i in range(24):
            bucket = (now - timedelta(hours=i)).strftime('%Y%m%d%H')
            # 使用 SCAN 避免阻塞（生产环境用 HSCAN + pipeline）
            pattern = f"heat:realtime:*:{bucket}"
            for key in self.redis.scan_iter(match=pattern, count=200):
                # 提取 content_id
                parts = key.decode().split(':')
                if len(parts) >= 4:
                    content_id = int(parts[2])
                    recent_active.add(content_id)
        
        return list(recent_active)
    
    def _calculate_batch(self, content_ids: list, snapshot_time: datetime):
        """批量计算热度分数"""
        for content_id in content_ids:
            try:
                self._calculate_single(content_id, snapshot_time)
            except Exception as e:
                logger.error(f"Error calculating heat for content {content_id}: {e}")
    
    def _calculate_single(self, content_id: int, snapshot_time: datetime):
        """计算单个内容的热度"""
        # 1. 聚合24小时窗口数据
        metrics_24h = self.aggregator.aggregate_24h(content_id)
        
        # 2. 聚合7天窗口数据
        metrics_7d = self.aggregator.aggregate_7d(content_id)
        
        # 3. 获取内容元信息
        meta = self._get_content_meta(content_id)
        if not meta:
            return
        
        # 4. 获取上一周期热度分
        prev_score = self._get_prev_heat_score(content_id)
        
        # 5. 计算各因子
        view_factor = f_view(metrics_24h['uv_24h'])
        complete_factor = f_complete(metrics_24h['completes_24h'])
        engage_factor = f_engage(
            metrics_24h['likes_24h'],
            metrics_24h['shares_24h'],
            metrics_24h['comments_24h']
        )
        
        # 6. 获取外部加权
        external_boost = self._get_external_boost(content_id, meta)
        seasonal_boost = self._get_seasonal_boost(meta)
        
        # 7. 计算原始分数
        raw_score = (
            self.config.W_VIEW * view_factor
            + self.config.W_COMPLETE * complete_factor
            + self.config.W_ENGAGE * engage_factor
            + self.config.W_EXTERNAL * external_boost
            + self.config.W_SEASONAL * seasonal_boost
        )
        
        # 8. 计算趋势
        direction, momentum = calculate_trend(
            raw_score, prev_score, has_prev=(prev_score > 0)
        )
        
        # 9. 叠加趋势加成
        trend_bonus = momentum_bonus(raw_score, prev_score) if prev_score > 0 else 0
        final_score = raw_score + self.config.W_TREND * trend_bonus
        
        # 10. 写入快照
        self._save_heat_snapshot(
            content_id=content_id,
            meta=meta,
            metrics=metrics_24h,
            metrics_7d=metrics_7d,
            heat_score=final_score,
            prev_score=prev_score,
            direction=direction,
            momentum=momentum,
            external_boost=external_boost,
            seasonal_boost=seasonal_boost,
            snapshot_time=snapshot_time,
        )
        
        # 11. 更新 Redis ZSet 排行榜
        self._update_redis_rankings(
            content_id, meta, final_score, direction
        )
```

## 6. 个性化热门推荐策略

### 6.1 个性化排序逻辑

个性化热门 Feed 不是简单返回全局榜单，而是在热门候选集中根据用户画像重排：

```python
class PersonalizedTrendingRecommender:
    """个性化热门推荐器"""
    
    def recommend(self, user_id: int, context: dict, limit: int = 10) -> list:
        """
        生成个性化热门推荐。
        
        策略：
        1. 从全局/学科/年级榜单中召回 Top 200 候选
        2. 排除用户已学过的内容
        3. 按用户学科偏好加权重排
        4. 插入"上升最快"内容（即使尚未进入Top榜单）
        5. 保证学科多样性
        """
        user_profile = self._get_user_profile(user_id)
        
        # 1. 召回候选
        candidates = self._recall_candidates(user_profile, context)
        
        # 2. 过滤已曝光内容
        candidates = self._filter_exposed(user_id, candidates)
        
        # 3. 计算个性化分数
        scored = []
        for item in candidates:
            personal_score = self._compute_personal_score(item, user_profile)
            scored.append((item, personal_score))
        
        # 4. 按分数排序
        scored.sort(key=lambda x: x[1], reverse=True)
        
        # 5. 学科多样性约束（同一学科不超过 40%）
        final = self._apply_diversity(scored, limit)
        
        # 6. 记录曝光
        self._record_exposure(user_id, final)
        
        return [item for item, _ in final[:limit]]
    
    def _compute_personal_score(self, item: dict, profile: dict) -> float:
        """计算个性化分数"""
        base_score = item['heat_score']
        
        # 学科偏好加权
        subject = item.get('subject', '')
        subject_pref = profile.get('subject_preferences', {})
        subject_weight = subject_pref.get(subject, 1.0)
        
        # 年级匹配加权
        grade_match = 1.0
        if item.get('grade_range') == profile.get('grade_range'):
            grade_match = 1.2
        
        # 教材版本匹配加权
        version_match = 1.0
        if item.get('textbook_version') == profile.get('textbook_version'):
            version_match = 1.1
        
        # 趋势加权（上升中的内容额外加权）
        trend_weight = 1.0
        if item.get('trend_direction') == 'RISING':
            trend_weight = 1.15
        elif item.get('trend_direction') == 'NEW':
            trend_weight = 1.08
        
        # 薄弱知识点加权（如果内容对应用户薄弱点）
        weak_kp_boost = 1.0
        if item.get('chapter_id') in profile.get('weak_chapters', set()):
            weak_kp_boost = 1.25  # 薄弱章节的热门内容优先推荐
        
        return (base_score * subject_weight * grade_match 
                * version_match * trend_weight * weak_kp_boost)
    
    def _apply_diversity(self, scored: list, limit: int) -> list:
        """学科多样性约束"""
        max_per_subject = max(2, int(limit * 0.4))
        subject_count = {}
        result = []
        
        for item, score in scored:
            subject = item.get('subject', 'UNKNOWN')
            if subject_count.get(subject, 0) >= max_per_subject:
                continue
            subject_count[subject] = subject_count.get(subject, 0) + 1
            result.append((item, score))
            if len(result) >= limit:
                break
        
        return result
```

```python
            }
        },

            'grade_boost_all': 1.0,

        }

        month = now.month
        
        if month in (5, 6):
            return SeasonalContext.NATIONAL_EXAM
        elif month in (4, 5):
            return SeasonalContext.EXAM_SEASON
        elif month in (10, 11):
            return SeasonalContext.MIDTERM
        elif month in (12, 1):
            return SeasonalContext.FINAL_EXAM
        elif month in (7, 8):
            return SeasonalContext.SUMMER_VACATION
        elif month == 2:
            return SeasonalContext.WINTER_VACATION
        elif month == 9:
            return SeasonalContext.BACK_TO_SCHOOL
        else:
            return SeasonalContext.NORMAL

        season = self.get_current_season()
        config = self.SEASONAL_CONFIG.get(season, self.SEASONAL_CONFIG[SeasonalContext.NORMAL])
        
        boost = 1.0
        
        # 年级加权
        grade_range = content_meta.get('grade_range', '')
        grade_boost = config.get(grade_range, config.get('grade_boost_all', 1.0))
        boost *= grade_boost
        
        # 学科加权
        subject = content_meta.get('subject', '')
        subject_boost = config.get('subject_boost', {}).get(subject, 1.0)
        boost *= subject_boost
        
        # 内容类型加权
        content_type = content_meta.get('content_type', '')
        type_boost = config.get('content_type_boost', {}).get(content_type, 1.0)
        boost *= type_boost
        
        # 最终加权分（转换为绝对分值，而非倍率）
        base_score = 100.0  # 基准分
        return (boost - 1.0) * base_score
```

```python
class ExternalHotBoostIntegrator:
```

```python
class TrendingCacheManager:
    """热度数据缓存管理器"""
    
    CACHE_TTL = 300  # 5分钟
    HOT_CACHE_TTL = 60  # 热门页面1分钟（允许稍微过期）
    
    def get_trending_feed(self, board_key: str, page: int, 
                          page_size: int) -> dict | None:
        """获取热门Feed（带缓存）"""
        cache_key = f"cache:trending:{board_key}:page_{page}"
        
        # 1. 先查 Redis 缓存
        cached = self.redis.get(cache_key)
        if cached:
            return json.loads(cached)
        
        # 2. 缓存未命中，查 ZSet
        zset_key = f"heat:ranking:{board_key}"
        start = (page - 1) * page_size
        end = start + page_size - 1
        
        content_ids_with_scores = self.redis.zrevrange(
            zset_key, start, end, withscores=True
        )
        
        if not content_ids_with_scores:
            # 3. ZSet 也空，查 MySQL
            return self._fallback_to_mysql(board_key, page, page_size)
        
        # 4. 批量获取内容详情
        content_ids = [int(cid) for cid, _ in content_ids_with_scores]
        details = self._batch_get_content_details(content_ids)
        
        # 5. 组装结果
        result = self._build_feed_response(details, content_ids_with_scores)
        
        # 6. 写回缓存
        self.redis.setex(cache_key, self.CACHE_TTL, json.dumps(result))
        
        return result
    
    def invalidate_board_cache(self, board_key: str):
        """榜单更新时使缓存失效"""
        pattern = f"cache:trending:{board_key}:*"
        for key in self.redis.scan_iter(match=pattern, count=100):
            self.redis.delete(key)
```

### 9.1 内容热度生命周期

```
                             └──────────┘
```

### 9.2 内容热度状态机

```python
from enum import Enum
from datetime import datetime, timedelta

class ContentHeatState(Enum):
    NEW = "NEW"
    ACTIVE = "ACTIVE"       # 在榜
    ARCHIVED = "ARCHIVED"   # 已下榜
    REMOVED = "REMOVED"     # 内容被删除

class HeatStateMachine:
    """内容热度状态机"""
    
    ARCHIVE_THRESHOLD = 50.0  # 热度分低于此值且持续3个周期 → ARCHIVED
    ARCHIVE_CONSECUTIVE_PERIODS = 3
    
    def transit_state(self, content_id: int, current_state: str,
                      heat_score: float, low_score_count: int) -> tuple[str, int]:
        """
        状态流转。
        
        Returns:
            (new_state, new_low_score_count)
        """
        if current_state == ContentHeatState.REMOVED.value:
            return current_state, 0
        
        if current_state == ContentHeatState.ARCHIVED.value:
            # 已归档内容如果突然有热度，重新激活
            if heat_score > self.ARCHIVE_THRESHOLD * 3:
                return ContentHeatState.ACTIVE.value, 0
            return current_state, 0
        
        # NEW 或 ACTIVE 状态
        if heat_score < self.ARCHIVE_THRESHOLD:
            new_low_count = low_score_count + 1
            if new_low_count >= self.ARCHIVE_CONSECUTIVE_PERIODS:
                return ContentHeatState.ARCHIVED.value, new_low_count
            return current_state, new_low_count
        else:
            # 热度恢复
            return ContentHeatState.ACTIVE.value, 0
```

```python
class TrendingFallbackHandler:
    """热度服务降级处理器"""
    
    def get_trending_with_fallback(self, board_key: str, 
                                    page: int, page_size: int) -> dict:
        """获取热门内容（带降级策略）"""
        
        try:
            # 正常流程：Redis ZSet
            result = self.cache_manager.get_trending_feed(board_key, page, page_size)
            if result and result.get('items'):
                return result
        
        except RedisConnectionError:
            logger.warning("Redis unavailable, falling back to MySQL")
            try:
                # 降级1：MySQL 查询
                result = self._query_mysql_board(board_key, page, page_size)
                if result:
                    return result
            except Exception as e:
                logger.error(f"MySQL fallback failed: {e}")
        
        except Exception as e:
            logger.error(f"Trending query failed: {e}")
        
        # 终极降级：返回静态推荐或最近缓存
        return self._get_emergency_fallback(board_key)
    
    def _get_emergency_fallback(self, board_key: str) -> dict:
        """紧急降级：返回预设或最近缓存数据"""
        # 尝试读取最近一次成功的缓存（即使已过期）
        stale_cache_key = f"cache:trending:{board_key}:page_1"
        stale = self.redis.get(stale_cache_key)  # 不检查TTL
        
        if stale:
            data = json.loads(stale)
            data['degraded'] = True
            data['degraded_reason'] = 'Using stale cache due to service degradation'
            return data
        
        # 如果连过期缓存都没有，返回空列表
        return {
            'items': [],
            'total': 0,
            'degraded': True,
            'degraded_reason': 'Trending service temporarily unavailable'
        }
```

```python
class HeatCalculationGuard:
    """热度计算任务守卫"""
    
    MAX_CALCULATION_TIME_SEC = 300  # 单次计算最大5分钟
    MAX_CONSECUTIVE_FAILURES = 3    # 连续失败上限
    
    def safe_run_calculation(self):
        """安全的计算执行（带超时和重试保护）"""
        try:
            # 检查上次计算是否成功
            last_status = self._get_last_calculation_status()
            if last_status == 'FAILED':
                consecutive_failures = self._get_consecutive_failures()
                if consecutive_failures >= self.MAX_CONSECUTIVE_FAILURES:
                    logger.critical(
                        "Heat calculation failed {n} times consecutively. "
                        "Using last successful snapshot. Alert required.".format(
                            n=consecutive_failures
                        )
                    )
                    # 发送告警
                    self._send_alert(
                        level='CRITICAL',
                        message='热度计算引擎连续失败，系统正在使用历史快照运行'
                    )
                    return
            
            # 执行计算（带超时）
            with timeout(self.MAX_CALCULATION_TIME_SEC):
                self.calculator.run_hourly()
                self._record_calculation_status('SUCCESS')
                
        except TimeoutError:
            logger.error("Heat calculation timed out")
            self._record_calculation_status('FAILED', reason='TIMEOUT')
            self._send_alert(
                level='WARNING',
                message='热度计算超时（>5分钟），请检查数据量或扩容'
            )
        except Exception as e:
            logger.error(f"Heat calculation failed: {e}")
            self._record_calculation_status('FAILED', reason=str(e))
```

```python
class HeatRedisPipeline:
    """Redis Pipeline 批量操作工具"""
    
    def batch_increment(self, events: list):
        """批量递增热度计数"""
        pipe = self.redis.pipeline()
        
        for event in events:
            content_id = event['content_id']
            bucket = event['bucket']
            field = event['field']
            
            key = f"heat:realtime:{content_id}:{bucket}"
            pipe.hincrby(key, field, 1)
            pipe.expire(key, 172800)
        
        pipe.execute()  # 单次网络往返完成所有操作
    
    def batch_uv_add(self, events: list):
        """批量UV去重"""
        pipe = self.redis.pipeline()
        
        for event in events:
            content_id = event['content_id']
            bucket = event['bucket']
            user_id = event['user_id']
            
            key = f"heat:uv:{content_id}:{bucket}"
            pipe.pfadd(key, str(user_id))
            pipe.expire(key, 172800)
        
        pipe.execute()
```

### 17.2 数字格式化工具

```python
def format_count(count: int) -> str:
    """将数字格式化为展示友好的文本"""
    if count >= 10000:
        return f"{count / 10000:.1f}万"
    elif count >= 1000:
        return f"{count / 1000:.1f}千"
    else:
        return str(count)

def format_heat_display(item: dict) -> dict:
    """格式化热度展示字段"""
    view_count = item.get('view_count_24h', 0)
    complete_count = item.get('complete_count_24h', 0)
    
    item['view_count_display'] = f"{format_count(view_count)}人在学"
    item['complete_count_display'] = f"{format_count(complete_count)}人已完成"
    
    # 趋势标签
    direction = item.get('trend_direction')
    if direction == 'RISING':
        item['trend_label'] = '🔥 热度上升'
    elif direction == 'NEW':
        item['trend_label'] = '✨ 新上榜'
    elif direction == 'STABLE':
        item['trend_label'] = '📌 持续热门'
    
    return item
```
