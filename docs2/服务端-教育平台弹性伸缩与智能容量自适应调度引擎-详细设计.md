# 服务端 - 教育平台弹性伸缩与智能容量自适应调度引擎 详细设计

## 1. 概述

### 1.1 模块定位

弹性伸缩与智能容量自适应调度引擎是 PrimeTop 生产环境的核心基础设施组件，负责根据实时流量、历史模式、业务事件和成本约束，自动调整各服务实例数量与资源配额，保障高并发场景下的服务稳定性，同时在低峰期回收资源以控制成本。

本文档与已有文档的关系：

| 已有文档 | 覆盖范围 | 本文增量 |
|----------|----------|----------|
| 《服务端容器化与Kubernetes生产编排设计》 | K8s 基础设施、HPA 基础配置 | 面向教育业务场景的智能伸缩策略、预测式扩容、多服务协调伸缩、成本感知调度 |
| 《非功能性需求-性能优化与容量规划》 | 静态容量规划方法论 | 动态容量自适应、自动化伸缩决策、实时反馈闭环 |
| 《服务端统一限流熔断与流量防护体系》 | 限流与熔断降级 | 主动式容量扩展与被动限流的协同机制 |

### 1.2 核心职责

1. **实时指标采集与汇聚**：从 Prometheus、业务监控、K8s Metrics API 多源采集负载指标。
2. **伸缩决策引擎**：基于规则 + 预测模型 + 业务事件，生成伸缩动作。
3. **多层级伸缩执行**：Pod 级（HPA/VPA）、节点级（Cluster Autoscaler）、服务级（自定义扩缩）。
4. **教育场景感知**：识别课后高峰、考试季、暑假低谷等教育特有流量模式。
5. **成本约束优化**：在满足 SLA 的前提下，最优化资源利用率与云计算成本。
6. **安全防护**：缩容保护窗口、会话亲和性保持、数据库连接池自适应。

### 3.2 指标采集层

```python
# metrics_collector.py

from dataclasses import dataclass
from typing import Optional
import asyncio
import aiohttp
import time

@dataclass
class ServiceMetrics:
    """服务实时指标快照"""
    service_name: str
    timestamp: float
    
    # K8s 资源指标
    cpu_usage: float              # 0.0 ~ 1.0
    memory_usage: float           # 0.0 ~ 1.0
    replicas_current: int
    replicas_ready: int
    
    # 流量指标
    request_rate: float           # requests/sec
    active_connections: int
    queue_depth: int
    
    # 性能指标
    p50_latency_ms: float
    p95_latency_ms: float
    p99_latency_ms: float
    error_rate: float             # 0.0 ~ 1.0
    
    # AI 专用指标
    ai_concurrent_requests: int = 0
    ai_tokens_per_second: float = 0.0
    ai_queue_wait_time_ms: float = 0.0
    
    # 数据库指标
    db_connection_pool_usage: float = 0.0  # 0.0 ~ 1.0
    db_slow_query_count: int = 0


class MetricsCollector:
    """多源指标采集器"""
    
    def __init__(self, prometheus_url: str, k8s_api_url: str, redis_client):
        self.prometheus_url = prometheus_url
        self.k8s_api_url = k8s_api_url
        self.redis = redis_client
        self._session: Optional[aiohttp.ClientSession] = None
    
    async def collect(self, service_name: str, namespace: str = "primetop-prod") -> ServiceMetrics:
        """采集服务完整指标快照"""
        tasks = [
            self._collect_k8s_metrics(service_name, namespace),
            self._collect_prom_metrics(service_name),
            self._collect_business_metrics(service_name),
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        k8s_data = results[0] if not isinstance(results[0], Exception) else {}
        prom_data = results[1] if not isinstance(results[1], Exception) else {}
        biz_data = results[2] if not isinstance(results[2], Exception) else {}
        
        metrics = ServiceMetrics(
            service_name=service_name,
            timestamp=time.time(),
            cpu_usage=k8s_data.get("cpu_usage", 0.0),
            memory_usage=k8s_data.get("memory_usage", 0.0),
            replicas_current=k8s_data.get("replicas_current", 0),
            replicas_ready=k8s_data.get("replicas_ready", 0),
            request_rate=prom_data.get("request_rate", 0.0),
            active_connections=prom_data.get("active_connections", 0),
            queue_depth=prom_data.get("queue_depth", 0),
            p50_latency_ms=prom_data.get("p50_latency_ms", 0.0),
            p95_latency_ms=prom_data.get("p95_latency_ms", 0.0),
            p99_latency_ms=prom_data.get("p99_latency_ms", 0.0),
            error_rate=prom_data.get("error_rate", 0.0),
            ai_concurrent_requests=biz_data.get("ai_concurrent_requests", 0),
            ai_tokens_per_second=biz_data.get("ai_tokens_per_second", 0.0),
            ai_queue_wait_time_ms=biz_data.get("ai_queue_wait_time_ms", 0.0),
            db_connection_pool_usage=biz_data.get("db_connection_pool_usage", 0.0),
            db_slow_query_count=biz_data.get("db_slow_query_count", 0),
        )
        
        # 写入 Redis 缓存
        await self._cache_metrics(metrics)
        
        return metrics
    
    async def _collect_prom_metrics(self, service_name: str) -> dict:
        """从 Prometheus 查询指标"""
        queries = {
            "request_rate": f'sum(rate(http_requests_total{{service="{service_name}"}}[2m]))',
            "active_connections": f'sum(http_active_connections{{service="{service_name}"}})',
            "queue_depth": f'sum(work_queue_depth{{service="{service_name}"}})',
            "p50_latency_ms": f'histogram_quantile(0.50, sum(rate(http_request_duration_ms_bucket{{service="{service_name}"}}[5m])) by (le))',
            "p95_latency_ms": f'histogram_quantile(0.95, sum(rate(http_request_duration_ms_bucket{{service="{service_name}"}}[5m])) by (le))',
            "p99_latency_ms": f'histogram_quantile(0.99, sum(rate(http_request_duration_ms_bucket{{service="{service_name}"}}[5m])) by (le))',
            "error_rate": f'sum(rate(http_requests_total{{service="{service_name}",status=~"5.."}}[5m])) / sum(rate(http_requests_total{{service="{service_name}"}}[5m]))',
        }
        
        results = {}
        for key, query in queries.items():
            value = await self._query_prometheus(query)
            if value is not None:
                results[key] = value
        return results
    
    async def _collect_k8s_metrics(self, service_name: str, namespace: str) -> dict:
        """从 K8s Metrics API 采集资源指标"""
        # 使用 K8s client 获取 Pod CPU/Memory
        # 简化示例
        return {
            "cpu_usage": await self._get_cpu_usage(service_name, namespace),
            "memory_usage": await self._get_memory_usage(service_name, namespace),
            "replicas_current": await self._get_replicas(service_name, namespace, "current"),
            "replicas_ready": await self._get_replicas(service_name, namespace, "ready"),
        }
    
    async def _collect_business_metrics(self, service_name: str) -> dict:
        """从业务监控采集自定义指标"""
        biz_keys = {
            "ai_concurrent_requests": f"primetop:metrics:{service_name}:ai_concurrent",
            "ai_tokens_per_second": f"primetop:metrics:{service_name}:ai_tps",
            "ai_queue_wait_time_ms": f"primetop:metrics:{service_name}:ai_queue_wait",
            "db_connection_pool_usage": f"primetop:metrics:{service_name}:db_pool_usage",
            "db_slow_query_count": f"primetop:metrics:{service_name}:db_slow_queries",
        }
        results = {}
        for key, redis_key in biz_keys.items():
            val = await self.redis.get(redis_key)
            if val is not None:
                results[key] = float(val)
        return results
    
    async def _query_prometheus(self, query: str) -> Optional[float]:
        """执行 PromQL 查询"""
        if not self._session:
            self._session = aiohttp.ClientSession()
        try:
            resp = await self._session.get(
                f"{self.prometheus_url}/api/v1/query",
                params={"query": query},
                timeout=aiohttp.ClientTimeout(total=5)
            )
            data = await resp.json()
            if data["status"] == "success" and data["data"]["result"]:
                return float(data["data"]["result"][0]["value"][1])
        except Exception:
            pass
        return None
    
    async def _cache_metrics(self, metrics: ServiceMetrics):
        """写入 Redis 缓存"""
        import json
        key = f"primetop:scaling:metrics:{metrics.service_name}"
        mapping = {
            "cpu_usage": str(metrics.cpu_usage),
            "memory_usage": str(metrics.memory_usage),
            "request_rate": str(metrics.request_rate),
            "active_connections": str(metrics.active_connections),
            "queue_depth": str(metrics.queue_depth),
            "p99_latency_ms": str(metrics.p99_latency_ms),
            "error_rate": str(metrics.error_rate),
            "timestamp": str(metrics.timestamp),
        }
        await self.redis.hset(key, mapping=mapping)
        await self.redis.expire(key, 120)  # 2分钟过期
```

### 3.3 决策引擎层

决策引擎是核心组件，采用**多信号融合决策模型**，综合反应式、预测式和事件驱动三类信号。

```python
# scaling_decision_engine.py

from enum import Enum
from dataclasses import dataclass, field
from typing import List, Optional
import asyncio
import logging

logger = logging.getLogger("scaling.engine")


class ScalingAction(Enum):
    SCALE_UP = "SCALE_UP"
    SCALE_DOWN = "SCALE_DOWN"
    NO_ACTION = "NO_ACTION"
    BLOCKED_COOLDOWN = "BLOCKED_COOLDOWN"
    BLOCKED_BOUNDARY = "BLOCKED_BOUNDARY"
    BLOCKED_COST = "BLOCKED_COST"


class TriggerType(Enum):
    REACTIVE = "REACTIVE"
    PREDICTIVE = "PREDICTIVE"
    EVENT_DRIVEN = "EVENT_DRIVEN"
    HYBRID = "HYBRID"


@dataclass
class ScalingDecision:
    """伸缩决策结果"""
    service_name: str
    action: ScalingAction
    trigger: TriggerType
    current_replicas: int
    target_replicas: int
    confidence: float = 1.0
    reasons: List[str] = field(default_factory=list)
    estimated_cost_delta: float = 0.0
    blocking_reason: Optional[str] = None


class DecisionEngine:
    """伸缩决策引擎 - 多信号融合"""
    
    def __init__(
        self,
        policy_store,        # PolicyRepository
        metrics_collector,   # MetricsCollector
        forecast_engine,     # ForecastEngine
        event_scheduler,     # EventDrivenScheduler
        cost_guard,          # CostGuard
        cooldown_manager,    # CooldownManager
    ):
        self.policy_store = policy_store
        self.metrics = metrics_collector
        self.forecast = forecast_engine
        self.events = event_scheduler
        self.cost = cost_guard
        self.cooldown = cooldown_manager
    
    async def evaluate(self, service_name: str, namespace: str) -> ScalingDecision:
        """评估服务伸缩需求，生成决策"""
        # 1. 加载策略
        policy = await self.policy_store.get_policy(service_name, namespace)
        if policy is None or not policy.enabled:
            return ScalingDecision(
                service_name=service_name,
                action=ScalingAction.NO_ACTION,
                trigger=TriggerType.REACTIVE,
                current_replicas=0,
                target_replicas=0,
                reasons=["未找到已启用策略"]
            )
        
        # 2. 采集当前指标
        current_metrics = await self.metrics.collect(service_name, namespace)
        current_replicas = current_metrics.replicas_current
        
        # 3. 并行评估三类信号
        reactive_signal, predictive_signal, event_signal = await asyncio.gather(
            self._evaluate_reactive(policy, current_metrics),
            self._evaluate_predictive(policy, service_name),
            self._evaluate_event_driven(policy, service_name),
            return_exceptions=True
        )
        
        # 异常容错
        for i, signal in enumerate([reactive_signal, predictive_signal, event_signal]):
            if isinstance(signal, Exception):
                signal_name = ["reactive", "predictive", "event"][i]
                logger.warning(f"{service_name} {signal_name} signal error: {signal}")
                if i == 0:
                    reactive_signal = ScalingDecision(
                        service_name=service_name, action=ScalingAction.NO_ACTION,
                        trigger=TriggerType.REACTIVE, current_replicas=current_replicas,
                        target_replicas=current_replicas
                    )
                elif i == 1:
                    predictive_signal = None
                elif i == 2:
                    event_signal = None
        
        # 4. 信号融合决策
        decision = self._fuse_signals(
            policy, current_replicas, reactive_signal, predictive_signal, event_signal
        )
        
        # 5. 前置检查（冷却期、边界、成本）
        decision = await self._apply_guardrails(policy, decision, current_metrics)
        
        return decision
    
    def _fuse_signals(
        self,
        policy,
        current_replicas: int,
        reactive: ScalingDecision,
        predictive: Optional[ScalingDecision],
        event: Optional[ScalingDecision],
    ) -> ScalingDecision:
        """多信号融合：取最保守的扩容决策 + 最保守的缩容决策"""
        
        # 收集所有建议扩容的信号
        scale_up_signals = []
        if reactive.action == ScalingAction.SCALE_UP:
            scale_up_signals.append(("REACTIVE", reactive))
        if predictive and predictive.action == ScalingAction.SCALE_UP:
            scale_up_signals.append(("PREDICTIVE", predictive))
        if event and event.action == ScalingAction.SCALE_UP:
            scale_up_signals.append(("EVENT", event))
        
        # 收集所有建议缩容的信号
        scale_down_signals = []
        if reactive.action == ScalingAction.SCALE_DOWN:
            scale_down_signals.append(("REACTIVE", reactive))
        if predictive and predictive.action == ScalingAction.SCALE_DOWN:
            scale_down_signals.append(("PREDICTIVE", predictive))
        
        # 决策逻辑
        if scale_up_signals:
            # 扩容：取最大目标副本数（最积极的扩容）
            best = max(scale_up_signals, key=lambda x: x[1].target_replicas)
            triggers = [s[0] for s in scale_up_signals]
            trigger = TriggerType.HYBRID if len(triggers) > 1 else TriggerType(triggers[0])
            
            reasons = [f"{t}: {', '.join(s.reasons)}" for t, s in scale_up_signals]
            return ScalingDecision(
                service_name=policy.service_name,
                action=ScalingAction.SCALE_UP,
                trigger=trigger,
                current_replicas=current_replicas,
                target_replicas=best[1].target_replicas,
                confidence=min(s[1].confidence for s in scale_up_signals),
                reasons=reasons,
                estimated_cost_delta=best[1].estimated_cost_delta,
            )
        
        if scale_down_signals:
            # 缩容：取最小目标副本数（最保守的缩容）
            # 但只有在所有信号都同意缩容时才缩容
            best = min(scale_down_signals, key=lambda x: x[1].target_replicas)
            
            # 如果预测信号要扩容但反应式要缩容，不缩容
            if predictive and predictive.action == ScalingAction.SCALE_UP:
                return ScalingDecision(
                    service_name=policy.service_name,
                    action=ScalingAction.NO_ACTION,
                    trigger=TriggerType.HYBRID,
                    current_replicas=current_replicas,
                    target_replicas=current_replicas,
                    reasons=["缩容信号与预测扩容信号冲突，保持当前副本"],
                )
            
            return ScalingDecision(
                service_name=policy.service_name,
                action=ScalingAction.SCALE_DOWN,
                trigger=TriggerType(best[0]),
                current_replicas=current_replicas,
                target_replicas=best[1].target_replicas,
                confidence=best[1].confidence,
                reasons=best[1].reasons,
            )
        
        return ScalingDecision(
            service_name=policy.service_name,
            action=ScalingAction.NO_ACTION,
            trigger=TriggerType.REACTIVE,
            current_replicas=current_replicas,
            target_replicas=current_replicas,
            reasons=["所有信号均无伸缩建议"],
        )
    
    async def _apply_guardrails(
        self, policy, decision: ScalingDecision, metrics
    ) -> ScalingDecision:
        """应用安全护栏：冷却期、边界检查、成本约束"""
        
        if decision.action == ScalingAction.NO_ACTION:
            return decision
        
        # 边界检查
        if decision.action == ScalingAction.SCALE_UP and decision.target_replicas > policy.max_replicas:
            decision.target_replicas = policy.max_replicas
            decision.reasons.append(f"目标副本数已达上限 {policy.max_replicas}")
            if decision.current_replicas >= policy.max_replicas:
                decision.action = ScalingAction.BLOCKED_BOUNDARY
                decision.blocking_reason = f"已达最大副本数 {policy.max_replicas}"
        
        if decision.action == ScalingAction.SCALE_DOWN:
            min_replicas = self._get_effective_min_replicas(policy)
            if decision.target_replicas < min_replicas:
                decision.target_replicas = min_replicas
                decision.reasons.append(f"目标副本数已降至下限 {min_replicas}")
                if decision.current_replicas <= min_replicas:
                    decision.action = ScalingAction.BLOCKED_BOUNDARY
                    decision.blocking_reason = f"已达最小副本数 {min_replicas}"
        
        # 冷却期检查
        if decision.action in (ScalingAction.SCALE_UP, ScalingAction.SCALE_DOWN):
            action_key = "scale_up" if decision.action == ScalingAction.SCALE_UP else "scale_down"
            in_cooldown = await self.cooldown.is_in_cooldown(
                policy.service_name, action_key
            )
            if in_cooldown:
                decision.action = ScalingAction.BLOCKED_COOLDOWN
                decision.blocking_reason = f"{action_key} 冷却中"
        
        # 成本约束（仅对扩容生效）
        if decision.action == ScalingAction.SCALE_UP:
            cost_ok, cost_reason = await self.cost.check_budget(
                decision.service_name,
                decision.current_replicas,
                decision.target_replicas,
            )
            if not cost_ok:
                decision.action = ScalingAction.BLOCKED_COST
                decision.blocking_reason = cost_reason
        
        # 夜间最小副本数保护
        if decision.action == ScalingAction.SCALE_DOWN:
            min_night = self._get_effective_min_replicas(policy)
            if min_night != policy.min_replicas:
                decision.reasons.append(f"夜间保护副本数: {min_night}")
        
        return decision
    
    def _get_effective_min_replicas(self, policy) -> int:
        """根据时段获取有效最小副本数"""
        import datetime
        now = datetime.datetime.now()
        hour = now.hour
        
        if policy.min_replicas_night is not None:
            is_night = hour >= policy.night_start_hour or hour < policy.night_end_hour
            if is_night:
                return policy.min_replicas_night
        
        return policy.min_replicas
    
    async def _evaluate_reactive(self, policy, metrics) -> ScalingDecision:
        """反应式评估：基于阈值规则"""
        rules = await self.policy_store.get_rules(policy.id)
        
        scale_up_votes = []
        scale_down_votes = []
        
        for rule in rules:
            metric_value = self._extract_metric(metrics, rule.metric_name)
            if metric_value is None:
                continue
            
            triggered = False
            if rule.action_type == "SCALE_UP":
                if self._compare(metric_value, rule.threshold_up, rule.operator):
                    triggered = True
                    scale_up_votes.append({
                        "rule": rule.rule_name,
                        "metric": rule.metric_name,
                        "value": metric_value,
                        "threshold": rule.threshold_up,
                    })
            elif rule.action_type == "SCALE_DOWN":
                if self._compare(metric_value, rule.threshold_down, rule.operator):
                    triggered = True
                    scale_down_votes.append({
                        "rule": rule.rule_name,
                        "metric": rule.metric_name,
                        "value": metric_value,
                        "threshold": rule.threshold_down,
                    })
        
        current = metrics.replicas_current
        
        if scale_up_votes:
            # 根据触发的规则数决定扩容幅度
            total_delta = self._calculate_scale_up_delta(scale_up_votes, current, policy)
            target = min(current + total_delta, policy.max_replicas)
            return ScalingDecision(
                service_name=policy.service_name,
                action=ScalingAction.SCALE_UP,
                trigger=TriggerType.REACTIVE,
                current_replicas=current,
                target_replicas=target,
                confidence=0.9,
                reasons=[f"指标触发: {v['metric']}={v['value']:.2f} (阈值{v['threshold']})" for v in scale_up_votes],
            )
        
        if scale_down_votes and len(scale_down_votes) == len([r for r in rules if r.action_type == "SCALE_DOWN"]):
            # 所有缩容规则都满足才缩容
            target = max(current - 1, policy.min_replicas)
            return ScalingDecision(
                service_name=policy.service_name,
                action=ScalingAction.SCALE_DOWN,
                trigger=TriggerType.REACTIVE,
                current_replicas=current,
                target_replicas=target,
                confidence=0.8,
                reasons=[f"指标恢复: {v['metric']}={v['value']:.2f}" for v in scale_down_votes],
            )
        
        return ScalingDecision(
            service_name=policy.service_name,
            action=ScalingAction.NO_ACTION,
            trigger=TriggerType.REACTIVE,
            current_replicas=current,
            target_replicas=current,
            reasons=["反应式规则未触发"],
        )
    
    async def _evaluate_predictive(self, policy, service_name) -> Optional[ScalingDecision]:
        """预测式评估：基于时序预测"""
        if policy.policy_type not in ("PREDICTIVE", "HYBRID"):
            return None
        
        forecast = await self.forecast.get_forecast(service_name)
        if forecast is None:
            return None
        
        current_metrics = await self.metrics.collect(service_name, policy.namespace)
        current = current_metrics.replicas_current
        
        # 如果预测15分钟内负载会超过阈值，提前扩容
        if forecast.predicted_load_15m > 0.75 and forecast.confidence > 0.85:
            pre_warm_target = self._calculate_needed_replicas(
                forecast.predicted_load_15m, 
                policy.max_replicas
            )
            if pre_warm_target > current:
                return ScalingDecision(
                    service_name=service_name,
                    action=ScalingAction.SCALE_UP,
                    trigger=TriggerType.PREDICTIVE,
                    current_replicas=current,
                    target_replicas=pre_warm_target,
                    confidence=forecast.confidence,
                    reasons=[
                        f"预测15分钟内负载达{forecast.predicted_load_15m:.0%}",
                        f"置信度{forecast.confidence:.0%}",
                        f"提前扩容至{pre_warm_target}副本"
                    ],
                )
        
        # 预测负载将持续走低
        if (forecast.predicted_load_30m < 0.35 
            and forecast.predicted_load_60m < 0.30 
            and forecast.confidence > 0.80):
            min_r = policy.min_replicas
            if current > min_r:
                return ScalingDecision(
                    service_name=service_name,
                    action=ScalingAction.SCALE_DOWN,
                    trigger=TriggerType.PREDICTIVE,
                    current_replicas=current,
                    target_replicas=max(current - 2, min_r),
                    confidence=forecast.confidence,
                    reasons=[
                        f"预测60分钟内负载持续低于35%",
                        f"提前缩容减少资源浪费"
                    ],
                )
        
        return ScalingDecision(
            service_name=service_name,
            action=ScalingAction.NO_ACTION,
            trigger=TriggerType.PREDICTIVE,
            current_replicas=current,
            target_replicas=current,
            reasons=["预测负载在正常范围"],
        )
    
    async def _evaluate_event_driven(self, policy, service_name) -> Optional[ScalingDecision]:
        """事件驱动评估：教育日历事件"""
        if policy.policy_type not in ("EVENT_DRIVEN", "HYBRID"):
            return None
        
        active_events = await self.events.get_active_events(service_name)
        if not active_events:
            return None
        
        current_metrics = await self.metrics.collect(service_name, policy.namespace)
        current = current_metrics.replicas_current
        
        # 计算事件驱动的目标副本数
        max_multiplier = 1.0
        active_reasons = []
        for event in active_events:
            max_multiplier = max(max_multiplier, event.expected_traffic_multiplier)
            active_reasons.append(f"{event.event_name}(x{event.expected_traffic_multiplier})")
        
        event_target = min(
            int(current * max_multiplier),
            policy.max_replicas
        )
        
        if event_target > current:
            return ScalingDecision(
                service_name=service_name,
                action=ScalingAction.SCALE_UP,
                trigger=TriggerType.EVENT_DRIVEN,
                current_replicas=current,
                target_replicas=event_target,
                confidence=0.95,
                reasons=active_reasons,
            )
        
        return ScalingDecision(
            service_name=service_name,
            action=ScalingAction.NO_ACTION,
            trigger=TriggerType.EVENT_DRIVEN,
            current_replicas=current,
            target_replicas=current,
            reasons=["事件已生效但当前副本足够"],
        )
    
    def _extract_metric(self, metrics, metric_name: str) -> Optional[float]:
        """从指标快照中提取值"""
        mapping = {
            "cpu_usage": metrics.cpu_usage,
            "memory_usage": metrics.memory_usage,
            "request_rate": metrics.request_rate,
            "p95_latency_ms": metrics.p95_latency_ms,
            "p99_latency_ms": metrics.p99_latency_ms,
            "error_rate": metrics.error_rate,
            "queue_depth": float(metrics.queue_depth),
            "ai_concurrent_requests": float(metrics.ai_concurrent_requests),
            "db_connection_pool_usage": metrics.db_connection_pool_usage,
        }
        return mapping.get(metric_name)
    
    def _compare(self, value: float, threshold: float, operator: str) -> bool:
        ops = {
            ">": value > threshold,
            ">=": value >= threshold,
            "<": value < threshold,
            "<=": value <= threshold,
            "==": value == threshold,
        }
        return ops.get(operator, False)
    
    def _calculate_scale_up_delta(self, votes: list, current: int, policy) -> int:
        """根据触发规则数计算扩容幅度"""
        base_delta = 1
        if len(votes) >= 3:
            base_delta = 3
        elif len(votes) >= 2:
            base_delta = 2
        
        # CPU 超过90%时激进扩容
        for v in votes:
            if v["metric"] == "cpu_usage" and v["value"] > 0.90:
                base_delta = max(base_delta, 3)
        
        return base_delta
    
    def _calculate_needed_replicas(self, predicted_load: float, max_replicas: int) -> int:
        """根据预测负载计算所需副本数"""
        # 假设每副本承载约 0.25 的负载（4副本=1.0负载容量）
        target = int(predicted_load / 0.20) + 1  # 留20%余量
        return min(target, max_replicas)
```

```python
import numpy as np

        # 先查缓存
        cached = await self._get_cached_forecast(service_name)
        if cached and self._is_fresh(cached):
            return cached
        
        # 重新计算
        try:
            forecast = await self._compute_forecast(service_name)
            if forecast:
                await self._cache_forecast(service_name, forecast)
            return forecast
        except Exception as e:
            logger.error(f"Forecast failed for {service_name}: {e}")
            return None
    
    async def _compute_forecast(self, service_name: str) -> Optional[ForecastResult]:

        # 1. 拉取历史负载数据
        history = await self._fetch_history(service_name, days=30)
        if len(history) < 144:  # 至少需要1天数据（每10分钟1个点）
            logger.info(f"{service_name}: 历史数据不足，跳过预测")
            return None
        
        # 2. 加载模型参数（如果已训练）
        model_params = await self.model_store.get_params(service_name)
        
        # 3. 执行预测
        if model_params and model_params.get("algorithm") == "HWINTER":
            predictions = self._hwinter_forecast(history, model_params)
        else:
            predictions = self._default_forecast(history)
        
        # 4. 计算置信度
        confidence = self._calculate_confidence(history, predictions)
        
        return ForecastResult(
            predicted_load_5m=predictions[0],
            predicted_load_15m=predictions[1],
            predicted_load_30m=predictions[2],
            predicted_load_60m=predictions[3],
            confidence=confidence,
            model_version=f"hwinter-v{model_params.get('version', 1) if model_params else 1}",
            last_updated=time.time(),
        )

    def _hwinter_forecast(
        self, 
        history: np.ndarray, 
        params: dict
    ) -> tuple:
        """Holt-Winters 三次指数平滑预测"""
        alpha = params.get("alpha", 0.3)
        beta = params.get("beta", 0.1)
        gamma = params.get("gamma", 0.2)
        season_length = params.get("season_length", 144)  # 日周期: 144个10分钟
        
        h = len(history)
        
        # 初始化
        level = np.mean(history[:season_length])
        trend = (np.mean(history[season_length:2*season_length]) - level) / season_length
        seasonals = np.array([
            history[i] - np.mean(history[:season_length])
            for i in range(season_length)
        ])
        
        # 递推
        for i in range(h):
            if i < season_length:
                continue
            last_level = level
            level = alpha * (history[i] - seasonals[i % season_length]) + (1 - alpha) * (level + trend)
            trend = beta * (level - last_level) + (1 - beta) * trend
            seasonals[i % season_length] = gamma * (history[i] - level) + (1 - gamma) * seasonals[i % season_length]
        
        # 预测未来
        forecasts = []
        for step in [1, 3, 6, 12]:  # 5m, 15m, 30m, 60m（10分钟粒度）
            forecast = level + step * trend + seasonals[(h + step) % season_length]
            forecasts.append(max(0.0, min(1.0, forecast)))
        
        return tuple(forecasts)
    
    def _default_forecast(self, history: np.ndarray) -> tuple:
        """默认预测：基于同时段历史均值"""
        season_length = 144  # 日周期
        if len(history) < season_length * 7:
            # 数据不足一周，用简单移动平均
            recent = history[-min(24, len(history)):]
            avg = np.mean(recent)
            return (avg, avg, avg, avg)
        
        # 取过去7天同时段的均值
        current_idx = len(history) % season_length
        same_period_values = []
        for d in range(1, 8):
            start_idx = len(history) - d * season_length + current_idx
            if start_idx >= 0:
                # 取前后各3个点做平滑
                window = history[max(0, start_idx-3):start_idx+4]
                same_period_values.append(np.mean(window))
        
        if same_period_values:
            avg = np.mean(same_period_values)
            # 加入微弱趋势
            if len(same_period_values) >= 3:
                trend = (same_period_values[0] - same_period_values[-1]) / len(same_period_values)
            else:
                trend = 0
        else:
            avg = np.mean(history[-24:])
            trend = 0
        
        # 预测
        result = []
        for step in [1, 3, 6, 12]:
            val = avg + step * trend * 0.1  # 衰减趋势
            result.append(max(0.0, min(1.0, val)))
        
        return tuple(result)
    
    def _calculate_confidence(self, history: np.ndarray, predictions: tuple) -> float:
        """计算预测置信度"""
        # 在历史数据上做回测
        if len(history) < 288:  # 至少2天
            return 0.6
        
        # 取最后一天做backtest
        test_data = history[-144:]
        train_data = history[:-144]
        
        if len(train_data) < 144:
            return 0.6
        
        # 计算历史预测误差
        try:
            backtest_preds = []
            for i in range(len(test_data)):
                partial = np.concatenate([train_data, test_data[:i]])
                pred = self._default_forecast(partial)
                backtest_preds.append(pred[0])
            
            mae = np.mean(np.abs(np.array(backtest_preds) - test_data))
            # MAE 越小，置信度越高
            confidence = max(0.5, min(0.98, 1.0 - mae * 2))
            return float(confidence)
        except Exception:
            return 0.7
    
    async def _get_cached_forecast(self, service_name: str) -> Optional[ForecastResult]:
        """从 Redis 获取缓存的预测"""
        import json
        key = f"primetop:scaling:forecast:{service_name}"
        data = await self.redis.hgetall(key)
        if not data:
            return None
        return ForecastResult(
            predicted_load_5m=float(data.get("predicted_load_5m", 0)),
            predicted_load_15m=float(data.get("predicted_load_15m", 0)),
            predicted_load_30m=float(data.get("predicted_load_30m", 0)),
            predicted_load_60m=float(data.get("predicted_load_60m", 0)),
            confidence=float(data.get("confidence", 0)),
            model_version=data.get("model_version", "unknown"),
            last_updated=float(data.get("last_updated", 0)),
        )
    
    def _is_fresh(self, forecast: ForecastResult, max_age_seconds: int = 300) -> bool:
        import time
        return (time.time() - forecast.last_updated) < max_age_seconds
    
    async def _cache_forecast(self, service_name: str, forecast: ForecastResult):
        """缓存预测结果"""
        key = f"primetop:scaling:forecast:{service_name}"
        mapping = {
            "predicted_load_5m": str(forecast.predicted_load_5m),
            "predicted_load_15m": str(forecast.predicted_load_15m),
            "predicted_load_30m": str(forecast.predicted_load_30m),
            "predicted_load_60m": str(forecast.predicted_load_60m),
            "confidence": str(forecast.confidence),
            "model_version": forecast.model_version,
            "last_updated": str(forecast.last_updated),
        }
        await self.redis.hset(key, mapping=mapping)
        await self.redis.expire(key, 600)  # 10分钟过期
```

```python
# scaling_executor.py

from kubernetes import client, config
from kubernetes.client.rest import ApiException
import asyncio
import logging
import uuid

logger = logging.getLogger("scaling.executor")


class ScalingExecutor:
    """伸缩执行器 - 调用 K8s API 执行副本变更"""
    
    def __init__(self, event_log_store, cooldown_manager):
        # 加载 K8s 配置
        try:
            config.load_incluster_config()
        except config.ConfigException:
            config.load_kube_config()
        
        self.apps_v1 = client.AppsV1Api()
        self.autoscaling_v2 = client.AutoscalingV2Api()
        self.event_log = event_log_store
        self.cooldown = cooldown_manager
    
    async def execute(self, decision: ScalingDecision) -> bool:
        """执行伸缩决策"""
        if decision.action not in (ScalingAction.SCALE_UP, ScalingAction.SCALE_DOWN):
            logger.info(f"Skipping {decision.service_name}: action={decision.action}")
            return False
        
        event_id = str(uuid.uuid4())
        
        # 记录伸缩事件
        await self.event_log.record(
            event_id=event_id,
            service_name=decision.service_name,
            action=decision.action.value,
            trigger_type=decision.trigger.value,
            replicas_before=decision.current_replicas,
            replicas_target=decision.target_replicas,
            trigger_detail={"reasons": decision.reasons, "confidence": decision.confidence},
        )
        
        try:
            # 执行方式1: 直接修改 Deployment replicas
            success = await self._patch_deployment(
                decision.service_name,
                decision.target_replicas
            )
            
            if success:
                # 更新冷却期
                action_key = "scale_up" if decision.action == ScalingAction.SCALE_UP else "scale_down"
                await self.cooldown.set_cooldown(decision.service_name, action_key)
                
                # 更新事件状态
                await self.event_log.mark_success(event_id, decision.target_replicas)
                
                logger.info(
                    f"Scaled {decision.service_name}: "
                    f"{decision.current_replicas} → {decision.target_replicas} "
                    f"({decision.trigger.value}, confidence={decision.confidence:.0%}) "
                    f"reasons: {'; '.join(decision.reasons)}"
                )
                return True
            else:
                await self.event_log.mark_failed(event_id, "Patch failed")
                return False
                
        except ApiException as e:
            logger.error(f"K8s API error scaling {decision.service_name}: {e}")
            await self.event_log.mark_failed(event_id, str(e))
            return False
        except Exception as e:
            logger.error(f"Unexpected error scaling {decision.service_name}: {e}", exc_info=True)
            await self.event_log.mark_failed(event_id, str(e))
            return False
    
    async def _patch_deployment(self, service_name: str, target_replicas: int) -> bool:
        """通过 patch Deployment replicas 实现伸缩"""
        namespace = "primetop-prod"
        
        # 使用 async wrapper
        body = {"spec": {"replicas": target_replicas}}
        
        loop = asyncio.get_event_loop()
        try:
            response = await loop.run_in_executor(
                None,
                lambda: self.apps_v1.patch_namespaced_deployment_scale(
                    name=service_name,
                    namespace=namespace,
                    body=body
                )
            )
            return True
        except ApiException as e:
            if e.status == 409:
                # 冲突，重试
                logger.warning(f"Conflict patching {service_name}, will retry")
                await asyncio.sleep(2)
                response = await loop.run_in_executor(
                    None,
                    lambda: self.apps_v1.patch_namespaced_deployment_scale(
                        name=service_name,
                        namespace=namespace,
                        body=body
                    )
                )
                return True
            raise
    
    async def verify_scaling(self, service_name: str, target_replicas: int, timeout: int = 180) -> bool:
        """验证伸缩是否完成（所有副本 Ready）"""
        namespace = "primetop-prod"
        start = asyncio.get_event_loop().time()
        
        while (asyncio.get_event_loop().time() - start) < timeout:
            loop = asyncio.get_event_loop()
            deployment = await loop.run_in_executor(
                None,
                lambda: self.apps_v1.read_namespaced_deployment(
                    name=service_name, namespace=namespace
                )
            )
            
            if (deployment.status.ready_replicas == target_replicas 
                and deployment.status.replicas == target_replicas):
                return True
            
            await asyncio.sleep(5)
        
        logger.warning(
            f"{service_name} scaling verification timeout: "
            f"target={target_replicas}, "
            f"current={deployment.status.replicas}, "
            f"ready={deployment.status.ready_replicas}"
        )
        return False
```

```python
# cooldown_manager.py

import time
import logging

logger = logging.getLogger("scaling.cooldown")


class CooldownManager:
    """冷却期管理 - 防止频繁伸缩"""
    
    def __init__(self, redis_client, default_cooldowns: dict = None):
        self.redis = redis_client
        self.default_cooldowns = default_cooldowns or {
            "scale_up": 60,     # 扩容冷却 60秒
            "scale_down": 300,  # 缩容冷却 5分钟
        }
    
    async def is_in_cooldown(self, service_name: str, action: str) -> bool:
        """检查是否在冷却期"""
        key = f"primetop:scaling:cooldown:{service_name}:{action}"
        ttl = await self.redis.ttl(key)
        return ttl > 0
    
    async def set_cooldown(self, service_name: str, action: str, duration: int = None):
        """设置冷却期"""
        if duration is None:
            duration = self.default_cooldowns.get(action, 120)
        key = f"primetop:scaling:cooldown:{service_name}:{action}"
        await self.redis.setex(key, duration, str(time.time()))
    
    async def get_remaining_cooldown(self, service_name: str, action: str) -> int:
        """获取剩余冷却时间"""
        key = f"primetop:scaling:cooldown:{service_name}:{action}"
        ttl = await self.redis.ttl(key)
        return max(0, ttl)
    
    async def clear_cooldown(self, service_name: str, action: str = None):
        """清除冷却期（紧急手动操作时）"""
        if action:
            key = f"primetop:scaling:cooldown:{service_name}:{action}"
            await self.redis.delete(key)
        else:
            # 清除所有冷却
            pattern = f"primetop:scaling:cooldown:{service_name}:*"
            keys = []
            async for key in self.redis.scan_iter(match=pattern):
                keys.append(key)
            if keys:
                await self.redis.delete(*keys)
```

```python
@dataclass
class ServiceCostModel:

    uses_gpu: bool = False
    
    # 成本上限
    daily_budget: float = 500.0  # 日预算上限
    monthly_budget: float = 10000.0  # 月预算上限


class CostGuard:
    """成本护栏 - 在扩容前检查预算约束"""
    
    def __init__(self, redis_client, cost_models: dict = None):
        self.redis = redis_client
        self.cost_models = cost_models or self._default_cost_models()
    
    async def check_budget(
        self,
        service_name: str,
        current_replicas: int,
        target_replicas: int,
    ) -> Tuple[bool, str]:
        """检查扩容是否超出预算"""
        model = self.cost_models.get(service_name)
        if model is None:
            return True, "无成本模型配置，放行"
        
        # 计算扩容后预估日成本
        additional_replicas = target_replicas - current_replicas
        additional_cost_per_hour = additional_replicas * model.cost_per_replica_hour
        if model.uses_gpu:
            additional_cost_per_hour += model.gpu_cost_per_hour * additional_replicas
        
        # 预估今日剩余成本
        today_spent = await self._get_daily_spend()
        remaining_budget = model.daily_budget - today_spent
        
        # 预估新增成本（按剩余小时计算）
        from datetime import datetime, timedelta
        now = datetime.now()
        hours_remaining = 24 - now.hour - now.minute / 60.0
        estimated_additional_cost = additional_cost_per_hour * hours_remaining
        
        if estimated_additional_cost > remaining_budget * 0.8:
            reason = (
                f"扩容成本预计+{estimated_additional_cost:.2f}元，"
                f"剩余预算{remaining_budget:.2f}元（超80%阈值）"
            )
            logger.warning(f"Cost guard blocked {service_name}: {reason}")
            return False, reason
        
        return True, "预算充足"
    
    async def record_cost(self, service_name: str, replicas: int, duration_hours: float):
        """记录实际成本"""
        model = self.cost_models.get(service_name)
        if model is None:
            return
        
        cost = replicas * model.cost_per_replica_hour * duration_hours
        if model.uses_gpu:
            cost += model.gpu_cost_per_hour * duration_hours
        
        # 累加到日/月统计
        from datetime import date
        today = date.today().isoformat()
        
        await self.redis.hincrbyfloat(
            f"primetop:scaling:cost:daily:{today}",
            service_name,
            cost
        )
        await self.redis.hincrbyfloat(
            f"primetop:scaling:cost:daily:{today}",
            "total_cost",
            cost
        )
        await self.redis.hincrbyfloat(
            f"primetop:scaling:cost:monthly:{today[:7]}",
            service_name,
            cost
        )
    
    async def _get_daily_spend(self) -> float:
        """获取今日已花费"""
        from datetime import date
        today = date.today().isoformat()
        val = await self.redis.hget(f"primetop:scaling:cost:daily:{today}", "total_cost")
        return float(val) if val else 0.0
```
