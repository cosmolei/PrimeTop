# 端到端流程设计 - 文科背诵辅助完整链路

## 1. 文档概述

### 1.1 文档目的

本文档细化"文科背诵"功能的端到端完整链路，覆盖从用户进入背诵模块、选择背诵材料、学习记忆、背诵检测、复习调度到数据沉淀的全流程。面向开发人员，提供可直接编码的数据结构、API接口、状态机、交互流程和跨模块集成规范。

### 1.2 适用范围

文科背诵覆盖以下学科场景：

| 学科 | 典型背诵内容 | 学段 |
|------|-------------|------|
| 语文 | 古诗文、文言文、课文段落、名句默写 | 小学~高中 |
| 英语 | 单词、短语、课文段落、经典句型 | 小学~高中 |
| 历史 | 时间线、事件、人物、条约/法案名称 | 初中~高中 |
| 政治 | 核心概念、原理表述、时政要点 | 初中~高中 |
| 地理 | 地名、气候类型、地形特征、区域知识 | 初中~高中 |
| 生物 | 名词定义、分类体系、实验步骤 | 初中~高中 |

### 1.3 链路总览

```
P1 入口与材料选择 → P2 预习与记忆辅助 → P3 背诵检测 → P4 结果评估与反馈
→ P5 复习调度与间隔重复 → P6 数据沉淀与跨模块联动
```

### 3.1 入口场景

| 入口 | 触发方式 | 说明 |

| 学科背诵页 | 底部Tab → 背诵模块 → 选择学科 | 主入口 |

| 复习提醒 | 通知推送 → 点击进入 | 到期复习材料 |
| AI辅导联动 | AI对话中提到背诵 → "帮你背诵" | 关联材料进入背诵 |

```python
class ReciteMaterial(Base):
    """背诵材料主表"""
    __tablename__ = "recite_material"

    id: Mapped[str] = Column(String(32), primary_key=True)  # RM+snowflake
    title: Mapped[str] = Column(String(200), nullable=False)  # 材料标题
    content: Mapped[str] = Column(Text, nullable=False)  # 完整原文
    subject: Mapped[str] = Column(String(20), nullable=False)  # 学科枚举
    material_type: Mapped[str] = Column(String(20), nullable=False)  # MATERIAL_TYPE
    grade_range: Mapped[str] = Column(String(20), nullable=False)  # 学段
    difficulty: Mapped[int] = Column(SmallInteger, default=3)  # 1-5难度

    # 来源关联
    textbook_id: Mapped[str | None] = Column(String(32))  # 关联教材
    chapter_id: Mapped[str | None] = Column(String(32))  # 关联章节
    knowledge_point_ids: Mapped[str] = Column(String(500), default="[]")  # JSON数组

    # 结构化内容
    author: Mapped[str | None] = Column(String(100))  # 作者(古诗文)
    dynasty: Mapped[str | None] = Column(String(50))  # 朝代(古诗文)
    segments: Mapped[str] = Column(Text, nullable=False)  # JSON: 分段数据
    total_chars: Mapped[int] = Column(Integer, nullable=False)  # 总字数
    total_segments: Mapped[int] = Column(Integer, nullable=False)  # 总段数

    # 辅助内容
    translation: Mapped[str | None] = Column(Text)  # 译文
    annotation: Mapped[str | None] = Column(Text)  # 注释(JSON)
    background: Mapped[str | None] = Column(Text)  # 背景/AI讲解
    keywords: Mapped[str | None] = Column(String(500))  # 关键词,逗号分隔

    # 配置
    recite_mode: Mapped[str] = Column(String(20), default="full")  # RECITE_MODE
    time_limit_seconds: Mapped[int | None] = Column(Integer)  # 建议背诵时长
    required_accuracy: Mapped[float] = Column(Float, default=0.85)  # 达标准确率

    # 元数据
    status: Mapped[str] = Column(String(20), default="active")  # MATERIAL_STATUS
    source: Mapped[str] = Column(String(20), default="official")  # MATERIAL_SOURCE
    created_at: Mapped[datetime] = Column(DateTime, server_default=func.now())
    updated_at: Mapped[datetime] = Column(DateTime, onupdate=func.now())

    # 倒排索引字段(ES同步)
    search_text: Mapped[str | None] = Column(Text)  # 搜索用文本

    __table_args__ = (
        Index("idx_rm_subject_grade", "subject", "grade_range"),
        Index("idx_rm_chapter", "chapter_id"),
        Index("idx_rm_type", "material_type"),
    )
```

```dart
/// 材料列表Tab内容
class _MaterialListTab extends ConsumerWidget {

  final String? chapterId;

        limit: limit,
```

```python
class MemoryAidEngine:
    """记忆辅助引擎"""

    def get_hint(self, content: str, key_chars: str | None,
                 hint_level: int = 1) -> str:
        """hint_level: 1=首字提示, 2=关键词提示, 3=结构提示"""
        if hint_level == 1:
            return self._first_char_hint(content)
        elif hint_level == 2:
            return self._keyword_hint(content, key_chars)
        else:
            return self._structure_hint(content)

    def _first_char_hint(self, text: str) -> str:
        """首字提示：每句只保留首字"""
        result = []
        for sentence in re.split(r'([。！？；，、\n])', text):
            if not sentence:
                continue
            if re.match(r'[。！？；，、\n]', sentence):
                result.append(sentence)
            else:
                result.append(sentence[0] + '＿' * (len(sentence) - 1))
        return ''.join(result)

    def _keyword_hint(self, text: str, key_chars: str | None) -> str:
        """关键词提示"""
        if not key_chars:
            return self._first_char_hint(text)
        keywords = set(key_chars.split(','))
        return ''.join(
            ch if (ch in keywords or re.match(r'[。！？；，、：""''（）\s]', ch))
            else '＿'
            for ch in text
        )

    def _structure_hint(self, text: str) -> str:
        """结构提示：仅展示开头和结尾"""
        if len(text) <= 20:
            return text
        return text[:10] + '...(省略)...' + text[-10:]
```

### 5.4 文本对比算法

```python
class ReciteComparator:
    """背诵文本对比引擎"""

    def compare(self, expected: str, actual: str,
                subject: str = "chinese") -> ReciteCompareResult:
        """对比用户输入与原文"""
        exp_clean = self._normalize(expected, subject)
        act_clean = self._normalize(actual, subject)

        alignment = self._align_sequences(exp_clean, act_clean)
        char_diff = self._build_char_diff(alignment)

        accuracy = self._calc_accuracy(alignment)
        completeness = self._calc_completeness(exp_clean, alignment)
        errors = self._classify_errors(char_diff)

        return ReciteCompareResult(
            accuracy=accuracy,
            completeness=completeness,
            char_diff=char_diff,
            error_positions=[e["position"] for e in errors if e["type"] == "wrong"],
            missing_parts=[e for e in errors if e["type"] == "missing"],
            extra_parts=[e for e in errors if e["type"] == "extra"],
        )

    def _normalize(self, text: str, subject: str) -> str:
        """文本标准化"""
        text = re.sub(r'\s+', '', text)
        if subject in ("chinese", "history", "politics", "geography"):
            text = re.sub(r'[，。！？；：""''（）]', '', text)
        if subject == "english":
            text = text.lower()
        return text

    def _align_sequences(self, expected: str, actual: str) -> list:
        """基于LCS的序列对齐"""
        m, n = len(expected), len(actual)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if expected[i-1] == actual[j-1]:
                    dp[i][j] = dp[i-1][j-1] + 1
                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])

        alignment = []
        i, j = m, n
        while i > 0 or j > 0:
            if i > 0 and j > 0 and expected[i-1] == actual[j-1]:
                alignment.append(("match", i-1, j-1))
                i -= 1; j -= 1
            elif j > 0 and (i == 0 or dp[i][j-1] >= dp[i-1][j]):
                alignment.append(("extra", None, j-1))
                j -= 1
            else:
                alignment.append(("missing", i-1, None))
                i -= 1
        alignment.reverse()
        return alignment

    def _calc_accuracy(self, alignment: list) -> float:
        matches = len([a for a in alignment if a[0] == "match"])
        total = len(alignment)
        return matches / max(total, 1)

    def _calc_completeness(self, expected: str, alignment: list) -> float:
        matches = len([a for a in alignment if a[0] == "match"])
        return matches / max(len(expected), 1)
```

### 5.5 语音背诵处理

```python
class VoiceReciteProcessor:
    """语音背诵处理器"""

    async def process_voice_recite(
        self, audio_file_id: str, expected_text: str,
        subject: str, user_id: str
    ) -> ReciteCompareResult:
        """处理语音背诵"""
        # 1. ASR识别
        asr_result = await self.asr_service.recognize(
            audio_file_id=audio_file_id,
            language="zh-CN" if subject == "chinese" else "en-US",
            enable_punctuation=True,
        )

        # 2. 文本对比
        comparator = ReciteComparator()
        result = comparator.compare(expected_text, asr_result.text, subject)

        # 3. 流利度评估
        result.fluency = self._assess_fluency(asr_result, expected_text)
        return result

    def _assess_fluency(self, asr_result, expected: str) -> float:
        """流利度评估：语速+停顿"""
        duration = asr_result.duration_seconds
        if duration <= 0:
            return 0.0
        speed = len(expected) / duration
        speed_score = max(0, 1.0 - abs(speed - 3.5) / 3.5)
        pause_ratio = len(asr_result.pauses) / max(len(expected), 1)
        pause_score = max(0, 1.0 - pause_ratio * 10)
        return round(0.6 * speed_score + 0.4 * pause_score, 3)
```

### 5.6 填空默写生成器

```python
class FillBlankGenerator:
    """填空默写生成器"""

    def generate_blanks(self, content: str, material_type: str,
                        difficulty: int = 3) -> list[BlankItem]:
        """根据难度生成填空项"""
        blanks = []
        if material_type == "ancient_poem":
            sentences = re.split(r'[。！？]', content)
            sentences = [s.strip() for s in sentences if s.strip()]
            for i, sent in enumerate(sentences):
                positions = self._select_blank_positions(sent, difficulty)
                display = self._apply_blanks(sent, positions)
                blanks.append(BlankItem(
                    segment_index=i,
                    display_text=display,
                    answer=sent,
                    blank_positions=positions,
                ))
        return blanks

    def _select_blank_positions(self, sentence: str, difficulty: int) -> list[int]:
        """按难度选择挖空位置"""
        n = len(sentence)
        if n == 0:
            return []
        blank_ratio = {1: 0.2, 2: 0.3, 3: 0.5, 4: 0.7, 5: 0.9}
        ratio = blank_ratio.get(difficulty, 0.5)
        num_blanks = max(1, int(n * ratio))
        # 优先挖非虚词位置
        positions = [i for i, ch in enumerate(sentence)
                     if not re.match(r'[的了着过是在和与而或，。、]', ch)]
        if len(positions) < num_blanks:
            positions = list(range(n))
        import random
        random.seed(hash(sentence))
        return sorted(random.sample(positions, min(num_blanks, len(positions))))
```

```python
class ReciteScorer:
    """背诵综合评分"""

    def calculate_score(self, accuracy: float, completeness: float,
                        fluency: float | None, input_type: str) -> float:
        """综合评分 0~100"""
        if input_type == "voice":
            weights = {"accuracy": 0.50, "completeness": 0.20, "fluency": 0.30}
            fluency = fluency or 0.0
        elif input_type == "text":
            weights = {"accuracy": 0.70, "completeness": 0.30, "fluency": 0.0}
        else:
            weights = {"accuracy": 0.60, "completeness": 0.40, "fluency": 0.0}

        score = (
            accuracy * weights["accuracy"]
            + completeness * weights["completeness"]
            + (fluency or 0.0) * weights["fluency"]
        ) * 100
        return round(min(100, max(0, score)), 1)

    def generate_feedback(self, score: float, errors: list,
                          grade_range: str) -> str:
        """生成分龄反馈"""
        if score >= 95:
            return "太棒了！你已经完全记住了！🌟"
        elif score >= 80:
            n = len(errors)
            return f"背诵得不错！还有{n}个小地方需要注意。"
        elif score >= 60:
            return "继续加油！有几个地方还不太熟悉。"
        else:
            return "还需要多加练习哦！建议先通读几遍原文。"
```

## 7. P5 - 复习调度与间隔重复

### 7.1 SM-2 算法实现

```python
class ReciteReviewScheduler:
    """背诵复习调度器 - SM-2 改良"""

    def schedule_next_review(
        self, progress: UserReciteProgress,
        score: float, accuracy: float
    ) -> ReviewSchedule:
        quality = self._score_to_quality(score)
        ef = progress.ease_factor
        rep = progress.repetition

        if quality >= 3:
            if rep == 0:
                interval = 1
            elif rep == 1:
                interval = 3
            else:
                interval = round(rep * ef)
            rep += 1
        else:
            rep = 0
            interval = 1

        ef = max(1.3, ef + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02)))

        if accuracy >= 0.95 and quality >= 4:
            interval = round(interval * 1.2)

        next_review = datetime.utcnow() + timedelta(days=max(1, interval))

        return ReviewSchedule(
            interval_days=max(1, interval),
            ease_factor=round(ef, 2),
            repetition=rep,
            next_review_at=next_review,
        )

    def _score_to_quality(self, score: float) -> int:
        """映射到SM-2 quality(0~5)"""
        if score >= 95: return 5
        if score >= 85: return 4
        if score >= 75: return 3
        if score >= 60: return 2
        if score >= 40: return 1
        return 0
```

### 7.2 更新背诵进度(内部服务)

```python
class ReciteService:
    async def update_progress_after_session(
        self, user_id: str, material_id: str, session: ReciteSession
    ) -> UserReciteProgress:
        """背诵会话结束后更新进度"""
        progress = await self._get_or_create_progress(user_id, material_id)

        progress.total_attempts += 1
        progress.best_accuracy = max(progress.best_accuracy, session.accuracy)
        progress.total_study_seconds += session.duration_seconds
        progress.last_recite_at = session.submitted_at

        if session.passed:
            progress.consecutive_correct += 1
        else:
            progress.consecutive_correct = 0

        scheduler = ReciteReviewScheduler()
        schedule = scheduler.schedule_next_review(
            progress, session.score, session.accuracy
        )
        progress.ease_factor = schedule.ease_factor
        progress.interval_days = schedule.interval_days
        progress.repetition = schedule.repetition
        progress.next_review_at = schedule.next_review_at

        progress.mastery_level = self._calc_mastery(progress, session)
        progress.status = self._determine_status(progress)

        await self.session.commit()

        await self.event_bus.publish(ReciteCompletedEvent(
            user_id=user_id,
            material_id=material_id,
            score=session.score,
            passed=session.passed,
            mastery_level=progress.mastery_level,
        ))
        return progress

    def _calc_mastery(self, p: UserReciteProgress, s: ReciteSession) -> float:
        return round(
            p.best_accuracy * 0.3
            + s.accuracy * 0.3
            + min(1.0, p.consecutive_correct / 3) * 0.2
            + min(1.0, p.repetition / 5) * 0.2,
            3,
        )

    def _determine_status(self, p: UserReciteProgress) -> str:
        if p.mastery_level >= 0.85 and p.consecutive_correct >= 2:
            return "mastered"
        if p.total_attempts > 0:
            return "studying" if p.mastery_level >= 0.6 else "needs_review"
        return "new"
```

## 10. 错误处理与降级策略

### 10.1 错误码定义

| 错误码 | 说明 | 处理方式 |
|--------|------|---------|
| 401001 | 材料不存在 | 提示用户返回列表 |
| 401002 | 材料已下架 | 提示并推荐替代材料 |
| 401003 | 无背诵权限(会员限制) | 引导开通会员 |
| 401010 | ASR识别失败 | 提示切换文本输入 |
| 401011 | ASR识别超时 | 提示重试或切换输入 |
| 401020 | 对比服务异常 | 降级为精确匹配 |
| 401030 | 复习调度失败 | 使用默认1天间隔 |
| 401040 | 会话已过期 | 提示重新开始 |
| 401050 | 录音权限未授予 | 引导设置权限 |

### 10.2 降级链路

```
正常链路:
  语音→ASR→LCS对比→AI反馈→SM-2调度

降级1(ASR异常):
  语音→客户端本地ASR→LCS对比→规则反馈→SM-2调度

降级2(ASR+对比服务异常):
  切换为文本输入模式→LCS对比→规则反馈→SM-2调度

降级3(全服务异常):
  本地精确匹配→简单正确/错误判断→本地缓存进度

降级4(AI反馈异常):
  使用预置模板反馈(按分数区间)
```
