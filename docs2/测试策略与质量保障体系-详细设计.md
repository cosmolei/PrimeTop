# 测试策略与质量保障体系 - 详细设计

## 1. 概述

### 1.1 文档定位

本文档为 PrimeTop 项目的测试策略与质量保障体系提供详细设计，覆盖客户端测试、服务端测试、AI 输出质量测试、性能测试、安全测试和持续集成流水线。目标是让开发团队直接建立可执行的测试体系，保障 MVP 至 V1.5 各阶段的产品质量。

### 1.2 质量目标

| 维度 | MVP 目标 | V1.0 目标 | 度量方式 |
|------|----------|-----------|----------|
| APP 崩溃率 | < 0.5% | < 0.2% | Crashlytics / Sentry 崩溃次数 ÷ 会话数 |
| 核心接口成功率 | ≥ 99% | ≥ 99.5% | 接口 5xx 比例 |
| AI 回答满意率 | ≥ 70% | ≥ 80% | 用户点赞 / (点赞+点踩) |
| 拍题识别准确率 | ≥ 85% | ≥ 92% | 正确识别题数 ÷ 总拍题数 |
| 首屏加载时间 | ≤ 2s | ≤ 1.5s | P95 客户端埋点 |
| AI 首 token 延迟 | ≤ 3s | ≤ 2s | P95 服务端埋点 |
| 回归测试通过率 | ≥ 95% | ≥ 98% | CI 流水线统计 |

### 1.3 测试分层策略

采用测试金字塔模型，各层比例建议：

```
         ╱  E2E Tests  ╲          ~10% — 端到端关键路径
        ╱────────────────╲
       ╱  Integration     ╲       ~25% — API 集成、服务间交互
      ╱    Tests           ╲
     ╱──────────────────────╲
    ╱    Unit Tests          ╲     ~65% — 业务逻辑、工具函数、状态管理
   ╱──────────────────────────╲
```

// test/providers/learning_plan_provider_test.dart

import 'package:flutter_test/flutter_test.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:mockito/mockito.dart';
import 'package:primetop/providers/learning_plan_provider.dart';
import 'package:primetop/services/learning_plan_service.dart';

class MockLearningPlanService extends Mock implements LearningPlanService {}

void main() {
  group('LearningPlanNotifier', () {
    late ProviderContainer container;
    late MockLearningPlanService mockService;

    setUp(() {
      mockService = MockLearningPlanService();
      container = ProviderContainer(
        overrides: [
          learningPlanServiceProvider.overrideWithValue(mockService),
        ],
      );
    });

    tearDown(() {
      container.dispose();
    });

    test('初始状态为 AsyncLoading', () {
      // 跳过自动加载，直接验证状态结构
      final notifier = container.read(learningPlanProvider.notifier);
      expect(notifier.debugState, isA<AsyncValue<List<LearningPlan>>>());
    });

    test('loadPlans 成功返回计划列表', () async {
      when(mockService.getPlans()).thenAnswer(
        (_) async => [
          LearningPlan(id: 'p1', title: '期末复习', status: PlanStatus.active),
        ],
      );

      await container.read(learningPlanProvider.notifier).loadPlans();

      final state = container.read(learningPlanProvider);
      expect(state.value, isNotNull);
      expect(state.value!.length, 1);
      expect(state.value!.first.title, '期末复习');
    });

    test('loadPlans 网络错误后可重试', () async {
      when(mockService.getPlans())
          .thenThrow(Exception('Network error'));

      await container.read(learningPlanProvider.notifier).loadPlans();

      final state = container.read(learningPlanProvider);
      expect(state.hasError, isTrue);

      // 重试成功
      when(mockService.getPlans()).thenAnswer(
        (_) async => [],
      );
      await container.read(learningPlanProvider.notifier).loadPlans();

      final retryState = container.read(learningPlanProvider);
      expect(retryState.hasValue, isTrue);
    });
  });
}

import pytest
import io
from PIL import Image

class TestPhotoSearchAPI:
    """拍照搜题 API 集成测试"""

    @pytest.fixture
    def mock_image(self):
        """生成测试用模拟题目图片"""
        img = Image.new('RGB', (800, 600), color='white')
        buf = io.BytesIO()
        img.save(buf, format='JPEG')
        buf.seek(0)
        return buf

    async def test_photo_search_success(self, client, auth_headers, mock_image):
        """正常拍照搜题"""
        response = await client.post(
            "/api/v1/questions/photo-search",
            headers=auth_headers,
            files={"image": ("test.jpg", mock_image, "image/jpeg")},
            data={"subject": "math"},
        )
        assert response.status_code == 200
        data = response.json()
        assert "questions" in data
        assert "session_id" in data

    async def test_photo_search_no_auth(self, client, mock_image):
        """未认证请求应返回 401"""
        response = await client.post(
            "/api/v1/questions/photo-search",
            files={"image": ("test.jpg", mock_image, "image/jpeg")},
        )
        assert response.status_code == 401

    async def test_photo_search_invalid_format(self, client, auth_headers):
        """非图片文件应返回 400"""
        fake_file = io.BytesIO(b"not an image")
        response = await client.post(
            "/api/v1/questions/photo-search",
            headers=auth_headers,
            files={"image": ("test.txt", fake_file, "text/plain")},
        )
        assert response.status_code == 400
        assert "image" in response.json()["detail"].lower()

    async def test_photo_search_rate_limit(self, client, auth_headers, mock_image):
        """超频请求应返回 429"""
        # 连续请求超过限制 (免费用户每日10次)
        for _ in range(11):
            response = await client.post(
                "/api/v1/questions/photo-search",
                headers=auth_headers,
                files={"image": ("test.jpg", mock_image, "image/jpeg")},
            )
        # 第 11 次应该被限流
        assert response.status_code == 429

    def ocr_service(self):

        {
            "file": "primary_math_001.jpg",
            "expected_text": "小明有 5 个苹果，小红给了他 3 个，小明现在有几个苹果？",
            "subject": "math",
            "expected_answer": "8",
            "tolerance": 0.9,  # 文本匹配容忍度
        },
        {
            "file": "junior_physics_001.jpg",
            "expected_text": "一个质量为 2kg 的物体",
            "subject": "physics",
            "tolerance": 0.85,
        },
        {
            "file": "senior_chemistry_001.jpg",
            "expected_text": "NaOH + HCl",
            "subject": "chemistry",
            "tolerance": 0.85,
        },
    ]

    @pytest.mark.parametrize("case", OCR_TEST_SET)
    async def test_ocr_accuracy(self, ocr_service, case):
        image_path = os.path.join(self.FIXTURES_DIR, case["file"])
        result = await ocr_service.recognize(image_path)

        # 文本匹配率
        similarity = self._text_similarity(result["text"], case["expected_text"])
        assert similarity >= case["tolerance"], \
            f"OCR 识别准确率不足: {similarity:.2f} < {case['tolerance']}"

    @staticmethod
    def _text_similarity(actual: str, expected: str) -> float:
        """简单的字符级相似度计算"""
        actual_clean = set(actual.replace(" ", ""))
        expected_clean = set(expected.replace(" ", ""))
        if not expected_clean:
            return 1.0
        intersection = actual_clean & expected_clean
        return len(intersection) / len(expected_clean)

### 8.2 自动评估规则

```python
# app/services/ai_quality_evaluator.py

from dataclasses import dataclass
from typing import List

@dataclass
class QualityCheckResult:
    score: float          # 0-1 综合评分
    passed: bool
    checks: dict          # 各维度检查结果


class AIQualityEvaluator:
    """AI 输出自动评估器"""

    # 评分权重
    WEIGHTS = {
        "completeness": 0.20,    # 完整性: 是否有实质性回答
        "accuracy_signal": 0.25, # 准确性信号: 公式/数值是否自洽
        "safety": 0.20,          # 安全性: 无敏感/不当内容
        "age_appropriate": 0.15, # 适龄性: 语言复杂度匹配学段
        "structure": 0.10,       # 结构性: 有步骤/分段/总结
        "helpfulness": 0.10,     # 有用性: 用户反馈信号
    }

    def evaluate(
        self,
        question: str,
        answer: str,
        subject: str,
        stage: str,
        grade: str,
    ) -> QualityCheckResult:
        checks = {}

        # 1. 完整性检查
        checks["completeness"] = self._check_completeness(question, answer)

        # 2. 准确性信号检查 (非绝对准确, 而是"看起来合理")
        checks["accuracy_signal"] = self._check_accuracy_signal(answer, subject)

        # 3. 安全性检查
        checks["safety"] = self._check_safety(answer)

        # 4. 适龄性检查
        checks["age_appropriate"] = self._check_age_appropriate(answer, stage)

        # 5. 结构性检查
        checks["structure"] = self._check_structure(answer)

        # 6. 有用性 (默认给中性分, 线上由用户反馈修正)
        checks["helpfulness"] = 0.7

        score = sum(
            self.WEIGHTS[k] * checks[k]
            for k in self.WEIGHTS
        )

        return QualityCheckResult(
            score=score,
            passed=score >= 0.6,
            checks=checks,
        )

    def _check_completeness(self, question: str, answer: str) -> float:
        if len(answer) < 20:
            return 0.0
        if "我无法" in answer or "抱歉" in answer[:20]:
            return 0.3  # 拒答给部分分
        return 1.0

    def _check_accuracy_signal(self, answer: str, subject: str) -> float:
        """检查公式和数值自洽性"""
        score = 1.0
        # 如果包含数学公式，检查 LaTeX 格式正确性
        if "$$" in answer or "\\(" in answer:
            # 简单检查: 开闭符号匹配
            open_count = answer.count("$$") + answer.count("\\(")
            close_count = answer.count("$$") + answer.count("\\)")
            if open_count != close_count:
                score -= 0.3
        return max(0.0, score)

    def _check_safety(self, answer: str) -> float:
        """基础安全检查"""
        unsafe_patterns = [
            "暴力", "色情", "赌博", "毒品",
            "自杀", "自残",
        ]
        for pattern in unsafe_patterns:
            if pattern in answer:
                return 0.0
        return 1.0

    def _check_age_appropriate(self, answer: str, stage: str) -> float:
        """检查语言复杂度是否匹配学段"""
        if stage == "kindergarten":
            # 幼儿: 短句、少复杂词
            avg_sentence_len = len(answer) / max(1, answer.count("。") + answer.count("！"))
            return 1.0 if avg_sentence_len < 30 else 0.5
        elif stage == "primary":
            avg_sentence_len = len(answer) / max(1, answer.count("。") + 1)
            return 1.0 if avg_sentence_len < 50 else 0.6
        return 1.0  # 初高中不限制

    def _check_structure(self, answer: str) -> float:
        """检查是否有结构化输出"""
        has_steps = "步骤" in answer or "第一步" in answer
        has_summary = "总结" in answer or "综上" in answer
        has_sections = answer.count("\n\n") >= 2
        score = 0.0
        if has_steps: score += 0.4
        if has_summary: score += 0.3
        if has_sections: score += 0.3
        return min(1.0, score)
```
