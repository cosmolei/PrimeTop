> **原始需求来源**：`docs/design/启硕-PrimeTop-全学段AI辅助学习软件项目设计文档.md` §6.5
> **关联文档**：

## 1. 设计目标

| 目标 | 说明 |
|------|------|
| 零门槛操作 | 3 岁幼儿无需阅读即可完成核心交互，依赖语音、图形和触摸 |
| 注意力维持 | 单次学习单元 ≤ 3 分钟，穿插奖励动画防止疲劳 |
| 分龄自适应 | 中班（4-5y）、大班（5-6y）、一年级（6-7y）三层难度自动切换 |
| 亲子友好 | 家长可一键介入，退出/跳转需家长验证 |
| 离线可用 | 核心学习包支持下载后离线使用 |
| 低端设备兼容 | 动画 60fps（中端设备），低端设备自动降级为静态反馈 |

```dart
/// 启蒙模块导航守卫
class LiteracyGuard {
  /// 家长验证守卫 — 退出模块、进入家长面板、修改设置需要家长验证
  static Future<bool> requireParentAuth(BuildContext context) async {
    final authService = context.read<ParentAuthService>();
    if (authService.isVerified && !authService.isExpired) {
      return true;
    }
    // 弹出家长验证弹窗（数字密码 / 生物识别）
    final result = await showDialog<bool>(
      context: context,
      barrierDismissible: false,
      builder: (_) => const ParentVerifyDialog(),
    );
    return result ?? false;
  }

  /// 学习单元时长守卫 — 超过设定时长时强制休息
  static Future<bool> checkTimeLimit(BuildContext context) async {
    final limitService = context.read<StudyTimeLimitService>();
    if (limitService.shouldBreak) {
      await _showBreakDialog(context);
      return false;
    }
    return true;
  }

  /// 离线资源检查守卫
  static Future<bool> checkOfflineResource(
    BuildContext context,
    String resourceId,
  ) async {
    final offlineService = context.read<OfflineResourceService>();
    if (!offlineService.isReady(resourceId) && !await offlineService.download(resourceId)) {
      if (context.mounted) {
        _showDownloadPrompt(context, resourceId);
      }
      return false;
    }
    return true;
  }
}
```

```dart
// lib/features/literacy/bloc/literacy_home/literacy_home_bloc.dart

@freezed
class LiteracyHomeState with _$LiteracyHomeState {
  const factory LiteracyHomeState({
    @Default(false) bool isLoading,
    @Default(false) bool isOffline,
    UserProfile? userProfile,
    List<DailyTask>? todayTasks,
    LiteracyProgress? progress,
    GrowthStats? growthStats,
    String? errorMessage,
  }) = _LiteracyHomeState;
}

// 事件
@freezed
class LiteracyHomeEvent with _$LiteracyHomeEvent {
  const factory LiteracyHomeLoad() = _Load;
  const factory LiteracyHomeRefresh() = _Refresh;
  const factory LiteracyHomeTaskTapped(String taskId) = _TaskTapped;
  const factory LiteracyHomeQuickAccess(QuickAccessType type) = _QuickAccess;
}

// BLoC
class LiteracyHomeBloc extends Bloc<LiteracyHomeEvent, LiteracyHomeState> {
  final LiteracyRepository literacyRepository;
  final ProgressRepository progressRepository;

  LiteracyHomeBloc({
    required this.literacyRepository,
    required this.progressRepository,
  }) : super(const LiteracyHomeState()) {
    on<_Load>(_onLoad);
    on<_Refresh>(_onRefresh);
    on<_TaskTapped>(_onTaskTapped);
    on<_QuickAccess>(_onQuickAccess);
  }

  Future<void> _onLoad(_Load event, Emitter<LiteracyHomeState> emit) async {
    emit(state.copyWith(isLoading: true));
    try {
      final results = await Future.wait([
        literacyRepository.getUserProfile(),
        literacyRepository.getDailyTasks(),
        progressRepository.getLiteracyProgress(),
        progressRepository.getGrowthStats(),
      ]);
      emit(state.copyWith(
        isLoading: false,
        userProfile: results[0] as UserProfile,
        todayTasks: results[1] as List<DailyTask>,
        progress: results[2] as LiteracyProgress,
        growthStats: results[3] as GrowthStats,
      ));
    } catch (e) {
      emit(state.copyWith(isLoading: false, errorMessage: e.toString()));
    }
  }

  Future<void> _onTaskTapped(
    _TaskTapped event,
    Emitter<LiteracyHomeState> emit,
  ) async {
    final task = state.todayTasks?.firstWhere((t) => t.id == event.taskId);
    if (task == null) return;
    // 导航到对应学习页面
    _navigateToTask(task);
  }

  void _navigateToTask(DailyTask task) {
    // 根据任务类型路由到不同页面
    switch (task.type) {
      case TaskType.pinyinFollowRead:
        context.go(LiteracyRoutes.pinyinDetail.replaceAll(':id', task.targetId));
      case TaskType.characterWriting:
        context.go(LiteracyRoutes.strokePractice.replaceAll(':charId', task.targetId));
      case TaskType.miniGame:
        context.go(LiteracyRoutes.miniGame.replaceAll(':type', task.targetId));
    }
  }
}
```

```dart
// lib/features/literacy/pages/pinyin_detail_page.dart

class PinyinDetailPage extends StatefulWidget {
  final String pinyinId;

  const PinyinDetailPage({super.key, required this.pinyinId});

  @override
  State<PinyinDetailPage> createState() => _PinyinDetailPageState();
}

class _PinyinDetailPageState extends State<PinyinDetailPage>
    with TickerProviderStateMixin {
  late final PinyinLearningCubit _cubit;
  late final AnimationController _scaleController;
  late final AnimationController _mouthController;

  @override
  void initState() {
    super.initState();
    _cubit = PinyinLearningCubit(
      pinyinRepository: context.read<PinyinRepository>(),
      audioService: context.read<AudioService>(),
      speechService: context.read<SpeechAssessmentService>(),
    )..loadPinyin(widget.pinyinId);

    // 拼音放大/缩小动画
    _scaleController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 600),
    );

    // 口型动画控制器
    _mouthController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1200),
    );
  }

  @override
  Widget build(BuildContext context) {
    return BlocProvider(
      create: (_) => _cubit,
      child: BlocBuilder<PinyinLearningCubit, PinyinLearningState>(
        builder: (context, state) {
          return Scaffold(
            body: Stack(
              children: [
                // 背景装饰
                const _PinyinBackground(),
                // 主内容区
                SafeArea(
                  child: Column(
                    children: [
                      // 顶部导航栏（返回 + 拼音名称 + 进度）
                      _TopBar(pinyin: state.pinyin, progress: state.progress),
                      // 主体内容
                      Expanded(
                        child: AnimatedSwitcher(
                          duration: const Duration(milliseconds: 300),
                          child: _buildPhaseWidget(state),
                        ),
                      ),
                    ],
                  ),
                ),
                // 完成奖励浮层
                if (state.showReward) const _CompletionRewardOverlay(),
              ],
            ),
          );
        },
      ),
    );
  }

  Widget _buildPhaseWidget(PinyinLearningState state) {
    return switch (state.phase) {
      PinyinLearnPhase.intro => _IntroPhaseWidget(
        pinyin: state.pinyin!,
        onPlaySound: _cubit.playSound,
        onStart: _cubit.startMouthGuide,
      ),
      PinyinLearnPhase.mouthGuide => _MouthGuideWidget(
        pinyin: state.pinyin!,
        mouthAnimation: _mouthController,
        onStartRecord: _cubit.startFollowRead,
      ),
      PinyinLearnPhase.followRead => _FollowReadWidget(
        pinyin: state.pinyin!,
        isRecording: state.isRecording,
        onRecordComplete: _cubit.evaluatePronunciation,
```

#### 笔画检测引擎

```dart
// lib/features/literacy/engine/stroke_detector.dart

/// 笔画路径匹配检测器
/// 将用户绘制路径与目标笔画路径进行相似度比较
class StrokeDetector {
  /// 匹配阈值：0.0 ~ 1.0，幼儿模式阈值更宽松
  final double threshold;

  /// 路径采样点数（统一化后）
  final int samplePoints;

  StrokeDetector({
    this.threshold = 0.65,
    this.samplePoints = 32,
  });

  /// 根据学段调整阈值
  factory StrokeDetector.forStage(AgeStage stage) {
    return StrokeDetector(
      threshold: switch (stage) {
        AgeStage.kindergarten => 0.55,  // 幼儿更宽松
        AgeStage.grade1 => 0.65,
        AgeStage.grade2 => 0.70,
        _ => 0.75,
      },
    );
  }

  /// 评估笔画
  StrokeEvaluateResult evaluate({
    required List<Offset> userPoints,
    required List<Offset> targetPoints,
  }) {
    if (userPoints.length < 5) {
      return const StrokeEvaluateResult(
        quality: StrokeQuality.tooShort,
        score: 0.0,
        feedback: '再多画一点哦',
      );
    }

    // 1. 归一化两组点到相同尺寸
    final normalizedUser = _normalize(userPoints);
    final normalizedTarget = _normalize(targetPoints);

    // 2. 重采样到统一点数
    final resampledUser = _resample(normalizedUser, samplePoints);
    final resampledTarget = _resample(normalizedTarget, samplePoints);

    // 3. 计算路径方向相似度
    final directionScore = _directionSimilarity(resampledUser, resampledTarget);

    // 4. 计算形状相似度（DTW 距离）
    final shapeScore = _shapeSimilarity(resampledUser, resampledTarget);

    // 5. 综合评分
    final score = directionScore * 0.4 + shapeScore * 0.6;

    if (score >= threshold) {
      return StrokeEvaluateResult(
        quality: StrokeQuality.correct,
        score: score,
        feedback: _positiveFeedback(),
      );
    } else if (score >= threshold - 0.15) {
      return StrokeEvaluateResult(
        quality: StrokeQuality.close,
        score: score,
        feedback: '差一点点，再试一次',
      );
    } else {
      return StrokeEvaluateResult(
        quality: StrokeQuality.incorrect,
        score: score,
        feedback: _encourageRetryFeedback(),
      );
    }
  }

  /// 归一化：将点集映射到 [0,1] x [0,1]
  List<Offset> _normalize(List<Offset> points) {
    if (points.isEmpty) return [];
    final xs = points.map((p) => p.dx);
    final ys = points.map((p) => p.dy);
    final minX = xs.reduce(min);
    final maxX = xs.reduce(max);
    final minY = ys.reduce(min);
    final maxY = ys.reduce(max);
    final rangeX = maxX - minX;
    final rangeY = maxY - minY;
    final scale = max(rangeX, rangeY);
    if (scale == 0) return [const Offset(0.5, 0.5)];
    return points
        .map((p) => Offset((p.dx - minX) / scale, (p.dy - minY) / scale))
        .toList();
  }

  /// 等距重采样
  List<Offset> _resample(List<Offset> points, int n) {
    final totalLength = _pathLength(points);
    final interval = totalLength / (n - 1);
    final result = <Offset>[points.first];
    var accumulated = 0.0;

    for (var i = 1; i < points.length; i++) {
      final d = (points[i] - points[i - 1]).distance;
      if (accumulated + d >= interval) {
        final ratio = (interval - accumulated) / d;
        final newPoint = Offset.lerp(points[i - 1], points[i], ratio)!;
        result.add(newPoint);
        accumulated = 0.0;
      } else {
        accumulated += d;
      }
    }
    while (result.length < n) {
      result.add(points.last);
    }
    return result;
  }

  double _pathLength(List<Offset> points) {
    var length = 0.0;
    for (var i = 1; i < points.length; i++) {
      length += (points[i] - points[i - 1]).distance;
    }
    return length;
  }

  double _directionSimilarity(List<Offset> user, List<Offset> target) {
    // 比较相邻点段的方向向量
    var score = 0.0;
    for (var i = 0; i < min(user.length, target.length) - 1; i++) {
      final dUser = (user[i + 1] - user[i]);
      final dTarget = (target[i + 1] - target[i]);
      final cos = _cosineSimilarity(dUser, dTarget);
      score += (cos + 1) / 2; // [-1,1] → [0,1]
    }
    return score / (min(user.length, target.length) - 1);
  }

  double _cosineSimilarity(Offset a, Offset b) {
    final dot = a.dx * b.dx + a.dy * b.dy;
    final magA = sqrt(a.dx * a.dx + a.dy * a.dy);
    final magB = sqrt(b.dx * b.dx + b.dy * b.dy);
    if (magA == 0 || magB == 0) return 0;
    return (dot / (magA * magB)).clamp(-1.0, 1.0);
  }

  double _shapeSimilarity(List<Offset> user, List<Offset> target) {
    // 简化 DTW：逐点欧氏距离均值
    var totalDist = 0.0;
    for (var i = 0; i < min(user.length, target.length); i++) {
      totalDist += (user[i] - target[i]).distance;
    }
    final avgDist = totalDist / min(user.length, target.length);
    // 距离越小分数越高，0 距离 → 1.0，最大距离 √2 → 0
    return (1.0 - avgDist / sqrt2).clamp(0.0, 1.0);
  }

  String _positiveFeedback() {
    const options = ['太棒了！🌟', '写得真好！✨', '真厉害！👏', '漂亮！🎉'];
    return options[DateTime.now().millisecond % options.length];
  }

  String _encourageRetryFeedback() {
    const options = ['没关系，再试试', '加油，你可以的！💪', '慢慢来，不着急'];
    return options[DateTime.now().millisecond % options.length];
  }
}

enum StrokeQuality { tooShort, incorrect, close, correct }

  final StrokeQuality quality;
  final double score;
  final String feedback;
  const StrokeEvaluateResult({
    required this.quality,
    required this.score,
    required this.feedback,
```

```dart
    return ListView.separated(
      padding: const EdgeInsets.symmetric(vertical: 8),
      itemCount: _leftItems.length,
      separatorBuilder: (_, __) => const SizedBox(height: 12),
      itemBuilder: (context, index) {
        final item = _leftItems[index];
        final isConnected = _connections.any((c) => c.leftIndex == index);
        final isSelected = _selectedLeftIndex == index;

        return _MatchItemCard(
          item: item,
          isSelected: isSelected,
          isConnected: isConnected,
          onTap: () => _onLeftTap(index),
        );
      },
    );
  }

  Widget _buildRightColumn() {
    return ListView.separated(
      padding: const EdgeInsets.symmetric(vertical: 8),
      itemCount: _rightItems.length,
      separatorBuilder: (_, __) => const SizedBox(height: 12),
      itemBuilder: (context, index) {
        final item = _rightItems[index];
        final isConnected = _connections.any((c) => c.rightIndex == index);

        return _MatchItemCard(
          item: item,
          isSelected: false,
          isConnected: isConnected,
          onTap: () => _onRightTap(index),
        );
      },
    );
  }

  void _onLeftTap(int index) {
    setState(() {
      _selectedLeftIndex = _selectedLeftIndex == index ? null : index;
    });
  }

  void _onRightTap(int rightIndex) {
    if (_selectedLeftIndex == null) return;

    final leftItem = _leftItems[_selectedLeftIndex!];
    final rightItem = _rightItems[rightIndex];

    // 检查配对是否正确
    final isCorrect = widget.config.pairs.any(
      (p) => p.left.id == leftItem.id && p.right.id == rightItem.id,
    );

    if (isCorrect) {
      setState(() {
        _connections.add(MatchPair(
          leftIndex: _selectedLeftIndex!,
          rightIndex: rightIndex,
        ));
        _score++;
        _selectedLeftIndex = null;
      });
      AudioService().playSfx('correct_match');

      if (_score >= _totalPairs) {
        _onGameComplete();
      }
    } else {
      AudioService().playSfx('wrong_match');
      _showWrongAnimation(rightIndex);
      setState(() => _selectedLeftIndex = null);
    }
  }

  void _onGameComplete() {
    _celebrationController.forward(from: 0);
    AudioService().playSfx('level_complete');
    Future.delayed(const Duration(seconds: 2), () {
      widget.onComplete?.call();
    });
  }
}

        ..strokeWidth = 2.0
        ..style = PaintingStyle.stroke;

    }
  }
```

```dart
// lib/features/literacy/services/offline_resource_service.dart

class OfflineResourceService {
  final String basePath;
  final Dio _dio;
  final OfflineStorage _storage;

  /// 资源包版本文件
  static const _manifestFile = 'manifest.json';

  Future<bool> isReady(String resourceId) async {
    final manifest = await _loadManifest();
    if (manifest == null) return false;
    return manifest.resources.containsKey(resourceId) &&
        manifest.resources[resourceId]!.isDownloaded;
  }

  Future<bool> download(String resourceId, {
    void Function(double progress)? onProgress,
  }) async {
    // 检查网络和存储空间
    final info = await _getResourceInfo(resourceId);
    final availableSpace = await _storage.availableSpace();
    if (info.size > availableSpace) {
      throw StorageException('存储空间不足，需要 ${(info.size / 1024 / 1024).toStringAsFixed(1)} MB');
    }

    // 下载资源包
    final savePath = '$basePath/${info.packageName}';
    await _dio.download(
      info.downloadUrl,
      savePath,
      onReceiveProgress: (received, total) {
        onProgress?.call(received / total);
      },
    );

    // 解压
    await _storage.extract(savePath, basePath);

    // 更新 manifest
    final manifest = await _loadManifest();
    manifest?.resources[resourceId] = ResourceEntry(
      id: resourceId,
      isDownloaded: true,
      version: info.version,
      downloadedAt: DateTime.now(),
    );
    await _saveManifest(manifest!);

    return true;
  }

  /// 获取资源文件路径
  Future<String?> getResourcePath(String relativePath) async {
    final fullPath = '$basePath/$relativePath';
    if (await File(fullPath).exists()) return fullPath;
    return null;
  }
}
```

| 录音权限被拒 | 跟读/朗读练习 | 弹窗引导开启权限 | 提供跳过选项 |
| 语音评测失败 | 发音评分超时 | "评分中…"等待后重试 | 自动重试一次 |
| 资源下载失败 | 离线包下载 | 进度条红色 + 重试按钮 | 断点续传 |
| 笔画数据缺失 | 某汉字无笔画数据 | 跳过描红阶段 | 降级到仅观看动画 |
| 存储空间不足 | 下载离线包 | 提示清理空间 | 计算所需空间 |

### 10.2 统一错误处理

```dart
// lib/features/literacy/errors/literacy_error_handler.dart

class LiteracyErrorHandler {
  static void handleError(BuildContext context, Object error) {
    if (error is NetworkException) {
      _showBubble(context, '网络不给力，试试离线学习吧 📴');
    } else if (error is MicrophonePermissionDenied) {
      _showPermissionDialog(context);
    } else if (error is SpeechEvaluationException) {
      _showBubble(context, '发音评分遇到问题，再试一次吧');
    } else if (error is StorageException) {
      _showBubble(context, '存储空间不够了，让爸爸妈妈帮忙清理一下');
    } else {
      _showBubble(context, '出了点小问题，重新试试看');
    }

    // 上报错误日志
    _reportError(error);
  }

  static void _showBubble(BuildContext context, String message) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
        duration: const Duration(seconds: 3),
      ),
    );
  }

  static void _showPermissionDialog(BuildContext context) {
    showDialog(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('需要录音权限'),
        content: const Text('跟读练习需要使用麦克风，请在设置中开启'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('跳过'),
          ),
          ElevatedButton(
            onPressed: () {
              openAppSettings();
              Navigator.pop(context);
            },
            child: const Text('去设置'),
          ),
        ],
      ),
    );
  }
}
```

```dart
// 资源预加载 — 在当前学习单元进行时，预加载下一个单元的资源
class LiteracyPreloadWorker {
  final OfflineResourceService _offlineService;
  final LiteracyAudioService _audioService;

  /// 预加载下一个学习单元的资源
  Future<void> preloadNextUnit(LearningUnit current) async {
    final nextUnitId = current.nextUnitId;
    if (nextUnitId == null) return;

    // 并行预加载：音频 + 图片 + 笔画数据
    await Future.wait([
      _audioService.preloadPinyinAudio(nextUnitId),
      _preloadImages(nextUnitId),
      _preloadStrokeData(nextUnitId),
    ]);
  }
}
```
