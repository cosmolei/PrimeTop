# 管理后台 - 学情数据分析与教育运营驾驶舱工作台 详细设计

## 1. 模块概述

### 1.1 功能定位

学情数据分析与教育运营驾驶舱工作台是 PrimeTop 管理后台的核心数据决策模块，面向运营人员、内容教研团队和管理层，提供全局学情数据的可视化分析、多维下钻、趋势对比、异常预警和报表导出能力。

该工作台与 `服务端-管理后台数据分析看板与运营驾驶舱服务` 的后端数据聚合服务对接，聚焦于**管理后台前端页面架构、交互流程、可视化组件和数据分析操作体验**的详细设计。

### 1.2 目标用户

| 角色 | 核心诉求 | 使用场景 |
| --- | --- | --- |
| 运营专员 | 监控日活/留存/转化，评估活动效果 | 每日巡检，活动复盘 |
| 内容教研 | 查看知识点掌握分布、内容使用率 | 内容迭代决策 |
| 产品经理 | 分析功能使用率、用户行为路径 | 需求优先级判断 |
| 管理层 | 掌握全局健康度、收入趋势 | 周报/月报汇报 |
| 数据分析师 | 多维下钻、自定义查询、导出原始数据 | 深度分析 |

### 1.3 设计目标

1. **一屏掌握全局**：核心指标首屏可见，3 秒内获取系统健康度
2. **多维自由下钻**：支持学段/年级/学科/地区/时间等多维度交叉分析
3. **异常主动预警**：关键指标异常波动时自动高亮并推送告警
4. **灵活报表导出**：支持定时报告和临时导出，满足汇报需求
5. **操作一致性好**：统一的筛选器、图表组件和交互模式

### 2.2 全局筛选器设计

```
┌─────────────────────────────────────────────────────────────┐
│  时间范围: [今天|昨天|近7天|近30天|本月|自定义] 📅         │
│  学段: [全部|幼儿|小学|初中|高中]                           │

│                                          [查询] [重置] [导出]│
└─────────────────────────────────────────────────────────────┘
```

**筛选器交互规范：**

- 时间范围默认"近7天"，切换后所有图表联动刷新
- 学段/年级/学科为级联关系：选"初中"后年级自动过滤为初一~初三
- 地区支持省→市级联展开
- 筛选条件变化后 URL Query 参数同步，支持分享链接还原视图
- 每次筛选变化记录埋点 `admin_dashboard_filter_changed`

### 3.2 核心指标卡定义

每个指标卡统一结构：

```typescript
interface MetricCard {
  key: string;                    // 指标唯一标识
  title: string;                  // 显示名称
  value: number | string;         // 当前值
  formattedValue: string;         // 格式化后显示 (如 "128,540")
  unit?: string;                  // 单位 (如 "%", "人", "元")
  trend: 'up' | 'down' | 'flat';  // 趋势方向
  trendValue?: string;            // "12.3%"
  trendLabel: string;             // "环比昨日" / "环比上周"
  trendGood: 'up' | 'down' | 'auto'; // 上升是否为好（DAU上升好，流失率上升坏）
  sparkline?: number[];           // 迷你折线图数据
  alertThreshold?: {              // 预警阈值
    min?: number;
    max?: number;
  };
  drillDownLink?: string;         // 点击跳转的下钻页面
}
```

### 4.2 留存矩阵组件设计

```typescript
// CohortRetentionMatrix.tsx
interface RetentionMatrixProps {
  data: RetentionRow[];
  granularity: 'daily' | 'weekly' | 'monthly';
  maxDays: number; // 展示最大天数
}

interface RetentionRow {
  cohortDate: string;      // 注册日期 "2026-07-15"
  cohortSize: number;      // 该日注册人数
  retentionRates: (number | null)[]; // [null, 0.45, 0.32, ...] 对应 Day0~DayN
}

// 单元格颜色映射
function getCellColor(rate: number): string {
  if (rate >= 0.40) return '#1a9876'; // 深绿
  if (rate >= 0.25) return '#69d2a0'; // 浅绿
  if (rate >= 0.15) return '#f5d061'; // 黄
  if (rate >= 0.08) return '#f5a25d'; // 橙
  if (rate >= 0.03) return '#e87b5d'; // 红
  return '#c94c4c';                   // 深红
}

// 组件渲染逻辑
function CohortRetentionMatrix({ data, granularity, maxDays }: RetentionMatrixProps) {
  const periods = ['Day1', 'Day3', 'Day7', 'Day14', 'Day30', 'Day60', 'Day90']
    .slice(0, maxDays);

  return (
    <table className="cohort-matrix">
      <thead>
        <tr>
          <th>注册日期</th>
          <th>注册人数</th>
          {periods.map(p => <th key={p}>{p}</th>)}
        </tr>
      </thead>
      <tbody>
        {data.map(row => (
          <tr key={row.cohortDate}>
            <td>{row.cohortDate}</td>
            <td>{row.cohortSize.toLocaleString()}</td>
            {row.retentionRates.map((rate, i) => (
              <td
                key={i}
                style={{
                  backgroundColor: rate != null ? getCellColor(rate) : '#f5f5f5',
                  color: rate != null && rate >= 0.15 ? '#333' : '#fff'
                }}
              >
                {rate != null ? `${(rate * 100).toFixed(1)}%` : '-'}
              </td>
            ))}
          </tr>
        ))}
      </tbody>
    </table>
  );
}
```

### 5.2 行为路径桑基图数据结构

```typescript
// 用户行为路径桑基图数据
interface BehaviorPathData {
  nodes: PathNode[];
  links: PathLink[];
}

interface PathNode {
  name: string;       // "首页" | "AI对话" | "拍题" | ...
  step: number;       // 0, 1, 2, ... (路径深度)
  userCount: number;  // 到达该节点的用户数
}

interface PathLink {
  source: string;     // 源节点名
  target: string;     // 目标节点名
  value: number;      // 流转用户数
  conversionRate: number; // 转化率
}

// 路径分析查询参数
interface PathQueryParams {
  startDate: string;
  endDate: string;
  startEvent?: string;     // 起始事件（默认 app_open）
  maxSteps: number;        // 最大路径深度
  minUserCount: number;    // 最小用户数过滤
  segmentBy?: 'grade' | 'subject' | 'channel'; // 分组维度
}
```

```typescript
// 预警状态机
type AlertStatus = 'open' | 'acknowledged' | 'resolved' | 'ignored';

const alertStatusTransitions: Record<AlertStatus, AlertStatus[]> = {
  open: ['acknowledged', 'resolved', 'ignored'],
  acknowledged: ['resolved', 'ignored'],
  resolved: [],     // 终态
  ignored: ['open'] // 可重新打开
};

function canTransition(from: AlertStatus, to: AlertStatus): boolean {
  return alertStatusTransitions[from]?.includes(to) ?? false;
```

```sql
CREATE MATERIALIZED VIEW analytics.daily_metrics_mv
TO analytics.daily_metrics AS
SELECT
    toDate(created_at) AS stat_date,
    stage,
    grade,
    subject,
    countDistinctIf(user_id, event_type = 'app_open') AS dau,
    countDistinctIf(user_id, event_type = 'register') AS new_users,
    sumIf(duration, event_type = 'learning_session') / nullIf(dau, 0) AS avg_learning_time,
    countIf(event_type = 'ai_call') AS ai_call_count,
    countIf(event_type = 'ai_call' AND status = 'error') / nullIf(countIf(event_type = 'ai_call'), 0) AS ai_error_rate
FROM events.user_behavior_log
WHERE created_at >= yesterday() - 1
GROUP BY stat_date, stage, grade, subject;
```

```typescript
// 缓存层级设计
interface CacheStrategy {
  // Level 1: 浏览器内存缓存（React state / context）
  browserMemory: {
    ttl: 'until-refresh',           // 用户手动刷新前
    invalidation: 'manual-refresh', // 下拉刷新或路由切换
  };

  // Level 2: 浏览器 localStorage（跨标签页共享）
  localStorage: {
    ttl: 300_000,                   // 5分钟
    key: (page: string, filters: object) =>
      `analytics:${page}:${JSON.stringify(filters)}`,
  };

  // Level 3: 服务端 Redis 缓存
  redis: {
    // 驾驶舱概览数据
    dashboardOverview: {
      ttl: 300,                     // 5分钟 (秒)
      key: 'analytics:dashboard:overview:{dateRange}:{stage}',
      refresh: 'incremental',       // 增量刷新
    },
    // 留存矩阵
    retentionMatrix: {
      ttl: 3600,                    // 1小时
      key: 'analytics:retention:{startDate}:{endDate}:{segment}',
    },
    // 通用查询结果
    genericQuery: {
      ttl: 600,                     // 10分钟
      key: 'analytics:query:{hash(params)}',
    },
  };
}

// 前端缓存 Hook
function useAnalyticsData<T>(
  fetcher: () => Promise<T>,
  cacheKey: string,
  ttl: number = 300_000
): { data: T | null; loading: boolean; error: Error | null; refresh: () => void } {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const fetchData = useCallback(async (force = false) => {
    // 检查本地缓存
    if (!force) {
      const cached = localStorage.getItem(cacheKey);
      if (cached) {
        const { data: cachedData, timestamp } = JSON.parse(cached);
        if (Date.now() - timestamp < ttl) {
          setData(cachedData);
          setLoading(false);
          return;
        }
      }
    }

    setLoading(true);
    try {
      const result = await fetcher();
      setData(result);
      setError(null);
      localStorage.setItem(cacheKey, JSON.stringify({
        data: result,
        timestamp: Date.now(),
      }));
    } catch (err) {
      setError(err as Error);
    } finally {
      setLoading(false);
    }
  }, [cacheKey, ttl]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  return { data, loading, error, refresh: () => fetchData(true) };
}
```

```typescript
  toggleSubject: (subject: string) => void;

  setChannel: (channels: string[]) => void;
  reset: () => void;

  // 序列化为 URL Query 参数

  // 从 URL Query 参数恢复

export const useGlobalFilter = create<GlobalFilterState>((set, get) => ({
  dateRange: '7d',
  startDate: dayjs().subtract(7, 'day').format('YYYY-MM-DD'),
  endDate: dayjs().format('YYYY-MM-DD'),
  stage: [],
  grade: [],
  subject: [],
  region: [],
  channel: [],

  setDateRange: (range) => {
    if (range === 'custom') {
      set({ dateRange: range });
      return;
    }
    const ranges = {
      today: [0, 0],
      yesterday: [-1, -1],
      '7d': [-6, 0],
      '30d': [-29, 0],
      month: ['monthStart', 0],
    };
    const [startOffset, endOffset] = ranges[range] || ranges['7d'];
    set({
      dateRange: range,
      startDate: dayjs().add(startOffset as number, 'day').format('YYYY-MM-DD'),
      endDate: dayjs().add(endOffset as number, 'day').format('YYYY-MM-DD'),
    });
  },

  setCustomDateRange: (start, end) =>
    set({ dateRange: 'custom', startDate: start, endDate: end }),

  toggleStage: (stage) => {
    const current = get().stage;
    set({
      stage: current.includes(stage)
        ? current.filter(s => s !== stage)
        : [...current, stage],
    });
  },

  toggleGrade: (grade) => {
    const current = get().grade;
    set({
      grade: current.includes(grade)
        ? current.filter(g => g !== grade)
        : [...current, grade],
    });
  },

  toggleSubject: (subject) => {
    const current = get().subject;
    set({
      subject: current.includes(subject)
        ? current.filter(s => s !== subject)
        : [...current, subject],
    });
  },

  setRegion: (regions) => set({ region: regions }),
  setChannel: (channels) => set({ channel: channels }),

  reset: () =>
    set({
      dateRange: '7d',
      startDate: dayjs().subtract(7, 'day').format('YYYY-MM-DD'),
      endDate: dayjs().format('YYYY-MM-DD'),
      stage: [], grade: [], subject: [], region: [], channel: [],

      start: s.startDate,
      end: s.endDate,

      grade: s.grade.join(','),
      subject: s.subject.join(','),
      region: s.region.join(','),
      channel: s.channel.join(','),

      dateRange: (params.range as any) || '7d',
      startDate: params.start || dayjs().subtract(7, 'day').format('YYYY-MM-DD'),
      endDate: params.end || dayjs().format('YYYY-MM-DD'),

      grade: params.grade ? params.grade.split(',') : [],
      subject: params.subject ? params.subject.split(',') : [],
      region: params.region ? params.region.split(',') : [],
      channel: params.channel ? params.channel.split(',') : [],

  });
```

```typescript
// 轮询导出任务状态
function useExportTask(taskId: string | null) {
  const [task, setTask] = useState<ExportTask | null>(null);

  useEffect(() => {
    if (!taskId) return;

    const poll = setInterval(async () => {
      const res = await fetch(`/api/admin/analytics/export/${taskId}`);
      const data: ExportTask = await res.json();
      setTask(data);

      if (data.status === 'completed' || data.status === 'failed') {
        clearInterval(poll);

        if (data.status === 'completed') {
          // 触发浏览器下载
          window.open(data.downloadUrl, '_blank');
          // 显示完成通知
          message.success('导出完成，文件已开始下载');
        } else {
          message.error(`导出失败: ${data.error}`);
        }
      }
    }, 3000); // 每3秒轮询

    return () => clearInterval(poll);
  }, [taskId]);

  return task;
}
```

```python
class AnalyticsExportService:

    async def create_export_task(self, request: ExportRequest) -> ExportTask:
        task_id = generate_uuid()
        task = ExportTask(
            task_id=task_id,
            status='pending',
            created_at=datetime.now()
        )
        await self.task_store.save(task)

        # 投递到异步队列
        await self.queue.enqueue(
            'analytics_export',
            {'task_id': task_id, 'request': request.dict()}
        )
        return task

    async def process_export(self, task_id: str, request: ExportRequest):
        task = await self.task_store.get(task_id)
        task.status = 'processing'
        await self.task_store.save(task)

        try:
            # 查询数据
            data = await self.query_service.query(
                metrics=request.filters.metrics,
                dimensions=request.filters.dimensions,
                filters=request.filters.filters,
                limit=request.max_rows or 10000
            )

            task.progress = 50
            await self.task_store.save(task)

            # 生成文件
            if request.type == 'csv':
                file_path = await self.generate_csv(data, request.columns)
            elif request.type == 'excel':
                file_path = await self.generate_excel(data, request.columns)
            elif request.type == 'pdf':
                file_path = await self.generate_pdf(data, request.page)

            # 上传到对象存储
            download_url = await self.oss.upload(
                file_path,
                key=f'exports/analytics/{task_id}.{request.type}',
                expire_seconds=3600  # 1小时过期
            )

            task.status = 'completed'
            task.download_url = download_url
            task.file_size = os.path.getsize(file_path)
            task.expires_at = datetime.now() + timedelta(hours=1)

        except Exception as e:
            task.status = 'failed'
            task.error = str(e)
            logger.error(f'Export task {task_id} failed: {e}', exc_info=True)

        finally:
            await self.task_store.save(task)
```
