# 端到端流程设计 — 练习测评完整链路

## 1. 文档概述

### 1.1 目的

本文档描述 PrimeTop **练习测评**功能的端到端完整链路，覆盖从学生发起练习到最终学习数据沉淀的全过程，包括客户端交互、服务端处理、跨模块协作、异常降级等环节，供开发人员直接编码实现。

### 1.2 范围

覆盖以下练习测评场景：
- 章节同步练习（按教材章节出题）
- 专项练习（按知识点/题型/难度）
- 智能推荐练习（自适应选题）
- 模拟测试（限时、固定题量）
- 错题重练（从错题本发起）

```python
# service/practice/service.py

class PracticeService:
    async def create_session(
        self,
        user_id: str,
        req: CreatePracticeRequest,
        ctx: RequestContext,
    ) -> PracticeSessionVO:
        """
        创建练习会话主流程
        """
        # 1. 额度预检
        await self.quota_svc.check_and_deduct(
            user_id=user_id,
            quota_type=QuotaType.PRACTICE_SESSION,
            amount=1,
            context={'mode': req.mode},
        )

        # 2. 获取用户画像上下文
        profile = await self.profile_repo.get_learning_profile(user_id)
        mastery_map = await self.mastery_repo.get_mastery_batch(
            user_id, subject=req.subject,
        )

        # 3. 根据模式分发到对应生成策略
        strategy = self._resolve_strategy(req.mode)
        questions = await strategy.assemble(
            req=req,
            profile=profile,
            mastery_map=mastery_map,
            user_id=user_id,
        )

        if not questions:
            raise BizError(ErrorCode.NO_SUITABLE_QUESTIONS,
                          "当前条件下没有可用的题目，请调整练习范围")

        # 4. 构建会话
        session = PracticeSession(
            id=generate_id('prac'),
            user_id=user_id,
            mode=req.mode,
            subject=req.subject,
            chapter_id=req.chapterId,
            question_ids=[q.id for q in questions],
            questions=[PracticeQuestionItem.from_question(q) for q in questions],
            total_count=len(questions),
            time_limit=req.timeLimit or self._default_time_limit(req.mode),
            status=PracticeStatus.NOT_STARTED,
            created_at=utcnow(),
            metadata={
                'difficulty': req.difficulty,
                'client_version': ctx.client_version,
                'device_id': ctx.device_id,
            },
        )

        # 5. 持久化
        await self.session_repo.create(session)

        # 6. 写入Redis缓存(答题过程中高频读写)
        await self._cache_session(session)

        # 7. 埋点
        await self.event_pub.publish(PracticeSessionCreatedEvent(
            session_id=session.id,
            user_id=user_id,
            mode=req.mode,
            subject=req.subject,
            question_count=len(questions),
            knowledge_point_ids=req.knowledgePointIds,
        ))

        # 8. 返回VO(不包含答案)
        return PracticeSessionVO.from_entity(session, include_answer=False)
```

```python
# service/practice/strategy.py

class ChapterStrategy(AssembleStrategy):
    """章节同步练习：按章节知识点均匀分布"""

    async def assemble(self, req, profile, mastery_map, **kwargs) -> list[Question]:
        # 获取章节下所有知识点
        kps = await self.kp_repo.list_by_chapter(req.chapterId)

        # 按掌握度分层: 薄弱知识点分配更多题目
        weak_kps = [kp for kp in kps if mastery_map.get(kp.id, 1.0) < 0.6]
        normal_kps = [kp for kp in kps if kp not in weak_kps]

        budget = req.questionCount or 10
        weak_budget = max(budget * 0.6, len(weak_kps) * 2)  # 薄弱知识点占60%
        normal_budget = budget - weak_budget

        questions = []
        questions += await self._pick_from_kps(weak_kps, int(weak_budget),
                                                difficulty='medium', mastery_map=mastery_map)
        questions += await self._pick_from_kps(normal_kps, int(normal_budget),
                                                difficulty='basic', mastery_map=mastery_map)

        # 去重: 排除用户近期做过的题(7天内)
        recent_ids = await self._get_recent_done_ids(kwargs['user_id'], days=7)
        questions = [q for q in questions if q.id not in recent_ids]

        # 补足数量
        while len(questions) < budget:
            fill = await self._pick_from_kps(kps, budget - len(questions),
                                              difficulty='medium', mastery_map=mastery_map,
                                              exclude_ids={q.id for q in questions} | recent_ids)
            if not fill:
                break
            questions.extend(fill)

        random.shuffle(questions)
        return questions[:budget]


class SmartRecommendStrategy(AssembleStrategy):
    """智能推荐：基于自适应模型(Elo)动态选题"""

    async def assemble(self, req, profile, mastery_map, **kwargs) -> list[Question]:
        count = req.questionCount or 6

        # 获取用户当前Elo能力估计
        user_elo = await self.elo_repo.get_user_elo(
            kwargs['user_id'], subject=req.subject,
        )

        # 目标难度区间: 用户Elo ± 200 (zone of proximal development)
        target_min = user_elo - 200
        target_max = user_elo + 200

        questions = []
        for i in range(count):
            # 每道题基于已选题目的反馈动态调整下道题难度
            expected_difficulty = self._map_elo_to_difficulty(user_elo, i, count)

            candidates = await self.question_repo.search(
                subject=req.subject,
                difficulty=expected_difficulty,
                exclude_ids={q.id for q in questions},
                limit=5,
            )

            if candidates:
                # 选择曝光度最低的题(均衡曝光)
                best = min(candidates, key=lambda q: q.exposure_count)
                questions.append(best)

        return questions


class MistakeRetryStrategy(AssembleStrategy):
    """错题重练：从错题本筛选待复习的错题"""

    async def assemble(self, req, profile, mastery_map, **kwargs) -> list[Question]:
        user_id = kwargs['user_id']
        count = req.questionCount or 8

        # 查询到期应复习的错题(遗忘曲线调度)
        due_mistakes = await self.mistake_repo.list_due_for_review(
            user_id=user_id,
            subject=req.subject,
            limit=count * 2,  # 多取一些用于去重
            date_range=req.mistakeDateRange,
        )

        # 优先级排序: 到期 > 即将到期 > 超期未复习
        due_mistakes.sort(key=lambda m: (
            m.review_overdue_days,   # 超期天数(降序)
            -m.keep_confidence,       # 保持信心度(升序=低信心优先)
            m.correct_streak,         # 连续正确次数(升序)
        ), reverse=True)

        questions = []
        seen_ids = set()
        for m in due_mistakes[:count]:
            if m.question_id not in seen_ids:
                q = await self.question_repo.get(m.question_id)
                if q:
                    # 包裹为重练题目,携带原始错因信息
                    q._mistake_meta = {
                        'mistake_id': m.id,
                        'original_error_type': m.error_type,
                        'review_count': m.review_count,
                        'last_review_at': m.last_review_at,
                    }
                    questions.append(q)
                    seen_ids.add(q.id)

        return questions


class MockExamStrategy(AssembleStrategy):
    """模拟测试：按试卷结构组卷(固定题量、限时)"""

    async def assemble(self, req, profile, mastery_map, **kwargs) -> list[Question]:
        count = req.questionCount or 25

        # 获取试卷模板(题型分布)
        template = await self.exam_template_repo.get_by_subject_grade(
            subject=req.subject,
            grade=profile.grade,
        )

        questions = []
        for slot in template.slots:
            # slot: { question_type, difficulty, count, knowledge_point_ids }
            batch = await self.question_repo.search(
                subject=req.subject,
                question_types=[slot.question_type],
                difficulty=slot.difficulty,
                knowledge_point_ids=slot.knowledge_point_ids,
                exclude_ids={q.id for q in questions},
                limit=slot.count * 3,
            )

            # 随机选取
            selected = random.sample(batch, min(slot.count, len(batch)))
            questions.extend(selected)

        # 确保总数正确
        if len(questions) > count:
            questions = questions[:count]

        # 按题型排序(选择→填空→计算→证明)
        questions.sort(key=lambda q: QUESTION_TYPE_ORDER.index(q.type))

        return questions
```

```dart
// 客户端: 练习会话本地缓存(离线支持)

class PracticeSessionCache {
  final HiveBox _box;

  /// 缓存会话(含题目数据)
  Future<void> cacheSession(PracticeSessionVO session) async {
    final key = 'practice_${session.sessionId}';
    await _box.put(key, jsonEncode(session.toJson()));

    // 预加载题目涉及的图片资源
    for (final q in session.questions) {
      if (q.images != null) {
        for (final url in q.images!) {
          CachedImagePreloader.instance.preload(url);
        }
      }
    }
  }

  /// 获取未完成的会话(断点续练)
  Future<PracticeSessionVO?> getResumableSession() async {
    final keys = _box.keys
        .where((k) => k.startsWith('practice_'))
        .toList();

    for (final key in keys) {
      final session = PracticeSessionVO.fromJson(
        jsonDecode(_box.get(key)),
      );
      // 检查是否未完成且未过期

        return session;
      }
    }
```

```python
# service/practice/grading_router.py

class GradingRouter:
    """根据题型路由到不同批改策略"""

    def __init__(self, exact_grader, rule_grader, ai_grader):
        self.exact = exact_grader       # 精确匹配(选择题/判断题)
        self.rule = rule_grader         # 规则匹配(填空题/计算题)
        self.ai = ai_grader             # AI批改(主观题/证明题)

    async def grade(self, question: Question, answer: UserAnswer) -> GradeResult:
        qtype = question.type

        # 一级路由: 客观题 vs 主观题
        if qtype in ('single_choice', 'multi_choice', 'true_false'):
            return await self.exact.grade(question, answer)

        elif qtype in ('fill_blank', 'calculation'):
            # 先尝试规则批改
            result = await self.rule.grade(question, answer)
            if result.confidence >= 0.95:
                return result
            # 规则不确定时降级到AI
            return await self.ai.grade(question, answer)

        elif qtype in ('short_answer', 'proof', 'essay', 'translation'):
            return await self.ai.grade(question, answer)

        else:
            return await self.ai.grade(question, answer)


class ExactGrader:
    """精确匹配批改"""

    async def grade(self, question, answer) -> GradeResult:
        standard = question.standard_answer
        user = answer.answer.strip()

        if question.type == 'single_choice':
            is_correct = (user.upper() == standard.upper())

        elif question.type == 'multi_choice':
            user_set = set(user.upper().replace(' ', '').split(','))
            standard_set = set(standard.upper().replace(' ', '').split(','))
            is_correct = (user_set == standard_set)

        elif question.type == 'true_false':
            is_correct = (user.lower() == standard.lower())

        return GradeResult(
            is_correct=is_correct,
            correctness=1.0 if is_correct else 0.0,
            grading_type='exact',
            confidence=1.0,
        )


class RuleGrader:
    """规则匹配批改(数学等价判定、填空容错)"""

    async def grade(self, question, answer) -> GradeResult:
        if question.type == 'fill_blank':
            return self._grade_fill(question, answer)
        elif question.type == 'calculation':
            return self._grade_calc(question, answer)
        return GradeResult(is_correct=None, correctness=None,
                          grading_type='rule', confidence=0.0)

    def _grade_fill(self, question, answer):
        standard_answers = question.acceptable_answers  # 可接受答案列表
        user = answer.answer.strip()

        for sa in standard_answers:
            # 精确匹配
            if user == sa:
                return GradeResult(is_correct=True, correctness=1.0,
                                  grading_type='rule', confidence=1.0)

            # 数学表达式等价判定
            if question.subject in ('math', 'physics', 'chemistry'):
                if MathEquivalence.check(user, sa):
                    return GradeResult(is_correct=True, correctness=1.0,
                                      grading_type='rule', confidence=0.98)

            # 模糊匹配(去空格/标点/大小写)
            if self._fuzzy_match(user, sa):
                return GradeResult(is_correct=True, correctness=1.0,
                                  grading_type='rule', confidence=0.95)

        return GradeResult(is_correct=False, correctness=0.0,
                          grading_type='rule', confidence=0.9)

    def _grade_calc(self, question, answer):
        # 提取最终数值结果
        user_value = MathParser.extract_final_value(answer.answer)
        standard_value = MathParser.extract_final_value(question.standard_answer)

        if user_value is not None and standard_value is not None:
            if abs(user_value - standard_value) < 1e-6:
                return GradeResult(is_correct=True, correctness=1.0,
                                  grading_type='rule', confidence=0.95)

        # 无法解析出数值, 降级到AI
        return GradeResult(is_correct=None, correctness=None,
                          grading_type='rule', confidence=0.3)
```

```python
# service/practice/service.py

class PracticeService:
    async def submit_session(
        self,
        session_id: str,
        user_id: str,
        force_submit: bool = False,
    ) -> PracticeReportVO:
        session = await self.session_repo.get(session_id)
        if not session or session.user_id != user_id:
            raise NotFoundError("练习会话不存在")

        if session.status in (PracticeStatus.SUBMITTED, PracticeStatus.COMPLETED):
            raise BizError(ErrorCode.ALREADY_SUBMITTED, "练习已提交")

        # 1. 标记所有未答题为skipped
        for q in session.questions:
            existing = await self.answer_repo.get_by_session_question(
                session_id, q.question_id,
            )
            if not existing:
                await self.answer_repo.create(PracticeAnswer(
                    id=generate_id('pans'),
                    session_id=session_id,
                    user_id=user_id,
                    question_id=q.question_id,
                    answer=None,
                    is_correct=None,
                    correctness=0.0,
                    grading_type='skipped',
                    seq_index=q.seq_index,
                ))

        # 2. 批量批改未批改的题(主要是主观题)
        answers = await self.answer_repo.list_by_session(session_id)
        ungraded = [a for a in answers if a.is_correct is None and a.answer is not None]

        grading_tasks = []
        for answer in ungraded:
            question = await self.question_repo.get(answer.question_id)
            grading_tasks.append(
                self.grading_router.grade(question, UserAnswer(answer=answer.answer))
            )

        if grading_tasks:
            results = await asyncio.gather(*grading_tasks, return_exceptions=True)
            for answer, result in zip(ungraded, results):
                if isinstance(result, Exception):
                    log.error(f"Grading failed for answer {answer.id}: {result}")
                    answer.is_correct = None
                    answer.grading_type = 'error'
                else:
                    answer.is_correct = result.is_correct
                    answer.correctness = result.correctness
                    answer.grading_type = result.grading_type
                    answer.grading_detail = result.detail

                await self.answer_repo.update(answer)

        # 3. 计算总分
        all_answers = await self.answer_repo.list_by_session(session_id)
        correct = sum(1 for a in all_answers if a.is_correct == True)
        wrong = sum(1 for a in all_answers if a.is_correct == False)
        partial = sum(1 for a in all_answers if a.correctness is not None
                      and 0 < a.correctness < 1.0)
        skipped = sum(1 for a in all_answers if a.answer is None)

        total_weight = sum(q.score_weight for q in session.questions)
        earned_weight = 0.0
        for a in all_answers:
            q = next(q for q in session.questions if q.question_id == a.question_id)
            earned_weight += q.score_weight * (a.correctness or 0.0)

        score = (earned_weight / total_weight * 100) if total_weight > 0 else 0

        # 4. 计算用时
        duration = int((utcnow() - session.started_at).total_seconds())

        # 5. 更新会话
        session.status = PracticeStatus.GRADED
        session.submitted_at = utcnow()
        session.correct_count = correct
        session.wrong_count = wrong
        session.partial_count = partial
        session.skipped_count = skipped
        session.score = round(score, 1)
        session.duration_seconds = duration
        await self.session_repo.update(session)

        # 6. 生成报告
        report = await self._build_report(session, all_answers)

        # 7. 发布事件(触发异步数据沉淀)
        await self.event_pub.publish(PracticeSubmittedEvent(
            session_id=session.id,
            user_id=user_id,
            mode=session.mode,
            subject=session.subject,
            score=session.score,
            correct_count=correct,
            wrong_count=wrong,
            total_count=session.total_count,
            duration_seconds=duration,
            chapter_id=session.chapter_id,
            knowledge_point_ids=session.knowledge_point_ids,
            answers=[
                AnswerEventData(
                    question_id=a.question_id,
                    is_correct=a.is_correct,
                    correctness=a.correctness,
                    time_spent_ms=a.time_spent_ms,
                    grading_type=a.grading_type,
                )
                for a in all_answers
            ],
        ))

        return report
```

```python
class PracticeReportBuilder:
    """构建测评报告"""

    async def build(
        self,
        session: PracticeSession,
        answers: list[PracticeAnswer],
    ) -> PracticeReportVO:
        # 1. 基础统计
        grade = self._calc_grade(session.score)

        # 2. 按知识点分析
        kp_stats = await self._build_kp_stats(session, answers)

        # 3. 按题型分析
        type_stats = self._build_type_stats(session, answers)

        # 4. 按难度分析
        diff_stats = self._build_difficulty_stats(session, answers)

        # 5. 用时分析
        time_analysis = self._build_time_analysis(answers)

        # 6. 生成AI建议(异步,有缓存时直接返回)
        advice = await self._generate_advice(session, kp_stats, answers)

        # 7. 推荐下一步
        next_suggestions = await self._build_next_suggestions(session, kp_stats)

        return PracticeReportVO(
            session_id=session.id,
            mode=session.mode,
            subject=session.subject,
            score=session.score,
            total_count=session.total_count,
            correct_count=session.correct_count,
            wrong_count=session.wrong_count,
            partial_count=session.partial_count,
            skipped_count=session.skipped_count,
            duration_seconds=session.duration_seconds,
            grade=grade,
            grade_label=GRADE_LABELS[grade],
            analysis={
                'by_knowledge_point': kp_stats,
                'by_question_type': type_stats,
                'by_difficulty': diff_stats,
                'time_analysis': time_analysis,
            },
            question_details=self._build_question_details(session, answers),
            recommendations={
                'advice': advice,
                'weak_points': [kp.name for kp in kp_stats if kp.accuracy < 0.6],
                'next_practice_suggestions': next_suggestions,
                'mistake_review_suggestions': await self._get_mistake_suggestions(
                    session, answers,
                ),
            },
        )

    async def _generate_advice(
        self,
        session: PracticeSession,
        kp_stats: list[KnowledgePointStat],
        answers: list[PracticeAnswer],
    ) -> str:
        """AI生成学习建议"""
        weak_kps = [kp for kp in kp_stats if kp.accuracy < 0.6]

        if not weak_kps:
            return self._get_random_encouragement(session.score)

        # 使用AI输出缓存(避免重复调用)
        cache_key = f"practice_advice:{session.subject}:{','.join(kp.name for kp in weak_kps)}"
        cached = await self.cache_svc.get(cache_key)
        if cached:
            return cached

        prompt = self._build_advice_prompt(session, weak_kps)
        advice = await self.ai_svc.generate(
            prompt=prompt,
            model='fast',    # 用轻量模型生成建议
            max_tokens=200,
        )

        await self.cache_svc.set(cache_key, advice, ttl=3600)
        return advice

    async def _build_next_suggestions(
        self,
        session: PracticeSession,
        kp_stats: list[KnowledgePointStat],
    ) -> list[NextPracticeSuggestion]:
        """推荐下一步练习"""
        weak_kps = [kp for kp in kp_stats if kp.accuracy < 0.6]
        suggestions = []

        for kp in weak_kps[:3]:  # 最多推荐3个专项练习
            suggestions.append(NextPracticeSuggestion(
                title=f"{kp.name} 专项练习",
                mode='topic',
                knowledge_point_ids=[kp.knowledge_point_id],
                estimated_count=max(3, int(kp.total_count * 1.5)),
                reason=f"本次正确率 {int(kp.accuracy*100)}%，建议加强练习",
            ))

        # 如果整体正确率较低,推荐章节复习
        if session.score < 60 and session.chapter_id:
            chapter = await self.chapter_repo.get(session.chapter_id)
            suggestions.append(NextPracticeSuggestion(
                title=f"复习「{chapter.title}」",
                mode='chapter',
                chapter_id=session.chapter_id,
                reason="章节整体掌握不足，建议重新学习",
            ))

        return suggestions
```

```python
class MasteryUpdateHandler(EventHandler):
    """知识点掌握度增量更新"""

    @handler('practice.submitted', channel='standard')
    async def handle(self, event: PracticeSubmittedEvent):
        for answer_data in event.answers:
            if answer_data.is_correct is None:
                continue

            # 获取题目关联的知识点
            kp_ids = await self.question_repo.get_knowledge_point_ids(
                answer_data.question_id,
            )

            for kp_id in kp_ids:
                # 贝叶斯增量更新
                current = await self.mastery_repo.get_or_init(
                    user_id=event.user_id,
                    knowledge_point_id=kp_id,
                )

                # 证据权重: 考虑题目难度和题型
                weight = self._calc_evidence_weight(
                    is_correct=answer_data.is_correct,
                    correctness=answer_data.correctness,
                    time_spent_ms=answer_data.time_spent_ms,
                )

                # Beta分布参数更新
                if answer_data.is_correct:
                    current.alpha += weight
                else:
                    current.beta += weight

                current.mastery = current.alpha / (current.alpha + current.beta)
                current.updated_at = utcnow()

                await self.mastery_repo.upsert(current)

    def _calc_evidence_weight(self, is_correct, correctness, time_spent_ms):
        """计算证据权重"""
        base = 1.0

        # 部分正确给部分权重
        if correctness is not None and 0 < correctness < 1.0:
            base = correctness

        # 用时因素: 太快可能蒙对, 太慢可能查了答案
        if time_spent_ms < 5000:    # <5秒
            base *= 0.7
        elif time_spent_ms > 300000: # >5分钟
            base *= 0.8

        return base


class MistakeCollectionHandler(EventHandler):
    """错题自动收录"""

    @handler('practice.submitted', channel='standard')
    async def handle(self, event: PracticeSubmittedEvent):
        for answer_data in event.answers:
            if answer_data.is_correct == False:
                await self.mistake_svc.add_mistake(
                    user_id=event.user_id,
                    question_id=answer_data.question_id,
                    source='practice',
                    source_session_id=event.session_id,
                    user_answer=answer_data.answer,
                    error_type=self._infer_error_type(answer_data),
                )

    def _infer_error_type(self, answer_data) -> str:
        """推测错因类型"""
        if answer_data.correctness is not None and answer_data.correctness > 0:
            return 'partial_understanding'  # 部分理解
        if answer_data.time_spent_ms and answer_data.time_spent_ms < 10000:
            return 'careless'               # 粗心
        return 'knowledge_gap'              # 知识缺口


class EloUpdateHandler(EventHandler):
    """用户Elo能力值更新"""

    @handler('practice.submitted', channel='standard')
    async def handle(self, event: PracticeSubmittedEvent):
        user_elo = await self.elo_repo.get_or_init(
            user_id=event.user_id,
            subject=event.subject,
        )

        for answer_data in event.answers:
            q = await self.question_repo.get(answer_data.question_id)
            q_elo = q.difficulty_elo or 1000  # 题目难度Elo

            # 标准Elo更新
            expected = 1.0 / (1.0 + 10 ** ((q_elo - user_elo.value) / 400))
            actual = answer_data.correctness or (1.0 if answer_data.is_correct else 0.0)

            # K因子: 根据做题数量调整
            k = self._calc_k_factor(user_elo.total_games)

            user_elo.value += k * (actual - expected)
            user_elo.total_games += 1

        await self.elo_repo.upsert(user_elo)

    def _calc_k_factor(self, total_games: int) -> float:
        if total_games < 10:
            return 40.0    # 新用户,快速校准
        elif total_games < 30:
            return 20.0
        else:
            return 10.0    # 稳定期,小幅调整
```

```python
                type='retry_mistakes',
                title='错题重练',
                description='趁热打铁，重新做一遍错题',

                    'source_session_id': session.id,

            actions.append(GuideAction(

                title=f'「{weak_types[0].typeName}」专项练习',
                description='针对薄弱题型加强训练',
                priority=80,

                type='share',
                title='分享今日练习成绩',
                priority=40,
            ))

        actions.sort(key=lambda a: a.priority, reverse=True)
```

| `PRACTICE_001` | 练习会话不存在 | 客户端清除本地缓存，返回首页 |
| `PRACTICE_002` | 会话已提交/已完成 | 展示已有报告 |

| `PRACTICE_004` | AI批改超时 | 降级为规则批改或标记待人工复核 |
| `PRACTICE_005` | 额度不足 | 引导会员购买/次数包 |
| `PRACTICE_006` | 超时自动提交 | 本地缓存答案一次性提交 |
| `PRACTICE_007` | 网络断开 | 离线答题队列，恢复后重试 |

| `PRACTICE_009` | 重复提交(幂等拦截) | 返回已有报告 |
| `PRACTICE_010` | 批改服务不可用 | 存入异步批改队列，稍后通知结果 |

### 10.2 降级矩阵

| 故障点 | 影响范围 | 降级策略 | 用户感知 |
|--------|---------|---------|---------|
| 题库服务不可用 | 无法组卷 | 使用客户端预缓存的练习包 | "当前使用离线题库" |
| AI批改不可用 | 主观题无法批改 | 规则批改兜底，标记待AI复核 | "部分题目稍后出结果" |
| Redis不可用 | 答题实时状态 | 降级到MySQL直接读写 | 响应变慢，功能正常 |

| 事件总线不可用 | 数据沉淀延迟 | 本地消息表+补偿扫描 | 用户无感知 |

```dart
// 客户端: 离线答题队列

class OfflineAnswerQueue {
  final HiveBox _box;

  /// 离线时缓存答题记录
  Future<void> enqueue({
    required String sessionId,
    required int seqIndex,
    required String answer,
    required int timeSpentMs,
    List<String>? answerImages,
  }) async {
    final record = OfflineAnswerRecord(
      sessionId: sessionId,
      seqIndex: seqIndex,
      answer: answer,
      timeSpentMs: timeSpentMs,
      answerImages: answerImages,
      createdAt: DateTime.now(),
      retryCount: 0,
    );
    await _box.add(record.toJson());
  }

  /// 网络恢复后批量提交
  Future<void> flushPending() async {
    final pending = _box.values.toList();
    if (pending.isEmpty) return;

    // 按会话分组
    final grouped = groupBy(pending, (r) => r['sessionId'] as String);

    for (final entry in grouped.entries) {
      final records = entry.value;
      // 按seqIndex排序后逐条提交
      records.sort((a, b) => (a['seqIndex'] as int).compareTo(b['seqIndex'] as int));

      for (final record in records) {
        try {
          await practiceApi.submitAnswer(
            sessionId: record['sessionId'],
            seqIndex: record['seqIndex'],
            answer: record['answer'],
            timeSpentMs: record['timeSpentMs'],
          );
          await _box.delete(record['localId']);
        } catch (e) {
          // 提交失败,增加重试计数
          record['retryCount'] = (record['retryCount'] as int) + 1;
          if (record['retryCount'] > 5) {
            // 超过5次放弃,标记失败
            record['status'] = 'failed';
          }
          await _box.put(record['localId'], record);
        }
      }
    }
  }
}
```

## 14. 安全考量

1. **答案防泄露**: 下发题目时不包含标准答案，答案由答案管控引擎按策略揭示
2. **防刷题**: 同一题目7天内不重复出现在同一用户的练习中
