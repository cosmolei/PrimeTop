# 服务端-学科思维导图自动生成与知识概念图智能构建引擎 - 详细设计

> **模块定位**：为全学段学生和教师提供思维导图/概念图的自动生成、智能构建、个性化定制与交互编辑能力。基于知识点体系、教材章节结构和 AI 大模型，自动生成学科思维导图；支持学生在复习、预习、知识总结等场景下创建和编辑个人思维导图；支持教师制作教学用概念图。
> **覆盖学段**：全学段（小学至高中）
> **优先级**：P2（V1.5 阶段交付，增强知识可视化与学习总结能力）
> **与知识图谱引擎的区别**：知识图谱引擎展示知识点间的图（Graph）拓扑关系（前置/关联/依赖）；思维导图引擎聚焦于层级化（Tree）知识分解与归纳总结，服务于复习梳理、知识沉淀和发散思考。

## 1. 问题域分析

### 1.1 核心场景

| 场景 | 用户角色 | 触发方式 | 典型示例 |
|------|----------|----------|----------|
| **章节复习导图** | 学生 | 章节学习完成 / 复习模式 | 学完"二次函数"后，自动生成以"二次函数"为中心的层级导图（定义→图像→性质→应用） |
| **考前知识梳理** | 学生 | 考点梳理 / 考试复习 | 中考前生成"函数"专题导图，串联一次函数、反比例函数、二次函数 |
| **预习导图** | 学生 | 课前预习 | 预习"牛顿第二定律"时生成概念导图，展示核心概念框架 |
| **AI 对话知识总结** | 学生 | AI 辅导对话结束后 | 多轮对话讨论"光合作用"后，一键生成讨论要点的思维导图 |
| **错题归因导图** | 学生 | 错题本 / 错因分析 | 按错因类型生成"我的易错知识树"，高亮高频错误分支 |
| **教师教学导图** | 教师 | 备课 / 课堂展示 | 教师创建"中国古代史"概念图，添加教学要点和补充说明 |
| **学科全景导图** | 学生/教师 | 学科总览 | "高中物理"全局知识树，可缩放查看各模块 |

### 1.3 设计挑战

| 挑战 | 描述 | 应对策略 |
|------|------|----------|
| **生成质量** | AI 生成的导图结构可能不合理、层级混乱 | Prompt 工程 + 知识点体系约束 + 后处理校验 |
| **性能** | 大型导图（500+节点）序列化和传输 | 分级加载 + 子树懒加载 + 增量更新 |
| **个性化** | 不同学段学生对同一知识的归纳深度不同 | 按学段/年级/认知水平动态调节层级和详细度 |
| **编辑协作** | 学生编辑后的导图与系统更新后的差异合并 | 版本快照 + 三方合并策略 |
| **与知识点同步** | 知识点体系更新后导图需要同步 | 变更事件订阅 + 影响范围计算 + 通知/自动迁移 |

```java
@Service
public class MindMapGenerationService {

    /**
     * 自动生成思维导图核心编排方法
     */
    @Transactional
    public MindMapDTO generateMindMap(GenerateMindMapRequest req, Long userId) {
        // 1. 参数校验
        validateRequest(req);

        // 2. 根据生成类型选择策略
        MindMapGenerationStrategy strategy = strategyFactory.get(req.getGenerateType());

        // 3. 生成导图骨架（树形结构）
        MindMapSkeleton skeleton = strategy.generate(req, userId);

        // 4. 后处理：知识点关联、掌握度叠加、考点标记
        postProcessor.process(skeleton, req, userId);

        // 5. 布局计算
        layoutCalculator.calculate(skeleton, req.getLayoutType());

        // 6. 持久化
        MindMap map = persistService.save(skeleton, req, userId);

        // 7. 生成版本快照
        snapshotService.createSnapshot(map.getId(), "auto", userId, "自动生成初版");

        // 8. 返回完整数据
        return assembleDTO(map);
    }
}
```

```java
@Component
public class ChapterBasedStrategy implements MindMapGenerationStrategy {

    @Override
    public MindMapSkeleton generate(GenerateMindMapRequest req, Long userId) {
        Chapter chapter = chapterService.getById(req.getChapterId());

        // 1. 获取章节下所有知识点（树形结构）
        List<KnowledgePoint> kpTree = knowledgePointService
            .getKpTreeByChapter(chapter.getId(), req.getDepth());

        // 2. 转换为导图骨架节点
        MindMapNode root = buildNodeFromChapter(chapter);

        // 3. 递归构建子节点
        buildChildrenFromKpTree(root, kpTree, req.getDepth(), 1);

        // 4. 限制节点数
        pruneExcessNodes(root, req.getMaxNodes());

        return MindMapSkeleton.builder()
            .root(root)
            .title(chapter.getName() + "知识梳理")
            .build();
    }

    private void buildChildrenFromKpTree(MindMapNode parent,
                                          List<KnowledgePoint> kpChildren,
                                          int maxDepth,
                                          int currentDepth) {
        if (currentDepth > maxDepth || kpChildren.isEmpty()) return;

        for (int i = 0; i < kpChildren.size(); i++) {
            KnowledgePoint kp = kpChildren.get(i);

            MindMapNode node = MindMapNode.builder()
                .nodeUid(UUID.randomUUID().toString())
                .nodeText(kp.getName())
                .level(currentDepth)
                .sortOrder(i)
                .kpId(kp.getId())
                .isKeyPoint(kp.getImportance() >= 4)
                .nodeColor(pickColorByLevel(currentDepth))
                .build();

            // 关联考点信息
            if (kp.getExamFrequency() != null && kp.getExamFrequency() >= 4) {
                node.setExamHot(true);
            }

            parent.addChild(node);

            // 递归
            if (CollectionUtils.isNotEmpty(kp.getChildren())) {
                buildChildrenFromKpTree(node, kp.getChildren(),
                    maxDepth, currentDepth + 1);
            }
        }
    }
}
```

```java
    private static final int MAX_SNAPSHOTS = 20;      // 保留最近20个快照
    private static final int AUTO_SNAPSHOT_INTERVAL = 5; // 每5次编辑自动快照

    @Transactional
    public EditResult applyEdit(Long mapId, EditRequest req, Long userId) {
        MindMap map = mindMapRepository.findById(mapId)
            .orElseThrow(() -> new BizException("MINDMAP_NOT_FOUND"));

        // 1. 乐观锁检查
        if (map.getVersion() != req.getBaseVersion()) {
            return handleVersionConflict(map, req);
        }

        // 2. 应用编辑操作
        applyNodeEdit(mapId, req);

        // 3. 版本递增
        int newVersion = map.getVersion() + 1;
        map.setVersion(newVersion);
        map.setUpdatedAt(LocalDateTime.now());
        mindMapRepository.save(map);

        // 4. 自动快照策略
        if (newVersion % AUTO_SNAPSHOT_INTERVAL == 0 || req.isForceSnapshot()) {
            snapshotService.createSnapshot(mapId, "auto", userId,
                String.format("第%d次编辑自动快照", newVersion));
            // 清理旧快照
            snapshotService.pruneOldSnapshots(mapId, MAX_SNAPSHOTS);
        }

        return EditResult.success(newVersion);
    }

    /**
     * 版本冲突处理 - 自动合并
     */
    private EditResult handleVersionConflict(MindMap map, EditRequest req) {
        // 获取服务端最新版本数据
        MindMapNode serverRoot = nodeService.getMapTree(map.getId());
        MindMapNode clientBaseRoot = req.getBaseRootNode();

        // 三方合并：base → server changes vs base → client changes
        MergeResult merge = threeWayMerge(
            clientBaseRoot,           // 客户端基线
            serverRoot,               // 服务端当前
            req.getEditOperations()    // 客户端编辑操作
        );

        if (merge.hasConflicts()) {
            // 返回冲突详情，让客户端/用户选择
            return EditResult.conflict(merge.getConflicts(), map.getVersion());
        } else {
            // 自动合并成功
            applyMergedChanges(map.getId(), merge.getMergedRoot());
            return EditResult.merged(map.getVersion() + 1, merge.getChanges());
        }
    }
}
```

#### 5.3.2 三方合并算法

```java
/**
 * 基于节点 UID 的树结构三方合并
 */
public class TreeThreeWayMerger {

    public MergeResult merge(MindMapNode base, MindMapNode server,
                             List<EditOperation> clientOps) {
        Map<String, EditOperation> clientOpsByUid = clientOps.stream()
            .collect(Collectors.toMap(EditOperation::getNodeUid, Function.identity()));

        List<NodeChange> changes = new ArrayList<>();
        List<Conflict> conflicts = new ArrayList<>();

        // 遍历 base 树的所有节点
        traverse(base, baseNode -> {
            String uid = baseNode.getNodeUid();
            EditOperation clientOp = clientOpsByUid.get(uid);
            NodeChange serverChange = diffNode(baseNode, findByUid(server, uid));

            if (clientOp != null && serverChange != null) {
                // 双方都修改了同一节点 - 冲突检测
                if (isContentConflict(clientOp, serverChange)) {
                    conflicts.add(new Conflict(uid, clientOp, serverChange));
                } else {
                    // 可自动合并（如修改了不同字段）
                    changes.add(mergeNonConflicting(clientOp, serverChange));
                }
            } else if (clientOp != null) {
                // 仅客户端修改 - 直接应用
                changes.add(new NodeChange(uid, clientOp));
            } else if (serverChange != null) {
                // 仅服务端修改 - 保留服务端变更
                changes.add(serverChange);
            }
        });

        // 检测客户端新增的节点
        for (EditOperation op : clientOps) {
            if (op.getAction() == Action.ADD_CHILD && findByUid(base, op.getNodeUid()) == null) {
                changes.add(new NodeChange(op.getNodeUid(), op));
            }
        }

        return new MergeResult(changes, conflicts);
    }
}
```

### 5.4 布局计算引擎

```java
@Component
public class MindMapLayoutCalculator {

    /**
     * 向右树状布局计算
     * 根节点在最左侧，子树向右展开
     */
    public void calculateRightTree(MindMapNode root, LayoutConfig config) {
        // 1. 计算每个子树的高度（占用的垂直空间）
        calculateSubtreeHeight(root, config);

        // 2. 分配坐标
        assignCoordinates(root, 0, 0, config);
    }

    private int calculateSubtreeHeight(MindMapNode node, LayoutConfig config) {
        if (node.getChildren().isEmpty()) {
            node.setSubtreeHeight(config.getNodeHeight());
            return config.getNodeHeight();
        }

        int totalHeight = 0;
        for (MindMapNode child : node.getChildren()) {
            totalHeight += calculateSubtreeHeight(child, config);
            totalHeight += config.getNodeSpacing();
        }
        totalHeight -= config.getNodeSpacing(); // 最后一个不需要间距

        int selfHeight = config.getNodeHeight();
        node.setSubtreeHeight(Math.max(selfHeight, totalHeight));
        return node.getSubtreeHeight();
    }

    private void assignCoordinates(MindMapNode node, int depth, int startY,
                                    LayoutConfig config) {
        // X 坐标 = 深度 × 层间距
        node.setX(depth * config.getLevelSpacing());

        if (node.getChildren().isEmpty()) {
            node.setY(startY);
            return;
        }

        int currentY = startY;
        int firstChildY = 0;
        int lastChildY = 0;

        for (int i = 0; i < node.getChildren().size(); i++) {
            MindMapNode child = node.getChildren().get(i);
            if (i == 0) firstChildY = currentY;
            assignCoordinates(child, depth + 1, currentY, config);
            currentY += child.getSubtreeHeight() + config.getNodeSpacing();
            if (i == node.getChildren().size() - 1) lastChildY = currentY - config.getNodeSpacing();
        }

        // 父节点 Y 坐标 = 第一个子节点和最后一个子节点的中点
        int centerY = (firstChildY + lastChildY) / 2;
        node.setY(centerY);
    }

    /**
     * 中心放射布局计算
     */
    public void calculateRadial(MindMapNode root, LayoutConfig config) {
        root.setX(0);
        root.setY(0);

        List<MindMapNode> level1 = root.getChildren();
        if (level1.isEmpty()) return;

        // 第一级节点均匀分布在圆周上
        double angleStep = 2 * Math.PI / level1.size();
        double radius = config.getLevelSpacing();

        for (int i = 0; i < level1.size(); i++) {
            MindMapNode node = level1.get(i);
            double angle = i * angleStep;
            node.setX((int)(radius * Math.cos(angle)));
            node.setY((int)(radius * Math.sin(angle)));

            // 递归布局子树（扇形区域内）
            layoutRadialSubtree(node, angle, angleStep, radius, 2, config);
        }
    }

    private void layoutRadialSubtree(MindMapNode node, double centerAngle,
                                      double parentAngularSpan,
                                      double parentRadius, int depth,
                                      LayoutConfig config) {
        if (node.getChildren().isEmpty()) return;

        double radius = parentRadius + config.getLevelSpacing();
        List<MindMapNode> children = node.getChildren();
        double angularSpan = parentAngularSpan / children.size();

        for (int i = 0; i < children.size(); i++) {
            MindMapNode child = children.get(i);
            double childAngle = centerAngle - parentAngularSpan / 2
                + angularSpan * (i + 0.5);

            child.setX((int)(radius * Math.cos(childAngle)));
            child.setY((int)(radius * Math.sin(childAngle)));

            layoutRadialSubtree(child, childAngle, angularSpan,
                radius, depth + 1, config);
        }
    }
}
```

## 6. 知识点变更同步机制

### 6.1 问题场景

当知识点体系发生变更（新增知识点、删除知识点、知识点合并、层级调整）时，已生成的导图可能出现：
- 节点关联的 kpId 失效
- 知识点名称变更导致不一致
- 新增知识点未在导图中体现

### 6.2 同步策略

```java
@Component
public class MindMapKpSyncService {

    /**
     * 监听知识点变更事件（通过事件总线）
     */
    @EventListener
    @Async("mindMapSyncExecutor")
    public void onKnowledgePointChanged(KpChangeEvent event) {
        switch (event.getChangeType()) {
            case RENAME:
                handleKpRename(event);
                break;
            case MERGE:
                handleKpMerge(event);
                break;
            case DELETE:
                handleKpDelete(event);
                break;
            case MOVE:
                handleKpMove(event);
                break;
            case ADD:
                handleKpAdd(event);
                break;
        }
    }

    /**
     * 知识点重命名：批量更新关联节点文本
     */
    private void handleKpRename(KpChangeEvent event) {
        // 查找所有引用了该 kpId 的导图节点
        List<MindMapNode> affected = nodeRepository.findByKpId(event.getKpId());

        for (MindMapNode node : affected) {
            // 仅更新自动生成的节点，保留用户自定义文本
            if (Boolean.TRUE.equals(node.getAutoGenerated())) {
                String oldText = node.getNodeText();
                node.setNodeText(event.getNewName());
                nodeRepository.save(node);

                // 记录变更日志
                changeLogService.log(node.getMapId(), node.getNodeUid(),
                    "kp_sync_rename", oldText, event.getNewName());
            }
        }

        // 标记受影响的导图需要刷新缓存
        Set<Long> affectedMapIds = affected.stream()
            .map(MindMapNode::getMapId)
            .collect(Collectors.toSet());
        cacheService.invalidateMindMapCache(affectedMapIds);
    }

    /**
     * 知识点删除：标记节点为"已失效"，通知用户
     */
    private void handleKpDelete(KpChangeEvent event) {
        List<MindMapNode> affected = nodeRepository.findByKpId(event.getKpId());

        Set<Long> mapIds = new HashSet<>();
        for (MindMapNode node : affected) {
            node.setKpId(null);
            node.setKpConfidence(null);
            node.setCustomTag("知识点已移除");
            nodeRepository.save(node);
            mapIds.add(node.getMapId());
        }

        // 发送通知
        notificationService.batchNotify(mapIds.stream()
            .map(mapId -> new MindMapSyncNotification(mapId,
                "知识点体系已更新，部分导图节点可能需要调整"))
            .collect(Collectors.toList()));
    }
}
```

```java
            return MindMapSkeleton.minimal(req.getTitle());

            throw new BizException("MINDMAP_GENERATION_FAILED",
                "导图生成服务暂时不可用，请稍后重试");
        }
    }
```

```dart
// 客户端离线编辑队列
class OfflineEditQueue {
  final List<MindMapEvent> _queue = [];
  bool _isSyncing = false;

  /// 添加离线编辑操作
  void enqueue(MindMapEvent event) {
    _queue.add(event);
    _trySync();
  }

  /// 网络恢复后同步
  Future<void> sync() async {
    if (_isSyncing || _queue.isEmpty) return;
    _isSyncing = true;

    while (_queue.isNotEmpty) {
      final event = _queue.first;
      try {
        final result = await mindMapApi.patch(event);
        if (result.hasConflict) {
          // 版本冲突：弹出合并界面
          _showConflictResolver(result);
          _isSyncing = false;
          return;
        }
        _queue.removeFirst();
      } catch (e) {
        // 重试3次后放弃，标记为同步失败
        log.error('Sync failed: $e');
        _isSyncing = false;
        return;
      }
    }
    _isSyncing = false;
  }
}
```

```java
    /**
     * 导出为 Markdown 大纲格式
     */
    private void exportAsMarkdown(MindMapTreeDTO tree, HttpServletResponse response) {
        StringBuilder sb = new StringBuilder();
        sb.append("# ").append(tree.getTitle()).append("\n\n");
        exportNodeAsMarkdown(tree.getRootNode(), 0, sb);

        response.setContentType("text/markdown; charset=UTF-8");
        response.getWriter().write(sb.toString());
    }

    private void exportNodeAsMarkdown(MindMapNodeDTO node, int level, StringBuilder sb) {
        if (level > 0) {
            sb.append("#".repeat(level + 1)).append(" ");
            sb.append(node.getNodeText()).append("\n");
            if (node.getSummary() != null) {
                sb.append("\n> ").append(node.getSummary()).append("\n");
            }
            sb.append("\n");
        }
        for (MindMapNodeDTO child : node.getChildren()) {
            exportNodeAsMarkdown(child, level + 1, sb);
        }
    }
}
```
