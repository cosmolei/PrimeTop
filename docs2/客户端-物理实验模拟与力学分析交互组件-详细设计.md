# 客户端 - 物理实验模拟与力学分析交互组件 详细设计

## 1. 概述

### 1.1 模块定位

本组件为 PrimeTop 客户端提供面向初高中物理学科的**交互式实验模拟与力学分析**能力，覆盖力学、运动学、电学、光学等核心实验场景。学生可通过可视化拖拽、参数调节实时观察物理过程，理解抽象概念。

### 1.2 设计目标

1. **可视化物理过程**：将抽象的力学、运动学、电学过程转化为可交互的动画模拟
2. **参数可调节**：支持调节质量、速度、角度、摩擦系数等参数，实时观察结果变化
3. **公式联动**：模拟过程同步展示对应公式推导和数值计算
4. **安全实验**：替代危险或需要昂贵器材的实验场景
5. **分龄适配**：初中侧重现象观察，高中侧重定量分析

### 1.3 覆盖实验场景

## 3. 核心物理引擎实现

### 3.1 积分器

采用 **Velocity Verlet** 积分，兼顾精度和稳定性：

```dart
class VelocityVerletIntegrator {
  /// 单步积分
  /// a(t) = F(t) / m
  /// x(t+dt) = x(t) + v(t)*dt + 0.5*a(t)*dt²
  /// a(t+dt) = F(t+dt) / m  (重新计算力)
  /// v(t+dt) = v(t) + 0.5*(a(t) + a(t+dt))*dt
  static void integrate(PhysicsBody body, double dt, Vector2 netForce) {
    final a0 = netForce / body.mass;
    
    // 更新位置
    body.position = body.position 
      + body.velocity * dt 
      + a0 * (0.5 * dt * dt);
    
    // 暂存旧加速度
    final aOld = a0.clone();
    
    // 标记需要重新计算力 (由引擎在下一步处理)
    body.acceleration = aOld; // 临时值，下一步更新
  }
  
  static void integrateVelocity(PhysicsBody body, Vector2 newNetForce, double dt) {
    final aNew = newNetForce / body.mass;
    body.velocity = body.velocity + 0.5 * (body.acceleration + aNew) * dt;
    body.acceleration = aNew;
  }
}
```

### 3.2 碰撞检测与响应

```dart
class CollisionDetector {
  /// 检测两个物体是否碰撞
  static CollisionResult? detect(PhysicsBody a, PhysicsBody b) {
    if (a.type == BodyType.circle && b.type == BodyType.circle) {
      return _circleVsCircle(a, b);
    }
    if (a.type == BodyType.circle && b.type == BodyType.rect) {
      return _circleVsRect(a, b);
    }
    if (a.type == BodyType.rect && b.type == BodyType.rect) {
      return _rectVsRect(a, b);
    }
    // ... 其他组合
    return null;
  }
  
  static CollisionResult? _circleVsCircle(PhysicsBody a, PhysicsBody b) {
    final delta = b.position - a.position;
    final dist = delta.length;
    final minDist = a.radius + b.radius;
    
    if (dist < minDist) {
      final normal = delta.normalized();
      final penetration = minDist - dist;
      return CollisionResult(
        bodyA: a,
        bodyB: b,
        normal: normal,
        penetration: penetration,
        contactPoint: a.position + normal * a.radius,
      );
    }
    return null;
  }
}

class CollisionResolver {
  /// 冲量法碰撞响应
  static void resolve(CollisionResult collision) {
    final a = collision.bodyA;
    final b = collision.bodyB;
    final normal = collision.normal;
    
    // 相对速度
    final relVel = b.velocity - a.velocity;
    final velAlongNormal = relVel.dot(normal);
    
    // 不处理正在分离的物体
    if (velAlongNormal > 0) return;
    
    // 弹性系数取最小
    final e = min(a.restitution, b.restitution);
    
    // 冲量标量
    final j = -(1 + e) * velAlongNormal 
              / (1 / a.mass + 1 / b.mass);
    
    // 施加冲量
    final impulse = normal * j;
    a.velocity = a.velocity - impulse / a.mass;
    b.velocity = b.velocity + impulse / b.mass;
    
    // 位置修正 (防止穿透)
    final correction = normal * (collision.penetration / (a.mass + b.mass)) * 0.8;
    a.position = a.position - correction * b.mass;
    b.position = b.position + correction * a.mass;
  }
}
```

### 3.3 力计算管线

```dart
class ForceAccumulator {
  final List<ForceGenerator> generators = [];
  
  Vector2 computeNetForce(PhysicsBody body, SimulationState state) {
    var netForce = Vector2.zero();
    
    for (final gen in generators) {
      final force = gen.compute(body, state);
      netForce = netForce + force;
    }
    
    return netForce;
  }
}

abstract class ForceGenerator {
  Vector2 compute(PhysicsBody body, SimulationState state);
}

/// 重力
class GravityGenerator extends ForceGenerator {
  final double g; // 重力加速度, 默认 9.8
  
  GravityGenerator({this.g = 9.8});
  
  @override
  Vector2 compute(PhysicsBody body, SimulationState state) {
    return Vector2(0, body.mass * g); // y 轴向下
  }
}

/// 弹簧力 (胡克定律)
class SpringGenerator extends ForceGenerator {
  final PhysicsBody other;
  final double stiffness;
  final double restLength;
  final double damping;
  
  SpringGenerator({
    required this.other,
    required this.stiffness,
    required this.restLength,
    this.damping = 0.1,
  });
  
  @override
  Vector2 compute(PhysicsBody body, SimulationState state) {
    final delta = other.position - body.position;
    final dist = delta.length;
    final direction = delta.normalized();
    
    // 胡克定律: F = -k * (x - x0)
    final springForce = direction * stiffness * (dist - restLength);
    
    // 阻尼力: F_d = -c * v_relative
    final relVel = other.velocity - body.velocity;
    final dampForce = direction * damping * relVel.dot(direction);
    
    return springForce + dampForce;
  }
}

/// 摩擦力
class FrictionGenerator extends ForceGenerator {
  final double mu; // 摩擦系数
  final Vector2 surfaceNormal;
  
  FrictionGenerator({required this.mu, required this.surfaceNormal});
  
  @override
  Vector2 compute(PhysicsBody body, SimulationState state) {
    final speed = body.velocity.length;
    if (speed < 0.001) return Vector2.zero();
    
    final normalForce = body.mass * 9.8; // 简化: 仅支持水平面
    final frictionMag = mu * normalForce;
    final direction = -body.velocity.normalized();
    
    // 静摩擦 vs 动摩擦
    return direction * min(frictionMag, body.mass * speed / 0.016);
  }
}
```

```dart
class InclinedPlaneSimulation implements PhysicsSimulation {
  double angle = 30.0;       // 斜面角度 (°)
  double mass = 2.0;         // 物块质量 (kg)
  double frictionMu = 0.3;   // 摩擦系数
  double initialVelocity = 0; // 初速度 (m/s)
  bool showForceDecomposition = true; // 显示力的分解
  
  // 计算量
  double get gravityComponent => mass * 9.8 * sin(angle * pi / 180);
  double get normalForce => mass * 9.8 * cos(angle * pi / 180);
  double get frictionForce => frictionMu * normalForce;
  double get netForce => gravityComponent - frictionForce;
  double get acceleration => netForce / mass;
}
```

#### 4.1.2 渲染

```dart
class InclinedPlaneRenderer {
  void paint(Canvas canvas, Size size, InclinedPlaneState state) {
    final paint = Paint()..style = PaintingStyle.fill;
    
    // 1. 绘制坐标网格
    _drawGrid(canvas, size);
    
    // 2. 绘制斜面
    final planePath = Path();
    final baseY = size.height * 0.8;
    final planeLength = size.width * 0.7;
    final radians = state.angle * pi / 180;
    
    planePath.moveTo(0, baseY);                                          // 直角点
    planePath.lineTo(planeLength * cos(radians), 
                     baseY - planeLength * sin(radians));                 // 斜面顶点
    planePath.lineTo(planeLength * cos(radians), baseY);                 // 垂足
    planePath.close();
    
    paint.color = const Color(0xFFE8E8E8);
    canvas.drawPath(planePath, paint);
    
    // 3. 绘制物块 (在斜面上)
    final blockX = state.blockPosition * cos(radians);
    final blockY = baseY - state.blockPosition * sin(radians);
    final blockSize = 30.0;
    
    canvas.save();
    canvas.translate(blockX, blockY);
    canvas.rotate(-radians);
    
    paint.color = const Color(0xFF4A90D9);
    canvas.drawRect(
      Rect.fromCenter(center: Offset.zero, width: blockSize, height: blockSize),
      paint,
    );
    
    // 4. 绘制力矢量 (可选)
    if (state.showForceDecomposition) {
      _drawForceArrows(canvas, state, blockSize);
    }
    
    canvas.restore();
    
    // 5. 绘制角度标注
    _drawAngleArc(canvas, Offset.zero, baseY, radians, state.angle);
  }
  
  void _drawForceArrows(Canvas canvas, InclinedPlaneState state, double blockSize) {
    final scale = 3.0; // 像素/牛顿
    
    // 重力 (向下)
    _drawArrow(canvas, Offset.zero, 
      Offset(0, state.mass * 9.8 * scale),
      Colors.red, 'G');
    
    // 重力沿斜面分量
    _drawArrow(canvas, Offset.zero,
      Offset(state.gravityComponent * scale, 0),
      Colors.orange, 'G∥');
    
    // 法向力 (垂直于斜面向外)
    _drawArrow(canvas, Offset.zero,
      Offset(-state.normalForce * scale * sin(0), -state.normalForce * scale),
      Colors.green, 'N');
    
    // 摩擦力 (沿斜面向上)
    _drawArrow(canvas, Offset.zero,
      Offset(-state.frictionForce * scale, 0),
      Colors.purple, 'f');
  }
}
```

#### 4.1.4 公式联动

```
当 angle=30°, mass=2kg, μ=0.3 时:

受力分析:
  G  = mg       = 2 × 9.8 = 19.6 N  ↓
  G∥ = mg sinθ  = 19.6 × sin30° = 9.8 N  (沿斜面向下)
  N  = mg cosθ  = 19.6 × cos30° = 17.0 N  (垂直斜面向外)
  f  = μN       = 0.3 × 17.0 = 5.1 N      (沿斜面向上)
  
  F合 = G∥ - f  = 9.8 - 5.1 = 4.7 N      (沿斜面向下)
  a  = F合/m    = 4.7 / 2 = 2.35 m/s²
```

```dart
class ProjectileSimulation implements PhysicsSimulation {
  double initialSpeed = 20.0;   // 初速度 (m/s)
  double launchAngle = 45.0;    // 发射角度 (°)
  double gravity = 9.8;         // 重力加速度 (m/s²)
  double initialHeight = 0.0;   // 初始高度 (m)
  double airResistance = 0.0;   // 空气阻力系数 (可选)
  
  // 计算量
  double get vx => initialSpeed * cos(launchAngle * pi / 180);
  double get vy => initialSpeed * sin(launchAngle * pi / 180);
  double get timeOfFlight => (vy + sqrt(vy*vy + 2*gravity*initialHeight)) / gravity;
  double get maxRange => vx * timeOfFlight;
  double get maxHeight => initialHeight + vy*vy / (2*gravity);
  
  @override
  void step(double dt) {
    final vx = state.bodies[0].velocity.x;
    final vy = state.bodies[0].velocity.y;
    
    // 无空气阻力: a_x=0, a_y=-g
    state.bodies[0].position.x += vx * dt;
    state.bodies[0].position.y += vy * dt - 0.5 * gravity * dt * dt;
    state.bodies[0].velocity.y -= gravity * dt;
    
    // 记录轨迹
    state.trajectories.add(state.bodies[0].position.clone());
  }
}
```

#### 4.3.2 电路求解 (基尔霍夫定律)

```dart
class CircuitSolver {
  /// 基于改进节点分析法 (MNA) 求解电路
  /// 返回各元件的电压、电流值
  CircuitSolution solve(List<CircuitComponent> components, List<Wire> wires) {
    // 1. 构建拓扑图
    final graph = _buildGraph(components, wires);
    
    // 2. 识别独立回路
    final loops = _findIndependentLoops(graph);
    
    // 3. 建立方程组 (基尔霍夫电压定律 + 电流定律)
    final equations = <List<double>>[];
    final constants = <double>[];
    
    for (final loop in loops) {
      final equation = List.filled(loops.length + 1, 0.0);
      for (final edge in loop.edges) {
        final comp = components.firstWhere((c) => c.id == edge.componentId);
        if (comp.type == ComponentType.battery) {
          constants[loop.index] += comp.emf * (edge.direction ? 1 : -1);
        } else if (comp.type == ComponentType.resistor) {
          equation[loop.index] += comp.resistance;
        }
      }
      equations.add(equation);
    }
    
    // 4. 高斯消元求解
    final currents = _solveLinearSystem(equations, constants);
    
    // 5. 计算各元件的电压、功率
    return _buildSolution(components, currents);
  }
}

class CircuitSolution {
  final Map<String, double> voltages;  // componentId -> 电压 (V)
  final Map<String, double> currents;  // componentId -> 电流 (A)
  final Map<String, double> powers;    // componentId -> 功率 (W)
  final double totalResistance;        // 等效电阻
  final bool isValid;                  // 是否有解 (短路/断路检测)
  final String? errorMessage;          // 错误信息 (如短路)
}
```

class CircuitRenderer {
  void paintComponent(Canvas canvas, CircuitComponent comp) {
    final paint = Paint()
      ..color = Colors.black
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.0;
    
    canvas.save();
    canvas.translate(comp.position.dx, comp.position.dy);
    canvas.rotate(comp.rotation);
    
    switch (comp.type) {
      case ComponentType.resistor:
        _drawResistor(canvas, paint, comp);
      case ComponentType.battery:
        _drawBattery(canvas, paint, comp);
      case ComponentType.switch_:
        _drawSwitch(canvas, paint, comp);
      case ComponentType.bulb:
        _drawBulb(canvas, paint, comp);
      case ComponentType.ammeter:
        _drawMeter(canvas, paint, comp, 'A');
      case ComponentType.voltmeter:
        _drawMeter(canvas, paint, comp, 'V');
      case ComponentType.wire:
        break; // 导线由 WireRenderer 单独绘制
      default:
        break;
    }
    
    canvas.restore();
  }
  
  void _drawResistor(Canvas canvas, Paint paint, CircuitComponent comp) {
    // 锯齿形电阻符号
    final path = Path();
    const w = 30.0, h = 6.0, segments = 6;
    path.moveTo(-w, 0);
    for (int i = 0; i < segments; i++) {
      final x = -w + (i + 0.25) * (2 * w / segments);
      path.lineTo(x, (i.isEven ? -h : h));
    }
    path.lineTo(w, 0);
    canvas.drawPath(path, paint);
    
    // 标注阻值
    _drawLabel(canvas, '${comp.resistance}Ω', const Offset(0, -15));
  }
  
  void _drawBulb(Canvas canvas, Paint paint, CircuitComponent comp) {
    // 灯泡: 圆 + 交叉线
    canvas.drawCircle(Offset.zero, 12, paint);
    canvas.drawLine(const Offset(-8, -8), const Offset(8, 8), paint);
    canvas.drawLine(const Offset(-8, 8), const Offset(8, -8), paint);
    
    // 灯泡通电发光效果
    if (comp.current > 0) {
      final glowPaint = Paint()
        ..color = Colors.yellow.withOpacity(0.6)
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8);
      canvas.drawCircle(Offset.zero, 16, glowPaint);
    }
  }
}
