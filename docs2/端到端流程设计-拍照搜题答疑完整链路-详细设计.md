## 3. 阶段详细设计

### 3.1 P1：端侧采集与预处理

#### 3.1.1 流程

```
用户操作 → 相机取景/相册选择 → 图像裁剪旋转 → 透视校正
         → 自适应增强 → 智能压缩 → 端侧多题检测 → 预览确认
```

#### 3.1.2 端侧预处理Pipeline

```dart
/// 客户端预处理管线
class PhotoPreprocessPipeline {
  final ImageEnhancer _enhancer;
  final TextDetector _textDetector; // 端侧 DBNet-Lite

  Future<PreprocessResult> process(Uint8List rawImage) async {
    // Step 1: 透视校正 (检测文档边缘 + 仿射变换)
    final corrected = await _enhancer.perspectiveCorrect(rawImage);

    // Step 2: 自适应增强 (对比度+锐化+去噪)
    final enhanced = await _enhancer.adaptiveEnhance(corrected);

    // Step 3: 智能压缩 (目标 ≤2MB, 保留文字清晰度)
    final compressed = await _enhancer.smartCompress(
      enhanced,
      maxSizeBytes: 2 * 1024 * 1024,
      qualityTarget: 0.85,
    );

    // Step 4: 端侧文字区域检测
    final textRegions = await _textDetector.detect(compressed);

    // Step 5: 多题分割
    final questionBoxes = _splitQuestions(textRegions);

    return PreprocessResult(
      imageBytes: compressed,
      textRegions: textRegions,
      questionBoxes: questionBoxes,
      questionCount: questionBoxes.length,
    );
  }

  /// 启发式题目分割
  List<Rect> _splitQuestions(List<TextRegion> regions) {
    if (regions.isEmpty) return [];
    // 规则1: 检测题号模式 (1. / (1) / 一、等)
    // 规则2: 垂直间距超阈值则分割
    // 规则3: 版面列数检测 (单栏/双栏)
    // 至少返回整图作为一个框
    final splits = <Rect>[];
    // ... 分割逻辑
    return splits.isNotEmpty ? splits : [Rect.largest];
  }
}
```

#### 3.2.2 服务端处理流程

```python
# search_orchestrator/service.py

class PhotoSearchOrchestrator:
    """拍照搜题编排服务"""

    def __init__(
        self,
        ocr_service: OCRService,
        question_parser: QuestionParser,
        question_repo: QuestionRepository,
        ai_service: AIService,
        quota_service: QuotaService,
        event_bus: EventBus,
    ):
        self._ocr = ocr_service
        self._parser = question_parser
        self._repo = question_repo
        self._ai = ai_service
        self._quota = quota_service
        self._events = event_bus

    async def upload_and_parse(
        self,
        user_id: int,
        image_bytes: bytes,
        crop_rect: Optional[dict],
        question_index: Optional[int],
        user_context: UserContext,  # grade, subject, textbook etc.
    ) -> PhotoSearchResult:
        """P2主流程: 上传+OCR+结构化"""

        # Step 1: 额度预检 (原子扣减)
        await self._quota.check_and_deduct(
            user_id=user_id,
            quota_type=QuotaType.PHOTO_SEARCH,
            amount=1,
        )

        # Step 2: 图片上传到OSS
        oss_url = await self._upload_to_oss(image_bytes, user_id)

        # Step 3: OCR识别 (带降级链路)
        ocr_result = await self._ocr.recognize_with_fallback(
            image_url=oss_url,
            crop_rect=crop_rect,
            timeout_ms=3000,
        )

        # Step 4: 题目结构化解析
        parsed_questions = await self._parser.parse_batch(
            ocr_texts=[r.text for r in ocr_result.regions],
            image_context={
                "grade": user_context.grade,
                "subject": user_context.subject,
            },
        )

        # Step 5: 题库匹配 (并行)
        match_tasks = [
            self._repo.find_similar(
                text=pq.structured_text,
                subject=user_context.subject,
                grade=user_context.grade,
                top_k=3,
            )
            for pq in parsed_questions
        ]
        match_results = await asyncio.gather(*match_tasks, return_exceptions=True)

        # Step 6: 组装结果
        request_id = generate_request_id("ps")
        questions = []
        for i, (pq, mr) in enumerate(zip(parsed_questions, match_results)):
            if isinstance(mr, Exception):
                mr = []

            questions.append(PhotoSearchQuestion(
                index=i,
                ocr_text=pq.raw_text,
                confidence=pq.confidence,
                structured=pq,
                cached_answer=mr[0] if mr else None,
                answer_source="cache" if mr else "ai",
            ))

        # Step 7: 缓存搜索请求 (供后续SSE查询)
        await self._cache_search_request(request_id, questions, user_context)

        # Step 8: 发布事件 (异步, 不阻塞)
        self._events.publish(PhotoUploadedEvent(
            user_id=user_id,
            request_id=request_id,
            question_count=len(questions),
            ocr_confidence=min(q.confidence for q in questions),
        ))

        return PhotoSearchResult(
            request_id=request_id,
            questions=questions,
            has_cache_hit=any(q.cached_answer for q in questions),
        )
```

#### 3.2.3 OCR降级链路

```
端侧OCR (DBNet-Lite + CRNN)
    │  confidence ≥ 0.85 且 coverage ≥ 80%
    │  → 直接使用
    ▼
云端主OCR (供应商A, 如百度/腾讯)
    │  confidence ≥ 0.90
    │  → 使用, 并用结构化解析校正
    ▼
云端备选OCR (供应商B)
    │  → 使用, 降级提示用户"识别可能不准确"
    ▼
手动输入 (降级UI) → 提供文本框让用户手动输入题目
```

#### 3.3.2 题库匹配策略

```python
async def find_similar(
    text: str,
    subject: Subject,
    grade: Grade,
    top_k: int = 3,
) -> List[QuestionMatch]:
    """三级匹配策略"""

    # Level 1: 精确匹配 (SHA-256 指纹)
    exact = await self._exact_match(text, subject)
    if exact and exact.similarity >= 0.98:
        return [exact]  # 高置信度精确匹配, 直接返回

    # Level 2: 向量语义检索 (Milvus)
    vector_hits = await self._vector_search(
        text=text,
        subject=subject,
        grade=grade,
        top_k=top_k * 2,  # 多召回, 后过滤
        threshold=0.85,
    )

    # Level 3: ES全文检索 (BM25)
    es_hits = await self._es_search(
        text=text,
        subject=subject,
        grade=grade,
        top_k=top_k * 2,
    )

    # RRF融合排序
    merged = self._rrf_merge(vector_hits, es_hits, weights=(0.6, 0.4))
    return merged[:top_k]
```

#### 3.3.3 匹配结果判定

| 相似度 | 处理策略 | 用户体验 |
|--------|----------|----------|

| 0.80-0.95 | 复用解析, 差异部分AI补充 | 快速出答案, 标注"类似题解析" |

#### 3.4.2 服务端SSE生成器

```python
class PhotoSearchAnswerService:
    """搜题答案生成与流式推送"""

    async def stream_answer(
        self,
        request_id: str,
        question_index: int,
        user_context: UserContext,
    ) -> AsyncIterator[SSEEvent]:
        """生成SSE流"""

        # 1. 恢复搜索请求上下文
        search_ctx = await self._get_cached_request(request_id)
        if not search_ctx:
            yield SSEEvent(event="error", data={"code": "REQUEST_EXPIRED"})
            return

        question = search_ctx.questions[question_index]

        # 2. 如果有缓存命中, 直接返回缓存答案
        if question.cached_answer:
            yield SSEEvent(event="meta", data={"source": "cache"})
            async for chunk in self._stream_cached(question.cached_answer):
                yield chunk
            yield SSEEvent(event="done", data={"source": "cache"})
            return

        # 3. AI全新解题
        yield SSEEvent(event="meta", data={
            "source": "ai",
            "question_type": question.structured.question_type.value,
        })

        # 3a. 构建Prompt
        prompt = await self._build_solve_prompt(
            question=question.structured,
            user_context=user_context,
        )

        # 3b. 调用AI (流式)
        full_response = StringBuilder()
        async for token in self._ai.stream_chat(
            messages=prompt.messages,
            model=prompt.model,
            temperature=0.3,  # 解题场景低温度
            max_tokens=2000,
        ):
            full_response.append(token)
            yield SSEEvent(event="delta", data={
                "content": token,
                "type": "text",
            })

        # 4. 后处理
        response_text = full_response.to_string()
        processed = await self._post_process(response_text, question)

        # 5. 答案管控分级
        tiers = self._answer_tier_service.split_into_tiers(
            full_text=processed.clean_text,
            question_type=question.structured.question_type,
            user_grade=user_context.grade,
        )
        for tier in tiers:
            yield SSEEvent(event=f"tier_{tier.level}", data=tier.to_dict())

        # 6. 知识点标注
        kps = await self._knowledge_service.extract_points(
            text=processed.clean_text,
            subject=user_context.subject,
        )
        yield SSEEvent(event="knowledge_points", data={
            "points": [kp.to_dict() for kp in kps],
        })

        # 7. 同类题推荐
        related = await self._question_repo.find_similar(
            text=question.structured.structured_text,
            subject=user_context.subject,
            grade=user_context.grade,
            top_k=3,
        )
        yield SSEEvent(event="related_questions", data={
            "questions": [{"id": r.id, "preview": r.preview} for r in related],
        })

        # 8. 异步保存 (不阻塞SSE流)
        asyncio.create_task(self._async_save_answer(
            request_id=request_id,
            question_index=question_index,
            answer_text=processed.clean_text,
            knowledge_points=kps,
            tokens_used=processed.token_count,
        ))

        yield SSEEvent(event="done", data={
            "source": "ai",
            "total_tokens": processed.token_count,
        })
```

```python
    self,
    question: ParsedQuestion,
    user_context: UserContext,

        "question_text": question.structured_text,
        "question_type": question.question_type.value,
        "options": question.options,
        "formulas": question.formulas,
        "has_image": question.has_image,
        "rag_context": rag_context,
    })
```

```python
def _select_model(self, question: ParsedQuestion) -> str:
    """根据题目特征选择最优模型"""

    # 理科复杂题 → 推理增强模型
    if question.subject in (Subject.MATH, Subject.PHYSICS, Subject.CHEMISTRY):
        if question.question_type in (QuestionType.PROOF, QuestionType.CALCULATION):
            if question.has_formula:
                return "reasoning_model"   # e.g. o3, deepseek-r1

    # 含图题 → 多模态模型
    if question.has_image:
        return "vision_model"              # e.g. gpt-4o, claude-3.5

    # 默认 → 通用问答模型 (成本优化)
    return "general_model"                 # e.g. glm-4, qwen-plus
```

```dart
/// 客户端搜题答案流式渲染器
class PhotoSearchAnswerStream {
  final Dio _dio;
  final AnswerTierController _tierController;
  final SubjectRenderer _renderer; // 富文本学科渲染器

  Stream<AnswerChunk> watchAnswer(String requestId, int questionIndex) {
    return _dio
        .getStream('/search/answer/$requestId?q=$questionIndex')
        .map(_parseSSE)
        .transform(_renderPipeline);
  }

  /// 渲染管线
  StreamTransformer< SSEEvent, AnswerChunk> get _renderPipeline {
    return StreamTransformer.fromHandlers(
      handleData: (event, sink) {
        switch (event.event) {
          case 'meta':
            sink.add(AnswerChunk.meta(source: event.data['source']));
          case 'delta':
            // 流式增量渲染 (Markdown + LaTeX)
            final rendered = _renderer.appendDelta(event.data['content']);
            sink.add(AnswerChunk.delta(rendered));
          case 'tier_hint':
            _tierController.showTier(TierLevel.hint, event.data['content']);
          case 'tier_analysis':
            _tierController.showTier(TierLevel.analysis, event.data['content']);
          case 'tier_steps':
            _tierController.showTier(TierLevel.steps, event.data['content']);
          case 'tier_answer':
            _tierController.lockTier(
              TierLevel.answer,
              event.data['content'],
              unlockCondition: event.data['unlock_condition'],
            );
          case 'knowledge_points':
            sink.add(AnswerChunk.knowledgePoints(
              (event.data['points'] as List).cast<Map>(),
            ));
          case 'related_questions':
            sink.add(AnswerChunk.relatedQuestions(
              (event.data['questions'] as List).cast<Map>(),
            ));
          case 'done':
            sink.add(AnswerChunk.done(event.data));
          case 'error':
            sink.add(AnswerChunk.error(event.data['code']));
        }
      },
    );
  }
}
```

```python
class PhotoSearchEventConsumer:
    """拍照搜题完成后置处理"""

    @subscribe("photo_search.completed")
    async def on_search_completed(self, event: PhotoSearchCompletedEvent):
        """搜题完成事件处理"""

        # 并行执行不互相依赖的任务
        await asyncio.gather(
            self._update_mastery(event),
            self._record_answer(event),
            self._track_cost(event),
            self._emit_analytics(event),
            return_exceptions=True,  # 单个失败不影响其他
        )

        # 新题入库 (依赖answer已保存)
        if event.is_new_question:
            await self._ingest_new_question(event)

    async def _update_mastery(self, event: PhotoSearchCompletedEvent):
        """更新知识点掌握度"""
        # 从AI回答中提取知识点
        knowledge_points = await self._kp_extractor.extract(
            event.question_text,
            event.answer_text,
            event.subject,
        )

        for kp in knowledge_points:
            # 证据强度 = OCR置信度 × 题目难度权重 × 用户是否看了完整答案
            evidence_strength = (
                event.ocr_confidence
                * event.difficulty_weight
                * (1.0 if event.fully_viewed else 0.5)
            )

            await self._mastery_service.update(
                user_id=event.user_id,
                knowledge_point_id=kp.id,
                evidence_type="photo_search",
                evidence_strength=evidence_strength,
            )

    async def _record_answer(self, event: PhotoSearchCompletedEvent):
        """记录答题"""
        await self._answer_repo.create(AnswerRecord(
            user_id=event.user_id,
            request_id=event.request_id,
            question_snapshot=event.question_text,
            answer_snapshot=event.answer_text,
            source="photo_search",
            knowledge_points=event.knowledge_point_ids,
            ai_model=event.ai_model,
            tokens_used=event.tokens_used,
            duration_ms=event.duration_ms,
            ocr_confidence=event.ocr_confidence,
            created_at=datetime.utcnow(),
        ))

        # 更新章节进度
        if event.chapter_id:
            await self._progress_service.touch_chapter(
                user_id=event.user_id,
                chapter_id=event.chapter_id,
                action="photo_search",
            )
```

## 12. 总结

本文档描述了拍照搜题答疑功能的完整端到端链路，覆盖 6 个处理阶段：

1. **P1 端侧采集与预处理**：相机取景→裁剪→增强→压缩→多题检测
2. **P2 图片上传与OCR识别**：额度预检→上传OSS→OCR识别(含降级链路)→结构化解析
3. **P3 题库匹配**：精确指纹→向量语义→ES全文→RRF融合
4. **P4 AI解题**：Prompt构建(RAG增强)→模型路由→SSE流式→后处理→答案管控分级
5. **P5 答案渲染与交互**：SSE消费→增量渲染→用户操作(错题/同类/追问/反馈)
6. **P6 异步后置处理**：掌握度更新→学习记录→题库质量→额度确认→数据埋点

关键设计要点：
- **多级降级**：每个阶段都有明确的降级策略，确保用户始终能得到某种形式的答案
- **幂等保障**：上传、SSE、错题收录均支持幂等，防止重复操作
- **异步解耦**：用户感知的操作(S1-S5)同步完成，后置处理(P6)异步执行
- **SSE流式**：AI回答采用SSE流式推送，确保用户尽快看到内容
- **成本优化**：题库命中直接复用，AI模型按题目特征智能路由
