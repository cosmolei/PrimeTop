## 1. 功能概述

### 1.1 模块目标

家长中心客户端模块为家长用户提供统一的孩子学习管理入口，核心能力包括：

1. **多孩子管理**：家长可绑定多个孩子，快速切换查看不同孩子的学情
2. **学情报告查看**：查看孩子的周报/月报、学习时长、薄弱点分析
3. **使用时长管控**：设置每日使用时长上限、禁用时段
4. **内容范围管控**：控制孩子可访问的学科和功能
5. **学习提醒配置**：设置学习提醒推送频率和时间段
6. **家长验证**：敏感操作需验证家长身份（PIN/生物识别）
7. **会员与订阅管理**：家庭套餐管理、续费操作

### 1.2 用户角色

| 角色 | 入口方式 | 权限范围 |
|------|---------|---------|
| 家长（已绑定孩子） | APP 内"家长中心"入口 / 家长角色登录 | 查看已绑定孩子的学情、设置管控 |
| 家长（未绑定） | APP 内"家长中心"入口 | 仅可看到"绑定引导"页面 |
| 学生（低龄） | 个人中心→家长中心入口 | 查看家长绑定状态（只读） |
| 学生（高中） | 无入口 | 家长中心入口隐藏 |

```dart
/// 禁用时段
class ForbiddenPeriod {
  final String id;
  final int startHour;        // 0-23
  final int startMinute;      // 0-59
  final int endHour;
  final int endMinute;
  final List<int> weekDays;   // 1-7（1=周一，7=周日），空=每天

  const ForbiddenPeriod({
    required this.id,
    required this.startHour,
    this.startMinute = 0,
    required this.endHour,
    this.endMinute = 0,
    this.weekDays = const [],
  });

  /// 判断当前时间是否在禁用时段内
  bool isCurrentlyForbidden() {
    final now = DateTime.now();
    final currentMinutes = now.hour * 60 + now.minute;
    final startMinutes = startHour * 60 + startMinute;
    final endMinutes = endHour * 60 + endMinute;
    final currentWeekDay = now.weekday; // 1=Mon, 7=Sun

    // 检查星期
    if (weekDays.isNotEmpty && !weekDays.contains(currentWeekDay)) {
      return false;
    }

    if (startMinutes <= endMinutes) {
      return currentMinutes >= startMinutes && currentMinutes < endMinutes;
    } else {
      // 跨午夜
      return currentMinutes >= startMinutes || currentMinutes < endMinutes;
    }
  }
}
```

```dart
// lib/features/parent_center/providers/parent_providers.dart

/// 家长中心全局状态
@riverpod
class ParentCenterState extends _$ParentCenterState {
  @override
  AsyncValue<ParentCenterData> build() {
    // 首次加载：并行获取绑定列表 + 全局概览
    _loadInitialData();
    return const AsyncLoading();
  }

  Future<void> _loadInitialData() async {
    try {
      final api = ref.read(parentApiServiceProvider);
      final bindings = await api.getBoundChildren();

      if (bindings.isEmpty) {
        state = AsyncData(ParentCenterData(
          bindings: [],
          selectedChildIndex: -1,
          hasChildren: false,
        ));
        return;
      }

      // 并行获取所有孩子概览
      final overviews = await api.getChildrenOverviewBatch();
      state = AsyncData(ParentCenterData(
        bindings: bindings,
        childOverviews: {for (var o in overviews) o.childUserId: o},
        selectedChildIndex: 0,
        hasChildren: true,
      ));
    } catch (e, st) {
      state = AsyncError(e, st);
    }
  }

  /// 切换选中的孩子
  void selectChild(int index) {
    state.whenData((data) {
      if (index >= 0 && index < data.bindings.length) {
        state = AsyncData(data.copyWith(selectedChildIndex: index));
      }
    });
  }

  /// 刷新某个孩子的概览数据
  Future<void> refreshChildOverview(String childUserId) async {
    state.whenData((data) async {
      try {
        final api = ref.read(parentApiServiceProvider);
        final overview = await api.getChildStudyOverview(
          childUserId: childUserId,
        );
        final newOverviews = Map<String, ChildStudyOverview>.from(
          data.childOverviews,
        )..[childUserId] = overview;
        state = AsyncData(data.copyWith(childOverviews: newOverviews));
      } catch (_) {
        // 静默失败，保留旧数据
      }
    });
  }

  /// 绑定新孩子成功后刷新
  Future<void> onChildBound(ParentBinding newBinding) async {
    final api = ref.read(parentApiServiceProvider);
    final overview = await api.getChildStudyOverview(
      childUserId: newBinding.childUserId,
    );
    state.whenData((data) {
      final newBindings = [...data.bindings, newBinding];
      final newOverviews = Map<String, ChildStudyOverview>.from(
        data.childOverviews,
      )..[newBinding.childUserId] = overview;
      state = AsyncData(data.copyWith(
        bindings: newBindings,
        childOverviews: newOverviews,
        selectedChildIndex: newBindings.length - 1,
        hasChildren: true,
      ));
    });
  }

  /// 解绑后刷新
  Future<void> onChildUnbound(String bindingId) async {
    state.whenData((data) {
      final idx = data.bindings.indexWhere((b) => b.bindingId == bindingId);
      if (idx < 0) return;
      final removed = data.bindings[idx];
      final newBindings = [...data.bindings]..removeAt(idx);
      final newOverviews = Map<String, ChildStudyOverview>.from(
        data.childOverviews,
      )..remove(removed.childUserId);
      state = AsyncData(data.copyWith(
        bindings: newBindings,
        childOverviews: newOverviews,
        selectedChildIndex: newBindings.isEmpty
            ? -1
            : (data.selectedChildIndex >= newBindings.length
                ? newBindings.length - 1
                : data.selectedChildIndex),
        hasChildren: newBindings.isNotEmpty,
      ));
    });
  }
}
```

```dart
/// 家长验证守卫
/// 敏感操作前弹出验证，获取 verifyToken
class ParentVerifyGuard {
  final ParentApiService _api;
  final BiometricService _biometric;

  ParentVerifyGuard(this._api, this._biometric);

  /// 缓存的验证 token 及过期时间
  static String? _cachedToken;
  static DateTime? _tokenExpiry;

  /// 检查是否有有效的验证 token
  bool get hasValidToken =>
      _cachedToken != null &&
      _tokenExpiry != null &&
      DateTime.now().isBefore(_tokenExpiry!);

  /// 获取有效 token（缓存未过期则直接返回）
  String? get validToken =>
      hasValidToken ? _cachedToken : null;

  /// 执行家长验证
  /// 返回 verifyToken，失败返回 null
  Future<String?> verify(BuildContext context) async {
    // 1. 检查缓存
    if (hasValidToken) return _cachedToken;

    // 2. 尝试生物识别
    if (await _biometric.isAvailable()) {
      final bioResult = await _biometric.authenticate(
        reason: '请验证家长身份以进行敏感操作',
      );
      if (bioResult) {
        // 生物识别成功后，通过服务端验证获取 token
        final result = await _api.verifyPin(pin: '__biometric_bypass__');
        if (result.success) {
          _cacheToken(result.token!);
          return result.token;
        }
      }
    }

    // 3. 回退到 PIN 码验证
    final token = await _showPinVerifyDialog(context);
    return token;
  }

  /// 显示 PIN 码验证弹窗
  Future<String?> _showPinVerifyDialog(BuildContext context) async {
    return showDialog<String>(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => const ParentPinVerifyDialog(),
    );
  }

  void _cacheToken(String token) {
    _cachedToken = token;
    _tokenExpiry = DateTime.now().add(const Duration(minutes: 5));
  }

  /// 清除缓存（退出家长中心时调用）
  void clearCache() {
    _cachedToken = null;
    _tokenExpiry = null;
  }
}
```

```dart
  bool _isExpired = false;

  @override
  void initState() {
    super.initState();
    // 每 5 秒轮询绑定状态
    _pollTimer = Timer.periodic(
      const Duration(seconds: 5),
      (_) => _checkBindStatus(),
    );

  @override

    _pollTimer?.cancel();
    super.dispose();
```

```dart
  const ParentPinVerifyDialog({super.key});

  @override
  ConsumerState<ParentPinVerifyDialog> createState() =>
      _ParentPinVerifyDialogState();
}

class _ParentPinVerifyDialogState
    extends ConsumerState<ParentPinVerifyDialog> {
  final _pinController = TextEditingController();
  bool _isVerifying = false;
  String? _errorMessage;
  int _remainingAttempts = 5;

  @override
  void dispose() {
    _pinController.dispose();
    super.dispose();
  }

  Future<void> _onPinComplete(String pin) async {
    if (_isVerifying) return;
    setState(() {
      _isVerifying = true;
      _errorMessage = null;
    });

    try {
      final api = ref.read(parentApiServiceProvider);
      final result = await api.verifyPin(pin: pin);

      if (result.success) {
        Navigator.of(context).pop(result.token);
      } else {
        setState(() {
          _errorMessage = result.error ?? '验证码错误';
          _remainingAttempts = result.remainingAttempts ?? 0;
          _isVerifying = false;
        });
        _pinController.clear();

        if (_remainingAttempts <= 0) {
          // 连续错误 5 次，锁定 15 分钟
          Navigator.of(context).pop(null);
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('验证码错误次数过多，请 15 分钟后再试'),
              duration: Duration(seconds: 5),
            ),
          );
        }
      }
    } catch (e) {
      setState(() {
        _errorMessage = '网络错误，请重试';
        _isVerifying = false;
      });
    }
  }

  Future<void> _tryBiometric() async {
    final biometric = ref.read(biometricServiceProvider);
    if (!await biometric.isAvailable()) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('设备不支持生物识别')),
      );
      return;
    }

    final result = await biometric.authenticate(
      reason: '请验证家长身份以进行敏感操作',
    );

    if (result) {
      // 生物识别成功，获取服务端 token
      final api = ref.read(parentApiServiceProvider);
      final verifyResult = await api.verifyPin(pin: '__biometric__');
      if (verifyResult.success) {
        Navigator.of(context).pop(verifyResult.token);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Dialog(
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
      child: Padding(
        padding: const EdgeInsets.all(24),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            const Text(
              '家长身份验证',
              style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 8),
            const Text(
              '为保护孩子安全，请验证您的身份',
              style: TextStyle(color: Colors.grey, fontSize: 14),
            ),
            const SizedBox(height: 24),
            // PIN 码输入框
            PinInputField(
              controller: _pinController,
              length: 6,
              obscureText: true,
              onComplete: _onPinComplete,
              errorText: _errorMessage,
            ),
            if (_remainingAttempts < 5) ...[
              const SizedBox(height: 8),
              Text(
                '剩余尝试次数: $_remainingAttempts 次',
                style: TextStyle(
                  color: _remainingAttempts <= 2 ? Colors.red : Colors.grey,
                  fontSize: 12,
                ),
              ),
            ],
            const SizedBox(height: 16),
            // 数字键盘

                  _pinController.text += d;
                  if (_pinController.text.length == 6) {

                  icon: const Icon(Icons.fingerprint),
                  label: const Text('使用指纹/面部识别'),
                );
              },
            ),
          ],
        ),
```

```dart
              _RestReminderSection(

              ),
```

```dart
// lib/features/parent_center/utils/parent_error_handler.dart

class ParentErrorHandler {
  /// 处理家长中心 API 错误，返回用户可见的提示
  static String getUserMessage(dynamic error) {
    if (error is ApiException) {
      switch (error.code) {
        case 110101:
          return '请先绑定学生账号';
        case 110102:
          return '邀请码已过期，请重新生成';
        case 110103:
          return '该邀请码已被使用';
        case 110104:
          return '该学生已绑定其他家长';
        case 110105:
          return '需要家长验证才能操作';
        case 110201:
          return '时长设置超出有效范围';
        case 110202:
          return '禁用时段不能重叠';
        case 110301:
          return '请先设置家长验证码';
        case 110302:
          return '验证码错误次数过多，请稍后再试';
        case 110401:
          return '暂无学情报告数据';
        case 110501:
          return '该功能需要升级家庭套餐';
        default:
          return '操作失败，请稍后重试';
      }
    }
    if (error is SocketException || error is TimeoutException) {
      return '网络连接失败，请检查网络后重试';
    }
    return '操作失败，请稍后重试';
  }

  /// 判断是否需要弹出家长验证
  static bool needsParentVerify(dynamic error) {
    if (error is ApiException) {
      return error.code == 110105;
    }
    return false;
  }

  /// 判断是否需要引导绑定
  static bool needsBinding(dynamic error) {
    if (error is ApiException) {
      return error.code == 110101;
    }
    return false;
  }
}
```

## 8. 分龄 UI 适配

### 8.1 家长中心分龄策略

家长中心本身的用户是家长，但被管控的孩子分布在不同学段。UI 需要根据孩子学段展示不同的管控选项：

| 管控项 | 幼儿 (3-6) | 小学 (6-12) | 初中 (12-15) | 高中 (15-18) |

```dart
// 仪表盘采用分步加载，优先展示骨架屏
//
// Step 1 (0ms):   显示骨架屏
// Step 2 (<1s):   加载绑定列表（本地缓存优先）
// Step 3 (<2s):   并行加载所有孩子概览
// Step 4 (<3s):   加载最新周报摘要
// Step 5 (懒加载): 管控配置（点击时才加载）

Future<ParentDashboardData> loadDashboard() async {
  // Step 2: 绑定列表
  final bindings = await _loadBindingsWithCache();
  if (bindings.isEmpty) {
    return ParentDashboardData.empty();
  }

  // Step 3: 并行加载概览
  final overviewFutures = bindings.map(
    (b) => _api.getChildStudyOverview(childUserId: b.childUserId)
        .timeout(const Duration(seconds: 3))
        .catchError((_) => ChildStudyOverview(childUserId: b.childUserId, childNickname: b.childNickname)),
  );
  final overviews = await Future.wait(overviewFutures);

  // Step 4: 最新周报（非阻塞，失败不影响主流程）
  StudyReportSummary? latestReport;
  try {
    final reports = await _api.getReportList(
      childUserId: bindings.first.childUserId,
      reportType: 'weekly',
      pageSize: 1,
    ).timeout(const Duration(seconds: 2));
    latestReport = reports.items.firstOrNull;
  } catch (_) {}

  return ParentDashboardData(
    bindings: bindings,
    overviews: {for (var o in overviews) o.childUserId: o},
    latestReport: latestReport,
  );
}
```

## 10. 安全设计

### 10.1 安全要点

1. **家长验证 token 不持久化**：验证 token 仅存内存，退出页面即清除
2. **PIN 码不明文传输**：客户端对 PIN 做 SHA256 摘要后传输
3. **管控配置变更审计**：每次配置变更均通过服务端审计日志记录
4. **防重放攻击**：verifyToken 含时间戳和 nonce，服务端校验
5. **绑定操作二次确认**：解绑需要双方确认，防止误操作
6. **数据隔离**：家长只能查看已绑定孩子的数据，API 层强制校验

### 10.2 客户端安全检查

```dart
// lib/features/parent_center/security/parent_security_checks.dart

class ParentSecurityChecks {
  /// 进入家长中心前的安全检查
  static Future<SecurityCheckResult> preEnterCheck({
    required String userRole,
    required String? ageGroup,
  }) async {
    // 1. 角色检查
    if (userRole != 'parent' && userRole != 'student') {
      return SecurityCheckResult.deny('仅家长和学生可访问家长中心');
    }

    // 2. 学生端只显示绑定状态，高中不显示入口
    if (userRole == 'student') {
      if (ageGroup == 'senior') {
        return SecurityCheckResult.hide();
      }
      return SecurityCheckResult.allow(studentView: true);
    }

    return SecurityCheckResult.allow(studentView: false);
  }

  /// 敏感操作前的安全检查
  static Future<bool> preActionCheck({
    required BuildContext context,
    required String action,
  }) async {
    // 检查是否需要家长验证
    final sensitiveActions = {
      'update_time_control',
      'update_content_control',
      'unbind_child',
      'set_pin',
      'purchase',
    };

    if (sensitiveActions.contains(action)) {
      // 检查锁定状态
      final isLocked = await _checkVerifyLock();
      if (isLocked) {
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('验证码错误次数过多，请稍后再试'),
              duration: Duration(minutes: 1),
            ),
          );
        }
        return false;
      }
      return true;
    }
    return true;
  }

  static Future<bool> _checkVerifyLock() async {
    // 检查本地缓存的锁定状态
    final prefs = await SharedPreferences.getInstance();
    final lockExpiry = prefs.getInt('parent_verify_lock_expiry');
    if (lockExpiry == null) return false;
    return DateTime.now().millisecondsSinceEpoch < lockExpiry;
  }
}

class SecurityCheckResult {
  final bool allow;
  final bool hidden;
  final bool studentView;
  final String? denyReason;

  const SecurityCheckResult._({
    this.allow = false,
    this.hidden = false,
    this.studentView = false,
    this.denyReason,
  });

  factory SecurityCheckResult.allow({bool studentView = false}) =>
      SecurityCheckResult._(allow: true, studentView: studentView);

  factory SecurityCheckResult.deny(String reason) =>
      SecurityCheckResult._(denyReason: reason);

  factory SecurityCheckResult.hide() =>
      const SecurityCheckResult._(hidden: true);
}
```
