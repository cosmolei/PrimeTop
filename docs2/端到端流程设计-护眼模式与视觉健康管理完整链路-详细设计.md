## 1. 链路概述

### 1.1 链路定位

本端到端链路覆盖"护眼模式与视觉健康管理"功能的完整用户旅程，从家长端管控设置、学生端自动启用、学习过程中实时监测、到异常用眼干预和数据报告生成的全链路闭环。

与模块文档的分工：
- **模块文档**（护眼模式与视觉健康管理系统）定义了功能架构、数据结构和技术实现
- **本链路文档**定义跨角色、跨系统的完整业务流程，供开发人员理解"一个护眼事件如何从触发到闭环"

### 1.2 涉及角色

| 角色 | 场景 |
|------|------|
| 家长 | 在家长中心配置护眼策略、查看用眼报告、远程调整设置 |
| 学生（幼儿/小学） | 被动接受强制护眼策略、看到休息提醒、完成眼保健操 |
| 学生（初中/高中） | 可自主微调护眼参数（在家长允许范围内）、查看个人护眼数据 |
| 系统 | 自动检测环境光/用眼距离、执行强制休息、生成护眼报告 |

### 2.3 策略校验与边界约束

家长调整配置时，服务端必须校验不超出安全边界。

```java
@Service
public class EyeHealthConfigValidator {

    // 各学段安全边界
    private static final Map<GradeLevel, SafetyBoundary> BOUNDARIES = Map.of(
        GradeLevel.KINDERGARTEN, new SafetyBoundary(
            EyeHealthLevel.L3, EyeHealthLevel.L4,
            10, 20,   // restInterval: 10-20 min
            3, 10,    // forcedBreak: 3-10 min
            30, 50    // minDistance: 30-50 cm
        ),
        GradeLevel.PRIMARY, new SafetyBoundary(
            EyeHealthLevel.L1, EyeHealthLevel.L3,
            15, 30,
            3, 8,
            25, 40
        ),
        GradeLevel.JUNIOR, new SafetyBoundary(
            EyeHealthLevel.L1, EyeHealthLevel.L2,
            25, 45,
            0, 5,
            0, 0     // 不强制距离检测
        ),
        GradeLevel.SENIOR, new SafetyBoundary(
            EyeHealthLevel.L0, EyeHealthLevel.L2,
            30, 60,
            0, 0,    // 不强制休息
            0, 0
        )
    );

    public ValidationResult validate(GradeLevel gradeLevel, EyeHealthConfig config) {
        SafetyBoundary boundary = BOUNDARIES.get(gradeLevel);
        List<String> violations = new ArrayList<>();

        // 校验护眼等级范围
        if (config.getLevel().ordinal() < boundary.minLevel.ordinal()
            || config.getLevel().ordinal() > boundary.maxLevel.ordinal()) {
            violations.add("护眼等级 " + config.getLevel() + " 超出允许范围 ["
                + boundary.minLevel + ", " + boundary.maxLevel + "]");
        }

        // 校验休息间隔
        if (config.getRestReminder().getIntervalMinutes() < boundary.minRestInterval
            || config.getRestReminder().getIntervalMinutes() > boundary.maxRestInterval) {
            violations.add("休息间隔 " + config.getRestReminder().getIntervalMinutes()
                + " 分钟超出允许范围 [" + boundary.minRestInterval
                + ", " + boundary.maxRestInterval + "]");
        }

        // 校验强制休息时长
        if (config.getRestReminder().getForcedBreakMinutes() < boundary.minForcedBreak
            || config.getRestReminder().getForcedBreakMinutes() > boundary.maxForcedBreak) {
            violations.add("强制休息时长超出允许范围");
        }

        // 幼儿/小学不可关闭护眼模式
        if ((gradeLevel == GradeLevel.KINDERGARTEN || gradeLevel == GradeLevel.PRIMARY)
            && !config.getBlueLightFilter().isEnabled()) {
            violations.add("该学段不允许关闭蓝光过滤");
        }

        if (!violations.isEmpty()) {
            return ValidationResult.rejected(violations);
        }
        return ValidationResult.accepted();
    }
}
```

### 3.2 客户端护眼引擎初始化

```dart
/// 客户端护眼引擎核心类
class EyeHealthEngine {
  static final EyeHealthEngine _instance = EyeHealthEngine._internal();
  factory EyeHealthEngine() => _instance;
  EyeHealthEngine._internal();

  EyeHealthConfig? _config;
  Timer? _restTimer;
  Timer? _distanceCheckTimer;
  Timer? _envLightTimer;
  DateTime? _sessionStartTime;
  final _eventBus = EventBus();

  /// 初始化引擎
  Future<void> initialize(String studentId) async {
    // 1. 加载配置
    _config = await _loadConfig(studentId);

    // 2. 应用护眼滤镜
    await _applyFilter();

    // 3. 启动定时任务
    _startRestTimer();
    if (_config!.distanceDetection.enabled) {
      _startDistanceDetection();
    }
    if (_config!.blueLightFilter.autoAdjust) {
      _startEnvLightDetection();
    }

    // 4. 记录会话开始
    _sessionStartTime = DateTime.now();

    // 5. 上报激活事件
    _reportActivation();
  }

  /// 加载配置：优先本地缓存，失败则请求服务端
  Future<EyeHealthConfig> _loadConfig(String studentId) async {
    // 本地缓存
    final cached = await _localCache.get('eye_health_config_$studentId');
    if (cached != null) {
      final config = EyeHealthConfig.fromJson(jsonDecode(cached));
      // 后台异步检查更新
      _checkRemoteUpdate(studentId, config.configVersion);
      return config;
    }
    // 请求服务端
    final remote = await _api.get('/api/v1/eye-health/config',
      params: {'studentId': studentId});
    final config = EyeHealthConfig.fromJson(remote.data);
    await _localCache.set('eye_health_config_$studentId', jsonEncode(config.toJson()));
    return config;
  }

  /// 应用护眼滤镜到全局 Overlay
  Future<void> _applyFilter() async {
    final level = _config!.level;
    final intensity = _config!.blueLightFilter.intensity;

    // 使用 Flutter OverlayEntry 实现全局色温滤镜
    final overlay = OverlayEntry(
      builder: (context) => Positioned.fill(
        child: IgnorePointer(
          child: Container(
            color: _calculateFilterColor(level, intensity),
          ),
        ),
      ),
    );

    // 注册到全局 Navigator overlay
    GlobalOverlayRegistry.register('eye_filter', overlay);
  }

  /// 根据护眼等级计算滤镜颜色
  Color _calculateFilterColor(EyeHealthLevel level, double intensity) {
    return switch (level) {
      EyeHealthLevel.L0 => Colors.transparent,
      EyeHealthLevel.L1 => Color.fromRGBO(255, 248, 220, 0.08),   // 微暖
      EyeHealthLevel.L2 => Color.fromRGBO(255, 235, 180, 0.15),   // 暖黄
      EyeHealthLevel.L3 => Color.fromRGBO(255, 218, 140, 0.25),   // 深暖
      EyeHealthLevel.L4 => Color.fromRGBO(255, 200, 120, 0.35),   // 极暖
    };
  }
}
```

```dart
/// 用眼时长追踪器
class ScreenTimeTracker {
  final EyeHealthConfig _config;
  DateTime? _segmentStart;
  Duration _totalScreenTime = Duration.zero;
  int _segmentCount = 0;

  /// 开始追踪一个学习片段
  void startSegment() {
    _segmentStart = DateTime.now();
    _segmentCount++;
  }

  /// 结束当前片段
  Duration endSegment() {
    if (_segmentStart == null) return Duration.zero;
    final duration = DateTime.now().difference(_segmentStart!);
    _totalScreenTime += duration;
    _segmentStart = null;
    return duration;
  }

  /// 获取当前连续使用时长
  Duration get continuousUsage {
    if (_segmentStart == null) return Duration.zero;
    return DateTime.now().difference(_segmentStart!);
  }

  /// 是否达到休息阈值
  bool get shouldRest {
    return continuousUsage.inMinutes >= _config.restReminder.intervalMinutes;
  }

  /// 获取剩余使用时间
  Duration get remainingTime {
    final limit = Duration(minutes: _config.restReminder.intervalMinutes);
    final used = continuousUsage;
    return used >= limit ? Duration.zero : limit - used;
  }
}
```

```dart
/// 环境光感应与亮度适配
class AmbientLightAdapter {
  static const platform = MethodChannel('com.primetop/sensors');

  /// 获取环境光照度（lux）
  Future<double> getAmbientLightLux() async {
    try {
      return await platform.invokeMethod<double>('getAmbientLight') ?? 300.0;
    } on PlatformException {
      return 300.0; // 默认中等光照
    }
  }

  /// 根据环境光计算推荐屏幕亮度
  double calculateBrightness(double lux) {
    // 暗环境(<50lux)：亮度 0.2-0.4
    // 正常环境(50-500lux)：亮度 0.4-0.7
    // 亮环境(>500lux)：亮度 0.7-1.0
    if (lux < 50) {
      return 0.2 + (lux / 50) * 0.2;
    } else if (lux < 500) {
      return 0.4 + ((lux - 50) / 450) * 0.3;
    } else {
      return min(0.7 + ((lux - 500) / 500) * 0.3, 1.0);
    }
  }

  /// 环境光过低警告
  bool isTooDark(double lux) => lux < 30;

  /// 环境光过高警告（可能导致反光）
  bool isTooBright(double lux) => lux > 2000;
}
```

```dart
/// 用眼距离检测（使用前置摄像头 + ML Kit 人脸检测）
class DistanceDetector {
  final CameraDescription _frontCamera;
  FaceDetector? _faceDetector;
  bool _isDetecting = false;

  static const double FACE_WIDTH_CM = 14.0; // 平均成人面宽(cm)
  static const double FOCAL_LENGTH_PX = 800.0; // 典型手机焦距(px)

  DistanceDetector(this._frontCamera) {
    _faceDetector = FaceDetector(
      options: FaceDetectorOptions(
        enableClassification: false,
        performanceMode: FaceDetectorMode.fast,
      ),
    );
  }

  /// 检测当前用眼距离（厘米）
  Future<DistanceResult> detectDistance() async {
    if (_isDetecting) return DistanceResult.skipped;
    _isDetecting = true;

    try {
      final image = await _captureFrontCameraFrame();
      final faces = await _faceDetector!.processImage(image);

      if (faces.isEmpty) {
        return DistanceResult.noFace;
      }

      final face = faces.first;
      final faceWidthPx = face.boundingBox.width;
      final distanceCm = (FACE_WIDTH_CM * FOCAL_LENGTH_PX) / faceWidthPx;

      return DistanceResult(
        distanceCm: distanceCm,
        isTooClose: distanceCm < 30,
        isWarning: distanceCm < 35,
        confidence: face.trackingId != null ? 0.8 : 0.6,
      );
    } catch (e) {
      return DistanceResult.error;
    } finally {
      _isDetecting = false;
    }
  }

  void dispose() {
    _faceDetector?.close();
  }
}

class DistanceResult {
  final double distanceCm;
  final bool isTooClose;
  final bool isWarning;
  final double confidence;
  final bool skipped;
  final bool noFace;
  final bool error;

  DistanceResult({
    required this.distanceCm,
    required this.isTooClose,
    required this.isWarning,
    required this.confidence,
    this.skipped = false,
    this.noFace = false,
    this.error = false,
  });

  DistanceResult.skipped()
      : distanceCm = 0, isTooClose = false, isWarning = false,
        confidence = 0, skipped = true, noFace = false, error = false;
  DistanceResult.noFace()
      : distanceCm = 0, isTooClose = false, isWarning = false,
        confidence = 0, skipped = false, noFace = true, error = false;
  DistanceResult.error()
      : distanceCm = 0, isTooClose = false, isWarning = false,
        confidence = 0, skipped = false, noFace = false, error = true;
}
```

```dart
/// 视觉疲劳综合评估
class FatigueEvaluator {
  /// 评估疲劳等级（0-100，越高越疲劳）
  FatigueAssessment evaluate({
    required Duration continuousTime,
    required Duration totalTime,
    required double blinkRate,        // 眨眼频率（次/分钟），通过摄像头估算
    required double ambientLight,     // 环境光照度
    required DistanceResult distance, // 距离检测结果
    required int segmentCount,        // 学习片段数
  }) {
    double score = 0;

    // 1. 连续使用时长因子（0-30分）
    score += _continuousTimeFactor(continuousTime);

    // 2. 累计使用时长因子（0-20分）
    score += _totalTimeFactor(totalTime);

    // 3. 眨眼频率因子（0-20分）
    // 正常眨眼 15-20次/分钟，低于10次/分钟视为疲劳信号
    score += _blinkRateFactor(blinkRate);

    // 4. 用眼距离因子（0-15分）
    score += _distanceFactor(distance);

    // 5. 环境光因子（0-15分）
    score += _ambientLightFactor(ambientLight);

    score = min(score, 100);

    return FatigueAssessment(
      score: score,
      level: _scoreToLevel(score),
      continuousTime: continuousTime,
      totalTime: totalTime,
      recommendations: _generateRecommendations(score, continuousTime, distance, ambientLight),
    );
  }

  FatigueLevel _scoreToLevel(double score) {
    if (score < 20) return FatigueLevel.fresh;
    if (score < 40) return FatigueLevel.slight;
    if (score < 60) return FatigueLevel.moderate;
    if (score < 80) return FatigueLevel.tired;
    return FatigueLevel.exhausted;
  }

  double _continuousTimeFactor(Duration t) {
    final min = t.inMinutes;
    if (min < 15) return 0;
    if (min < 25) return 5;
    if (min < 40) return 12;
    if (min < 60) return 20;
    return 30;
  }

  double _totalTimeFactor(Duration t) {
    final min = t.inMinutes;
    if (min < 30) return 0;
    if (min < 60) return 5;
    if (min < 120) return 10;
    return 20;
  }

  double _blinkRateFactor(double rate) {
    if (rate >= 15) return 0;
    if (rate >= 10) return 8;
    if (rate >= 5) return 14;
    return 20;
  }

  double _distanceFactor(DistanceResult d) {
    if (d.noFace || d.error || d.skipped) return 5; // 无法检测时给中间分
    if (d.distanceCm >= 35) return 0;
    if (d.distanceCm >= 30) return 5;
    if (d.distanceCm >= 20) return 10;
    return 15;
  }

  double _ambientLightFactor(double lux) {
    if (lux >= 100 && lux <= 500) return 0;
    if (lux >= 50 && lux < 100) return 5;
    if (lux >= 30 && lux < 50) return 10;
    if (lux < 30) return 15;
    if (lux > 1000) return 10;
    return 5;
  }

  List<String> _generateRecommendations(double score, Duration time,
    DistanceResult dist, double lux) {
    final recs = <String>[];
    if (score > 40) recs.add('建议休息 5-10 分钟，远眺窗外放松');
    if (time.inMinutes > 40) recs.add('已连续学习较长时间，请起来活动');
    if (!dist.noFace && dist.isTooClose) recs.add('请保持 30cm 以上的用眼距离');
    if (lux < 50) recs.add('环境光线过暗，请开灯或移至光线充足处');
    if (lux > 1000) recs.add('光线过强可能造成屏幕反光，请调整位置');
    return recs;
  }
}

enum FatigueLevel { fresh, slight, moderate, tired, exhausted }

class FatigueAssessment {
  final double score;
  final FatigueLevel level;
  final Duration continuousTime;
  final Duration totalTime;
  final List<String> recommendations;

  FatigueAssessment({
    required this.score,
    required this.level,
    required this.continuousTime,
    required this.totalTime,
    required this.recommendations,
  });
}
```
