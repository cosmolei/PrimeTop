# 客户端应用包体积优化与资源管理策略 - 详细设计

## 1. 模块概述

### 1.1 功能定位

PrimeTop 客户端基于 Flutter 构建，面向 Android / iOS 双端分发。随着功能模块持续扩展（AI 辅导、拍题 OCR、同步课堂、错题本、多媒体播放、端侧推理等），APK / IPA 体积将快速膨胀。本文档定义包体积管控目标、资源管理策略、构建优化手段和持续监控体系，确保安装包在各学段目标设备上保持合理的体积和下载体验。

### 1.2 设计目标

| 指标 | Android (APK/AAB) | iOS (IPA) |
|------|-------------------|-----------|
| MVP 阶段目标 | ≤ 40 MB | ≤ 45 MB |
| V1.0 阶段目标 | ≤ 60 MB | ≤ 65 MB |
| V2.0 阶段目标 | ≤ 80 MB | ≤ 85 MB |
| 下载转化率影响 | 首屏下载 ≤ 20 MB（Play Store / App Store 压缩后） | 同左 |
| 安装后增量 | 按需下载资源包，首装增量 ≤ 15 MB | 同左 |

### 1.3 设计原则

1. **预算驱动**：每个模块 / 功能有包体积预算，超预算必须优化
2. **按需加载**：非核心资源在首次使用时按需下载，不打入主包
3. **自动化管控**：CI/CD 自动检测体积变化，超阈值阻断合并
4. **分层治理**：主包 → 资源包 → 端侧模型三层分离

```

推荐：使用 App Bundle (.aab)，Google Play 自动按设备 ABI 分发
```

### 4.1 图片格式策略

| 场景 | 格式 | 说明 |
|------|------|------|
| 图标 / 简单图形 | SVG (flutter_svg) | 矢量，无分辨率限制 |
| 照片 / 复杂图像 | WebP (lossy, quality=80) | 比 PNG 小 25-35%，比 JPEG 小 25% |

```python
import os
import subprocess
import hashlib
from pathlib import Path

MAX_SINGLE_IMAGE_KB = 200  # 单张图片体积上限（主包内）
WARN_SINGLE_IMAGE_KB = 50  # 单张图片告警阈值
SUPPORTED_FORMATS = {'.png', '.jpg', '.jpeg', '.webp', '.gif'}

def check_image_sizes(assets_dir: str) -> list:
    """检查所有图片资源体积"""
    violations = []
    for root, _, files in os.walk(assets_dir):
        for f in files:
            path = Path(root) / f
            if path.suffix.lower() in SUPPORTED_FORMATS:
                size_kb = path.stat().st_size / 1024
                if size_kb > MAX_SINGLE_IMAGE_KB:
                    violations.append({
                        'path': str(path),
                        'size_kb': round(size_kb, 1),
                        'level': 'ERROR',
```

```dart
/// 统一图片加载 Widget
class OptimizedImage extends StatelessWidget {
  final String? assetPath;
  final String? networkUrl;
  final double? width;
  final double? height;
  final BoxFit fit;

  const OptimizedImage({
    super.key,
    this.assetPath,
    this.networkUrl,
    this.width,
    this.height,
    this.fit = BoxFit.cover,
  });

  @override
  Widget build(BuildContext context) {
    if (assetPath != null) {
      return Image.asset(
        assetPath!,
        width: width,
        height: height,
        fit: fit,
        cacheWidth: _calcCacheWidth(context),
        cacheHeight: _calcCacheHeight(context),
        errorBuilder: (_, __, ___) => _errorWidget(),
      );
    }

    if (networkUrl != null) {
      return CachedNetworkImage(
        imageUrl: networkUrl!,
        width: width,
        height: height,
        fit: fit,
        memCacheWidth: _calcCacheWidth(context),
        memCacheHeight: _calcCacheHeight(context),
        placeholder: (_, __) => _shimmerPlaceholder(),
        errorWidget: (_, __, ___) => _errorWidget(),
      );
    }

    return _errorWidget();
  }

  /// 根据设备像素比计算缓存尺寸，避免解码过大的图片
  int? _calcCacheWidth(BuildContext context) {
    if (width == null) return null;
    return (width! * MediaQuery.devicePixelRatioOf(context)).ceil();
  }

  int? _calcCacheHeight(BuildContext context) {
    if (height == null) return null;
    return (height! * MediaQuery.devicePixelRatioOf(context)).ceil();
  }
}
```

### 5.2 字体子集化工具

```python
# scripts/subset_fonts.py
"""
字体子集化脚本
使用 fonttools + harfbuzz 进行字体子集化
"""
import subprocess
from pathlib import Path

# 按学段定义字符集
CHARSET = {
    'common': 'charset/common_3500.txt',      # 常用 3500 字
    'junior': 'charset/junior_6500.txt',       # 初中扩展
    'senior': 'charset/senior_8000.txt',       # 高中扩展
    'math':   'charset/math_symbols.txt',      # 数学符号集
}

def subset_font(
    input_font: str,
    output_font: str,
    charset_file: str,
    flavor: str = 'woff2',
):
    """使用 pyftsubset 进行字体子集化"""
    cmd = [
        'pyftsubset',
        input_font,
        f'--text-file={charset_file}',
        f'--flavor={flavor}',
        '--layout-features=*',
        '--name-IDs=*',
        '--no-hinting',           # 移除 hinting 减少体积
        '--desubroutinize',       # 优化 CFF 字体
        f'--output-file={output_font}',
    ]
    subprocess.run(cmd, check=True)

def build_font_pack():
    """构建分学段字体包"""
    # 基础包：common 字符集（打入主包）
    subset_font(
        'assets/fonts/NotoSansSC-Regular.otf',
        'assets/fonts/NotoSansSC-Common.subset.woff2',
        CHARSET['common'],
    )

    # 数学符号包（打入主包，理科解题需要）
    subset_font(
        'assets/fonts/KaTeX-Main.ttf',
        'assets/fonts/KaTeX-Main.subset.woff2',
        CHARSET['math'],
    )
```

### 5.4 动态字体加载

```dart
/// 动态字体加载器 - 用于加载按需下载的扩展字体包
class DynamicFontLoader {
  static final _loadedFonts = <String>{};

  /// 加载从资源包下载的字体
  static Future<void> loadFontFromFile({
    required String fontName,
    required String fontPath, // 本地文件路径
  }) async {
    if (_loadedFonts.contains(fontName)) return;

    final fontLoader = FontLoader(fontName);
    final file = File(fontPath);
    if (!await file.exists()) {
      throw FontNotFoundException(fontName, fontPath);
    }

    final bytes = await file.readAsBytes();
    fontLoader.addFont(Future.value(ByteData.sublistView(bytes)));
    await fontLoader.load();

    _loadedFonts.add(fontName);
  }

  /// 检查字体是否已加载
  static bool isFontLoaded(String fontName) => _loadedFonts.contains(fontName);
}
```

```python
# scripts/size_analysis.py
"""
包体积分析工具
对比基准包，输出体积变化报告
"""
import json
import subprocess
import sys
from pathlib import Path
from dataclasses import dataclass, asdict

@dataclass
class SizeEntry:
    path: str
    size_bytes: int
    category: str

@dataclass
class SizeDelta:
    path: str
    old_size: int
    new_size: int
    delta_bytes: int
    delta_percent: float

def analyze_apk(apk_path: str) -> list[SizeEntry]:
    """使用 Android Build Tools 分析 APK 组成"""
    result = subprocess.run(
        ['bundletool', 'analyze', 'apk', '--apk', apk_path],
        capture_output=True, text=True
    )
    # 解析 APK 内容
    entries = []
    # ... 解析逻辑
    return entries

def compare_with_baseline(
    current: list[SizeEntry],
    baseline: list[SizeEntry],
) -> list[SizeDelta]:
    """对比当前包与基准包的差异"""
    baseline_map = {e.path: e for e in baseline}
    deltas = []
    for entry in current:
        if entry.path in baseline_map:
            old = baseline_map[entry.path].size_bytes
            deltas.append(SizeDelta(
                path=entry.path,
                old_size=old,
                new_size=entry.size_bytes,
                delta_bytes=entry.size_bytes - old,
                delta_percent=(entry.size_bytes - old) / old * 100 if old > 0 else 0,
            ))
        else:
            deltas.append(SizeDelta(
                path=entry.path,
                old_size=0,
                new_size=entry.size_bytes,
                delta_bytes=entry.size_bytes,
                delta_percent=100.0,
            ))
    return deltas

def check_threshold(
    deltas: list[SizeDelta],
    max_delta_bytes: int,
) -> bool:
    """检查是否超过体积增长阈值"""
    total_delta = sum(d.delta_bytes for d in deltas)
    if total_delta > max_delta_bytes:
        print(f"❌ 包体积增加 {total_delta / 1024 / 1024:.2f} MB "
              f"超过阈值 {max_delta_bytes / 1024 / 1024:.2f} MB")
        return False
    print(f"✅ 包体积变化 {total_delta / 1024 / 1024:+.2f} MB")
    return True
```

```dart
/// 本地存储配额管理
class StorageQuotaManager {
  /// 各类资源最大占用空间
  static const quotas = StorageQuotas(
    maxCacheImages: 100 * 1024 * 1024,     // 图片缓存 100 MB
    maxCacheAiResponses: 50 * 1024 * 1024,  // AI 回答缓存 50 MB
    maxResourcePacks: 200 * 1024 * 1024,    // 资源包 200 MB
    maxOfflineContent: 500 * 1024 * 1024,   // 离线内容 500 MB
    maxLogFiles: 20 * 1024 * 1024,          // 日志文件 20 MB
    maxTempFiles: 30 * 1024 * 1024,         // 临时文件 30 MB
  );

  /// 检查存储空间是否足够下载资源包
  static Future<StorageCheckResult> checkBeforeDownload(
    int requiredBytes,
  ) async {
    final available = await getAvailableStorage();
    final totalUsed = await getTotalUsedSpace();
    final minReserve = 200 * 1024 * 1024; // 保留 200 MB 给系统

    if (available - requiredBytes < minReserve) {
      return StorageCheckResult.insufficient(
        required: requiredBytes,
        available: available,
        suggestion: '请清理离线内容或释放设备存储空间',
      );
    }
    return StorageCheckResult.sufficient();
  }

  /// LRU 淘汰：清理最久未使用的离线内容
  static Future<int> evictLeastUsed({int targetFreeBytes = 0}) async {
    int freed = 0;
    final offlineContents = await _getOfflineContentsSortedByAccess();

    for (final content in offlineContents) {
      if (freed >= targetFreeBytes) break;
      await content.delete();
      freed += content.sizeBytes;
    }
    return freed;
  }
}
```

## 10. 错误处理

### 10.1 错误码体系

| 错误码 | 场景 | 处理策略 |
|--------|------|----------|
| SIZE_001 | 资源包下载空间不足 | 提示用户清理存储或移除离线内容 |
| SIZE_002 | 资源包校验失败(sha256) | 自动重试(最多3次)，失败后清理残留 |
| SIZE_003 | 资源包解压失败 | 清理残留，重新下载 |
| SIZE_004 | 存储空间低于阈值(200MB) | 触发 LRU 淘汰，提醒用户 |
| SIZE_005 | 字体子集加载失败 | 降级使用系统默认字体 |
| SIZE_006 | 延迟组件加载失败 | 降级使用云端服务或提示重试 |
| SIZE_007 | 图片优化管线失败 | 跳过优化，使用原图 |

### 10.2 降级策略

```dart
/// 包体积相关降级策略
class SizeDegradationHandler {
  /// 资源包不可用时的降级
  static Widget handlePackUnavailable(
    BuildContext context,
    String packId,
  PackUnavailableReason reason,
  Widget Function() normalBuilder,
  ) {
    return switch (reason) {
      PackUnavailableReason.notDownloaded => _DownloadPrompt(
        packId: packId,
        child: normalBuilder(),
      ),
      PackUnavailableReason.downloadFailed => _RetryPrompt(
        packId: packId,
        child: _CloudFallback(),
      ),
      PackUnavailableReason.corrupted => _CorruptedAlert(
        packId: packId,
        onRetry: () => ResourcePackManager.I.redownload(packId),
      ),
    };
  }

  /// 端侧模型不可用时的降级到云端
  static Future<T> withCloudFallback<T>({
    required String modelId,
    required Future<T> Function() localInference,
    required Future<T> Function() cloudInference,
  }) async {
    try {
      if (await ResourcePackManager.I.isPackReady(modelId)) {
        return await localInference();
      }
    } catch (e) {
      _log.warning('端侧模型 $modelId 推理失败，降级到云端', e);
    }
    return await cloudInference();
  }
}
```
