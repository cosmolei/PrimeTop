## 1. 模块概述

### 1.1 模块定位

多媒体内容播放与交互引擎是 PrimeTop 客户端的统一媒体基础设施，为同步课堂视频课、英语听力练习、文科背诵检测、理科实验动画、知识点微课等场景提供一致的播放、交互和进度管理能力。

### 1.2 设计目标

| 目标 | 说明 |
| --- | --- |
| 统一播放内核 | 视频、音频、动画共用一套播放器内核，避免各业务模块各自实现 |
| 学段适配 | 低龄段大按钮/语音引导，高龄段快进/倍速/笔记标记 |
| 离线优先 | 已缓存内容无网可播，播放进度本地持久 |
| 低端机兼容 | 主流 Android 6.0+、iOS 13+，内存占用 < 80MB（视频场景） |
| 可扩展 | 支持后续嵌入互动题、弹幕笔记、画中画等增强能力 |

### 1.3 使用场景矩阵

| 场景 | 媒体类型 | 交互需求 | 来源模块 |
| --- | --- | --- | --- |
| 视频课程播放 | 视频（MP4/HLS） | 倍速、断点续播、章节跳转、笔记标记 | 同步课堂 |
| 知识点微课 | 视频（短） | 播完后弹出练习题 | 同步课堂、考点梳理 |
| 英语听力练习 | 音频 | A-B 循环、倍速、逐句听写、填空 | 英语专项 |
| 课文朗读示范 | 音频 | 同步高亮文字、逐句跟读 | 文科背诵 |
| 背诵检测 | 音频（录音） | 语音识别对比、分段评分 | 文科背诵 |
| 物理实验动画 | 动画（Lottie/WebP 序列帧） | 参数调节、暂停回放 | 理科解题 |
| 化学分子动画 | 3D 模型（GLTF） | 旋转、缩放、步骤演示 | 理科解题 |
| 拼音识字音频 | 音频 | 播放+跟读+评分 | 拼音识字启蒙 |

```typescript
interface MediaContent {
  /** 内容唯一标识 */
  id: string;
  /** 内容类型 */
  type: 'video' | 'audio' | 'animation' | 'model3d';
  /** 业务来源 */
  source: 'course' | 'listening' | 'recite' | 'experiment' | 'micro_lesson' | 'phonics';
  /** 标题 */
  title: string;
  /** 时长（毫秒），-1 表示直播流 */
  duration: number;
  /** 资源列表（支持多清晰度/多格式） */
  resources: MediaResource[];
  /** 字幕/歌词轨道 */
  subtitles?: SubtitleTrack[];
  /** 章节标记 */
  chapters?: Chapter[];
  /** 交互时间轴事件（如播到某秒弹出练习题） */
  interactiveEvents?: InteractiveEvent[];
  /** 元数据 */
  metadata: {
    /** 关联教材章节 ID */
    chapterId?: string;
    /** 关联知识点 ID 列表 */
    knowledgePointIds?: string[];
    /** 适用学段 */
    gradeRange: [number, number];
    /** 内容大小（字节） */
    fileSize: number;
    /** 加密标识 */
    encrypted: boolean;
  };
}

interface MediaResource {
  /** 清晰度/质量标签 */
  quality: 'auto' | 'sd' | 'hd' | 'fhd' | 'original';
  /** 播放地址，支持多协议 */
  urls: {
    /** HLS 流地址（优先） */
    hls?: string;
    /** DASH 流地址 */
    dash?: string;
    /** 直连 MP4/WebM/MP3 */
    direct?: string;

  audio?: {

    bitrate: number;  // kbps
```

```typescript
 * 播放器的完整状态模型
 * 遵循有限状态机设计

type PlaybackStatus =
  | 'idle'           // 初始化完成，未开始
  | 'loading'        // 缓冲中
  | 'ready'          // 首帧已缓冲，可播放
  | 'playing'        // 正在播放
  | 'paused'         // 用户暂停
  | 'buffering'      // 播放中缓冲
  | 'seeking'        // 跳转中
  | 'ended'          // 播放结束
  | 'error'          // 错误状态
  | 'released';      // 已释放资源

interface PlaybackState {
  /** 当前状态 */
  status: PlaybackStatus;
  /** 当前播放位置（毫秒） */
  position: number;
  /** 缓冲位置（毫秒） */
  bufferedPosition: number;
  /** 播放速度 (0.5 ~ 3.0) */
  speed: number;
  /** 音量 (0.0 ~ 1.0) */
  volume: number;
  /** 是否静音 */
  muted: boolean;
  /** 当前清晰度 */
  currentQuality: string;
  /** A-B 循环区间 */
  abRepeat: { start: number; end: number } | null;
  /** 错误信息 */
  error: PlaybackError | null;
```

```typescript
/**
 * 播放器工厂 - 根据内容类型和平台创建合适的播放器实例

class PlayerFactory {
  private static instance: PlayerFactory;

  static getInstance(): PlayerFactory {
    if (!this.instance) {
      this.instance = new PlayerFactory();
    }
    return this.instance;
  }

  /**
   * 创建播放器实例
   * @param content 媒体内容描述
   * @param config 播放配置
   */
  createPlayer(
    content: MediaContent,
    config: PlaybackConfig
  ): IPlatformPlayer {
    switch (content.type) {
      case 'video':
        return new VideoPlayer(content, config);
      case 'audio':
        return new AudioPlayer(content, config);
      case 'animation':
        return new AnimationPlayer(content, config);
      case 'model3d':
        return new Model3DViewer(content, config);
      default:
        throw new UnsupportedMediaError(content.type);
    }
  }
}
```

### 4.3 播放状态机

```
           ┌──────────────────────────────────────────┐
           │                                          │
           ▼                                          │
  ┌────────┐  load   ┌─────────┐  ready   ┌───────┐  │
  │  idle  │───────►│ loading │────────►│ ready │  │
  └────────┘        └────┬────┘         └───┬───┘  │
                         │ error            │       │
                         ▼                  │ play  │
                   ┌──────────┐             ▼       │
                   │  error   │◄───────┌──────────┐ │
                   └──────────┘  error │ playing  │ │
                         ▲             └──┬───┬───┘ │
                         │                │   │     │
                         │    ┌───────────┘   │     │
                         │    │ pause         │     │
                         │    ▼               │ seek│
                         │  ┌────────┐        │     │
                         │  │ paused │────────┘     │
                         │  └────┬───┘              │
                         │       │ play              │
                         │       ▼                   │
                         │   ┌──────────┐            │
                         │   │ seeking  │────────────┘
                         │   └────┬─────┘
                         │        │ complete
                         │        ▼
                         │   ┌────────┐   seek
                         │   │ ended  │──────────┘
                         │   └────────┘
                         │
                         └─── (retry/reload → loading)
```

状态机规则：
- **idle → loading**：调用 `initialize()` 或 `play()`
- **loading → ready**：首帧数据已加载
- **ready → playing**：`play()` 或 `autoPlay=true`
- **playing → paused**：用户暂停或交互事件暂停
- **playing → buffering**：缓冲不足，自动触发
- **buffering → playing**：缓冲充足
- **playing → seeking**：`seekTo()` 调用
- **seeking → playing**：跳转完成
- **playing → ended**：播放结束
- **任意状态 → error**：平台错误、网络错误、解码错误
- **ended → loading**：重播

### 4.4 Android 平台实现（ExoPlayer）

```kotlin
/**
 * Android 视频播放器实现
 * 基于 ExoPlayer 2.19+
 */
class ExoVideoPlayer(
    private val context: Context,
    private val content: MediaContent,
    private val config: PlaybackConfig
) : IPlatformPlayer {

    private var player: ExoPlayer? = null
    private val stateRelay = MutableStateFlow<PlaybackState>(
        PlaybackState(status = "idle", position = 0, ...)
    )

    override fun initialize() {
        player = ExoPlayer.Builder(context)
            .setLoadControl(
                DefaultLoadControl.Builder()
                    .setMinBufferMs((config.preloadDuration ?: 10) * 1000)
                    .setMaxBufferMs((config.maxBufferDuration ?: 60) * 1000)
                    .build()
            )
            .setTrackSelector(DefaultTrackSelector(context).apply {
                // 根据配置设置清晰度参数
                setParameters(buildTrackParameters())
            })
            .build()

        // 构建 MediaSource
        val mediaSource = buildMediaSource()
        player?.setMediaSource(mediaSource)

        // 监听器
        player?.addListener(object : Player.Listener {
            override fun onPlaybackStateChanged(state: Int) {
                updateState(state)
            }
            override fun onPlayerError(error: PlaybackException) {
                handleError(error)
            }
        })

        // 断点续播
        if (config.startPosition != null && config.startPosition > 0) {
            player?.seekTo(config.startPosition)
        }

        player?.prepare()
    }

    private fun buildMediaSource(): MediaSource {
        val resource = selectBestResource()
        return when {
            resource.urls.hls != null -> {
                val factory = HlsMediaSource.Factory(
                    DefaultHttpDataSource.Factory()
                        .setUserAgent("PrimeTop/1.0")
                )
                if (content.metadata.encrypted) {
                    // DRM 处理
                    factory.createMediaSource(
                        MediaItem.Builder()
                            .setUri(resource.urls.hls)
                            .setDrmConfiguration(
                                MediaItem.DrmConfiguration.Builder(C.WIDEVINE_UUID)
                                    .setLicenseUri(DRM_LICENSE_URL)
                                    .build()
                            )
                            .build()
                    )
                } else {
                    factory.createMediaSource(
                        MediaItem.fromUri(resource.urls.hls!!)
                    )
                }
            }
            resource.urls.direct != null -> {
                ProgressiveMediaSource.Factory(
                    DefaultHttpDataSource.Factory()
                ).createMediaSource(
                    MediaItem.fromUri(resource.urls.direct!!)
                )
            }
            else -> throw PlaybackError(
                code = "NO_AVAILABLE_URL",
                message = "无可用播放地址"
            )
        }
    }

    /**
     * 根据网络条件和设备性能选择最优资源
     */
    private fun selectBestResource(): MediaResource {
        val bandwidth = BandwidthEstimator.estimate()
        val screenRes = getScreenResolution()

        return content.resources
            .filter { res ->
                // 检查设备是否支持该编码
                isCodecSupported(res.video?.codec)
            }
            .sortedByDescending { res ->
                // 评分：带宽匹配度 * 质量权重
                val bandwidthScore = calculateBandwidthScore(bandwidth, res)
                val qualityScore = calculateQualityScore(screenRes, res)
                bandwidthScore * 0.7 + qualityScore * 0.3
            }
            .first()
    }

    override fun release() {
        player?.release()
        player = null
        stateRelay.value = stateRelay.value.copy(status = "released")
    }

    // ... 其他 IPlatformPlayer 方法实现
}
```

### 4.5 iOS 平台实现（AVPlayer）

```swift
/**
 * iOS 视频播放器实现
 * 基于 AVPlayer + AVFoundation
 */
class AVVideoPlayer: IPlatformPlayer {
    private var player: AVPlayer?
    private var playerItem: AVPlayerItem?
    private var timeObserver: Any?
    private let stateSubject = CurrentValueSubject<PlaybackState, Never>(
        PlaybackState(status: "idle", position: 0, ...)
    )

    func initialize(content: MediaContent, config: PlaybackConfig) {
        let resource = selectBestResource(content: content)
        guard let url = resource.urls.hls ?? resource.urls.direct else {
            handleError(PlaybackError(code: "NO_URL", message: "无可用地址"))
            return
        }

        let asset = AVURLAsset(url: URL(string: url)!)
        playerItem = AVPlayerItem(asset: asset)

        // 配置 preferredForwardBufferDuration
        playerItem?.preferredForwardBufferDuration = TimeInterval(
            config.preloadDuration ?? 10
        )

        player = AVPlayer(playerItem: playerItem)

        // 断点续播
        if let start = config.startPosition, start > 0 {
            player?.seek(
                to: CMTime(milliseconds: start),
                toleranceBefore: .zero,
                toleranceAfter: .zero
            )
        }

        // 定时器更新位置
        let interval = CMTime(milliseconds: 250)
        timeObserver = player?.addPeriodicTimeObserver(
            forInterval: interval,
            queue: .main
        ) { [weak self] time in
            self?.updatePosition(time)
        }

        // 监听状态
        setupNotifications()
    }

    private func setupNotifications() {
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(playerDidFinishPlaying),
            name: .AVPlayerItemDidPlayToEndTime,
            object: playerItem
        )

        playerItem?.addObserver(
            self,
            forKeyPath: "status",
            options: [.new],
            context: nil
        )
        playerItem?.addObserver(
            self,
            forKeyPath: "playbackBufferEmpty",
            options: [.new],
            context: nil
        )
        playerItem?.addObserver(
            self,
            forKeyPath: "playbackLikelyToKeepUp",
            options: [.new],
            context: nil
        )
    }

    // ... 其他方法实现
}
```

### 5.1 倍速控制

```typescript
/**
 * 倍速控制器
 * 学段相关策略：
 * - 幼儿/小学（G1-G6）：不允许超过 1.5x
 * - 初中（G7-G9）：允许 0.5x-2.0x
 * - 高中（G10-G12）：允许 0.5x-3.0x
 */
class SpeedController {
  private static readonly GRADE_SPEED_POLICY: Record<string, number[]> = {
    'elementary': [0.75, 1.0, 1.25, 1.5],
    'middle':     [0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0],
    'high':       [0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0],
  };

  private currentSpeed: number = 1.0;
  private allowedSpeeds: number[];
  private player: IPlatformPlayer;

  constructor(player: IPlatformPlayer, gradeRange: [number, number]) {
    this.player = player;
    const stage = gradeToStage(gradeRange[0]);
    this.allowedSpeeds = SpeedController.GRADE_SPEED_POLICY[stage];
  }

  setSpeed(speed: number): void {
    // 校验速度是否允许
    if (!this.allowedSpeeds.includes(speed)) {
      // 找最接近的允许值
      speed = this.allowedSpeeds.reduce((prev, curr) =>
        Math.abs(curr - speed) < Math.abs(prev - speed) ? curr : prev
      );
    }
    this.currentSpeed = speed;
    this.player.setSpeed(speed);
  }

  /** 增加一档速度 */
  speedUp(): void {
    const idx = this.allowedSpeeds.indexOf(this.currentSpeed);
    if (idx < this.allowedSpeeds.length - 1) {
      this.setSpeed(this.allowedSpeeds[idx + 1]);
    }
  }

  /** 降低一档速度 */
  speedDown(): void {
    const idx = this.allowedSpeeds.indexOf(this.currentSpeed);
    if (idx > 0) {
      this.setSpeed(this.allowedSpeeds[idx - 1]);
    }
  }

  getAllowedSpeeds(): number[] {
    return [...this.allowedSpeeds];
  }
}

function gradeToStage(grade: number): string {
  if (grade <= 0) return 'elementary';  // 幼儿
  if (grade <= 6) return 'elementary';  // 小学
  if (grade <= 9) return 'middle';      // 初中
  return 'high';                        // 高中
}
```
