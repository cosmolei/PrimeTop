# 客户端性能监控与用户体验度量系统 - 详细设计

## 1. 概述

### 1.1 模块定位

客户端性能监控与用户体验度量系统是 PrimeTop 的客户端可观测性基础设施，负责采集、上报、聚合和展示客户端运行时性能指标与用户体验数据。区别于服务端的「日志与监控告警体系」（关注 API 延迟、服务健康），本系统聚焦于客户端侧的性能瓶颈、崩溃、ANR、卡顿、首屏加载、交互延迟等用户可感知的质量问题。

### 1.2 设计目标

| 目标 | 指标 |
|------|------|
| 崩溃率 < 0.1% | 每日崩溃用户 / 日活 < 0.1% |
| ANR率 < 0.3% | Android 严格模式 ANR |
| 首屏加载 < 2s | P90 冷启动到首页可交互 |
| AI首Token < 3s | P90 从发送到首个Token渲染 |
| 帧率 > 55fps | 关键交互页面的P90帧率 |
| 内存稳定 | 单次会话内存增长 < 50MB |

### 1.3 与已有模块的关系

| 已有模块 | 关系 |
|----------|------|
| 日志与监控告警体系 | 服务端监控，本系统是客户端侧的补充 |
| 数据埋点与关键指标系统 | 业务埋点，本系统是技术性能埋点 |
| 错误处理与服务降级策略 | 错误处理后上报到本系统 |
| 客户端版本管理与升级策略 | 版本升级后对比性能基线 |
| 灰度发布与特性开关系统 | 灰度期间监控新版本性能指标 |

```dart
class PerfEvent {
  /// 事件唯一ID (UUID v7)
  final String eventId;
  
  /// 事件类型
  final PerfEventType type;
  
  /// 事件名称 (如 "cold_start", "page_render", "ai_first_token")
  final String name;
  
  /// 事件发生时间 (ms 精度)
  final int timestamp;
  
  /// 持续时间 (ms), 0 表示瞬时事件
  final int duration;
  
  /// 附加标签
  final Map<String, String> tags;
  
  /// 附加数值指标
  final Map<String, double> metrics;
  
  /// 设备快照 (采样, 非每次都带)
  final DeviceSnapshot? deviceSnapshot;
  
  /// 应用状态快照
  final AppStateSnapshot appState;
  
  /// 会话ID
  final String sessionId;
  
  /// 用户ID (可选, 未登录时为空)
  final String? userId;
}

/// 事件类型枚举
enum PerfEventType {

  frameRate,
```

```dart
    return StartupReport(
      totalDuration: _milestoneTimes['home_interactive'] ?? 0,
      milestones: Map.from(_milestoneTimes),
      startupType: _determineStartupType(),
    );
  }
}
```

#### 3.2.3 帧率与卡顿采集

```dart
class FrameRateCollector {
  Timer? _sampleTimer;
  final List<FrameTiming> _frameTimings = [];
  
  static const _jankThresholdMs = 16.67 * 2; // 2帧掉帧 = 卡顿
  static const _sampleInterval = Duration(seconds: 1);
  static const _reportInterval = Duration(seconds: 30);

  void start() {
    // 使用 Flutter 的 FrameTiming 回调
    WidgetsBinding.instance.addTimingsCallback(_onFrameTimings);
    _sampleTimer = Timer.periodic(_reportInterval, (_) => _report());
  }

  void _onFrameTimings(List<FrameTiming> timings) {
    _frameTimings.addAll(timings);
    
    for (final timing in timings) {
      final frameDuration = timing.totalSpan.inMilliseconds;
      if (frameDuration > _jankThresholdMs) {
        // 检测到卡顿帧
        MonitorManager.instance.track(PerfEvent(
          eventId: Uuid().v7(),
          type: PerfEventType.frameRate,
          name: 'jank_detected',
          timestamp: DateTime.now().millisecondsSinceEpoch,
          duration: frameDuration,
          tags: {
            'build_ms': timing.buildDuration.inMilliseconds.toString(),
            'raster_ms': timing.rasterDuration.inMilliseconds.toString(),
          },
          metrics: {
            'frame_ms': frameDuration.toDouble(),
            'dropped_frames': (frameDuration / 16.67).floorToDouble(),
          },
          appState: AppStateSnapshot.current(),
          sessionId: SessionManager.currentSessionId,
        ));
      }
    }
  }

  void _report() {
    if (_frameTimings.isEmpty) return;
    
    final avgFps = _calculateAvgFps();
    final p90Fps = _calculatePercentileFps(0.9);
    final jankCount = _frameTimings.where(
      (t) => t.totalSpan.inMilliseconds > _jankThresholdMs
    ).length;
    
    MonitorManager.instance.track(PerfEvent(
      eventId: Uuid().v7(),
      type: PerfEventType.frameRate,
      name: 'fps_sample',
      timestamp: DateTime.now().millisecondsSinceEpoch,
      duration: 0,
      tags: {},
      metrics: {
        'avg_fps': avgFps,
        'p90_fps': p90Fps,
        'jank_count': jankCount.toDouble(),
        'total_frames': _frameTimings.length.toDouble(),
      },
      appState: AppStateSnapshot.current(),
      sessionId: SessionManager.currentSessionId,
    ));
    
    _frameTimings.clear();
  }

  double _calculateAvgFps() {
    if (_frameTimings.isEmpty) return 0;
    final avgMs = _frameTimings.map(
      (t) => t.totalSpan.inMilliseconds
    ).reduce((a, b) => a + b) / _frameTimings.length;
    return avgMs > 0 ? 1000 / avgMs : 0;
  }

  void stop() {
    WidgetsBinding.instance.removeTimingsCallback(_onFrameTimings);
    _sampleTimer?.cancel();
  }
}
```

```dart
class CrashCollector {
  static CrashCollector? _instance;
  static CrashCollector get instance => _instance ??= CrashCollector._();
  CrashCollector._();

  void init() {
    // 1. Dart 异常捕获
    FlutterError.onError = _onFlutterError;
    
    // 2. 异步未捕获异常
    PlatformDispatcher.instance.onError = _onPlatformError;
    
    // 3. Isolate 异常 (后台 Isolate)
    Isolate.current.addErrorListener(
      RawReceivePort((dynamic error) {
        _reportCrash(
          type: 'isolate_error',
          error: error.toString(),
          stackTrace: null,
        );
      }).sendPort,
    );

    // 4. 原生崩溃 (通过 MethodChannel 桥接)
    _setupNativeCrashHandler();
  }

  void _onFlutterError(FlutterErrorDetails details) {
    // 过滤掉已知的非致命异常
    if (_isNonFatal(details)) return;
    
    _reportCrash(
      type: 'dart_exception',
      error: details.exceptionAsString(),
      stackTrace: details.stack?.toString(),
      context: details.context?.toString(),
      library: details.library,
    );
  }

  bool _onPlatformError(Object error, StackTrace stack) {
    _reportCrash(
      type: 'platform_error',
      error: error.toString(),
      stackTrace: stack.toString(),
    );
    return true; // 已处理
  }

  void _reportCrash({
    required String type,
    required String error,
    String? stackTrace,
    String? context,
    String? library,
  }) {
    final event = CrashEvent(
      eventId: Uuid().v7(),
      crashType: type,
      errorMessage: error.length > 4000 ? error.substring(0, 4000) : error,
      stackTrace: stackTrace != null 
        ? (stackTrace.length > 8000 ? stackTrace.substring(0, 8000) : stackTrace)
        : null,
      context: context,
      library: library,
      deviceSnapshot: DeviceSnapshot.current(),
      appState: AppStateSnapshot.current(),
      sessionId: SessionManager.currentSessionId,
      userId: AuthManager.currentUserId,
      timestamp: DateTime.now().millisecondsSinceEpoch,
      // 附带最近 10 条操作日志用于复现
      recentActions: UserActionLogger.getRecentActions(limit: 10),
    );
    
    // 崩溃事件立即写入本地持久化
    _persistCrashEvent(event);
    
    // 尝试立即上报
    MonitorManager.instance.trackCrash(event);
  }

  void _persistCrashEvent(CrashEvent event) {
    final dir = Directory(
      path.join(AppPaths.cacheDir, 'crash_reports')
    );
    if (!dir.existsSync()) dir.createSync(recursive: true);
    
    final file = File(
      path.join(dir.path, '${event.eventId}.json')
    );
    file.writeAsStringSync(jsonEncode(event.toJson()));
  }

  bool _isNonFatal(FlutterErrorDetails details) {
    // 过滤: 图片加载失败、布局溢出等非致命异常
    final nonFatalPatterns = [
      'Unable to load asset',
      'A RenderFlex overflowed',
      'Looking up a deactivated widget',
    ];
    return nonFatalPatterns.any(
      (p) => details.exceptionAsString().contains(p)
    );
  }
}
```

```kotlin
// NativeCrashHandler.kt
class NativeCrashHandler : Thread.UncaughtExceptionHandler {
    private val defaultHandler = Thread.getDefaultUncaughtExceptionHandler()
    
    override fun uncaughtException(thread: Thread, throwable: Throwable) {
        val crashReport = buildCrashReport(throwable)
        saveCrashReport(crashReport)
        
        // 传递给默认处理器让系统处理
        defaultHandler?.uncaughtException(thread, throwable)
    }
    
    private fun buildCrashReport(throwable: Throwable): String {
        val sw = StringWriter()
        throwable.printStackTrace(PrintWriter(sw))
        return JSONObject().apply {
            put("crash_type", "native_crash")
            put("thread_name", thread.name)
            put("error_message", throwable.message ?: "Unknown")
            put("stack_trace", sw.toString().take(8000))
            put("timestamp", System.currentTimeMillis())
            put("device_info", DeviceInfoCollector.collect().toJson())
        }.toString()
    }
    
    private fun saveCrashReport(report: String) {
        val dir = File(context.cacheDir, "native_crash_reports")
        dir.mkdirs()
        File(dir, "${System.currentTimeMillis()}.json")
            .writeText(report)
    }
}
```

```dart
class MemoryCollector {
  Timer? _monitorTimer;
  
  static const _warningThresholdMB = 300;  // 内存警告阈值
  static const _criticalThresholdMB = 500; // 临界阈值
  static const _sampleInterval = Duration(seconds: 30);

  void start() {
    _monitorTimer = Timer.periodic(_sampleInterval, (_) => _sample());
  }

  void _sample() {
    final info = ProcessInfo.currentRss; // bytes
    
    MonitorManager.instance.track(PerfEvent(
      eventId: Uuid().v7(),
      type: PerfEventType.memory,
      name: 'memory_sample',
      timestamp: DateTime.now().millisecondsSinceEpoch,
      duration: 0,
      tags: {},
      metrics: {
        'rss_mb': info / (1024 * 1024),
        'heap_mb': _getHeapSize() / (1024 * 1024),
      },
      appState: AppStateSnapshot.current(),
      sessionId: SessionManager.currentSessionId,
    ));

    // 内存预警
    final rssMB = info / (1024 * 1024);
    if (rssMB > _criticalThresholdMB) {
      MonitorManager.instance.track(PerfEvent(
        eventId: Uuid().v7(),
        type: PerfEventType.memory,
        name: 'memory_critical',
        timestamp: DateTime.now().millisecondsSinceEpoch,
        duration: 0,
        tags: {'level': 'critical'},
        metrics: {'rss_mb': rssMB},
        appState: AppStateSnapshot.current(),
        sessionId: SessionManager.currentSessionId,
      ));
      // 触发主动内存回收
      _triggerMemoryCleanup();
    }
  }

  void _triggerMemoryCleanup() {
    // 清理图片缓存
    PaintingBinding.instance.imageCache.clear();
    PaintingBinding.instance.imageCache.clearLiveImages();
    // 通知各模块释放缓存
    EventBus.instance.fire(MemoryPressureEvent());
  }
}
```

## 4. 本地缓冲与上报策略

### 4.1 EventBuffer 设计

```dart
class EventBuffer {
  /// SQLite 本地存储, 保证崩溃不丢失
  final Database _db;
  
  /// 内存缓冲区 (减少磁盘IO)
  final List<PerfEvent> _memoryBuffer = [];
  
  /// 内存缓冲区上限
  static const _memoryBufferCapacity = 50;
  
  /// 本地存储上限
  static const _maxStoredEvents = 5000;
  
  /// 单次上报上限
  static const _batchSize = 100;

  Future<void> add(PerfEvent event) async {
    // 崩溃事件立即持久化
    if (event.type == PerfEventType.crash) {
      await _persistEvents([event]);
      return;
    }
    
    _memoryBuffer.add(event);
    
    if (_memoryBuffer.length >= _memoryBufferCapacity) {
      await _flushToDisk();
    }
  }

  Future<void> _flushToDisk() async {
    if (_memoryBuffer.isEmpty) return;
    await _persistEvents(List.from(_memoryBuffer));
    _memoryBuffer.clear();
  }

  Future<void> _persistEvents(List<PerfEvent> events) async {
    await _db.transaction((txn) async {
      for (final event in events) {
        await txn.insert(
          'perf_events',
          _eventToRow(event),
          conflictAlgorithm: ConflictAlgorithm.replace,
        );
      }
    });
    
    // 超限清理: 保留最新的 _maxStoredEvents 条
    await _db.rawDelete(
      'DELETE FROM perf_events WHERE id NOT IN '
      '(SELECT id FROM perf_events ORDER BY timestamp DESC LIMIT ?)',
      [_maxStoredEvents],
    );
  }

  /// 获取待上报事件
  Future<List<PerfEvent>> getPendingEvents({int limit = _batchSize}) async {
    final rows = await _db.query(
      'perf_events',
      orderBy: 'timestamp ASC',
      limit: limit,
    );
    return rows.map(_rowToEvent).toList();
  }

  /// 删除已上报事件
  Future<void> removeEvents(List<String> eventIds) async {
    await _db.delete(
      'perf_events',
      where: 'event_id IN (${List.filled(eventIds.length, '?').join(',')})',
      whereArgs: eventIds,
    );
  }
}
```

class UploadScheduler {
  final EventBuffer _buffer;
  final PerfApiClient _apiClient;
  Timer? _periodicTimer;
  
  // 上报策略配置
  static const _uploadInterval = Duration(seconds: 30);
  static const _maxBatchSize = 100;
  static const _maxRetries = 3;
  static const _retryBaseDelay = Duration(seconds: 5);

  void start() {
    // 定时上报
    _periodicTimer = Timer.periodic(_uploadInterval, (_) => _upload());
    
    // 应用回到前台时立即上报
    AppLifecycle.instance.onResume.listen((_) => _upload());
    
    // 崩溃恢复后立即上报
    _reportPendingNativeCrashes();
  }

  Future<void> _upload() async {
    final events = await _buffer.getPendingEvents(limit: _maxBatchSize);
    if (events.isEmpty) return;

    // 按类型分组聚合 (减少数据量)
    final aggregated = _aggregateEvents(events);
    
    // 压缩上报
    final payload = gzip.encode(utf8.encode(jsonEncode({
      'events': aggregated.map((e) => e.toJson()).toList(),
      'device_fingerprint': DeviceFingerprint.current,
      'app_version': AppInfo.version,
      'sdk_version': '1.0.0',
    })));

    try {
      await _apiClient.uploadPerfData(payload);
      // 上报成功, 删除本地记录
      await _buffer.removeEvents(events.map((e) => e.eventId).toList());
    } catch (e) {
      // 失败重试 (下次定时器触发时自动重试)
      _incrementRetryCount(events);
    }
  }

  /// 事件聚合: 同一秒内相同类型的指标取均值/最大值

      groups.putIfAbsent(key, () => []).add(event);
    }
