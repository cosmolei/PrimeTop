# 成本管理与优化系统 - 详细设计

> 模块定位：全平台成本核算、监控、预警、分析与优化策略管理系统，实现精细化成本控制，确保产品可持续运营。
> 优先级：P1（MVP阶段基础能力，V1.0完善）
> 依赖模块：多模型调度、API网关、日志监控、数据分析

---

## 1. 概述

### 1.1 背景与目标

PrimeTop 作为 AI 驱动的学习平台，涉及多种高成本服务：
- **大模型调用**：GPT-4、Claude、文心一言等模型API调用成本高昂
- **OCR识别**：拍照搜题需要商用OCR服务
- **语音服务**：ASR语音识别、TTS语音合成
- **CDN与存储**：多媒体内容分发与对象存储
- **数据库与缓存**：云数据库、Redis等

**目标：**
1. 实现全链路成本精准核算与归属
2. 建立实时成本监控与预警机制
3. 提供多维度成本分析与洞察
4. 自动化成本优化策略执行
5. 支持商业化决策与盈利分析

### 1.2 核心职责

| 职责 | 说明 |
|------|------|
| 成本核算 | 按服务类型、用户、功能、时间维度核算成本 |
| 成本监控 | 实时监控成本消耗，异常告警 |
| 成本预警 | 设定阈值，超限预警并触发防护策略 |
| 成本分析 | 多维度分析成本趋势、构成、效率 |
| 成本优化 | 自动化执行成本优化策略 |
| 成本报表 | 生成成本报表，支持财务对账 |

### 1.3 成本指标体系

| 指标类别 | 指标名称 | 计算方式 | 目标值 |
|---------|---------|---------|--------|
| 总体成本 | 日总成本 | 当日所有服务成本总和 | 按预算控制 |
| 总体成本 | 月总成本 | 当月所有服务成本总和 | 按预算控制 |
| 人均成本 | 单用户日均成本 | 日总成本 / DAU | ≤ 0.5元 |
| 人均成本 | 付费用户月均成本 | 月总成本 / 月活付费用户数 | ≤ 15元 |
| AI成本 | 单次AI问答成本 | AI总成本 / AI问答次数 | ≤ 0.1元 |
| AI成本 | 单次OCR成本 | OCR总成本 / OCR调用次数 | ≤ 0.05元 |
| 转化成本 | 获客成本 (CAC) | 营销成本 / 新增付费用户数 | ≤ 50元 |
| 转化成本 | 生命周期价值 (LTV) | 用户平均生命周期收入 | ≥ 200元 |
| 效率指标 | 成本收入比 (CIR) | 总成本 / 总收入 | ≤ 40% |
| 效率指标 | 毛利率 | (总收入 - 总成本) / 总收入 | ≥ 60% |

## 4. 核心业务逻辑

### 4.1 成本核算流程

```python
class CostAccountingService:
    """成本核算服务"""

    def __init__(self, db_session, config_service):
        self.db = db_session
        self.config = config_service

    async def record_cost(
        self,
        service_type: str,
        provider: str,
        usage_quantity: float,
        usage_unit: str,
        input_quantity: float = None,
        output_quantity: float = None,
        user_id: int = None,
        feature: str = None,
        request_id: str = None,
        session_id: str = None
    ) -> float:
        """
        记录成本流水

        Returns:
            float: 计算出的成本金额
        """
        # 1. 获取服务定价配置
        cost_config = await self.config.get_cost_config(
            service_type, provider
        )
        if not cost_config:
            logger.warning(f"No cost config for {service_type}/{provider}")
            return 0.0

        # 2. 计算成本
        total_cost = self._calculate_cost(
            cost_config,
            usage_quantity,
            input_quantity,
            output_quantity
        )

        # 3. 记录流水
        await self._insert_cost_ledger(
            cost_date=date.today(),
            service_type=service_type,
            provider=provider,
            usage_quantity=usage_quantity,
            usage_unit=usage_unit,
            input_quantity=input_quantity,
            output_quantity=output_quantity,
            total_cost=total_cost,
            user_id=user_id,
            feature=feature,
            request_id=request_id,
            session_id=session_id
        )

        # 4. 更新实时成本统计
        await self._update_cost_statistics(
            cost_date=date.today(),
            service_type=service_type,
            provider=provider,
            total_cost=total_cost
        )

        return total_cost

    def _calculate_cost(
        self,
        cost_config: dict,
        usage_quantity: float,
        input_quantity: float = None,
        output_quantity: float = None
    ) -> float:
        """计算成本"""
        # 阶梯定价
        if cost_config.get('tiered_pricing'):
            return self._calculate_tiered_cost(
                cost_config['tiered_pricing'],
                usage_quantity,
                input_quantity,
                output_quantity
            )

        # 标准定价
        if input_quantity and output_quantity:
            cost = (
                input_quantity * cost_config['input_unit_price'] +
                output_quantity * cost_config['output_unit_price']
            )
        else:
            cost = usage_quantity * cost_config.get('unit_price', 0)

        # 汇率转换
        if cost_config.get('currency') != 'CNY':
            cost = cost * cost_config.get('exchange_rate', 1.0)

        # 最低收费
        min_charge = cost_config.get('min_charge', 0)
        cost = max(cost, min_charge)

        return round(cost, 4)

    def _calculate_tiered_cost(
        self,
        tiered_pricing: dict,
        usage_quantity: float,
        input_quantity: float = None,
        output_quantity: float = None
    ) -> float:
        """计算阶梯定价成本"""
        tiers = tiered_pricing.get('tiers', [])
        total_cost = 0.0
        remaining_quantity = usage_quantity

        for tier in tiers:
            tier_min = tier.get('min_units', 0)
            tier_max = tier.get('max_units')

            # 计算该阶梯的使用量
            if tier_max is None:
                tier_quantity = remaining_quantity
            else:
                tier_quantity = min(remaining_quantity, tier_max - tier_min + 1)

            if tier_quantity <= 0:
                continue

            # 计算该阶梯的成本
            if input_quantity and output_quantity:
                tier_cost = (
                    input_quantity * tier.get('input_price', 0) +
                    output_quantity * tier.get('output_price', 0)
                )
            else:
                tier_cost = tier_quantity * tier.get('input_price', 0)

            total_cost += tier_cost
            remaining_quantity -= tier_quantity

            if remaining_quantity <= 0:
                break

        return round(total_cost, 4)

    async def _insert_cost_ledger(self, **kwargs):
        """插入成本流水"""
        query = """
            INSERT INTO cost_ledger (
                cost_date, service_category, service_type, provider,
                usage_quantity, usage_unit, input_quantity, output_quantity,
                total_cost, currency, user_id, feature,
                request_id, session_id
            ) VALUES (
                :cost_date, :service_category, :service_type, :provider,
                :usage_quantity, :usage_unit, :input_quantity, :output_quantity,
                :total_cost, :currency, :user_id, :feature,
                :request_id, :session_id
            )
        """
        await self.db.execute(query, kwargs)

    async def _update_cost_statistics(self, cost_date, service_type, provider, total_cost):
        """更新实时成本统计（缓存）"""
        # 使用 Redis 存储实时统计
        cache_key = f"cost_stats:{cost_date}:{service_type}:{provider}"
        await self.redis.incrbyfloat(cache_key, total_cost)
        await self.redis.expire(cache_key, 86400 * 30)  # 30天过期
```

```python
    async def monitor_daily_budget(self, cost_date: date = None):
        """监控日预算"""
        if cost_date is None:
            cost_date = date.today()

        # 1. 获取日预算
        budget = await self._get_budget(cost_date, 'daily')
        if not budget:
            logger.info(f"No daily budget for {cost_date}")
            return

        # 2. 获取实际成本
        actual_cost = await self._get_actual_cost(cost_date, 'daily')

        # 3. 计算使用率
        usage_rate = (actual_cost / budget['total_budget'] * 100) if budget['total_budget'] > 0 else 0

        # 4. 更新预算执行状态
        await self._update_budget_status(
            budget['id'],
            actual_cost,
            usage_rate
        )

        # 5. 检查告警阈值
        if usage_rate >= budget['critical_threshold']:
            await self._create_alert(
                alert_type='budget_exceed',
                alert_level='critical',
                message=f'日预算严重超限！使用率: {usage_rate:.2f}% (阈值: {budget["critical_threshold"]}%)',
                context={
                    'budget_id': budget['id'],
                    'budget': budget['total_budget'],
                    'actual': actual_cost,
                    'usage_rate': usage_rate
                }
            )
        elif usage_rate >= budget['warning_threshold']:
            await self._create_alert(
                alert_type='budget_exceed',
                alert_level='warning',
                message=f'日预算即将超限！使用率: {usage_rate:.2f}% (阈值: {budget["warning_threshold"]}%)',
                context={
                    'budget_id': budget['id'],
                    'budget': budget['total_budget'],
                    'actual': actual_cost,
                    'usage_rate': usage_rate
                }
            )

    async def detect_cost_spike(self):
        """检测成本异常激增"""
        # 1. 获取最近7天的成本数据
        cost_history = await self._get_cost_history(days=7)

        # 2. 计算平均成本和标准差
        costs = [day['total_cost'] for day in cost_history]
        avg_cost = statistics.mean(costs)
        std_cost = statistics.stdev(costs)

        # 3. 检测今日成本是否异常
        today_cost = await self._get_today_cost()
        if today_cost > avg_cost + 3 * std_cost:  # 3σ 原则
            await self._create_alert(
                alert_type='cost_spike',
                alert_level='critical',
                message=f'检测到成本异常激增！今日成本: ¥{today_cost:.2f}, 7日均值: ¥{avg_cost:.2f}',
                context={
                    'today_cost': today_cost,
                    'avg_cost': avg_cost,
                    'std_cost': std_cost,
                    'deviation': (today_cost - avg_cost) / avg_cost * 100
                }
            )

    async def monitor_service_cost(self, service_type: str, provider: str):
        """监控特定服务成本"""
        # 1. 获取服务配置
        config = await self._get_service_config(service_type, provider)
        if not config or not config.get('daily_limit'):
            return

        # 2. 获取今日服务成本
        today_cost = await self._get_service_cost_today(service_type, provider)

        # 3. 检查是否超限
        if today_cost >= config['daily_limit']:
            await self._create_alert(
                alert_type='abnormal_surge',
                alert_level='critical',
                message=f'{service_type} ({provider}) 日限额超限！今日成本: ¥{today_cost:.2f}, 限额: ¥{config["daily_limit"]:.2f}',
                context={
                    'service_type': service_type,
                    'provider': provider,
                    'today_cost': today_cost,
                    'daily_limit': config['daily_limit']
                }
            )

            # 触发限流策略
            await self._trigger_rate_limit(service_type, provider)

    async def _trigger_rate_limit(self, service_type: str, provider: str):
        """触发限流策略"""
        # 调用多模型调度服务，降低该服务的限流阈值
        await self.model_scheduler.update_rate_limit(
            service_type=service_type,
            provider=provider,
            new_limit=10  # 降低到 10 req/min
        )

        # 记录日志
        logger.warning(f"Rate limit triggered for {service_type}/{provider}")
```

```python
class CostAnalysisService:
    """成本分析服务"""

    def __init__(self, db_session):
        self.db = db_session

    async def get_cost_summary(
        self,
        start_date: date,
        end_date: date,
        dimensions: list = None
    ) -> dict:
        """
        获取成本汇总

        Args:
            start_date: 起始日期
            end_date: 结束日期
            dimensions: 维度列表，如 ['service_type', 'provider', 'feature']

        Returns:
            dict: 成本汇总数据
        """
        dimensions = dimensions or ['service_type']

        # 1. 获取总成本
        total_cost = await self._get_total_cost(start_date, end_date)

        # 2. 获取按维度分组的成本
        cost_by_dimension = {}
        for dimension in dimensions:
            cost_by_dimension[dimension] = await self._get_cost_by_dimension(
                start_date, end_date, dimension
            )

        # 3. 获取趋势数据
        cost_trend = await self._get_cost_trend(start_date, end_date)

        # 4. 获取效率指标
        efficiency_metrics = await self._get_efficiency_metrics(start_date, end_date)

        return {
            'period': {
                'start_date': start_date.isoformat(),
                'end_date': end_date.isoformat(),
                'days': (end_date - start_date).days + 1
            },
            'total_cost': total_cost,
            'cost_by_dimension': cost_by_dimension,
            'cost_trend': cost_trend,
            'efficiency_metrics': efficiency_metrics
        }

    async def _get_total_cost(self, start_date: date, end_date: date) -> dict:
        """获取总成本"""
        query = """
            SELECT
                service_category,
                SUM(total_cost) as cost
            FROM cost_ledger
            WHERE cost_date BETWEEN :start_date AND :end_date
            GROUP BY service_category
        """
        results = await self.db.fetch_all(
            query,
            {'start_date': start_date, 'end_date': end_date}
        )

        categories = {
            'AI': 0,
            'INFRA': 0,
            'THIRD_PARTY': 0
        }

        for row in results:
            if row['service_category'] in categories:
                categories[row['service_category']] = float(row['cost'])

        return {
            'total': sum(categories.values()),
            'by_category': categories
        }

    async def _get_cost_by_dimension(
        self,
        start_date: date,
        end_date: date,
        dimension: str
    ) -> list:
        """按维度获取成本"""
        query = f"""
            SELECT
                {dimension},
                SUM(total_cost) as cost,
                COUNT(*) as count
            FROM cost_ledger
            WHERE cost_date BETWEEN :start_date AND :end_date
            GROUP BY {dimension}
            ORDER BY cost DESC
        """
        results = await self.db.fetch_all(
            query,
            {'start_date': start_date, 'end_date': end_date}
        )

        return [
            {
                'dimension': dimension,
                'value': row[dimension],
                'cost': float(row['cost']),
                'count': row['count'],
                'percentage': (row['cost'] / await self._get_total_amount(start_date, end_date)) * 100
            }
            for row in results
        ]

    async def _get_cost_trend(self, start_date: date, end_date: date) -> list:
        """获取成本趋势（按天）"""
        query = """
            SELECT
                cost_date,
                SUM(total_cost) as cost
            FROM cost_ledger
            WHERE cost_date BETWEEN :start_date AND :end_date
            GROUP BY cost_date
            ORDER BY cost_date
        """
        results = await self.db.fetch_all(
            query,
            {'start_date': start_date, 'end_date': end_date}
        )

        return [
            {
                'date': row['cost_date'].isoformat(),
                'cost': float(row['cost'])
            }
            for row in results
        ]

    async def get_cost_per_user_analysis(
        self,
        start_date: date,
        end_date: date
    ) -> dict:
        """获取单用户成本分析"""
        query = """
            SELECT
                user_id,
                COUNT(*) as usage_count,
                SUM(total_cost) as total_cost,
                AVG(total_cost) as avg_cost_per_usage
            FROM cost_ledger
            WHERE cost_date BETWEEN :start_date AND :end_date
              AND user_id IS NOT NULL
            GROUP BY user_id
            ORDER BY total_cost DESC
        """
        results = await self.db.fetch_all(
            query,
            {'start_date': start_date, 'end_date': end_date}
        )

        # 计算分位数
        costs = [float(row['total_cost']) for row in results]
        p50 = statistics.quantiles(costs, n=100)[49] if len(costs) >= 100 else costs[len(costs)//2]
        p90 = statistics.quantiles(costs, n=100)[89] if len(costs) >= 100 else costs[int(len(costs)*0.9)]
        p95 = statistics.quantiles(costs, n=100)[94] if len(costs) >= 100 else costs[int(len(costs)*0.95)]

        return {
            'total_users': len(results),
            'total_cost': sum(costs),
            'avg_cost_per_user': sum(costs) / len(costs) if results else 0,
            'median_cost_per_user': p50,
            'p90_cost_per_user': p90,
            'p95_cost_per_user': p95,
            'top_users': [
                {
                    'user_id': row['user_id'],
                    'usage_count': row['usage_count'],
                    'total_cost': float(row['total_cost']),
                    'avg_cost_per_usage': float(row['avg_cost_per_usage'])
                }
                for row in results[:20]  # 前20名
            ]
        }
```

```python
class CostOptimizationService:
    """成本优化服务"""

    def __init__(self, db_session, strategy_service, model_scheduler):
        self.db = db_session
        self.strategy = strategy_service
        self.scheduler = model_scheduler

    async def execute_optimization_strategies(self):
        """执行成本优化策略"""
        # 1. 获取启用的优化策略
        strategies = await self.strategy.get_active_strategies()

        for strategy in strategies:
            try:
                # 2. 检查触发条件
                if await self._check_trigger_condition(strategy):
                    # 3. 执行策略
                    if strategy['auto_execute']:
                        savings = await self._execute_strategy(strategy)
                        await self._record_strategy_execution(
                            strategy['id'],
                            savings
                        )
                    else:
                        # 通知需要人工审核
                        await self._notify_manual_review(strategy)
            except Exception as e:
                logger.error(f"Failed to execute strategy {strategy['id']}: {e}")

    async def _execute_strategy(self, strategy: dict) -> float:
        """执行单个优化策略"""
        strategy_type = strategy['strategy_type']
        config = strategy['config']

        if strategy_type == 'caching':
            return await self._execute_caching_strategy(config)
        elif strategy_type == 'model_selection':
            return await self._execute_model_selection_strategy(config)
        elif strategy_type == 'rate_limit':
            return await self._execute_rate_limit_strategy(config)
        elif strategy_type == 'offload':
            return await self._execute_offload_strategy(config)
        else:
            logger.warning(f"Unknown strategy type: {strategy_type}")
            return 0.0

    async def _execute_caching_strategy(self, config: dict) -> float:
        """执行缓存策略"""
        # 1. 启用 AI 回答缓存
        cache_key_pattern = config.get('cache_key_pattern')
        cache_ttl = config.get('cache_ttl', 3600)

        await self.scheduler.enable_caching(
            cache_key_pattern=cache_key_pattern,
            cache_ttl=cache_ttl
        )

        # 2. 评估节省成本（基于历史命中率）
        cache_hit_rate = await self._get_cache_hit_rate()
        estimated_savings = await self._get_daily_ai_cost() * cache_hit_rate * 0.7

        return estimated_savings

    async def _execute_model_selection_strategy(self, config: dict) -> float:
        """执行模型选择策略"""
        # 1. 配置模型路由规则
        user_tier_routing = config.get('user_tier_routing', {})
        fallback_models = config.get('fallback_models', [])

        await self.scheduler.configure_model_routing(
            user_tier_routing=user_tier_routing,
            fallback_models=fallback_models
        )

        # 2. 评估节省成本
        # 假设 30% 的 FREE 用户切换到 GPT-3.5
        free_user_ratio = await self._get_user_tier_ratio('FREE')
        estimated_savings = (
            await self._get_daily_ai_cost() *
            free_user_ratio *
            0.3 *
            (1 - 0.2)  # GPT-3.5 比 GPT-4 便宜 80%
        )

        return estimated_savings

    async def _execute_rate_limit_strategy(self, config: dict) -> float:
        """执行限流策略"""
        # 1. 降低免费用户的请求频率
        await self.scheduler.update_user_rate_limit(
            user_tier='FREE',
            requests_per_minute=config.get('rate_limit', 10)
        )

        # 2. 评估节省成本
        estimated_reduction = 1 - (config.get('rate_limit', 10) / 60)
        estimated_savings = (
            await self._get_daily_ai_cost() *
            estimated_reduction *
            0.4  # 假设 40% 的用户是免费用户
        )

        return estimated_savings

    async def _execute_offload_strategy(self, config: dict) -> float:
        """执行卸载策略（将部分任务转移到低成本服务）"""
        # 例如：将简单的数学计算转移到本地计算引擎
        # 而不是调用大模型

        # 这里只是一个示例
        offload_ratio = config.get('offload_ratio', 0.2)
        estimated_savings = (
            await self._get_daily_ai_cost() *
            offload_ratio *
            0.9  # 本地计算几乎免费
        )

        return estimated_savings
```

#### 6.2.1 成本核算集成流程

```python
# 在 AI 模型调用完成后，调用成本核算服务
class AIModelService:
    async def call_model(self, user_id, prompt, model, feature):
        # 1. 调用模型
        response = await self.model_provider.call(prompt, model)

        # 2. 记录成本
        await cost_accounting.record_cost(
            service_type=model,
            provider=self._get_provider(model),
            usage_quantity=response.total_tokens,
            usage_unit='tokens',
            input_quantity=response.input_tokens,
            output_quantity=response.output_tokens,
            user_id=user_id,
            feature=feature,
            request_id=response.request_id,
            session_id=response.session_id
        )

        return response
```
