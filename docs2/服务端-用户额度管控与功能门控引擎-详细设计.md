# 服务端-用户额度管控与功能门控引擎-详细设计

## 1. 模块概述

### 1.1 定位

用户额度管控与功能门控引擎（Quota & Feature Gate Engine）是 PrimeTop 全站的**运行时权益执行层**。每当用户触发 AI 对话、拍题、练习、作文批改等高成本或会员限定功能时，本引擎负责：

1. **额度扣减**：按用户会员等级、功能类型、时间窗口检查并扣减可用额度
2. **功能门控**：判断用户是否有权访问某功能（基于会员等级、学段、灰度开关等）
3. **超限处理**：额度耗尽时返回友好降级响应，引导升级会员或等待重置
4. **用量统计**：实时聚合用量数据，供运营分析和计费使用

### 1.2 与现有模块的关系

```
┌─────────────────────────────────────────────────────┐
│                   API Gateway 层                      │
│  (鉴权 → QuotaMiddleware → 业务处理 → 响应)           │
└──────────────────────┬──────────────────────────────┘
                       │
         ┌─────────────┴─────────────┐
         │  本模块 (额度管控引擎)       │
         │  - QuotaService            │
         │  - FeatureGateService      │
         │  - QuotaCounter            │
         └─────────────┬─────────────┘
                       │ 依赖
    ┌──────────┬───────┴───────┬──────────────┐
    │ 支付服务  │ 会员权益定义   │ 配置中心      │
    │ (订单)   │ (权益矩阵)     │ (动态额度)    │
    └──────────┴───────────────┴──────────────┘
```

- **支付与会员订阅**：管理会员等级、订阅生命周期、订单，但不负责运行时检查
- **配置中心**：可动态调整各等级额度上限，无需发版
- **增值服务与商品系统**：额度包/次数包购买后充值到本引擎管理的额度池
- **灰度发布与特性开关**：功能级别的开关控制

## 3. Redis 额度计数器设计

### 3.1 Key 结构

```
# 日额度计数器
quota:{user_id}:{quota_type}:daily:{YYYY-MM-DD}    →  STRING (int, INCRBY)

# 周额度计数器
quota:{user_id}:{quota_type}:weekly:{YYYY-WNN}     →  STRING (int, INCRBY)

# 月额度计数器
quota:{user_id}:{quota_type}:monthly:{YYYY-MM}     →  STRING (int, INCRBY)

# 总量计数器（如存储空间）
quota:{user_id}:{quota_type}:total                 →  STRING (int, INCRBY)

# TTL：daily=48h / weekly=8d / monthly=35d / total=不过期

# 额度包剩余次数（来自购买）
topup:{user_id}:{quota_type}                       →  STRING (int, DECRBY)
# TTL = 额度包过期时间

# 用户会员等级缓存
user:membership:{user_id}                          →  HASH {tier, expires_at, ...}
# TTL = 5min
```

### 3.2 原子扣减脚本（Lua）

```lua
-- Lua: quota_check_and_consume.lua
-- KEYS[1] = quota counter key
-- KEYS[2] = topup counter key (optional, "" if not used)
-- ARGV[1] = limit value (-1 = unlimited)
-- ARGV[2] = cost weight (integer, 1 for simple, 2+ for heavy)
-- ARGV[3] = current timestamp (for audit)
-- Returns: {allowed(0/1), remaining, used, topup_remaining(-1 if N/A)}

local limit_val = tonumber(ARGV[1])
local cost = tonumber(ARGV[2])
local counter_key = KEYS[1]
local topup_key = KEYS[2]

local current = tonumber(redis.call('GET', counter_key) or '0')

-- 如果有额度包，优先扣额度包
if topup_key ~= '' then
    local topup_remaining = tonumber(redis.call('GET', topup_key) or '0')
    if topup_remaining >= cost then
        redis.call('DECRBY', topup_key, cost)
        return {1, -1, current, topup_remaining - cost}
    end
end

-- 额度包不足或无额度包，走正常额度
if limit_val == -1 then
    -- 无限制
    redis.call('INCRBY', counter_key, cost)
    return {1, -1, current + cost, -1}
end

local remaining = limit_val - current
if remaining >= cost then
    redis.call('INCRBY', counter_key, cost)
    return {1, remaining - cost, current + cost, -1}
else
    return {0, remaining, current, -1}
end
```

### 4.1 QuotaService — 额度服务

```python
# services/quota_service.py

from datetime import date, datetime
from typing import Optional
import logging

logger = logging.getLogger(__name__)


class QuotaService:
    """
    额度管控核心服务
    负责：检查、扣减、退还、查询
    """

    def __init__(
        self,
        redis: Redis,
        db: AsyncSession,
        config_service: ConfigService,
        membership_service: MembershipService,
    ):
        self.redis = redis
        self.db = db
        self.config = config_service
        self.membership = membership_service

    async def check_and_consume(
        self,
        user_id: str,
        quota_type: QuotaType,
        cost_weight: float = 1.0,
        request_id: Optional[str] = None,
    ) -> QuotaCheckResponse:
        """
        检查并扣减额度（原子操作）

        返回:
            QuotaCheckResponse:
                allowed: bool
                remaining: int (-1 = 无限)
                used: int
                limit: int (-1 = 无限)
                result: QuotaCheckResult
                degrade_info: Optional[dict]  # 降级信息
                upgrade_hint: Optional[str]    # 升级引导
        """
        # 1. 获取用户会员等级
        tier = await self.membership.get_effective_tier(user_id)

        # 2. 获取该等级的额度配置
        limit_config = await self.config.get_quota_limit(tier, quota_type)
        if limit_config is None:
            # 配置缺失 = 不限制
            return QuotaCheckResponse(
                allowed=True, remaining=-1, used=0,
                limit=-1, result=QuotaCheckResult.ALLOWED
            )

        # 3. 检查功能是否对该等级开放
        features = await self.config.get_tier_features(tier)
        if quota_type.value not in features:
            return QuotaCheckResponse(
                allowed=False, remaining=0, used=0,
                limit=0, result=QuotaCheckResult.MEMBER_ONLY,
                upgrade_hint=self._build_upgrade_hint(tier, quota_type)
            )

        # 4. 构造 Redis Key
        window = limit_config.window
        window_key = self._window_key(window)
        counter_key = f"quota:{user_id}:{quota_type.value}:{window.value}:{window_key}"
        topup_key = f"topup:{user_id}:{quota_type.value}"

        # 5. 原子检查并扣减
        lua_script = self._load_lua("quota_check_and_consume.lua")
        result = await self.redis.eval(
            lua_script, 2,
            counter_key, topup_key,
            str(limit_config.limit_value),
            str(int(cost_weight * 100)),  # 转整数避免浮点问题
            str(int(datetime.now().timestamp()))
        )
        allowed, remaining, used, topup_remaining = result

        # 6. 异步持久化审计日志
        await self._audit_log(
            user_id=user_id,
            quota_type=quota_type,
            action="consume" if allowed else "reject",
            result=QuotaCheckResult.ALLOWED if allowed else QuotaCheckResult.EXCEEDED,
            cost_weight=cost_weight,
            membership_tier=tier,
            window_key=window_key,
            used_count=used,
            limit_value=limit_config.limit_value,
            request_id=request_id,
        )

        response = QuotaCheckResponse(
            allowed=bool(allowed),
            remaining=remaining,
            used=used,
            limit=limit_config.limit_value,
            result=QuotaCheckResult.ALLOWED if allowed else QuotaCheckResult.EXCEEDED,
        )

        # 7. 超限降级处理
        if not allowed and limit_config.overage_policy == "degrade":
            degrade_rule = await self.config.get_degrade_rule(tier, quota_type)
            response.degrade_info = {
                "degrade_model": degrade_rule.downgrade_model,
                "message": degrade_rule.degrade_message,
                "throttle_rps": degrade_rule.throttle_rps,
            }
            response.result = QuotaCheckResult.ALLOWED  # 允许但降级
            response.allowed = True

        if not allowed and limit_config.overage_policy == "pay":
            response.upgrade_hint = self._build_upgrade_hint(tier, quota_type)

        return response

    async def refund(
        self,
        user_id: str,
        quota_type: QuotaType,
        cost_weight: float = 1.0,
        request_id: Optional[str] = None,
    ) -> None:
        """退还额度（用于AI调用失败等场景）"""
        tier = await self.membership.get_effective_tier(user_id)
        limit_config = await self.config.get_quota_limit(tier, quota_type)
        if limit_config is None or limit_config.limit_value == -1:
            return  # 无限额度无需退还

        window = limit_config.window
        window_key = self._window_key(window)
        counter_key = f"quota:{user_id}:{quota_type.value}:{window.value}:{window_key}"

        await self.redis.incrby(counter_key, -int(cost_weight * 100))

        await self._audit_log(
            user_id=user_id,
            quota_type=quota_type,
            action="refund",
            result=QuotaCheckResult.ALLOWED,
            cost_weight=cost_weight,
            membership_tier=tier,
            window_key=window_key,
            request_id=request_id,
        )

    async def get_usage_summary(self, user_id: str) -> dict:
        """获取用户当前额度使用概览（供客户端展示）"""
        tier = await self.membership.get_effective_tier(user_id)
        configs = await self.config.get_all_quota_limits(tier)
        today = self._window_key(QuotaWindow.DAILY)

        summary = {}
        for quota_type, limit_cfg in configs.items():
            counter_key = f"quota:{user_id}:{quota_type.value}:{limit_cfg.window.value}:{self._window_key(limit_cfg.window)}"
            topup_key = f"topup:{user_id}:{quota_type.value}"

            used = int(await self.redis.get(counter_key) or 0) // 100
            topup = await self.redis.get(topup_key)
            topup_val = int(topup) if topup else None

            summary[quota_type.value] = {
                "used": used,
                "limit": limit_cfg.limit_value,
                "remaining": max(0, limit_cfg.limit_value - used) if limit_cfg.limit_value > 0 else -1,
                "topup_remaining": topup_val,
                "window": limit_cfg.window.value,
                "resets_at": self._next_reset_time(limit_cfg.window),
            }

        return {"tier": tier, "quotas": summary}

    # ---- 私有方法 ----

    def _window_key(self, window: QuotaWindow) -> str:
        now = datetime.now()
        if window == QuotaWindow.DAILY:
            return now.strftime("%Y-%m-%d")
        elif window == QuotaWindow.WEEKLY:
            return f"{now.strftime('%Y')}-W{now.isocalendar()[1]:02d}"
        elif window == QuotaWindow.MONTHLY:
            return now.strftime("%Y-%m")
        return "total"

    def _next_reset_time(self, window: QuotaWindow) -> str:
        now = datetime.now()
        if window == QuotaWindow.DAILY:
            return (now.replace(hour=0, minute=0, second=0, microsecond=0)
                    .replace(day=now.day + 1)).isoformat()
        elif window == QuotaWindow.WEEKLY:
            # 下周一
            days_ahead = 7 - now.weekday()
            return (now + timedelta(days=days_ahead)).replace(
                hour=0, minute=0, second=0, microsecond=0).isoformat()
        elif window == QuotaWindow.MONTHLY:
            return now.replace(month=now.month + 1, day=1,
                             hour=0, minute=0, second=0, microsecond=0).isoformat()
        return None

    def _build_upgrade_hint(self, current_tier: str, quota_type: QuotaType) -> str:
        """根据超额类型生成升级提示"""
        hints = {
            (QuotaType.AI_CHAT, "free"): "升级月度会员，每日可进行更多AI对话",
            (QuotaType.PHOTO_SEARCH, "free"): "升级月度会员，解锁更多拍照搜题次数",
            (QuotaType.ESSAY_CORRECT, "free"): "作文批改为会员专属功能",
            (QuotaType.EXAM_SIM, "monthly"): "升级年度会员，解锁更多模拟考试次数",
        }
        return hints.get((quota_type, current_tier), "升级会员获取更多使用次数")

    def _load_lua(self, name: str) -> str:
```

### 4.2 FeatureGateService — 功能门控服务

```python
# services/feature_gate_service.py

class FeatureGateService:
    """
    功能门控服务
    负责：功能开关检查、学段适配、灰度控制
    """

    # 功能 → 最低会员等级要求
    FEATURE_MIN_TIER = {
        "ai_chat_basic": "free",           # 基础AI对话（限次）
        "ai_chat_unlimited": "monthly",    # 无限AI对话
        "photo_search": "free",            # 拍题（限次）
        "essay_correct": "monthly",        # 作文批改
        "essay_deep_polish": "yearly",     # 深度润色
        "exam_simulation": "monthly",      # 考试模拟
        "exam_real_paper": "exam_prep",    # 真题试卷
        "knowledge_graph": "free",         # 知识图谱浏览
        "study_plan_advanced": "monthly",  # 高级学习规划
        "parent_report": "monthly",        # 家长学情报告
        "export_pdf": "free",              # PDF导出（限次）
        "offline_download": "yearly",      # 离线内容下载
        "voice_coach": "monthly",          # 口语陪练
    }

    # 功能 → 适用学段
    FEATURE_STAGE_SCOPE = {
        "pinyin_literacy": ["preschool", "primary_low"],
        "exam_simulation": ["junior", "senior"],
        "college_entrance_sprint": ["senior"],
        "voice_coach": ["primary_high", "junior", "senior"],
        "parent_report": ["preschool", "primary_low", "primary_high", "junior", "senior"],
    }

    def __init__(
        self,
        redis: Redis,
        config_service: ConfigService,
        membership_service: MembershipService,
    ):
        self.redis = redis
        self.config = config_service
        self.membership = membership_service

    async def check_access(
        self,
        user_id: str,
        feature: str,
        context: Optional[dict] = None,
    ) -> FeatureGateResult:
        """
        检查用户是否有权访问某功能

        Args:
            user_id: 用户ID
            feature: 功能标识
            context: 附加上下文 {stage, grade, ...}

        Returns:
            FeatureGateResult {allowed, reason, message}
        """
        # 1. 灰度开关检查
        enabled = await self.config.is_feature_enabled(feature)
        if not enabled:
            return FeatureGateResult(
                allowed=False,
                reason="feature_disabled",
                message="该功能暂未开放，敬请期待"
            )

        # 2. 灰度百分比检查
        if not await self._check_rollout(feature, user_id):
            return FeatureGateResult(
                allowed=False,
                reason="rollout_pending",
                message="该功能正在灰度开放中，即将全量上线"
            )

        # 3. 会员等级检查
        min_tier = await self.config.get_feature_min_tier(feature)
        if min_tier is None:
            min_tier = self.FEATURE_MIN_TIER.get(feature, "free")

        user_tier = await self.membership.get_effective_tier(user_id)
        if not self._tier_gte(user_tier, min_tier):
            return FeatureGateResult(
                allowed=False,
                reason="member_only",
                message=f"该功能需要{self._tier_display_name(min_tier)}及以上",
                required_tier=min_tier,
                current_tier=user_tier,
            )

        # 4. 学段适配检查
        stage_scope = self.FEATURE_STAGE_SCOPE.get(feature)
        if stage_scope and context:
            user_stage = context.get("stage")
            if user_stage and user_stage not in stage_scope:
                return FeatureGateResult(
                    allowed=False,
                    reason="stage_not_supported",
                    message="该功能当前不适用于您的学段"
                )

        # 5. 特殊场景检查（如未成年人消费限制）
        if context and context.get("is_minor"):
            age = context.get("age", 0)
            if feature in ("exam_real_paper",) and age < 12:
                return FeatureGateResult(
                    allowed=False,
                    reason="age_restricted",
                    message="该功能需要家长授权后使用"
                )

        return FeatureGateResult(allowed=True, reason="ok", message="")

    async def _check_rollout(self, feature: str, user_id: str) -> bool:
        """灰度百分比检查（基于用户ID哈希）"""
        rollout_pct = await self.config.get_feature_rollout(feature)
        if rollout_pct >= 100:
            return True
        # 稳定哈希：同一用户永远落到同一桶
        bucket = int(hashlib.md5(f"{feature}:{user_id}".encode()).hexdigest(), 16) % 100
        return bucket < rollout_pct

    @staticmethod
    def _tier_gte(user_tier: str, required_tier: str) -> bool:
        """会员等级比较"""
        tier_order = {"free": 0, "monthly": 1, "yearly": 2, "exam_prep": 2}
        return tier_order.get(user_tier, 0) >= tier_order.get(required_tier, 0)

    @staticmethod
    def _tier_display_name(tier: str) -> str:
        names = {"free": "免费用户", "monthly": "月度会员", "yearly": "年度会员", "exam_prep": "备考专项会员"}
        return names.get(tier, tier)


@dataclass
class FeatureGateResult:
    allowed: bool
    reason: str       # ok / member_only / feature_disabled / rollout_pending / stage_not_supported / age_restricted
    message: str
    required_tier: Optional[str] = None
    current_tier: Optional[str] = None
```

```python
# middleware/quota_decorator.py

from functools import wraps
from fastapi import Request, HTTPException


def require_quota(quota_type: QuotaType, cost_weight: float = 1.0):
    """
    额度检查装饰器，用于需要额度控制的 API 接口

    用法:
        @router.post("/api/v1/ai/chat")
        @require_quota(QuotaType.AI_CHAT, cost_weight=1.0)
        async def chat(request: Request, ...):
            ...
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(request: Request, *args, **kwargs):
            quota_service: QuotaService = request.app.state.quota_service
            user_id = request.state.user_id  # 由鉴权中间件注入

            result = await quota_service.check_and_consume(
                user_id=user_id,
                quota_type=quota_type,
                cost_weight=cost_weight,
                request_id=request.state.request_id,
            )

            if not result.allowed:
                raise HTTPException(
                    status_code=429,
                    detail={
                        "code": "QUOTA_EXCEEDED",
                        "message": "今日使用次数已达上限",
                        "quota_type": quota_type.value,
                        "used": result.used,
                        "limit": result.limit,
                        "remaining": result.remaining,
                        "upgrade_hint": result.upgrade_hint,
                        "degrade_info": result.degrade_info,
                        "resets_at": None,  # 由客户端根据 window 自行计算
                    }
                )

            # 将额度信息注入 request state，供业务逻辑使用
            request.state.quota_result = result

            # 如果有降级信息，透传给业务层
            if result.degrade_info:
                request.state.model_override = result.degrade_info.get("degrade_model")
                request.state.throttle_rps = result.degrade_info.get("throttle_rps")

            return await func(request, *args, **kwargs)
        return wrapper
    return decorator


def require_feature(feature: str):
    """
    功能门控装饰器

    用法:
        @router.post("/api/v1/essay/correct")
        @require_feature("essay_correct")
        async def correct_essay(request: Request, ...):
            ...
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(request: Request, *args, **kwargs):
            gate_service: FeatureGateService = request.app.state.feature_gate_service
            user_id = request.state.user_id

            # 构建上下文
            student_profile = request.state.student_profile  # 由鉴权中间件注入
            context = {
                "stage": student_profile.stage if student_profile else None,
                "grade": student_profile.grade if student_profile else None,
                "is_minor": student_profile.is_minor if student_profile else False,
                "age": student_profile.age if student_profile else None,
            }

            result = await gate_service.check_access(
                user_id=user_id,
                feature=feature,
                context=context,
            )

            if not result.allowed:
                status_code = 403 if result.reason == "member_only" else 404
                raise HTTPException(
                    status_code=status_code,
                    detail={
                        "code": f"FEATURE_{result.reason.upper()}",
                        "message": result.message,
                        "required_tier": result.required_tier,
                        "current_tier": result.current_tier,
                    }
                )

            return await func(request, *args, **kwargs)
        return wrapper
    return decorator
```

```python
# middleware/quota_refund.py

class QuotaRefundMiddleware:
    """
    AI调用失败时自动退还额度
    """

    async def __call__(self, request: Request, call_next):
        response = await call_next(request)

        # 只处理被额度装饰器标记的请求
        quota_result = getattr(request.state, "quota_result", None)
        if quota_result is None:
            return response

        # 5xx 错误或 AI 调用超时 → 退还额度
        if response.status_code >= 500:
            quota_type = request.state.consumed_quota_type
            cost_weight = request.state.consumed_cost_weight
            await request.app.state.quota_service.refund(
                user_id=request.state.user_id,
                quota_type=quota_type,
                cost_weight=cost_weight,
                request_id=request.state.request_id,
            )
            logger.info(f"额度已退还: user={request.state.user_id} type={quota_type}")

        return response
```

## 8. 状态流转

### 8.1 额度检查主流程

```
请求进入
    │
    ▼
┌─────────────────┐
│ 会员等级查询     │ ← Redis缓存 + DB回源 (TTL=5min)
└────────┬────────┘
         │
    tier已知
         │
         ▼
┌─────────────────┐
│ 功能门控检查     │ ← 灰度开关 + 学段 + 会员等级
└────────┬────────┘
         │
    通过？
    ├── NO → 403/404 返回限制原因
    │
    YES
         ▼
┌─────────────────┐
│ 额度上限查询     │ ← 配置中心 (Redis缓存)
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ 原子扣减 (Lua)   │ ← Redis + Lua脚本
│ 优先扣额度包     │
└────────┬────────┘
         │
    成功？
    ├── YES → 放行 → 业务处理 → 完成后记录审计
    │
    NO（超限）
         │
         ▼
┌─────────────────┐
│ 超限策略判断     │
├─────────────────┤
│ block   → 429   │
│ degrade → 放行但降级模型 │
│ pay     → 429 + 升级引导 │
└─────────────────┘
```

### 8.2 会员等级变更时额度重置

```
会员升级/降级事件
    │
    ▼
清除用户会员等级缓存 (user:membership:{user_id})
    │
    ▼
下次请求自动获取新等级
（不重置已用额度，仅改变上限判定）
```

### 8.3 额度包生命周期

```
购买额度包
    │
    ▼
写入 quota_topup 表 + Redis topup:{user_id}:{type}
设置 TTL = expires_at
    │
    ▼
使用时优先扣 topup key
    │
    ▼
topup 耗尽 → 回退到会员额度

    ▼
```

## 9. 错误处理

### 9.1 错误码

| 错误码 | HTTP Status | 说明 |
|-------|-------------|------|
| QUOTA_EXCEEDED | 429 | 额度已耗尽 |
| QUOTA_MEMBER_ONLY | 403 | 需要更高会员等级 |
| QUOTA_FEATURE_DISABLED | 404 | 功能未开放 |
| QUOTA_ROLLOUT_PENDING | 404 | 功能灰度中 |
| QUOTA_STAGE_NOT_SUPPORTED | 403 | 学段不适用 |
| QUOTA_AGE_RESTRICTED | 403 | 年龄限制 |
| QUOTA_TOPUP_EXPIRED | 410 | 额度包已过期 |
| QUOTA_INTERNAL_ERROR | 500 | 额度服务内部错误 |

### 9.2 降级策略

| 故障场景 | 降级方案 |
|---------|---------|
| Redis 不可用 | 切换到 DB 计数（SELECT ... FOR UPDATE），降级日志记录 |
| 配置中心不可用 | 使用内存默认配置（Free=5次/日 AI对话等） |
| 会员服务不可用 | 按 Free 等级处理（宁限勿放） |
| Lua 脚本执行失败 | 回退到 GET + INCRBY 两步操作（非原子，可容忍短暂超扣） |
| 审计日志写入失败 | 异步重试队列，最多重试3次，超限则丢弃（审计丢失优于业务阻塞） |

```python
# QuotaService 中 Redis 故障的兜底逻辑

class QuotaService:
    async def check_and_consume(self, user_id, quota_type, cost_weight=1.0, request_id=None):
        try:
            # 正常 Redis 路径
            return await self._redis_check_and_consume(...)
        except RedisError as e:
            logger.error(f"Redis额度服务异常: {e}", exc_info=True)

            # 降级到数据库路径
            try:
                return await self._db_check_and_consume(user_id, quota_type, cost_weight)
            except Exception as db_err:
                logger.error(f"DB额度服务降级失败: {db_err}", exc_info=True)
                # 最终兜底：按 Free 等级 + 保守限制
                return QuotaCheckResponse(
                    allowed=True,  # 放行避免阻断用户
                    remaining=0,
                    used=0,
                    limit=5,
                    result=QuotaCheckResult.ALLOWED,
                    degrade_info={"message": "服务波动中，部分功能可能受限"},
                )

    async def _db_check_and_consume(self, user_id, quota_type, cost_weight):
        """数据库降级路径：行锁保证原子性"""
        async with self.db.begin():
            row = await self.db.execute(
                select(QuotaUsage)
                .where(
                    QuotaUsage.user_id == user_id,
                    QuotaUsage.quota_type == quota_type.value,
                    QuotaUsage.window_key == self._window_key(QuotaWindow.DAILY),
                )
                .with_for_update()
            )
            usage = row.scalar_one_or_none()
            if usage is None:
                usage = QuotaUsage(
                    user_id=user_id,
                    quota_type=quota_type.value,
                    window=QuotaWindow.DAILY.value,
                    window_key=self._window_key(QuotaWindow.DAILY),
                    used_count=0,
                )
                self.db.add(usage)

            limit_config = await self.config.get_quota_limit("free", quota_type)
            if limit_config and limit_config.limit_value > 0:
                if usage.used_count + cost_weight > limit_config.limit_value:
                    return QuotaCheckResponse(
                        allowed=False,
                        remaining=max(0, limit_config.limit_value - usage.used_count),
                        used=usage.used_count,
                        limit=limit_config.limit_value,
                        result=QuotaCheckResult.EXCEEDED,
                    )

            usage.used_count += cost_weight
            return QuotaCheckResponse(
                allowed=True,
                remaining=-1,
                used=usage.used_count,
                limit=limit_config.limit_value if limit_config else -1,
                result=QuotaCheckResult.ALLOWED,
            )
```

```python
# tasks/quota_reconcile.py

class QuotaReconcileTask:
    """
    每日凌晨执行，将 Redis 计数器持久化到 MySQL quota_usage 表
    用于对账和冷启动恢复
    """

    async def run(self):
        yesterday = (date.today() - timedelta(days=1)).strftime("%Y-%m-%d")

        # 扫描所有昨日活跃的额度 key
        pattern = f"quota:*:*:{yesterday}"
        cursor = 0
        while True:
            cursor, keys = await self.redis.scan(cursor, match=pattern, count=1000)
            for key in keys:
                # 解析 key: quota:{user_id}:{quota_type}:{window}:{window_key}
                parts = key.split(":")
                if len(parts) != 5:
                    continue
                user_id, quota_type, window, window_key = parts[1], parts[2], parts[3], parts[4]
                used = int(await self.redis.get(key) or 0) // 100

                # Upsert 到 MySQL
                await self.db.execute(
                    insert(QuotaUsage)
                    .values(
                        user_id=user_id,
                        quota_type=quota_type,
                        window=window,
                        window_key=window_key,
                        used_count=used,
                    )
                    .on_conflict_do_update(
                        index_elements=["user_id", "quota_type", "window", "window_key"],
                        set_{"used_count": used, "updated_at": func.now()},
                    )
                )

            if cursor == 0:
                break

        logger.info(f"额度对账完成: {yesterday}")
```

```dart
  }

  /// 获取某额度类型的剩余次数（-1=无限）
  int getRemaining(QuotaType type) {
    final quota = state.value?.quotas[type.jsonKey];
    if (quota == null) return -1;
    final remaining = quota['remaining'] as int;
    final topup = quota['topup_remaining'] as int?;
    if (topup != null && topup > 0) return remaining + topup;
    return remaining;
  }

  /// 本地乐观扣减（发送对话后立即更新UI，不等服务端确认）
  void consumeLocally(QuotaType type, double costWeight) {
    final current = state.value;
    if (current == null) return;
    final quota = current.quotas[type.jsonKey];
    if (quota == null) return;
    quota['used'] = (quota['used'] as int) + costWeight.toInt();
    state = AsyncData(current);
  }
}
```
