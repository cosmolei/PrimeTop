> 版本：1.0 | 日期：2026-05-23 | 状态：已完成  
> 优先级：P1（V1.0 阶段核心输入体验）

## 1. 模块概述

### 1.1 功能定位

为学生提供便捷的学科特殊内容输入能力，覆盖数学公式、化学方程式、物理符号、希腊字母等跨学科符号的高效输入。支持键盘快捷输入、手写识别、语音输入三种模式，并与 AI 对话、练习答题、错题手动录入等场景深度集成。

### 1.2 核心问题

| 痛点 | 说明 |
|------|------|
| 移动端输入困难 | 手机键盘无法直接输入上标、下标、分式、根号、积分号等数学符号 |
| 学生年龄跨度大 | 幼儿/小学生需要更简单的输入方式，高中生需要更丰富的符号集 |
| 学科符号差异大 | 数学（公式）、化学（方程式/元素）、物理（单位/矢量）各有特殊需求 |
| 手写识别准确率 | 学生手写（尤其低龄）规范性差，识别难度高 |
| 混合内容输入 | 一段文字中混合普通文本和公式（如"求 x²+2x-3=0 的解"） |

### 1.3 设计目标

| 目标 | 指标 |
|------|------|
| 符号覆盖 | 覆盖小学至高中全学段常用数学/物理/化学符号 ≥ 800 个 |
| 输入速度 | 熟练用户输入常见公式速度 ≥ 纸笔书写速度的 60% |
| 手写识别准确率 | 印刷体 ≥ 95%，规范手写 ≥ 85%，低龄手写 ≥ 70% |
| 集成无感 | 在 AI 对话、练习等场景中一键唤起，≤ 2 次点击完成输入 |
| 离线可用 | 键盘模式完全离线可用，手写模式核心符号集离线可用 |

### 1.4 覆盖学段

| 学段 | 核心输入需求 | 推荐输入方式 |
|------|-------------|-------------|
| 小学 | 四则运算、分数、简单方程、面积/体积公式 | 符号键盘 + 语音 |
| 初中 | 一元二次方程、函数、三角函数、化学方程式 | 符号键盘 + 手写 |
| 高中 | 导数/积分、概率统计、向量、复杂化学方程式 | 手写 + LaTeX 快捷 + 符号键盘 |

#### L1 本地模型

```typescript
// 本地模型配置
interface LocalModelConfig {
  modelPath: string;          // 'assets/models/math_ocr_v3.tflite'
  labelPath: string;          // 'assets/models/math_symbols_labels.json'
  inputSize: [number, number]; // [224, 224]
  maxStrokes: number;         // 30
  supportedSymbols: string[]; // 约 200 个常用符号
  confidenceThreshold: number; // 0.6，低于此值回退到云端
}

// 笔画预处理管线
class StrokePreprocessor {
  /**
   * 1. 重采样：等间距采样点（20个点/笔画）
   * 2. 归一化：缩放到 [0,1]×[0,1]，中心化
   * 3. 平滑：Bezier 曲线平滑降噪
   * 4. 组装：多笔画组合为单一特征向量
   */
  preprocess(strokes: Stroke[]): Float32Array {
    const resampled = strokes.map(s => this.resample(s, 20));
    const normalized = this.normalize(resampled);
    const smoothed = this.smooth(normalized);
    return this.toFeatureVector(smoothed);
  }
  
  private resample(stroke: Stroke, n: number): Point[] {
    // 等弧长重采样
    const totalLen = stroke.totalLength();
    const step = totalLen / (n - 1);
    const result: Point[] = [stroke.points[0]];
    let accum = 0;
    for (let i = 1; i < stroke.points.length; i++) {
      accum += stroke.points[i].distanceTo(stroke.points[i-1]);
      while (accum >= step) {
        result.push(stroke.interpolate(i, accum - step));
        accum -= step;
      }
    }
    return result.slice(0, n);
  }
}
```

```typescript
// 云端识别服务抽象
interface CloudOCRProvider {

}

interface OCRContext {
  subject: 'math' | 'chemistry' | 'physics';
  stage: 'primary' | 'junior' | 'senior';
  previousText?: string;   // 前文内容（帮助上下文识别）
}

// 多供应商降级策略
class CloudOCRService {
  private providers: CloudOCRProvider[] = [
    new MyScriptProvider(),    // 主力：MyScript Math SDK
    new MathpixProvider(),     // 备用：Mathpix API
    new InternalAIProvider(),  // 兜底：自研多模态模型
  ];
  
  async recognize(strokes: Stroke[], context?: OCRContext): Promise<OCRResult> {
    for (const provider of this.providers) {
      try {
        const result = await provider.recognize(strokes, context);
        if (result.candidates[0].confidence > 0.5) {
          return result;
        }
      } catch (e) {
        logger.warn(`OCR provider failed`, { provider: provider.name, error: e });
        continue;
      }
    }
    // 全部失败 → 返回低置信度结果
    return { candidates: [], fallback: true };
  }
}
```

### 5.2 语音解析引擎

```typescript
class VoiceMathParser {
  /**
   * 将 ASR 文本解析为 LaTeX 表达式
   * 
   * 示例：
   * "x 的平方加 2x 减 3 等于 0" → "x^2 + 2x - 3 = 0"
   * "根号下 b 方减 4ac" → "\\sqrt{b^2 - 4ac}"
   * "二分之根号下..." → "\\frac{}{2}\\sqrt{...}"
   */
  parse(asrText: string): ParsedMathResult {
    // 1. 预处理：去除语气词
    let cleaned = this.removeFillerWords(asrText);
    
    // 2. 按优先级匹配语音数学规则
    const tokens = this.tokenize(cleaned);
    
    // 3. 构建语法树
    const ast = this.buildAST(tokens);
    
    // 4. 转换为 LaTeX
    const latex = this.astToLatex(ast);
    
    return {
      originalText: asrText,
      latex,
      confidence: this.calculateConfidence(tokens),
      ambiguous: this.checkAmbiguity(tokens),
    };
  }
  
  private removeFillerWords(text: string): string {
    return text.replace(/(嗯|那个|就是|然后|呃|额)/g, '');
  }
}

interface ParsedMathResult {
  originalText: string;
  latex: string;
  confidence: number;
  ambiguous: boolean;    // 是否存在歧义需要确认
}
```

## 6. 中间表示层 (IR)

### 6.1 内部数学表达式树

所有输入模式最终转换为统一的数学表达式树（AST），再输出为 LaTeX：

```typescript
// 数学表达式 AST 节点类型
type MathNode =
  | NumberNode
  | VariableNode
  | BinaryOpNode
  | UnaryOpNode
  | FunctionNode
  | FractionNode
  | RadicalNode
  | SuperscriptNode
  | SubscriptNode
  | MatrixNode
  | IntegralNode
  | SumNode
  | ChemicalNode;

interface NumberNode {
  type: 'number';
  value: string;   // '3.14', '-5', '1/2'
}

interface VariableNode {
  type: 'variable';
  name: string;     // 'x', 'θ', 'α'
}

interface BinaryOpNode {
  type: 'binary_op';
  operator: '+' | '-' | '*' | '/' | '=' | '!=' | '<' | '>' | '<=' | '>=';
  left: MathNode;
  right: MathNode;
}

interface FractionNode {
  type: 'fraction';
  numerator: MathNode;
  denominator: MathNode;
}

interface RadicalNode {
  type: 'radical';
  index: MathNode | null;   // null = 平方根
  radicand: MathNode;
}

interface SuperscriptNode {
  type: 'superscript';
  base: MathNode;
  exponent: MathNode;
}

interface SubscriptNode {
  type: 'subscript';
  base: MathNode;
  subscript: MathNode;
}

interface FunctionNode {
  type: 'function';
  name: string;    // 'sin', 'cos', 'log', 'ln', 'lim'
  arguments: MathNode[];
}

interface ChemicalNode {
  type: 'chemical';
  reactants: ChemicalTerm[];
  products: ChemicalTerm[];
  reversible: boolean;
  conditions?: string[];   // 'Δ', '催化剂', '高温高压'
}

interface ChemicalTerm {
  coefficient: number;
  formula: string;      // 'H2O', 'CO2'
  state?: string;       // '↑', '↓', '(g)', '(l)', '(s)', '(aq)'
}

// LaTeX 生成器
class LatexGenerator {
  generate(node: MathNode): string {
    switch (node.type) {
      case 'number':
        return node.value;
      case 'variable':
        return this.escapeVariable(node.name);
      case 'binary_op':
        return `${this.generate(node.left)} ${this.opToLatex(node.operator)} ${this.generate(node.right)}`;
      case 'fraction':
        return `\\frac{${this.generate(node.numerator)}}{${this.generate(node.denominator)}}`;
      case 'radical':
        if (node.index === null) return `\\sqrt{${this.generate(node.radicand)}}`;
        return `\\sqrt[${this.generate(node.index)}]{${this.generate(node.radicand)}}`;
      case 'superscript':
        return `${this.generate(node.base)}^{${this.generate(node.exponent)}}`;
      case 'subscript':
        return `${this.generate(node.base)}_{${this.generate(node.subscript)}}`;
      case 'function':
        return `\\${node.name}${node.arguments.length > 1
          ? `(${node.arguments.map(a => this.generate(a)).join(', ')})`
          : `{${this.generate(node.arguments[0])}}`
        }`;
      case 'chemical':
        return this.chemicalToLatex(node);
      default:
        throw new Error(`Unknown node type: ${(node as any).type}`);
    }
  }
}
```

```dart
/// 公式输入管理器 - 管理三种输入模式的切换和数据流
class FormulaInputController extends ChangeNotifier {
  InputMode _currentMode = InputMode.symbolKeyboard;
  final TextEditingController _textController = TextEditingController();
  final LatexGenerator _latexGenerator = LatexGenerator();
  final MathASTBuilder _astBuilder = MathASTBuilder();
  
  /// 当前 LaTeX 内容
  String get latex => _textController.text;
  
  /// 当前渲染预览 HTML
  String get previewHtml => _renderLatex(_textController.text);
  
  /// AST 根节点（用于高级操作）
  MathNode? _astRoot;
  
  /// 切换输入模式
  void switchMode(InputMode mode) {
    _currentMode = mode;
    notifyListeners();
  }
  
  /// 插入符号（来自符号键盘）
  void insertSymbol(SymbolAction action) {
    final selection = _textController.selection;
    final text = _textController.text;
    
    final newText = text.replaceRange(
      selection.start, selection.end, action.latexTemplate,
    );
    _textController.text = newText;
    
    // 设置光标到占位符位置
    final cursorPos = selection.start + action.latexTemplate.length + action.cursorOffset;
    _textController.selection = TextSelection.collapsed(offset: cursorPos);
    
    _rebuildAST();
    notifyListeners();
  }
  
  /// 插入手写识别结果
  void insertHandwritingResult(RecognitionResult result) {
    if (result.candidates.isEmpty) return;
    final best = result.candidates.first;
    _insertLatex(best.latex);
  }
  
  /// 插入语音解析结果
  void insertVoiceResult(ParsedMathResult result) {
    _insertLatex(result.latex);
  }
  
  void _insertLatex(String latex) {
    final selection = _textController.selection;
    final text = _textController.text;
    _textController.text = text.replaceRange(selection.start, selection.end, latex);
    _textController.selection = TextSelection.collapsed(
      offset: selection.start + latex.length,
    );
    _rebuildAST();
    notifyListeners();
  }
  
  /// 重新构建 AST（用于智能补全和校验）
  void _rebuildAST() {
    try {
      _astRoot = _astBuilder.build(_textController.text);
    } catch (_) {
      _astRoot = null; // 语法不完整时不阻塞输入
    }
  }
  
  String _renderLatex(String latex) {
    try {
      return KatexRenderer.render(latex);
    } catch (_) {
      return '<span class="formula-raw">$latex</span>';
    }
  }
}

enum InputMode {
  symbolKeyboard,  // 符号键盘
  handwriting,     // 手写识别
  voice,           // 语音输入
}
```

```dart
        top: offset.dy + size.height,
        width: size.width,
```

| 场景 | 处理策略 | 用户提示 |

| 自定义片段超限 | 拒绝创建并提示 | "自定义公式已达上限(50个)" |
