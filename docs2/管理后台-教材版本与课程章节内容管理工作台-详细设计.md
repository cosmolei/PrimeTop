# 管理后台 - 教材版本与课程章节内容管理工作台 详细设计

## 1. 概述

### 1.1 模块定位

教材版本与课程章节内容管理工作台是 PrimeTop 管理后台的核心 P0 模块，面向内容运营人员和教研团队，提供教材版本的全生命周期管理、章节目录结构维护、知识点关联配置、内容挂载与预览等可视化操作能力。

该工作台是教材数据进入平台的"入口"——所有学生端同步课堂、题库匹配、学习路径推荐、考点梳理等功能都依赖于此处维护的教材数据质量。

### 1.2 核心职责

| 职责 | 说明 |
|------|------|
| 教材版本管理 | 创建、编辑、归档、克隆教材版本，管理版本状态流转 |
| 章节目录维护 | 可视化树形编辑器，支持拖拽排序、层级调整、批量导入 |
| 知识点关联 | 为章节挂载/卸载知识点，配置权重与关系类型 |
| 内容挂载管理 | 查看章节关联的课件、练习、视频等资源，快速跳转资源管理 |
| 变更审计 | 记录所有编辑操作，支持版本对比与回滚 |
| 数据校验 | 章节完整性检查、知识点覆盖率检测、空节点预警 |

### 1.3 目标用户

| 角色 | 权限范围 | 典型操作 |
|------|---------|---------|
| 超级管理员 | 全部操作 | 教材版本创建、归档、删除、数据校验 |
| 内容运营 | 版本与章节编辑 | 日常章节维护、知识点关联、批量导入 |
| 教研人员 | 只读 + 知识点编辑 | 审核章节内容、调整知识点映射 |
| 访客（只读） | 仅查看 | 浏览教材结构、导出数据 |

### 4.4 克隆版本功能

克隆操作允许运营人员基于已有版本快速创建新版本（如新学年教材更新）：

```typescript
interface CloneVersionPayload {
  sourceVersionId: string;
  newName: string;           // 新版本名称
  newYear: number;           // 新版本年份
  cloneChapters: boolean;    // 是否克隆章节结构（默认true）
  cloneKnowledgeMappings: boolean; // 是否克隆知识点映射（默认true）
  cloneContentAttachments: boolean; // 是否克隆内容挂载（默认false，内容需重新关联）
}

// 克隆流程
async function handleCloneVersion(payload: CloneVersionPayload) {
  // 1. 创建新版本（状态为DRAFT）
  const newVersion = await api.createVersion({
    ...sourceVersion.basicInfo,
    name: payload.newName,
    year: payload.newYear,
    status: 'DRAFT',
  });
  
  // 2. 克隆章节树（递归复制）
  if (payload.cloneChapters) {
    await api.cloneChapterTree(sourceVersion.id, newVersion.id, {
      cloneKnowledgeMappings: payload.cloneKnowledgeMappings,
    });
  }
  
  // 3. 返回新版本编辑页
  navigate(`/admin/textbook/versions/${newVersion.id}/edit`);
}
```

### 5.4 拖拽排序逻辑

```typescript
interface DragInfo {
  dragNode: ChapterTreeNode;  // 被拖拽的节点
  dropNode: ChapterTreeNode;  // 目标节点
  dropPosition: number;       // -1: 前方插入, 0: 成为子节点, 1: 后方插入
}

async function handleDrop({ dragNode, dropNode, dropPosition }: DragInfo) {
  // 校验1: 不允许将学期节点拖入课时节点下（层级约束）
  const targetType = dropNode.type;
  const sourceType = dragNode.type;
  if (!isValidHierarchy(sourceType, targetType, dropPosition)) {
    message.error(`${getTypeName(sourceType)}不能作为${getTypeName(targetType)}的子节点`);
    return;
  }

  // 校验2: 不允许跨版本拖拽
  if (dragNode.versionId !== dropNode.versionId) {
    message.error('不允许跨版本拖拽');
    return;
  }

  // 校验3: 拖入目标成为子节点时，检查层级深度不超过4层
  if (dropPosition === 0) {
    const targetDepth = await getNodeDepth(dropNode.key);
    const sourceDepth = await getSubtreeMaxDepth(dragNode.key);
    if (targetDepth + sourceDepth > 4) {
      message.error('章节层级最多4层');
      return;
    }
  }

  // 调用后端更新
  await api.moveChapter({
    chapterId: dragNode.key,
    newParentId: dropPosition === 0 ? dropNode.key : dropNode.parentId,
    newOrder: dropPosition === 0 ? (dropNode.children?.length ?? 0) : dropNode.order + (dropPosition > 0 ? 1 : 0),
  });

  // 刷新树（React Query 自动重新拉取）

  UNIT: ['SEMESTER'],              // 单元只能作为学期的子节点

function isValidHierarchy(sourceType: string, targetType: string, dropPos: number): boolean {
  if (dropPos !== 0) return true; // 前后插入不改变层级关系
  return HIERARCHY_RULES[sourceType]?.includes(targetType) ?? false;
}
```

### 6.3 前端解析与校验逻辑

```typescript
import * as XLSX from 'xlsx';

interface ParsedRow {
  rowNumber: number;      // Excel行号
  level: number;          // 层级深度（1-4）
  name: string;
  type: string;
  code?: string;
  estimatedHours?: number;
  tags?: string[];
  knowledgePoints?: string[]; // 知识点名称列表（逗号分隔）
  contentSummary?: string;
  // 解析时计算
  parentRowNumber?: number;   // 父节点行号（根据层级缩进计算）
  validationErrors: ValidationError[];
  validationWarnings: ValidationWarning[];
}

async function parseAndValidateExcel(file: File): Promise<{
  rows: ParsedRow[];
  summary: { total: number; valid: number; warnings: number; errors: number };
}> {
  const buffer = await file.arrayBuffer();
  const workbook = XLSX.read(buffer, { type: 'array' });
  const worksheet = workbook.Sheets[workbook.SheetNames[0]];
  const rawRows = XLSX.utils.sheet_to_json<Record<string, any>>(worksheet, { header: 1 });
  
  const parsed: ParsedRow[] = [];
  const levelStack: ParsedRow[] = []; // 维护当前层级路径栈
  
  for (let i = 1; i < rawRows.length; i++) { // 跳过表头
    const row = rawRows[i];
    const level = inferLevel(row[0] as string); // 根据缩进推断层级
    const rowData: ParsedRow = {
      rowNumber: i + 1,
      level,
      name: (row[0] as string)?.trim() ?? '',
      type: row[1]?.trim() ?? '',
      code: row[2]?.trim(),
      estimatedHours: parseFloat(row[3]) || undefined,
      tags: row[4]?.split(',').map(s => s.trim()).filter(Boolean),
      knowledgePoints: row[5]?.split(',').map(s => s.trim()).filter(Boolean),
      contentSummary: row[6]?.trim(),
      validationErrors: [],
      validationWarnings: [],
    };
    
    // 校验必填字段
    if (!rowData.name) {
      rowData.validationErrors.push({ field: 'name', message: '章节名称不能为空' });
    }
    if (!rowData.type || !['SEMESTER', 'UNIT', 'LESSON', 'SECTION'].includes(rowData.type)) {
      rowData.validationErrors.push({ field: 'type', message: `章节类型"${rowData.type}"无效` });
    }
    
    // 校验层级关系
    if (level < 1 || level > 4) {
      rowData.validationErrors.push({ field: 'level', message: '层级深度超出范围(1-4)' });
    }
    
    // 计算父节点
    while (levelStack.length > 0 && levelStack[levelStack.length - 1].level >= level) {
      levelStack.pop();
    }
    rowData.parentRowNumber = levelStack[levelStack.length - 1]?.rowNumber;
    levelStack.push(rowData);
    
    // 软校验：无知识点关联
    if (!rowData.knowledgePoints?.length && level >= 3) {
      rowData.validationWarnings.push({ field: 'knowledgePoints', message: '未关联知识点' });
    }
    
    parsed.push(rowData);
  }
  
  const summary = {
    total: parsed.length,
    valid: parsed.filter(r => r.validationErrors.length === 0).length,
    warnings: parsed.filter(r => r.validationWarnings.length > 0).length,
    errors: parsed.filter(r => r.validationErrors.length > 0).length,
  };
  
  return { rows: parsed, summary };
}

// 根据前导空格数推断层级
function inferLevel(text: string): number {
  const leadingSpaces = text.match(/^\s*/)?.[0].length ?? 0;
  return Math.floor(leadingSpaces / 2) + 1; // 每2个空格代表一级
}
```

```typescript
async function validateBeforePublish(versionId: string): Promise<ValidationResult> {
  const tree = await api.getChapterTree(versionId);
  const errors: string[] = [];
  const warnings: string[] = [];
  
  // 校验1: 至少有一个学期节点
  const semesters = tree.filter(n => n.type === 'SEMESTER');
  if (semesters.length === 0) {
    errors.push('教材版本必须包含至少一个学期节点');
  }
  
  // 校验2: 每个学期下至少有一个单元
  for (const semester of semesters) {
    if (!semester.children?.length) {
      errors.push(`学期"${semester.name}"下没有单元`);
    }
  }
  
  // 校验3: 每个单元下至少有一个课时
  const units = semesters.flatMap(s => s.children ?? []);
  for (const unit of units) {
    if (!unit.children?.length) {
      warnings.push(`单元"${unit.name}"下没有课时`);
    }
  }
  
  // 校验4: 知识点覆盖率 ≥ 80%
  const totalLessons = countNodesByType(tree, 'LESSON');
  const lessonsWithKp = countNodesWithKnowledgePoints(tree, 'LESSON');
  const coverage = totalLessons > 0 ? lessonsWithKp / totalLessons : 0;
  if (coverage < 0.8) {
    warnings.push(`知识点覆盖率为 ${(coverage * 100).toFixed(1)}%，建议达到 80% 以上`);
  }
  
  // 校验5: 章节编码唯一性
  const allCodes = collectAllCodes(tree);
  const duplicateCodes = findDuplicates(allCodes);
  if (duplicateCodes.length > 0) {
    errors.push(`章节编码重复: ${duplicateCodes.join(', ')}`);
  }
  
  return { errors, warnings, coverage, totalChapters: countAllNodes(tree) };
}
```

```typescript
function calculateChapterStatus(node: ChapterTreeNode): 'COMPLETE' | 'PARTIAL' | 'EMPTY' {
  let score = 0;
  
  // 有名称和编码 (+2)
  if (node.name && node.code) score += 2;
  
  // 有内容摘要 (+1)
  if (node.contentSummary) score += 1;
  
  // 有关联知识点 (+3)
  if (node.knowledgePointCount > 0) score += 3;
  
  // 有预估学时 (+1)
  if (node.estimatedHours && node.estimatedHours > 0) score += 1;
  
  // 有标签 (+1)
  if (node.tags && node.tags.length > 0) score += 1;
  
  // 有子节点（非SECTION类型）(+2)
  if (node.type !== 'SECTION' && node.children && node.children.length > 0) score += 2;
  
  if (score >= 7) return 'COMPLETE';
  if (score >= 4) return 'PARTIAL';
  return 'EMPTY';
}
```

### 11.1 前端错误处理策略

```typescript
// 全局错误拦截
const apiErrorHandler = (error: AxiosError) => {
  const status = error.response?.status;
  const code = error.response?.data?.code;
  
  switch (status) {
    case 401:
      message.error('登录已过期，请重新登录');
      navigate('/admin/login');
      break;
    case 403:
      message.error('您没有权限执行此操作');
      break;
    case 409:
      // 版本名称冲突等并发冲突
      if (code === 'VERSION_NAME_DUPLICATE') {
        message.error('版本名称已存在，请修改后重试');
      } else if (code === 'CHAPTER_CODE_DUPLICATE') {
        message.error('章节编码重复，请检查后重试');
      } else {
        message.error('数据冲突，请刷新后重试');
      }
      break;
    case 422:
      // 数据校验失败
      const details = error.response?.data?.details;
      if (Array.isArray(details)) {
        showValidationErrors(details);
      } else {
        message.error(error.response?.data?.message ?? '数据校验失败');
      }
      break;
    case 429:
      message.warning('操作过于频繁，请稍后再试');
      break;
    default:
      message.error('网络异常，请稍后重试');
  }
};

// 操作级错误处理（不影响页面其他部分）
async function safeAction<T>(
  action: () => Promise<T>,
  options: { 
    successMessage?: string;
    errorMessage?: string;
    rollback?: () => void;
  } = {}
): Promise<T | undefined> {
  try {
    const result = await action();
    if (options.successMessage) {
      message.success(options.successMessage);
    }
    return result;
  } catch (error) {
    if (options.errorMessage) {
      message.error(options.errorMessage);
    }
    if (options.rollback) {
      options.rollback();
    }
    return undefined;
  }
}
```

## 17. 设计决策记录

| 决策点 | 选择 | 理由 |
|--------|------|------|
| 树编辑器布局 | 左树右面板（Split Pane） | 符合内容管理系统的常见模式，树形导航和详情编辑可同时查看 |
| 拖拽实现 | Ant Design Tree 原生 + 自定义校验 | 减少自定义代码量，利用框架稳定性 |
| 批量导入格式 | Excel (非JSON/CSV) | 运营人员对Excel更熟悉，支持复杂层级表达 |
| 层级表示 | 缩进（非编号） | 更直观，减少人为编号错误 |
| 状态机限制 | ACTIVE下不允许增删章节 | 保护学生端数据稳定性，避免运行时结构变更导致异常 |
| 克隆功能 | 默认不克隆内容挂载 | 内容挂载可能引用版本特定的资源路径，需人工确认 |
| 预览模式 | 手机框架模拟 | 直观展示学生端实际效果，非开发者也能理解 |
| 知识点选择 | 树浏览+搜索双模式 | 覆盖"知道大概分类"和"知道关键词"两种使用场景 |
