# 客户端 - 题目展示与答题交互组件库 详细设计

> **版本**: v1.0  
> **创建日期**: 2026-05-27  
> **关联模块**: 练习与测评、考试模拟、错题复习、同步课堂练习、互动练习、AI组卷  
> **优先级**: P0（核心共享组件）  
> **所属端**: 客户端（Flutter）

---

## 1. 概述

### 1.1 背景

题目展示与答题交互是 PrimeTop 最高频的用户交互场景之一。以下模块均依赖统一的题目渲染与答题收集能力：

| 依赖模块 | 优先级 | 使用场景 |
| --- | --- | --- |
| 互动练习与即时反馈引擎 | P0 | 实时答题、即时判分、提示引导 |
| 练习与测评系统 | P0 | 课后练习、单元测评、综合练习 |
| 考试模拟与真题练习系统 | P1 | 模拟考试、限时测试、真题训练 |
| 错题复习 | P0 | 错题重做、订正练习 |
| 同步课堂·同步练习 | P0 | 章节配套练习 |
| AI 智能组卷 | P1 | 自适应组卷练习 |
| AI 辅导·生成同类题 | P1 | AI 生成练习题后答题 |

若每个模块各自实现题目渲染逻辑，将导致严重代码重复、渲染不一致、维护成本失控。因此，需要设计一套**统一的题目展示与答题交互组件库**，作为基础设施层供所有上层业务模块调用。

### 1.2 设计目标

1. **统一渲染**：所有题型通过统一组件渲染，保证视觉与交互一致性。
2. **可扩展**：新增题型只需新增渲染器，不修改框架代码。
3. **高性能**：支持百题级试卷流畅滚动与状态管理。
4. **分龄适配**：根据学段自动调整字号、间距、交互复杂度。
5. **无障碍**：支持 TalkBack/VoiceOver 读屏。
6. **离线可用**：答题过程不依赖网络，本地暂存。

enum QuestionType {
  /// 单选题

  readingComprehension('reading_comprehension'),
  /// 听力题
  listening('listening'),
  /// 排序题
  ordering('ordering'),
  /// 连线题/匹配题
  matching('matching'),
  /// 计算题（含步骤分）
  calculation('calculation'),
  /// 画图/标注题
  drawing('drawing'),
  /// 口语题
  oral('oral');

  const QuestionType(this.value);
  final String value;

  static QuestionType fromString(String value) =>
      QuestionType.values.firstWhere(
        (e) => e.value == value,
        orElse: () => QuestionType.singleChoice,
      );

  final Map<String, UserAnswer> _answers = {};

  final Set<String> _bookmarkedIds = {};

  /// 会话状态
  SessionState _state = SessionState.notStarted;

  // ── 公共接口 ──

  /// 当前题目

  /// 当前索引
  int get currentIndex => _currentIndex;

  /// 题目总数
  int get totalCount => _questions.length;

  /// 已答题数
  int get answeredCount => _answers.length;

  /// 会话状态
  SessionState get state => _state;

  /// 获取某题的用户答案
  UserAnswer? getAnswer(String questionId) => _answers[questionId];

  /// 获取某题的判题结果
  JudgeResult? getJudgeResult(String questionId) => _judgeResults[questionId];

  /// 所有题目（含子题平铺后的列表）
  List<QuestionModel> get flatQuestions => _flattenQuestions();

  /// 开始会话
  void start() {
    _state = SessionState.inProgress;
    _timer?.start();
    _restoreLocalCache(); // 恢复上次未完成的答题进度
    notifyListeners();
  }

  /// 提交当前题目答案
  Future<void> submitCurrentAnswer(AnswerPayload payload) async {
    final question = currentQuestion;
    final answer = UserAnswer(
      questionId: question.questionId,
      payload: payload,
      timeSpent: _timer?.currentQuestionTime ?? Duration.zero,
      answeredAt: DateTime.now(),
      isSkipped: payload is SkipPayload,
      isBookmarked: _bookmarkedIds.contains(question.questionId),
      clientId: _generateClientId(),
    );

    _answers[question.questionId] = answer;
    await _localCache.saveAnswer(_sessionKey, answer);

    if (_config.revealImmediately && !(payload is SkipPayload)) {
      // 请求判题
      await _requestJudge(question.questionId);
    }

    notifyListeners();
  }

  /// 跳到下一题
  void next() {
    if (_canGoNext()) {
      _timer?.switchQuestion();
      _currentIndex++;
      notifyListeners();
    }
  }

  /// 跳到上一题
  void previous() {
    if (_config.allowGoBack && _currentIndex > 0) {
      _timer?.switchQuestion();
      _currentIndex--;
      notifyListeners();
    }
  }

  /// 跳到指定题号
  void jumpTo(int index) {
    if (index >= 0 && index < _questions.length) {
      _timer?.switchQuestion();
      _currentIndex = index;
      notifyListeners();
    }
  }

  /// 切换标记/疑问
  void toggleBookmark() {
    final qid = currentQuestion.questionId;
    if (_bookmarkedIds.contains(qid)) {
      _bookmarkedIds.remove(qid);
    } else {
      _bookmarkedIds.add(qid);
    }
    notifyListeners();
  }

  /// 交卷
  Future<SessionResult> finish() async {
    _state = SessionState.finished;
    _timer?.stop();

    // 批量请求未判题的答案

        .toList();

      await _batchJudge(unanswered);
    }

    final result = _calculateResult();
    await _localCache.clearSession(_sessionKey);
    notifyListeners();
    return result;
  }

      totalScore += q.score;
      final result = _judgeResults[q.questionId];
      if (result != null) {
        earnedScore += result.earnedScore;
        if (result.isCorrect) correctCount++;
        else wrongCount++;
      } else {
        skippedCount++;
      }
    }

    return SessionResult(
      totalQuestions: _questions.length,
      answeredQuestions: answeredCount,
      correctCount: correctCount,
      wrongCount: wrongCount,
      skippedCount: skippedCount,
      totalScore: totalScore,
      earnedScore: earnedScore,
      accuracy: answeredCount > 0 ? correctCount / answeredCount : 0,
      totalTime: _timer?.elapsed ?? Duration.zero,
      questionResults: {
        for (final q in _questions)
          q.questionId: _judgeResults[q.questionId],
      },
    );
  }

  String get _sessionKey => '${_userId}_${_questions.hashCode}';
}

enum SessionState { notStarted, inProgress, paused, finished }

  final int skippedCount;
  final double totalScore;
  final double earnedScore;

class FillBlankCollector implements AnswerCollector {
  final UiConfig uiConfig;
  final Map<String, TextEditingController> _controllers = {};

  FillBlankCollector({required this.uiConfig});

  @override
  AnswerPayload? getPayload() {
    final slotAnswers = <String, String>{};
    for (final entry in _controllers.entries) {
      slotAnswers[entry.key] = entry.value.text.trim();
    }
    return FillBlankPayload(slotAnswers: slotAnswers);
  }

  @override
  ValidationResult validate() {
    for (final entry in _controllers.entries) {
      if (entry.value.text.trim().isEmpty) {
        return ValidationResult.error('请填写所有空格');
      }
    }
    return ValidationResult.ok;
  }

  @override
  void reset() {
    for (final controller in _controllers.values) {
      controller.clear();
    }
  }

  @override
  void restore(UserAnswer previousAnswer) {
    if (previousAnswer.payload is FillBlankPayload) {
      final payload = previousAnswer.payload as FillBlankPayload;
      for (final entry in payload.slotAnswers.entries) {
        _controllers[entry.key]?.text = entry.value;
      }
    }
  }

  /// 注册填空位的 controller
  void registerSlot(String slotId, TextEditingController controller) {
    _controllers[slotId] = controller;
  }

  void dispose() {
    for (final controller in _controllers.values) {
      controller.dispose();
    }
  }
}

    _totalStopwatch.start();
    _questionStopwatch.start();
  }

  void stop() {
    _totalStopwatch.stop();
    _questionStopwatch.stop();
  }

  /// 切换到下一题时调用

/// 本地答题缓存 — 用于断点续答和进度恢复
class LocalAnswerCache {
  final SharedPreferences _prefs;

  LocalAnswerCache({required SharedPreferences prefs}) : _prefs = prefs;

  /// 保存单题答案
  Future<void> saveAnswer(String sessionKey, UserAnswer answer) async {
    final json = jsonEncode(answer.toJson());
    await _prefs.setString('answer_${sessionKey}_${answer.questionId}', json);

    // 更新会话索引
    final sessionKeys = _prefs.getStringList('active_sessions') ?? [];
    if (!sessionKeys.contains(sessionKey)) {
      sessionKeys.add(sessionKey);
      await _prefs.setStringList('active_sessions', sessionKeys);
    }
  }

  /// 恢复会话所有答案
  Map<String, UserAnswer> restoreSession(String sessionKey) {
    final answers = <String, UserAnswer>{};
    final prefix = 'answer_${sessionKey}_';
    for (final key in _prefs.getKeys()) {
      if (key.startsWith(prefix)) {
        final json = _prefs.getString(key);
        if (json != null) {
          final answer = UserAnswer.fromJson(jsonDecode(json));
          answers[answer.questionId] = answer;
        }
      }
    }
    return answers;
  }

  /// 检查是否存在未完成会话
  bool hasUnfinishedSession(String sessionKey) {
    final prefix = 'answer_${sessionKey}_';
    return _prefs.getKeys().any((k) => k.startsWith(prefix));
  }

  /// 清除会话缓存
  Future<void> clearSession(String sessionKey) async {
    final prefix = 'answer_${sessionKey}_';
    final keysToRemove = _prefs.getKeys().where((k) => k.startsWith(prefix)).toList();
    for (final key in keysToRemove) {
      await _prefs.remove(key);
    }

    final sessionKeys = _prefs.getStringList('active_sessions') ?? [];
    sessionKeys.remove(sessionKey);
    await _prefs.setStringList('active_sessions', sessionKeys);
  }
}

## 14. 错误处理

### 14.1 常见错误场景

| 场景 | 处理策略 |
| --- | --- |
| 网络断开 | 本地暂存答案，恢复网络后自动同步 |
| 判题请求超时 | 本地判题（选择题/填空题），简答题标记"待评" |
| 题目数据格式异常 | 降级为纯文本渲染，日志上报 |
| 公式渲染失败 | 显示 LaTeX 源码 + 提示"公式加载失败" |
| 图片加载失败 | 显示占位图 + 重试按钮 |
| 交卷失败 | 重试 3 次，失败则提示"已本地保存，稍后重试" |
| 音频播放失败 | 显示文字版题目 + 反馈按钮 |
| 会话恢复数据损坏 | 清除缓存，提示用户重新开始 |
| 内存不足（大试卷） | 分页加载题目，已答题回收渲染树 |

        body: {
          'questionId': questionId,
          'answer': answer.payload.toJson(),
          'timeSpentMs': answer.timeSpent.inMilliseconds,
          'clientId': answer.clientId,
        },
        timeout: const Duration(seconds: 10),
      );

      if (response.statusCode == 200) {
        return JudgeResult.fromJson(response.data['data']);
      }
    } on TimeoutException {
      _logger.warning('判题请求超时 (attempt ${attempt + 1})');
    } on SocketException {
      _logger.warning('网络异常 (attempt ${attempt + 1})');
    } catch (e) {
      _logger.error('判题异常: $e');
    }

    if (attempt < maxRetries - 1) {
      await Future.delayed(Duration(seconds: pow(2, attempt).toInt()));
    }
  }

  // 重试全部失败 → 降级处理
  return _localJudgeFallback(questionId, answer);
}

/// 本地降级判题（仅支持客观题）
JudgeResult? _localJudgeFallback(String questionId, UserAnswer answer) {
  // 从预加载的题目数据中获取标准答案
  final question = _questions.firstWhereOrNull((q) => q.questionId == questionId);
  if (question?.standardAnswer == null) return null;

  final stdAnswer = question!.standardAnswer!;

  switch (question.type) {
    case QuestionType.singleChoice:
      final payload = answer.payload as ChoicePayload;
      final isCorrect = payload.selectedOptionIds.length == 1 &&
          stdAnswer.choiceAnswers?.contains(payload.selectedOptionIds.first) == true;
      return JudgeResult(
        questionId: questionId,
        isCorrect: isCorrect,
        earnedScore: isCorrect ? question.score : 0,
        totalScore: question.score,
        correctAnswer: stdAnswer,
        explanation: question.explanation ?? RichContent(blocks: [], plainTextSummary: ''),
      );

    case QuestionType.trueFalse:
      final payload = answer.payload as TrueFalsePayload;
      final isCorrect = payload.answer == stdAnswer.tfAnswer;
      return JudgeResult(/* ... */);

    // 简答题等无法本地判题
    default:
      return null;
  }
}

/// 题干预渲染缓存
class StemRenderCache {
  final int maxSize;
  final LinkedHashMap<String, _CachedStem> _cache = LinkedHashMap();

  /// 获取或创建渲染结果
  InlineSpan getOrCreateSpan({
    required String questionId,
    required RichContent content,
    required TextStyle baseStyle,
  }) {
    final cached = _cache[questionId];
    if (cached != null) return cached.span;

    final span = _buildSpan(content, baseStyle);
    _cache[questionId] = _CachedStem(span);
    _evictIfNeeded();
    return span;
  }

  void _evictIfNeeded() {
    while (_cache.length > maxSize) {
      _cache.remove(_cache.keys.first);
    }
  }
}

## 20. 总结

题目展示与答题交互组件库是 PrimeTop 的**核心共享基础设施**，承载了全学段、全学科的高频答题交互。通过以下设计确保其可靠性：

1. **注册机制**保证题型可扩展，新增题型无需修改框架代码。
2. **会话控制器**统一管理答题生命周期，业务模块只需关心配置和结果。
3. **分龄 UI 配置**让同一套代码适配从幼儿到高中的不同视觉需求。
4. **本地缓存**确保断网、闪退等异常场景下答题进度不丢失。
5. **判题降级**保证客观题在网络异常时仍能即时反馈。
6. **虚拟化渲染**保证百题级试卷的流畅体验。
