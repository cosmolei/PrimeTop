# 安全与内容合规系统 - 详细设计

## 1. 模块概述

### 1.1 定位

安全与内容合规系统是 PrimeTop 的跨领域基础设施，贯穿所有业务模块。负责内容安全过滤、未成年人保护（防沉迷、时长管控、消费限制）、数据隐私保护、操作审计和合规管理。

### 1.2 设计原则

| 原则 | 说明 |
|------|------|
| 默认安全 | 所有用户输入和 AI 输出默认经过安全检查，无需业务方显式调用 |
| 零信任 | 任何接口调用均需鉴权，不依赖"上游已校验"假设 |
| 最小可见 | 未成年人仅可见适龄内容，敏感数据脱敏展示 |
| 可审计 | 关键操作全链路可追溯，审计日志不可篡改 |
| 降级不宕机 | 安全服务不可用时，采用"保守拒绝"策略，不暴露风险内容 |

```kotlin
class LocalContentFilter {

    private val blockedPatterns: List<Regex> = listOf(
        // 越狱提示词常见模式
        Regex("(?i)(ignore.previous|forget.instructions|system.prompt|你是谁|developer.mode)"),
        // 明显非学习内容
        Regex("(?i)(帮我写情书|如何黑入|制造炸弹)"),
        // 手机号/身份证号模式
        Regex("1[3-9]\\d{9}"),           // 手机号
        Regex("\\d{17}[\\dXx]"),          // 身份证号
    )

    data class FilterResult(
        val pass: Boolean,
        val reason: String? = null,
        val riskType: ContentType? = null
    )

    fun check(input: String): FilterResult {
        for ((index, pattern) in blockedPatterns.withIndex()) {
            if (pattern.containsMatchIn(input)) {
                return FilterResult(
                    pass = false,
                    reason = "LOCAL_RULE_${index}",
                    riskType = when (index) {
                        0 -> ContentType.GATEWAY_BYPASS
                        1 -> ContentType.INAPPROPRIATE_QUESTION
                        else -> ContentType.PERSONAL_INFO
                    }
                )
            }
        }
        return FilterResult(pass = true)
    }
}
```

#### Level 2: 服务端规则引擎

```python
class RuleEngine:
    """服务端规则引擎 - 关键词 + 正则 + 黑名单"""

    def __init__(self, config_repo: SecurityConfigRepository):
        self.config = config_repo
        self._compile_rules()

    def _compile_rules(self):
        """加载并编译规则（应用启动时 + 配置变更时热更新）"""
        self.keyword_blacklist = self.config.get_keyword_blacklist()
        self.regex_rules = [
            (re.compile(rule["pattern"], re.IGNORECASE), rule)
            for rule in self.config.get_regex_rules()
        ]

    def check(self, text: str, context: SecurityContext) -> RuleCheckResult:
        # 1. 精确关键词匹配（ Trie 树加速）
        matched_keywords = self._match_keywords(text)
        if matched_keywords:
            return RuleCheckResult(
                action=FilterAction.BLOCK,
                matched_rules=matched_keywords
            )

        # 2. 正则规则匹配
        for pattern, rule in self.regex_rules:
            if pattern.search(text):
                return RuleCheckResult(
                    action=FilterAction(rule["action"]),  # BLOCK / REVIEW
                    matched_rules=[rule["id"]]
                )

        return RuleCheckResult(action=FilterAction.PASS)

    def _match_keywords(self, text: str) -> list:
        """Trie 树关键词匹配，O(n) 复杂度"""
        results = []
        node = self.keyword_trie
        for i, char in enumerate(text):
            while char not in node.children and node != self.keyword_trie:
                node = node.fail  # AC 自动机失败指针
            if char in node.children:
                node = node.children[char]
                if node.is_end:
                    results.append(node.keyword_id)
        return results
```

```python
class CloudContentModerator:
    """第三方内容安全审核（阿里云/腾讯云/百度）"""

    def __init__(self, providers: list[ContentSafetyProvider]):
        # 主备双供应商
        self.primary = providers[0]
        self.fallback = providers[1] if len(providers) > 1 else None

    async def moderate_text(
        self, text: str, scene: str = "education"
    ) -> ModerationResult:
        try:
            result = await self.primary.check_text(text, scene=scene)
        except Exception as e:
            logger.warning(f"Primary moderation failed: {e}")
            if self.fallback:
                result = await self.fallback.check_text(text, scene=scene)
            else:
                # 降级：保守拒绝
                return ModerationResult(
                    action=FilterAction.BLOCK,
                    reason="MODERATION_SERVICE_UNAVAILABLE"
                )

        # 记录审核日志（无论是否拦截）
        await self._log_moderation(text, result)

        return result

    async def moderate_image(self, image_bytes: bytes) -> ModerationResult:
        """图片安全审核：色情、暴力、政治、广告"""
        result = await self.primary.check_image(image_bytes)
        await self._log_moderation("[IMAGE]", result)
        return result
```

```python
class AIOutputSafetyLayer:
    """AI 模型输出的安全处理"""

    # 学习场景特殊规则：不允许直接输出完整答案
    DIRECT_ANSWER_PATTERNS = [
        r"答案[是为]\s*[A-D]",
        r"正确答案[是为]",
        r"最终结果[是为]\s*\d+",
    ]

    async def process(
        self,
        raw_output: str,
        context: AIOutputContext
    ) -> SafeOutput:
        # 1. 通用内容安全检查
        moderation = await self.moderator.moderate_text(raw_output)
        if moderation.action == FilterAction.BLOCK:
            return SafeOutput(
                content=self._safe_replacement("content_unsafe"),
                was_filtered=True,
                filter_reason=moderation.reason
            )

        # 2. 学习场景：答案渐进式展示检查
        if context.scene == "homework_help":
            if self._contains_direct_answer(raw_output):
                return SafeOutput(
                    content=self._convert_to_hint(raw_output),
                    was_filtered=True,
                    filter_reason="direct_answer_converted"
                )

        # 3. 个人信息泄露检查（AI 不应输出用户真实个人信息）
        if self._contains_personal_info(raw_output):
            raw_output = self._mask_personal_info(raw_output)

        return SafeOutput(content=raw_output, was_filtered=False)

    def _convert_to_hint(self, output: str) -> str:
        """将直接答案转换为启发式提示"""
        # 例如："答案是B" → "你可以从以下几个方面思考这道题..."
        prompt = """请将以下内容改写为启发式讲解，不要直接给出最终答案，
        而是引导学生思考解题思路和关键步骤：

        原始内容：
        {content}
        """
        # 调用轻量模型进行改写
        return self.llm_client.rewrite(prompt.format(content=output))

    def _contains_direct_answer(self, text: str) -> bool:
        for pattern in self.DIRECT_ANSWER_PATTERNS:
            if re.search(pattern, text):
                return True
        return False
```

#### 时长管控服务

```python
class UsageTimeService:

    def __init__(self, repo: UsageTimeRepository):
        self.repo = repo

    async def check_and_enter(self, user_id: int) -> UsageCheckResult:
        """用户启动APP或新会话时检查"""
        age_group = await self._get_age_group(user_id)
        if age_group == AgeGroup.ADULT:
            return UsageCheckResult(allowed=True)

        rule = await self.repo.get_rule(age_group)
        record = await self.repo.get_today_record(user_id)
        now = datetime.now()

        # 1. 禁用时段检查
        if rule.forbidden_start and rule.forbidden_end:
            if self._in_forbidden_period(now, rule):
                return UsageCheckResult(
                    allowed=False,
                    reason="forbidden_period",
                    message=f"当前为休息时间（{rule.forbidden_start}-{rule.forbidden_end}），请好好休息",
                    resume_at=self._next_allowed_time(now, rule)
                )

        # 2. 每日总时长检查
        if rule.max_daily_min > 0:
            used_min = record.total_seconds // 60 if record else 0
            if used_min >= rule.max_daily_min:
                return UsageCheckResult(
                    allowed=False,
                    reason="daily_limit",
                    message=f"今日学习时长已达上限（{rule.max_daily_min}分钟），明天继续加油！",
                    used_minutes=used_min,
                    limit_minutes=rule.max_daily_min
                )

        # 3. 强制休息检查
        if record and record.rest_until and now < record.rest_until:
            return UsageCheckResult(
                allowed=False,
                reason="rest_required",
                message=f"休息一下，{record.rest_until.strftime('%H:%M')}后可以继续",
                resume_at=record.rest_until
            )

        return UsageCheckResult(
            allowed=True,
            remaining_today=(rule.max_daily_min * 60 - (record.total_seconds if record else 0)),
            session_limit_sec=rule.max_session_min * 60
        )

    async def heartbeat(self, user_id: int, session_duration_sec: int):
        """客户端每60秒心跳上报"""
        age_group = await self._get_age_group(user_id)
        if age_group == AgeGroup.ADULT:
            return

        rule = await self.repo.get_rule(age_group)
        record = await self.repo.get_or_create_today(user_id)
        new_total = record.total_seconds + 60  # 每次心跳+60秒

        # 更新记录
        await self.repo.update_usage(user_id, new_total, record.session_count + 1)

        # 单次会话超时检查
        if rule.max_session_min > 0 and session_duration_sec >= rule.max_session_min * 60:
            rest_until = datetime.now() + timedelta(minutes=rule.rest_min)
            await self.repo.set_rest_until(user_id, rest_until)
            return HeartbeatResult(
                action="force_rest",
                message=f"已经学习{rule.max_session_min}分钟了，休息{rule.rest_min}分钟吧！",
                rest_until=rest_until
            )

        # 接近日限时警告
        used_min = new_total // 60
        remaining_min = rule.max_daily_min - used_min
        if remaining_min <= 5 and remaining_min > 0:
            return HeartbeatResult(
                action="warning",
                message=f"今日还剩{remaining_min}分钟学习时间"
            )

        return HeartbeatResult(action="continue")

    def _in_forbidden_period(self, now: datetime, rule) -> bool:
        current = now.time()
        start = rule.forbidden_start
        end = rule.forbidden_end
        if start < end:
            return start <= current < end
        else:  # 跨午夜（如22:00-06:00）
            return current >= start or current < end
```

```python
        limits = self.LIMITS[age_group]

        if limits["single"] == 0:
            return PurchaseCheckResult(
                allowed=False,
                reason="age_restricted",
                message="当前年龄不支持自主购买，请联系家长操作"

            return PurchaseCheckResult(
                allowed=False,
                reason="single_limit_exceeded",

        # 检查月累计

            return PurchaseCheckResult(
                allowed=False,
                reason="monthly_limit_exceeded",
                message=f"本月累计充值已达上限{limits['monthly']}元"

        return PurchaseCheckResult(allowed=True)
```

### 4.2 数据加密策略

```python
class EncryptionService:
    """数据加密服务"""

    def __init__(self, kms_client: KMSClient):
        self.kms = kms_client
        # 数据库字段级加密密钥（通过KMS获取，定期轮换）
        self._field_key_cache = {}

    async def encrypt_field(self, plaintext: str, sensitivity: DataSensitivity) -> str:
        """字段级加密"""
        if sensitivity.value < DataSensitivity.HIGHLY_SENSITIVE.value:
            return plaintext  # 低敏感度数据不加密（依赖传输加密和存储加密）

        key_id = f"field_encryption_key_v{await self.kms.get_current_version()}"
        key = await self._get_key(key_id)

        iv = os.urandom(12)
        cipher = AESGCM(key)
        ciphertext = cipher.encrypt(iv, plaintext.encode('utf-8'), None)
        # 格式: key_version | iv | ciphertext | tag (base64)
        return base64.b64encode(
            struct.pack('>I', await self.kms.get_current_version()) + iv + ciphertext
        ).decode('ascii')

    async def decrypt_field(self, encrypted: str) -> str:
        """字段级解密"""
        raw = base64.b64decode(encrypted)
        version = struct.unpack('>I', raw[:4])[0]
        iv = raw[4:16]
        ciphertext = raw[16:]

        key_id = f"field_encryption_key_v{version}"
        key = await self._get_key(key_id)
        cipher = AESGCM(key)
        return cipher.decrypt(iv, ciphertext, None).decode('utf-8')
```

```python
class DataMasking:
    """数据脱敏工具"""

    @staticmethod
    def mask_phone(phone: str) -> str:
        """手机号脱敏: 138****8888"""
        if len(phone) == 11:
            return phone[:3] + "****" + phone[7:]
        return "***"

    @staticmethod
    def mask_name(name: str) -> str:
        """姓名脱敏: 张* / 张*三"""
        if len(name) == 1:
            return name
        if len(name) == 2:
            return name[0] + "*"
        return name[0] + "*" * (len(name) - 2) + name[-1]

    @staticmethod
    def mask_id_card(id_card: str) -> str:
        """身份证号脱敏: 110***********1234"""
        if len(id_card) >= 15:
            return id_card[:3] + "*" * (len(id_card) - 7) + id_card[-4:]
        return "****"

    @staticmethod
    def mask_email(email: str) -> str:
        """邮箱脱敏: a***@example.com"""
        parts = email.split("@")
        if len(parts) == 2 and len(parts[0]) > 1:
            return parts[0][0] + "***@" + parts[1]
        return "***"
```

```python
class RiskControlEngine:
    """风控引擎 - 检测异常行为"""

    async def evaluate(self, event: RiskEvent) -> RiskDecision:
        score = 0

        # 规则1: 同一IP短时间大量注册
        if event.action == "register":
            ip_count = await self.repo.count_by_ip(event.ip, minutes=60)
            if ip_count > 5: score += 30
            if ip_count > 20: score += 50

        # 规则2: 同一设备多账号
        if event.device_id:
            device_users = await self.repo.count_users_on_device(event.device_id)
            if device_users > 3: score += 20
            if device_users > 10: score += 40

        # 规则3: AI调用频率异常
        if event.action == "ai_call":
            hourly_calls = await self.repo.count_ai_calls(event.user_id, hours=1)
            if hourly_calls > 100: score += 30
            if hourly_calls > 500: score += 50

        # 规则4: 支付行为异常
        if event.action == "payment":
            daily_payments = await self.repo.count_payments(event.user_id, days=1)
            if daily_payments > 3: score += 20

        # 规则5: 跨地区登录
        if event.action == "login":
            last_location = await self.repo.get_last_login_location(event.user_id)
            if last_location and self._distance_km(last_location, event.location) > 500:
                score += 25

        # 决策
        if score >= 80:
            return RiskDecision(action="block", score=score, reason="high_risk")
        if score >= 40:
            return RiskDecision(action="challenge", score=score, reason="medium_risk",
                              challenge_type="sms_verify")
        if score >= 20:
            return RiskDecision(action="monitor", score=score, reason="low_risk")

        return RiskDecision(action="allow", score=score)
```

```python
    """设备信任评估"""

    async def evaluate(self, fingerprint: DeviceFingerprint) -> DeviceTrust:
        trust_score = 100  # 满分100

        # 模拟器检测
        if fingerprint.is_emulator:
            trust_score -= 50
            await self._flag(fingerprint.device_id, "emulator_detected")

        # Root/越狱检测
        if fingerprint.is_rooted:
            trust_score -= 30
            await self._flag(fingerprint.device_id, "root_detected")

        # 调试模式
        if fingerprint.is_debug_mode:
            trust_score -= 20

        # 设备ID稳定性检查（同一物理设备频繁更换deviceId）
        stability = await self.repo.get_id_stability(fingerprint.device_id)
        if stability.change_count > 3:
            trust_score -= 30

        return DeviceTrust(
            device_id=fingerprint.device_id,
            trust_score=max(0, trust_score),
            level=self._score_to_level(trust_score)
        )

    @staticmethod
    def _score_to_level(score: int) -> str:
        if score >= 80: return "trusted"
        if score >= 50: return "suspicious"
        return "untrusted"
```

| 规则引擎延迟 | < 10ms |
| 第三方内容安全API延迟 | < 200ms（P99） |
| 时长检查延迟 | < 20ms |
| 心跳处理延迟 | < 50ms |
| 安全过滤对AI响应的整体额外延迟 | < 500ms |

### 10.2 可靠性策略

| 策略 | 说明 |
|------|------|
| 本地规则兜底 | 客户端内置基础规则，网络异常时仍可拦截高风险内容 |
| 第三方API双供应商 | 主备切换，避免单点故障 |
| 保守降级 | 审核服务不可用时，AI输出走保守策略（只显示提示不展示完整内容） |
| 规则缓存 | 关键词黑名单和正则规则缓存到 Redis，本地进程缓存10秒TTL |
| 异步审计 | 审计日志写入走消息队列，不阻塞主流程 |

## 12. 状态流转

### 12.1 内容安全检查流转

```
用户输入 ──→ 客户端本地预检 ──→ [不通过] ──→ 拦截提示（客户端直接处理）
                │
             [通过]
                │
                ▼
         服务端规则引擎 ──→ [命中BLOCK] ──→ 返回40001 + 记录安全事件
                │
             [通过/REVIEW]
                │
                ▼
         第三方内容安全API ──→ [命中] ──→ 返回40001 + 记录安全事件
                │                      REVIEW → 人工审核队列
             [通过]
                │
                ▼
         正常业务处理（如调用AI）
                │
                ▼
         AI输出安全层 ──→ [不安全] ──→ 替换/改写 + 记录
                │
             [安全]
                │
                ▼
         返回给客户端
```

### 12.2 使用时长管控流转

```
APP启动 ──→ usage-check ──→ [不允许] ──→ 展示限制页面（倒计时/提示）
               │
            [允许]
               │
               ▼
         进入主界面 + 启动心跳定时器（60秒）
               │
               ▼
         每60秒 heartbeat ──→ [continue] ──→ 继续
               │
               ├──→ [warning] ──→ 展示剩余时间提示
               │
               ├──→ [force_rest] ──→ 锁定界面 + 倒计时展示
               │
               └──→ [daily_limit] ──→ 展示今日已达上限页面
```

### 12.3 支付风控流转

```
用户点击支付 ──→ 消费限制检查 ──→ [不允许] ──→ 提示限制原因
                    │
                 [允许]
                    │
                    ▼
              风控引擎评估 ──→ [block] ──→ 拒绝交易 + 安全事件
                    │
                 [challenge] ──→ 短信验证 → 验证通过 → 继续
                    │
                 [allow]
                    │
                    ▼
              正常支付流程
```
