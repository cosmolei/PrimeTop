## 1. 模块概述

### 1.1 功能定位

为学生提供操作系统桌面级别的小组件(Widget)和快捷入口，将核心学习信息（今日任务、学习进度、连续打卡、快捷操作）前置到系统桌面，降低使用门槛，促进学习习惯养成。

### 1.2 核心价值

| 价值维度 | 说明 |
|----------|------|
| 降低启动成本 | 无需打开 APP 即可查看学习状态，一键进入核心功能 |
| 促进学习习惯 | 桌面常驻展示连续打卡天数、今日任务，形成心理锚点 |
| 增强产品粘性 | 小组件作为"被动触达"渠道，提升 DAU 和留存率 |
| 分龄差异化 | 幼儿/小学展示趣味化内容，初高中展示任务和倒计时 |

### 1.3 支持平台

| 平台 | 技术 | 最低版本 |
|------|------|----------|
| iOS | WidgetKit (SwiftUI) | iOS 16+ |
| Android | Glance (Jetpack Glance) | Android 12+ (API 31) |
| Android (兼容) | AppWidgetProvider (XML布局) | Android 8+ (API 26) |

### 1.4 与现有模块的关系

| 关联模块 | 交互方式 |
|----------|----------|
| 首页与学习工作台 | 共享数据源（WidgetDataService），展示相同的今日任务/学习概览 |
| 学习提醒与智能通知调度系统 | Widget 刷新与通知调度协同，避免重复触达 |
| 学习记录与进度追踪服务 | 消费学习时长、连续天数、完成率数据 |
| 学习习惯养成与打卡系统 | 展示打卡状态、连续天数、徽章进度 |
| 运营活动与成长激励体系 | 展示当前活动进度、限时任务倒计时 |
| 学期与学年管理服务 | 考试倒计时（期中/期末/中考/高考） |
| 客户端主题引擎与动态外观系统 | Widget 配色跟随应用主题（Light/Dark/护眼） |
| 会员订阅 | 部分高级 Widget 样式可能需要会员权益 |

### 2.2 数据流设计

```text
主APP数据变更 → WidgetDataService.aggregate()
                    ↓
            WidgetDataStore.persist()
                    ↓
         OS Widget Runtime 读取本地数据
                    ↓
          渲染 Widget UI → 桌面展示
                    ↓
          用户点击 → DeepLink → Flutter 路由
```

**关键约束**：
- Widget 进程与主 APP 进程隔离，通过共享文件/数据库通信
- Widget 不能直接调用 Flutter 插件或网络请求（iOS WidgetKit 限制）
- 数据必须由主 APP 预计算并写入共享存储

```swift
    // 写入 payload JSON
    func savePayload(_ payload: WidgetPayload) {
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601

        Self.sharedDefaults.set(data, forKey: "widget_payload")
        Self.sharedDefaults.set(Date(), forKey: "widget_payload_timestamp")

}
```

```kotlin
// Glance / AppWidget 数据共享
class WidgetDataStore(context: Context) {
    private val prefs = context.getSharedPreferences(
        "widget_data", Context.MODE_PRIVATE
    )
    
    fun savePayload(payload: WidgetPayload) {
        val json = Json.encodeToString(payload)
        prefs.edit()
            .putString("widget_payload", json)
            .putLong("widget_payload_timestamp", System.currentTimeMillis())
            .apply()
    }
    
    companion object {
        fun getPayload(context: Context): WidgetPayload? {
            val prefs = context.getSharedPreferences(
                "widget_data", Context.MODE_PRIVATE
            )
            val json = prefs.getString("widget_payload", null) ?: return null
            return Json.decodeFromString(json)
        }
    }
}
```

#### 4.2.3 Flutter 侧统一写入接口

```dart
/// Flutter 端统一 Widget 数据管理器
/// 聚合各模块数据后通过平台通道写入
class WidgetDataManager {
  static const _channel = MethodChannel('com.primetop.app/widget');
  
  final StudyRecordService _studyRecord;
  final DailyTaskService _dailyTask;
  final StreakService _streak;
  final SemesterService _semester;
  final CourseService _course;
  final ThemeService _theme;
  final MembershipService _membership;
  
  /// 聚合并写入所有 Widget 数据
  Future<void> refreshWidgetData() async {
    final payload = await _aggregatePayload();
    
    // 通过 MethodChannel 写入平台原生共享存储
    await _channel.invokeMethod('updateWidgetData', {
      'payload': jsonEncode(payload.toJson()),
    });
    
    // 通知 Widget 刷新
    await _channel.invokeMethod('refreshWidgets');
  }
  
  Future<WidgetPayload> _aggregatePayload() async {
    final userId = _auth.currentUserId;
    final grade = _user.grade;
    
    // 并行获取各数据源
    final results = await Future.wait([
      _studyRecord.getTodayOverview(userId),
      _dailyTask.getTodayTasks(userId, limit: 5),
      _streak.getCurrentStreak(userId),
      _semester.getNextExam(userId),
      _course.getFeaturedCourse(userId),
      _theme.getCurrentTheme(),
      _membership.getStatus(userId),
    ]);
    
    return WidgetPayload(
      version: 1,
      generatedAt: DateTime.now().toIso8601String(),
      userId: userId,
      studyOverview: results[0],
      dailyTasks: results[1],
      streak: results[2],
      examCountdown: results[3],
      featuredCourse: results[4],
      theme: results[5],
      membership: results[6],
    );
  }
}
```

## 5. Widget 刷新策略

### 5.1 刷新机制对比

| 机制 | 平台 | 特点 | 限制 |
|------|------|------|------|
| Background Fetch | iOS | 系统调度，最少15分钟间隔 | 不保证执行时机，低频 |
| Timeline Provider | iOS WidgetKit | 定义未来时间线，系统自动刷新 | 最多70条时间线条目 |
| WorkManager | Android | 周期性任务，最小15分钟 | Doze模式下受限 |
| Glance Update | Android Glance | 响应式更新，支持Flow | 需要 Android 12+ |
| App Foreground | 双端 | APP 在前台时主动更新 | 仅 APP 运行时 |
| Push Notification (静默) | 双端 | 服务端触发，实时性最高 | 需要推送基础设施 |

### 5.2 刷新策略矩阵

```dart
/// Widget 刷新调度器
class WidgetRefreshScheduler {
  final WidgetDataManager _dataManager;
  
  /// 注册所有刷新触发点
  void registerTriggers() {
    // 1. APP 回到前台 → 立即刷新
    WidgetsBinding.instance.addObserver(_lifecycleObserver);
    
    // 2. 关键数据变更事件 → 延迟3秒批量刷新
    _eventBus.on<StudySessionCompletedEvent>().debounce(3.seconds).listen((_) {
      _dataManager.refreshWidgetData();
    });
    _eventBus.on<TaskCompletedEvent>().debounce(3.seconds).listen((_) {
      _dataManager.refreshWidgetData();
    });
    _eventBus.on<StreakUpdatedEvent>().listen((_) {
      _dataManager.refreshWidgetData();
    });
    
    // 3. 主题变更 → 刷新主题相关数据
    _eventBus.on<ThemeChangedEvent>().listen((_) {
      _dataManager.refreshWidgetData();
    });
  }
  
  // APP 生命周期回调
  void _onAppResumed() {
    _dataManager.refreshWidgetData();
  }
}
```

### 5.4 iOS Timeline 策略

```swift
// WidgetKit Timeline Provider
struct StudyTimelineProvider: TimelineProvider {
    func getTimeline(in context: Context, completion: @escaping (Timeline<StudyEntry>) -> Void) {
        let payload = WidgetDataStore.loadPayload()
        let now = Date()
        
        var entries: [StudyEntry] = []
        
        // 当前条目
        entries.append(StudyEntry(date: now, payload: payload))
        
        // 预测未来时间线条目
        // 在关键时间点刷新：早晨起床、放学后、晚饭后、睡前
        let refreshTimes = generateRefreshTimes(from: now)
        for time in refreshTimes {
            entries.append(StudyEntry(date: time, payload: payload))
        }
        
        let timeline = Timeline(entries: entries, policy: .after(nextRefreshDate))
        completion(timeline)
    }
    
    /// 根据分龄策略生成关键刷新时间
    private func generateRefreshTimes(from now: Date) -> [Date] {
        let calendar = Calendar.current
        let grade = payload.grade
        
        // 通用刷新点
        var times: [Date] = []
        
        // 早晨 7:00 - 起床提醒
        if let morning = calendar.date(bySettingHour: 7, minute: 0, second: 0, of: now), morning > now {
            times.append(morning)
        }
        
        // 放学后 16:30 - 作业时间
        if let afterSchool = calendar.date(bySettingHour: 16, minute: 30, second: 0, of: now), afterSchool > now {
            times.append(afterSchool)
        }
        
        // 晚饭后 19:00 - 复习时间
        if let evening = calendar.date(bySettingHour: 19, minute: 0, second: 0, of: now), evening > now {
            times.append(evening)
        }
        
        // 睡前 21:30 - 当日总结
        if let bedtime = calendar.date(bySettingHour: 21, minute: 30, second: 0, of: now), bedtime > now {
            times.append(bedtime)
        }
        
        // 高中生增加晚间学习时段
        if grade >= 10 {
            if let nightStudy = calendar.date(bySettingHour: 22, minute: 0, second: 0, of: now), nightStudy > now {
                times.append(nightStudy)
            }
        }
        
        return times
    }
}
```

### 5.5 Android WorkManager 策略

```kotlin
class WidgetRefreshWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {
    
    override suspend fun doWork(): Result {
        return try {
            // 从服务端拉取最新数据（如果网络可用）
            val freshData = fetchLatestWidgetData()
            
            if (freshData != null) {
                // 写入共享存储
                WidgetDataStore.savePayload(freshData)
                // 通知 Widget 更新
                GlanceAppWidgetManager(applicationContext)
                    .getGlanceIds(StudyOverviewWidget::class.java)
                    .forEach { glanceId ->
                        StudyOverviewWidget().update(applicationContext, glanceId)
                    }
                Result.success()
            } else {
                // 网络失败，保持现有数据
                Result.retry()
            }
        } catch (e: Exception) {
            Result.retry()
        }
    }
    
    companion object {
        fun schedulePeriodicRefresh(context: Context) {
            val request = PeriodicWorkRequestBuilder<WidgetRefreshWorker>(
                15, TimeUnit.MINUTES  // 最小间隔15分钟
            )
            .setConstraints(
                Constraints.Builder()
                    .setRequiredNetworkType(NetworkType.CONNECTED)
                    .setRequiresBatteryNotLow(true)
                    .build()
            )
            .setBackoffCriteria(
                BackoffPolicy.LINEAR,
                5, TimeUnit.MINUTES
            )
            .build()
            
            WorkManager.getInstance(context)
                .enqueueUniquePeriodicWork(
                    "widget_refresh",
                    ExistingPeriodicWorkPolicy.KEEP,
                    request
                )
        }
    }
}
```

```dart
class ExamCountdownCalculator {
  /// 计算距离考试天数
  /// 返回 null 表示没有即将到来的考试
  ExamCountdown? calculate(StudentProfile profile) {
    // 1. 查找下一个最近的考试
    final exams = _semesterService.getUpcomingExams(profile.userId);
    if (exams.isEmpty) return null;
    
    // 2. 按优先级排序：自定义考试 > 学期考试 > 中高考
    final exam = exams.first;
    final daysLeft = exam.targetDate.difference(DateTime.now()).inDays;
    
    // 3. 考试已过则跳到下一个
    if (daysLeft < 0) {
      return exams.length > 1 
        ? _calculate(exams[1]) 
        : null;
    }
    
    return ExamCountdown(
      examName: exam.name,
      targetDate: exam.targetDate,
      daysLeft: daysLeft,
      slogan: _getSlogan(daysLeft, profile.grade),
    );
  }
  
  String _getSlogan(int daysLeft, int grade) {
    if (daysLeft <= 3) return '最后冲刺！';
    if (daysLeft <= 7) return '一周内，稳住！';
    if (daysLeft <= 30) return '有条不紊地复习';
    return '提前准备，从容应对';
  }
}
```

```swift
    do {
        _ = try Activity.request(
            attributes: attributes,
            content: content,
            pushToken: nil  // 可选：用于服务端推送更新
        )
    } catch {
        logger.error("Failed to start Live Activity: \(error)")
    }
}

                sessionType: activity.contentState.sessionType
```

```dart
class WidgetRefreshRetryPolicy {
  static const maxRetries = 3;
  static const backoffDurations = [5, 15, 60]; // 秒
  
  static Future<void> refreshWithRetry() async {
    for (var attempt = 0; attempt < maxRetries; attempt++) {
      try {
        await _dataManager.refreshWidgetData();
        return; // 成功，退出
      } catch (e) {
        if (attempt < maxRetries - 1) {
          await Future.delayed(
            Duration(seconds: backoffDurations[attempt])
          );
        }
        // 最后一次失败则静默放弃，等待下次触发
      }
    }
  }
}
```

```dart
class WidgetMembershipFilter {
  /// 根据会员等级过滤 Widget 数据
  WidgetPayload filterByMembership(WidgetPayload raw, MembershipTier tier) {
    return WidgetPayload(
      ...raw,
      // 基础数据始终返回
      studyOverview: raw.studyOverview,
      streak: raw.streak,
      wordOfDay: raw.wordOfDay,
      
      // 按权益过滤
      dailyTasks: tier == MembershipTier.free
        ? raw.dailyTasks.take(3).toList()
        : raw.dailyTasks,
      examCountdown: tier == MembershipTier.free
        ? null
        : raw.examCountdown,
      quickActions: tier == MembershipTier.free
        ? raw.quickActions.take(3).toList()
        : raw.quickActions,
      featuredCourse: tier == MembershipTier.free
        ? null
        : raw.featuredCourse,
      theme: tier == MembershipTier.annual
        ? raw.theme  // 年度会员保留自定义色
        : raw.theme.copyWith(accentColor: kDefaultAccentColor),
    );
  }
}
```

## 15. 容量估算

### 15.1 数据存储

| 项 | 估算 |
|----|------|
| 单个 WidgetPayload JSON | ~2-3 KB |
| 资源图片（Logo + 图标） | ~200 KB (bundled) |
| 缓存图片（吉祥物、徽章） | ~500 KB (最多 10 张) |
| **单用户总存储** | **~1 MB** |

### 15.2 刷新流量

| 场景 | 频率 | 数据量 |
|------|------|--------|
| APP 前台切换 | ~20次/天 | 3KB × 20 = 60 KB/天 |
| 后台 WorkManager | ~4次/天 | 3KB × 4 = 12 KB/天 |
| 静默推送触发 | ~2次/天 | 3KB × 2 = 6 KB/天 |
| **单用户日均流量** | | **~80 KB** |

### 15.3 影响评估

- **对主 APP 性能影响**：每次数据聚合 < 50ms，写入共享存储 < 10ms
- **对电池影响**：WorkManager 使用约束（非低电量），影响可忽略
- **对网络影响**：Widget 自身不发网络请求，流量来自主 APP 正常请求

```dart
void main() {
  // Widget 数据通道
  const widgetChannel = MethodChannel('com.primetop.app/widget');
  
  widgetChannel.setMethodCallHandler((call) async {
    switch (call.method) {
      case 'updateWidgetData':
        final json = call.arguments as String;
        // 已在 Flutter 侧写入完成，此处仅触发原生刷新
        break;
      case 'refreshWidgets':
        // 通知原生侧刷新所有 Widget
        break;
      case 'getWidgetConfig':
        // 返回 Widget 配置（如用户选择的 Widget 类型）
        return jsonEncode(await _getWidgetConfig());
    }
  });
  
  runApp(PrimeTopApp());
```
