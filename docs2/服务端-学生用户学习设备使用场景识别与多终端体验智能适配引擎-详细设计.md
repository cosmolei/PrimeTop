# 服务端-学生用户学习设备使用场景识别与多终端体验智能适配引擎 详细设计

## 1. 概述

### 1.1 功能定位

本引擎负责实时感知学生当前使用的设备类型、网络环境、时间段和运动状态等上下文信号，推断学生的学习场景（如"书桌前专注学习"、"通勤途中碎片学习"、"睡前轻度复习"等），并据此向各业务模块输出**场景化适配策略**——包括内容形态选择、推荐会话时长、交互模式、通知策略和UI密度等。

### 1.2 设计目标

| 目标 | 说明 |
| --- | --- |
| 场景识别准确率 ≥ 85% | 基于多信号融合推断当前学习场景 |
| 适配决策延迟 < 50ms | 场景策略查询为内存级操作 |
| 隐私优先 | 所有场景推断在服务端完成，不采集 GPS 等精确位置 |
| 渐进增强 | 弱网/信号缺失时使用默认策略，不影响核心学习流程 |

### 1.3 适用范围

- 小学、初中、高中学生在手机、平板、Web 端的学习场景
- 不涉及幼儿段（幼儿场景由亲子共学模块单独管理）

## 4. 场景推断引擎设计

### 4.1 信号预处理

客户端上报的原始信号需要经过清洗和特征增强：

```python
class SignalPreprocessor:
    """信号预处理器：清洗、补全、时区对齐"""

    TIME_SLOT_RULES = [
        (5, 8,   "EARLY_MORNING"),
        (8, 12,  "FORENOON"),
        (12, 14, "NOON"),
        (14, 17, "AFTERNOON"),
        (17, 19, "DUSK"),
        (19, 22, "EVENING"),
        (22, 24, "LATE_NIGHT"),
        (0, 5,   "DEEP_NIGHT"),
    ]

    def process(self, signal: DeviceContextSignal, user_timezone: str) -> ProcessedSignal:
        tz = ZoneInfo(user_timezone)
        local_time = signal.client_timestamp.astimezone(tz)
        hour = local_time.hour

        # 补全时间段
        time_slot = self._resolve_time_slot(hour)
        is_weekday = local_time.weekday() < 5

        # 带宽缺失时根据网络类型粗估
        if signal.estimatedBandwidth is None:
            signal.estimatedBandwidth = self._estimate_bandwidth(signal.networkType)

        # 电量缺失时默认满电
        if signal.batteryLevel is None:
            signal.batteryLevel = 100

        # motion 缺失默认 STATIONARY
        if signal.motionState is None:
            signal.motionState = 'STATIONARY'

        return ProcessedSignal(
            raw=signal,
            local_hour=hour,
            time_slot=time_slot,
            is_weekday=is_weekday,
            low_battery=signal.batteryLevel < 20,
            on_wifi=signal.networkType == 'WIFI',
            on_cellular=signal.networkType.startswith('CELLULAR'),
            is_charging=signal.isCharging or False,
        )
```

### 4.2 场景分类器

采用 **规则优先 + ML 兜底** 的双层分类策略：

#### 4.2.1 规则层

```python
class RuleBasedSceneClassifier:
    """基于专家规则的场景分类器"""

    def classify(self, sig: ProcessedSignal, history: DeviceHistory) -> list[SceneCandidate]:
        candidates: list[SceneCandidate] = []

        # ---- 书桌前专注学习 ----
        if (sig.on_wifi
            and sig.raw.deviceType in ('DESKTOP', 'TABLET', 'WEB')
            and sig.time_slot in ('FORENOON', 'AFTERNOON', 'EVENING')
            and sig.raw.motionState == 'STATIONARY'
            and (sig.raw.idleSeconds or 0) < 30
            and history.avg_session_minutes_recent >= 20
        ):
            candidates.append(SceneCandidate('DESK_FOCUSED_STUDY', 0.90))

        # ---- 书桌前轻度复习 ----
        if (sig.on_wifi
            and sig.time_slot in ('FORENOON', 'AFTERNOON', 'EVENING')
            and sig.raw.motionState == 'STATIONARY'
            and (sig.raw.idleSeconds or 0) > 60
            and history.avg_session_minutes_recent < 15
        ):
            candidates.append(SceneCandidate('DESK_LIGHT_REVIEW', 0.75))

        # ---- 通勤途中碎片学习 ----
        if (sig.on_cellular
            and sig.raw.motionState in ('WALKING', 'IN_VEHICLE')
            and sig.raw.deviceType == 'PHONE'
        ):
            candidates.append(SceneCandidate('COMMUTE_FRAGMENT', 0.85))

        # ---- 睡前轻度学习 ----
        if (sig.time_slot in ('LATE_NIGHT', 'DEEP_NIGHT')
            and sig.raw.deviceType == 'PHONE'
            and sig.raw.orientation == 'PORTRAIT'
            and sig.low_battery == False
        ):
            candidates.append(SceneCandidate('BEDTIME_LIGHT', 0.80))

        # ---- 课堂内隐蔽使用 ----
        if (sig.is_weekday
            and sig.time_slot in ('FORENOON', 'AFTERNOON')
            and sig.raw.deviceType == 'PHONE'
            and sig.raw.orientation == 'PORTRAIT'
            and history.school_hours_active_ratio > 0.3
        ):
            candidates.append(SceneCandidate('CLASSROOM_STEALTH', 0.65))

        # ---- 等待间隙偶发学习 ----
        if (sig.raw.motionState == 'STATIONARY'
            and sig.raw.idleSeconds is not None
            and sig.raw.idleSeconds < 10
            and history.app_switch_count_recent >= 3
            and sig.time_slot in ('NOON', 'DUSK')
        ):
            candidates.append(SceneCandidate('WAITING_SPORADIC', 0.60))

        # ---- 考前长时间备考 ----
        if (sig.on_wifi
            and sig.raw.deviceType in ('DESKTOP', 'TABLET')
            and sig.time_slot in ('FORENOON', 'AFTERNOON', 'EVENING', 'LATE_NIGHT')
            and history.exam_within_days <= 14
            and history.avg_session_minutes_recent >= 45
        ):
            candidates.append(SceneCandidate('LONG_SESSION_EXAM_PREP', 0.88))

        # ---- 考后放松态 ----
        if (history.days_since_last_exam <= 3
            and history.avg_session_minutes_recent < 10
            and sig.time_slot in ('EVENING', 'LATE_NIGHT')
        ):
            candidates.append(SceneCandidate('POST_EXAM_RELAXED', 0.55))

        # 默认兜底
        if not candidates:
            candidates.append(SceneCandidate('UNKNOWN', 0.30))

        # 排序
        candidates.sort(key=lambda c: c.confidence, reverse=True)
        return candidates[:3]
```

#### 4.2.2 ML 兜底层

当规则层输出的最高置信度 < 0.60 时，启用轻量梯度提升模型：

```python
class MLSceneClassifier:
    """轻量GBDT场景分类器（ONNX推理）"""

    FEATURE_COLUMNS = [
        'hour', 'is_weekday', 'device_type_code', 'screen_size',
        'network_type_code', 'on_wifi', 'on_cellular',
        'battery_level', 'is_charging', 'motion_state_code',
        'orientation_code', 'idle_seconds', 'is_split_screen',
        'recent_app_switch_count', 'avg_session_min_recent',
        'days_to_next_exam', 'current_subject_code',
    ]

    def __init__(self, model_path: str):
        self.session = ort.InferenceSession(model_path)
        self.label_encoder = SceneLabelEncoder()  # 场景码 ↔ 数值

    def predict(self, features: np.ndarray) -> list[SceneCandidate]:
        input_feed = {self.session.get_inputs()[0].name: features.astype(np.float32)}
        output = self.session.run(None, input_feed)[0]  # softmax 概率
        top3_idx = np.argsort(output)[::-1][:3]
        return [
            SceneCandidate(
                scene=self.label_encoder.decode(idx),
                confidence=float(output[idx]),
            )
            for idx in top3_idx
        ]
```

**模型训练**：使用历史场景推断结果（规则标注）作为弱标签训练，每月使用用户反馈校准数据重训。模型大小控制在 2MB 以内，ONNX 部署。

### 4.3 场景推断融合

```python
class SceneInferenceEngine:
    """场景推断引擎入口"""

    def __init__(
        self,
        preprocessor: SignalPreprocessor,
        rule_classifier: RuleBasedSceneClassifier,
        ml_classifier: MLSceneClassifier,
        strategy_resolver: AdaptationStrategyResolver,
        cache: SceneCache,
    ):
        self.preprocessor = preprocessor
        self.rule_classifier = rule_classifier
        self.ml_classifier = ml_classifier
        self.strategy_resolver = strategy_resolver
        self.cache = cache

    async def infer(
        self,
        signal: DeviceContextSignal,
        student_id: str,
        user_timezone: str,
    ) -> AdaptationStrategy:
        # 1. 检查缓存（同一设备 5 分钟内复用）
        cache_key = f"scene:{student_id}:{signal.deviceId}"
        if cached := await self.cache.get(cache_key):
            if not cached.is_expired():
                return cached.strategy

        # 2. 信号预处理
        processed = self.preprocessor.process(signal, user_timezone)

        # 3. 加载设备历史
        history = await self._load_device_history(student_id, signal.deviceId)

        # 4. 规则分类
        candidates = self.rule_classifier.classify(processed, history)

        # 5. 规则置信度不足，启用 ML 兜底
        if candidates[0].confidence < 0.60:
            features = self._build_features(processed, history)
            ml_candidates = self.ml_classifier.predict(features)
            # 融合：取两者加权平均
            candidates = self._merge_candidates(candidates, ml_candidates)

        # 6. 构建推断结果
        result = SceneInferenceResult(
            primary_scene=candidates[0].scene,
            ranked_scenes=candidates,
            evidence=self._build_evidence(processed, history),
            inferred_at=datetime.utcnow(),
        )

        # 7. 解析适配策略
        strategy = self.strategy_resolver.resolve(result, processed, history)

        # 8. 写入缓存
        await self.cache.set(cache_key, CachedScene(result, strategy), ttl=strategy.ttl_seconds)

        # 9. 异步持久化
        asyncio.create_task(self._persist(student_id, signal.deviceId, result, strategy))

        return strategy

    def _merge_candidates(
        self, rule_cands: list[SceneCandidate], ml_cands: list[SceneCandidate]
    ) -> list[SceneCandidate]:
        """规则与ML结果加权融合，规则权重 0.6，ML 权重 0.4"""
        merged: dict[str, float] = {}
        for c in rule_cands:
            merged[c.scene] = merged.get(c.scene, 0) + c.confidence * 0.6
        for c in ml_cands:
            merged[c.scene] = merged.get(c.scene, 0) + c.confidence * 0.4
        result = [SceneCandidate(s, p) for s, p in sorted(merged.items(), key=lambda x: -x[1])]
        return result[:3]
```

### 5.2 策略动态调整

```python
class AdaptationStrategyResolver:
    """根据场景推断结果 + 个性化偏好，输出最终策略"""

    async def resolve(
        self,
        scene_result: SceneInferenceResult,
        processed: ProcessedSignal,
        history: DeviceHistory,
    ) -> AdaptationStrategy:
        scene_code = scene_result.primary_scene

        # 获取场景默认策略
        base_strategy = await self._load_strategy(scene_code)
        if base_strategy is None:
            base_strategy = await self._load_strategy('UNKNOWN')

        # 个性化覆写
        strategy = self._apply_personal_overrides(base_strategy, history)

        # 低电量覆写：电量<15%时降级处理
        if processed.low_battery and not processed.is_charging:
            strategy = self._apply_low_battery_override(strategy)

        # 弱网覆写：蜂窝3G时禁用视频
        if processed.on_cellular and processed.raw.networkType == 'CELLULAR_3G':
            strategy = self._apply_weak_network_override(strategy)

        return strategy

    def _apply_personal_overrides(
        self, base: AdaptationStrategy, history: DeviceHistory
    ) -> AdaptationStrategy:
        """根据用户历史偏好微调策略"""
        # 用户习惯长会话 → 放宽最大时长
        if history.avg_session_minutes_recent > base.maxSessionMinutes * 1.5:
            base.maxSessionMinutes = int(base.maxSessionMinutes * 1.3)

        # 用户偏好语音交互
        if history.voice_interaction_ratio > 0.4:
            base.preferVoiceInteraction = True

        # 用户频繁忽略休息提醒 → 降低休息提醒频率
        if history.break_remind_dismiss_rate > 0.7:
            base.breakIntervalMinutes = 0

        return base

    def _apply_low_battery_override(self, strategy: AdaptationStrategy) -> AdaptationStrategy:
        """低电量降级：禁用视频，推荐文本/闪卡"""
        strategy.contentFormatPriority = [
            f for f in strategy.contentFormatPriority
            if f not in ('LONG_VIDEO', 'MICRO_VIDEO')
        ]
        strategy.contentFormatPriority.extend(['TEXT_CARD', 'FLASHCARD'])
        strategy.maxSessionMinutes = min(strategy.maxSessionMinutes, 15)
        return strategy

    def _apply_weak_network_override(self, strategy: AdaptationStrategy) -> AdaptationStrategy:
        """弱网降级：禁用视频，优先文本和音频"""
        strategy.contentFormatPriority = [
            f for f in strategy.contentFormatPriority
            if f not in ('LONG_VIDEO', 'MICRO_VIDEO')
        ]
        if 'AUDIO_ONLY' not in strategy.contentFormatPriority:
            strategy.contentFormatPriority.insert(0, 'AUDIO_ONLY')
        return strategy
```

```python
class RecommendationService:
    async def recommend(self, student_id: str, device_id: str) -> list[ContentItem]:
        # 获取场景策略
        strategy = await self.scene_engine.get_strategy(student_id, device_id)

            c for c in candidates
            if c.estimated_minutes <= strategy.maxSessionMinutes

        return candidates[:20]
```

```python
class DeviceHistoryUpdater:
    """消费Kafka事件流，实时更新设备历史数据"""

    async def on_learning_session_end(self, event: SessionEndEvent):
        """学习会话结束时更新平均时长"""
        key = f"device_history:{event.student_id}:{event.device_id}"
        pipe = self.redis.pipeline()

        # 维护近3次会话时长的滑动窗口
        pipe.lpush(f"{key}:recent_sessions", event.duration_minutes)
        pipe.ltrim(f"{key}:recent_sessions", 0, 2)
        pipe.set(f"{key}:last_updated", time.time())
        await pipe.execute()

    async def on_app_switch(self, event: AppSwitchEvent):
        """APP切换事件：维护10分钟内切换计数"""
        key = f"device_history:{event.student_id}:{event.device_id}:switches"
        now = time.time()
        pipe = self.redis.pipeline()
        pipe.zadd(key, {str(now): now})
        pipe.zremrangebyscore(key, 0, now - 600)  # 清理10分钟前
        pipe.execute()

    async def on_break_remind(self, event: BreakRemindEvent):
        """休息提醒响应事件"""
        key = f"device_history:{event.student_id}:{event.device_id}:break"
        # 记录接受/忽略
        if event.action == 'DISMISS':
            await self.redis.hincrby(key, 'dismiss_count', 1)
        else:
            await self.redis.hincrby(key, 'accept_count', 1)
```

## 9. 错误处理与降级策略

### 9.1 异常场景处理

| 异常场景 | 处理策略 | 错误码 |
| --- | --- | --- |
| 客户端信号缺失字段 | 用默认值补全，不阻断流程 | SCENE_SIGNAL_INCOMPLETE |

| 上报频率异常（>10次/分钟） | 触发限流，返回最近缓存策略 | SCENE_RATE_LIMITED |

### 9.2 降级链路

```python
class SceneServiceWithFallback:
    """带降级链路的场景服务"""

    async def get_strategy(self, student_id: str, device_id: str) -> AdaptationStrategy:
        try:
            # 正常链路：缓存 → 推断
            return await self.engine.infer_and_get(student_id, device_id)
        except CacheDownException:
            # 降级1：跳过缓存，直接推断
            return await self.engine.infer_without_cache(student_id, device_id)
        except ModelUnavailableException:
            # 降级2：仅规则推断
            return await self.engine.infer_rules_only(student_id, device_id)
        except Exception:
            # 最终兜底：返回UNKNOWN默认策略
            return self.default_strategy
```

```typescript
class ContextHeartbeatCollector {
  private buffer: DeviceContextSignal[] = [];
  private timer: NodeJS.Timeout | null = null;

  push(signal: DeviceContextSignal) {
    this.buffer.push(signal);

    // 弱网时降低上报频率，攒满10条或60秒后批量发送
    if (this.buffer.length >= 10 || !this.timer) {
      this.timer = setTimeout(() => this.flush(), 60000);
    }
  }

  async flush() {
    if (this.buffer.length === 0) return;
    const batch = this.buffer.splice(0);
    await api.post('/api/v1/scene/context-heartbeat-batch', { signals: batch });
  }
}
```
