# 客户端系统权限管理与运行时授权框架 - 详细设计

> 版本：v1.0 | 日期：2026-05-24 | 状态：初稿

## 1. 模块概述

### 1.1 功能定位

客户端系统权限管理框架是 PrimeTop Flutter 客户端的基础设施层，统一管理应用所需的全部系统级权限（相机、麦克风、存储、通知、定位等），提供运行时权限请求、状态查询、拒绝降级和引导设置的标准化方案。

作为面向全学段未成年人的教育产品，PrimeTop 对系统权限有特殊的合规要求：
- **最小必要原则**：仅申请功能所必需的权限
- **即时请求原则**：在用户触发具体功能时才请求对应权限，不提前批量请求
- **透明告知原则**：请求前需说明用途，符合《儿童个人信息网络保护规定》
- **家长知情原则**：敏感权限（相机、麦克风）需在家长中心同步展示授权状态

### 1.2 设计目标

| 目标 | 说明 |
|------|------|
| 统一管理 | 所有权限通过单一框架管理，避免各模块自行处理 |
| 优雅降级 | 权限被拒绝时提供替代方案，不阻塞核心学习流程 |
| 用户体验 | 请求时机合理，说明文案适龄，引导操作清晰 |
| 可观测性 | 权限状态可查询、可追踪，用于埋点和运营分析 |
| 合规审计 | 记录所有权限请求历史，支持合规审查 |

### 1.3 权限全景图

| 权限 | Android 声明 | iOS Info.plist Key | 使用场景 | 优先级 | 分龄要求 |

### 2.2 权限状态

```dart
/// 权限授权状态
enum PermissionStatus {
  /// 尚未请求过（首次）
  notDetermined,

  /// 已授权
  granted,

  /// 已拒绝，但可以再次弹出系统对话框
  denied,

  /// 已永久拒绝（用户勾选"不再询问"并拒绝）
  permanentlyDenied,

  /// 仅在使用期间授权（iOS 14+ 精确位置等）
  grantedForUsage,

  /// 受限（家长控制等）
  restricted,

  /// 该平台不支持此权限
  notSupported;
}

/// 权限请求结果
class PermissionResult {
  final AppPermission permission;
  final PermissionStatus status;
  final bool isFromUserAction;   // 是否由用户主动触发
  final Duration requestDuration; // 请求耗时
  final DateTime timestamp;

  const PermissionResult({
    required this.permission,
    required this.status,
    required this.isFromUserAction,
    required this.requestDuration,
    required this.timestamp,
  });

  /// 是否可以继续执行功能
  bool get canProceed => status == PermissionStatus.granted
      || status == PermissionStatus.grantedForUsage;

  Map<String, dynamic> toAnalyticsMap() => {
    'permission': permission.key,
    'status': status.name,
    'from_user_action': isFromUserAction,
    'duration_ms': requestDuration.inMilliseconds,
    'timestamp': timestamp.toIso8601String(),
  };
}
```

```dart
/// 权限被拒绝后的降级策略
class DenialStrategy {
  /// 降级类型
  final DenialType type;

  /// 降级后提示文案
  final String fallbackMessage;

  /// 降级后可使用的替代功能
  final List<FallbackAction> fallbackActions;

  /// 引导跳转设置页的文案
  final String settingsGuideText;

  const DenialStrategy({
    required this.type,
    required this.fallbackMessage,
    this.fallbackActions = const [],
    required this.settingsGuideText,
  });
}

enum DenialType {
  /// 提供替代输入方式（如相机被拒 → 文字输入）
  alternativeInput,

  /// 使用只读/受限模式（如存储被拒 → 仅在线模式）
  limitedMode,

  /// 功能完全不可用，展示引导卡片
  featureUnavailable,

  /// 静默降级，不影响用户体验
  silent,
}

class FallbackAction {
  final String label;
  final String route;       // 跳转路由
  final String? description;

  const FallbackAction({
    required this.label,
    required this.route,
    this.description,
  });
}
```

## 4. 权限管理器核心实现

### 4.1 PermissionManager 接口

```dart
/// 权限管理器 — 应用级单例，通过 Riverpod Provider 注入
abstract class PermissionManager {
  /// 查询单个权限当前状态
  Future<PermissionStatus> checkStatus(AppPermission permission);

  /// 批量查询权限状态
  Future<Map<AppPermission, PermissionStatus>> checkStatuses(
    List<AppPermission> permissions,
  );

  /// 请求单个权限（核心方法）
  /// [context] 用于展示 Rationale 对话框
  /// [fromUserAction] 是否由用户主动点击触发
  Future<PermissionResult> request(
    AppPermission permission, {
    required BuildContext context,
    bool fromUserAction = true,
  });

  /// 请求多个权限（按顺序逐个请求）
  Future<Map<AppPermission, PermissionResult>> requestMultiple(
    List<AppPermission> permissions, {
    required BuildContext context,
  });

  /// 打开系统应用设置页
  Future<bool> openAppSettings();

  /// 监听权限状态变化流
  Stream<PermissionStatus> watchStatus(AppPermission permission);

  /// 获取权限请求历史记录（用于埋点和审计）
  List<PermissionResult> getRequestHistory({AppPermission? filter});

  /// 清除请求历史（用户注销时调用）
  void clearHistory();
}
```

### 4.2 核心实现

```dart
class PermissionManagerImpl implements PermissionManager {
  final PermissionPlugin _plugin;  // 底层权限插件封装
  final PermissionConfigProvider _configProvider;
  final AnalyticsService _analytics;
  final Logger _logger;
  final PermissionHistoryStore _historyStore;

  /// 请求计数器：记录本次会话中各权限的请求次数
  final Map<AppPermission, int> _sessionRequestCount = {};

  @override
  Future<PermissionResult> request(
    AppPermission permission, {
    required BuildContext context,
    bool fromUserAction = true,
  }) async {
    final config = PermissionRegistry.getConfig(permission);
    final stopwatch = Stopwatch()..start();

    // 1. 先检查当前状态
    var status = await checkStatus(permission);
    if (status == PermissionStatus.granted) {
      return PermissionResult(
        permission: permission,
        status: status,
        isFromUserAction: fromUserAction,
        requestDuration: stopwatch.elapsed,
        timestamp: DateTime.now(),
      );
    }

    // 2. 判断是否已永久拒绝
    if (status == PermissionStatus.permanentlyDenied) {
      // 直接引导到设置页
      _logger.info('权限已永久拒绝: ${permission.key}, 引导设置页');
      final opened = await openAppSettings();
      // 等待用户从设置页返回后重新检查
      if (opened) {
        await _waitForSettingsReturn();
        status = await checkStatus(permission);
      }
      final result = PermissionResult(
        permission: permission,
        status: status,
        isFromUserAction: fromUserAction,
        requestDuration: stopwatch.elapsed,
        timestamp: DateTime.now(),
      );
      _recordResult(result);
      return result;
    }

    // 3. 判断请求次数是否已达上限
    final count = _sessionRequestCount[permission] ?? 0;
    if (count >= config.maxRetryCount) {
      _logger.info('权限请求次数已达上限: ${permission.key} ($count/${config.maxRetryCount})');
      return PermissionResult(
        permission: permission,
        status: status,
        isFromUserAction: fromUserAction,
        requestDuration: stopwatch.elapsed,
        timestamp: DateTime.now(),
      );
    }

    // 4. 展示 Rationale 说明（Android / iOS 均适用）
    if (fromUserAction && context.mounted) {
      final shouldRequest = await _showRationaleDialog(
        context: context,
        config: config,
      );
      if (!shouldRequest) {
        return PermissionResult(
          permission: permission,
          status: PermissionStatus.denied,
          isFromUserAction: fromUserAction,
          requestDuration: stopwatch.elapsed,
          timestamp: DateTime.now(),
        );
      }
    }

    // 5. 调用系统权限请求
    _sessionRequestCount[permission] = count + 1;
    status = await _plugin.request(permission);

    stopwatch.stop();

    final result = PermissionResult(
      permission: permission,
      status: status,
      isFromUserAction: fromUserAction,
      requestDuration: stopwatch.elapsed,
      timestamp: DateTime.now(),
    );

    // 6. 记录结果
    _recordResult(result);

    // 7. 如果被拒绝，展示降级引导
    if (!result.canProceed && context.mounted) {
      _showDenialGuidance(context: context, config: config, result: result);
    }

    return result;
  }

  /// 记录请求结果（用于埋点和审计）
  void _recordResult(PermissionResult result) {
    _historyStore.add(result);
    _analytics.trackEvent(
      eventName: 'permission_request',
      properties: result.toAnalyticsMap(),
    );
    _logger.info(
      '权限请求完成: ${result.permission.key} => ${result.status.name} '
      '(${result.requestDuration.inMilliseconds}ms)',
    );
  }

  /// 等待用户从系统设置页返回
  Future<void> _waitForSettingsReturn() async {
    // 使用 AppLifecycleState 监听应用回到前台
    final completer = Completer<void>();
    final sub = WidgetsBinding.instance.addObserver(
      _AppResumeObserver(() => completer.complete()),
    );
    // 超时 30 秒自动继续
    return completer.future.timeout(
      const Duration(seconds: 30),
      onTimeout: () {},
    ).whenComplete(() {
      WidgetsBinding.instance.removeObserver(sub);
    });
  }
}
```

```dart
/// 分龄文案 + 功能说明图示
Future<bool> _showRationaleDialog({
  required BuildContext context,
  required PermissionConfig config,
}) async {
  final gradeLevel = context.read<GradeLevelProvider>().current;
  final rationale = config.rationaleText[gradeLevel]
      ?? config.rationaleText[GradeLevel.middle]!;

  final result = await showDialog<bool>(
    context: context,
    barrierDismissible: false,
    builder: (ctx) => _PermissionRationaleDialog(
      permission: config.permission,
      message: rationale,
      featureModules: config.permission.featureModules,
    ),
  );

  return result ?? false;
}
```

```dart
/// 权限被拒绝后的降级引导
void _showDenialGuidance({
  required BuildContext context,
  required PermissionConfig config,
  required PermissionResult result,
}) {
  final strategy = config.denialStrategy;

  switch (strategy.type) {
    case DenialType.alternativeInput:
      // 展示替代操作 SnackBar
      if (strategy.fallbackActions.isNotEmpty) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(strategy.fallbackMessage),
            duration: const Duration(seconds: 5),
            action: strategy.fallbackActions.isNotEmpty
                ? SnackBarAction(
                    label: strategy.fallbackActions.first.label,
                    onPressed: () => context.push(strategy.fallbackActions.first.route),
                  )
                : null,
          ),
        );
      }

    case DenialType.limitedMode:
      // 展示受限模式提示
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(strategy.fallbackMessage),
          duration: const Duration(seconds: 4),
        ),
      );

    case DenialType.featureUnavailable:
      // 展示完整引导卡片
      _showPermissionGuideCard(
        context: context,
        permission: config.permission,
        guideText: strategy.settingsGuideText,
        fallbackActions: strategy.fallbackActions,
      );

    case DenialType.silent:
      // 静默降级，不展示任何 UI
      break;
  }
}
```

```dart
/// 单个权限状态 Provider（自动刷新）
final permissionStatusProvider =
    FutureProvider.family<PermissionStatus, AppPermission>((ref, permission) {
  final manager = ref.watch(permissionManagerProvider);

  // 监听应用生命周期，回到前台时刷新
  ref.listen(appLifecycleProvider, (_, __) {
    ref.invalidateSelf();
  });

  return manager.checkStatus(permission);
});

/// 批量权限状态 Provider
final permissionStatusesProvider =
    FutureProvider.family<Map<AppPermission, PermissionStatus>,
                        List<AppPermission>>((ref, permissions) {
  final manager = ref.watch(permissionManagerProvider);
  return manager.checkStatuses(permissions);
});

/// 权限请求历史 Provider
final permissionHistoryProvider = Provider<List<PermissionResult>>((ref) {
  final manager = ref.watch(permissionManagerProvider);
```
