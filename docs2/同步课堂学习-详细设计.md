# 同步课堂学习 - 详细设计文档

## 1. 模块概述

### 1.1 功能定位

同步课堂学习模块围绕国内主流教材版本，按学段、年级、学科、章节组织学习内容，使 AI 辅导能力与学生当前课堂进度保持一致。是 MVP 阶段 P0 核心功能。

### 1.2 核心能力

| 能力 | 说明 | 优先级 |
|------|------|--------|
| 教材版本选择 | 用户选择使用的教材版本（人教版、苏教版等） | P0 |
| 章节目录浏览 | 按学科 → 学期 → 单元 → 课时层级展示 | P0 |
| 章节学习内容 | 课前预习、知识讲解、例题、课后复习 | P0 |
| 同步练习 | 基于章节知识点生成/推荐练习题 | P1 |
| 学习进度记录 | 记录章节完成状态与掌握程度 | P0 |

### 1.3 依赖关系

```
用户账号体系 → 学生档案（年级、学段、教材版本）
AI 智能辅导 → 章节知识点讲解、RAG 检索
拍照搜题与习题答疑 → 同步练习题目
内容与运营后台 → 教材、章节、知识点数据维护
```

## 2. 数据结构定义

### 2.1 教材版本 (textbook)

```sql
CREATE TABLE textbook (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    name            VARCHAR(100) NOT NULL COMMENT '教材名称，如：人教版数学',
    publisher       VARCHAR(50) NOT NULL COMMENT '出版社：人教版、苏教版、北师大版、外研版等',
    subject_code    VARCHAR(20) NOT NULL COMMENT '学科编码：MATH, CHINESE, ENGLISH, PHYSICS 等',
    stage           VARCHAR(20) NOT NULL COMMENT '学段：PRESCHOOL, PRIMARY, JUNIOR, SENIOR',
    grade_start     TINYINT NOT NULL COMMENT '适用起始年级 1=一年级, 7=初一, 10=高一',
    grade_end       TINYINT NOT NULL COMMENT '适用结束年级',
    semester_type   VARCHAR(10) NOT NULL COMMENT 'FIRST=上册, SECOND=下册, FULL=全一册',
    edition_year    VARCHAR(10) COMMENT '版本年份，如 2019, 2022',
    is_active       TINYINT(1) DEFAULT 1 COMMENT '是否启用',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_subject_stage (subject_code, stage),
    INDEX idx_publisher (publisher)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='教材版本表';
```

### 2.2 章节 (chapter)

```sql
CREATE TABLE chapter (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    textbook_id     BIGINT NOT NULL COMMENT '关联教材版本',
    parent_id       BIGINT DEFAULT NULL COMMENT '父章节ID（支持多级树形结构）',
    level           TINYINT NOT NULL COMMENT '层级：1=册/学期, 2=单元, 3=课时/小节',
    sort_order      INT NOT NULL DEFAULT 0 COMMENT '同层级排序号',
    title           VARCHAR(200) NOT NULL COMMENT '章节标题',
    description     TEXT COMMENT '章节简介',
    knowledge_tags  JSON COMMENT '知识点标签列表 ["二次函数", "顶点式"]',

    -- 学习内容（富文本，支持 MathJax/LaTeX 公式）
    preview_content MEDIUMTEXT COMMENT '课前预习内容（Markdown/HTML）',
    summary_content MEDIUMTEXT COMMENT '课后复习总结（Markdown/HTML）',
    key_points      JSON COMMENT '重点要点列表 [{"title":"...", "content":"..."}]',

    -- 练习配置
    practice_config JSON COMMENT '练习配置 {"basic_count":5, "medium_count":3, "hard_count":2}',

    is_active       TINYINT(1) DEFAULT 1,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_textbook (textbook_id),
    INDEX idx_parent (parent_id),
    FOREIGN KEY (textbook_id) REFERENCES textbook(id),
    FOREIGN KEY (parent_id) REFERENCES chapter(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='教材章节表';
```

**knowledge_tags JSON 结构示例：**

```json
["一元二次方程", "求根公式", "判别式", "韦达定理"]
```

**key_points JSON 结构示例：**

```json
[
  {
    "title": "一元二次方程的标准形式",
    "content": "ax² + bx + c = 0 (a ≠ 0)，其中 a、b、c 分别叫二次项系数、一次项系数和常数项。"
  },
  {
    "title": "求根公式",
    "content": "x = (-b ± √(b²-4ac)) / 2a，其中 Δ = b²-4ac 叫做判别式。"
  }
]
```

### 2.3 知识点 (knowledge_point)

```sql
CREATE TABLE knowledge_point (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    subject_code    VARCHAR(20) NOT NULL COMMENT '学科编码',
    stage           VARCHAR(20) NOT NULL COMMENT '学段',
    code            VARCHAR(50) NOT NULL UNIQUE COMMENT '知识点编码，如 MATH-J3-QUADRATIC-001',
    name            VARCHAR(200) NOT NULL COMMENT '知识点名称',
    parent_id       BIGINT DEFAULT NULL COMMENT '父知识点（支持层级结构）',
    description     TEXT COMMENT '知识点说明',
    difficulty      TINYINT COMMENT '难度等级 1-5',
    prerequisites   JSON COMMENT '前置知识点ID列表 [1001, 1002]',
    exam_weight     DECIMAL(3,1) COMMENT '考试权重/占比（百分比），如 15.0',
    
    -- RAG 向量检索用
    embedding_id    VARCHAR(100) COMMENT '向量数据库中对应的文档ID',
    
    is_active       TINYINT(1) DEFAULT 1,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_subject_stage (subject_code, stage),
    INDEX idx_parent (parent_id),
    FOREIGN KEY (parent_id) REFERENCES knowledge_point(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='知识点表';
```

### 2.4 章节-知识点关联 (chapter_knowledge)

```sql
CREATE TABLE chapter_knowledge (
    id                  BIGINT PRIMARY KEY AUTO_INCREMENT,
    chapter_id          BIGINT NOT NULL,
    knowledge_point_id  BIGINT NOT NULL,
    relevance_type      VARCHAR(20) NOT NULL COMMENT '关联类型：CORE=核心, EXTENSION=拓展, PREREQUISITE=前置',
    sort_order          INT DEFAULT 0,
    UNIQUE KEY uk_ch_kp (chapter_id, knowledge_point_id),
    FOREIGN KEY (chapter_id) REFERENCES chapter(id),
    FOREIGN KEY (knowledge_point_id) REFERENCES knowledge_point(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='章节-知识点关联表';
```

### 2.5 学习进度 (learning_progress)

```sql
CREATE TABLE learning_progress (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id         BIGINT NOT NULL COMMENT '用户ID',
    chapter_id      BIGINT NOT NULL COMMENT '章节ID',
    
    -- 进度状态
    status          VARCHAR(20) NOT NULL DEFAULT 'NOT_STARTED' 
                    COMMENT '状态：NOT_STARTED, PREVIEWING, STUDYING, PRACTICING, COMPLETED',
    mastery_level   TINYINT DEFAULT 0 COMMENT '掌握程度 0-100（百分制）',
    
    -- 时间记录
    preview_done    TINYINT(1) DEFAULT 0 COMMENT '是否完成预习',
    preview_at      DATETIME COMMENT '预习完成时间',
    study_done      TINYINT(1) DEFAULT 0 COMMENT '是否完成学习',
    study_at        DATETIME COMMENT '学习完成时间',
    practice_done   TINYINT(1) DEFAULT 0 COMMENT '是否完成练习',
    practice_at     DATETIME COMMENT '练习完成时间',
    completed_at    DATETIME COMMENT '整体完成时间',
    
    -- 学习时长（秒）
    total_seconds   INT DEFAULT 0 COMMENT '累计学习时长',
    
    -- 练习结果
    practice_total  TINYINT DEFAULT 0 COMMENT '练习题目总数',
    practice_correct TINYINT DEFAULT 0 COMMENT '正确题目数',
    
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    UNIQUE KEY uk_user_chapter (user_id, chapter_id),
    INDEX idx_user_status (user_id, status),
    INDEX idx_user_mastery (user_id, mastery_level),
    FOREIGN KEY (chapter_id) REFERENCES chapter(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='学习进度表';
```

### 2.6 学生教材配置 (student_textbook)

```sql
CREATE TABLE student_textbook (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id         BIGINT NOT NULL COMMENT '学生用户ID',
    subject_code    VARCHAR(20) NOT NULL COMMENT '学科编码',
    textbook_id     BIGINT NOT NULL COMMENT '选择的教材版本ID',
    is_current      TINYINT(1) DEFAULT 1 COMMENT '是否当前使用',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    UNIQUE KEY uk_user_subject (user_id, subject_code),
    FOREIGN KEY (textbook_id) REFERENCES textbook(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='学生教材配置表';
```

## 3. API 接口设计

### 3.1 基础约定

- **Base URL**: `/api/v1/sync-classroom`
- **认证**: Bearer Token（JWT）
- **通用响应结构**:

```json
{
  "code": 0,
  "message": "success",
  "data": { ... }
}
```

**错误码约定：**

| 错误码 | 含义 |
|--------|------|
| 0 | 成功 |
| 40001 | 参数校验失败 |
| 40101 | 未登录或 Token 过期 |
| 40301 | 无权限 |
| 40401 | 资源不存在 |
| 40901 | 数据冲突（如教材已选） |
| 50001 | 服务内部错误 |

## 4. 状态流转设计

### 4.1 学习进度状态机

```
                    ┌──────────────────────────────────────────┐
                    │                                          ▼
              NOT_STARTED ──START_PREVIEW──▶ PREVIEWING ──COMPLETE_PREVIEW──▶ STUDYING
                                                                  │
                                              COMPLETE_STUDY     │
                                                                  ▼
                                                          PRACTICING
                                                                  │
                                              COMPLETE_PRACTICE   │
                                                                  ▼
                                                           COMPLETED
```

**状态流转规则：**

| 当前状态 | 动作 | 目标状态 | 说明 |
|----------|------|----------|------|
| NOT_STARTED | START_PREVIEW | PREVIEWING | 首次打开章节预习 |
| PREVIEWING | COMPLETE_PREVIEW | STUDYING | 完成预习，进入学习 |
| STUDYING | COMPLETE_STUDY | PRACTICING | 完成学习，进入练习 |
| PRACTICING | COMPLETE_PRACTICE | COMPLETED | 完成练习，标记完成 |
| COMPLETED | START_STUDY | STUDYING | 允许重新学习已完成的章节 |
| PREVIEWING | START_STUDY | STUDYING | 允许跳过预习直接学习 |
| STUDYING | COMPLETE_STUDY | COMPLETED | 允许跳过练习直接完成 |
| * | UPDATE_TIME | * | 不改变状态，仅累加时长 |
| * | UPDATE_MASTERY | * | 不改变状态，仅更新掌握度 |

### 4.2 掌握度计算规则

```
masteryLevel = w1 * previewScore + w2 * studyScore + w3 * practiceScore

其中：
- previewScore: 预习完成得分，完成=20，未完成=0
- studyScore: 学习完成得分，完成=40，未完成=0  
- practiceScore: 练习得分 = 40 * (correctCount / totalCount)
  - 未做练习时 practiceScore = 0
  - 总分 0-100
```

**掌握度等级标签：**

| 分数 | 等级 | 标签 | 颜色 |
|------|------|------|------|
| 0-20 | E | 需要学习 | 🔴 红 |
| 21-40 | D | 初步了解 | 🟠 橙 |
| 41-60 | C | 基本掌握 | 🟡 黄 |
| 61-80 | B | 熟练掌握 | 🟢 绿 |
| 81-100 | A | 优秀 | 🔵 蓝 |

## 5. Prompt 编排设计

### 5.1 课前预习讲解 Prompt

```
你是一位 {stage} {subjectName} 老师，正在给 {grade} 年级的学生做课前预习讲解。

## 章节信息
- 教材：{textbookName}
- 章节：{chapterTitle}
- 核心知识点：{knowledgePoints}

## 教材内容参考
{chapterContent}
```

```python
async def retrieve_context(chapter_id: int, mode: str, user_id: int) -> str:
    """获取章节学习的 RAG 上下文"""
    
    # 1. 获取章节信息
    chapter = await get_chapter(chapter_id)
    
    # 2. 获取关联知识点
    knowledge_points = await get_chapter_knowledge_points(chapter_id)
    
    # 3. 构建查询向量
    query_text = f"{chapter.title} {' '.join(kp.name for kp in knowledge_points)}"
    
    # 4. 向量检索
    results = await vector_search(
        query=query_text,
        filters={
            "subject_code": chapter.subject_code,
            "stage": chapter.stage,
        },
        top_k=5,
        min_score=0.7
    )
    
    # 5. 如果是复习模式，加入薄弱知识点相关内容
    if mode == "REVIEW":
        weak_points = await get_weak_knowledge_points(user_id, chapter_id)
        weak_results = await vector_search(
            query=" ".join(wp.name for wp in weak_points),
            top_k=3,
            min_score=0.65
        )
        results.extend(weak_results)
    
    # 6. 组装上下文
    context_parts = []
    for r in results:
        context_parts.append(f"【{r.title}】\n{r.content}")
    
    return "\n\n---\n\n".join(context_parts)
```

```java
// ChapterService.java
public ChapterTreeDTO getChapterTree(Long userId, String subjectCode, String semesterType) {
    // 1. 获取学生配置的教材
    StudentTextbook st = studentTextbookRepository
        .findByUserIdAndSubjectCodeAndIsCurrent(userId, subjectCode, true)
        .orElseThrow(() -> new BusinessException(40401, "请先选择教材版本"));
    
    // 2. 查询该教材所有章节
    List<Chapter> chapters = chapterRepository
        .findByTextbookIdAndSemesterTypeAndIsActiveTrueOrderBySortOrder(
            st.getTextbookId(), semesterType);
    
    // 3. 查询用户学习进度
    List<Long> chapterIds = chapters.stream().map(Chapter::getId).toList();
    Map<Long, LearningProgress> progressMap = learningProgressRepository
        .findByUserIdAndChapterIdIn(userId, chapterIds)
        .stream().collect(Collectors.toMap(LearningProgress::getChapterId, p -> p));
    
    // 4. 构建树形结构
    List<ChapterNodeDTO> tree = buildTree(chapters, progressMap);
    
    // 5. 统计
    ChapterStatsDTO stats = calculateStats(chapters, progressMap);
    
    return ChapterTreeDTO.builder()
        .textbookId(st.getTextbookId())
        .textbookName(st.getTextbook().getName())
        .chapters(tree)
        .stats(stats)
        .build();
}

private List<ChapterNodeDTO> buildTree(List<Chapter> chapters, 
                                         Map<Long, LearningProgress> progressMap) {
    Map<Long, ChapterNodeDTO> nodeMap = new LinkedHashMap<>();
    List<ChapterNodeDTO> roots = new ArrayList<>();
    
    for (Chapter ch : chapters) {
        LearningProgress p = progressMap.get(ch.getId());
        ChapterNodeDTO node = ChapterNodeDTO.builder()
            .id(ch.getId())
            .title(ch.getTitle())
            .level(ch.getLevel())
            .sortOrder(ch.getSortOrder())
            .status(p != null ? p.getStatus().name() : "NOT_STARTED")
            .masteryLevel(p != null ? p.getMasteryLevel() : 0)
            .children(new ArrayList<>())
            .build();
        nodeMap.put(ch.getId(), node);
        
        if (ch.getParentId() == null) {
            roots.add(node);
        }
    }
    
    // 建立父子关系
    for (Chapter ch : chapters) {
        if (ch.getParentId() != null) {
            ChapterNodeDTO parent = nodeMap.get(ch.getParentId());
            ChapterNodeDTO child = nodeMap.get(ch.getId());
            if (parent != null && child != null) {
                parent.getChildren().add(child);
            }
        }
    }
    
    return roots;
}
```

```java
// LearningProgressService.java
@Transactional
public ProgressDTO updateProgress(Long userId, ProgressUpdateRequest req) {
    LearningProgress progress = learningProgressRepository
        .findByUserIdAndChapterId(userId, req.getChapterId())
        .orElseGet(() -> createNewProgress(userId, req.getChapterId()));
    
    switch (req.getAction()) {
        case START_PREVIEW -> {
            assertCanTransition(progress, ProgressStatus.NOT_STARTED);
            progress.setStatus(ProgressStatus.PREVIEWING);
        }
        case COMPLETE_PREVIEW -> {
            assertCanTransition(progress, ProgressStatus.PREVIEWING);
            progress.setPreviewDone(true);
            progress.setPreviewAt(LocalDateTime.now());
            progress.setStatus(ProgressStatus.STUDYING);
        }
        case COMPLETE_STUDY -> {
            assertCanTransition(progress, ProgressStatus.STUDYING, ProgressStatus.PREVIEWING);
            progress.setStudyDone(true);
            progress.setStudyAt(LocalDateTime.now());
            // 如果已完成练习则直接标记完成，否则进入练习
            if (progress.getPracticeDone()) {
                progress.setStatus(ProgressStatus.COMPLETED);
                progress.setCompletedAt(LocalDateTime.now());
            } else {
                progress.setStatus(ProgressStatus.PRACTICING);
            }
        }
        case COMPLETE_PRACTICE -> {
            assertCanTransition(progress, ProgressStatus.PRACTICING, ProgressStatus.STUDYING);
            progress.setPracticeDone(true);
            progress.setPracticeAt(LocalDateTime.now());
            progress.setPracticeTotal(req.getPracticeTotal());
            progress.setPracticeCorrect(req.getPracticeCorrect());
            progress.setStatus(ProgressStatus.COMPLETED);
            progress.setCompletedAt(LocalDateTime.now());
        }
        case UPDATE_TIME -> {
            progress.setTotalSeconds(progress.getTotalSeconds() + req.getSeconds());
        }
        case UPDATE_MASTERY -> {
            progress.setMasteryLevel(req.getMasteryLevel());
        }
        default -> throw new BusinessException(40001, "不支持的动作: " + req.getAction());
    }
    
    // 自动计算掌握度
    if (req.getAction() != ProgressAction.UPDATE_MASTERY) {
        progress.setMasteryLevel(calculateMastery(progress));
    }
    
    learningProgressRepository.save(progress);
    return ProgressDTO.from(progress);
}

private int calculateMastery(LearningProgress p) {
    int previewScore = p.getPreviewDone() ? 20 : 0;
    int studyScore = p.getStudyDone() ? 40 : 0;
    int practiceScore = 0;
    if (p.getPracticeDone() && p.getPracticeTotal() > 0) {
        practiceScore = (int) (40.0 * p.getPracticeCorrect() / p.getPracticeTotal());
    }
    return Math.min(100, previewScore + studyScore + practiceScore);
}
```
