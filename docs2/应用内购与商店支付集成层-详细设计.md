# 应用内购与商店支付集成层 — 详细设计

> 版本：v1.0 | 更新日期：2026-06-04 | 状态：草案

---

## 1. 模块概述

### 1.1 背景

PrimeTop 作为面向未成年人的教育类 APP，需在 Android（Google Play、国内应用商店）和 iOS（Apple App Store）双平台上架并实现付费订阅和增值服务购买。各支付渠道的 SDK 接入、票据验证、订阅管理、合规要求和退款策略差异显著，需要统一封装一层**商店支付集成层**，对上层业务屏蔽底层差异。

### 1.2 设计目标

1. **统一接口**：上层业务（支付与会员订阅模块）通过统一抽象接口调用支付能力，无需关心具体支付渠道。
2. **安全可靠**：所有支付凭据必须经过服务端二次验证，防止伪造和重放攻击。
3. **合规达标**：满足 Apple App Store Review Guidelines、Google Play政策以及国内应用商店对儿童/教育类 APP 的特殊要求。
4. **对账准确**：支持多渠道订单对账，保证服务端订单状态与商店实际状态一致。
5. **优雅降级**：商店接口异常时提供明确错误反馈，避免用户资金损失。

### 1.3 范围与优先级

| 功能 | 渠道 | 优先级 | MVP |
|------|------|--------|-----|
| Apple IAP — 消耗型商品（次数包） | iOS | P0 | ✅ |
| Apple IAP — 自动续期订阅 | iOS | P0 | ✅ |
| Apple IAP — 非消耗型商品（专项包） | iOS | P1 | ❌ |
| Google Play Billing — 一次性购买 | Android | P0 | ✅ |
| Google Play Billing — 自动续期订阅 | Android | P0 | ✅ |
| 微信支付 APP 支付 | 国内 Android | P0 | ✅ |
| 支付宝 APP 支付 | 国内 Android | P0 | ✅ |
| Apple IAP 退款通知处理 | iOS | P1 | ❌ |

| 华为 IAP | 华为渠道 | P2 | ❌ |
| 小米/OPPO/vivo 应用商店支付 | 国内渠道 | P3 | ❌ |

## 2. 核心概念

| 术语 | 定义 |
|------|------|
| IAP | In-App Purchase，苹果应用内购买 |
| SKU / Product ID | 商店中配置的商品标识符，与 PrimeTop 内部商品码映射 |
| Receipt / 票据 | 购买成功后商店返回的凭据数据，需服务端验证 |
| Transaction ID | 商店侧唯一交易标识，用于去重和对账 |
| Original Transaction ID | Apple 订阅的原始交易 ID（续费交易共享同一 original ID） |
| Purchase Token | Google Play 购买令牌，用于服务端 API 查询和确认 |
| Subscription | 自动续期订阅，周期结束自动扣款 |
| One-time Purchase | 一次性购买（消耗型/非消耗型） |
| Grace Period | 宽限期，订阅扣款失败后保留权益的缓冲时间 |
| Billing Retry | 计费重试，Apple/Google 在扣款失败后自动重试 |
| Voided Purchase | 已退款/撤销的购买记录 |

Future<List<ProductDetails>> loadProducts(List<String> internalCodes) async {
  // 1. 根据内部码查询对应 Apple SKU
  final skuMap = await _paymentApi.getSkuMapping(
    store: 'apple',
    internalCodes: internalCodes,

  final appleSkus = skuMap.values.toList();

    Set.from(appleSkus),

    title: p.title,
    description: p.description,
    price: p.price,
    priceValue: p.rawPrice,
  )).toList();

/// 发起购买
Future<PurchaseResult> initiatePurchase(ProductDetails product) async {
  final completer = Completer<PurchaseResult>();

  // 监听购买流
  _purchaseStream.listen(
    (purchaseDetailsList) async {
      for (final details in purchaseDetailsList) {
        if (details.productID != product.storeSku) continue;

        switch (details.status) {
          case PurchaseStatus.purchased:
            // 3. 提交 Receipt 到服务端验证
            final result = await _verifyAndDeliver(details);
            if (result.success) {
              details.completePurchase; // 告知 StoreKit 交易完成
              completer.complete(PurchaseResult.success(result.orderId!));
            } else {
              completer.complete(PurchaseResult.verificationFailed(result.error!));
            }
            break;

          case PurchaseStatus.error:
            completer.complete(PurchaseResult.error(details.error!));
            break;

          case PurchaseStatus.canceled:
            completer.complete(PurchaseResult.canceled);
            break;

          case PurchaseStatus.pending:
            // 需要家长批准（Ask to Buy）或其他审批
            completer.complete(PurchaseResult.pending);
            break;

          default:
            break;
        }
      }
    },
    onDone: () => completer.complete(PurchaseResult.unknown),
  );

  // 发起购买请求
  final purchaseParam = PurchaseParam(
    productDetails: product.storeProduct,
  );
  _inAppPurchase.buyNonConsumable(purchaseParam: purchaseParam);

  return completer.future;
}

#### 4.3.2 Receipt 验证服务实现

```python
# payment/iap/apple_receipt_verifier.py

import base64
import httpx
from dataclasses import dataclass
from enum import Enum
from typing import Optional

class AppleEnvironment(Enum):
    SANDBOX = "sandbox"
    PRODUCTION = "buy"

@dataclass
class ReceiptVerifyResult:
    success: bool
    transaction_id: Optional[str] = None
    original_transaction_id: Optional[str] = None
    product_id: Optional[str] = None
    purchase_date_ms: Optional[int] = None
    expires_date_ms: Optional[int] = None  # 订阅专用
    is_trial: bool = False
    environment: Optional[str] = None
    error: Optional[str] = None
    # App Store Server API V2 JWS 数据
    jws_transaction: Optional[dict] = None

class AppleReceiptVerifier:
    """Apple Receipt 验证器"""

    PRODUCTION_URL = "https://buy.itunes.apple.com/verifyReceipt"
    SANDBOX_URL = "https://sandbox.itunes.apple.com/verifyReceipt"

    def __init__(self, shared_secret: str, bundle_id: str):
        self.shared_secret = shared_secret
        self.bundle_id = bundle_id

    async def verify(
        self,
        receipt_data: str,
        expected_product_id: Optional[str] = None,
    ) -> ReceiptVerifyResult:
        """验证 Receipt 并返回解析结果"""

        # 1. 先尝试生产环境
        result = await self._call_verify(self.PRODUCTION_URL, receipt_data)
        if result is None:
            return ReceiptVerifyResult(success=False, error="APPLE_API_ERROR")

        # 2. 状态码 21007 表示沙盒 Receipt 发到了生产环境，需重试沙盒
        if result.get("status") == 21007:
            result = await self._call_verify(self.SANDBOX_URL, receipt_data)
            if result is None:
                return ReceiptVerifyResult(success=False, error="APPLE_API_ERROR")

        status = result.get("status")
        if status != 0:
            error_map = {
                21000: "APP_STORE_NOT_ACCESSIBLE",
                21002: "RECEIPT_MALFORMED",
                21003: "RECEIPT_AUTHENTICATED_FAILED",
                21004: "SHARED_SECRET_MISMATCH",
                21005: "RECEIPT_SERVER_UNAVAILABLE",
                21006: "RECEIPT_VALID_BUT_SUB_EXPIRED",
                21008: "NOT_PRODUCTION_RECEIPT",
                21009: "RECEIPT_WITHOUT_PURCHASE",
                21010: "RECEIPT_ACCOUNT_NOT_FOUND",
            }
            return ReceiptVerifyResult(
                success=False,
                error=error_map.get(status, f"UNKNOWN_STATUS_{status}"),
            )

        # 3. 解析 latest_receipt_info（订阅）或 receipt（一次性）
        receipt_info = self._extract_latest_transaction(result)
        if not receipt_info:
            return ReceiptVerifyResult(success=False, error="NO_TRANSACTION_FOUND")

        # 4. 校验 bundle_id
        actual_bundle = result.get("receipt", {}).get("bundle_id", "")
        if actual_bundle != self.bundle_id:
            return ReceiptVerifyResult(
                success=False,
                error="BUNDLE_ID_MISMATCH",
            )

        # 5. 校验 product_id
        product_id = receipt_info.get("product_id", "")
        if expected_product_id and product_id != expected_product_id:
            return ReceiptVerifyResult(
                success=False,
                error="PRODUCT_ID_MISMATCH",
            )

        environment = result.get("environment", "Production")

        return ReceiptVerifyResult(
            success=True,
            transaction_id=receipt_info.get("transaction_id"),
            original_transaction_id=receipt_info.get("original_transaction_id"),
            product_id=product_id,
            purchase_date_ms=int(receipt_info.get("purchase_date_ms", 0)),
            expires_date_ms=self._safe_int(receipt_info.get("expires_date_ms")),
            is_trial=receipt_info.get("is_trial_period") == "true",
            environment=environment,
        )

    async def _call_verify(self, url: str, receipt_data: str) -> Optional[dict]:
        """调用 Apple verifyReceipt API"""
        payload = {
            "receipt-data": receipt_data,
            "password": self.shared_secret,
            "exclude-old-transactions": True,
        }
        async with httpx.AsyncClient(timeout=30) as client:
            try:
                resp = await client.post(url, json=payload)
                resp.raise_for_status()
                return resp.json()
            except (httpx.HTTPError, ValueError):
                return None

    def _extract_latest_transaction(self, result: dict) -> Optional[dict]:
        """提取最新交易记录"""
        # 订阅型：取 latest_receipt_info 的最后一条
        latest = result.get("latest_receipt_info", [])
        if latest:
            return max(latest, key=lambda x: int(x.get("purchase_date_ms", 0)))
        # 一次性购买：取 receipt 中的 in_app 最后一条
        in_app = result.get("receipt", {}).get("in_app", [])
        if in_app:
            return max(in_app, key=lambda x: int(x.get("purchase_date_ms", 0)))
        return None

    @staticmethod
    def _safe_int(val) -> Optional[int]:
        if val is None:
            return None
        try:
            return int(val)
        except (TypeError, ValueError):
            return None
```

### 4.4 App Store Server Notification V2

#### 4.4.1 通知类型与处理策略

| Notification Type | 触发场景 | 处理动作 |
|-------------------|----------|----------|
| `DID_RENEW` | 订阅续期成功 | 延长会员到期时间 |
| `DID_FAIL_TO_RENEW` | 续期失败（计费问题） | 进入宽限期，保留权益 N 天 |
| `DID_CHANGE_RENEWAL_STATUS` | 用户关闭/开启自动续期 | 更新订阅状态，发送提醒 |
| `SUBSCRIBED` | 首次订阅或重新订阅 | 激活会员权益 |
| `EXPIRED` | 订阅到期 | 降级处理，触发挽留 |
| `GRACE_PERIOD_EXPIRED` | 宽限期结束仍未续费 | 立即降级 |
| `REVOKE` | 家庭共享撤销 | 检查该用户权益 |
| `REFUND` | 用户获退款 | 撤销权益，标记订单已退款 |
| `CONSUMPTION_REQUEST` | 用户申请退款，Apple 请求消费数据 | 48小时内响应消费数据 |
| `PRICE_INCREASE` | 价格上涨，用户需同意 | 更新价格信息 |
| `RENEWAL_EXTENDED` | Apple 延长续期日期 | 更新到期时间 |

#### 4.4.2 通知处理服务

```python
# payment/iap/apple_notification_handler.py

import hashlib
import hmac
import json
import time
from typing import Dict, Any

class AppleNotificationHandler:
    """处理 App Store Server Notification V2"""

    # 宽限期（秒），对应 Apple 配置的 Billing Grace Period
    GRACE_PERIOD_SECONDS = 16 * 24 * 3600  # 16天

    # 消费数据响应超时（Apple要求48小时内）
    CONSUMPTION_RESPONSE_DEADLINE_HOURS = 48

    def __init__(
        self,
        signing_key_provider,  # Apple 签名验证密钥提供者
        order_service,         # 订单服务
        membership_service,    # 会员服务
    ):
        self.signing_key_provider = signing_key_provider
        self.order_service = order_service
        self.membership_service = membership_service

    async def handle_notification(self, raw_body: bytes, signature_headers: dict) -> dict:
        """
        处理 Apple Server Notification V2
        Apple 使用 JWS (JSON Web Signature) 格式
        """
        # 1. 验证 JWS 签名
        payload = await self._verify_jws(raw_body, signature_headers)
        if not payload:
            return {"status": "INVALID_SIGNATURE"}

        notification_type = payload.get("notificationType")
        subtype = payload.get("subtype", "")
        transaction_info = payload.get("data", {})

        original_txn_id = transaction_info.get("originalTransactionId")
        product_id = transaction_info.get("productId")

        # 2. 查找关联订单
        order = await self.order_service.find_by_original_transaction_id(
            original_txn_id,
            store="apple",
        )

        if not order:
            # 可能是通知先于客户端确认到达，记录待处理通知
            await self._save_pending_notification(original_txn_id, payload)
            return {"status": "ORDER_NOT_FOUND_PENDING"}

        # 3. 根据类型分发处理
        handler_map = {
            "DID_RENEW": self._handle_renewal,
            "DID_FAIL_TO_RENEW": self._handle_renewal_failure,
            "DID_CHANGE_RENEWAL_STATUS": self._handle_renewal_status_change,
            "SUBSCRIBED": self._handle_subscribed,
            "EXPIRED": self._handle_expired,
            "GRACE_PERIOD_EXPIRED": self._handle_grace_expired,
            "REFUND": self._handle_refund,
            "REVOKE": self._handle_revoke,
            "CONSUMPTION_REQUEST": self._handle_consumption_request,
            "RENEWAL_EXTENDED": self._handle_renewal_extended,
        }

        handler = handler_map.get(notification_type, self._handle_unknown)
        return await handler(order, transaction_info, subtype)

    async def _handle_renewal(self, order, txn_info: dict, subtype: str) -> dict:
        """处理续期成功"""
        expires_date_ms = txn_info.get("expiresDate", 0)
        new_expires_at = datetime.fromtimestamp(expires_date_ms / 1000, tz=utc)

        # 更新订单到期时间
        await self.order_service.update_expiry(order["id"], new_expires_at)
        # 延长会员权益
        await self.membership_service.extend(
            user_id=order["user_id"],
            plan=order["product_code"],
            expires_at=new_expires_at,
            reason="APPLE_RENEWAL",
        )
        return {"status": "OK"}

    async def _handle_renewal_failure(self, order, txn_info: dict, subtype: str) -> dict:
        """处理续期失败 → 进入宽限期"""
        grace_until = datetime.now(tz=utc) + timedelta(seconds=self.GRACE_PERIOD_SECONDS)

        await self.membership_service.enter_grace_period(
            user_id=order["user_id"],
            grace_until=grace_until,
            reason="APPLE_RENEWAL_FAILED",
        )
        # 发送提醒：续期失败，请更新支付方式
        await self._notify_user(order["user_id"], "RENEWAL_FAILED", grace_until)
        return {"status": "OK"}

    async def _handle_refund(self, order, txn_info: dict, subtype: str) -> dict:
        """处理退款通知"""
        refund_time = datetime.now(tz=utc)

        # 撤销会员权益
        await self.membership_service.revoke(
            user_id=order["user_id"],
            reason="APPLE_REFUND",
            effective_at=refund_time,
        )
        # 标记订单已退款
        await self.order_service.mark_refunded(
            order_id=order["id"],
            refund_amount=order["paid_amount"],
            refund_reason="APPLE_STORE_REFUND",
            refund_time=refund_time,
        )
        return {"status": "OK"}
