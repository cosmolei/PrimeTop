# 服务端 - 学习目标达成度评估与计划偏离预警服务

> 本文档为 PrimeTop 服务端"学习目标达成度评估与计划偏离预警服务"的详细设计。
> 负责量化评估学生学习目标的完成进度，检测计划偏离行为，触发分级预警与干预策略。

---

## 1. 模块定位与职责边界

### 1.1 模块定位

学习目标达成度评估与计划偏离预警服务是 **学习规划模块** 与 **学情分析模块** 之间的桥梁。学生在系统中设定学习目标（如期中考试数学90分、本周完成3个章节复习等），本服务持续追踪目标完成进度，识别偏离情况，并通过多渠道预警推动纠偏。

### 1.2 核心职责

| 职责 | 说明 |
|------|------|
| 目标达成度计算 | 根据目标类型和进度数据，实时/定时计算达成率 |
| 计划偏离检测 | 比对实际学习行为与计划安排，识别偏离模式 |
| 分级预警触发 | 基于偏离严重程度触发不同级别的预警 |
| 干预策略推荐 | 为偏离学生生成个性化纠偏建议 |
| 预警通知推送 | 将预警信息推送到学生端、家长端、学习规划模块 |
| 达成度趋势分析 | 追踪目标完成度的变化趋势，预测最终达成概率 |

### 1.3 与相关模块的边界

| 模块 | 本服务与其关系 |
|------|---------------|
| 学习规划服务 | 本服务 **消费** 规划服务生成的学习计划和目标定义；预警结果 **回写** 触发计划调整 |
| 学习记录与进度追踪服务 | 本服务 **消费** 学习时长、答题记录、章节完成度等原始进度数据 |
| 学情分析服务 | 本服务 **消费** 知识点掌握度、能力维度评分；本服务的达成度评估结果 **供** 学情报告使用 |
| 消息与推送服务 | 本服务 **调用** 推送服务发送预警通知 |
| 用户学习画像 | 本服务 **消费** 学习风格、自律性标签等画像数据用于个性化预警策略 |
| 学习动力量化与倦怠预警引擎 | 本服务 **消费** 倦怠信号作为偏离检测的辅助因子 |

#### 3.2.1 考试成绩目标

```go
// ExamScoreCalculator 考试成绩目标达成度计算
type ExamScoreCalculator struct {
    repo AchievementRepo
}

func (c *ExamScoreCalculator) Calculate(ctx context.Context, goal *LearningGoal) (*AchievementResult, error) {
    // 获取目标信息
    targetScore := goal.TargetValue          // 如 90 分
    baselineScore := goal.BaselineValue      // 上次考试成绩，如 75 分
    gap := targetScore - baselineScore        // 需提升 15 分

    // 收集中间指标
    metrics := c.collectProgressMetrics(ctx, goal)

    // 加权计算达成度
    // 权重分配：知识点掌握提升(40%) + 专题练习正确率(30%) + 模拟考试成绩趋势(30%)
    achievementRate := 0.0

    // 1. 知识点掌握提升
    kpImprovement := metrics.KPMasteryImprovement // 0.0~1.0
    achievementRate += 0.4 * kpImprovement

    // 2. 专题练习正确率改善
    accuracyImprovement := (metrics.CurrentAccuracy - metrics.BaselineAccuracy) / gap * 10
    if accuracyImprovement > 1.0 { accuracyImprovement = 1.0 }
    if accuracyImprovement < 0 { accuracyImprovement = 0 }
    achievementRate += 0.3 * accuracyImprovement

    // 3. 模拟考试趋势（线性回归斜率外推）
    if len(metrics.MockExamScores) >= 2 {
        slope := linearRegressionSlope(metrics.MockExamScores)
        predictedFinal := metrics.LastMockScore + slope * metrics.RemainingWeeks
        scoreProgress := clamp((predictedFinal - baselineScore) / gap, 0, 1)
        achievementRate += 0.3 * scoreProgress
    } else {
        // 无模拟考试数据时，用前两项加权补偿
        achievementRate += 0.3 * kpImprovement
    }

    // 预测最终达成率
    predictedRate := c.predictFinalRate(ctx, goal, achievementRate, metrics)

    return &AchievementResult{
        AchievementRate: clamp(achievementRate, 0, 1),
        PredictedRate:   predictedRate,
        CurrentValue:    metrics.CurrentEstimatedScore,
        TargetValue:     targetScore,
        BaselineValue:   baselineScore,
    }, nil
}
```

```go
// ChapterFinishCalculator 章节完成目标达成度
func (c *ChapterFinishCalculator) Calculate(ctx context.Context, goal *LearningGoal) (*AchievementResult, error) {
    // 获取需要完成的章节列表
    chapters, _ := c.repo.GetGoalChapters(ctx, goal.ID)
    totalChapters := len(chapters)
    if totalChapters == 0 {
        return &AchievementResult{AchievementRate: 0}, nil
    }

    // 统计完成状态
    completedCount := 0
    partialCount := 0
    for _, ch := range chapters {
        progress := c.chapterProgressSvc.GetProgress(ctx, goal.UserID, ch.ChapterID)
        if progress >= 0.9 {       // 完成度 ≥90% 视为完成
            completedCount++
        } else if progress >= 0.3 { // 30%-90% 视为部分完成
            partialCount += int(progress * 10) / 10
        }
    }

    // 加权达成率：已完成章节全量计入 + 部分完成章节折半计入
    achievementRate := (float64(completedCount) + 0.5*float64(partialCount)) / float64(totalChapters)

    // 计算所需节奏 vs 当前节奏
    elapsedDays := daysBetween(goal.StartedAt, time.Now())
    remainingDays := daysBetween(time.Now(), goal.DeadlineAt)
    totalDays := daysBetween(goal.StartedAt, goal.DeadlineAt)
    if totalDays == 0 { totalDays = 1 }

    currentPace := float64(completedCount) / float64(max(elapsedDays, 1))
    requiredPace := float64(totalChapters-completedCount) / float64(max(remainingDays, 1))

    return &AchievementResult{
        AchievementRate: clamp(achievementRate, 0, 1),
        CurrentValue:    float64(completedCount),
        TargetValue:     float64(totalChapters),
        CurrentPace:     currentPace,
        RequiredPace:    requiredPace,
    }, nil
}
```

```go
// CumulativeTargetCalculator 累计量目标通用计算（适用于练习量、学习时长、错题复习量）
func (c *CumulativeTargetCalculator) Calculate(ctx context.Context, goal *LearningGoal) (*AchievementResult, error) {
    // 从学习记录服务获取累计值
    current, _ := c.progressSvc.GetCumulativeValue(ctx, goal.UserID, goal.GoalType, goal.StartedAt, time.Now())

    achievementRate := current / goal.TargetValue
    if achievementRate > 1.0 { achievementRate = 1.0 }

    // 节奏计算
    elapsedDays := daysBetween(goal.StartedAt, time.Now())
    remainingDays := daysBetween(time.Now(), goal.DeadlineAt)
    if elapsedDays < 1 { elapsedDays = 1 }
    if remainingDays < 1 { remainingDays = 1 }

    currentPace := current / float64(elapsedDays)           // 日均完成量
    remaining := goal.TargetValue - current
    requiredPace := remaining / float64(remainingDays)      // 剩余日均需完成量

    // 预测最终完成量（线性外推）
    predictedFinal := current + currentPace * float64(remainingDays)
    predictedRate := predictedFinal / goal.TargetValue
    if predictedRate > 1.0 { predictedRate = 1.0 }

    return &AchievementResult{
        AchievementRate: clamp(achievementRate, 0, 1),
        PredictedRate:   clamp(predictedRate, 0, 1),
        CurrentValue:    current,
        TargetValue:     goal.TargetValue,
        CurrentPace:     currentPace,
        RequiredPace:    requiredPace,
    }, nil
}
```

```go
// ErrorRateCalculator 错误率改善目标达成度
func (c *ErrorRateCalculator) Calculate(ctx context.Context, goal *LearningGoal) (*AchievementResult, error) {
    // 获取基线错误率和近期错误率
    baselineRate := goal.BaselineValue       // 如 0.35 (35%)
    targetRate := goal.TargetValue           // 如 0.15 (15%)
    recentRate := c.answerRecordSvc.GetRecentErrorRate(ctx, goal.UserID, goal.SubjectID, 14) // 近14天

    totalImprovement := baselineRate - targetRate   // 需改善 0.20
    actualImprovement := baselineRate - recentRate  // 实际已改善

    if totalImprovement <= 0 {
        return &AchievementResult{AchievementRate: 1.0}, nil
    }

    achievementRate := actualImprovement / totalImprovement
    if achievementRate < 0 { achievementRate = 0 }
    if achievementRate > 1 { achievementRate = 1 }

    return &AchievementResult{
        AchievementRate: achievementRate,
        CurrentValue:    recentRate,
        TargetValue:     targetRate,
```

### 3.3 预测算法

```go
// GoalPredictor 目标达成预测器
type GoalPredictor struct {
    snapshotRepo SnapshotRepo
}

// PredictFinalRate 预测最终达成率
// 综合考虑：历史趋势 + 学习节奏变化 + 距离deadline天数
func (p *GoalPredictor) PredictFinalRate(ctx context.Context, goalID int64, currentRate float64, remainingDays int) (predictedRate float64, confidence float64) {
    // 1. 获取历史快照
    snapshots, _ := p.snapshotRepo.GetRecentSnapshots(ctx, goalID, 30)
    if len(snapments) < 3 {
        // 数据不足，用线性外推
        return currentRate, 0.3 // 低置信度
    }

    // 2. 提取达成率序列
    rates := make([]float64, len(snapshots))
    for i, s := range snapshots {
        rates[i] = s.AchievementRate
    }

    // 3. 线性回归预测趋势
    slope, intercept := linearRegression(rates)
    totalDays := len(snapshots) + remainingDays
    predictedRate = slope*float64(totalDays) + intercept

    // 4. 置信度计算：基于样本量和趋势稳定性
    variance := calculateVariance(rates)
    confidence = 1.0 / (1.0 + variance*10) // 方差越大置信度越低
    if len(snapshots) < 7 {
        confidence *= 0.7 // 少于7天数据降低置信度
    }

    // 5. 边界约束
    if predictedRate > 1.2 { predictedRate = 1.2 }
    if predictedRate < 0 { predictedRate = 0 }

    return predictedRate, clamp(confidence, 0.1, 0.95)
}
```

```go
// DeviationDetector 偏离检测器
type DeviationDetector struct {
    achievementSvc AchievementService
    progressSvc    ProgressService
    alertRepo      AlertRepo
    deviationRepo  DeviationRepo
}

type DeviationCheckResult struct {
    Detected     bool
    Type         DeviationType
    Severity     DeviationSeverity
    Score        float64
    ExpectedDesc string
    ActualDesc   string
}

// CheckDeviations 综合偏离检测入口
func (d *DeviationDetector) CheckDeviations(ctx context.Context, userID int64, planID int64) []DeviationCheckResult {
    var results []DeviationCheckResult

    // 并行执行各类偏离检测
    checks := []func(context.Context, int64, int64) *DeviationCheckResult{
        d.checkPaceLag,         // 进度节奏滞后
        d.checkTaskCompletion,  // 任务完成率
        d.checkDurationDrop,    // 学习时长骤降
        d.checkQualityDrop,     // 答题质量下降
        d.checkInactivePattern, // 学习中断
        d.checkSkipPattern,     // 跳过困难任务
    }

    for _, check := range checks {
        if result := check(ctx, userID, planID); result != nil && result.Detected {
            results = append(results, *result)
        }
    }

    return results
}

// checkPaceLag 进度节奏滞后检测
func (d *DeviationDetector) checkPaceLag(ctx context.Context, userID int64, planID int64) *DeviationCheckResult {
    // 获取所有活跃目标的达成度
    achievements := d.achievementSvc.GetActiveAchievements(ctx, userID, planID)

    for _, ach := range achievements {
        if ach.RequiredPace <= 0 { continue }

        gapRatio := (ach.RequiredPace - ach.CurrentPace) / ach.RequiredPace

        // 偏离阈值
        // gapRatio > 0.2 → 轻微; > 0.4 → 中度; > 0.6 → 严重; > 0.8 → 危急
        if gapRatio > 0.2 {
            severity := classifySeverity(gapRatio)
            return &DeviationCheckResult{
                Detected:     true,
                Type:         DeviationPaceLag,
                Severity:     severity,
                Score:        gapRatio,
                ExpectedDesc: fmt.Sprintf("日均需完成 %.1f 单位", ach.RequiredPace),
                ActualDesc:   fmt.Sprintf("实际日均完成 %.1f 单位", ach.CurrentPace),
            }
        }
    }

    return nil
}

// checkInactivePattern 学习中断检测
func (d *DeviationDetector) checkInactivePattern(ctx context.Context, userID int64, planID int64) *DeviationCheckResult {
    lastActiveTime := d.progressSvc.GetLastActiveTime(ctx, userID)
    inactiveDays := daysBetween(lastActiveTime, time.Now())

    if inactiveDays < 2 { return nil }

    severity := DeviationSeverityLow
    switch {
    case inactiveDays >= 7:
        severity = DeviationSeverityCritical
    case inactiveDays >= 5:
        severity = DeviationSeverityHigh
    case inactiveDays >= 3:
        severity = DeviationSeverityMedium
    }

    return &DeviationCheckResult{
        Detected:     true,
        Type:         DeviationInactive,
        Severity:     severity,
        Score:        float64(inactiveDays) / 7.0,
        ExpectedDesc: "每日持续学习",
        ActualDesc:   fmt.Sprintf("已 %d 天未学习", inactiveDays),
    }
}

// checkSkipPattern 跳过困难任务模式检测
func (d *DeviationDetector) checkSkipPattern(ctx context.Context, userID int64, planID int64) *DeviationCheckResult {
    // 获取最近7天的任务完成记录
    recentTasks := d.progressSvc.GetRecentTasks(ctx, userID, planID, 7)
    if len(recentTasks) < 3 { return nil }

    skippedHard := 0
    totalHard := 0
    for _, task := range recentTasks {
        if task.DifficultyLevel >= 3 { // 困难任务
            totalHard++
            if task.Status == TaskStatusSkipped || (task.Status == TaskStatusIncomplete && task.IsOverdue) {
                skippedHard++
            }
        }
    }

    if totalHard < 2 { return nil }
    skipRate := float64(skippedHard) / float64(totalHard)

    if skipRate >= 0.5 { // 跳过50%以上的困难任务
        return &DeviationCheckResult{
            Detected:     true,
            Type:         DeviationSkipPattern,
            Severity:     classifySeverity(skipRate),
            Score:        skipRate,
            ExpectedDesc: "按计划完成困难任务",
            ActualDesc:   fmt.Sprintf("跳过了 %d/%d 个困难任务", skippedHard, totalHard),
        }
    }

    return nil
}

// checkDurationDrop 学习时长骤降检测
func (d *DeviationDetector) checkDurationDrop(ctx context.Context, userID int64, planID int64) *DeviationCheckResult {
    // 获取最近14天的每日学习时长
    dailyDurations := d.progressSvc.GetDailyDurations(ctx, userID, 14)
    if len(dailyDurations) < 7 { return nil }

    // 前一周平均 vs 后一周平均
    week1Avg := average(dailyDurations[:7])
    week2Avg := average(dailyDurations[7:])

    if week1Avg == 0 { return nil }
    dropRate := (week1Avg - week2Avg) / week1Avg

    if dropRate >= 0.3 { // 学习时长下降30%以上
        return &DeviationCheckResult{
            Detected:     true,
            Type:         DeviationDurationDrop,
            Severity:     classifySeverity(dropRate),
            Score:        dropRate,
            ExpectedDesc: fmt.Sprintf("维持日均 %.0f 分钟", week1Avg),
            ActualDesc:   fmt.Sprintf("近一周日均 %.0f 分钟", week2Avg),
        }
    }

    return nil
}

// checkQualityDrop 答题质量下降检测
func (d *DeviationDetector) checkQualityDrop(ctx context.Context, userID int64, planID int64) *DeviationCheckResult {
    accuracyTrend := d.progressSvc.GetAccuracyTrend(ctx, userID, 14) // 近14天正确率趋势
    if len(accuracyTrend) < 7 { return nil }

    recentAcc := average(accuracyTrend[:3])   // 最近3天
    baselineAcc := average(accuracyTrend[4:]) // 之前10天

    if baselineAcc == 0 { return nil }
    dropRate := (baselineAcc - recentAcc) / baselineAcc

    if dropRate >= 0.2 { // 正确率下降20%以上
        return &DeviationCheckResult{
            Detected:     true,
            Type:         DeviationQualityDrop,
            Severity:     classifySeverity(dropRate),
            Score:        dropRate,
            ExpectedDesc: fmt.Sprintf("维持 %.0f%% 正确率", baselineAcc*100),
            ActualDesc:   fmt.Sprintf("近3天正确率 %.0f%%", recentAcc*100),
        }
    }

    return nil
}

func classifySeverity(score float64) DeviationSeverity {
    switch {
    case score >= 0.8:
        return DeviationSeverityCritical
    case score >= 0.6:
        return DeviationSeverityHigh
    case score >= 0.4:
        return DeviationSeverityMedium
    default:
        return DeviationSeverityLow
    }
}
```

```go
// AlertPolicy 预警策略配置
type AlertPolicy struct {
    // 冷却期：同一用户同一偏离类型的最小通知间隔
    CooldownPeriods map[AlertLevel]time.Duration
    // 每日最大预警数（防打扰）
    MaxAlertsPerDay int
    // 允许通知的时间窗口
    NotifyWindowStart int // 小时，如 8
    NotifyWindowEnd   int // 小时，如 22
    // 家长通知阈值（低于此级别不通知家长）
    ParentNotifyThreshold AlertLevel
}

// DefaultAlertPolicy 默认预警策略
func DefaultAlertPolicy() *AlertPolicy {
    return &AlertPolicy{
        CooldownPeriods: map[AlertLevel]time.Duration{
            AlertLevelInfo:     7 * 24 * time.Hour,  // Info: 7天冷却
            AlertLevelWarning:  3 * 24 * time.Hour,  // Warning: 3天冷却
            AlertLevelUrgent:   2 * 24 * time.Hour,  // Urgent: 2天冷却
            AlertLevelCritical: 24 * time.Hour,       // Critical: 1天冷却
        },
        MaxAlertsPerDay:        3,
        NotifyWindowStart:      8,
        NotifyWindowEnd:        22,
        ParentNotifyThreshold:  AlertLevelWarning,  // Warning及以上通知家长
    }
}

// ShouldSendAlert 判断是否应该发送预警（冷却+去重+频率控制）
func (s *AlertService) ShouldSendAlert(ctx context.Context, userID int64, alertLevel AlertLevel, deviationType DeviationType) (bool, *AlertPolicy) {
    policy := s.getPolicy(ctx, userID)

    // 1. 检查时间窗口
    hour := time.Now().Hour()
    if hour < policy.NotifyWindowStart || hour >= policy.NotifyWindowEnd {
        return false, policy
    }

    // 2. 检查每日最大预警数
    todayCount := s.alertRepo.CountTodayAlerts(ctx, userID)
    if todayCount >= policy.MaxAlertsPerDay {
        return false, policy
    }

    // 3. 检查冷却期
    cooldown := policy.CooldownPeriods[alertLevel]
    lastAlert := s.alertRepo.GetLastAlertByType(ctx, userID, string(deviationType))
    if lastAlert != nil && time.Since(lastAlert.CreatedAt) < cooldown {
        return false, policy
    }

    return true, policy
}
```

### 7.2 每日达成度评估流程

```go
// DailyAchievementEvalJob 每日达成度评估定时任务
func (j *DailyAchievementEvalJob) Run(ctx context.Context) error {
    log := logger.FromContext(ctx)
    log.Info("开始每日达成度评估")

    // 1. 分页查询所有活跃目标（避免一次性加载过多）
    batchSize := 500
    offset := 0

    for {
        goals, err := j.goalRepo.ListActiveGoals(ctx, offset, batchSize)
        if err != nil {
            return fmt.Errorf("查询活跃目标失败: %w", err)
        }
        if len(goals) == 0 { break }

        // 2. 按用户分组（避免同一用户的目标并发评估导致数据竞争）
        userGoals := groupByUser(goals)

        for userID, userGoalList := range userGoals {
            // 3. 为每个目标计算达成度
            for _, goal := range userGoalList {
                calculator := j.calculatorFactory.GetCalculator(goal.GoalType)
                result, err := calculator.Calculate(ctx, goal)
                if err != nil {
                    log.Warn("目标达成度计算失败", "goal_id", goal.ID, "error", err)
                    continue
                }

                // 4. 更新或创建 GoalAchievement 记录
                j.achievementRepo.UpsertAchievement(ctx, goal, result)
            }
        }

        offset += batchSize
    }

    log.Info("每日达成度评估完成")
    return nil
}
```

### 7.3 每日偏离检测流程

```go
// DailyDeviationCheckJob 每日偏离检测定时任务
func (j *DailyDeviationCheckJob) Run(ctx context.Context) error {
    log := logger.FromContext(ctx)

    // 1. 获取所有有活跃计划的用户
    users := j.planRepo.GetUsersWithActivePlans(ctx)

    for _, userID := range users {
        plans := j.planRepo.GetActivePlans(ctx, userID)

        for _, plan := range plans {
            // 2. 执行偏离检测
            results := j.deviationDetector.CheckDeviations(ctx, userID, plan.ID)

            for _, result := range results {
                // 3. 查找已有偏离记录
                existing := j.deviationRepo.FindActiveByType(ctx, userID, plan.ID, result.Type)

                if existing != nil {
                    // 更新已有偏离
                    existing.LastDetectedAt = time.Now()
                    existing.DurationDays++
                    existing.Severity = maxSeverity(existing.Severity, result.Severity)
                    existing.DeviationScore = result.Score
                    j.deviationRepo.Update(ctx, existing)
                } else {
                    // 创建新偏离
                    deviation := &PlanDeviation{
                        UserID:          userID,
                        PlanID:          plan.ID,
                        DeviationType:   result.Type,
                        Severity:        result.Severity,
                        DeviationScore:  result.Score,
                        ExpectedValue:   result.ExpectedDesc,
                        ActualValue:     result.ActualDesc,
                        FirstDetectedAt: time.Now(),
                        LastDetectedAt:  time.Now(),
                        DurationDays:    1,
                        InterventionStatus: InterventionPending,
                    }
                    j.deviationRepo.Create(ctx, deviation)
                }

                // 4. 判断是否需要发送预警
                if j.shouldEscalate(ctx, userID, result) {
                    j.alertService.CreateAndSendAlert(ctx, userID, plan.ID, result)
                }
            }
        }
    }

    return nil
}
```

### 12.2 容错策略

```go
// 评估失败容错：跳过失败目标，不影响其他目标的评估
// 定时任务中使用 continue 而非 return

// 推送失败容错：记录失败状态，由重试任务兜底
func (s *AlertService) sendWithRetry(ctx context.Context, alert *DeviationAlert, maxRetries int) {
    for i := 0; i < maxRetries; i++ {
        err := s.pushService.Send(ctx, alert)
        if err == nil {
            alert.NotifyPushedAt = timePtr(time.Now())
            s.alertRepo.Update(ctx, alert)
            return
        }
        log.Warn("预警推送失败，准备重试", "attempt", i+1, "error", err)
        time.Sleep(time.Duration(i+1) * 5 * time.Second)
    }
    // 最终失败：标记为推送失败，由离线补偿任务处理
    log.Error("预警推送最终失败", "alert_id", alert.ID)
}
```

## 14. 关键设计决策

| 决策 | 选择 | 理由 |
|------|------|------|
| 达成度计算采用策略工厂模式 | ✅ | 不同目标类型的计算逻辑差异大，策略模式便于扩展 |
| 偏离检测在定时任务中执行而非事件驱动 | 定时任务为主 | 偏离检测需要综合多日数据趋势，单次事件无法判断偏离 |
| 预警通知有冷却期和频率限制 | ✅ | 避免过度打扰，尤其是学生用户 |
| AI纠偏建议在预警时实时生成 | ✅ | 个性化建议需要结合最新数据，预生成会过时 |
| 达成度快照按天保存 | ✅ | 支持趋势分析和预测，同时存储成本可控 |
| 使用线性回归而非复杂模型做预测 | ✅ | 教育场景数据周期短（通常1-4周），简单模型足够且可解释性强 |
