# 服务端-学生综合学习健康度评估与多维度发展均衡性智能诊断引擎-详细设计

## 1. 概述

### 1.1 功能定位

学生综合学习健康度评估引擎是 PrimeTop 平台的**顶层健康聚合服务**，负责将分散在各垂直分析引擎中的维度数据（学业掌握度、心理状态、认知负荷、学习习惯、体能/视力健康、社交互动等）汇聚为统一的 **学习健康度指数（Learning Health Index, LHI）**，并输出多维度均衡性诊断报告和优先级改进建议。

不同于单一维度的分析引擎（如知识掌握度评估、心理状态建模等），本引擎的职责是：

1. **聚合**——从各维度引擎拉取标准化指标快照。
2. **加权计算**——根据学生学段、目标、季节等因素动态调整各维度权重。
3. **均衡性诊断**——检测维度间的严重失衡（如学业高但心理极低）。
4. **综合报告**——面向学生、家长、教师输出不同视角的健康报告。
5. **预警**——当综合健康度或关键维度跌破阈值时触发实时预警。

### 1.2 设计目标

| 目标 | 指标 |
| --- | --- |
| 指标聚合延迟 | ≤5 秒（从各维度引擎拉取 + 计算） |
| 健康度评分准确率 | 与教研专家人工评估一致率 ≥80% |
| 日评估吞吐量 | ≥100 万学生/天（批量 + 增量） |
| 报告生成时间 | 单学生全量报告 ≤3 秒 |
| 预警触达延迟 | 关键维度异常 → 推送 ≤30 秒 |

### 1.3 适用范围

- 学生端：首页健康度仪表盘、周报/月报中的健康板块
- 家长端：学情报告中的综合健康评价
- 教师端：班级健康管理看板
- 管理后台：平台级学生健康度统计分析

### 2.2 维度权重模板

权重并非固定，而是根据学生特征动态调整：

```python
# 维度权重计算逻辑伪代码
class DimensionWeightCalculator:
    """维度权重计算器"""

    # 基础权重模板（按学段）
    BASE_WEIGHTS = {
        "kindergarten": {  # 幼儿
            "D1": 0.10, "D2": 0.20, "D3": 0.10, "D4": 0.15,
            "D5": 0.10, "D6": 0.25, "D7": 0.05, "D8": 0.05,
        },
        "primary": {  # 小学
            "D1": 0.20, "D2": 0.15, "D3": 0.10, "D4": 0.15,
            "D5": 0.05, "D6": 0.15, "D7": 0.10, "D8": 0.10,
        },
        "junior": {  # 初中
            "D1": 0.25, "D2": 0.15, "D3": 0.10, "D4": 0.15,
            "D5": 0.05, "D6": 0.10, "D7": 0.10, "D8": 0.10,
        },
        "senior": {  # 高中
            "D1": 0.30, "D2": 0.15, "D3": 0.10, "D4": 0.10,
            "D5": 0.05, "D6": 0.10, "D7": 0.10, "D8": 0.10,
        },
    }

    # 场景叠加调整（备考期间、假期等）
    SCENE_ADJUSTMENTS = {
        "exam_prep": {"D1": +0.05, "D2": +0.05, "D6": -0.05, "D5": -0.05},
        "holiday": {"D1": -0.05, "D4": +0.05, "D6": +0.05, "D5": +0.05},
        "new_semester": {"D2": +0.05, "D4": +0.05, "D1": -0.05, "D8": +0.05},
    }

    def calculate_weights(
        self,
        student_stage: str,
        scene: str | None = None,
        custom_overrides: dict[str, float] | None = None,
    ) -> dict[str, float]:
        """
        计算最终维度权重。

        Args:
            student_stage: 学段 (kindergarten/primary/junior/senior)
            scene: 当前场景 (exam_prep/holiday/new_semester/None)
            custom_overrides: 自定义权重覆盖（家长/教师设定的偏好）

        Returns:
            归一化后的各维度权重字典
        """
        weights = dict(self.BASE_WEIGHTS[student_stage])

        # 叠加场景调整
        if scene and scene in self.SCENE_ADJUSTMENTS:
            for dim, delta in self.SCENE_ADJUSTMENTS[scene].items():
                weights[dim] = max(0.02, weights[dim] + delta)

        # 叠加自定义覆盖
        if custom_overrides:
            for dim, val in custom_overrides.items():
                weights[dim] = val

        # 归一化
        total = sum(weights.values())
        return {k: v / total for k, v in weights.items()}
```

## 4. 核心算法设计

### 4.1 综合健康度计算引擎

```python
from typing import Optional
from dataclasses import dataclass
from datetime import date
import math

@dataclass
class DimensionSnapshot:
    """单维度快照数据"""
    dim_code: str           # D1~D8
    dim_name: str
    score: float            # 0~100
    confidence: float       # 0~1，数据可信度
    sub_metrics: dict       # 子指标明细
    source_engine: str      # 来源引擎标识
    is_missing: bool = False

@dataclass
class HealthEvaluationResult:
    """健康度评估结果"""
    overall_score: float
    balance_score: float
    dimension_scores: dict[str, float]
    dimension_weights: dict[str, float]
    risk_level: str          # normal/warning/danger
    trend_direction: str     # up/down/stable
    missing_dims: list[str]
    diagnosis: str
    suggested_actions: list[dict]


class LearningHealthEvaluator:
    """学习健康度评估引擎核心"""

    # 维度数据源映射
    DIMENSION_SOURCES = {
        "D1": "knowledge_tracing_engine",
        "D2": "psychological_state_engine",
        "D3": "cognitive_load_engine",
        "D4": "study_habit_engine",
        "D5": "social_learning_engine",
        "D6": "health_monitor_engine",
        "D7": "engagement_engine",
        "D8": "ai_tutoring_effectiveness_engine",
    }

    # 缺失维度默认分（保守取中间偏低值）
    MISSING_DIM_DEFAULT = 55.0

    # 缺失维度置信度
    MISSING_DIM_CONFIDENCE = 0.3

    def evaluate(
        self,
        student_id: int,
        eval_date: date,
        stage: str,
        scene: Optional[str] = None,
        custom_weights: Optional[dict] = None,
        force_refresh: bool = False,
    ) -> HealthEvaluationResult:
        """
        执行完整的健康度评估流程。
        """
        # Step 1: 拉取各维度快照
        snapshots = self._fetch_dimension_snapshots(
            student_id, eval_date, force_refresh
        )

        # Step 2: 处理缺失维度
        snapshots = self._handle_missing_dimensions(snapshots)

        # Step 3: 计算维度权重
        weights = self._calculate_weights(stage, scene, custom_weights, snapshots)

        # Step 4: 加权计算综合分
        overall_score = self._calculate_overall(snapshots, weights)

        # Step 5: 均衡性诊断
        balance_score = self._calculate_balance(snapshots)

        # Step 6: 风险等级判定
        risk_level = self._determine_risk_level(overall_score, snapshots, balance_score)

        # Step 7: 趋势方向
        trend = self._calculate_trend(student_id, overall_score)

        # Step 8: 生成诊断和建议
        diagnosis = self._generate_diagnosis(snapshots, overall_score, balance_score)
        actions = self._generate_actions(snapshots, weights, risk_level)

        return HealthEvaluationResult(
            overall_score=round(overall_score, 2),
            balance_score=round(balance_score, 2),
            dimension_scores={s.dim_code: s.score for s in snapshots.values()},
            dimension_weights=weights,
            risk_level=risk_level,
            trend_direction=trend,
            missing_dims=[k for k, v in snapshots.items() if v.is_missing],
            diagnosis=diagnosis,
            suggested_actions=actions,
        )

    def _fetch_dimension_snapshots(
        self, student_id: int, eval_date: date, force_refresh: bool
    ) -> dict[str, DimensionSnapshot]:
        """
        从各维度引擎并行拉取快照数据。
        实际实现中使用并行 RPC/HTTP 调用 + 超时降级。
        """
        snapshots = {}
        for dim_code, engine_id in self.DIMENSION_SOURCES.items():
            try:
                snapshot = self._call_dimension_engine(
                    engine_id, student_id, eval_date, timeout_ms=2000
                )
                if snapshot and snapshot.score is not None:
                    snapshots[dim_code] = snapshot
                else:
                    snapshots[dim_code] = DimensionSnapshot(
                        dim_code=dim_code,
                        dim_name=self._get_dim_name(dim_code),
                        score=0, confidence=0,
                        sub_metrics={},
                        source_engine=engine_id,
                        is_missing=True,
                    )
            except Exception as e:
                # 引擎调用失败，标记为缺失
                logger.warning(
                    f"Dimension {dim_code} fetch failed for student {student_id}: {e}"
                )
                snapshots[dim_code] = DimensionSnapshot(
                    dim_code=dim_code,
                    dim_name=self._get_dim_name(dim_code),
                    score=0, confidence=0,
                    sub_metrics={},
                    source_engine=engine_id,
                    is_missing=True,
                )
        return snapshots

    def _handle_missing_dimensions(
        self, snapshots: dict[str, DimensionSnapshot]
    ) -> dict[str, DimensionSnapshot]:
        """
        对缺失维度进行补偿：使用历史均值或保守默认值。
        """
        for dim_code, snap in snapshots.items():
            if snap.is_missing:
                # 尝试从最近7天的历史数据取均值
                history_avg = self._get_history_avg(dim_code, snap_dim=dim_code)
                if history_avg is not None:
                    # 历史均值衰减：越久的数据权重越低
                    snap.score = history_avg * 0.9 + self.MISSING_DIM_DEFAULT * 0.1
                else:
                    snap.score = self.MISSING_DIM_DEFAULT
                snap.confidence = self.MISSING_DIM_CONFIDENCE
        return snapshots

    def _calculate_overall(
        self,
        snapshots: dict[str, DimensionSnapshot],
        weights: dict[str, float],
    ) -> float:
        """
        加权计算综合健康度，考虑置信度调整。
        """
        numerator = 0.0
        denominator = 0.0

        for dim_code, snap in snapshots.items():
            w = weights.get(dim_code, 0)
            # 置信度低于0.5的维度降低权重
            adjusted_w = w * (0.5 + 0.5 * snap.confidence)
            numerator += snap.score * adjusted_w
            denominator += adjusted_w

        if denominator == 0:
            return 0.0

        raw_score = numerator / denominator

        # 均衡性惩罚：如果维度间差异极大，适度降低总分
        scores = [s.score for s in snapshots.values()]
        std_dev = np.std(scores)
        max_std = 30.0  # 经验值：标准差超过30说明严重失衡
        balance_penalty = max(0, (std_dev - 15.0) / max_std) * 5.0  # 最多扣5分

        return max(0, min(100, raw_score - balance_penalty))

    def _calculate_balance(self, snapshots: dict[str, DimensionSnapshot]) -> float:
        """
        计算维度均衡度。
        使用标准差的逆归一化：标准差越小越均衡。
        """
        scores = [s.score for s in snapshots.values()]
        if len(scores) < 2:
            return 100.0

        std_dev = float(np.std(scores))
        # 标准差 0 → 均衡度 100，标准差 40+ → 均衡度 0
        balance = max(0, 100 - std_dev * 2.5)
        return balance

    def _determine_risk_level(
        self,
        overall_score: float,
        snapshots: dict[str, DimensionSnapshot],
        balance_score: float,
    ) -> str:
        """
        风险等级判定逻辑。
        """
        # 危险：综合分 <40 或 任一维度 <30 或 均衡度 <30
        if overall_score < 40:
            return "danger"
        for snap in snapshots.values():
            if not snap.is_missing and snap.score < 30:
                return "danger"
        if balance_score < 30:
            return "danger"

        # 警告：综合分 <55 或 任一维度 <45 或 均衡度 <50
        if overall_score < 55:
            return "warning"
        for snap in snapshots.values():
            if not snap.is_missing and snap.score < 45:
                return "warning"
        if balance_score < 50:
            return "warning"

        return "normal"

    def _calculate_trend(self, student_id: int, current_score: float) -> str:
        """
        与昨天对比计算趋势方向。
        """
        yesterday_score = self._get_yesterday_score(student_id)
        if yesterday_score is None:
            return "stable"

        delta = current_score - yesterday_score
        if delta > 2:
            return "up"
        elif delta < -2:
            return "down"
        else:
            return "stable"
```

### 4.2 均衡性诊断算法

```python
class BalanceDiagnosticEngine:
    """维度均衡性诊断引擎"""

    # 失衡阈值：两个维度得分差超过此值判定为失衡
    IMBALANCE_THRESHOLD = 15.0

    # 严重失衡阈值
    SEVERE_IMBALANCE_THRESHOLD = 25.0

    def diagnose(
        self, snapshots: dict[str, DimensionSnapshot]
    ) -> dict:
        """
        执行均衡性诊断。
        """
        valid_snaps = {
            k: v for k, v in snapshots.items() if not v.is_missing
        }

        if len(valid_snaps) < 2:
            return {
                "balance_score": 100.0,
                "strongest_dims": [],
                "weakest_dims": [],
                "imbalance_pairs": [],
                "improvement_priority": [],
            }

        # 排序
        sorted_dims = sorted(
            valid_snaps.values(), key=lambda s: s.score, reverse=True
        )

        # 最强/最弱维度
        strongest = [
            {"dim": s.dim_code, "name": s.dim_name, "score": s.score}
            for s in sorted_dims[:3]
        ]
        weakest = [
            {"dim": s.dim_code, "name": s.dim_name, "score": s.score}
            for s in sorted_dims[-3:]
        ]

        # 失衡维度对检测
        imbalance_pairs = []
        dim_list = list(valid_snaps.values())
        for i in range(len(dim_list)):
            for j in range(i + 1, len(dim_list)):
                gap = abs(dim_list[i].score - dim_list[j].score)
                if gap >= self.IMBALANCE_THRESHOLD:
                    level = "severe" if gap >= self.SEVERE_IMBALANCE_THRESHOLD else "moderate"
                    imbalance_pairs.append({
                        "dimA": dim_list[i].dim_code,
                        "scoreA": round(dim_list[i].score, 1),
                        "dimB": dim_list[j].dim_code,
                        "scoreB": round(dim_list[j].score, 1),
                        "gap": round(gap, 1),
                        "level": level,
                    })

        # 按差距排序
        imbalance_pairs.sort(key=lambda x: x["gap"], reverse=True)

        # 优先改进维度排序
        # 规则：得分最低 + 权重最高 = 最优先改进
        priority = self._calculate_priority(valid_snaps)

        return {
            "balance_score": self._calc_balance_score(valid_snaps),
            "strongest_dims": strongest,
            "weakest_dims": weakest,
            "imbalance_pairs": imbalance_pairs[:5],  # 取前5对
            "improvement_priority": priority,
        }

    def _calculate_priority(self, snapshots: dict) -> list[dict]:
        """
        优先级 = normalize(1 - score) * normalize(weight)
        得分低且权重高的维度最优先改进。
        """
        priorities = []
        for dim_code, snap in snapshots.items():
            urgency = (100 - snap.score) / 100  # 得分越低越紧急
            # 注意：这里 weight 需要外部传入，简化处理用默认权重
            impact = 0.15  # 平均权重
            priority_score = urgency * impact
            priorities.append({
                "dim": dim_code,
                "name": snap.dim_name,
                "score": round(snap.score, 1),
                "priority_score": round(priority_score, 4),
            })
        priorities.sort(key=lambda x: x["priority_score"], reverse=True)
        return priorities
```

```python
class LHIDailyEvalJob:
    """每日全量健康度评估任务"""

    BATCH_SIZE = 500  # 每批500个学生
    MAX_WORKERS = 10  # 最大并行 worker

    async def run(self, eval_date: date):
        """
        执行全量评估。
        """
        # 1. 获取需要评估的学生列表
        student_ids = await self._get_active_students(eval_date)
        total = len(student_ids)
        logger.info(f"LHI daily eval started: {total} students for {eval_date}")

        # 2. 分批并行处理
        success_count = 0
        fail_count = 0

        for batch in self._chunk(student_ids, self.BATCH_SIZE):
            tasks = [
                self._eval_single(sid, eval_date) for sid in batch
            ]
            results = await asyncio.gather(*tasks, return_exceptions=True)

            for sid, result in zip(batch, results):
                if isinstance(result, Exception):
                    logger.error(f"Eval failed for student {sid}: {result}")
                    fail_count += 1
                else:
                    success_count += 1

            # 批次间短暂等待，避免对维度引擎造成压力
            await asyncio.sleep(0.5)

        logger.info(
            f"LHI daily eval completed: {success_count} success, {fail_count} fail"
        )

        # 3. 触发预警扫描
        await self._trigger_alert_scan(eval_date)

    async def _eval_single(self, student_id: int, eval_date: date):
        """评估单个学生"""
        # 获取学生元信息
        meta = await self._get_student_meta(student_id)

        # 调用评估引擎
        result = self.evaluator.evaluate(
            student_id=student_id,
            eval_date=eval_date,
            stage=meta.stage,
            scene=meta.current_scene,
        )

        # 持久化
        await self._save_snapshot(student_id, eval_date, result)

        # 检查预警
        await self._check_alerts(student_id, result)
```

### 7.1 预警规则定义

```python
class AlertRuleEngine:
    """健康度预警规则引擎"""

    RULES = [
        # 规则1：综合健康度骤降
        {
            "id": "R001",
            "name": "综合健康度骤降",
            "condition": lambda prev, curr: (prev - curr) >= 10,
            "severity": "danger",
            "title": "学习健康度骤降{delta}分",
            "description": "综合健康度从{prev}分降至{curr}分，可能存在学习困扰",
            "actions": ["notify_parent", "recommend_check_in", "adjust_plan"],
        },
        # 规则2：任一维度进入 critical
        {
            "id": "R002",
            "name": "维度进入危险区",
            "condition": lambda dim_score: dim_score < 30,
            "severity": "danger",
            "title": "{dim_name}维度处于危险水平({score}分)",
            "description": "{dim_name}健康度仅为{score}分，需要立即关注",
            "actions": ["notify_parent", "notify_teacher", "urgent_intervention"],
        },
        # 规则3：综合健康度连续3天下降
        {
            "id": "R003",
            "name": "健康度持续下降",
            "condition": lambda trend_3d: all(t < 0 for t in trend_3d),
            "severity": "warning",
            "title": "学习健康度连续3天下降",
            "description": "综合健康度已连续3天呈下降趋势（{d1}→{d2}→{d3}），建议排查原因",
            "actions": ["notify_student", "adjust_plan", "recommend_rest"],
        },
        # 规则4：均衡度严重不足
        {
            "id": "R004",
            "name": "维度严重失衡",
            "condition": lambda balance_score: balance_score < 30,
            "severity": "warning",
            "title": "学习发展严重不均衡（均衡度{balance}分）",
            "description": "各维度差距过大，存在明显的偏科或偏废情况",
            "actions": ["notify_parent", "recommend_balanced_plan"],
        },
        # 规则5：心理状态维度持续低位
        {
            "id": "R005",
            "name": "心理状态持续低迷",
            "condition": lambda d2_history: all(s < 45 for s in d2_history[-3:]),
            "severity": "danger",
            "title": "心理状态健康持续偏低",
            "description": "心理状态维度连续3天低于45分，可能存在学习焦虑或倦怠",
            "actions": ["notify_parent", "recommend_counseling", "reduce_load"],
        },
    ]

    def check_and_alert(
        self,
        student_id: int,
        current_result: HealthEvaluationResult,
        prev_result: HealthEvaluationResult | None,
        history_scores: list[float],
        dim_history: dict[str, list[float]],
    ) -> list[dict]:
        """
        执行预警规则检查，返回触发的预警列表。
        """
        triggered = []

        for rule in self.RULES:
            rule_id = rule["id"]

            # 去重：同一天同一规则不重复触发
            if self._alert_exists(student_id, rule_id, current_result.snapshot_date):
                continue

            fired = False
            context = {}

            if rule_id == "R001" and prev_result:
                delta = prev_result.overall_score - current_result.overall_score
                if delta >= 10:
                    fired = True
                    context = {
                        "delta": round(delta, 1),
                        "prev": round(prev_result.overall_score, 1),
                        "curr": round(current_result.overall_score, 1),
                    }

            elif rule_id == "R002":
                for dim_code, snap in current_result.dimension_scores.items():
                    if snap < 30:
                        fired = True
                        context = {"dim_name": DIM_NAMES[dim_code], "score": snap}
                        break

            elif rule_id == "R003" and len(history_scores) >= 3:
                trend_3d = [
                    history_scores[-2] - history_scores[-3],
                    history_scores[-1] - history_scores[-2],
                    current_result.overall_score - history_scores[-1],
                ]
                if all(t < 0 for t in trend_3d):
                    fired = True
                    context = {
                        "d1": round(history_scores[-3], 1),
                        "d2": round(history_scores[-2], 1),
                        "d3": round(history_scores[-1], 1),
                    }

            elif rule_id == "R004":
                if current_result.balance_score < 30:
                    fired = True
                    context = {"balance": round(current_result.balance_score, 1)}

            elif rule_id == "R005":
                d2_hist = dim_history.get("D2", [])
                if len(d2_hist) >= 3 and all(s < 45 for s in d2_hist[-3:]):
                    fired = True
                    context = {"avg_score": round(sum(d2_hist[-3:]) / 3, 1)}

            if fired:
                alert = self._build_alert(student_id, rule, context, current_result)
                triggered.append(alert)
                self._persist_alert(alert)
                self._dispatch_alert(alert)

        return triggered

    def _build_alert(self, student_id, rule, context, result) -> dict:
        return {
            "student_id": student_id,
            "alert_type": rule["id"],
            "severity": rule["severity"],
            "title": rule["title"].format(**context),
            "description": rule["description"].format(**context),
            "suggested_actions": rule["actions"],
            "metric_value": result.overall_score,
            "threshold": 40.0 if rule["severity"] == "danger" else 55.0,
            "status": "active",
            "triggered_at": datetime.now(),
        }

    def _dispatch_alert(self, alert: dict):
        """
        分发预警消息到对应渠道。
        """
        severity = alert["severity"]
        actions = alert.get("suggested_actions", [])

        if "notify_parent" in actions and severity == "danger":
            # 通过推送服务发送给家长
            message_service.send_to_parent(
                student_id=alert["student_id"],
                title=alert["title"],
                body=alert["description"],
                priority="high",
            )

        if "notify_teacher" in actions:
            # 通知班主任
            message_service.send_to_teacher(
                student_id=alert["student_id"],
                title=alert["title"],
                body=alert["description"],
            )

        # 写入学生端站内信（所有级别）
        message_service.send_in_app(
            user_id=alert["student_id"],
            title=alert["title"],
            body=alert["description"],
            category="health_alert",
        )
```

```python
class DimensionFallbackHandler:
    """维度引擎不可用时的降级处理"""

    def handle_unavailable_dim(
        self, dim_code: str, student_id: int, eval_date: date
    ) -> DimensionSnapshot:
        """
        降级策略优先级：
        1. 使用昨天缓存数据（衰减置信度）

        3. 使用全局默认值

        # 策略1：昨天缓存
```

```python
import asyncio

class ParallelDimensionFetcher:
    """并行拉取各维度数据"""

    async def fetch_all(
        self,
        student_id: int,
        eval_date: date,
        adapters: list[DimensionEngineAdapter],
    ) -> dict[str, DimensionResult]:
        """
        并行调用所有维度适配器，设置超时降级。
        """
        tasks = []
        for adapter in adapters:
            task = asyncio.wait_for(
                asyncio.to_thread(
                    adapter.fetch,
                    DimensionQuery(student_id, eval_date)
                ),
                timeout=2.0,  # 单维度最多2秒
            )
            tasks.append((adapter.get_dimension_code(), task))

        results = {}
        for dim_code, task in tasks:
            try:
                result = await task
                if result:
                    results[dim_code] = result
            except asyncio.TimeoutError:
                logger.warning(f"Dimension {dim_code} timed out for student {student_id}")
            except Exception as e:
                logger.error(f"Dimension {dim_code} error: {e}")

        return results
```

```python
class BatchEvalPipeline:
    """批量评估流水线 - 分阶段并行"""

    async def eval_batch(self, student_ids: list[int], eval_date: date):
        # Stage 1: 并行拉取维度数据（按维度分组批量请求）
        dim_data = await self._batch_fetch_dimensions(student_ids, eval_date)

        # Stage 2: 并行计算健康度（CPU密集，用进程池）
        results = await asyncio.gather(*[
            asyncio.to_thread(self._compute_health, sid, dim_data[sid], eval_date)
            for sid in student_ids
        ])

        # Stage 3: 批量写入数据库
        await self._batch_save(results)

        # Stage 4: 批量检查预警
        await self._batch_check_alerts(results)
```

```python
class HealthDataMasker:
    """健康度数据脱敏"""

    MASK_RULES = {
        # 心理状态子指标对家长部分隐藏
        "D2": {
            "parent": {
                "hide_fields": ["anxietyIndex", "burnoutIndex"],
                "show_level_only": ["motivationLevel"],
            },
            # 教师仅看风险级别
            "teacher": {
                "show_level_only": True,
            },
        },
        # 社交互动细节对家长仅展示概要
        "D5": {
            "parent": {"hide_fields": ["interactionsDetail"]},
            "teacher": {"show_level_only": True},
        },
    }

    def mask(self, data: dict, dim_code: str, viewer_role: str) -> dict:
        rules = self.MASK_RULES.get(dim_code, {}).get(viewer_role)
        if not rules:
            return data

        result = dict(data)
        for field in rules.get("hide_fields", []):
            result.pop(field, None)

        if rules.get("show_level_only"):
            # 只返回等级，不返回具体分数
            if isinstance(result.get("sub_metrics"), dict):
                score = result["sub_metrics"].get("score", 0)
                result["sub_metrics"] = {"level": self._score_to_level(score)}

        return result
```
