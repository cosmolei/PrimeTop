# 服务端-学习任务完成质量多维度评估与完成度智能计算引擎-详细设计

## 1. 概述

### 1.1 功能定位

学习任务完成质量多维度评估引擎是 PrimeTop 学习闭环中的核心计算组件。当学生执行各类学习任务（看视频、读文章、做练习、AI 问答、错题复习等）时，本引擎负责以多维度指标计算任务的「完成度」与「完成质量」，为每日任务推进、学习计划调度、学情报告生成和自动进阶决策提供基础数据支撑。

### 1.2 设计目标

| 目标 | 说明 |
| --- | --- |
| 统一度量 | 为视频、图文、练习、AI 对话、错题复习等异构任务提供统一的完成度计算框架 |
| 多维评估 | 不再使用二元「已完成/未完成」，而是从投入度、正确率、交互深度、理解度、保持度 5 个维度评分 |
| 实时计算 | 任务执行过程中实时输出当前完成度，驱动 UI 进度条与即时反馈 |
| 可配置 | 不同任务类型、不同学段、不同场景下的完成度判定标准可灵活配置 |
| 低延迟 | 单次完成度计算 P99 ≤ 50ms，支持高频心跳上报 |

### 1.3 适用范围

- 每日学习任务的进度推进与完成判定
- 学习计划中子任务的完成度聚合
- 章节学习进度的细粒度计算（与章节完成度评估引擎互补）
- 学情报告中的「任务完成质量」数据源
- 自适应推荐引擎的输入信号

### 2.2 完成度维度定义

```python
from dataclasses import dataclass
from typing import Optional

@dataclass
class CompletionDimensions:
    """
    五维完成度评分体系
    每个维度取值范围 [0.0, 1.0]
    """

    # 投入度 Engagement：学习时长、专注度、交互频率
    engagement: float = 0.0

    # 正确率 Accuracy：答题正确率、知识检测通过率
    accuracy: float = 0.0

    # 交互深度 Interaction Depth：内容内交互行为覆盖率（高亮、笔记、问答等）
    interaction_depth: float = 0.0

    # 理解度 Comprehension：即时检测得分、AI 辅导理解信号
    comprehension: float = 0.0

    # 保持度 Retention：间隔复习表现、延迟回忆正确率
    retention: float = 0.0

    def weighted_score(self, weights: 'DimensionWeights') -> float:
        """计算加权总分"""
        return (
            self.engagement * weights.engagement +
            self.accuracy * weights.accuracy +
            self.interaction_depth * weights.interaction_depth +
            self.comprehension * weights.comprehension +
            self.retention * weights.retention
        )
```

### 2.4 完成度等级

```python
class CompletionLevel(Enum):
    """任务完成度等级"""
    NOT_STARTED = "not_started"          # 未开始 (score = 0)
    EARLY_STAGE = "early_stage"          # 初期 (0 < score < 0.3)
    IN_PROGRESS = "in_progress"          # 进行中 (0.3 ≤ score < 0.6)
    NEAR_COMPLETE = "near_complete"      # 接近完成 (0.6 ≤ score < 0.8)
    COMPLETED = "completed"              # 已完成 (0.8 ≤ score < 0.95)
    MASTERED = "mastered"                # 掌握 (score ≥ 0.95)

    @classmethod
    def from_score(cls, score: float) -> 'CompletionLevel':
        if score <= 0:
            return cls.NOT_STARTED
        elif score < 0.3:
            return cls.EARLY_STAGE
        elif score < 0.6:
            return cls.IN_PROGRESS
        elif score < 0.8:
            return cls.NEAR_COMPLETE
        elif score < 0.95:
            return cls.COMPLETED
        else:
            return cls.MASTERED
```

## 3. 核心算法设计

### 3.1 投入度（Engagement）计算

```python
import time
from dataclasses import dataclass

@dataclass
class EngagementMetrics:
    """投入度原始指标"""
    accumulated_time_ms: int = 0       # 累计学习时长
    expected_time_ms: int = 0          # 预期学习时长（按任务复杂度）
    active_interaction_count: int = 0  # 主动交互次数（点击/滚动/输入等）
    expected_interaction_count: int = 0  # 预期交互次数
    focus_ratio: float = 1.0           # 专注比例（前台时长/总会话时长）
    pause_count: int = 0               # 暂停次数
    avg_pause_duration_ms: int = 0     # 平均暂停时长
    tab_switch_count: int = 0          # 切屏/切应用次数（客户端上报）


def calculate_engagement(metrics: EngagementMetrics) -> float:
    """
    计算投入度评分 [0, 1]

    策略：
    1. 时长完成率 (权重 50%)：实际学习时长 / 预期时长
    2. 交互活跃度 (权重 30%)：主动交互频率
    3. 专注度 (权重 20%)：前台聚焦比例，减去切屏惩罚
    """
    # 时长完成率
    if metrics.expected_time_ms > 0:
        time_ratio = min(metrics.accumulated_time_ms / metrics.expected_time_ms, 1.0)
    else:
        time_ratio = 0.5  # 无预期时长时给中等分

    # 交互活跃度
    if metrics.expected_interaction_count > 0:
        interaction_ratio = min(
            metrics.active_interaction_count / metrics.expected_interaction_count, 1.0
        )
    else:
        interaction_ratio = 0.5

    # 专注度：从基准分开始，切屏次数扣分
    focus_score = metrics.focus_ratio
    focus_score -= min(metrics.tab_switch_count * 0.05, 0.3)  # 每次切屏扣0.05，最多扣0.3
    focus_score = max(focus_score, 0.0)

    score = time_ratio * 0.50 + interaction_ratio * 0.30 + focus_score * 0.20
    return round(min(max(score, 0.0), 1.0), 3)
```

### 3.2 正确率（Accuracy）计算

```python
@dataclass
class AccuracyMetrics:
    """正确率原始指标"""
    total_questions: int = 0           # 总题数
    correct_first_try: int = 0         # 首次作答正确数
    correct_after_hint: int = 0        # 提示后正确数
    correct_after_retry: int = 0       # 重试后正确数
    wrong_answers: int = 0             # 错误作答数
    partial_credit_ratios: list = None # 部分得分率列表（主观题/分步题）


def calculate_accuracy(metrics: AccuracyMetrics) -> float:
    """
    计算正确率评分 [0, 1]

    策略：
    - 首次正确：满分
    - 提示后正确：75% 分
    - 重试后正确：50% 分
    - 部分得分：按比例
    """
    if metrics.total_questions == 0:
        return 0.0  # 无答题环节的任务，正确率维度为0（权重也为0）

    total_score = 0.0
    # 客观题计分
    total_score += metrics.correct_first_try * 1.0
    total_score += metrics.correct_after_hint * 0.75
    total_score += metrics.correct_after_retry * 0.50

    # 主观题部分得分
    if metrics.partial_credit_ratios:
        for ratio in metrics.partial_credit_ratios:
            total_score += ratio

    # 未答对的题得0分
    answered = (metrics.correct_first_try + metrics.correct_after_hint
                + metrics.correct_after_retry + metrics.wrong_answers)
    partial_count = len(metrics.partial_credit_ratios or [])
    total_items = max(metrics.total_questions, answered + partial_count)

    if total_items == 0:
        return 0.0

    return round(min(total_score / total_items, 1.0), 3)
```

### 3.3 交互深度（Interaction Depth）计算

```python
@dataclass
class InteractionMetrics:
    """交互深度原始指标"""
    # 内容定义的交互点总数
    defined_interaction_points: int = 0
    # 学生触发的交互点数
    triggered_interaction_points: int = 0
    # 交互类型权重映射
    interaction_type_weights: dict = None  # {"highlight": 0.3, "note": 0.4, "answer_inline": 0.3}
    # 各类型交互的触发次数
    interaction_type_counts: dict = None   # {"highlight": 5, "note": 2, "answer_inline": 8}
    # 交互触发的知识点关联次数
    knowledge_link_triggers: int = 0
    # 使用了哪些学习辅助工具（查词、翻译、公式等）
    tools_used: list = None


def calculate_interaction_depth(metrics: InteractionMetrics) -> float:
    """
    计算交互深度评分 [0, 1]

    策略：
    1. 交互覆盖率 (权重 50%)：触发的交互点 / 总交互点
    2. 交互类型丰富度 (权重 30%)：不同类型交互的熵
    3. 学习工具使用 (权重 20%)：辅助工具的使用广度
    """
    # 交互覆盖率
    if metrics.defined_interaction_points > 0:
        coverage = min(
            metrics.triggered_interaction_points / metrics.defined_interaction_points, 1.0
        )
    else:
        coverage = 0.5  # 无预定义交互点的内容

    # 交互类型丰富度（Shannon熵归一化）
    import math
    type_counts = metrics.interaction_type_counts or {}
    total_interactions = sum(type_counts.values())
    if total_interactions > 0 and type_counts:
        entropy = 0.0
        for count in type_counts.values():
            p = count / total_interactions
            entropy -= p * math.log2(p) if p > 0 else 0
        max_entropy = math.log2(len(type_counts)) if len(type_counts) > 1 else 1
        richness = entropy / max_entropy if max_entropy > 0 else 0.0
    else:
        richness = 0.0

    # 工具使用广度
    tools_used = metrics.tools_used or []
    tool_score = min(len(tools_used) / 3.0, 1.0)  # 使用3种以上工具即满分

    score = coverage * 0.50 + richness * 0.30 + tool_score * 0.20
    return round(min(max(score, 0.0), 1.0), 3)
```

### 3.4 理解度（Comprehension）计算

```python
@dataclass
class ComprehensionMetrics:
    """理解度原始指标"""
    # 嵌入式即时检测题得分
    inline_quiz_scores: list = None     # [0.8, 0.9, 0.7]  每个小测的得分率
    # AI 辅导对话中的理解度信号
    ai_understanding_signals: list = None  # [{"type": "SUMMARY_QUALITY", "score": 0.85}, ...]
    # 费曼学习法自讲解评分
    self_explanation_score: float = 0.0
    # 知识点关联准确率（能否正确指出知识点间关系）
    knowledge_mapping_accuracy: float = 0.0
    # 提问质量分（学生提出的问题深度）
    question_quality_score: float = 0.0


def calculate_comprehension(metrics: ComprehensionMetrics) -> float:
    """
    计算理解度评分 [0, 1]

    策略：
    有嵌入式检测题时以检测题为主；
    无检测题时以 AI 信号和自讲解为补充。
    """
    scores = []

    # 嵌入式检测题（权重 45%）
    inline_scores = metrics.inline_quiz_scores or []
    if inline_scores:
        avg_inline = sum(inline_scores) / len(inline_scores)
        scores.append(("inline", avg_inline, 0.45))

    # AI 理解信号（权重 25%）
    ai_signals = metrics.ai_understanding_signals or []
    if ai_signals:
        avg_ai = sum(s["score"] for s in ai_signals) / len(ai_signals)
        scores.append(("ai", avg_ai, 0.25))

    # 自讲解评分（权重 15%）
    if metrics.self_explanation_score > 0:
        scores.append(("explanation", metrics.self_explanation_score, 0.15))

    # 知识点关联准确率（权重 10%）
    if metrics.knowledge_mapping_accuracy > 0:
        scores.append(("mapping", metrics.knowledge_mapping_accuracy, 0.10))

    # 提问质量（权重 5%）
    if metrics.question_quality_score > 0:
        scores.append(("question", metrics.question_quality_score, 0.05))

    if not scores:
        return 0.0

    # 归一化权重（实际权重之和可能<1，需要归一化）
    total_weight = sum(w for _, _, w in scores)
    weighted_sum = sum(score * weight for _, score, weight in scores)

    return round(min(weighted_sum / total_weight if total_weight > 0 else 0.0, 1.0), 3)
```

### 3.5 保持度（Retention）计算

```python
@dataclass
class RetentionMetrics:
    """保持度原始指标"""
    # 是否有间隔复习数据
    has_review_data: bool = False
    # 最近一次复习正确率
    latest_review_accuracy: float = 0.0
    # 复习间隔天数（距上次学习）
    days_since_last_study: int = 0
    # 遗忘曲线预测保持率
    predicted_retention_rate: float = 0.0
    # 复习次数
    review_count: int = 0
    # 复习正确率趋势
    review_accuracy_trend: list = None  # [0.6, 0.7, 0.85] 越来越好


def calculate_retention(metrics: RetentionMetrics) -> float:
    """
    计算保持度评分 [0, 1]

    策略：
    - 有复习数据时，以复习正确率 + 趋势为主
    - 无复习数据时，使用遗忘曲线预测值
    - 首次学习（无复习）时给基准分
    """
    if not metrics.has_review_data:
        if metrics.predicted_retention_rate > 0:
            # 使用遗忘曲线模型预测值
            return round(metrics.predicted_retention_rate, 3)
        else:
            # 首次学习，无复习数据，给予基准分
            return 0.5

    # 有复习数据
    score = metrics.latest_review_accuracy * 0.60  # 最近正确率占 60%

    # 趋势加分（正确率上升则加分）
    trend = metrics.review_accuracy_trend or []
    if len(trend) >= 2:
        trend_delta = trend[-1] - trend[0]
        if trend_delta > 0:
            score += min(trend_delta * 0.5, 0.20)  # 趋势上升最多加 0.20
        else:
            score += max(trend_delta * 0.3, -0.15)  # 趋势下降最多扣 0.15

    # 复习次数加分（更多复习 = 更好保持，但边际递减）
    review_bonus = min(0.20 * (1 - pow(0.7, metrics.review_count)), 0.20)
    score += review_bonus

    return round(min(max(score, 0.0), 1.0), 3)
```

### 3.6 综合完成度计算

```python
class CompletionEvaluationEngine:
    """完成度评估引擎核心"""

    def __init__(self, config_service: 'ConfigService', cache: 'RedisClient'):
        self.config = config_service
        self.cache = cache

    async def evaluate(
        self,
        user_id: int,
        task_id: str,
        task_type: TaskType,
        stage_code: str,
        raw_metrics: dict,
        is_final: bool = False
    ) -> 'CompletionResult':
        """
        执行完成度评估

        Args:
            user_id: 用户ID
            task_id: 任务ID
            task_type: 任务类型
            stage_code: 学段编码
            raw_metrics: 各维度的原始指标数据
            is_final: 是否为最终评估（任务结束时）

        Returns:
            CompletionResult: 完成度评估结果
        """
        # 1. 获取权重配置
        weights = await self._get_weights(task_type, stage_code)

        # 2. 分维度计算
        engagement = calculate_engagement(
            EngagementMetrics(**raw_metrics.get("engagement", {}))
        )
        accuracy = calculate_accuracy(
            AccuracyMetrics(**raw_metrics.get("accuracy", {}))
        )
        interaction_depth = calculate_interaction_depth(
            InteractionMetrics(**raw_metrics.get("interaction", {}))
        )
        comprehension = calculate_comprehension(
            ComprehensionMetrics(**raw_metrics.get("comprehension", {}))
        )
        retention = calculate_retention(
            RetentionMetrics(**raw_metrics.get("retention", {}))
        )

        dimensions = CompletionDimensions(
            engagement=engagement,
            accuracy=accuracy,
            interaction_depth=interaction_depth,
            comprehension=comprehension,
            retention=retention,
        )

        # 3. 加权总分
        overall = dimensions.weighted_score(weights)

        # 4. 确定等级
        level = CompletionLevel.from_score(overall)

        # 5. 构建结果
        result = CompletionResult(
            user_id=user_id,
            task_id=task_id,
            task_type=task_type,
            dimensions=dimensions,
            overall_score=round(overall, 3),
            completion_level=level,
            raw_metrics=raw_metrics,
            is_final=is_final,
            timestamp=int(time.time() * 1000),
        )

        # 6. 持久化
        if is_final:
            await self._persist_final_result(result)
        else:
            await self._update_realtime_cache(result)

        # 7. 检查等级变化并发出事件
        await self._check_and_emit_level_change(result)

        return result

    async def _get_weights(self, task_type: TaskType, stage_code: str) -> DimensionWeights:
        """获取权重配置（优先缓存，其次DB，最后默认值）"""
        cache_key = WEIGHTS_CACHE_KEY.format(
            task_type=task_type.value, stage_code=stage_code
        )
        cached = await self.cache.hgetall(cache_key)
        if cached:
            return DimensionWeights(
                engagement=float(cached.get("w_engagement", 0.20)),
                accuracy=float(cached.get("w_accuracy", 0.30)),
                interaction_depth=float(cached.get("w_interaction_depth", 0.15)),
                comprehension=float(cached.get("w_comprehension", 0.25)),
                retention=float(cached.get("w_retention", 0.10)),
            )
        # 回退到默认值
        return DEFAULT_WEIGHTS.get(task_type, DimensionWeights())
```

```python
class CompletionStateMachine:
    """完成度等级状态机"""

    TRANSITIONS = {
        # (当前等级, 事件) → 目标等级
        (CompletionLevel.NOT_STARTED, "START"): CompletionLevel.EARLY_STAGE,
        (CompletionLevel.EARLY_STAGE, "SCORE_UPDATE"): None,  # 动态计算
        (CompletionLevel.IN_PROGRESS, "SCORE_UPDATE"): None,
        (CompletionLevel.NEAR_COMPLETE, "SCORE_UPDATE"): None,
        (CompletionLevel.COMPLETED, "SCORE_UPDATE"): None,
        (CompletionLevel.COMPLETED, "RETENTION_BOOST"): CompletionLevel.MASTERED,
        (CompletionLevel.MASTERED, "RETENTION_DECAY"): CompletionLevel.COMPLETED,
        # 任意状态 → ABANDONED
        (None, "ABANDON"): CompletionLevel.NOT_STARTED,  # 归零
        # 任意状态 → 超时
        (None, "TIMEOUT"): None,  # 保持当前分数，状态变为 EXPIRED
    }

    @classmethod
    def transition(
        cls,
        current: CompletionLevel,
        new_score: float,
        event: str = "SCORE_UPDATE"
    ) -> CompletionLevel:
        """根据分数变化确定新等级"""
        if event in ("ABANDON",):
            return CompletionLevel.NOT_STARTED

        # SCORE_UPDATE：直接从分数计算等级
        target = CompletionLevel.from_score(new_score)

        # 特殊规则：保持度提升可从 COMPLETED → MASTERED
        if (current == CompletionLevel.COMPLETED
            and event == "RETENTION_BOOST"
            and new_score >= 0.95):
            return CompletionLevel.MASTERED

        # 保持度衰减：MASTERED → COMPLETED
        if (current == CompletionLevel.MASTERED
            and event == "RETENTION_DECAY"
            and new_score < 0.95):
            return CompletionLevel.COMPLETED

        # 不允许回退（除了保持度衰减场景）
        # 即：分数下降不会让 COMPLETED 变回 IN_PROGRESS
        order = [
            CompletionLevel.NOT_STARTED,
            CompletionLevel.EARLY_STAGE,
            CompletionLevel.IN_PROGRESS,
            CompletionLevel.NEAR_COMPLETE,
            CompletionLevel.COMPLETED,
            CompletionLevel.MASTERED,
        ]
        if event == "SCORE_UPDATE" and order.index(target) < order.index(current):
            # 分数下降但不低于当前等级的下限 → 保持当前等级
            return current

        return target
```

### 5.3 关键业务规则

| 规则 | 说明 |
| --- | --- |
| 单向递进 | 正常学习过程中等级只升不降（分数下降时保持当前等级） |
| 保持度衰减例外 | MASTERED 状态在间隔复习后如果正确率 < 80%，降级为 COMPLETED |
| 任务超时 | 超过预期完成时间 3 倍后任务标记为 EXPIRED，冻结当前完成度 |
| 重做任务 | 用户重做已完成的任务时，取两次完成度的较高值 |
| 首次完成奖励 | 首次达到 COMPLETED 时发放积分，达到 MASTERED 时额外发放奖励 |

## 6. 混合任务完成度聚合

### 6.1 混合任务结构

一个「混合任务」包含多个子任务，需要按子任务权重聚合完成度。

```python
@dataclass
class SubTask:
    """子任务定义"""
    sub_task_id: str
    task_type: TaskType
    weight: float          # 子任务在混合任务中的权重
    is_required: bool      # 是否为必须完成的子任务


class MixedTaskAggregator:
    """混合任务完成度聚合器"""

    def aggregate(
        self,
        sub_tasks: list[SubTask],
        sub_results: dict[str, CompletionDimensions],
        weights_map: dict[str, DimensionWeights]
    ) -> CompletionDimensions:
        """
        聚合子任务完成度

        策略：
        1. 如果有必须完成的子任务未达标，整体不达标
        2. 按子任务权重加权平均各维度分数
        3. 必须子任务未完成的维度分数打 50% 折扣
        """
        total_weight = sum(st.weight for st in sub_tasks)
        agg = {dim: 0.0 for dim in
               ["engagement", "accuracy", "interaction_depth", "comprehension", "retention"]}

        for sub in sub_tasks:
            result = sub_results.get(sub.sub_task_id)
            if result is None:
                continue

            weight_ratio = sub.weight / total_weight if total_weight > 0 else 0

            # 检查必须子任务是否达标
            if sub.is_required:
                sub_overall = result.weighted_score(weights_map.get(sub.sub_task_id,
                                                                     DimensionWeights()))
                if sub_overall < 0.60:  # 未达标折扣
                    weight_ratio *= 0.5

            agg["engagement"] += result.engagement * weight_ratio
            agg["accuracy"] += result.accuracy * weight_ratio
            agg["interaction_depth"] += result.interaction_depth * weight_ratio
            agg["comprehension"] += result.comprehension * weight_ratio
            agg["retention"] += result.retention * weight_ratio

        return CompletionDimensions(**agg)
```

```python
import asyncio

from typing import Optional

class HeartbeatHandler:

            "completion_level": result.completion_level.value,
            "dimensions": {
                "engagement": result.dimensions.engagement,
                "accuracy": result.dimensions.accuracy,
                "interaction_depth": result.dimensions.interaction_depth,
                "comprehension": result.dimensions.comprehension,
                "retention": result.dimensions.retention,
            },
            "progress_percent": round(result.overall_score * 100, 1),
        }

    def _merge_metrics(self, cached: dict, delta: dict) -> dict:
        """合并增量指标"""
        result = json.loads(cached.get("raw_metrics", "{}"))
        for dim, dim_delta in delta.items():
            if dim not in result:
                result[dim] = {}
            for key, value in dim_delta.items():
                if key.endswith("_ms") or key.endswith("_count"):
                    result[dim][key] = result[dim].get(key, 0) + value
                elif key == "new_answers":
                    result[dim].setdefault("answers", []).extend(value)
                elif isinstance(value, list):
                    result[dim].setdefault(key, []).extend(value)
                else:
                    result[dim][key] = value
        return result
```

```python
class CompletionEvaluationEngine:
    # ... 前文代码 ...

    async def evaluate_with_fallback(
        self,
        user_id: int,
        task_id: str,
        task_type: TaskType,
        raw_metrics: dict,
        **kwargs
    ) -> 'CompletionResult':
        """带降级的完成度评估"""

        try:
            # 正常流程
            return await self.evaluate(user_id, task_id, task_type, ...)
        except RedisUnavailableError:
            # Redis 不可用 → 降级为纯内存计算
            logger.warning("Redis unavailable, fallback to in-memory calculation")
            return self._evaluate_in_memory(user_id, task_id, task_type, raw_metrics)
        except ConfigServiceUnavailableError:
            # 配置服务不可用 → 使用默认配置
            logger.warning("Config service unavailable, using default weights")
            weights = DEFAULT_WEIGHTS.get(task_type, DimensionWeights())
            return self._evaluate_with_weights(raw_metrics, weights)
        except Exception as e:
            # 完全降级：使用最简模型（只看时长 + 正确率）
            logger.error(f"Completion evaluation failed, using minimal model: {e}")
            return self._minimal_evaluate(user_id, task_id, task_type, raw_metrics)

    def _minimal_evaluate(
        self, user_id: int, task_id: str, task_type: TaskType, raw_metrics: dict
    ) -> 'CompletionResult':
        """
        最简完成度模型（完全降级模式）
        仅基于时长和答题正确率
        """
        engagement_data = raw_metrics.get("engagement", {})
        accuracy_data = raw_metrics.get("accuracy", {})

        # 简单时长比率
        time_ms = engagement_data.get("accumulated_time_ms", 0)
        expected_ms = engagement_data.get("expected_time_ms", 600000)  # 默认10分钟
        engagement = min(time_ms / expected_ms, 1.0) if expected_ms > 0 else 0.5

        # 简单正确率
        total = accuracy_data.get("total_questions", 0)
        correct = accuracy_data.get("correct_first_try", 0)
        accuracy = correct / total if total > 0 else 0.0

        overall = engagement * 0.5 + accuracy * 0.5
        level = CompletionLevel.from_score(overall)

        return CompletionResult(
            user_id=user_id,
            task_id=task_id,
            task_type=task_type,
            dimensions=CompletionDimensions(
                engagement=engagement,
                accuracy=accuracy,
                interaction_depth=0.0,
                comprehension=0.0,
                retention=0.0,
            ),
            overall_score=round(overall, 3),
            completion_level=level,
            raw_metrics=raw_metrics,
            is_final=False,
            timestamp=int(time.time() * 1000),
            degraded=True,  # 标记为降级结果
        )
```

### 8.3 幂等性保障

```python
import hashlib

class CompletionIdempotency:
    """完成度提交幂等性控制"""

    IDEMPOTENCY_TTL = 86400  # 24小时

    @staticmethod
    def make_key(user_id: int, task_id: str, action: str) -> str:
        """生成幂等键"""
        raw = f"{user_id}:{task_id}:{action}"
        return f"tc:idem:{hashlib.sha256(raw.encode()).hexdigest()[:16]}"

    async def check_and_mark(
        self, user_id: int, task_id: str, action: str
    ) -> tuple[bool, Optional[dict]]:
        """
        检查幂等性，如果已处理则返回之前的结果

        Returns:
            (is_duplicate, previous_result)
        """
        key = self.make_key(user_id, task_id, action)
        previous = await self.cache.get(key)
        if previous:
            return True, json.loads(previous)

        # 标记为处理中（SET NX）
        acquired = await self.cache.set(key, json.dumps({"status": "processing"}),
                                        nx=True, ex=self.IDEMPOTENCY_TTL)
        if not acquired:
            # 被其他请求抢先处理
            previous = await self.cache.get(key)
            return True, json.loads(previous) if previous else (True, None)

        return False, None

    async def complete(
        self, user_id: int, task_id: str, action: str, result: dict
    ):
        """标记处理完成"""
        key = self.make_key(user_id, task_id, action)
        await self.cache.set(key, json.dumps(result), ex=self.IDEMPOTENCY_TTL)
```

```python
class CompletionFlushJob:
    """
    定时任务：将 Redis 中的完成度数据批量写入 DB
    执行频率：每 5 分钟
    """

    async def run(self):
        # 1. 扫描活跃任务的所有缓存键
        keys = await self.cache.scan(match="tc:realtime:*")

        if not keys:
            return

        # 2. 批量读取
        pipe = self.cache.pipeline()
        for key in keys:
            pipe.hgetall(key)
        results = await pipe.execute()

        # 3. 批量 UPSERT
        upsert_batch = []
        for key, data in zip(keys, results):
            if not data:
                continue
            user_id, task_id = self._parse_key(key)
            upsert_batch.append({
                "user_id": user_id,
                "task_id": task_id,
                "engagement_score": float(data.get("engagement", 0)),
                "accuracy_score": float(data.get("accuracy", 0)),
                "interaction_depth_score": float(data.get("interaction_depth", 0)),
                "comprehension_score": float(data.get("comprehension", 0)),
                "retention_score": float(data.get("retention", 0)),
                "overall_score": float(data.get("overall", 0)),
                "completion_level": data.get("level", "NOT_STARTED"),
                "last_active_at": datetime.fromtimestamp(
                    int(data.get("last_heartbeat_ts", 0)) / 1000
                ),
            })

        if upsert_batch:
            await self.db.batch_upsert(
                "task_completion_records",
                upsert_batch,
                conflict_on=["user_id", "task_id"]
            )
            logger.info(f"Flushed {len(upsert_batch)} completion records to DB")
```

```python
class AntiCheatValidator:
    """完成度防作弊校验"""

    # 可疑行为阈值
    SUSPICIOUS_THRESHOLDS = {
        "max_time_per_day_ms": 12 * 3600 * 1000,   # 单任务日学习超12小时
        "max_heartbeat_interval_ms": 120 * 1000,     # 心跳间隔超2分钟（非暂停）
        "min_answer_time_ms": 1000,                  # 答题时间少于1秒
        "max_tab_switch_per_min": 10,                # 每分钟切屏超过10次
    }

    async def validate(
        self, user_id: int, task_id: str, metrics: dict
    ) -> tuple[bool, list[str]]:
        """
        校验指标数据合理性
        Returns: (is_valid, warning_codes)
        """
        warnings = []

        # 时间合理性
        time_ms = metrics.get("engagement", {}).get("accumulated_time_ms", 0)
        if time_ms > self.SUSPICIOUS_THRESHOLDS["max_time_per_day_ms"]:
            warnings.append("EXCESSIVE_TIME")

        # 答题速度
        answers = metrics.get("accuracy", {}).get("new_answers", [])
        for ans in answers:
            if ans.get("time_spent_ms", 999999) < self.SUSPICIOUS_THRESHOLDS["min_answer_time_ms"]:
                warnings.append("TOO_FAST_ANSWER")
                break

        if warnings:
            logger.warning(
                f"Suspicious completion metrics: user={user_id}, "
                f"task={task_id}, warnings={warnings}"
            )
            # 不直接拒绝，而是标记为需人工审核
            await self._flag_for_review(user_id, task_id, warnings)

        return len(warnings) == 0, warnings
```
