# 服务端-学生学习卡点智能诊断与突破辅导策略推荐引擎 详细设计

## 1. 概述

### 1.1 模块定位

学生学习卡点智能诊断引擎是 PrimeTop 学习闭环中的核心诊断中枢。不同于"薄弱知识点分析"（静态的知识掌握度评估）或"错题模式挖掘"（对错误类型进行分类），**学习卡点诊断**聚焦于动态识别学生在学习过程中"卡住"的时刻——即学生反复尝试仍无法突破的具体学习障碍，并生成针对性的突破策略。

**核心区别：**
| 维度 | 薄弱知识点分析 | 错题模式挖掘 | 学习卡点诊断（本模块） |
|------|--------------|------------|---------------------|
| 关注点 | 哪些知识点掌握度低 | 学生常犯什么类型的错误 | 学生在哪个具体环节卡住无法推进 |
| 时效性 | 静态/周期性 | 准实时 | 实时感知 + 会话级诊断 |
| 输出 | 知识点掌握度分数 | 错误模式标签 | 卡点类型 + 根因 + 突破策略 |
| 触发 | 定时计算 | 答题错误时 | 连续失败/长时间停滞/放弃行为 |

### 1.2 核心职责

1. **卡点实时感知**：在学生答题、学习、对话过程中实时检测"卡住"信号
2. **卡点根因诊断**：分析卡点产生的深层原因（概念缺失/程序性错误/前置依赖不足/认知超载/心因性障碍）
3. **突破策略生成**：基于卡点类型和根因，生成个性化突破方案
4. **干预效果追踪**：持续监控突破策略的执行效果并动态调整
5. **卡点趋势分析**：长期跟踪学生卡点模式，为教学策略优化提供数据支撑

## 2. 数据模型

### 2.1 核心实体定义

#### 2.1.1 学习卡点记录 (learning_stuck_point)

```sql
CREATE TABLE learning_stuck_point (
    id              BIGINT       PRIMARY KEY,
    student_id      BIGINT       NOT NULL COMMENT '学生ID',
    subject_code    VARCHAR(20)  NOT NULL COMMENT '学科编码',
    knowledge_point_id BIGINT    COMMENT '关联知识点ID',
    question_id     BIGINT       COMMENT '触发卡点的题目ID（如有）',
    
    -- 卡点基本信息
    stuck_type      VARCHAR(40)  NOT NULL COMMENT '卡点类型: CONCEPT_GAP|PROCEDURAL_ERROR|PREREQUISITE_MISSING|COGNITIVE_OVERLOAD|PSYCHOLOGICAL_BLOCK|MISCONCEPTION',
    severity_level  TINYINT      NOT NULL COMMENT '严重程度: 1=轻微 2=中等 3=严重 4=极高',
    status          VARCHAR(20)  NOT NULL DEFAULT 'ACTIVE' COMMENT '状态: ACTIVE|INTERVENING|RESOLVED|ABANDONED',
    
    -- 卡点信号
    trigger_signals JSON         NOT NULL COMMENT '触发信号集合',
    stuck_context   JSON         COMMENT '卡点发生时的学习上下文',
    
    -- 根因诊断
    root_cause      JSON         COMMENT '根因分析结果',
    diagnosis_confidence DECIMAL(3,2) DEFAULT 0 COMMENT '诊断置信度 0.00-1.00',
    
    -- 关联数据
    attempt_count   INT          DEFAULT 0 COMMENT '在卡点上的尝试次数',
    first_stuck_at  DATETIME(3)  NOT NULL COMMENT '首次卡住时间',
    last_stuck_at   DATETIME(3)  NOT NULL COMMENT '最近卡住时间',
    duration_seconds INT         DEFAULT 0 COMMENT '总卡住时长（秒）',
    
    -- 解决信息
    resolved_at     DATETIME(3)  COMMENT '突破时间',
    resolution_strategy_id BIGINT COMMENT '采用的突破策略ID',
    resolution_summary TEXT     COMMENT '突破过程摘要',
    
    -- 元数据
    tenant_id       VARCHAR(32)  NOT NULL,
    created_at      DATETIME(3)  DEFAULT CURRENT_TIMESTAMP(3),
    updated_at      DATETIME(3)  DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
    
    INDEX idx_student_subject (student_id, subject_code, status),
    INDEX idx_knowledge_point (knowledge_point_id, status),
    INDEX idx_stuck_time (first_stuck_at),
    INDEX idx_severity (severity_level, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='学习卡点记录';
```

#### 2.1.2 卡点触发信号 (stuck_trigger_signal)

```sql
CREATE TABLE stuck_trigger_signal (
    id              BIGINT       PRIMARY KEY,
    stuck_point_id  BIGINT       NOT NULL COMMENT '关联卡点ID',
    
    signal_type     VARCHAR(50)  NOT NULL COMMENT '信号类型',
    signal_value    JSON         NOT NULL COMMENT '信号值',
    signal_strength DECIMAL(3,2) NOT NULL COMMENT '信号强度 0.00-1.00',
    detected_at     DATETIME(3)  NOT NULL COMMENT '检测时间',
    
    INDEX idx_stuck_point (stuck_point_id),
    INDEX idx_signal_type (signal_type, detected_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='卡点触发信号明细';
```

#### 2.1.3 突破策略推荐 (breakthrough_strategy)

```sql
CREATE TABLE breakthrough_strategy (
    id                  BIGINT       PRIMARY KEY,
    stuck_point_id      BIGINT       NOT NULL COMMENT '关联卡点ID',
    
    strategy_type       VARCHAR(40)  NOT NULL COMMENT '策略类型',
    strategy_content    JSON         NOT NULL COMMENT '策略详细内容',
    priority            INT          DEFAULT 0 COMMENT '推荐优先级（越大越优先）',
    
    -- 执行追踪
    execution_status    VARCHAR(20)  DEFAULT 'PENDING' COMMENT '执行状态: PENDING|EXECUTING|COMPLETED|FAILED',
    executed_at         DATETIME(3)  COMMENT '开始执行时间',
    completed_at        DATETIME(3)  COMMENT '完成执行时间',
    effectiveness_score DECIMAL(3,2) COMMENT '有效性评分 0.00-1.00',
    feedback            TEXT         COMMENT '执行效果反馈',
    
    created_at          DATETIME(3)  DEFAULT CURRENT_TIMESTAMP(3),
    updated_at          DATETIME(3)  DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
    
    INDEX idx_stuck_point (stuck_point_id),
    INDEX idx_status (execution_status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='突破策略推荐';
```

### 2.3 卡点类型定义

```java
/**
 * 学习卡点类型枚举
 */
public enum StuckType {
    
    /**
     * 概念缺失 — 学生对核心概念根本不理解
     * 典型信号：连续答错 + 反复查看 + 困惑追问
     * 典型策略：回到前置概念重新讲解 + 使用生活化类比
     */
    CONCEPT_GAP("概念缺失", 3),
    
    /**
     * 程序性错误 — 理解概念但解题步骤/方法出错
     * 典型信号：反复修改 + 分步得分但某步总错
     * 典型策略：分步骤拆解 + 每步单独练习
     */
    PROCEDURAL_ERROR("程序性错误", 2),
    
    /**
     * 前置依赖不足 — 缺少必要的前置知识
     * 典型信号：话题回退 + 进度停滞 + 错误集中在综合题
     * 典型策略：前置知识快速复习 + 桥梁练习
     */
    PREREQUISITE_MISSING("前置依赖不足", 3),
    
    /**
     * 认知超载 — 内容难度超出当前认知能力
     * 典型信号：长时间犹豫 + 完成率低 + 中断
     * 典型策略：降低难度 + 分块学习 + 视觉化辅助
     */
    COGNITIVE_OVERLOAD("认知超载", 2),
    
    /**
     * 心因性障碍 — 非能力原因，而是心理因素
     * 典型信号：挫败表达 + 快速乱猜 + 主动放弃
     * 典型策略：鼓励反馈 + 成功体验设计 + 降低期望门槛
     */
    PSYCHOLOGICAL_BLOCK("心因性障碍", 4),
    
    /**
     * 迷思概念 — 存在错误但自洽的认知模型
     * 典型信号：同一类型反复犯同样错误 + 自信作答但错误
     * 典型策略：认知冲突设计 + 对比辨析 + 纠正性练习
     */
    MISCONCEPTION("迷思概念", 3),
    ;

    private final String displayName;
    private final int defaultSeverity;
}
```

### 2.4 Redis 缓存结构

```
# 学生当前活跃卡点（Hash）
stuck:active:{student_id} → {
    "{subject_code}:{knowledge_point_id}": "{stuck_point_id}:{stuck_type}:{severity}",
    ...
}
TTL: 7天

# 学生卡点信号计数器（Hash, 滑动窗口）
stuck:signals:{student_id}:{subject_code}:{signal_type} → {
    "{knowledge_point_id_or_question_id}": "{count}:{first_detected_timestamp}",
    ...
}
TTL: 24小时

# 卡点诊断结果缓存（String JSON）
stuck:diagnosis:{stuck_point_id} → JSON
TTL: 1小时

# 学生最近突破策略（List）
stuck:strategies:recent:{student_id}:{subject_code} → [strategy_id, ...]
TTL: 7天

# 全局卡点热点（ZSet, 用于教研洞察）
stuck:hotspots:{subject_code}:{grade_level} → [{knowledge_point_id: score}, ...]
TTL: 1小时
```

### 3.1 接口列表

| # | 接口 | 方法 | 路径 | 说明 |
|---|------|------|------|------|
| 1 | 上报卡点信号 | POST | /api/v1/stuck-point/signals | 客户端/服务端模块上报触发信号 |

### 4.2 信号收集与阈值判断

#### 4.2.1 信号强度计算模型

```java
/**
 * 卡点信号强度计算器
 * 综合多维度信号计算卡点置信度
 */
@Component
public class StuckSignalStrengthCalculator {
    
    // 各信号类型的默认权重
    private static final Map<StuckSignalType, Double> SIGNAL_WEIGHTS = Map.of(
        StuckSignalType.CONSECUTIVE_WRONG,      0.25,
        StuckSignalType.REPEATED_ERASURE,       0.10,
        StuckSignalType.LONG_HESITATION,        0.10,
        StuckSignalType.RAPID_GUESS,            0.15,
        StuckSignalType.QUESTION_ABANDON,       0.15,
        StuckSignalType.REPEATED_HINT_REQUEST,  0.15,
        StuckSignalType.REPEATED_QUESTION,      0.15,
        StuckSignalType.CONFUSED_FOLLOWUP,      0.12,
        StuckSignalType.TOPIC_REGRESSION,       0.20,
        StuckSignalType.FRUSTRATION_EXPRESSION, 0.20,
        StuckSignalType.SESSION_INTERRUPTION,   0.08,
        StuckSignalType.PROGRESS_STAGNATION,    0.15,
        StuckSignalType.REPEATED_REVIEW,        0.10,
        StuckSignalType.LOW_COMPLETION_RATE,    0.12
    );
    
    // 触发阈值：信号强度之和达到此值则创建卡点
    private static final double CREATION_THRESHOLD = 0.45;
    
    // 合并阈值：在此值以下但已有活跃卡点时，将信号合并到现有卡点
    private static final double MERGE_THRESHOLD = 0.20;
    
    /**
     * 计算综合信号强度
     * 
     * @param signals 当前时间窗口内收集到的信号列表
     * @param studentProfile 学生画像（用于个性化调整权重）
     * @return 综合信号强度 [0.0, 1.0]
     */
    public double calculateStrength(List<StuckSignal> signals, 
                                     StudentProfile studentProfile) {
        double totalStrength = 0.0;
        
        for (StuckSignal signal : signals) {
            double weight = getAdjustedWeight(signal.getType(), studentProfile);
            double normalizedValue = normalizeSignalValue(signal);
            totalStrength += weight * normalizedValue;
            
            // 信号去重：同一类型同一目标的信号只取最强值
        }
        
        // 时间衰减：越早的信号贡献越小
        totalStrength *= calculateTimeDecayFactor(signals);
        
        // 信号多样性奖励：多类型信号同时出现增强置信度
        long uniqueTypes = signals.stream()
            .map(StuckSignal::getType)
            .distinct()
            .count();
        if (uniqueTypes >= 3) {
            totalStrength *= 1.15; // 15% 多样性奖励
        }
        
        return Math.min(totalStrength, 1.0);
    }
    
    /**
     * 根据学生画像调整信号权重
     * 例如：对低年级学生，挫败表达信号的权重更高
     */
    private double getAdjustedWeight(StuckSignalType type, 
                                      StudentProfile profile) {
        double baseWeight = SIGNAL_WEIGHTS.getOrDefault(type, 0.10);
        
        // 低年级学生的心理信号权重提升
        if (profile.getGradeLevel() <= GradeLevel.G3 && 
            type == StuckSignalType.FRUSTRATION_EXPRESSION) {
            return baseWeight * 1.5;
        }
        
        // 已知有学习焦虑的学生，心理信号权重提升
        if (profile.hasTrait("LEARNING_ANXIETY") &&
            (type == StuckSignalType.FRUSTRATION_EXPRESSION ||
             type == StuckSignalType.QUESTION_ABANDON)) {
            return baseWeight * 1.3;
        }
        
        // 高年级学生的行为信号权重提升（表达更隐晦）
        if (profile.getGradeLevel() >= GradeLevel.G10 &&
            type == StuckSignalType.LONG_HESITATION) {
            return baseWeight * 1.3;
        }
        
        return baseWeight;
    }
    
    /**
     * 归一化信号值到 [0, 1]
     */
    private double normalizeSignalValue(StuckSignal signal) {
        switch (signal.getType()) {
            case CONSECUTIVE_WRONG:
                // 连续错误次数：3次=0.6, 5次=0.8, 7+次=1.0
                int wrongCount = signal.getIntValue("wrongCount", 1);
                return Math.min(0.2 + wrongCount * 0.15, 1.0);
                
            case LONG_HESITATION:
                // 停留时长倍数：2x=0.5, 3x=0.7, 5x+=1.0
                double ratio = signal.getDoubleValue("ratio", 1.0);
                return Math.min(0.3 + (ratio - 1) * 0.2, 1.0);
                
            case REPEATED_HINT_REQUEST:
                // 请求提示次数：1次=0.3, 3次=0.7, 5+次=1.0
                int hintCount = signal.getIntValue("hintCount", 1);
                return Math.min(0.3 + hintCount * 0.15, 1.0);
                
            case FRUSTRATION_EXPRESSION:
                // 挫败表达直接映射
                return signal.getDoubleValue("sentimentScore", 0.5);
                
            default:
                return 0.5; // 默认中等强度
        }
    }
    
    private double calculateTimeDecayFactor(List<StuckSignal> signals) {
        if (signals.isEmpty()) return 0;
        
        long now = System.currentTimeMillis();
        double factor = 0;
        for (StuckSignal signal : signals) {
            long ageHours = (now - signal.getDetectedAt()) / (1000 * 60 * 60);
            // 24小时内全权重，之后线性衰减，72小时后为0
            if (ageHours <= 24) {
                factor += 1.0;
            } else if (ageHours <= 72) {
                factor += 1.0 - (ageHours - 24) / 48.0;
            }
        }
        return factor / signals.size();
    }
}
```

### 4.3 卡点根因诊断算法

#### 4.3.1 多因素诊断决策树

```java
/**
 * 卡点根因诊断引擎
 * 使用规则引擎 + 加权评分模型进行根因诊断
 */
@Component
public class StuckPointDiagnosticEngine {
    
    @Autowired private KnowledgeGraphService knowledgeGraphService;
    @Autowired private KnowledgeTracker knowledgeTracker;
    @Autowired private MisconceptionDetector misconceptionDetector;
    @Autowired private StudentProfileService profileService;
    
    /**
     * 执行根因诊断
     * 
     * @param stuckPoint 待诊断的卡点
     * @param context 额外诊断上下文
     * @return 根因诊断结果
     */
    public RootCauseDiagnosis diagnose(StuckPoint stuckPoint, 
                                        DiagnosticContext context) {
        
        RootCauseDiagnosis result = new RootCauseDiagnosis();
        
        // 1. 迷思概念检测（优先级最高，因为迷思概念容易与其他类型混淆）
        MisconceptionResult miscResult = misconceptionDetector.detect(
            stuckPoint.getStudentId(),
            stuckPoint.getKnowledgePointId(),
            context.getRecentAnswers()
        );
        if (miscResult.isDetected() && miscResult.getConfidence() > 0.7) {
            result.setType(StuckType.MISCONCEPTION)
                   .setConfidence(miscResult.getConfidence())
                   .setDetail("迷思概念", miscResult.getDescription())
                   .setEvidence(miscResult.getEvidence());
            return result;
        }
        
        // 2. 前置依赖检测
        PrerequisiteCheckResult prereqResult = checkPrerequisites(stuckPoint, context);
        if (prereqResult.hasGap()) {
            result.setType(StuckType.PREREQUISITE_MISSING)
                   .setConfidence(prereqResult.getConfidence())
                   .setDetail("前置知识缺陷", prereqResult.getGapDescription())
                   .setMissingPrerequisites(prereqResult.getMissingPoints());
            return result;
        }
        
        // 3. 概念理解 vs 程序性技能分类
        ConceptProcedureClassification cpResult = classifyConceptVsProcedure(stuckPoint, context);
        
        // 4. 心因性因素检测
        PsychologicalFactor psychFactor = detectPsychologicalFactors(stuckPoint, context);
        
        // 5. 认知负荷评估
        CognitiveLoadAssessment loadAssessment = assessCognitiveLoad(stuckPoint, context);
        
        // 6. 综合加权决策
        return makeWeightedDecision(stuckPoint, cpResult, psychFactor, 
                                     loadAssessment, context);
    }
    
    /**
     * 前置依赖检测
     */
    private PrerequisiteCheckResult checkPrerequisites(StuckPoint stuckPoint,
                                                        DiagnosticContext context) {
        Long kpId = stuckPoint.getKnowledgePointId();
        Long studentId = stuckPoint.getStudentId();
        
        // 从知识图谱获取所有前置知识点
        List<KnowledgePoint> prerequisites = 
            knowledgeGraphService.getPrerequisites(kpId, 2); // 最多回溯2层
        
        List<KnowledgePoint> missingPoints = new ArrayList<>();
        double totalGap = 0;
        int checkedCount = 0;
        
        for (KnowledgePoint prereq : prerequisites) {
            double mastery = knowledgeTracker.getMastery(studentId, prereq.getId());
            if (mastery < 0.6) { // 掌握度低于60%视为不足
                missingPoints.add(prereq);
                totalGap += (0.6 - mastery);
            }
            checkedCount++;
        }
        
        if (missingPoints.isEmpty()) {
            return PrerequisiteCheckResult.noGap();
        }
        
        double confidence = Math.min(totalGap / (checkedCount * 0.6), 1.0);
        String desc = String.format("发现%d个前置知识点掌握不足：%s",
            missingPoints.size(),
            missingPoints.stream()
                .map(p -> p.getName() + "(" + String.format("%.0f%%", 
                    knowledgeTracker.getMastery(studentId, p.getId()) * 100) + ")")
                .collect(Collectors.joining(", "))
        );
        
        return PrerequisiteCheckResult.hasGap(missingPoints, confidence, desc);
    }
    
    /**
     * 概念理解 vs 程序性技能分类
     * 基于答题模式分析
     */
    private ConceptProcedureClassification classifyConceptVsProcedure(
            StuckPoint stuckPoint, DiagnosticContext context) {
        
        List<AnswerRecord> recentAnswers = context.getRecentAnswers();
        
        // 分析错误模式
        int conceptualErrorCount = 0; // 概念性错误（方向性错误）
        int proceduralErrorCount = 0; // 程序性错误（步骤性错误）
        int partialCorrectCount = 0;  // 部分正确（部分步骤对）
        
        for (AnswerRecord answer : recentAnswers) {
            ErrorAnalysis analysis = answer.getErrorAnalysis();
            if (analysis != null) {
                if (analysis.getErrorPosition() == ErrorPosition.APPROACH) {
                    conceptualErrorCount++;
                } else if (analysis.getErrorPosition() == ErrorPosition.CALCULATION ||
                           analysis.getErrorPosition() == ErrorPosition.STEP) {
                    proceduralErrorCount++;
                }
                if (analysis.getCorrectSteps() > 0 && !answer.isCorrect()) {
                    partialCorrectCount++;
                }
            }
        }
        
        double conceptRatio = (double) conceptualErrorCount / Math.max(recentAnswers.size(), 1);
        double procedureRatio = (double) proceduralErrorCount / Math.max(recentAnswers.size(), 1);
        
        StuckType type;
        double confidence;
        
        if (conceptRatio > 0.6) {
            type = StuckType.CONCEPT_GAP;
            confidence = conceptRatio;
        } else if (procedureRatio > 0.6 || partialCorrectCount > recentAnswers.size() / 2) {
            type = StuckType.PROCEDURAL_ERROR;
            confidence = Math.max(procedureRatio, 
                                   (double) partialCorrectCount / recentAnswers.size());
        } else {
            type = StuckType.CONCEPT_GAP; // 默认归为概念缺失
            confidence = 0.5;
        }
        
        return new ConceptProcedureClassification(type, confidence, 
            conceptualErrorCount, proceduralErrorCount, partialCorrectCount);
    }
    
    /**
     * 心因性因素检测
     */
    private PsychologicalFactor detectPsychologicalFactors(
            StuckPoint stuckPoint, DiagnosticContext context) {
        
        double frustrationScore = 0;
        int signals = 0;
        
        // 检查挫败表达信号
        if (context.hasSignal(StuckSignalType.FRUSTRATION_EXPRESSION)) {
            frustrationScore += context.getSignalStrength(StuckSignalType.FRUSTRATION_EXPRESSION);
            signals++;
        }
        
        // 检查快速乱猜信号
        if (context.hasSignal(StuckSignalType.RAPID_GUESS)) {
            frustrationScore += 0.8;
            signals++;
        }
        
        // 检查主动放弃信号
        if (context.hasSignal(StuckSignalType.QUESTION_ABANDON)) {
            frustrationScore += 0.7;
            signals++;
        }
        
        // 学生画像中的心理特质
        StudentProfile profile = profileService.getProfile(stuckPoint.getStudentId());
        if (profile.hasTrait("LOW_CONFIDENCE")) {
            frustrationScore += 0.2;
        }
        if (profile.hasTrait("PERFECTIONISM") && stuckPoint.getAttemptCount() > 3) {
            frustrationScore += 0.2;
        }
        
        double avgScore = signals > 0 ? frustrationScore / signals : 0;
        boolean isPsychological = avgScore > 0.5 && 
                                   profile.getSubjectConfidence(stuckPoint.getSubjectCode()) < 0.4;
        
        return new PsychologicalFactor(isPsychological, avgScore);
    }
    
    /**
     * 认知负荷评估
     */
    private CognitiveLoadAssessment assessCognitiveLoad(
            StuckPoint stuckPoint, DiagnosticContext context) {
        
        // 获取知识点难度
        double kpDifficulty = knowledgeGraphService
            .getDifficulty(stuckPoint.getKnowledgePointId());
        
        // 获取学生当前认知能力水平
        double studentAbility = knowledgeTracker
            .getAbilityEstimate(stuckPoint.getStudentId(), 
                                stuckPoint.getSubjectCode());
        
        // 负荷 = 难度 - 能力
        double loadGap = kpDifficulty - studentAbility;
        
        // 参考近期学习时长（疲劳度）
        double recentStudyHours = context.getRecentStudyHours();
        double fatigueFactor = Math.max(0, (recentStudyHours - 2) / 4.0); // 超2小时开始疲劳
        
        boolean isOverload = loadGap > 0.3 || fatigueFactor > 0.5;
        double confidence = Math.min(loadGap + fatigueFactor * 0.3, 1.0);
        
        return new CognitiveLoadAssessment(isOverload, loadGap, fatigueFactor, confidence);
    }
    
    /**
     * 综合加权决策
     */
    private RootCauseDiagnosis makeWeightedDecision(
            StuckPoint stuckPoint,
            ConceptProcedureClassification cpResult,
            PsychologicalFactor psychFactor,
            CognitiveLoadAssessment loadAssessment,
            DiagnosticContext context) {
        
        Map<StuckType, Double> scores = new EnumMap<>(StuckType.class);
        
        // 概念/程序分类得分
        scores.merge(cpResult.getType(), cpResult.getConfidence() * 0.40, Double::sum);
        
        // 心因性因素得分
        if (psychFactor.isDetected()) {
            scores.merge(StuckType.PSYCHOLOGICAL_BLOCK, 
                        psychFactor.getScore() * 0.35, Double::sum);
        }
        
        // 认知负荷得分
        if (loadAssessment.isOverload()) {
            scores.merge(StuckType.COGNITIVE_OVERLOAD, 
                        loadAssessment.getConfidence() * 0.25, Double::sum);
        }
        
        // 找到得分最高的类型
        Map.Entry<StuckType, Double> top = scores.entrySet().stream()
            .max(Map.Entry.comparingByValue())
            .orElse(Map.entry(StuckType.CONCEPT_GAP, 0.3));
        
        return new RootCauseDiagnosis()
            .setType(top.getKey())
            .setConfidence(top.getValue())
            .setAllScores(scores)
            .setDetail("综合诊断", buildDiagnosisDetail(top.getKey(), 
                cpResult, psychFactor, loadAssessment));
    }
    
    private String buildDiagnosisDetail(StuckType type,
                                         ConceptProcedureClassification cpResult,
                                         PsychologicalFactor psychFactor,
                                         CognitiveLoadAssessment loadAssessment) {
        StringBuilder sb = new StringBuilder();
        sb.append("卡点类型：").append(type.getDisplayName()).append("\n");
        sb.append("错误模式：概念性错误").append(cpResult.getConceptualErrors())
          .append("次，程序性错误").append(cpResult.getProceduralErrors())
          .append("次，部分正确").append(cpResult.getPartialCorrectCount()).append("次\n");
        if (psychFactor.isDetected()) {
            sb.append("检测到心理因素影响，挫败指数：")
              .append(String.format("%.2f", psychFactor.getScore())).append("\n");
        }
        if (loadAssessment.isOverload()) {
            sb.append("认知负荷偏高，难度差距：")
              .append(String.format("%.2f", loadAssessment.getLoadGap()))
              .append("，疲劳度：")
              .append(String.format("%.2f", loadAssessment.getFatigueFactor()))
              .append("\n");
        }
        return sb.toString();
    }
}
```

        )

    /**
     * 为卡点生成突破策略列表
     */
    public List<BreakthroughStrategy> generateStrategies(StuckPoint stuckPoint,
                                                          RootCauseDiagnosis diagnosis,
                                                          StudentProfile profile) {
        List<BreakthroughStrategy> strategies = new ArrayList<>();
        
        List<StrategyPriority> templates = STRATEGY_MAP.getOrDefault(
            diagnosis.getType(), 
            List.of(new StrategyPriority(BreakthroughStrategyType.AI_TUTORING, 50))
        );
        
        for (StrategyPriority template : templates) {
            BreakthroughStrategy strategy = buildStrategy(
                template, stuckPoint, diagnosis, profile);
            if (strategy != null) {
                strategies.add(strategy);
            }
        }
        
        // 根据学生画像微调优先级
        adjustPrioritiesByProfile(strategies, profile);
        
        // 按优先级排序
        strategies.sort(Comparator.comparing(BreakthroughStrategy::getPriority).reversed());
        
        return strategies;
    }
    
    /**
     * 构建具体策略实例
     */
    private BreakthroughStrategy buildStrategy(StrategyPriority template,
                                                StuckPoint stuckPoint,
                                                RootCauseDiagnosis diagnosis,
                                                StudentProfile profile) {
        BreakthroughStrategy strategy = new BreakthroughStrategy();
        strategy.setStuckPointId(stuckPoint.getId());
        strategy.setStrategyType(template.type);
        strategy.setPriority(template.priority);
        
        switch (template.type) {
            case STEP_DECOMPOSITION:
                return buildStepDecompositionStrategy(strategy, stuckPoint, profile);
                
            case PREREQUISITE_REVIEW:
                return buildPrerequisiteReviewStrategy(strategy, diagnosis, profile);
                
            case VISUAL_AID:
                return buildVisualAidStrategy(strategy, stuckPoint, profile);
                
            case COGNITIVE_CONFLICT:
                return buildCognitiveConflictStrategy(strategy, stuckPoint, profile);
                
            case DIFFICULTY_REDUCTION:
                return buildDifficultyReductionStrategy(strategy, stuckPoint, profile);
                
            case SUCCESS_EXPERIENCE:
                return buildSuccessExperienceStrategy(strategy, stuckPoint, profile);
                
            case AI_TUTORING:
                return buildAITutoringStrategy(strategy, stuckPoint, diagnosis, profile);
                
            case CONCEPT_RETEACH:
                return buildConceptReteachStrategy(strategy, stuckPoint, profile);
                
            case TARGETED_PRACTICE:
                return buildTargetedPracticeStrategy(strategy, stuckPoint, diagnosis, profile);
                
            case ANALOGY_TRANSFER:
                return buildAnalogyTransferStrategy(strategy, stuckPoint, profile);
                
            default:
                return null;
        }
    }
    
    /**
     * 分步拆解策略
     */
    private BreakthroughStrategy buildStepDecompositionStrategy(
            BreakthroughStrategy strategy, StuckPoint stuckPoint, 
            StudentProfile profile) {
        
        // 从知识图谱获取该知识点的标准解题步骤
        List<SolutionStep> steps = knowledgeGraphService
            .getStandardSolutionSteps(stuckPoint.getKnowledgePointId());
        
        // 根据学生错误记录标注每步的薄弱程度
        Map<Integer, Double> stepWeaknessMap = analyzeStepWeakness(
            stuckPoint, steps);
        
        List<StrategyStep> strategySteps = new ArrayList<>();
        for (int i = 0; i < steps.size(); i++) {
            SolutionStep step = steps.get(i);
            double weakness = stepWeaknessMap.getOrDefault(i, 0.0);
            
            strategySteps.add(new StrategyStep(
                i + 1,
                step.getDescription(),
                step.getSuggestedPracticeType(),
                step.getRecommendedQuestionCount(),
                weakness
            ));
        }
        
        StepDecompositionContent content = new StepDecompositionContent();
        content.setSteps(strategySteps);
        content.setTotalEstimatedTime(
            strategySteps.size() * (profile.getAverageStepTimeSeconds() + 30));
        
        strategy.setStrategyContent(JsonUtils.toJson(content));
        strategy.setDescription(String.format("将%s拆分为%d个独立步骤逐步练习",
            knowledgeGraphService.getKnowledgePointName(stuckPoint.getKnowledgePointId()),
            steps.size()));
        strategy.setEstimatedTime(formatTime(content.getTotalEstimatedTime()));
        
        return strategy;
    }
    
    /**
     * 前置知识复习策略
     */
    private BreakthroughStrategy buildPrerequisiteReviewStrategy(
            BreakthroughStrategy strategy, RootCauseDiagnosis diagnosis,
            StudentProfile profile) {
        
        @SuppressWarnings("unchecked")
        List<KnowledgePoint> missingPoints = (List<KnowledgePoint>) 
            diagnosis.getDetail("missingPrerequisites");
        
        if (missingPoints == null || missingPoints.isEmpty()) {
            return null;
        }
        
        List<PrerequisiteReviewItem> reviewItems = new ArrayList<>();
        for (KnowledgePoint kp : missingPoints) {
            double currentMastery = knowledgeTracker.getMastery(
                profile.getStudentId(), kp.getId());
            
            // 为每个前置知识点推荐复习内容
            ReviewContent reviewContent = contentService
                .findBestReviewContent(kp.getId(), profile.getGradeLevel());
            
            reviewItems.add(new PrerequisiteReviewItem(
                kp.getId(),
                kp.getName(),
                currentMastery,
                reviewContent.getContentId(),
                reviewContent.getContentType(), // VIDEO, INTERACTIVE, TEXT
                reviewContent.getEstimatedTime()
            ));
        }
        
        PrerequisiteReviewContent content = new PrerequisiteReviewContent();
        content.setReviewItems(reviewItems);
        content.setTotalEstimatedTime(reviewItems.stream()
            .mapToInt(PrerequisiteReviewItem::getEstimatedTime)
            .sum());
        
        strategy.setStrategyContent(JsonUtils.toJson(content));
        strategy.setDescription(String.format("复习%d个前置知识点：%s",
            missingPoints.size(),
            missingPoints.stream()
                .map(KnowledgePoint::getName)
                .collect(Collectors.joining("、"))));
        strategy.setEstimatedTime(formatTime(content.getTotalEstimatedTime()));
        
        return strategy;
    }
    
    /**
     * 成功体验策略
     */
    private BreakthroughStrategy buildSuccessExperienceStrategy(
            BreakthroughStrategy strategy, StuckPoint stuckPoint,
            StudentProfile profile) {
        
        // 选择该知识点最简单的题目，确保学生能做对
        List<Question> easyQuestions = questionService.findQuestionsByDifficulty(
            stuckPoint.getKnowledgePointId(),
            DifficultyLevel.VERY_EASY,
            5 // 先做5道最简单的
        );
        
        // 逐渐增加难度的题目
        List<Question> mediumQuestions = questionService.findQuestionsByDifficulty(
            stuckPoint.getKnowledgePointId(),
            DifficultyLevel.EASY,
            5
        );
        
        SuccessExperienceContent content = new SuccessExperienceContent();
        content.setPhase1Questions(easyQuestions);  // 必对阶段
        content.setPhase2Questions(mediumQuestions); // 巩固阶段
        content.setEncouragementEnabled(true);
        content.setCelebrationThreshold(3); // 连续做对3道后给予奖励反馈
        
        strategy.setStrategyContent(JsonUtils.toJson(content));
        strategy.setDescription("从最简单的题目开始，逐步建立信心");
        strategy.setEstimatedTime("15-20分钟");
        
        return strategy;
    }

     * AI一对一辅导策略
     */
    private BreakthroughStrategy buildAITutoringStrategy(
            BreakthroughStrategy strategy, StuckPoint stuckPoint,
            RootCauseDiagnosis diagnosis, StudentProfile profile) {
        
        // 构建AI辅导的上下文Prompt
        AITutoringPrompt prompt = new AITutoringPrompt();
        prompt.setStudentGrade(profile.getGradeLevel());
        prompt.setSubject(stuckPoint.getSubjectCode());
        prompt.setKnowledgePointId(stuckPoint.getKnowledgePointId());

        prompt.setRootCause(diagnosis.getDetail("description"));
        prompt.setStudentErrors(getRecentErrorSummary(stuckPoint));
        prompt.setPreferredStyle(profile.getLearningStyle());

            profile.getGradeLevel());
        prompt.setTemplateId(templateId);
        
        AITutoringContent content = new AITutoringContent();
        content.setPrompt(prompt);
        content.setSessionType("FOCUSED_TUTORING");
        content.setMaxTurns(15);
        content.setSuccessCriteria("学生能独立解答该知识点中等难度题目");
        
        strategy.setStrategyContent(JsonUtils.toJson(content));
        strategy.setDescription("AI一对一深度辅导，针对性解决卡点");
        strategy.setEstimatedTime("10-15分钟");
        
        return strategy;
    }
