# 互动练习与即时反馈引擎 - 详细设计

## 1. 模块概述

### 1.1 定位与边界

互动练习与即时反馈引擎（Interactive Exercise Engine, IEE）是 PrimeTop 学习闭环中的核心练习组件，负责在学习会话中生成、渲染、批改和反馈各类微型互动练习。

**与相邻模块的关系：**

| 模块 | 区别 |
|------|------|
| 练习与测评系统 | 测评系统管理完整练习册/试卷的生命周期；IEE 提供单题级别的互动渲染与即时反馈 |
| 智能组卷与自适应测试引擎 | 组卷引擎负责题目筛选和 IRT 自适应；IEE 消费组卷结果，负责前端交互呈现 |
| 答案管控与渐进式提示引擎 | 答案管控控制揭示节奏（5 级 TIER）；IEE 调用答案管控接口获取当前可展示内容 |
| 同步课堂 / 错题复习 / 学习任务 | 上游模块调用 IEE 的练习生成 API 产出练习单元；IEE 是被消费的"练习能力层" |

**一句话边界：IEE = 题目选择 + 交互渲染 + 即时批改 + 反馈生成 + 练习会话状态管理。**

### 1.2 设计目标

1. **互动丰富**：支持 10+ 种练习题型，覆盖全学科交互需求
2. **即时反馈**：作答后 < 500ms 返回批改结果与反馈（本地批改），< 2s 返回 AI 增强反馈
3. **分龄适配**：同一题型在不同学段呈现不同的视觉风格与交互复杂度
4. **可嵌入**：提供标准化 Widget / API，可嵌入任何学习场景（同步课堂、错题复习、课前预习等）
5. **数据闭环**：每次练习产生结构化作答记录，流入学习记录与学情分析系统

### 3.3 自适应模式

```python
class AdaptiveMode(str, Enum):
    FIXED = "fixed"                # 固定题目列表，不做调整
    EASY_FIRST = "easy_first"      # 从易到难排列，不动态调整
    ADAPTIVE = "adaptive"          # 实时根据作答调整下一题难度


class AdaptiveEngine:
    """自适应选题引擎——基于 Elo 评分模型"""

    def __init__(self, config: AdaptiveConfig):
        self.initial_theta = 0.0       # 用户初始能力值
        self.k_factor = 0.5            # Elo K因子
        self.target_difficulty_offset = 0.2  # 目标题目难度比能力值高0.2

    def select_next(
        self,
        theta: float,
        pool: list[Exercise],
        history: list[AttemptResult],
        constraints: SelectionConstraints
    ) -> Exercise | None:
        """
        选择下一道题
        - theta: 当前估计能力值
        - pool: 可选题池
        - history: 本次会话历史作答
        - constraints: 知识点覆盖、题型分布等约束
        """
        # 1. 更新能力估计
        if history:
            theta = self._update_theta(theta, history[-1])

        # 2. 计算目标难度区间
        target_diff = theta + self.target_difficulty_offset
        diff_range = (target_diff - 0.3, target_diff + 0.3)

        # 3. 过滤已做过的题目和已覆盖的知识点
        done_ids = {a.exercise_id for a in history}
        covered_kps = set()
        for a in history:
            covered_kps.update(a.knowledge_point_ids)

        candidates = [
            ex for ex in pool
            if ex.id not in done_ids
            and diff_range[0] <= ex.difficulty <= diff_range[1]
        ]

        # 4. 如果目标区间无可用题目，逐步放宽
        if not candidates:
            candidates = [ex for ex in pool if ex.id not in done_ids]
            candidates.sort(key=lambda ex: abs(ex.difficulty - target_diff))
            candidates = candidates[:5]

        # 5. 应用约束（知识点优先覆盖未覆盖的）
        if covered_kps and candidates:
            uncovered = [ex for ex in candidates
                        if not set(ex.knowledge_point_ids).issubset(covered_kps)]
            if uncovered:
                candidates = uncovered

        # 6. 从候选中随机选一题（避免完全确定性）
        if candidates:
            return random.choice(candidates[:3])
        return None

    def _update_theta(self, theta: float, last: AttemptResult) -> float:
        """Elo 更新公式"""
        expected = 1.0 / (1.0 + 10 ** ((last.difficulty - theta) / 400))
        actual = 1.0 if last.is_correct else 0.0
        return theta + self.k_factor * (actual - expected)


class SelectionConstraints(BaseModel):
    """选题约束"""
    max_from_same_kp: int = 2          # 同一知识点最多连续2题
    ensure_type_variety: bool = True   # 保证题型多样性
    excluded_ids: list[str] = []       # 排除的题目ID
    required_kp_coverage: float = 0.8  # 要求覆盖80%指定知识点
```

```python
class ObjectiveGrader:
    """选择题 / 判断题 / 排序 / 匹配 / 分类——纯逻辑判定"""

    async def grade(self, exercise: Exercise, user_answer: UserAnswer, _) -> GradeResult:
        correct = exercise.answer
        ua = user_answer

        match exercise.type:
            case "single_choice" | "multi_choice" | "true_false":
                is_correct = set(ua.selected_keys) == set(correct.correct_keys)
                score = exercise.scoring.max_score if is_correct else 0

            case "sort":
                is_correct = ua.ordered_ids == correct.correct_order
                score = exercise.scoring.max_score if is_correct else 0

            case "match":
                correct_pairs = {(p.left_id, p.right_id) for p in correct.pairs}
                user_pairs = {(p.left_id, p.right_id) for p in ua.pairs}
                matched = correct_pairs & user_pairs
                total = len(correct_pairs)
                if total > 0:
                    ratio = len(matched) / total
                    is_correct = ratio == 1.0
                    score = round(ratio * exercise.scoring.max_score, 1)
                else:
                    is_correct = False
                    score = 0

            case "classify":
                # 同 match 逻辑，逐组比对
                score, is_correct = self._grade_classify(correct, ua, exercise.scoring)

            case "click_area":
                is_correct = ua.selected_area_id in correct.correct_keys
                score = exercise.scoring.max_score if is_correct else 0

            case _:
                raise ValueError(f"Unknown objective type: {exercise.type}")

        return GradeResult(
            is_correct=is_correct,
            score=score,
            max_score=exercise.scoring.max_score,
            detail=self._build_detail(exercise, correct, ua, is_correct),
            grading_method="local",
        )
```

### 4.3 半客观题批改（< 100ms）

```python
class SemiObjectiveGrader:
    """填空题——精确 / 模糊 / 数学等价匹配"""

    async def grade(self, exercise: Exercise, user_answer: UserAnswer, _) -> GradeResult:
        blanks = exercise.answer.blanks
        results = []
        total_score = 0.0

        for blank in blanks:
            user_val = user_answer.values.get(blank.blank_id, "")
            if not user_val.strip():
                results.append(BlankResult(blank_id=blank.blank_id, correct=False, score=0))
                continue

            matched = False
            for accepted in blank.accepted_values:
                match blank.match_mode:
                    case "exact":
                        matched = (user_val.strip() == accepted.strip())
                    case "fuzzy":
                        sim = self.fuzzy_matcher.similarity(user_val, accepted)
                        matched = (sim >= blank.fuzzy_threshold)
                    case "math_equiv":
                        matched = self.math_equiv.is_equivalent(user_val, accepted)
                    case "regex":
                        matched = bool(re.fullmatch(accepted, user_val.strip()))
                if matched:
                    break

            blank_score = (exercise.scoring.max_score / len(blanks)) if matched else 0
            total_score += blank_score
            results.append(BlankResult(blank_id=blank.blank_id, correct=matched, score=blank_score))

        all_correct = all(r.correct for r in results)
        return GradeResult(
            is_correct=all_correct,
            score=round(total_score, 1),
            max_score=exercise.scoring.max_score,
            detail={"blank_results": [r.dict() for r in results]},
            grading_method="local",
        )
```

### 4.4 数学等价判定

```python
class MathEquivalenceChecker:
    """
    判断两个 LaTeX 数学表达式是否数学等价
    策略：先尝试符号化简，失败则数值采样验证
    """

    def is_equivalent(self, user_latex: str, reference_latex: str) -> bool:
        # 1. 标准化 LaTeX
        u = self._normalize(user_latex)
        r = self._normalize(reference_latex)

        # 2. 精确匹配（标准化后）
        if u == r:
            return True

        # 3. 符号化简比较（sympy）
        try:
            u_expr = parse_latex(u)
            r_expr = parse_latex(r)
            diff = simplify(u_expr - r_expr)
            if diff == 0:
                return True
        except Exception:
            pass

        # 4. 数值采样验证
        return self._numerical_verify(u_expr if 'u_expr' in dir() else None,
                                       r_expr if 'r_expr' in dir() else None,
                                       u, r)

    def _normalize(self, latex: str) -> str:
        """标准化 LaTeX 字符串"""
        s = latex.strip()
        s = re.sub(r'\\left|\\right', '', s)
        s = re.sub(r'\\,', '', s)
        s = re.sub(r'\s+', ' ', s)
        return s

    def _numerical_verify(self, u_expr, r_expr, u_str, r_str, n_samples=10) -> bool:
        """随机采样数值验证"""
        import random
        for _ in range(n_samples):
            x = random.uniform(-10, 10)
            try:
                u_val = self._safe_eval(u_str, x)
                r_val = self._safe_eval(r_str, x)
                if u_val is None or r_val is None:
                    continue
                if abs(u_val - r_val) > 1e-6:
                    return False
            except Exception:
                continue
        return True
```

## 8. 练习生成服务

### 8.1 生成管线

```python
class ExerciseGenerator:
    """根据场景参数生成一组练习题"""

    def __init__(
        self,
        question_repo: QuestionRepository,
        ai_engine: AIExerciseEngine,
        difficulty_scorer: DifficultyScorer,
    ):
        self.question_repo = question_repo
        self.ai_engine = ai_engine
        self.difficulty_scorer = difficulty_scorer

    async def generate(
        self,
        request: GenerateRequest
    ) -> list[Exercise]:
        """
        生成策略优先级：
        1. 从题库精确匹配（知识点 + 难度 + 题型）
        2. 从题库模糊匹配（知识点匹配，难度放宽）
        3. AI 生成（题库不够时）
        """
        exercises = []

        # 1. 精确匹配题库
        pool = await self.question_repo.search(
            knowledge_point_ids=request.knowledge_point_ids,
            types=request.exercise_types,
            difficulty_range=request.difficulty_range,
            subject=request.subject,
            grade=request.grade,
            exclude_ids=request.exclude_ids,
        )

        # 2. 按约束筛选
        selected = self._select_with_constraints(
            pool=pool,
            count=request.exercise_count,
            constraints=SelectionConstraints(
                max_from_same_kp=2,
                ensure_type_variety=True,
                required_kp_coverage=0.8,
            ),
            knowledge_point_ids=request.knowledge_point_ids,
        )

        exercises.extend(selected)

        # 3. 题库不足时 AI 补充
        remaining = request.exercise_count - len(exercises)
        if remaining > 0:
            ai_exercises = await self.ai_engine.generate_batch(
                knowledge_point_ids=request.knowledge_point_ids,
                types=request.exercise_types,
                count=remaining,
                difficulty_target=request.difficulty_range[0],
                subject=request.subject,
                grade=request.grade,
            )
            exercises.extend(ai_exercises)

        # 4. 填充 difficulty / bloom_level 元数据
        for ex in exercises:
            if ex.difficulty is None:
                ex.difficulty = await self.difficulty_scorer.score(ex)

        # 5. 排序（fixed/easy_first 模式）
        if request.adaptive_mode == "easy_first":
            exercises.sort(key=lambda e: e.difficulty or 0)
        elif request.adaptive_mode == "fixed":
            random.shuffle(exercises)
        # adaptive 模式不排序，由 AdaptiveEngine 动态选择

        return exercises[:request.exercise_count]

    def _select_with_constraints(
        self,
        pool: list[Exercise],
        count: int,
        constraints: SelectionConstraints,
        knowledge_point_ids: list[str],
    ) -> list[Exercise]:
        """按约束从题池中选择题目"""
        selected = []
        kp_count = defaultdict(int)  # 知识点出现次数

        for ex in pool:
            if len(selected) >= count:
                break

            # 检查同知识点限制
            kp_hit = False
            for kp_id in ex.knowledge_point_ids:
                if kp_count[kp_id] >= constraints.max_from_same_kp:
                    kp_hit = True
                    break
            if kp_hit:
                continue

            # 检查题型多样性（避免连续3题同类型）
            if (constraints.ensure_type_variety
                and len(selected) >= 2
                and all(selected[-i].type == ex.type for i in range(1, min(3, len(selected)+1)))):
                continue

            selected.append(ex)
            for kp_id in ex.knowledge_point_ids:
                kp_count[kp_id] += 1

        return selected
```

## 9. 错误处理与降级策略

### 9.1 错误码定义

| 错误码 | 说明 | 客户端处理 |
|--------|------|-----------|
| `EXERCISE_SESSION_NOT_FOUND` | 会话不存在 | 返回首页 |
| `EXERCISE_SESSION_EXPIRED` | 会话已过期 | 提示重新开始 |
| `EXERCISE_ALREADY_ANSWERED` | 题目已作答 | 刷新当前状态 |
| `EXERCISE_SUBMIT_CONFLICT` | 并发提交冲突 | 重试（指数退避） |
| `EXERCISE_POOL_EMPTY` | 无可用题目 | 降级为 AI 生成 |
| `GRADING_AI_UNAVAILABLE` | AI 批改不可用 | 降级为规则批改或延迟批改 |
| `GRADING_TIMEOUT` | 批改超时 | 显示"批改中"，异步通知结果 |
| `HINT_LIMIT_REACHED` | 提示次数用尽 | 禁用提示按钮 |
| `ADAPTIVE_ENGINE_ERROR` | 自适应引擎异常 | 降级为固定顺序 |

### 9.2 降级链路

```
AI批改不可用 ──► 规则关键词匹配批改 ──► 延迟批改（显示"批改中"）

自适应选题异常 ──► 降级为 easy_first 模式 ──► 降级为 fixed 随机模式

题库不足 ──► AI 生成补充 ──► 减少题目数量 ──► 提示"练习资源不足"

数学等价判定失败 ──► 精确字符串匹配 ──► 标记为"待人工复核"
```

### 9.3 离线练习支持

```python
class OfflineExerciseManager:
    """离线练习包管理"""

    async def preload_pack(
        self,
        user_id: int,
        subject: str,
        grade: str,
        knowledge_point_ids: list[str],
    ) -> OfflinePack:
        """预下载练习包到本地"""
        exercises = await self.generator.generate(GenerateRequest(
            subject=subject,
            grade=grade,
            knowledge_point_ids=knowledge_point_ids,
            exercise_count=20,  # 多生成一些供离线使用
            adaptive_mode="fixed",
        ))

        pack = OfflinePack(
            id=uuid4(),
            exercises=exercises,
            created_at=datetime.utcnow(),
            valid_until=datetime.utcnow() + timedelta(days=7),
        )

        # 客观题答案一并缓存（离线可批改）
        # 主观题标记为 needs_online_grading=True

        return pack

    async def sync_offline_attempts(
        self,
        user_id: int,
        attempts: list[OfflineAttempt],
    ) -> SyncResult:
        """离线作答结果同步到服务端"""
        results = []
        for attempt in attempts:
            if attempt.needs_online_grading:
                grade_result = await self.grading_router.grade(
                    attempt.exercise, attempt.user_answer, attempt.context
                )
                attempt.grade_result = grade_result
            results.append(attempt)
        return SyncResult(synced_count=len(results))
```

```python
async def _check_error_collection(
    self,
```

```python
async def _build_feedback(
    self,
    exercise: Exercise,
    attempt: ExerciseAttempt,
    session: ExerciseSession,
) -> FeedbackDto:
    # 查询当前答案管控层级
    tier_result = await self.answer_control.get_accessible_tier(
        user_id=session.user_id,
        exercise_id=exercise.id,
        context=AnswerControlContext(
            source_type=session.source_type,
            is_correct=attempt.is_correct,
            hint_count=attempt.hint_count,
            attempt_no=attempt.attempt_no,
        )
    )

    feedback = FeedbackDto(
        level=session.feedback_level,
        correct=attempt.is_correct,
        # 根据 tier 决定展示多少内容
        steps=exercise.explanation.steps if tier_result.tier >= 2 else None,
        hint=exercise.explanation.hint if tier_result.tier >= 1 else None,
        common_mistake=exercise.explanation.common_mistake if tier_result.tier >= 3 else None,
    )

    return feedback
```
