## 1. 模块概述

### 1.1 定位

异步任务与事件驱动架构是 PrimeTop 后端的"神经系统"，负责所有跨模块解耦通信、后台耗时处理、定时调度和分布式事务编排。本文档详细定义任务队列、事件总线、Saga 编排器的设计规范，以及各业务模块需要注册的完整任务与事件清单，使开发人员可以直接编码实现。

### 1.2 核心目标

1. **解耦**：模块间通过事件/任务通信，避免直接数据库交叉访问
2. **可靠**：任务至少执行一次（at-least-once），关键业务精确一次（exactly-once）
3. **可观测**：每个任务/事件可追踪、可回溯、可告警
4. **可扩展**：支持按模块独立扩缩 Worker，按优先级分队列
5. **可降级**：队列积压或第三方故障时自动降级，不阻塞主流程

### 1.3 术语定义

| 术语 | 含义 |
|------|------|
| Task | Celery 异步任务，有明确输入/输出，可重试 |
| Event | 领域事件，表示业务状态变更，触发下游处理 |
| Saga | 分布式事务编排模式，通过事件链协调多模块 |
| EventBus | 进程内事件分发器，用于单体架构内的模块解耦 |
| DLQ | Dead Letter Queue，处理失败且重试耗尽的消息 |
| Cron Job | 定时任务，通过 Celery Beat 调度 |

### 1.4 技术选型

| 组件 | 选型 | 说明 |
|------|------|------|
| 任务队列 | Celery 5.x + Redis 7.x (Broker) | 成熟稳定，支持优先级队列、任务链、结果后端 |
| 结果后端 | Redis（短期）/ MySQL（持久化） | 短期结果用 Redis TTL，需要持久追踪的用 DB |
| 定时调度 | Celery Beat + Redis（锁） | 支持动态 crontab，Beat 多实例用 Redis 锁选主 |
| 事件总线（进程内） | 自研 EventBus（asyncio） | MVP 阶段进程内分发，预留 Kafka 接口 |
| 事件总线（跨进程） | Redis Stream（Phase 1）→ Kafka（Phase 2） | Phase 1 轻量方案，Phase 2 按需升级 |
| 分布式锁 | Redis（Redlock） | 防止定时任务重复执行 |

### 2.2 数据流模式

```text
模式1: 同步请求 → 异步任务 → 结果回调
  用户请求 → API处理 → 投递Celery任务 → 轮询/SSE获取结果

模式2: 领域事件 → 多个监听器
  业务操作 → 发出Event → EventBus同步分发本进程 → 异步投递Celery

模式3: 定时触发
  Celery Beat → Cron任务 → 调用业务逻辑

模式4: Saga编排
  操作A → event_a → 操作B → event_b → 操作C → event_c
  任一步骤失败 → 补偿链回滚
```

## 4. 任务基类与开发规范

### 4.1 基础任务类

```python
# primetop/core/task_base.py
import logging
import time
from functools import wraps
from typing import Any, Callable
from celery import Task
from primetop.core.celery_app import celery_app
from primetop.core.exceptions import RetryableError, FatalError

logger = logging.getLogger(__name__)


class PrimeTopTask(Task):
    """所有 Celery 任务的基类，提供统一日志、指标、重试策略"""

    # 子类可覆盖
    max_retries = 3
    default_retry_delay = 60  # 秒
    soft_time_limit = 300
    time_limit = 600
    acks_late = True
    track_started = True

    def on_success(self, retval: Any, task_id: str, args: tuple, kwargs: dict):
        duration = time.time() - self._start_time if hasattr(self, "_start_time") else 0
        logger.info(
            "task_success",
            extra={
                "task_name": self.name,
                "task_id": task_id,
                "duration_ms": round(duration * 1000),
            },
        )
        # 上报指标
        self._report_metric("success", duration)

    def on_failure(self, exc: Exception, task_id: str, args: tuple, kwargs: tuple, einfo):
        logger.error(
            "task_failed",
            extra={
                "task_name": self.name,
                "task_id": task_id,
                "error": str(exc),
                "error_type": type(exc).__name__,
            },
        )
        self._report_metric("failure")

    def on_retry(self, exc: Exception, task_id: str, args: tuple, kwargs: dict, einfo):
        logger.warning(
            "task_retry",
            extra={
                "task_name": self.name,
                "task_id": task_id,
                "retry_count": self.request.retries,
                "error": str(exc),
            },
        )

    def __call__(self, *args, **kwargs):
        self._start_time = time.time()
        # 注入 trace_id 用于链路追踪
        from primetop.core.tracing import set_trace_id
        trace_id = kwargs.pop("_trace_id", None) or self.request.headers.get("trace_id")
        if trace_id:
            set_trace_id(trace_id)
        return super().__call__(*args, **kwargs)

    def _report_metric(self, status: str, duration: float = 0):
        """上报到监控系统（Prometheus / StatsD）"""
        try:
            from primetop.core.metrics import task_counter, task_histogram
            task_counter.labels(
                task_name=self.name, status=status
            ).inc()
            if duration > 0:
                task_histogram.labels(
                    task_name=self.name
                ).observe(duration)
        except Exception:
            pass  # 指标上报不应影响业务


def task(
    func: Callable = None,
    *,
    name: str = None,
    queue: str = None,
    max_retries: int = 3,
    retry_delay: int = 60,
    soft_time_limit: int = 300,
    time_limit: int = 600,
    **kwargs,
):
    """任务装饰器，简化任务定义"""

    def decorator(fn: Callable):
        task_name = name or f"primetop.modules.{fn.__module__.split('.')[-2]}.tasks.{fn.__name__}"

        @celery_app.task(
            name=task_name,
            base=PrimeTopTask,
            max_retries=max_retries,
            default_retry_delay=retry_delay,
            soft_time_limit=soft_time_limit,
            time_limit=time_limit,
            bind=True,
            **kwargs,
        )
        @wraps(fn)
        def wrapper(self, *args, **kw):
            try:
                return fn(self, *args, **kw)
            except RetryableError as e:
                # 可重试错误：自动重试
                raise self.retry(exc=e, countdown=retry_delay * (self.request.retries + 1))
            except FatalError as e:
                # 致命错误：不重试，直接失败
                logger.error(f"Fatal error in {task_name}: {e}", exc_info=True)
                raise
            except Exception as e:
                # 未知错误：重试
                if self.request.retries < max_retries:
                    raise self.retry(exc=e, countdown=retry_delay * (self.request.retries + 1))
                raise

        return wrapper

    if func is not None:
        return decorator(func)
    return decorator
```

```python
# primetop/core/schedule_registry.py
"""
定时任务注册表。
所有 cron 任务在此集中注册，支持动态调整。
"""
from celery.schedules import crontab
from typing import Dict, Any

class ScheduleRegistry:
    _schedules: Dict[str, Dict[str, Any]] = {}

    @classmethod
    def register(
        cls,
        name: str,
        task: str,
        cron_expr: str,
        args: tuple = (),
        kwargs: dict = None,
        enabled: bool = True,
        description: str = "",
    ):
        """注册定时任务
        
        Args:
            cron_expr: cron 表达式，如 "0 8 * * *" (每天8:00)
        """
        parts = cron_expr.split()
        schedule = crontab(
            minute=parts[0],
            hour=parts[1],
            day_of_month=parts[2],
            month_of_year=parts[3],
            day_of_week=parts[4],
        )
        cls._schedules[name] = {
            "task": task,
            "schedule": schedule,
            "args": args,
            "kwargs": kwargs or {},
            "options": {"queue": "cron", "priority": 5},
            "enabled": enabled,
            "description": description,
        }

    @classmethod
    def get_beat_schedule(cls) -> Dict:
        """返回 Celery Beat 需要的 schedule dict（过滤 disabled）"""
        return {
            name: config
            for name, config in cls._schedules.items()
            if config.get("enabled", True)
        }
```

```python
# primetop/core/beat_lock.py
"""Celery Beat 分布式锁，防止多实例重复执行"""
import redis
import os
from datetime import datetime

redis_client = redis.from_url(os.getenv("REDIS_URL", "redis://localhost:6379/0"))

class BeatLock:
    """使用 Redis 实现的 Beat 选主锁
    
    Beat 实例启动时竞争锁，获得锁的实例成为 active scheduler。
    锁自动续期，实例崩溃后锁自动释放。
    """
    LOCK_KEY = "primetop:beat:lock"
    LOCK_TTL = 30  # 秒，锁续期间隔

    def __init__(self):
        self._lock = None

    def acquire(self) -> bool:
        """尝试获取 Beat 锁"""
        instance_id = os.getenv("HOSTNAME", "beat-1")
        acquired = redis_client.set(
            self.LOCK_KEY, instance_id, nx=True, ex=self.LOCK_TTL
        )
        if acquired:
            self._start_renewal(instance_id)
            return True
        return False

    def _start_renewal(self, instance_id: str):
        """后台线程定期续期锁"""
        import threading

        def renew():
            while True:
                import time
                time.sleep(self.LOCK_TTL // 2)
                current = redis_client.get(self.LOCK_KEY)
                if current and current.decode() == instance_id:
                    redis_client.expire(self.LOCK_KEY, self.LOCK_TTL)

        t = threading.Thread(target=renew, daemon=True)
        t.start()
```

### 6.1 进程内事件总线

```python
# primetop/core/events.py
"""
进程内事件总线 (MVP阶段)

设计原则:
1. 同步事件：在当前事务内同步处理，用于模块内解耦
2. 异步事件：投递到 Celery，用于跨模块/耗时操作
3. 事件不可变：已发布的事件不可修改
4. 事件有序：同一 aggregate 的事件按序处理
"""
import logging
from typing import Any, Callable, Dict, List, Optional
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import json

logger = logging.getLogger(__name__)


class EventPriority(int, Enum):
    LOW = 1
    NORMAL = 5
    HIGH = 8
    CRITICAL = 10


@dataclass
class DomainEvent:
    """领域事件基类"""
    event_type: str
    aggregate_id: str          # 聚合根ID（如 user_id, conversation_id）
    aggregate_type: str        # 聚合根类型（如 "user", "conversation"）
    payload: Dict[str, Any]    # 事件数据
    timestamp: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    trace_id: Optional[str] = None
    event_id: Optional[str] = None

    def __post_init__(self):
        if self.event_id is None:
            import uuid
            self.event_id = str(uuid.uuid4())

    def to_dict(self) -> Dict:
        return {
            "event_id": self.event_id,
            "event_type": self.event_type,
            "aggregate_id": self.aggregate_id,
            "aggregate_type": self.aggregate_type,
            "payload": self.payload,
            "timestamp": self.timestamp,
            "trace_id": self.trace_id,
        }


class EventHandler:
    """事件处理器基类"""
    
    # 子类覆盖
    is_async: bool = False          # True = 投递到 Celery
    queue: str = "default"          # Celery 队列
    max_retries: int = 3
    priority: EventPriority = EventPriority.NORMAL

    def handle(self, event: DomainEvent):
        raise NotImplementedError


class EventBus:
    """进程内事件总线"""

    _instance: Optional["EventBus"] = None
    _handlers: Dict[str, List[EventHandler]] = {}
    _interceptors: List[Callable] = []  # 全局拦截器（安全、审计）

    @classmethod
    def initialize(cls):
        cls._instance = cls()
        cls._handlers = {}
        cls._interceptors = []

    @classmethod
    def get(cls) -> "EventBus":
        if cls._instance is None:
            cls.initialize()
        return cls._instance

    def subscribe(self, event_type: str, handler: EventHandler):
        """订阅事件"""
        if event_type not in self._handlers:
            self._handlers[event_type] = []
        self._handlers[event_type].append(handler)
        logger.info(f"EventBus: registered handler {handler.__class__.__name__} for {event_type}")

    def add_interceptor(self, interceptor: Callable):
        """添加全局拦截器（如审计日志、安全过滤）"""
        self._interceptors.append(interceptor)

    def emit(self, event_type: str, aggregate_id: str, aggregate_type: str,
             payload: Dict[str, Any], trace_id: str = None):
        """发布事件"""
        event = DomainEvent(
            event_type=event_type,
            aggregate_id=aggregate_id,
            aggregate_type=aggregate_type,
            payload=payload,
            trace_id=trace_id,
        )

        # 全局拦截器
        for interceptor in self._interceptors:
            try:
                interceptor(event)
            except Exception as e:
                logger.error(f"Event interceptor error: {e}")

        # 分发给订阅者
        handlers = self._handlers.get(event_type, [])
        for handler in handlers:
            try:
                if handler.is_async:
                    self._dispatch_async(handler, event)
                else:
                    handler.handle(event)
            except Exception as e:
                logger.error(
                    f"Event handler error: {handler.__class__.__name__} "
                    f"for {event_type}: {e}",
                    exc_info=True,
                )
                # 同步处理器失败不阻塞其他处理器
                continue

        # 持久化事件日志
        self._persist_event(event)

        return event

    def _dispatch_async(self, handler: EventHandler, event: DomainEvent):
        """异步投递到 Celery"""
        from primetop.core.celery_app import celery_app
        
        task_name = f"{handler.__class__.__module__}.{handler.__class__.__qualname__}"
        celery_app.send_task(
            task_name,
            args=[event.to_dict()],
            queue=handler.queue,
            priority=handler.priority,
            headers={"trace_id": event.trace_id},
        )

    def _persist_event(self, event: DomainEvent):
        """持久化事件到 event_store 表"""
        try:
            from primetop.core.database import get_session
            from primetop.models.event_store import EventStore
            
            session = get_session()
            record = EventStore(
                event_id=event.event_id,
                event_type=event.event_type,
                aggregate_id=event.aggregate_id,
                aggregate_type=event.aggregate_type,
                payload=event.payload,
                trace_id=event.trace_id,
                created_at=event.timestamp,
            )
            session.add(record)
            session.commit()
        except Exception as e:
            logger.error(f"Failed to persist event: {e}")
            # 事件持久化失败不阻塞主流程
```

### 7.1 Saga 编排器框架

```python
# primetop/core/saga.py
"""
Saga 编排器：管理跨模块的分布式事务。
适用于需要多步操作且要求最终一致性的场景。

模式：Orchestration（编排）- 由中央协调器控制流程。
"""
import logging
from typing import Any, Callable, Dict, List, Optional
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import json

logger = logging.getLogger(__name__)


class SagaStepStatus(str, Enum):
    PENDING = "pending"
    EXECUTING = "executing"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    COMPENSATED = "compensated"
    FAILED = "failed"


class SagaStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    COMPENSATED = "compensated"
    FAILED = "failed"


@dataclass
class SagaStep:
    """Saga 步骤定义"""
    name: str
    action: Callable                       # 正向操作（接收 saga_context，返回 result）
    compensate: Callable                   # 补偿操作（接收 saga_context + step_result）
    timeout: int = 300                     # 超时秒数
    retry_count: int = 2                   # 重试次数
    status: SagaStepStatus = SagaStepStatus.PENDING
    result: Optional[Dict] = None
    error: Optional[str] = None


@dataclass
class SagaDefinition:
    """Saga 定义"""
    saga_type: str                         # Saga 类型名
    steps: List[SagaStep] = field(default_factory=list)
    status: SagaStatus = SagaStatus.PENDING
    saga_id: Optional[str] = None
    context: Dict[str, Any] = field(default_factory=dict)
    current_step: int = 0
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())


class SagaOrchestrator:
    """Saga 编排器"""

    _saga_registry: Dict[str, "SagaDefinition"] = {}

    @classmethod
    def register(cls, saga_type: str, steps: List[SagaStep]):
        """注册 Saga 定义"""
        cls._saga_registry[saga_type] = SagaDefinition(
            saga_type=saga_type,
            steps=steps,
        )

    @classmethod
    async def execute(cls, saga_type: str, context: Dict[str, Any]) -> Dict:
        """执行 Saga"""
        import uuid

        definition = cls._saga_registry.get(saga_type)
        if not definition:
            raise ValueError(f"Unknown saga type: {saga_type}")

        saga = SagaDefinition(
            saga_type=saga_type,
            steps=[SagaStep(**s.__dict__) for s in definition.steps],
            saga_id=str(uuid.uuid4()),
            context=context,
        )

        # 持久化 Saga 状态
        cls._persist_saga(saga)

        logger.info(f"Saga {saga.saga_id} ({saga_type}) started")

        # 执行步骤
        for i, step in enumerate(saga.steps):
            saga.current_step = i
            saga.status = SagaStatus.RUNNING
            step.status = SagaStepStatus.EXECUTING
            cls._persist_saga(saga)

            try:
                step.result = await step.action(saga.context)
                step.status = SagaStepStatus.COMPLETED
                # 将步骤结果合并到 context
                saga.context[f"step_{i}_result"] = step.result
                cls._persist_saga(saga)
                logger.info(f"Saga {saga.saga_id} step {step.name} completed")

            except Exception as e:
                step.status = SagaStepStatus.FAILED
                step.error = str(e)
                logger.error(
                    f"Saga {saga.saga_id} step {step.name} failed: {e}"
                )
                # 开始补偿
                await cls._compensate(saga, i)
                raise

        saga.status = SagaStatus.COMPLETED
        cls._persist_saga(saga)
        logger.info(f"Saga {saga.saga_id} completed successfully")
        return saga.context

    @classmethod
    async def _compensate(cls, saga: SagaDefinition, failed_step: int):
        """执行补偿链（反向顺序）"""
        saga.status = SagaStatus.COMPENSATING
        cls._persist_saga(saga)

        for i in range(failed_step - 1, -1, -1):
            step = saga.steps[i]
            step.status = SagaStepStatus.COMPENSATING
            cls._persist_saga(saga)

            try:
                await step.compensate(saga.context, step.result)
                step.status = SagaStepStatus.COMPENSATED
                cls._persist_saga(saga)
                logger.info(f"Saga {saga.saga_id} compensated step {step.name}")

            except Exception as e:
                step.status = SagaStepStatus.FAILED
                step.error = str(e)
                cls._persist_saga(saga)
                logger.error(
                    f"Saga {saga.saga_id} compensate step {step.name} failed: {e}"
                )
                # 补偿失败需要人工介入
                cls._alert_compensation_failure(saga, i, str(e))

        saga.status = SagaStatus.COMPENSATED
        cls._persist_saga(saga)

    @classmethod
    def _persist_saga(cls, saga: SagaDefinition):
        """持久化 Saga 状态到数据库"""
        try:
            from primetop.core.database import get_session
            from primetop.models.saga_store import SagaInstance
            
            session = get_session()
            record = session.query(SagaInstance).filter_by(
                saga_id=saga.saga_id
            ).first()
            
            if record:
                record.status = saga.status
                record.current_step = saga.current_step
                record.context = saga.context
                record.steps_state = [
                    {"name": s.name, "status": s.status, "result": s.result, "error": s.error}
                    for s in saga.steps
                ]
                record.updated_at = datetime.utcnow()
            else:
                record = SagaInstance(
                    saga_id=saga.saga_id,
                    saga_type=saga.saga_type,
                    status=saga.status,
                    current_step=saga.current_step,
                    context=saga.context,
                    steps_state=[
                        {"name": s.name, "status": s.status, "result": s.result, "error": s.error}
                        for s in saga.steps
                    ],
                )
                session.add(record)
            session.commit()
        except Exception as e:
            logger.error(f"Failed to persist saga state: {e}")

    @classmethod
    def _alert_compensation_failure(cls, saga: SagaDefinition, step_index: int, error: str):
        """补偿失败告警"""
        logger.critical(
            f"SAGA COMPENSATION FAILURE: saga_id={saga.saga_id}, "
            f"saga_type={saga.saga_type}, failed_step={step_index}, error={error}"
        )
```

## 10. 错误处理与降级

### 10.1 任务失败分类

| 错误类型 | 示例 | 处理策略 |
|----------|------|----------|
| **可重试** | AI模型限流 (429)、网络超时、Redis 暂时不可用 | 指数退避重试，最多 N 次 |
| **业务失败** | 额度不足、订单已过期、用户不存在 | 记录日志，不重试，通知调用方 |
| **数据错误** | 参数缺失、JSON 解析失败 | 不重试，记录到 DLQ 供排查 |
| **系统错误** | 数据库连接失败、OOM | 重试 + 告警 |

### 10.2 Dead Letter Queue 处理

```python
# primetop/core/dlq.py
"""
Dead Letter Queue 处理器。
重试耗尽的任务进入 DLQ，需要人工或自动处理。
"""
import json
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

DLQ_REDIS_KEY = "primetop:dlq"

class DeadLetterHandler:
    
    @classmethod
    def push(cls, task_name: str, task_id: str, args: tuple, 
             kwargs: dict, error: str, retries: int):
        """将失败任务推入 DLQ"""
        import redis
        r = redis.from_url(os.getenv("REDIS_URL"))
        
        dlq_entry = {
            "task_name": task_name,
            "task_id": task_id,
            "args": args,
            "kwargs": kwargs,
            "error": error,
            "retries": retries,
            "failed_at": datetime.utcnow().isoformat(),
            "status": "pending_review",  # pending_review / requeued / dismissed
        }
        r.lpush(DLQ_REDIS_KEY, json.dumps(dlq_entry, default=str))
        
        # 发送告警
        if retries >= 3:
            cls._send_alert(dlq_entry)

    @classmethod
    def get_pending(cls, limit: int = 50) -> list:
        """获取待处理的 DLQ 条目"""
        import redis
        r = redis.from_url(os.getenv("REDIS_URL"))
        items = r.lrange(DLQ_REDIS_KEY, 0, limit - 1)
        return [json.loads(item) for item in items]

    @classmethod
    def requeue(cls, task_id: str):
        """重新入队指定任务"""
        from primetop.core.celery_app import celery_app
        
        entries = cls.get_pending(limit=1000)
        for entry in entries:
            if entry["task_id"] == task_id:
                celery_app.send_task(
                    entry["task_name"],
                    args=entry["args"],
                    kwargs=entry["kwargs"],
                )
                entry["status"] = "requeued"
                # 更新 DLQ 中的状态
                cls._update_entry(entry)
                return True
        return False

        """发送 DLQ 告警"""
        logger.warning(
            f"DLQ Alert: task={entry['task_name']} "
            f"id={entry['task_id']} retries={entry['retries']}"
        )

# 集成到 Celery 任务失败回调
@celery_app.on_after_configure.connect
def setup_failure_handler(sender, **kwargs):
    """注册全局任务失败处理器"""
    
    @sender.on_task_failure.connect
    def handle_task_failure(sender=None, **kwargs):
        task = kwargs.get("task")
        if task and task.request.retries >= task.max_retries:
            DeadLetterHandler.push(
```

### 10.3 降级策略

```python
# primetop/core/task_fallback.py
"""
任务降级策略。
当关键依赖不可用时，自动降级到备选方案。
"""

class TaskFallback:
    """任务降级策略注册"""

    _strategies: Dict[str, Callable] = {}

    @classmethod
    def register(cls, task_name: str, fallback: Callable):
        cls._strategies[task_name] = fallback

    @classmethod
    async def execute_or_fallback(cls, task_name: str, primary: Callable, 
                                   *args, **kwargs):
        """执行主逻辑，失败时降级"""
        try:
            return await primary(*args, **kwargs)
        except Exception as e:
            fallback = cls._strategies.get(task_name)
            if fallback:
                logger.warning(f"Task {task_name} degraded: {e}")
                return await fallback(*args, **kwargs)
            raise


# --- 具体降级策略 ---

# AI回答降级：从高级模型降级到基础模型

)

    "primetop.modules.question.tasks.ocr_recognize",
    fallback=lambda image_url, **kw: {"status": "failed", "message": "识别服务暂时不可用，请手动输入题目"},
)

# 周报生成降级：生成简化版报告
```

```python
# Phase 2: Kafka 事件总线（接口兼容层）
# primetop/core/events_kafka.py

class KafkaEventBus(EventBus):
    """Kafka 事件总线，接口兼容 Redis 版本"""

    def __init__(self):
        from aiokafka import AIOKafkaProducer
        self.producer = AIOKafkaProducer(
            bootstrap_servers=os.getenv("KAFKA_BOOTSTRAP_SERVERS"),
            value_serializer=lambda v: json.dumps(v).encode("utf-8"),
        )

    def emit(self, event_type: str, aggregate_id: str, aggregate_type: str,
             payload: dict, trace_id: str = None):
        event = DomainEvent(
            event_type=event_type,
            aggregate_id=aggregate_id,
            aggregate_type=aggregate_type,
            payload=payload,
            trace_id=trace_id,
        )
        
        # 同时写入 Kafka（异步）和 DB（同步）
        topic = f"primetop.{aggregate_type}.events"
        self.producer.send_and_wait(
            topic=topic,
            key=aggregate_id.encode("utf-8"),
            value=event.to_dict(),
            headers=[("trace_id", trace_id.encode("utf-8") if trace_id else b"")],
        )
        
        # 本地同步处理器
        handlers = self._handlers.get(event_type, [])
        for handler in handlers:
            if not handler.is_async:
                handler.handle(event)
```
