```dart
class StartupPerformanceMonitor {
  final String TAG = 'StartupPerformance';

  /// 各阶段时间戳
  final Map<String, int> timestamps = {};

  /// 记录时间戳
  void record(String phase) {
    timestamps[phase] = DateTime.now().millisecondsSinceEpoch;
    log('$TAG: $phase - ${timestamps[phase]}');
  }

  /// 获取启动总耗时
  int getTotalTime() {
    if (timestamps.containsKey('start') && timestamps.containsKey('ready')) {
      return timestamps['ready']! - timestamps['start']!;
    }
    return -1;
  }

  /// 上报启动性能数据
  void report() {
    final totalTime = getTotalTime();
    if (totalTime > 0) {
      AnalyticsService.trackEvent('app_startup', {
        'total_time_ms': totalTime,
        'phases': timestamps,
        'device_info': DeviceInfo.getBasicInfo(),
      });
    }
  }
}
```

```dart
  @override
  void initState() {
    super.initState();
    _scrollController.addListener(_onScroll);

    _scrollController.removeListener(_onScroll);
    _scrollController.dispose();
    super.dispose();

    if (_scrollController.position.pixels >=
        _scrollController.position.maxScrollExtent * 0.7) {
      _preloadNextPage();
    }
  }

  /// 预加载下一页
  void _preloadNextPage() {
    final nextIndex = _preloadIndex + 20;
    if (nextIndex < widget.items.length) {
      _preloadIndex = nextIndex;
      // 触发数据预加载
      DataService.preloadData(nextIndex, 20);
    }
  }

  @override
  Widget build(BuildContext context) {
    return ListView.builder(
      controller: _scrollController,
      itemCount: widget.items.length,
      itemBuilder: (context, index) {
        return RepaintBoundary(
          child: widget.itemBuilder(widget.items[index]),
        );
      },
    );
  }
```

```dart
/// 多级缓存管理器
class CacheManager {
  /// L1缓存：内存缓存
  final Map<String, dynamic> _memoryCache = {};

  /// L2缓存：本地存储缓存
  final SharedPreferences _prefs;

  /// L3缓存：图片缓存
  final BaseCacheManager _imageCache;

  CacheManager({
    required SharedPreferences prefs,
    BaseCacheManager? imageCache,
  })  : _prefs = prefs,
        _imageCache = imageCache ?? DefaultCacheManager();

  /// 获取缓存
  Future<T?> get<T>(String key) async {
    // 先从L1缓存获取
    if (_memoryCache.containsKey(key)) {
      return _memoryCache[key] as T;
    }

    // 再从L2缓存获取
    final value = _prefs.get(key);
    if (value != null) {
      // 回填到L1缓存
      _memoryCache[key] = value;
      return value as T;
    }

    return null;
  }

  /// 设置缓存
  Future<void> set(String key, dynamic value) async {
    // 设置L1缓存
    _memoryCache[key] = value;

    // 设置L2缓存
    await _prefs.setString(key, jsonEncode(value));
  }

  /// 移除缓存
  Future<void> remove(String key) async {
    // 移除L1缓存
    _memoryCache.remove(key);

    // 移除L2缓存
    await _prefs.remove(key);
  }

  /// 清空缓存
  Future<void> clearAll() async {
    // 清空L1缓存
    _memoryCache.clear();

    // 清空L2缓存
    await _prefs.clear();

    // 清空L3缓存
    await _imageCache.emptyCache();
  }
}
```

```go
// API中间件 - 请求性能监控
func PerformanceMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        startTime := time.Now()

        // 记录请求开始
        requestID := getRequestID(r)
        logger.Info("Request started",
            "request_id", requestID,
            "method", r.Method,
            "path", r.URL.Path,
        )

        // 创建响应包装器
        wrapped := &responseWrapper{
            ResponseWriter: w,
            statusCode:     http.StatusOK,
        }

        // 调用下一个处理器
        next.ServeHTTP(wrapped, r)

        // 记录请求结束
        duration := time.Since(startTime)
        logger.Info("Request completed",
            "request_id", requestID,
            "method", r.Method,
            "path", r.URL.Path,
            "status", wrapped.statusCode,
            "duration_ms", duration.Milliseconds(),
        )

        // 性能告警
        if duration.Milliseconds() > 500 {
            logger.Warn("Slow request detected",
                "request_id", requestID,
                "duration_ms", duration.Milliseconds(),
            )
        }
    })
}

// 响应包装器
type responseWrapper struct {
    http.ResponseWriter
    statusCode int
}

func (w *responseWrapper) WriteHeader(statusCode int) {
    w.statusCode = statusCode
    w.ResponseWriter.WriteHeader(statusCode)
}
```

```go
// 批量查询题目（使用IN查询和缓存）
func (s *QuestionService) GetQuestionsByIDs(ids []string) ([]*Question, error) {
    // 1. 先从缓存获取
    var missingIDs []string
    questions := make([]*Question, 0, len(ids))

    for _, id := range ids {
        cacheKey := fmt.Sprintf("question:%s", id)
        cached, err := s.cache.Get(context.Background(), cacheKey).Result()
        if err == nil {
            var question Question
            if err := json.Unmarshal([]byte(cached), &question); err == nil {
                questions = append(questions, &question)
                continue
            }
        }
        missingIDs = append(missingIDs, id)
    }

    // 2. 批量查询缺失的题目
    if len(missingIDs) > 0 {
        query, args, err := sqlx.In(
            "SELECT id, content, options, answer, analysis FROM questions WHERE id IN (?)",
            missingIDs,
        )
        if err != nil {
            return nil, err
        }

        var dbQuestions []Question
        if err := s.db.Select(&dbQuestions, query, args...); err != nil {
            return nil, err
        }

        // 3. 将查询结果写入缓存
        pipe := s.cache.Pipeline()
        for _, q := range dbQuestions {
            cacheKey := fmt.Sprintf("question:%s", q.ID)
            data, _ := json.Marshal(q)
            pipe.Set(context.Background(), cacheKey, data, time.Hour*24)
        }
        if _, err := pipe.Exec(context.Background()); err != nil {
            logger.Error("Failed to cache questions", "error", err)
        }

        questions = append(questions, &dbQuestions...)
    }

    return questions, nil
}
```

```go
// 限流中间件
func RateLimitMiddleware(next http.Handler) http.Handler {
    // 使用令牌桶算法
    limiter := rate.NewLimiter(rate.Limit(1000), 100) // 1000 req/s, burst 100

    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // 从请求头获取用户ID
        userID := r.Header.Get("X-User-ID")

        // 使用IP地址作为key
        ip := r.RemoteAddr
        key := ip
        if userID != "" {
            key = userID
        }

        // 检查是否超过限制
        if !limiter.Allow() {
            http.Error(w, "Rate limit exceeded", http.StatusTooManyRequests)
            return
        }

        next.ServeHTTP(w, r)
    })
}
```

```go
func (s *LearningService) GetLearningReport(userID string, startDate, endDate time.Time) (*LearningReport, error) {
    query := `
        SELECT
            lr.subject,
            COUNT(*) as total_records,
            SUM(lr.learning_duration) as total_duration,
            AVG(lr.score) as avg_score,
            COUNT(DISTINCT lr.knowledge_point_id) as unique_knowledge_points,

                SELECT COUNT(*)
                FROM mistake_records mr
                WHERE mr.user_id = lr.user_id
                AND mr.subject = lr.subject
                AND mr.created_at BETWEEN ? AND ?
            ) as mistake_count
        FROM learning_records lr
        WHERE lr.user_id = ?
        AND lr.learning_date BETWEEN ? AND ?
        GROUP BY lr.subject

        StartDate:   startDate,
        EndDate:     endDate,
        Subjects:    subjects,
        TotalRecords: sumSubjectRecords(subjects),
```

```go
// AI服务 - 流式响应处理
func (s *AIService) StreamAnswer(ctx context.Context, req *AnswerRequest) (<-chan string, error) {
    outputChan := make(chan string, 100)

    go func() {
        defer close(outputChan)

        // 1. 构建Prompt
        prompt, err := s.buildPrompt(req)
        if err != nil {
            logger.Error("Failed to build prompt", "error", err)
            return
        }

        // 2. 调用大模型API（流式）
        stream, err := s.llmClient.CreateCompletionStream(ctx, prompt)
        if err != nil {
            logger.Error("Failed to create completion stream", "error", err)
            return
        }

        // 3. 处理流式响应
        var buffer strings.Builder
        var lastFlushTime time.Time

        for chunk := range stream {
            // 处理内容
            content := chunk.Choices[0].Delta.Content

            // 安全过滤
            if s.isContentSafe(content) {
                buffer.WriteString(content)

                // 控制输出频率
                now := time.Now()
                if now.Sub(lastFlushTime) > 200*time.Millisecond || buffer.Len() > 100 {
                    outputChan <- buffer.String()
                    buffer.Reset()
                    lastFlushTime = now
                }
            }
        }

        // 刷新剩余内容
        if buffer.Len() > 0 {
            outputChan <- buffer.String()
        }

        // 4. 记录使用情况
        s.recordUsage(req.UserID, req.SessionID)
    }()

    return outputChan, nil
}
```

```go
    // 1. 获取可用模型列表
    availableModels := s.getAvailableModels()

    // 2. 根据请求类型选择模型
    candidates := s.filterModelsByType(availableModels, req.Type)

    // 3. 根据性能指标排序
    ranked := s.rankModelsByPerformance(candidates)

    // 4. 负载均衡
    selected := s.balanceLoad(ranked)

    return selected, nil
}

// 模型性能排名
func (s *ModelScheduler) rankModelsByPerformance(models []*ModelConfig) []*ModelConfig {
    sort.Slice(models, func(i, j int) bool {
        scoreI := s.calculateScore(models[i])
        scoreJ := s.calculateScore(models[j])
        return scoreI > scoreJ
    })
    return models
}

// 计算模型综合得分
func (s *ModelScheduler) calculateScore(model *ModelConfig) float64 {
    metrics := s.metrics.GetMetrics(model.Name)

    // 综合考虑响应时间、成功率、成本等指标
    responseTimeScore := 1.0 / (metrics.AvgResponseTime / 1000.0)
    successRateScore := metrics.SuccessRate
    costScore := 1.0 / (metrics.CostPerRequest + 0.01)

    totalScore := responseTimeScore*0.4 + successRateScore*0.4 + costScore*0.2
    return totalScore
}

// 负载均衡
func (s *ModelScheduler) balanceLoad(models []*ModelConfig) *ModelConfig {
    // 选择当前负载最低的模型
    var selected *ModelConfig
    minLoad := float64(math.MaxFloat64)

    for _, model := range models {
        metrics := s.metrics.GetMetrics(model.Name)
        currentLoad := float64(metrics.ActiveRequests) / float64(model.MaxConcurrency)

        if currentLoad < minLoad {
            minLoad = currentLoad
            selected = model
        }
    }

    return selected
}
```

```dart
/// 网络请求优化工具
class NetworkOptimizer {
  /// 请求队列
  final Queue<NetworkRequest> _requestQueue = Queue();

  /// 请求去重
  final Map<String, Set<Completer>> _pendingRequests = {};

  /// 批量请求管理器
  final BatchRequestManager _batchManager = BatchRequestManager();

  /// 发送请求（带去重）
  Future<T> sendRequest<T>(NetworkRequest request) async {
    // 生成请求key
    final requestKey = _generateRequestKey(request);

    // 检查是否有相同的请求正在进行
    if (_pendingRequests.containsKey(requestKey)) {
      final completer = Completer<T>();
      _pendingRequests[requestKey]!.add(completer);
      return completer.future;
    }

    // 创建新的请求
    final completer = Completer<T>();
    _pendingRequests[requestKey] = {completer};

    try {
      final response = await _executeRequest(request);
      completer.complete(response as T);
      return response;
    } catch (e) {
      completer.completeError(e);
      rethrow;
    } finally {
      // 移除待处理请求
      _pendingRequests.remove(requestKey);
    }
  }

  /// 生成请求key
  String _generateRequestKey(NetworkRequest request) {
    return '${request.method}:${request.path}:${request.parameters.toString()}';
  }

  /// 执行请求
  Future<dynamic> _executeRequest(NetworkRequest request) async {
    // 添加重试逻辑
    int retryCount = 0;
    const maxRetries = 3;

    while (retryCount < maxRetries) {
      try {
        return await _doRequest(request);
      } catch (e) {
        retryCount++;
        if (retryCount >= maxRetries) {
          rethrow;
        }

        // 指数退避
        final delay = Duration(milliseconds: 100 * (1 << (retryCount - 1)));
        await Future.delayed(delay);
      }
    }

    throw Exception('Request failed after $maxRetries retries');
  }

  /// 实际发送请求
  Future<dynamic> _doRequest(NetworkRequest request) async {
    final response = await HttpClient().sendRequest(
      request.method,
      request.path,
      parameters: request.parameters,
      headers: request.headers,
    );

    if (response.statusCode >= 200 && response.statusCode < 300) {
      return response.data;
    } else {
      throw Exception('Request failed: ${response.statusCode}');
    }
  }
}
```

```dart
    if (!metrics.containsKey(name)) {

    }

    // 上报到服务端
    _uploadMetric(name, durationMs);
  }

    return PerformanceReport(

      generatedAt: DateTime.now(),
    );
  }

      'metric_name': name,
      'duration_ms': durationMs,
      'device_info': DeviceInfo.getBasicInfo(),
    });
  }
}

/// 性能指标
class PerformanceMetric {
  final String name;
  final List<int> samples = [];

  PerformanceMetric({required this.name});

  void record(int durationMs) {
    samples.add(durationMs);

    // 只保留最近1000个样本
    if (samples.length > 1000) {
      samples.removeAt(0);
    }
  }

  /// 获取平均值
  double get average {
    if (samples.isEmpty) return 0;
    return samples.reduce((a, b) => a + b) / samples.length;
  }

  /// 获取P95
  int getP95() {
    if (samples.isEmpty) return 0;
    final sorted = List.from(samples)..sort();
    final index = (sorted.length * 0.95).floor();
    return sorted[index];
  }

  /// 获取P99
  int getP99() {
    if (samples.isEmpty) return 0;
    final sorted = List.from(samples)..sort();
    final index = (sorted.length * 0.99).floor();
    return sorted[index];
  }
}
```
