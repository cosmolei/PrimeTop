## 1. 模块概述

### 1.1 功能定位

本文档描述 PrimeTop 的核心价值链——**端到端学习闭环**。主设计文档多次强调"从知识讲解、练习答疑、错题整理、复习规划到学情反馈形成闭环"，但各模块的详细设计文档独立成篇，缺少一份贯穿所有模块的全景图。

本文档解决以下问题：

1. **用户旅程全景**：一个学生从打开 APP 到完成一轮完整学习循环，经过哪些页面、触发哪些服务、产生哪些数据。
2. **模块协作关系**：AI辅导、拍题答疑、同步课堂、错题整理、学情分析、学习规划等模块如何通过 API 和事件协同工作。
3. **数据流转路径**：学习行为数据从产生到分析到驱动个性化推荐的完整链路。
4. **状态机定义**：核心业务实体（学习任务、错题、学习计划等）的状态流转。
5. **端到端异常处理**：跨模块调用的失败策略和降级方案。

### 1.2 适用读者

- 服务端开发工程师：理解自己负责的模块在全局链路中的位置和上下游依赖。
- 移动端开发工程师：理解完整的页面跳转和数据驱动逻辑。
- AI 工程师：理解 AI 输出如何被下游模块消费。
- 产品经理：验证闭环体验是否顺畅。
- 测试工程师：设计端到端集成测试用例。

#### 错题数据结构

```typescript
interface MistakeRecord {
  id: string;                          // 主键
  userId: string;                      // 学生ID
```

```typescript
interface KnowledgePointMastery {
  userId: string;
  knowledgePointId: string;
  
  // 掌握度评分（0-100）
  masteryScore: number;
  
  // 计算因子
  totalQuestions: number;          // 该知识点总做题数
  correctCount: number;            // 正确数
  mistakeCount: number;            // 错题数（当前未掌握的）
  consecutiveCorrect: number;      // 连续正确
  avgTimeSpentSeconds: number;     // 平均用时
  lastPracticeAt: Date;            // 最近练习时间
  
  // 遗忘曲线衰减
  lastReinforcedAt: Date;          // 最近一次正确作答时间
  reinforcementCount: number;      // 巩固次数
  
  // 掌握等级
  level: 'unknown'    // 未接触（初始状态）
       | 'exposed'    // 已接触但未练习
       | 'learning'   // 学习中（正确率 < 60%）
       | 'familiar'   // 熟悉（正确率 60%-80%）
       | 'proficient' // 熟练（正确率 80%-95%，连续正确≥2）
       | 'mastered';  // 掌握（正确率 > 95%，连续正确≥3，间隔>3天）
  
  updatedAt: Date;
}

// 掌握度计算公式
function calculateMasteryScore(m: KnowledgePointMastery): number {
  if (m.totalQuestions === 0) return 0;
  
  const accuracy = m.correctCount / m.totalQuestions;
  
  // 基础分：正确率权重
  let score = accuracy * 60;
  
  // 连续正确加分
  score += Math.min(m.consecutiveCorrect * 5, 20);
  
  // 遗忘衰减（距上次练习越久分越低）
  const daysSinceLastPractice = (Date.now() - m.lastPracticeAt.getTime()) / 86400000;
  const decayFactor = Math.max(0, 1 - daysSinceLastPractice / 30); // 30天半衰期
  score *= (0.7 + 0.3 * decayFactor);
  
  // 巩固次数加分
  score += Math.min(m.reinforcementCount * 2, 20);
  
  return Math.round(Math.min(100, Math.max(0, score)));
}
```

### 4.1 拍题答疑端到端编排

这是最复杂的服务编排场景，涉及5个服务的串行/并行调用。

```typescript
// 拍题答疑服务编排器
class PhotoSolveOrchestrator {
  
  async execute(request: PhotoSolveRequest): Promise<PhotoSolveResponse> {
    const timer = new Stopwatch();
    const { userId, imageBase64, selectedQuestionIndex } = request;
    
    // ── Phase 1: OCR识别（≤ 5s） ──
    timer.start('ocr');
    const ocrResult = await this.ocrService.recognize({
      image: imageBase64,
      multiQuestion: true,
      timeout: 4500  // 预留500ms余量
    });
    timer.stop('ocr');
    
    if (ocrResult.questions.length === 0) {
      return { success: false, error: 'OCR_RECOGNITION_FAILED', 
               message: '未识别到题目，请重新拍摄' };
    }
    
    const question = ocrResult.questions[selectedQuestionIndex || 0];
    
    // ── Phase 2: 并行处理 ──
    // 2a: 获取学生画像（缓存优先）
    // 2b: 题库匹配（尝试匹配已有题目）
    // 2c: 知识点识别
    timer.start('parallel');
    const [studentProfile, questionMatch, knowledgePoints] = await Promise.all([
      this.userService.getStudentProfile(userId),
      this.questionService.findSimilarQuestion(question.text, { threshold: 0.95 }),
      this.knowledgeService.identifyKnowledgePoints(question.text, {
        subject: request.subjectId
      })
    ]);
    timer.stop('parallel');
    
    // ── Phase 3: RAG检索 + Prompt编排 ──
    timer.start('rag');
    const ragContext = await this.ragService.retrieve({
      query: question.text,
      knowledgePointIds: knowledgePoints.map(kp => kp.id),
      textbookVersion: studentProfile.textbookVersions[request.subjectId],
      grade: studentProfile.grade,
      topK: 5,
      minScore: 0.7
    });
    timer.stop('rag');
    
    // 编排Prompt
    const prompt = this.promptEngine.assemble('photo_solve_analysis', {
      studentProfile,
      question,
      knowledgePoints,
      ragContext: ragContext.documents,
      analysisMode: 'step_by_step',
      grade: studentProfile.grade
    });
    
    // ── Phase 4: 大模型调用（流式） ──
    timer.start('llm');
    const llmResponse = await this.llmService.chat({
      model: this.selectModel(question, knowledgePoints),
      messages: prompt.messages,
      stream: true,
      temperature: 0.3,  // 解题场景低温度
      maxTokens: 2000
    });
    timer.stop('llm');
    
    // ── Phase 5: 安全审核 ──
    const safetyResult = await this.safetyService.check(llmResponse.content, {
      mode: 'student_content',
      stage: studentProfile.stage
    });
    
    if (!safetyResult.safe) {
      // 记录并返回安全提示
      await this.loggingService.warn('AI_OUTPUT_UNSAFE', { 
        userId, content: llmResponse.content, flags: safetyResult.flags 
      });
      return { success: false, error: 'CONTENT_FILTERED',
               message: '内容审核未通过，请换个问题' };
    }
    
    // ── Phase 6: 数据沉淀（异步，不阻塞响应） ──
    const analysisId = generateId();
    
    // 使用 Promise.allSettled 确保不因后台任务失败影响用户
    setImmediate(async () => {
      try {
        await Promise.allSettled([
          // 6a: 保存分析记录
          this.analysisRepo.save({
            id: analysisId,
            userId, question, knowledgePoints, 
            response: llmResponse.content,
            tokenUsage: llmResponse.usage,
            durationMs: timer.totalMs
          }),
          
          // 6b: 发送领域事件
          this.eventBus.publish('answer.recorded', {
            userId,
            answerRecordId: analysisId,
            questionId: questionMatch?.questionId,
            knowledgePointIds: knowledgePoints.map(kp => kp.id),
            isCorrect: null,  // 拍题场景尚不确定对错
            timeSpentSeconds: Math.round(timer.totalMs / 1000),
            source: 'photo_solve',
            subjectId: request.subjectId,
            chapterId: knowledgePoints[0]?.chapterId
          }),
          
          // 6c: 如果匹配到题库题目，更新统计
          questionMatch ? this.questionService.incrementStats(
            questionMatch.questionId, 'view_count'
          ) : this.questionService.saveNewQuestion({
            sourceType: 'photo_ocr',
            content: question.text,
            options: question.options,
            identifiedKnowledgePoints: knowledgePoints
          })
        ]);
      } catch (err) {
        this.loggingService.error('PhotoSolve background task failed', err);
      }
    });
    
    return {
      success: true,
      analysisId,
      question: { text: question.text, options: question.options },
      steps: this.parseResponseToSteps(llmResponse.content),
      knowledgePoints: knowledgePoints.map(kp => ({ id: kp.id, name: kp.name })),
      similarQuestions: questionMatch?.similarQuestionIds || [],
      quickActions: ['加入错题本', '练同类题', '换一种讲法', '继续追问'],
      timing: timer.getMetrics()
    };
  }
  
  private selectModel(question, knowledgePoints): string {
    // 理科复杂题用推理增强模型
    if (knowledgePoints.some(kp => 
      ['math', 'physics', 'chemistry'].includes(kp.subject) 
      && kp.difficulty === 'hard'
    )) {
      return 'reasoning_model';
    }
    // 默认用通用模型
    return 'general_model';
  }
}
```

### 4.2 学习计划动态调整编排

```typescript
class StudyPlanAdjustmentOrchestrator {
  
  /**
   * 学习计划的动态调整触发条件：
   * 1. 每日定时检查（cron: 06:00）
   * 2. 学习任务完成后实时检查
   * 3. 错题复习结果变更后检查
   * 4. 用户主动请求调整
   */
  
  async adjustIfNeeded(userId: string, trigger: AdjustmentTrigger): Promise<void> {
    const profile = await this.userService.getStudentProfile(userId);
    const currentPlan = await this.planService.getActivePlan(userId);
    
    if (!currentPlan) return;
    
    // 收集调整依据
    const [weakPoints, overdueReviews, completionRate, upcomingExams] = 
      await Promise.all([
        this.analyticsService.getWeakPoints(userId, { topN: 5, minUrgency: 'medium' }),
        this.mistakeService.getOverdueReviewCount(userId),
        this.planService.getWeeklyCompletionRate(userId),
        this.calendarService.getUpcomingExams(userId, { daysAhead: 30 })
      ]);
    
    // 调整策略判断
    const adjustments: PlanAdjustment[] = [];
    
    // 策略1: 新增薄弱点练习任务
    for (const wp of weakPoints) {
      if (!this.isCoveredInPlan(currentPlan, wp)) {
        adjustments.push({
          type: 'add_task',
          taskType: 'weak_point_practice',
          title: `薄弱点突破：${wp.knowledgePointName}`,
          knowledgePointId: wp.knowledgePointId,
          reason: `最近${wp.recentErrorCount}次错误，掌握度${wp.masteryScore}%`,
          priority: 'high'
        });
      }
    }
    
    // 策略2: 插入错题复习任务（如果逾期超过5题）
    if (overdueReviews > 5) {
      adjustments.push({
        type: 'add_task',
        taskType: 'mistake_review',
        title: `复习 ${Math.min(overdueReviews, 10)} 道逾期错题`,
        reason: `${overdueReviews}道错题已超过复习时间`,
        priority: 'high'
      });
    }
    
    // 策略3: 考前冲刺模式
    if (upcomingExams.length > 0) {
      const nearestExam = upcomingExams[0];
      const daysUntil = daysBetween(new Date(), nearestExam.date);
      
      if (daysUntil <= 14 && daysUntil > 0) {
        adjustments.push({
          type: 'mode_switch',
          newMode: 'exam_sprint',
          examName: nearestExam.name,
          daysRemaining: daysUntil,
          reason: `${nearestExam.name}还有${daysUntil}天，进入冲刺模式`
        });
      }
    }
    
    // 策略4: 计划完成率过低时简化任务
    if (completionRate < 0.4 && currentPlan.dailyTaskCount > 4) {
      adjustments.push({
        type: 'reduce_load',
        targetTaskCount: Math.max(3, currentPlan.dailyTaskCount - 2),
        reason: `本周完成率${Math.round(completionRate * 100)}%，适当减少任务量`
      });
    }
    
    // 策略5: 完成率高时增加挑战
    if (completionRate > 0.9 && currentPlan.dailyTaskCount < 6) {
      adjustments.push({
        type: 'increase_load',
        targetTaskCount: currentPlan.dailyTaskCount + 1,
        reason: '本周完成率很高，适当增加挑战'
      });
    }
    
    // 应用调整
    if (adjustments.length > 0) {
      await this.planService.applyAdjustments(currentPlan.id, adjustments);
      
      // 通知用户
      await this.notificationService.send({
        userId,
        type: 'plan_adjusted',
        title: '学习计划已更新',
        body: this.summarizeAdjustments(adjustments),
        priority: adjustments.some(a => a.priority === 'high') ? 'high' : 'normal'
      });
    }
  }
}
```

```typescript
// 通用重试执行器
async function withRetry<T>(
  fn: () => Promise<T>,
  policy: RetryPolicy,
  context: string
): Promise<T> {
  let lastError: Error;
  
  for (let attempt = 1; attempt <= policy.maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error;
      
      if (!policy.retryableErrors.includes(error.code)) {
        throw error; // 不可重试错误直接抛出
      }
      
      const delay = calculateDelay(policy, attempt);
      logger.warn(`${context} attempt ${attempt}/${policy.maxAttempts} failed, retrying in ${delay}ms`, {
        error: error.message,
        code: error.code
      });
      
      await sleep(delay);
    }
  }
  
  // 所有重试失败，执行降级
  logger.error(`${context} all ${policy.maxAttempts} attempts failed`, {
    lastError: lastError.message
  });
  throw new RetryExhaustedError(context, lastError);
}
```

### 8.2 补偿机制

```typescript
// 补偿任务：处理事件丢失或处理失败的场景
class CompensationScheduler {
  
  // 每小时执行一次
  @Cron('0 * * * *')
  async checkAndCompensate() {
    
    // 检查1: 答题记录与学情分析的一致性
    const unmatchedAnswers = await this.db.query(`
      SELECT ar.* FROM answer_records ar
      LEFT JOIN mastery_updates mu ON ar.id = mu.source_record_id
      WHERE ar.created_at > NOW() - INTERVAL 2 HOUR
        AND mu.id IS NULL
        AND ar.compensation_attempts < 3
    `);
    
    for (const answer of unmatchedAnswers) {
      await this.eventBus.publish('answer.recorded', {
        userId: answer.userId,
        answerRecordId: answer.id,
        knowledgePointIds: answer.knowledgePointIds,
        isCorrect: answer.isCorrect,
        // ... 完整事件数据
      });
      await this.db.execute(
        'UPDATE answer_records SET compensation_attempts = compensation_attempts + 1 WHERE id = ?',
        [answer.id]
      );
    }
    
    // 检查2: 错题分析超时
    const pendingAnalysis = await this.db.query(`
      SELECT * FROM mistake_records
      WHERE status = 'raw' 
        AND created_at < NOW() - INTERVAL 10 MINUTE
        AND analysis_attempts < 3
    `);
    
    for (const mistake of pendingAnalysis) {
      await this.mistakeService.triggerAnalysis(mistake.id);
    }
    
    // 检查3: 学习计划进度一致性
    // ...验证已完成的学习任务是否都反映在计划进度中
  }
}
```
