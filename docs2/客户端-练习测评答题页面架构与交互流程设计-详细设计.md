/// 练习会话
class PracticeSession {
  final String sessionId;
  final String title;
  final PracticeMode mode;        // 练习模式
  final PracticeSource source;    // 来源
  final List<PracticeQuestion> questions;
  final int totalCount;
  final Duration? timeLimit;      // null = 不限时
  final Duration? elapsedTime;    // 已用时间
  final PracticeConfig config;
  final SessionStatus status;
  final DateTime createdAt;
  final DateTime? completedAt;

  /// 当前题目索引
  final int currentIndex;

  /// 标记的题目索引集合
  final Set<int> markedIndices;

  /// 已答题的答案映射: questionIndex -> answer
  final Map<int, UserAnswer> answers;
}

/// 练习模式枚举
enum PracticeMode {
  practice,       // 普通练习
  quiz,           // 随堂测验
  exam,           // 考试模拟（严格限时）
  adaptive,       // 自适应练习
  mistakeReview,  // 错题重练
  dailyTask,      // 每日任务
}

/// 练习来源
enum PracticeSource {
  chapter,        // 同步课堂章节
  knowledgePoint, // 知识点专项
  mistake,        // 错题本
  aiRecommend,    // AI推荐
  examSimulation, // 考试模拟
  customGroup,    // 自定义组卷
  dailyTask,      // 每日任务
  campTask,       // 学习营任务
}

/// 会话状态
enum SessionStatus {
  notStarted,     // 未开始
  inProgress,     // 进行中
  paused,         // 已暂停（非限时模式）
  submitted,      // 已提交
  autoSubmitted,  // 自动交卷（超时）
  abandoned,      // 已放弃
}

/// 练习配置
class PracticeConfig {
  final bool showAnswerImmediately;   // 答完即显答案
  final bool allowGoBack;             // 允许返回上一题
  final bool allowSkip;               // 允许跳过
  final bool shuffleOrder;            // 随机题目顺序
  final bool showProgress;            // 显示进度
  final bool showTimer;               // 显示计时器
  final FeedbackLevel feedbackLevel;  // 反馈详细程度
  final AnswerRevealStrategy answerReveal; // 答案揭示策略
  final bool enableMarking;           // 允许标记题目
  final int maxAttempts;              // 单题最大尝试次数（0=无限）
}

/// 反馈级别
enum FeedbackLevel {
  none,           // 无反馈（考试模式）
  resultOnly,     // 仅对错
  withExplanation,// 对错+解析
  detailed,       // 详细反馈+知识点
  guided,         // 启发式引导反馈
}

/// 答案揭示策略（对接答案管控引擎）
enum AnswerRevealStrategy {
  immediate,      // 答题后立即揭示
  afterSubmit,    // 提交后揭示
  tiered,         // 分级渐进揭示（TIER 体系）
  onDemand,       // 按需点击揭示
}

/// 练习中的题目（扩展自通用题目模型）
class PracticeQuestion {
  final String questionId;
  final QuestionType type;
  final DifficultyLevel difficulty;
  final String stem;                 // 题干（富文本 Markdown）
  final List<String>? materials;     // 材料段落
  final List<Attachment>? attachments; // 附件（图片/音频/视频）
  final dynamic answerStructure;     // 题型专属答题结构
  final List<KnowledgeTag> knowledgeTags;
  final double? score;               // 分值
  final int? estimatedSeconds;       // 预估答题秒数

  /// 运行时状态（不序列化到服务端）
  final QuestionRuntimeState runtimeState;
}

/// 题目运行时状态
class QuestionRuntimeState {
  final bool isAnswered;
  final bool isMarked;
  final int attemptCount;
  final FeedbackState? feedbackState;
  final bool isRevealed;             // 答案是否已揭示
  final Duration? timeSpent;
}

/// 反馈状态
class FeedbackState {
  final bool isCorrect;
  final String? correctAnswer;
  final String? userAnswer;
  final String? explanation;         // 解析（Markdown）
  final List<String>? knowledgePoints;
  final String? aiHint;              // AI提示（guided模式）
  final FeedbackLevel revealedLevel; // 当前揭示到哪一级
}

/// 用户答案（多态）
@immutable
sealed class UserAnswer {
  final String questionId;
  final DateTime answeredAt;
  final Duration? timeSpent;
}

class ChoiceAnswer extends UserAnswer {
  final Set<String> selectedOptionIds; // 多选用 Set，单选用 Set.single
}

class FillBlankAnswer extends UserAnswer {
  final List<String> blanks;          // 每个空一个值
}

class TrueFalseAnswer extends UserAnswer {
  final bool value;
}

class ShortAnswer extends UserAnswer {
  final String text;
  final List<String>? imageUrls;      // 手写拍照上传
}

class MathAnswer extends UserAnswer {
  final String latex;                 // LaTeX 格式
}

class EssayAnswer extends UserAnswer {
  final String text;
  final List<String>? imageUrls;
  final int? wordCount;
}

class SortAnswer extends UserAnswer {
  final List<String> orderedIds;      // 排序后的选项ID序列
}

class MatchAnswer extends UserAnswer {
  final Map<String, String> pairs;    // leftId -> rightId
}

}

/// 答题统计（实时计算）
@riverpod
PracticeStats practiceStats(Ref ref) {
  final session = ref.watch(practiceSessionNotifierProvider(
    ref.watch(currentSessionIdProvider)
  ));
  int answered = session.answers.length;
  int correct = 0;
  int wrong = 0;
  for (final entry in session.answers.entries) {
    final fb = session.questions[entry.key].runtimeState.feedbackState;
    if (fb != null) {
      if (fb.isCorrect) correct++;
      else wrong++;
    }
  }
  return PracticeStats(
    total: session.totalCount,
    answered: answered,
    correct: correct,
    wrong: wrong,
    unanswered: session.totalCount - answered,
    marked: session.markedIndices.length,
  );
}

class PracticeStats {
  final int total;
  final int answered;
  final int correct;
  final int wrong;
  final int unanswered;
  final int marked;
  double get accuracy => answered > 0 ? correct / answered : 0;
}

          ],

    // 显示加载
    final scaffoldMessenger = ScaffoldMessenger.of(context);
    scaffoldMessenger.showSnackBar(
      const SnackBar(content: Text('正在生成练习...'), duration: Duration(seconds: 1)),
    );

    try {
      final sessionId = await ref.read(practiceRepositoryProvider).createSession(
        CreatePracticeRequest(

          mode: params.mode,
          source: params.source,
          sourceId: params.sourceId,
          config: params.defaultConfig,

      }

      scaffoldMessenger.showSnackBar(
        SnackBar(content: Text('创建练习失败：${_friendlyError(e)}')),

// lib/features/practice/pages/practice_session_page.dart

class PracticeSessionPage extends ConsumerStatefulWidget {
  final String sessionId;

  const PracticeSessionPage({super.key, required this.sessionId});

  @override
  ConsumerState<PracticeSessionPage> createState() =>
      _PracticeSessionPageState();
}

class _PracticeSessionPageState extends ConsumerState<PracticeSessionPage>
    with WidgetsBindingAware {

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    // 启动计时器
    Future.microtask(() {
      ref.read(practiceTimerProvider(widget.sessionId).notifier).start();
    });
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    // 自动保存进度
    _saveProgress();
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused) {
      _saveProgress();
    }
    if (state == AppLifecycleState.resumed) {
      // 恢复计时器
      ref.read(practiceTimerProvider(widget.sessionId).notifier).resume();
    }
  }

  @override
  Widget build(BuildContext context) {
    final sessionAsync = ref.watch(
      practiceSessionNotifierProvider(widget.sessionId),
    );
    final ageGroup = ref.watch(currentAgeGroupProvider);
    final theme = AgeTheme.of(ageGroup);

    return sessionAsync.when(
      loading: () => const _LoadingSkeleton(),
      error: (e, _) => _ErrorView(error: e, onRetry: _retry),
      data: (session) {
        // 检查是否需要退出
        if (session.status == SessionStatus.submitted ||
            session.status == SessionStatus.autoSubmitted) {
          // 延迟导航到结果页
          WidgetsBinding.instance.addPostFrameCallback((_) {
            _navigateToResult(session);
          });
          return const _LoadingSkeleton();
        }

        return PopScope(
          canPop: false,
          onPopInvokedWithResult: (didPop, _) {
            if (!didPop) _showExitConfirmDialog(context, session);
          },
          child: Scaffold(
            appBar: _buildAppBar(session, theme),
            body: _buildBody(session, theme),
            bottomNavigationBar: _buildBottomBar(session, theme),
          ),
        );
      },
    );
  }

  /// ── AppBar：进度 + 计时 + 设置 ──
  PreferredSizeWidget _buildAppBar(PracticeSession session, AgeTheme theme) {
    return AppBar(
      leading: IconButton(
        icon: const Icon(Icons.close),
        onPressed: () => _showExitConfirmDialog(context, session),
      ),
      title: _SessionAppBarContent(sessionId: widget.sessionId),
      actions: [
        // 题号盘按钮
        IconButton(
          icon: const Icon(Icons.grid_view),
          onPressed: () => _showQuestionPad(context, session),
          tooltip: '题号导航',
        ),
      ],
    );
  }

  /// ── 主体内容 ──
  Widget _buildBody(PracticeSession session, AgeTheme theme) {
    final currentIndex = session.currentIndex;
    final question = session.questions[currentIndex];
    final answer = session.answers[currentIndex];
    final feedback = question.runtimeState.feedbackState;

    return Column(
      children: [
        // 进度条
        _ProgressBar(
          current: currentIndex + 1,
          total: session.totalCount,
          answeredCount: session.answers.length,
          markedIndices: session.markedIndices,
          theme: theme,
        ),

        // 题目+答题区（可滚动）
        Expanded(
          child: SingleChildScrollView(
            padding: theme.contentPadding,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                // 题目渲染
                QuestionRenderWidget(
                  question: question,
                  index: currentIndex,
                  showScore: session.config.mode == PracticeMode.exam,
                ),

                const SizedBox(height: 16),

                // 答题交互区
                AnswerInteractionWidget(
                  question: question,
                  currentAnswer: answer,
                  feedback: feedback,
                  isEnabled: feedback == null, // 有反馈后不可修改
                  onAnswerChanged: (ans) => _onAnswerChanged(currentIndex, ans),
                  onSubmit: () => _submitCurrentAnswer(currentIndex),
                  config: session.config,
                  theme: theme,
                ),

                // 反馈展示
                if (feedback != null) ...[
                  const SizedBox(height: 16),
                  FeedbackDisplayWidget(
                    feedback: feedback,
                    question: question,
                    onRevealNext: () => _revealNextTier(currentIndex),
                    onAskAI: () => _askAI(currentIndex),
                    onAddToMistakes: () => _addToMistakes(currentIndex),
                    onPracticeSimilar: () => _practiceSimilar(currentIndex),
                    config: session.config,
                    theme: theme,
                  ),
                ],
              ],
            ),
          ),
        ),
      ],
    );
  }

  /// ── 底部操作栏 ──
  Widget _buildBottomBar(PracticeSession session, AgeTheme theme) {
    final currentIndex = session.currentIndex;
    final canGoBack = session.config.allowGoBack && currentIndex > 0;
    final canGoForward = currentIndex < session.totalCount - 1;
    final isLast = currentIndex == session.totalCount - 1;
    final hasAnswer = session.answers.containsKey(currentIndex);

    return Container(
      decoration: BoxDecoration(
        color: theme.surfaceColor,
        border: Border(top: BorderSide(color: theme.dividerColor)),
      ),
      child: SafeArea(
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
          child: Row(
            children: [
              // 标记按钮
              IconButton(
                icon: Icon(
                  session.markedIndices.contains(currentIndex)
                    ? Icons.bookmark
                    : Icons.bookmark_border,
                  color: session.markedIndices.contains(currentIndex)
                    ? theme.accentColor
                    : null
