/// 练习会话
class PracticeSession {
  final String sessionId;
  final String title;
  final PracticeMode mode;        // 练习模式
  final PracticeSource source;    // 来源
  final List<PracticeQuestion> questions;
  final int totalCount;
  final Duration? timeLimit;      // null = 不限时
  final Duration? elapsedTime;    // 已用时间
  final PracticeConfig config;
  final SessionStatus status;
  final DateTime createdAt;
  final DateTime? completedAt;

  /// 当前题目索引
  final int currentIndex;

  /// 标记的题目索引集合
  final Set<int> markedIndices;

  /// 已答题的答案映射: questionIndex -> answer
  final Map<int, UserAnswer> answers;
}

/// 练习模式枚举
enum PracticeMode {
  practice,       // 普通练习
  quiz,           // 随堂测验
  exam,           // 考试模拟（严格限时）
  adaptive,       // 自适应练习
  mistakeReview,  // 错题重练
  dailyTask,      // 每日任务
}

/// 练习来源
enum PracticeSource {
  chapter,        // 同步课堂章节
  knowledgePoint, // 知识点专项
  mistake,        // 错题本
  aiRecommend,    // AI推荐
  examSimulation, // 考试模拟
  customGroup,    // 自定义组卷
  dailyTask,      // 每日任务
  campTask,       // 学习营任务
}

/// 会话状态
enum SessionStatus {
  notStarted,     // 未开始
  inProgress,     // 进行中
  paused,         // 已暂停（非限时模式）
  submitted,      // 已提交
  autoSubmitted,  // 自动交卷（超时）
  abandoned,      // 已放弃
}

/// 练习配置
class PracticeConfig {
  final bool showAnswerImmediately;   // 答完即显答案
  final bool allowGoBack;             // 允许返回上一题
  final bool allowSkip;               // 允许跳过
  final bool shuffleOrder;            // 随机题目顺序
  final bool showProgress;            // 显示进度
  final bool showTimer;               // 显示计时器
  final FeedbackLevel feedbackLevel;  // 反馈详细程度
  final AnswerRevealStrategy answerReveal; // 答案揭示策略
  final bool enableMarking;           // 允许标记题目
  final int maxAttempts;              // 单题最大尝试次数（0=无限）
}

/// 反馈级别
enum FeedbackLevel {
  none,           // 无反馈（考试模式）
  resultOnly,     // 仅对错
  withExplanation,// 对错+解析
  detailed,       // 详细反馈+知识点
  guided,         // 启发式引导反馈
}

/// 答案揭示策略（对接答案管控引擎）
enum AnswerRevealStrategy {
  immediate,      // 答题后立即揭示
  afterSubmit,    // 提交后揭示
  tiered,         // 分级渐进揭示（TIER 体系）
  onDemand,       // 按需点击揭示
}

/// 练习中的题目（扩展自通用题目模型）
class PracticeQuestion {
  final String questionId;
  final QuestionType type;
  final DifficultyLevel difficulty;
  final String stem;                 // 题干（富文本 Markdown）
  final List<String>? materials;     // 材料段落
  final List<Attachment>? attachments; // 附件（图片/音频/视频）
  final dynamic answerStructure;     // 题型专属答题结构
  final List<KnowledgeTag> knowledgeTags;
  final double? score;               // 分值
  final int? estimatedSeconds;       // 预估答题秒数

  /// 运行时状态（不序列化到服务端）
  final QuestionRuntimeState runtimeState;
}

/// 题目运行时状态
class QuestionRuntimeState {
  final bool isAnswered;
  final bool isMarked;
  final int attemptCount;
  final FeedbackState? feedbackState;
  final bool isRevealed;             // 答案是否已揭示
  final Duration? timeSpent;
}

/// 反馈状态
class FeedbackState {
  final bool isCorrect;
  final String? correctAnswer;
  final String? userAnswer;
  final String? explanation;         // 解析（Markdown）
  final List<String>? knowledgePoints;
  final String? aiHint;              // AI提示（guided模式）
  final FeedbackLevel revealedLevel; // 当前揭示到哪一级
}

/// 用户答案（多态）
@immutable
sealed class UserAnswer {
  final String questionId;
  final DateTime answeredAt;
  final Duration? timeSpent;
}

class ChoiceAnswer extends UserAnswer {
  final Set<String> selectedOptionIds; // 多选用 Set，单选用 Set.single
}

class FillBlankAnswer extends UserAnswer {
  final List<String> blanks;          // 每个空一个值
}

}

class ShortAnswer extends UserAnswer {

class MathAnswer extends UserAnswer {

class EssayAnswer extends UserAnswer {
  final String text;

}

class SortAnswer extends UserAnswer {

}

class MatchAnswer extends UserAnswer {

/// 答题统计（实时计算）
@riverpod
PracticeStats practiceStats(Ref ref) {
  final session = ref.watch(practiceSessionNotifierProvider(
    ref.watch(currentSessionIdProvider)
  ));
  int answered = session.answers.length;
  int correct = 0;
  int wrong = 0;
  for (final entry in session.answers.entries) {
    final fb = session.questions[entry.key].runtimeState.feedbackState;
    if (fb != null) {
      if (fb.isCorrect) correct++;
      else wrong++;
    }
  }
  return PracticeStats(
    total: session.totalCount,
    answered: answered,
    correct: correct,
    wrong: wrong,
    unanswered: session.totalCount - answered,
    marked: session.markedIndices.length,
  );
}

class PracticeStats {
  final int total;
  final int answered;
  final int correct;
  final int wrong;
  final int unanswered;
  final int marked;
  double get accuracy => answered > 0 ? correct / answered : 0;
}

    // 显示加载
    final scaffoldMessenger = ScaffoldMessenger.of(context);
    scaffoldMessenger.showSnackBar(
      const SnackBar(content: Text('正在生成练习...'), duration: Duration(seconds: 1)),
    );

          mode: params.mode,

          sourceId: params.sourceId,

// lib/features/practice/pages/practice_session_page.dart

class PracticeSessionPage extends ConsumerStatefulWidget {
  final String sessionId;

  const PracticeSessionPage({super.key, required this.sessionId});

  @override
  ConsumerState<PracticeSessionPage> createState() =>
      _PracticeSessionPageState();
}

class _PracticeSessionPageState extends ConsumerState<PracticeSessionPage>
    with WidgetsBindingAware {

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    // 启动计时器
    Future.microtask(() {
      ref.read(practiceTimerProvider(widget.sessionId).notifier).start();
    });
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    // 自动保存进度
    _saveProgress();
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused) {
      _saveProgress();
    }
    if (state == AppLifecycleState.resumed) {
      // 恢复计时器
      ref.read(practiceTimerProvider(widget.sessionId).notifier).resume();
    }
  }

  @override
  Widget build(BuildContext context) {
    final sessionAsync = ref.watch(
      practiceSessionNotifierProvider(widget.sessionId),
    );
    final ageGroup = ref.watch(currentAgeGroupProvider);
    final theme = AgeTheme.of(ageGroup);

    return sessionAsync.when(
      loading: () => const _LoadingSkeleton(),
      error: (e, _) => _ErrorView(error: e, onRetry: _retry),
      data: (session) {
        // 检查是否需要退出
        if (session.status == SessionStatus.submitted ||
            session.status == SessionStatus.autoSubmitted) {
          // 延迟导航到结果页
          WidgetsBinding.instance.addPostFrameCallback((_) {
            _navigateToResult(session);
          });
          return const _LoadingSkeleton();
        }

        return PopScope(
          canPop: false,

    final notifier = ref.read(
      practiceSessionNotifierProvider(widget.sessionId).notifier,
    );
    final answer = notifier.current.answers[index];
    if (answer == null) return;

    // 按钮防连点：提交期间禁用（Notifier 内部以 submittingIndex 标记）
    if (notifier.isSubmitting(index)) return;

    try {
      final feedback = await notifier.submitAnswer(index, answer);
      if (!mounted) return;

      if (feedback != null && !feedback.isCorrect) {
        // 答错震动反馈（仅移动端、可在设置中关闭）
        unawaited(HapticFeedback.lightImpact());
      }
    } on PracticeApiException catch (e) {
      if (!mounted) return;
      _handleApiError(e);
    }
  }

        .read(practiceSessionNotifierProvider(widget.sessionId).notifier)

      _confirmSubmitAll(session);
    } else {
      _goTo(currentIndex + 1);
    }

                      .notifier)
                  .pause();
              if (context.mounted) context.pop();

          if (session.source != PracticeSource.examSimulation)
            TextButton(
              onPressed: () async {
                Navigator.of(ctx).pop();
                await _abandon();
              },
              style: TextButton.styleFrom(foregroundColor: Colors.red),
              child: const Text('放弃练习'),
            ),
        ],
      ),

  /// 交卷确认（未答 >30% 时服务端会要求 confirm=true，见守卫 G4）
  Future<void> _confirmSubmitAll(PracticeSession session) async {
    final unanswered = session.totalCount - session.answers.length;
    final needConfirm = unanswered > (session.totalCount * 0.3).floor();

    if (needConfirm && unanswered > 0) {
      final ok = await showDialog<bool>(
        context: context,
        builder: (ctx) => AlertDialog(
          title: const Text('还有题目未完成'),
          content: Text('剩余 $unanswered 题未作答，确定交卷吗？'),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(ctx).pop(false),
              child: const Text('继续答题'),
            ),
            FilledButton(
              onPressed: () => Navigator.of(ctx).pop(true),
              child: const Text('确认交卷'),
            ),
          ],
        ),
      );
      if (ok != true) return;
    }

    await _submitAll(confirm: true);
  }

  Future<void> _submitAll({bool confirm = false}) async {
    try {
      final result = await ref
          .read(practiceSessionNotifierProvider(widget.sessionId).notifier)
          .submitAll(confirm: confirm);
      if (!mounted) return;
      context.go('/practice/result', extra: result);
    } on PracticeApiException catch (e) {
      if (!mounted) return;
      if (e.code == 55118) {
        // 服务端二次确认裁决：重新弹确认框
        _confirmSubmitAll(
          ref.read(practiceSessionNotifierProvider(widget.sessionId)),
        );
        return;
      }
      _handleApiError(e);
    }
  }

  Future<void> _abandon() async {
    try {
      await ref
          .read(practiceSessionNotifierProvider(widget.sessionId).notifier)
          .abandon();
      if (mounted) context.pop();
    } on PracticeApiException catch (e) {
      if (mounted) _handleApiError(e);
    }
  }

  // ─────────────────────────── 反馈动作 ───────────────────────────

  /// 问 AI（携带题目上下文跳转 AI 对话）
  Future<void> _askAI(int index) async {
    final session = ref.read(
      practiceSessionNotifierProvider(widget.sessionId),
    );
    final question = session.questions[index];
    final answer = session.answers[index];
    context.push('/ai/chat', extra: AiChatLaunchContext(
      entry: AiEntry.practiceQuestion,
      questionId: question.questionId,
      stemSummary: question.stem.length > 200
          ? '${question.stem.substring(0, 200)}...'
          : question.stem,
      userAnswerSummary: answer?.toSummary(),
    ));
  }

  /// 加入错题本
  Future<void> _addToMistakes(int index) async {
    final session = ref.read(
      practiceSessionNotifierProvider(widget.sessionId),
    );
    try {
      await ref.read(mistakeRepositoryProvider).addFromPractice(
        sessionId: widget.sessionId,
        questionId: session.questions[index].questionId,
        userAnswer: session.answers[index],
        source: MistakeSource.practice,
      );
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('已加入错题本')),
        );
      }
    } on MistakeApiException catch (e) {
      if (e.code == 40901 || e.code == 40902) {
        // 已收录：视为成功，不重复提示
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('该题已在错题本中')),
          );
        }
        return;
      }
      if (mounted) _handleApiError(e);
    }
  }

  /// 练一道同类题
  Future<void> _practiceSimilar(int index) async {
    final session = ref.read(
      practiceSessionNotifierProvider(widget.sessionId),
    );
    final question = session.questions[index];
    try {
      final sessionId = await ref
          .read(practiceRepositoryProvider)
          .createSimilarSession(questionId: question.questionId);
      if (mounted) {
        context.pushReplacement('/practice/session', extra: sessionId);
      }
    } on PracticeApiException catch (e) {
      if (mounted) _handleApiError(e);
    }
  }

  // ─────────────────────────── 工具方法 ───────────────────────────

  void _saveProgress() {
    ref
        .read(practiceSessionNotifierProvider(widget.sessionId).notifier)
        .persistLocal();
  }

  void _navigateToResult(PracticeSession session) {
    // 终态会话直接拉取报告跳转
    unawaited(_submitAll(confirm: true));
  }

  void _retry() {
    ref.invalidate(practiceSessionNotifierProvider(widget.sessionId));
  }

  /// 统一 API 错误处理（详见 §8 错误码映射表）
  void _handleApiError(PracticeApiException e) {
    switch (e.code) {
      case 55116: // 考试会话被其他设备锁定
        _enterReadOnlyMode('该考试正在其他设备上进行，本设备仅可查看');
      case 55119: // 非持锁设备写操作（练习类）
        _enterReadOnlyMode('另一台设备正在答题，本设备已切换为只读');
      case 55111: // 已终态
        _navigateToResult(
          ref.read(practiceSessionNotifierProvider(widget.sessionId)),
        );
      case 55112: // 已过期
        _showExpiredDialog();
      case 55114: // 防重窗重复提交：忽略
        return;
      case 55121: // 计时漂移：强制硬校准
        ref
            .read(practiceTimerProvider(widget.sessionId).notifier)
            .hardCalibrate(e.serverNow!);
      default:
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text(e.friendlyMessage)),
        );
    }
  }

  void _enterReadOnlyMode(String message) {
    ref
        .read(practiceSessionNotifierProvider(widget.sessionId).notifier)
        .enterReadOnly();
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message), duration: const Duration(seconds: 3)),
    );
  }

  void _showExpiredDialog() {
    showDialog<void>(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => AlertDialog(
        title: const Text('练习已过期'),
        content: const Text('本次练习已超过有效时限，请重新开始。'),
        actions: [
          FilledButton(
            onPressed: () {
              Navigator.of(ctx).pop();
              context.pop();
            },
            child: const Text('知道了'),
          ),
        ],
      ),
    );
  }

  String _formatDuration(Duration d) {
    final m = d.inMinutes;
    final s = d.inSeconds % 60;
    return s > 0 ? '$m 分 $s 秒' : '$m 分钟';
  }
}

      // 已收录等幂等场景静默处理
    }
  }

  void _askAI(int index) {

## 6. 题号盘组件（QuestionPadSheet）

```dart
// lib/features/practice/widgets/question_pad_sheet.dart

/// 题号盘：网格化展示每题状态，支持点击跳转
class QuestionPadSheet extends StatelessWidget {
  final PracticeSession session;
  final ValueChanged<int> onSelect;

  const QuestionPadSheet({
    super.key,
    required this.session,
    required this.onSelect,
  });

  @override
  Widget build(BuildContext context) {
    final theme = AgeTheme.of(context);
    final crossAxisCount = MediaQuery.of(context).size.width > 600 ? 10 : 7;

    return Container(
      decoration: BoxDecoration(
        color: theme.surfaceColor,
        borderRadius: const BorderRadius.vertical(top: Radius.circular(16)),
      ),
      padding: const EdgeInsets.fromLTRB(16, 12, 16, 24),
      child: SafeArea(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            // 拖动把手
            Container(
              width: 40, height: 4,
              decoration: BoxDecoration(
                color: theme.dividerColor,
                borderRadius: BorderRadius.circular(2),
              ),
            ),
            const SizedBox(height: 12),

            // 图例
            _LegendRow(theme: theme),
            const SizedBox(height: 12),

            // 题号网格
            Flexible(
              child: SingleChildScrollView(
                child: GridView.builder(
                  shrinkWrap: true,
                  physics: const NeverScrollableScrollPhysics(),
                  gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
                    crossAxisCount: crossAxisCount,
                    mainAxisSpacing: 8,
                    crossAxisSpacing: 8,
                  ),
                  itemCount: session.totalCount,
                  itemBuilder: (context, index) {
                    final state = _resolveState(index);
                    return _QuestionCell(
                      index: index,
                      state: state,
                      isCurrent: index == session.currentIndex,
                      theme: theme,
                      onTap: () => onSelect(index),
                    );
                  },
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  /// 派生每格状态（读时派生，不落库）
  _CellState _resolveState(int index) {
    final q = session.questions[index];
    if (session.markedIndices.contains(index)) return _CellState.marked;
    if (!session.answers.containsKey(index)) return _CellState.unanswered;
    final fb = q.runtimeState.feedbackState;
    if (fb == null) return _CellState.answered;      // 已答未批改（异步判分中）
    return fb.isCorrect ? _CellState.correct : _CellState.wrong;
  }
}

enum _CellState { unanswered, answered, correct, wrong, marked }
```

**题号盘交互规则：**

| 规则 | 说明 |
|------|------|
| R1 | 限时考试模式（`sequential_only=true`）下点击后续题号不跳转，Toast 提示"请按顺序作答"（对应服务端 55113 语义前置拦截） |
| R2 | 有反馈（已批改）的题允许反复查看；未答的题跳转后自动定位答题区 |
| R3 | 标记态优先显示：标记 > 对错 > 已答 > 未答 |
| R4 | 题量 >100 时网格分屏，支持按 20 题分段滚动（防止超长网格卡顿） |

## 7. 客户端状态机守卫

客户端不重复实现服务端状态机（权威归《服务端-练习测评会话服务与答题流程状态机》），仅做**前置拦截与兜底**。以下守卫在 Notifier/页面层执行：

| 守卫 | 规则 | 拦截位置 | 违例处理 |
|------|------|----------|----------|
| G1 | 考试模式（exam/MOCK_EXAM）隐藏暂停入口、退出仅允许"交卷"或"放弃" | SessionPage build | 隐藏按钮，不依赖服务端兜底（55117 为最后防线） |
| G2 | 有反馈的题目答案不可修改（`isEnabled = feedback == null`） | AnswerInteractionWidget | 禁用交互控件 |
| G3 | `sequential_only` 模式禁止跳题与提前翻页 | 底部栏/题号盘 | 前置拦截 + Toast |
| G4 | 未答 >30% 时交卷必须二次确认（与服务端 55118 对齐） | `_confirmSubmitAll` | 本地弹窗；服务端仍拒绝时按 55118 重弹 |
| G5 | 提交中防连点：`isSubmitting(index)` 为 true 时忽略重复提交 | `_submitCurrentAnswer` | 静默忽略（服务端 55114 防重窗双保险） |
| G6 | 只读快照模式下一切写操作（草稿/提交/标记）禁用 | `_enterReadOnlyMode` | 置 `readOnly=true`，UI 整体置灰，写操作入口隐藏 |
| G7 | 计时器以服务端为权威：心跳漂移 >5s 触发硬校准 | PracticeTimer | 对齐 `server_now`，漂移超限（55121）立即校准 |
| G8 | 交卷/放弃前必须 `persistLocal()` 落本地快照 | dispose / 弹窗分支 | 防进程被杀丢失进度 |
| G9 | 自动交卷（倒计时归零）必须走与手动交卷相同的 `submitAll(confirm:true)` 路径 | PracticeTimer.onExpire | 不旁路，保证事件埋点一致 |
| G10 | 终态会话（submitted/autoSubmitted）禁止再进入答题页 | SessionPage build | 直接导航结果页 |

**多端竞态裁决（与服务端对齐）：**

- 考试独占：本地收到 55116 → 切只读快照，展示"考试正在其他设备进行"横幅（不可关闭，可退出页面）。
- 练习只读：本地收到 55119 → 切只读快照，提示"另一设备正在答题"，允许退出与查看。
- 心跳断 35s 后其他设备可申请接管（服务端裁决），本端重连收到 `session.taken_over` WS 事件 → 弹窗告知并退出。

## 8. 错误处理与降级

### 8.1 错误码映射表（服务端 551xx 段 → 客户端处理）

| 错误码 | 语义 | 客户端处理 |
|--------|------|------------|
| 55100 | 会话不存在 | 返回列表页并刷新 |
| 55101 | 越权访问 | 静默退出 + 轻提示"无权限" |
| 55103 | 每日额度不足 | 弹会员引导（每日 1 次频控），提供"明日再来" |
| 55104 | 题目装配失败 | 重试 1 次；仍失败提示更换练习类型 |
| 55110 | start 时已非 CREATED | 拉详情按当前状态续答（幂等恢复） |
| 55111 | 已终态 | 直接跳报告页 |
| 55112 | 已过期/超时处理 | 过期对话框（不可关闭），退出重开 |
| 55113 | 禁止跳题/越序 | Toast"请按顺序作答" |
| 55114 | 2s 防重窗重复提交 | 静默忽略，展示上次结果 |
| 55115 | 已提交答案不可写草稿 | 引导进入修改模式或查看结果 |
| 55116 | 考试会话被其他设备锁定 | 只读快照 + 横幅（见 G6） |
| 55117 | 模拟考试不允许暂停 | 隐藏暂停入口（G1 已前置，此为兜底兜底） |
| 55118 | 未答比例过高需二次确认 | 重弹确认框，带 `confirm=true` 重提 |
| 55119 | 非持锁设备写操作 | 只读快照（G6） |
| 55120 | 同步判分失败转异步 | 展示"批改中"占位，轮询或 WS 等待 grading 完成 |
| 55121 | 计时漂移超限 | 强制硬校准本地计时器（G7） |
| 55122 | 内部依赖异常 | 通用重试提示，不清本地进度 |

### 8.2 降级矩阵

| 编号 | 场景 | 降级策略 |
|------|------|----------|
| D1 | 创建会话接口失败 | 重试 1 次 → 提示失败留在准备页（不生成假会话） |
| D2 | 答案提交失败（网络错误非业务码） | 本地排队（Write-ahead Hive 草稿），恢复网络后自动重放；重放前校验会话仍 ACTIVE |
| D3 | 拉取反馈/解析失败 | 展示"解析加载失败 [重试]"，已答状态不受影响 |
| D4 | 计时心跳失败 ≥3 次 | 切换本地计时并以进入页面时的服务端时间戳为锚点，恢复后硬校准 |
| D5 | 结果页报告生成中（grading_pending） | 展示骨架屏 + 每 3s 轮询，最长 60s 后转"报告稍后可在学习记录中查看" |
| D6 | WS 推送不可用 | 全部状态以 REST 轮询兜底（题态 10s、反馈 3s），功能不缺失仅延迟增加 |
| D7 | 雷达图/掌握度数据缺失 | 隐藏对应卡片，核心分数卡必显 |
| D8 | 埋点上报失败 | 本地缓冲，下次启动补发（不阻塞任何交互） |

**总红线：** 任何降级不得伪造答题结果或提前展示未确认的答案判定；批改结论一律以服务端为权威。

## 9. 离线支持与弱网

| 能力 | 策略 |
|------|------|
| 进度保存 | 每题草稿实时写 Hive（`kPracticeDraftBox`）；页面 dispose / 切后台 30s 触发 `persistLocal()` |

| 考试禁离线 | MOCK_EXAM 模式进入前强制网络探测，弱网（RTT>2s 或失败）阻止进入并提示（防计时作弊） |
